4. Evaluate g(t) at each such candidate t (clipped to [lo, hi]) and at lo, hi
5. Take the minimum

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])
        g[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv])
        g[:nv] += (-h[kk:])
    return g

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        grad += w[m] * grad_lag(h, n, m)
    return grad * dx

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alpha_list=None, max_iters=300):
    if alpha_list is None:
        alpha_list = [5000, 20000, 100000, 500000, 2000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 1.0
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        t_alloc = min(40.0, deadline - time.time())
        if t_alloc < 5:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, deadline, top_lines=8, sweeps=50):
    """Coordinate descent: for each coordinate, find optimal value via 1D optimization"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    nlags = 2 * n - 1
    stagnant = 0
    for sweep in range(sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > deadline:
                return best_h, best_c
            c = compute_c5_vec(h, n)
            other_sum = h.sum() - h[i]
            lo = max(0.0, target - other_sum)
            hi = min(1.0, target - other_sum)
            if hi - lo < 1e-15:
                continue
            # Compute slope a_k for each lag
            a = np.zeros(nlags)
            for m in range(nlags):
                a[m] = grad_lag(h, n, m)[i]
            dx = 2.0 / n
            a *= dx
            # Current values
            # c_k + (t - h[i]) * a_k
            # Find top lines at t = h[i]
            vals = c  # c_k at current t
            top_idx = np.argsort(vals)[-top_lines:]
            # Candidate t values: lo, hi, and crossings of top lines
            candidates = [lo, hi, h[i]]
            for p in range(len(top_idx)):
                for q in range(p + 1, len(top_idx)):
                    kp, kq = top_idx[p], top_idx[q]
                    if abs(a[kp] - a[kq]) > 1e-15:
                        t_cross = (c[kq] - c[kp]) / (a[kp] - a[kq]) + h[i]
                        if lo <= t_cross <= hi:
                            candidates.append(t_cross)
            best_t = h[i]
            best_val = best_c
            for t in candidates:
                t = np.clip(t, lo, hi)
                new_c = np.max(c + (t - h[i]) * a)
                if new_c < best_val - 1e-15:
                    best_val = new_c
                    best_t = t
            if best_t != h[i] and best_val < best_c:
                h[i] = best_t
                best_c = best_val
                improved = True
        if improved:
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5:
                break
    return best_h, best_c

def subgrad_optimize(h, n, deadline, top_k=3):
    """Subgradient method on the top-k active lags with line search"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.003
    stagnant = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            g += grad_lag(h, n, m)
        g /= top_k
        dx = 2.0 / n
        g *= dx
        # Project gradient onto feasible direction (sum constraint)
        g = g - g.mean()
        # Line search
        best_step_c = best_c
        best_step_h = h
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            step = lr * mult
            h_trial = project_feasible(h - step * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_step_c = c_trial
                best_step_h = h_trial
                break
        if best_step_c < best_c:
            h = best_step_h
            best_c = best_step_c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            lr *= 0.6
            if stagnant > 40:
                h = project_feasible(h + 0.005 * np.random.randn(n), n)
                best_c = compute_c5(h, n)
                lr = 0.003
                stagnant = 0
            if lr < 1e-12:
                break
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def con_fun(x, nn):
        return float(x.sum() - nn / 2.0)
    def con_jac(x, nn):
        return np.ones(nn)
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
