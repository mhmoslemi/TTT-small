        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=200):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 8000000]
    mus = [10.0, 50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 50000.0, 100000.0]
    for idx in range(len(alphas)):
        if time.time() > deadline - 3:
            break
        alpha = alphas[idx]
        mu = mus[idx]
        per_alpha = min(80, (deadline - time.time()) / (len(alphas) - idx))
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=5000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, jac=slsqp_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair_search(h, n, deadline, k=50):
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in step_sizes:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, 100000, 0.0)
            idx_dec = np.argsort(g)[-k:]
            idx_inc = np.argsort(g)[:k]
            for i in idx_dec:
                if time.time() > deadline:
                    return h, best_c
                if h[i] < d:
                    continue
                for j in idx_inc:
                    if i == j or h[j] > 1.0 - d:
                        continue
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
            if not improved:
                break
    return h, best_c

def max_tracking_descent(h, n, deadline, lr_init=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(50000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_star = np.argmax(c)
        alpha = 1000000
        g = smooth_max_grad(h, n, alpha, 0.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS with aggressive alpha annealing
    t1 = min(t0 + 280, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, best_n, t1, max_iters=250)

    # Phase 2: SLSQP with very high alpha
    for a in [2000000, 8000000]:
        if time.time() > deadline - 350:
            break
        t2 = min(time.time() + 80, deadline - 300)
        h2, c2 = slsqp_phase(best_h, best_n, t2, alpha=a, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Max-tracking gradient descent
    if time.time() < deadline - 250:
        t3 = min(time.time() + 80, deadline - 200)
        h3, c3 = max_tracking_descent(best_h, best_n, t3, lr_init=0.003)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 120)
        best_h, best_c = grad_pair_search(best_h, best_n, t4, k=55)

    # Phase 5: Second round of SLSQP + pair search
    if time.time() < deadline - 80:
        h5, c5 = slsqp_phase(best_h, best_n, min(time.time() + 50, deadline - 60),
                            alpha=10000000, maxiter=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 30:
            best_h, best_c = grad_pair_search(best_h, best_n, deadline - 20, k=40)

    # Phase 6: Try different resolutions
    for n_try in [200, 250, 400]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 15)
        if alloc < 20:
            break
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6, max_iters=150)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Final polish
    best_h = project_feasible(best_h, best_n)
    if time.time() < deadline - 2:
        best_h, best_c = grad_pair_search(best_h, best_n, deadline - 1, k=35)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
