```python
    im = i + m_range  # i + m for all m
    valid_im = (im >= 0) & (im < n)
    delta[valid_im] += (1.0 - h[im[valid_im]])
    imn = i - m_range  # i - m for all m
    valid_imn = (imn >= 0) & (imn < n)
    delta[valid_imn] -= h[imn[valid_imn]]
    jm = j + m_range
    valid_jm = (jm >= 0) & (jm < n)
    delta[valid_jm] -= (1.0 - h[jm[valid_jm]])
    jmn = j - m_range
    valid_jmn = (jmn >= 0) & (jmn < n)
    delta[valid_jmn] += h[jmn[valid_jmn]]
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    sh = alpha * (corr - m)
    ms = sh.max()
    e = np.exp(sh - ms)
    lh = ms + np.log(e.sum())
    obj = m + lh / alpha
    w = e / e.sum()
    ome = 1.0 - x
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        mo = idx - (n - 1)
        wi = w[idx] * dx
        if abs(wi) < 1e-17:
            continue
        if mo >= 0:
            if mo < n:
                grad[:n-mo] += wi * ome[mo:]
                grad[mo:] -= wi * x[:n-mo]
        else:
            oo = -mo
            if oo < n:
                grad[oo:] += wi * ome[:n-oo]
                grad[:n-oo] -= wi * x[oo:]
    return obj, grad

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj_grad(x, n, alpha)[0]
    def fg(x):
        return smooth_obj_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fg, h, jac=True, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def lbfgsb_opt(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def fg(x):
        o, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        penalty = mu * s * s
        gp = g + 2.0 * mu * s * np.ones(n)
        return o + penalty, gp
    try:
        res = minimize(fg, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta_vec(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    v = (im >= 0) & (im < n)
    delta[v] += 1.0 - h[im[v]]
    imn = i - m_range
    v = (imn >= 0) & (imn < n)
    delta[v] -= h[imn[v]]
    jm = j + m_range
    v = (jm >= 0) & (jm < n)
    delta[v] -= 1.0 - h[jm[v]]
    jmn = j - m_range
    v = (jmn >= 0) & (jmn < n)
    delta[v] += h[jmn[v]]
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.028, t_end=0.00003, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start
    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        if i == j:
            continue
        delta = swap_delta_vec(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c - 1e-15:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii], h[jj] = 1.0, 0.0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = max(T, t_start * 0.5)
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=300, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for sweep_round in range(4):
        if time.time() > deadline:
            break
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            for i in zeros:
                best_j = -1
                best_nc = cur_c
                for j in ones:
                    delta = swap_delta_vec(h, n, i, j, m_range)
                    nc = (corr + delta).max() * dx
                    if nc < best_nc:
                        best_nc = nc
                        best_j = j
                        best_delta = delta
                if best_j >= 0 and best_nc < cur_c - 1e-15:
                    h[i], h[best_j] = 1.0, 0.0
                    corr = corr + best_delta
                    cur_c = best_nc
                    improved = True
            if not improved:
                break
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
        if sweep_round < 3:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                for _ in range(4):
                    a = oi[rng.randint(len(oi))]
                    b = zi[rng.randint(len(zi))]
                    h[a], h[b] = 0.0, 1.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = corr.max() * dx
    return best_h, best_c

def continuous_refine(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

        if time.time() > deadline:

        rem = deadline - time.time()
        if rem < 20:
            break
        h = slsqp_opt(h, n, alpha, mi, deadline)

        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:

            best_h = h.copy()
        if time.time() > deadline:

        h = slsqp_opt(h, n, alpha * 2, mi, deadline)
```
