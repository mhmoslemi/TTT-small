import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def sa_binary(h_init, n, deadline, seed, t_start, t_end, patience):

    rng = np.random.RandomState(seed)

    best_h = h.copy()
    zeros = np.where(h == 0)[0]

def slsqp_refine(h, n, deadline, seed):
    """SLSQP with log-sum-exp for continuous refinement."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                       options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def subgradient_refine(h, n, max_iter, step_size):
    """Projected subgradient method for the max of correlations."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        m = corr.max()
        # Find active lags (within 5% of max)
        active = np.where(corr > 0.95 * m)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k_shift = k - (n - 1)
            # d(corr_k)/dh[i] = (1-h[i+k_shift]) for forward + (-h[i-k_shift]) for backward
            # More precisely: corr_k = sum_i h[i]*(1-h[i+k_shift])
            # d/dh[i]: (1-h[i+k_shift]) - h[i-k_shift] (if both in range)
            for i in range(n):
                val = 0.0
                ip = i + k_shift
                if 0 <= ip < n:
                    val += (1.0 - h[ip])
                im = i - k_shift
                if 0 <= im < n:
                    val -= h[im]
                grad[i] += val
        grad /= len(active)
        
        # Projected step
        eta = step_size / (it + 10)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def compute_subgradient(h, n, active_lags):
    grad = np.zeros(n)
    for k in active_lags:
        k_shift = k - (n - 1)
        if k_shift == 0:
            continue
        # Forward term: (1-h[i+k_shift]) for valid i
        if k_shift > 0:
            fwd = 1.0 - h[k_shift:]
            grad[:n-k_shift] += fwd
        else:
            fwd = 1.0 - h[:n+k_shift]
            grad[-k_shift:] += fwd
        # Backward term: -h[i-k_shift] for valid i
        if k_shift > 0:
            bwd = -h[:n-k_shift]
            grad[k_shift:] += bwd
        else:
            bwd = -h[-k_shift:]
            grad[:n+k_shift] += bwd
    return grad / len(active_lags)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def sa_binary(h_init, n, deadline, seed, t_start=0.04, t_end=0.00001, patience=40000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 500000.0)
    T = t_start

    while time.time() < deadline:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = swap_delta(h, n, int(i), int(j))
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[int(i)] = 1.0
            h[int(j)] = 0.0
            corr = new_corr
            cur_c = nc
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(6)
            for _ in range(nk):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                h[int(ii)] = 1.0
                h[int(jj)] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 4.0)
    return best_h, best_c

def slsqp_refine(h, n, deadline, seed):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def subgrad_refine(h, n, max_iter=5000, base_eta=0.05):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        m = corr.max()
        thresh = m - 0.5
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.argmax(corr)
            active = np.array([active])
        grad = np.zeros(n)
        for k in active:
            ks = k - (n - 1)
            if ks == 0:
                continue
            if ks > 0:
                grad[:n - ks] += (1.0 - h[ks:])
                grad[ks:] += (-h[:n - ks])
            else:
                grad[-ks:] += (1.0 - h[:n + ks])
                grad[:-ks] += (-h[-ks:])
        grad /= max(1, len(active))
        eta = base_eta / (1.0 + 0.01 * it)
        h = project_feasible(h - eta * grad, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def greedy_local(h, n, max_rounds=100, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        improved = False
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        for i in rng.choice(zeros, min(50, len(zeros)), replace=False):
            for j in rng.choice(ones, min(50, len(ones)), replace=False):
                hb = h.copy()
                hb[int(i)] = min(1.0, hb[int(i)] + 0.05)
                hb[int(j)] = max(0.0, hb[int(j)] - 0.05)
                hb = project_feasible(hb, n)
                c = compute_c5(hb, n)
                if c < best_c - 1e-8:
                    h = hb
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
