        x_new = np.linspace(0, 2, 500)
        h_500 = project_feasible(np.interp(x_new, x_old, best_h), 500)
        alloc2 = min(200, (deadline - time.time()) * 0.25)
        if alloc2 > 50:
            h_opt500, c_opt500 = optimize_with_sg(h_500, 500, time.time() + alloc2, rng)
            if c_opt500 < best_c:
                best_c, best_h = c_opt500, h_opt500
    # Multi-restart with subgradient perturbations
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        c_cur = compute_corr(best_h, n_cur)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 60))
        scales = [0.02, 0.03, 0.04, 0.05, 0.03, 0.02, 0.04, 0.06, 0.03, 0.05]
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = scales[r % len(scales)]
            direction = 'neg' if r % 2 == 0 else 'pos'
            c_now = compute_corr(best_h, n_cur)
            h_pert = sg_perturb(best_h, n_cur, c_now, scale, direction)
            # Add small random noise
            h_pert += rng.normal(0, scale * 0.3, n_cur)
            h_pert = project_feasible(h_pert, n_cur)
            alloc_r = min(55, (deadline - time.time() - 20) * 0.35)
            if alloc_r < 20:
                break
            h_opt_r, c_opt_r = optimize_with_sg(h_pert, n_cur, time.time() + alloc_r, rng)
            if c_opt_r < best_c:
                best_c, best_h = c_opt_r, h_opt_r
    # Final SLSQP + CD polish
    if time.time() < deadline - 20:
        n_f = len(best_h)
        h_f = slsqp_opt(best_h, n_f, alpha=200000, maxiter=2000)
        c_f = compute_c5(h_f, n_f)
        if c_f <= best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        n_f = len(best_h)
        h_f, c_f = coord_descent(best_h, n_f, deadline - 2, rng, grid=120)
        h_f = project_feasible(h_f, n_f)
        c_f = compute_c5(h_f, n_f)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
