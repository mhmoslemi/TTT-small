def project_feasible(h, n):

            s = room.sum()

            if s < 1e-14:

def targeted_best_swap(h, n, k_star):
    """Find the (i,j) swap that most reduces corr at lag k_star,
    without increasing any lag above current max."""
    m_range = np.arange(-(n-1), n)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    best_delta = 0  # we want negative
    best_i, best_j = -1, -1
    for i in zeros:
        # Change at lag k_star from h[i] 0->1:
        # (1-h[i+k_star]) if 0<=i+k_star<n, plus -h[i-k_star] if 0<=i-k_star<n
        val_i = 0.0
        if 0 <= i + k_star < n:
            val_i += 1.0 - h[i + k_star]
        if 0 <= i - k_star < n:
            val_i -= h[i - k_star]
        for j in ones:
            val_j = 0.0
            if 0 <= j + k_star < n:
                val_j -= 1.0 - h[j + k_star]
            if 0 <= j - k_star < n:
                val_j += h[j - k_star]
            total = val_i + val_j
            # Handle collision at k_star
            if i == j + k_star and 0 <= i < n and 0 <= j < n:
                total += 1.0
            if j == i + k_star and 0 <= i < n and 0 <= j < n:
                total += 1.0
            if total < best_delta:
                best_delta = total
                best_i, best_j = i, j
    return best_i, best_j, best_delta

def targeted_swaps(h, n, k_star, top_k=50):
    """Find top_k swaps that most reduce corr at lag k_star."""
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    candidates = []
    for i in zeros:
        val_i = 0.0
        if 0 <= i + k_star < n:
            val_i += 1.0 - h[i + k_star]
        if 0 <= i - k_star < n:
            val_i -= h[i - k_star]
        for j in ones:
            val_j = 0.0
            if 0 <= j + k_star < n:
                val_j -= 1.0 - h[j + k_star]
            if 0 <= j - k_star < n:
                val_j += h[j - k_star]
            # collision corrections
            if i - j == k_star and 0 <= i < n and 0 <= j < n:
                val_j += 1.0
            if j - i == k_star and 0 <= i < n and 0 <= j < n:
                val_j += 1.0
            candidates.append((val_i + val_j, i, j))
    candidates.sort()
    return candidates[:top_k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def smooth_slsqp(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def tabu_sa(h_bin, n, deadline, seed, t_start=0.02, t_end=0.00001,
            patience=40000, tabu_len=80):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 150000.0)
    T = t_start
    tabu_last = np.full(n, -10**9, dtype=int)
    step = 0
    while time.time() < deadline:
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        for attempt in range(5):
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
            if step - tabu_last[i] >= tabu_len and step - tabu_last[j] >= tabu_len:
                break
        else:
            step += 1
            continue
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            tabu_last[i] = step
            tabu_last[j] = step
            if nc < best_c - 1e-15:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        step += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                a = zi[rng.randint(len(zi))]
                b = oi[rng.randint(len(oi))]
                h[a] = 1.0
                h[b] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_start, T * 2.5)
    return best_h, best_c

def targeted_local_search(h_bin, n, deadline, max_rounds=2000, top_k=30):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline and no_improve < 500:
        k_star = int(np.argmax(corr)) - (n - 1)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        candidates = []
        for i in zeros:
            vi = 0.0
            if 0 <= i + k_star < n:
                vi += 1.0 - h[i + k_star]
            if 0 <= i - k_star < n:
                vi -= h[i - k_star]
            for j in ones:
                vj = 0.0
                if 0 <= j + k_star < n:
                    vj -= 1.0 - h[j + k_star]
                if 0 <= j - k_star < n:
                    vj += h[j - k_star]
                if i - j == k_star:
                    vj += 1.0
                if j - i == k_star:
                    vj += 1.0
                candidates.append((vi + vj, i, j))
        candidates.sort()
        improved = False
        for _, i, j in candidates[:top_k]:
            delta = swap_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            if nc < cur_c - 1e-14:
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_c = nc
                improved = True
                break
        if improved:
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            k2 = k_star + (1 if k_star >= 0 else -1)
            if abs(k2) < n:
                idx2 = (n - 1) + k2
                if idx2 != np.argmax(corr):
                    k_third = k2
                else:
                    k_third = -k_star
            else:
                k_third = -k_star
            for _ in range(3):
                delta = swap_delta(h, n, i, j, m_range)
                break
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            if len(zi) > 0 and len(oi) > 0:
                ra = zi[np.random.randint(len(zi))]
                rb = oi[np.random.randint(len(oi))]
                h[ra] = 1.0
                h[rb] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
                cur_c = corr.max() * dx
    return best_h, best_c

def greedy_polish(h_bin, n, deadline, max_rounds=100):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(12345 + r)
        perm = rng.permutation(len(zeros) * len(ones))
        for p in perm[:20000]:
            i_idx = p % len(zeros)
            j_idx = p // len(zeros)
            i = zeros[i_idx]
            j = ones[j_idx]
            delta = swap_delta(h, n, i, j, m_range)
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-14:
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta
                cur_c = nc
                improved = True
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
