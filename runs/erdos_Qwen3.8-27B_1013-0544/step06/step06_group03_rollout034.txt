    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in [2000, 10000, 50000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_run(best_h, n, alpha)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 5:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial n
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try other n values
    n_list = [200, 250, 300, 350, 400, 500]
    if h_init is not None:
        n_list = [x for x in n_list if x != n_init]
    for n_try in n_list:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time() - 30
        if remaining < 35:
            break
        alloc = min(70, remaining * 0.2)
        if best_h is not None and len(best_h) != n_try:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multi-start at best n
    if best_h is not None and time.time() < deadline - 35:
        n = len(best_h)
        remaining = deadline - time.time() - 15
        n_restarts = max(1, int(remaining / 40))
        for r in range(min(n_restarts, 8)):
            if time.time() > deadline - 15:
                break
            scale = 0.008 * (1.0 + 0.6 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 15) * 0.3)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h_new = slsqp_run(best_h, n, 200000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 1:
            best_h, best_c = fast_local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    
    return best_h, best_c, len(best_h)
