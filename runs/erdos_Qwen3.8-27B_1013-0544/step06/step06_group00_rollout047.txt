    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

    s = alpha * (c - M)

```python
def subgradient_step(h, n, lr, n_iters):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    for it in range(n_iters):
        c = np.correlate(h_curr, 1-h_curr, mode='full') * dx
        k_max = np.argmax(c)
        l = k_max - (n-1)
        g = np.zeros(n)
        lo1 = max(0, -l)
        hi1 = min(n, n-l)
        g[lo1:hi1] += (1.0 - h_curr[lo1+l:hi1+l])
        lo2 = max(0, l)
        hi2 = min(n, n+l)
        g[lo2:hi2] -= h_curr[lo2-l:hi2-l]
        g *= dx
        h_new = project_feasible(h_curr - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c
```
