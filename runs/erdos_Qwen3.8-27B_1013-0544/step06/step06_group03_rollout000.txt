import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]
            g[m:] += 1.0 - h[:n-m]
        else:
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g

def _eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(len(x))

def slsqp_solve(h, n, alpha, active=None, maxiter=5000, ftol=1e-16):
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        if active is not None:
            c = c[active]
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(s))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_descent(h, n, deadline, max_iters=3000):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    avg_h = np.zeros(n)
    for t in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_grad(h_curr, n, lag)
        C_target = max(0.1, best_c - 0.01)
        step = max(0.0, c[k_star] - C_target) / (np.dot(g, g) + 1e-10)
        step = min(step, 0.08)
        h_new = project_feasible(h_curr - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        avg_h = (t * avg_h + h_new) / (t + 1)
        h_curr = h_new
    avg_h = project_feasible(avg_h, n)
    c_avg = compute_c5(avg_h, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = avg_h
    return best_h, best_c
