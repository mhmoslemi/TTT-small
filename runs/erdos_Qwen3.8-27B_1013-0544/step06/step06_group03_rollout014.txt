import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Analytical gradient of c_lag w.r.t. h, including dx factor."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] += -h[:n - m]
    else:
        grad[m:] += 1.0 - h[:n - m]
        grad[:n - m] += -h[m:]
    return grad * dx

def smooth_max(c, alpha):
    """Log-sum-exp approximation of max."""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def projected_grad_descent(h, n, deadline, n_iter_max=50000):
    """Fast projected gradient on the active (max) lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.1
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_gradient(h, n, lag)
        # Line search: try a few step sizes
        best_local = best_c
        best_h_local = best_h
        for s in [step, step * 0.5, step * 0.25, step * 2]:
            h_trial = h - s * g
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_local:
                best_local = c_trial
                best_h_local = h_trial.copy()
        if best_local < best_c:
            best_c = best_local
            best_h = best_h_local
            h = best_h_local
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-6:

    return best_h, best_c

def smooth_obj_full(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return float(smooth_max(c, alpha))

def smooth_obj_active(x, n, alpha, top_idx):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    active = c[top_idx]
    return float(smooth_max(active, alpha))

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, n, alpha, top_idx=None, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        obj_fn = smooth_obj_active
        obj_args = (n, alpha, top_idx)
    else:
        obj_fn = smooth_obj_full
        obj_args = (n, alpha)
    try:
        res = minimize(obj_fn, h, args=obj_args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sparse_cutting_plane(h, n, deadline, max_active=20, max_iters=50):
    """LP-based cutting plane with sparse active set."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-max_active:]
        
        # Solve LP: minimize t s.t. c_k + grad_k . (h - h_curr) <= t for k in top_idx
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for k in top_idx:
            lag = k - (n - 1)
            grad = lag_gradient(h_curr, n, lag)
            cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        if step < 1e-10:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] += -h[:n - m]
    else:
        grad[m:] += 1.0 - h[:n - m]
        grad[:n - m] += -h[m:]
    return grad * dx

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def obj_full(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return float(smooth_max_val(c, alpha))

def obj_active(x, n, alpha, top_idx):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return float(smooth_max_val(c[top_idx], alpha))

def slsqp_solve(h, n, alpha, top_idx=None, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(obj_active, h, args=(n, alpha, top_idx), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(obj_full, h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

def projected_grad_descent(h, n, deadline, n_iter_max=80000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.15
    stuck = 0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_gradient(h, n, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        best_local = best_c
        best_h_local = None
        for s in [step, step * 0.5, step * 0.25, step * 1.5]:
            h_trial = project_feasible(h - s * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_local - 1e-15:
                best_local = c_trial
                best_h_local = h_trial
        if best_h_local is not None:
            best_c = best_local
            best_h = best_h_local.copy()
            h = best_h_local
            stuck = 0
            step = min(step * 1.08, 0.6)
        else:
            stuck += 1
            step *= 0.6
            if step < 1e-7 or stuck > 50:
                break
    return best_h, best_c

def sparse_cutting_plane(h, n, deadline, max_active=40, max_iters=60):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-max_active:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for k in top_idx:
            lag = k - (n - 1)
            grad = lag_gradient(h_curr, n, lag)
            cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        if step < 1e-10:
            break
    return best_h, best_c

    best_c = compute_c5(h, n)
    for delta in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        n_iter = 0
        improved = True
        while improved and n_iter < 500:
            if time.time() > deadline:
                break
            improved = False
            n_iter += 1
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            if lag > 0:
                contrib = h[:n - m] * (1.0 - h[m:])
            else:
                contrib = h[m:] * (1.0 - h[:n - m])
            n_top = min(30, n - m)
            top_pairs = np.argsort(contrib)[-n_top:]
            for idx in top_pairs:
                if time.time() > deadline:
                    break
                ii = int(idx)
                if lag > 0:
                    i, j = ii, ii + m
                else:
                    i, j = m + ii, ii
                if h[i] > 0.3 and h[j] < 0.7:
                    d = min(delta, h[i] - 0.3, 0.7 - h[j])
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
                elif h[j] > 0.3 and h[i] < 0.7:
                    d = min(delta, h[j] - 0.3, 0.7 - h[i])
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
    return h, best_c
