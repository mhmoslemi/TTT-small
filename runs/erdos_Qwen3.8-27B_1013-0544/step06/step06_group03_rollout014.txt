def project_feasible(h,n):
    h=np.asarray(h,dtype=float)
    if h.size==0: return h
    target=0.5*n
    lo=float(np.min(h))-1.0
    hi=float(np.max(h))+1.0
    mid=0.5*(lo+hi)
    for _ in range(45):
        mid=0.5*(lo+hi)
        p=np.clip(h-mid,0.0,1.0)
        if p.sum() > target:
            lo=mid
        else:
            hi=mid
    p=np.clip(h-mid,0.0,1.0)
    r=target-p.sum()
    free=(p>1e-10)&(p<1.0-1e-10)

        p=np.clip(p,0.0,1.0)

def lag_gradient_int(h,n,lag):
   g=np.zeros(n,float)
   m=abs(int(lag))
   if m==0: return 1.0-2.0*h
   L=n-m
   if lag>0:
      g[:L]-=h[m:]
      g[m:]+=1.0-h[:L]
   else:
      g[:L]+=1.0-h[m:]
      g[m:]-=h[:L]
   return g

def lp_step(h,n,trust,top_count=40):
   if cp is None: return h.copy()
   s=np.correlate(h,1.0-h,mode='full')
   top=np.argsort(s)[-top_count:]
   hv=cp.Variable(n); tv=cp.Variable()
   cons=[cp.sum(hv)==0.5*n, hv>=0.0, hv<=1.0, hv-h<=trust, h-hv<=trust]
   for idx in top:
      g=lag_gradient_int(h,n,int(idx-(n-1)))
      cons.append(float(s[idx]) + cp.sum(cp.multiply(g, hv-h)) <= tv)
   prob=cp.Problem(cp.Minimize(tv),cons)
   for name in ('GLOP','ECOS','CBC','CVXOPT'):

      if sol is None: continue
      try:
         prob.solve(solver=sol)
         if prob.status in ('optimal','optimal_inaccurate'):
            val=hv.value
            if val is not None:
               return project_feasible(val,n)
      except Exception:
         continue
   return h.copy()

def exact_c5(h, n):

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = np.resize(h, n).astype(float)
    target = 0.5 * n
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mn = float(np.min(h))
    mx = float(np.max(h))
    sm = float(np.sum(h))
    if mn >= -1e-10 and mx <= 1.0 + 1e-10 and abs(sm - target) <= 1e-9:
        return h.copy()
    mid = 0.5 * (lo + hi)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - mid, 0.0, 1.0)
    r = target - p.sum()
    free = (p > 1e-10) & (p < 1.0 - 1e-10)
    if free.any():
        p[free] += r / free.sum()
        p = np.clip(p, 0.0, 1.0)
    r = target - p.sum()
    if abs(r) > 1e-12:
        free = np.where((p > 1e-8) & (p < 1.0 - 1e-8))[0]
        if free.size:
            p[int(free[0])] += r
            p = np.clip(p, 0.0, 1.0)
    return p

def soft_int_obj(x, n, alpha):
    s = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(s))
    z = alpha * (s - m)
    z = z - z.max()
    return float(m + np.log(np.sum(np.exp(z))) / alpha)

def build_weighted_grad(x, n, w):
    grad = w[n - 1] * (1.0 - 2.0 * x)
    for d in range(1, n):
        a = w[n - 1 + d]
        b = w[n - 1 - d]
        sm = a + b
        if sm == 0.0:
            continue
        L = n - d
        grad[:L] += b - sm * x[d:d + L]
        grad[d:n] += a - sm * x[:L]
    return grad

def soft_int_grad_only(x, n, alpha):
    s = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(s))
    z = alpha * (s - m)
    z = z - z.max()
    w = np.exp(z)
    sw = float(w.sum())
    if sw > 0.0:
        w = w / sw
    else:
        w = np.full_like(s, 1.0 / s.size)
    return build_weighted_grad(x, n, w)

def soft_info(x, n, alpha):
    s = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(s))
    z = alpha * (s - m)
    z = z - z.max()
    w = np.exp(z)
    sw = float(w.sum())
    if sw > 0.0:
        nnz = w / sw
    else:
        sw = 1.0
        nnz = np.full_like(s, 1.0 / s.size)
    soft = m + np.log(sw) / alpha if alpha > 0.0 else m
    return soft, m * (2.0 / n), build_weighted_grad(x, n, nnz)

def _sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)

def _sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_optimize(h, n, alpha, maxiter=150):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(
            soft_int_obj, h, args=(n, alpha), method='SLSQP',
            bounds=bounds, constraints=cons,
            jac=soft_int_grad_only, jac_args=(n, alpha),
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def lag_gradient_int(h, n, lag):
    g = np.zeros(n, dtype=float)
    m = abs(int(lag))
    if m == 0:
        return 1.0 - 2.0 * h
    L = n - m
    if lag > 0:
        g[:L] -= h[m:]
        g[m:] += 1.0 - h[:L]
    else:
        g[:L] += 1.0 - h[m:]
        g[m:] -= h[:L]
    return g

def lp_step(h, n, trust, top_count=40):
    if cp is None:
        return h.copy()
    s = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(s)[-top_count:]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [
        cp.sum(hv) == 0.5 * n,
        hv >= 0.0,
        hv <= 1.0,
        hv - h <= trust,
        h - hv <= trust,
    ]
    for idx in top:
        g = lag_gradient_int(h, n, int(idx - (n - 1)))
        cons.append(float(s[idx]) + cp.sum(cp.multiply(g, hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for name in ('GLOP', 'ECOS', 'CBC', 'CVXOPT'):
        sol = getattr(cp, name, None)
        if sol is None:
            continue
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = hv.value
                if val is not None and np.all(np.isfinite(val)):
                    return project_feasible(val, n)
        except Exception:
            continue
    return h.copy()

def projected_gradient(h, n, end):
    best_h = h.copy()
    best_c = exact_c5(best_h, n)
    alphas = (20.0, 100.0, 500.0, 2000.0, 10000.0, 40000.0)
    cnt = len(alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() >= end:
            break
        rem = max(0.0, end - time.time())
        phase = min(end, time.time() + max(1.0, rem / (cnt - ai)))
        lr = 0.05
        fail = 0
        cur = best_h.copy()
        while time.time() < phase and fail < 40:
            try:
                _, _, g = soft_info(cur, n, alpha)
            except Exception:
                break
            if not np.all(np.isfinite(g)):
                break
            nxt = project_feasible(cur - lr * g, n)
            cnew = exact_c5(nxt, n)
            if cnew < best_c - 1e-14:
                best_c = cnew
                best_h = nxt
                cur = nxt
                fail = 0
                if lr < 0.35:
                    lr *= 1.08
            else:
                lr *= 0.6
                fail += 1
            if lr < 1e-5:
                break
    return best_h, best_c

def optimize_n(h, n, end, rng=None):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = exact_c5(best_h, n)
    if end - time.time() <= 3.0:
        return best_h, best_c
    rem0 = max(0.0, end - time.time())
    pg_end = min(end - 30.0, time.time() + max(5.0, rem0 * 0.45))
    hg, cg = projected_gradient(best_h, n, pg_end)
    if cg < best_c:
        best_h, best_c = hg, cg
    if n <= 700:
        iters = 140 if n <= 400 else 70
        for alpha in (200.0, 2000.0, 10000.0):
            if time.time() > end - 15.0:
                break
            hnew = slsqp_optimize(best_h, n, alpha, iters)
            cnew = exact_c5(hnew, n)
            if cnew < best_c - 1e-13:
                best_h, best_c = hnew, cnew
    if n <= 500 and time.time() < end - 90.0:
        for trust in (0.15, 0.08, 0.04, 0.02):
            if time.time() > end - 90.0:
                break
            hnew = lp_step(best_h, n, trust)
            cnew = exact_c5(hnew, n)
            if cnew < best_c - 1e-13:
                best_h, best_c = hnew, cnew
        if time.time() < end - 10.0:
            h2, c2 = projected_gradient(best_h, n, min(end - 5.0, time.time() + 20.0))
            if c2 < best_c:
                best_h, best_c = h2, c2
    if rng is not None and time.time() < end - 40.0:
        for scale in (0.02, 0.005):
            if time.time() > end - 15.0:
                break
            hp = project_feasible(best_h + rng.normal(0.0, scale, n), n)
            hp2, c2 = projected_gradient(hp, n, min(end - 10.0, time.time() + 25.0))
            if c2 < best_c:
                best_h, best_c = hp2, c2
    if n <= 700 and time.time() < end - 4.0:
        iters = max(30, (140 if n <= 400 else 70) // 2)
        hnew = slsqp_optimize(best_h, n, 50000.0, iters)
        cnew = exact_c5(hnew, n)
        if cnew < best_c:
            best_h, best_c = hnew, cnew
    best_h = project_feasible(best_h, n)
    return best_h, exact_c5(best_h, n)

def candidate_list(h, n, rng):
    h = project_feasible(h, n)
    out = [h]
    out.append(project_feasible(0.5 * (h + h[::-1]), n))
    out.append(project_feasible(h + 0.30 * (h - 0.5), n))
    out.append(project_feasible(h - 0.15 * (h - 0.5), n))
    if n >= 4:
        b = np.zeros(n)
        idx = np.argsort(h)[::-1][:max(1, n // 2)]
        b[idx] = 1.0
        out.append(b)
    try:
        mpts = 400
        a = rng.uniform(0.0, 1.0, mpts)
        x = np.linspace(0.0, 2.0, n)
        counts = ((x[None, :] >= a[:, None]) & (x[None, :] <= a[:, None] + 1.0)).sum(axis=0)
        out.append(project_feasible(counts / mpts, n))
    except Exception:
        pass
    out.append(project_feasible(0.5 + rng.normal(0.0, 0.2, n), n))
    return out

def pick_best_candidate(cands, n):
    best = cands[0]
    bc = exact_c5(best, n)
    for c in cands[1:]:
        cc = exact_c5(c, n)
        if cc < bc:
            best, bc = c, cc
    return best, bc

def resample(h, n):
    if h.size == n:
        return project_feasible(h, n)
    x_old = np.linspace(0.0, 2.0, h.size)
    x_new = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(x_new, x_old, h), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(3.0, float(budget_s) - 4.0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        hp = None
    if hp is not None and hp.size > 4:
        if hp.size % 2:
            hp = hp[:-1]
        n0 = int(hp.size)
        h0 = project_feasible(hp, n0)
    else:
        n0 = 240
        h0 = project_feasible(np.full(n0, 0.5), n0)
    cands = candidate_list(h0, n0, rng)
    hstart, _ = pick_best_candidate(cands, n0)
    end1 = t0 + max(15.0, min(520.0, (deadline - t0) * 0.55))
    hopt, c_opt = optimize_n(hstart, n0, end1, rng)
    if c_opt < best_c:
        best_h, best_c = hopt, c_opt
    n1 = min(600, 2 * n0)
    if n1 % 2:
        n1 -= 1
    if n1 != n0 and time.time() < deadline - 120.0:
        hs = resample(best_h, n1)
        cands = candidate_list(hs, n1, rng)
        hstart, _ = pick_best_candidate(cands, n1)
        rem = deadline - time.time()
        end2 = min(deadline - 60.0, time.time() + max(60.0, rem * 0.55))
        hopt, c_opt = optimize_n(hstart, n1, end2, rng)
        if c_opt < best_c:
            best_h, best_c = hopt, c_opt
    n2 = min(900, int(1.5 * n1))
    if n2 % 2:
        n2 -= 1
    if n2 != len(best_h) and n2 > len(best_h) and time.time() < deadline - 120.0:
        hs = resample(best_h, n2)
        cands = candidate_list(hs, n2, rng)
        hstart, _ = pick_best_candidate(cands, n2)
        rem = deadline - time.time()
        end3 = min(deadline - 80.0, time.time() + max(50.0, rem * 0.55))
        hopt, c_opt = optimize_n(hstart, n2, end3, rng)
        if c_opt < best_c:
            best_h, best_c = hopt, c_opt
    if best_h is not None and time.time() < deadline - 5.0:
        n = len(best_h)
        hfin = project_feasible(best_h, n)
        endf = min(deadline - 1.0, time.time() + max(5.0, min(80.0, deadline - time.time())))
        hfin, cfin = projected_gradient(hfin, n, endf)
        if cfin < best_c:
            best_h, best_c = hfin, cfin
        if n <= 700 and time.time() < deadline - 2.0:
            hnew = slsqp_optimize(hfin, n, 50000.0, 80)
            cnew = exact_c5(hnew, n)
            if cnew < best_c:
                best_h, best_c = hnew, cnew
        best_h = project_feasible(best_h, n)
        best_c = exact_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = exact_c5(best_h, n0)
    return best_h, float(best_c), int(len(best_h))
