```python
        if 0 <= idx1 < n:
            coeff += (1 - h[idx1])
        if 0 <= idx2 < n:
            coeff -= h[idx2]
        a[k] = coeff
    
    # Minimize max_k (c_arr[k] + t*a[k]) for t in [-h[j], 1-h[j]]
    lo = -h[j]
    hi = 1.0 - h[j]
    
    if lo >= hi:
        return 0.0, c_arr
    
    # Bisection on subgradient
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c_arr + mid * a
        k_star = np.argmax(vals)
        slope = a[k_star]
        if slope < -1e-15:
            lo = mid
        elif slope > 1e-15:
            hi = mid
        else:
            break
    
    t_opt = (lo + hi) / 2
    new_c = c_arr + t_opt * a
    return t_opt, new_c
```

Wait, but this doesn't handle the constraint sum(h)=n/2. After changing h[j], the sum changes by t_opt. I need to compensate.

One approach: use a Lagrange multiplier. Add lambda * (sum(h) - n/2) to the objective. But this makes the coordinate-wise optimization more complex.

Simpler approach: after each coordinate step, project back. The projection will change all coordinates slightly, which might undo some of the progress. But if the steps are small, this should work.

Even simpler: use the "swap" approach. When I change h[j] by t, I also change h[j'] by -t (for some j'). This maintains the constraint exactly. The optimal j' would be the one that minimizes the resulting max.

But this is more complex. Let me just use the projection approach.

Actually, you know what, let me just go with L-BFGS-B and the smooth max. It's simpler, well-tested, and should work well. The key is to use high enough beta and enough iterations.

Let me benchmark: L-BFGS-B on n=400 variables with a smooth objective that requires O(n) to evaluate (the correlate). Each iteration of L-BFGS-B evaluates the objective once (and maybe a few more times for line search). With 200 iterations, that's about 400-800 function evaluations = 400-800 correlates of length 400 = 160K-320K operations. That's very fast (< 1 second).

The issue is that L-BFGS-B uses numerical gradients by default, which requires n+1 = 401 function evaluations per gradient. With 200 iterations, that's 80,200 function evaluations = 32M operations. Still fast (< 5 seconds).

If I provide the analytical gradient, it's even faster. But let me first try without it.

Actually wait, let me reconsider. For L-BFGS-B with numerical gradients:
- Each gradient = n+1 function evaluations = 401 * O(400) = 160K ops
- Each iteration uses 1 gradient + a few line search evaluations
- 200 iterations: 200 * (160K + 5*400) = 200 * 162K = 32M ops
- In Python with numpy, this is maybe 10-30 seconds.

That's fine for our budget.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, 'full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def binary_ls(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, 'full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    z_list = np.where(h == 0)[0].tolist()
    o_list = np.where(h == 1)[0].tolist()
    
    while time.time() < deadline:
        zi = rng.randint(len(z_list))
        oi = rng.randint(len(o_list))
        i = z_list[zi]
        j = o_list[oi]
        
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + j - i
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + i - j
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        nc = (corr + delta).max() * dx
        if nc < cur_c - 1e-15:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            z_list[zi] = j
            o_list[oi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def lbgfsb_refine(h_init, n, deadline, seed=42):
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    
    betas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    
    for beta in betas:
        if time.time() > deadline:
            break
        
        def obj(x):
            c = np.correlate(x, 1-x, 'full') * dx
            m = c.max()
            shifted = beta * (c - m)
            # Add small epsilon for numerical safety
            logsum = np.log(np.exp(shifted).sum())
            return m + logsum / beta
        
        def obj_grad(x):
            c = np.correlate(x, 1-x, 'full') * dx
            m = c.max()
            shifted = beta * (c - m)
            w = np.exp(shifted)
            w_sum = w.sum()
            w /= w_sum
            # grad = beta * dx * w @ J where J is Jacobian of c
            # J[k,j] = dc[k]/dh[j]
            # For now, use numerical gradient (simpler)
            return None  # Let scipy handle it
        
        res = minimize(obj, best_h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10, 'maxlines': 30})
        
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c
