def unit_swap_delta(h, i, j, n):
    c = n - 1
    d = np.zeros(2*n - 1)
    rev = h[::-1]
    # i leaves: a=-1
    d[c - i : c - i + n] += h - 1.0
    start = c - (n - 1) + i
    d[start : start + n] += rev
    # j enters: a=+1
    d[c - j : c - j + n] += 1.0 - h
    start = c - (n - 1) + j
    d[start : start + n] += -rev
    q1 = c + (j - i)
    q2 = c + (i - j)
    return d, q1, q2

def candidate_pairs(h, corr, curmax, max_top, eps=1e-7):
    n=len(h)
    active = np.where(corr >= curmax - eps)[0]
    if active.size == 0:
        active = np.where(corr >= corr.max() - 1e-8)[0]
    deriv = -np.inf * np.ones((n,n))
    for pos in active:
       d = int(pos - (n-1))
       g = grad_lag(d, h, n)
       mat = g[None,:] - g[:,None]
       deriv = np.fmax(deriv, mat)
    tmax = np.minimum(h[:,None], 1.0 - h[None,:])

    idx = np.argwhere(mask)
    vals = deriv[idx]
    order = np.argsort(vals)
    k = min(max_top, len(order))
    return idx[order[:k]], vals[order[:k]]

import time
import math
import numpy as np
from scipy.optimize import minimize

TARGET_C = 0.38078
LINE_FRACS = np.linspace(0.0, 1.0, 17)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))


def project_feasible(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(20000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] - d)
    return h


def initial_solution():

        h = np.array(initial_h_values, dtype=float)

def grad_lag(d, h, n):
    if d == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if d > 0:
        r = d
        g[:n - r] += 1.0 - h[r:]
        g[r:] += -h[:-r]
    else:
        r = -d
        g[r:] += 1.0 - h[:-r]
        g[:-r] += -h[r:]
    return g


def soft_obj(x, n, alpha, mu):
    g = np.correlate(x, 1.0 - x, mode="full")
    m = float(g.max())
    z = alpha * (g - m)
    zm = float(z.max())
    s = float(np.exp(z - zm).sum())
    val = m + (zm + math.log(s)) / alpha
    d = x.sum() - 0.5 * n
    return val + 0.5 * mu * d * d


def soft_jac(x, n, alpha, mu):
    g = np.correlate(x, 1.0 - x, mode="full")
    m = float(g.max())
    z = alpha * (g - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    w = e / e.sum()
    grad = np.zeros(n)
    h = x
    center = n - 1
    for pos in range(2 * n - 1):
        wd = w[pos]
        if wd <= 1e-12:
            continue
        d = pos - center
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        elif d > 0:
            r = d
            grad[:n - r] += wd * (1.0 - h[r:])
            grad[r:] += -wd * h[:-r]
        else:
            r = -d
            grad[r:] += wd * (1.0 - h[:-r])
            grad[:-r] += -wd * h[r:]
    d = x.sum() - 0.5 * n
    grad += mu * d
    return grad


def smooth_l1(h, n, alpha, mu, maxiter):
    x0 = np.clip(h, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            soft_obj,
            x0,
            method="L-BFGS-B",
            jac=soft_jac,
            args=(n, alpha, mu),
            bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-9, "maxls": 12},
        )
        return project_feasible(res.x, 0.5 * n)
    except Exception:
        return x0


def unit_swap_delta(h, i, j, n):
    center = n - 1
    d = np.zeros(2 * n - 1)
    rev = h[::-1]
    st = center - i
    d[st:st + n] += h - 1.0
    d[i:i + n] += rev
    st = center - j
    d[st:st + n] += 1.0 - h
    d[j:j + n] += -rev
    return d, center + (j - i), center + (i - j)


def pair_eval(corr, base, q1, q2, l0, t):
    if t == 0.0:
        return float(corr.max())
    v = float((corr + t * base).max())
    vt = corr[q1] + t * base[q1] + t * t
    if vt > v:
        v = vt
    vt = corr[q2] + t * base[q2] + t * t
    if vt > v:
        v = vt
    vt = corr[l0] + t * base[l0] - 2.0 * t * t
    if vt > v:
        v = vt
    return v


def golden_min_pair(corr, base, q1, q2, l0, a, b, iters):
    if b - a <= 1e-14:
        return 0.5 * (a + b), pair_eval(corr, base, q1, q2, l0, 0.5 * (a + b))
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = pair_eval(corr, base, q1, q2, l0, c)
    fd = pair_eval(corr, base, q1, q2, l0, d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = pair_eval(corr, base, q1, q2, l0, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = pair_eval(corr, base, q1, q2, l0, d)
    t = 0.5 * (a + b)
    return t, pair_eval(corr, base, q1, q2, l0, t)


def pair_line_search(corr, base, q1, q2, l0, tmax, fracs):
    if tmax <= 1e-14:
        return 0.0, float(corr.max())
    best_t = 0.0
    best_f = float(corr.max())
    for f in fracs:
        t = float(f) * tmax
        v = pair_eval(corr, base, q1, q2, l0, t)
        if v < best_f:
            best_f = v
            best_t = t
    step = tmax / max(1, len(fracs) - 1)
    a = max(0.0, best_t - step)
    b = min(float(tmax), best_t + step)
    if b - a > 1e-12:
        t2, f2 = golden_min_pair(corr, base, q1, q2, l0, a, b, 12)
        if f2 < best_f:
            best_f = f2
            best_t = t2
    return best_t, best_f


def candidate_pairs(h, corr, curmax, max_top):
    n = len(h)
    near = np.where(corr >= curmax - 1e-7)[0]
    if near.size == 0:
        near = np.argsort(corr)[-1:]
    if near.size > 24:
        active = near[np.argsort(corr[near])[-24:]]
    else:
        active = near
    deriv = -np.inf * np.ones((n, n))
    for pos in active:
        d = int(pos - (n - 1))
        g = grad_lag(d, h, n)
        mat = g[None, :] - g[:, None]
        np.fmax(deriv, mat, out=deriv)
    tmaxm = np.minimum(h[:, None], 1.0 - h[None, :])
    mask = (deriv < -1e-7) & (tmaxm > 1e-6)
    np.fill_diagonal(mask, False)
    idx = np.argwhere(mask)
    if idx.size == 0:
        return idx, np.empty(0)
    vals = deriv[idx]
    order = np.argsort(vals)
    k = min(max_top, len(order))
    return idx[order[:k]], vals[order[:k]]


def pair_local_refine(h_init, n, deadline, seed=42, stop_c=None):

    h = project_feasible(np.clip(h_init, 0.0, 1.0), target).copy()

    dx = 2.0 / n

    stalls = 0

    while time.time() < deadline:
        h = project_feasible(h, target)
        corr = compute_corr(h).astype(float)

        if cur < best_count:

            if stop_c is not None and best_count * dx <= stop_c + 1e-12:
