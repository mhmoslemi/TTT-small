def gradient_from_weights(p, x):
    n=x.size; n1=n-1
    q = p + p[::-1]
    q[n1] = 0.0
    T = np.convolve(x, q, mode='full')[n1:n1+n]
    pos = p[n1+1:]
    pos_cum = np.concatenate(([0.0], np.cumsum(pos)))  # length n
    negw = p[:n1][::-1]

import numpy as np
import time
import math
from scipy.optimize import minimize, differential_evolution

OBJ_ALPHA = 100.0
OBJ_RHO = 0.0
STOP_TIME = 1e100


def compute_c5(h):
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def project_feasible(y):
    y = np.asarray(y, dtype=float)
    n = y.size
    target = 0.5 * n
    low = float(np.min(y)) - 1.0
    high = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (low + high)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            low = mid
        else:
            high = mid
    h = np.clip(y - 0.5 * (low + high), 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        idx = int(np.argmax(1.0 - h))
        if h[idx] < 1.0:
            h[idx] = min(1.0, h[idx] + diff)
    elif diff < -1e-14:
        idx = int(np.argmax(h))
        if h[idx] > 0.0:
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def make_binary(h):
    h = np.asarray(h, dtype=float)
    target = h.size // 2
    if h.size % 2 == 0 and np.all((h <= 0.0) | (h >= 1.0)) and abs(h.sum() - target) < 1e-9:
        return (h > 0.5).astype(float)
    out = np.zeros(h.size, dtype=float)
    out[np.argsort(h)[::-1][:target]] = 1.0
    return out


def upscale_to(h, n2):
    h = np.asarray(h, dtype=float)
    n1 = h.size
    if n1 == n2:
        return h.copy()
    if n2 % n1 == 0:
        return np.repeat(h, n2 // n1)
    return np.interp(np.linspace(0.0, 2.0, n2), np.linspace(0.0, 2.0, n1), h)


def grad_from_weights(p, x):
    n = x.size
    n1 = n - 1
    q = p + p[::-1]
    q[n1] = 0.0
    t = np.convolve(x, q, mode="full")[n1:n1 + n]
    pos_cum = np.concatenate((np.zeros(1), np.cumsum(p[n1 + 1:])))
    neg_cum = np.concatenate((np.zeros(1), np.cumsum(p[:n1][::-1])))
    idx = np.arange(n)
    w = pos_cum + neg_cum[n1 - idx]
    return (w - t + p[n1] * (1.0 - 2.0 * x)) * (2.0 / n)


def soft_value_grad(x):
    n = x.size
    f = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(np.max(f))
    a = OBJ_ALPHA
    L = 2 * n - 1
    if a < 1e-8:
        p = np.zeros(L)
        p[int(np.argmax(f))] = 1.0
        val = m
    else:
        z = a * (f - m)
        e = np.exp(z)
        s = e.sum()
        if s <= 0.0 or not np.isfinite(s):
            p = np.full(L, 1.0 / L)
        else:
            p = e / s
        val = m + math.log(s) / a
    g = grad_from_weights(p, x)
    if OBJ_RHO > 0.0:
        err = x.sum() - 0.5 * n
        val += OBJ_RHO * err * err
        g += 2.0 * OBJ_RHO * err
    return float(val), g


def minimizer_callback(xk):
    return time.time() > STOP_TIME


def de_callback(xk, conv):
    return time.time() > STOP_TIME


def optim_l_bfgs(x0, alpha, rho, limit, maxiter=500):
    global OBJ_ALPHA, OBJ_RHO, STOP_TIME
    x = project_feasible(x0)
    OBJ_ALPHA = alpha
    OBJ_RHO = rho
    STOP_TIME = limit
    try:
        res = minimize(
            soft_value_grad, x, method="L-BFGS-B", jac=True,
            options={"maxiter": maxiter, "ftol": 1e-12, "maxls": 50,
                     "callback": minimizer_callback}
        )
        return project_feasible(res.x)
    except Exception:
        return x


def eq_fun(x):
    return x.sum() - 0.5 * x.size


def eq_jac(x):
    return np.ones_like(x)


def optim_slsqp(x0, alpha, limit, maxiter=200):
    global OBJ_ALPHA, OBJ_RHO, STOP_TIME
    x = project_feasible(x0)
    OBJ_ALPHA = alpha
    OBJ_RHO = 0.0
    STOP_TIME = limit
    bounds = [(0.0, 1.0)] * x.size
    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    try:
        res = minimize(
            soft_value_grad, x, method="SLSQP", bounds=bounds, constraints=cons,
            jac=True, options={"maxiter": maxiter, "ftol": 1e-12, "maxls": 50,
                               "callback": minimizer_callback}
        )
        return project_feasible(res.x)
    except Exception:
        return x


def continuous_work(h0, t_start, t_end, seed):
    h = project_feasible(h0)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    total = max(1.0, t_end - time.time())
    plans = ((20.0, 1.0, 0.25), (100.0, 10.0, 0.50), (500.0, 50.0, 0.75), (2000.0, 150.0, 1.0))
    for a, rho, frac in plans:
        sub = min(t_start + total * frac, t_end)
        if time.time() >= sub:
            continue
        h2 = optim_l_bfgs(cur_h, a, rho, sub, 500 if cur_h.size <= 400 else 300)
        h2 = project_feasible(h2)
        c2 = compute_c5(h2)
        if c2 <= cur_c + 1e-10:
            cur_h = h2
            cur_c = c2
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    if cur_h.size <= 350 and time.time() < t_end - 2.0:
        h3 = optim_slsqp(cur_h, 2000.0, min(t_end, time.time() + (t_end - time.time()) * 0.30), 180)
        c3 = compute_c5(h3)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
        if c3 <= cur_c + 1e-10:
            cur_h = h3
            cur_c = c3
    return best_h, best_c


def de_objective(x):
    return compute_c5(project_feasible(x))


def de_work(n, t_start, t_end, seed):
    global STOP_TIME
    if t_end - time.time() < 25.0:
        return np.full(n, 0.5), 1e9
    bounds = [(0.0, 1.0)] * n
    STOP_TIME = t_end - 1.0
    try:
        res = differential_evolution(
            de_objective, bounds=bounds, seed=seed, maxiter=10000,
            popsize=10, tol=1e-8, init="latinhypercube", polish=False,
            updating="immediate", workers=1, callback=de_callback
        )
        h = project_feasible(res.x)
        return h, compute_c5(h)
    except Exception:
        return np.full(n, 0.5), 1e9


def binary_sa(h0, limit, seed=42, t_start=0.03, t_end=1e-5, nsteps=250000, patience=30000):
    h = make_binary(h0)
    n = h.size
    if n < 4 or time.time() >= limit:
        return h, compute_c5(h)
    n1 = n - 1
    dx = 2.0 / n
    hrev = h[::-1]
    tmp = np.zeros(2 * n - 1, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(c.max()) * dx
    best = cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T = t_start
    cool = (t_end / t_start) ** (1.0 / float(nsteps))
    no_best = 0
    step = 0
    while time.time() < limit:
        i = rng.randint(n)
        while h[i] != 0.0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] != 1.0:
            j = rng.randint(n)
        tmp.fill(0.0)
        s = n1 - i
        tmp[s:s + n] -= h
        s = i
        tmp[s:s + n] += 1.0
        tmp[s:s + n] -= hrev
        s = n1 - j
        tmp[s:s + n] += h
        s = j
        tmp[s:s + n] -= 1.0
        tmp[s:s + n] += hrev
        tmp[n1] -= 2.0
        tmp[n1 + j - i] += 1.0
        tmp[n1 + i - j] += 1.0
        c += tmp
        nc = float(c.max()) * dx
        d = nc - cur
        if d < 0.0 or (T > 1e-12 and rng.random() < math.exp(-d / T if d / T < 50.0 else 1.0)):
            h[i] = 1.0
            h[j] = 0.0
            cur = nc
            if nc < best:
                best = nc
                best_h = h.copy()
                no_best = 0
            else:
                no_best += 1
        else:
            c -= tmp
            no_best += 1
        step += 1
        T *= cool
        if step % 10000 == 0:
            c = np.rint(c)
            cur = float(c.max()) * dx
            if cur < best:
                best = cur
                best_h = h.copy()
                no_best = 0
        if no_best > patience or T < t_end:
            if rng.random() < 0.20:
                h = np.roll(h, rng.randint(1, n))
                hrev = h[::-1]
            else:
                jump = 3 + rng.randint(12)
                for _ in range(jump):
                    ii = rng.randint(n)
                    while h[ii] != 0.0:
                        ii = rng.randint(n)
                    jj = rng.randint(n)
                    while h[jj] != 1.0:
                        jj = rng.randint(n)
                    h[ii] = 1.0
                    h[jj] = 0.0
            c = np.correlate(h, 1.0 - h, mode="full")
            cur = float(c.max()) * dx
            no_best = 0
            if cur < best:
                best = cur
                best_h = h.copy()
            T = t_start * (0.25 + 0.75 * rng.random())
            step = 0
    return best_h, best
