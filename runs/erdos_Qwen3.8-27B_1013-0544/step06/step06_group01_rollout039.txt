def lse_count(q, alpha):
   mfloat = float(np.max(q))
   a=alpha
   sh=a*(q-mfloat)
   mx=float(np.max(sh))
   s=float(np.sum(np.exp(sh-mx)))
   if not np.isfinite(s): return mfloat
   return float(mfloat + (mx + np.log(s))/a)

def binary_project(h, n):

        return project_feasible(h, n)
    b = (h >= 0.5).astype(float)

def lse_weights(q, alpha):
    m = float(np.max(q))
    sh = alpha * (q - m)
    mx = float(np.max(sh))
    ex = np.exp(sh - mx)
    s = float(np.sum(ex))
    if s <= 1e-300:

    return ex / s


def grad_lse(h, w):
    n = len(h)
    N = n - 1
    g = np.zeros(n, dtype=float)
    om = 1.0 - h
    hr = h[::-1]
    pref = np.empty(w.size + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    for p in range(n):

        b = 2 * N - p + 1
        s1 = pref[b] - pref[a]
        d1 = float(np.dot(w[a:b], om))
        d2 = float(np.dot(w[p:p + n], hr))
        g[p] = s1 - d1 - d2
    return g

def golden_pair(a, b, q, L, Q, iters=24):
    phi = 0.6180339887498949
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = pair_max(q, L, Q, c)
    fd = pair_max(q, L, Q, d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = pair_max(q, L, Q, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = pair_max(q, L, Q, d)
    return 0.5 * (a + b)


def try_pair_update(h, q, L, Q, p, rv, best, grid=41):
    if p == rv:
        return 0.0, best
    hp = float(h[p])
    hq = float(h[rv])
    lo = max(-hp, hq - 1.0)
    hi = min(1.0 - hp, hq)
    if hi - lo <= 1e-12:
        return 0.0, best

    n = len(h)
    N = n - 1
    L.fill(0.0)
    Q.fill(0.0)

    om = 1.0 - h
    hr = h[::-1]

    L[N - p:N - p + n] += om
    L[p:p + n] -= hr
    L[N - rv:N - rv + n] -= om
    L[rv:rv + n] += hr

    Q[N] = -2.0
    Q[N + rv - p] += 1.0
    Q[N + p - rv] += 1.0

    ds = np.linspace(lo, hi, grid)
    vals_c = pair_eval(q, L, Q, ds)
    fd = np.clip(FINE, lo, hi)
    vals_f = pair_eval(q, L, Q, fd)

    vals = np.concatenate((vals_c, vals_f))
    ds_all = np.concatenate((ds, fd))
    bi = int(np.argmin(vals))
    bv = float(vals[bi])
    bd = float(ds_all[bi])

    if bv < best - TOL_IMP:
        half = max(1e-4, 0.04)
        a = max(lo, bd - half)
        b = min(hi, bd + half)
        if b - a > 1e-7:
            dg = golden_pair(a, b, q, L, Q, 22)
            vg = pair_max(q, L, Q, dg)
            if vg < bv:
                bv = vg
                bd = dg

        if bv < best - TOL_IMP:
            q += bd * L + bd * bd * Q
            hp += bd
            hq -= bd
            changed = False
            if hp < 0.0:
                hp = 0.0
                changed = True
            elif hp > 1.0:
                hp = 1.0
                changed = True
            if hq < 0.0:
                hq = 0.0
                changed = True
            elif hq > 1.0:
                hq = 1.0
                changed = True
            h[p] = hp
            h[rv] = hq
            if changed:
                q[:] = compute_q(h)
                bv = float(np.max(q))
            return bd, bv
    return 0.0, best


def targeted_pair_search(h, deadline, rng, grid=41, random_grid=25):
    n = len(h)
    N = n - 1
    m = 2 * N + 1
    dx = 2.0 / n
    h = project_feasible(h, n)
    q = compute_q(h)
    best = float(np.max(q))

    L = np.zeros(m, dtype=float)
    Q = np.zeros(m, dtype=float)
    seen = np.zeros(n * n, dtype=bool)
    stall = 0
    accept_count = 0

    max_lags = 40
    take_lag = 50
    max_c = max(1600, 6 * n)
    if max_c > 5000:
        max_c = 5000

    while time.time() < deadline:
        margin = max(0.5, 0.02 * best)
        active = np.where(q >= best - margin)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(q))])
        elif active.size > max_lags:
            order = np.argsort(-q[active])
            active = active[order[:max_lags]]

        seen.fill(False)
        cands = []
        partner_min = np.argsort(h)[:3]
        partner_max = np.argsort(-h)[:3]
        partners = np.concatenate((partner_min, partner_max))

        for kk in active:
            s = int(kk - N)
            istart = max(0, -s)
            iend = min(N, N - s)
            if iend < istart:
                continue

            if s == 0:
                val0 = h * (1.0 - h)
                take = min(50, val0.size)
                if take < val0.size:
                    idx0 = np.argpartition(-val0, take)[:take]
                else:
                    idx0 = np.arange(val0.size)
                for ii in idx0:
                    i = int(ii)
                    for pt in partners:
                        pt = int(pt)
                        if i == pt:
                            continue
                        key = (i * n + pt) if i < pt else (pt * n + i)
                        if not seen[key]:
                            seen[key] = True
                            cands.append((i, pt))
                            if len(cands) >= max_c:
                                break
                    if len(cands) >= max_c:
                        break
                if len(cands) >= max_c:
                    break
                continue

            ai = istart
            aj = istart + s
            contrib = h[ai:iend + 1] * (1.0 - h[aj:iend + s + 1])
            take = min(take_lag, contrib.size)
            if take < contrib.size:
                idx = np.argpartition(-contrib, take)[:take]
            else:
                idx = np.arange(contrib.size)
            for ii in idx:
                i = istart + int(ii)
                j = i + s
                if i == j:
                    continue
                key = (i * n + j) if i < j else (j * n + i)
                if not seen[key]:
                    seen[key] = True
                    cands.append((i, j))
                    if len(cands) >= max_c:
                        break
            if len(cands) >= max_c:
                break

        nrand = max(700, int(2.0 * n))
        if nrand > 4000:
            nrand = 4000
        for _ in range(nrand):
            if len(cands) >= max_c:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            key = (i * n + j) if i < j else (j * n + i)
            if not seen[key]:
                seen[key] = True
                cands.append((i, j))

        if len(cands) > 1:
            arr = np.array(cands, dtype=int)
            rng.shuffle(arr)
            cands = arr.tolist()

        improved = False
        for p, rv in cands:
            if time.time() >= deadline:
                break
            d, newbest = try_pair_update(h, q, L, Q, int(p), int(rv), best, grid)
            if d != 0.0:
                improved = True
                best = newbest
                accept_count += 1
                if accept_count % 25 == 0:
                    q[:] = compute_q(h)
                    best = float(np.max(q))

        if improved:
            q[:] = compute_q(h)
            best = float(np.max(q))
            stall = 0
            continue

        count = max(700, int(2.5 * n))
        if count > 3500:
            count = 3500
        rand_imp = False
        for _ in range(count):
            if time.time() >= deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d, nb = try_pair_update(h, q, L, Q, i, j, best, random_grid)
            if d != 0.0:
                rand_imp = True
                best = nb
                accept_count += 1
                if accept_count % 25 == 0:
                    q[:] = compute_q(h)
                    best = float(np.max(q))

        if rand_imp:
            q[:] = compute_q(h)
            best = float(np.max(q))
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break

    h = project_feasible(h, n)
    q = compute_q(h)
    return h, float(np.max(q)) * dx


def resample_h(h, n_new):
    n = len(h)
    old = np.linspace(0.0, 2.0, n)
    new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(new, old, h), n_new)


def optimize_start(h0, deadline, rng):
    n = len(h0)
    h = project_feasible(h0, n)
    c = compute_c5(h)

    if deadline - time.time() > 45.0:
        pd = min(deadline - 40.0, time.time() + min(45.0, 0.25 * max(0.0, deadline - time.time())))
        hp, cp = pgt_refine(h, pd, (80.0, 350.0), lr=0.12, max_iter=50)
        if cp < c:
            h, c = hp, cp

    if deadline - time.time() > 25.0:
        ht, ct = targeted_pair_search(h, deadline - 18.0, rng, grid=41, random_grid=25)
        if ct < c:
            h, c = ht, ct

    if deadline - time.time() > 25.0:
        pd = min(deadline - 12.0, time.time() + min(25.0, 0.20 * max(0.0, deadline - time.time())))
        hp, cp = pgt_refine(h, pd, (250.0, 800.0), lr=0.08, max_iter=30)
        if cp < c:
            h, c = hp, cp

    if deadline - time.time() > 12.0:
        ht, ct = targeted_pair_search(h, deadline - 4.0, rng, grid=33, random_grid=17)
        if ct < c:
            h, c = ht, ct

    h = project_feasible(h, n)
    c = compute_c5(h)
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        x = np.linspace(0.0, 2.0, n0)
        h0 = np.clip(0.5 + 0.45 * np.sin(3.0 * np.pi * x), 0.0, 1.0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    reserve = max(150.0, 0.22 * float(budget_s))
    phase1 = min(deadline - reserve, t0 + max(180.0, 0.42 * float(budget_s)))
    if phase1 > time.time() + 60.0:
        h1, c1 = optimize_start(h0, phase1, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if deadline - time.time() > 160.0:
        nb = len(best_h)
        hb = binary_project(best_h, nb)
        dl = min(deadline - 100.0, time.time() + min(130.0, 0.14 * float(budget_s)))
        h2, c2 = optimize_start(hb, dl, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if deadline - time.time() > 210.0 and len(best_h) > 120:
        ns = len(best_h) // 2
        if ns < 80:
            ns = 80
        ns = 2 * (ns // 2)
        if 0 < ns < len(best_h):
            hs = resample_h(best_h, ns)
            dl = min(deadline - 90.0, time.time() + min(170.0, 0.16 * float(budget_s)))
            h3, c3 = optimize_start(hs, dl, rng)
            if c3 < best_c:
                best_h, best_c = h3, c3

    if deadline - time.time() > 190.0 and len(best_h) < 450:
        nl = 2 * len(best_h)
        if nl > 700:
            nl = 700
        nl = 2 * (nl // 2)
        if nl > len(best_h):
            hl = resample_h(best_h, nl)
            dl = min(deadline - 60.0, time.time() + min(160.0, 0.16 * float(budget_s)))
            h4, c4 = optimize_start(hl, dl, rng)
            if c4 < best_c:
                best_h, best_c = h4, c4

    if deadline - time.time() > 20.0:
        hf, cf = targeted_pair_search(best_h, deadline - 2.0, rng, grid=33, random_grid=17)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
