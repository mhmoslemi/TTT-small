    target = 0.5 * n
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(70):
       mid = 0.5*(lo+hi)
       s = float(np.sum(np.clip(v-mid, 0.0, 1.0)))
       if s > target:
          lo = mid
       else:
          hi = mid
    lam = 0.5*(lo+hi)
    h = np.clip(v-lam,0.0,1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
       if diff > 0:
           room = 1.0 - h
           tot = float(room.sum())
           if tot > 1e-15:
              h += (diff/tot)*room
           else:
              h = np.clip(h + diff/n,0,1)
       else:
           tot = float(h.sum())
           if tot > 1e-15:
              h += (diff/tot)*h
           else:
              h = np.clip(h + diff/n,0,1)
       h = np.clip(h,0,1)
       # tiny final adjustment
       diff = target - float(h.sum())
       if abs(diff) > 1e-14:
           # adjust first coordinate with room
           h[0] += diff
    h = np.clip(h,0,1)
    return h

import numpy as np
import time

POS_STEPS = np.array([1e-5,2e-5,5e-5,1e-4,2e-4,5e-4,1e-3,2e-3,5e-3,1e-2,2e-2,5e-2], dtype=float)
FRACTIONS = (0.1,0.2,0.3,0.5,0.7,0.9,1.0)

def compute_c5(h,n=None):
    if n is None: n=len(h)
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*(2.0/float(n)))

def project_box_sum(v,n):
    v=np.asarray(v,dtype=float)
    target=0.5*float(n)
    lo=float(np.min(v))-1.0
    hi=float(np.max(v))+1.0
    for _ in range(70):
       mid=0.5*(lo+hi)
       s=float(np.sum(np.clip(v-mid,0.0,1.0)))
       if s>target: lo=mid
       else: hi=mid
    h=np.clip(v-0.5*(lo+hi),0.0,1.0)
    diff=target-float(h.sum())
    if abs(diff)>1e-14:
       interior=(h>1e-12)&(h<1.0-1e-12)
       cnt=int(np.count_nonzero(interior))
       if cnt:
          h[interior]+=diff/cnt
       else:
          if diff>0:
             room=1.0-h; sr=float(room.sum())
             if sr>1e-18: h+=diff*room/sr
          else:
             sr=float(h.sum())
             if sr>1e-18: h+=diff*h/sr
       h=np.clip(h,0.0,1.0)
    return h

def compute_U(h,n):
    L=2*n-1
    U=np.zeros((n,L),dtype=float)
    for r in range(L):
       l=n-1-r
       a = 0 if l >= 0 else -l
       b = min(n, n-l)
       if b > a:
          U[a:b,r] += 1.0
          U[a:b,r] -= h[a+l:b+l]
          lo2 = max(0, a+l)
          hi2 = min(n, b+l)
          if hi2 > lo2:
             U[lo2:hi2,r] -= h[lo2-l:hi2-l]
    return U

def softmax_weights(c,beta):
    mc=float(np.max(c))
    z=beta*(c-mc)
    e=np.exp(z)
    s=float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
       p=np.zeros_like(c); p[int(np.argmax(c))]=1.0; return p
    return e/s

def projected_gradient(h,n,deadline,rng,beta=30.0,eta0=0.15):
    h=project_box_sum(h,n)
    best=compute_c5(h,n)
    eta=eta0
    stag=0

       c=np.correlate(h,1.0-h,mode='full')
       U=compute_U(h,n)
       p=softmax_weights(c,beta)
       grad=U.T @ p
       if not np.all(np.isfinite(grad)): break
       gmax=float(np.max(np.abs(grad)))

          stag+=1

import numpy as np
import time

POS_STEPS = np.array(
    [1e-5, 2e-5, 5e-5, 1e-4, 2e-4, 5e-4,
     1e-3, 2e-3, 5e-3, 1e-2, 2e-2, 5e-2], dtype=float
)
FRACTIONS = (0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0)


def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def project_box_sum(v, n):
    v = np.asarray(v, dtype=float)
    target = 0.5 * float(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior))
        if cnt:
            h[interior] += diff / float(cnt)
        elif diff > 0:
            room = 1.0 - h
            sr = float(room.sum())
            if sr > 1e-18:
                h += (diff / sr) * room
        else:
            sr = float(h.sum())
            if sr > 1e-18:
                h += (diff / sr) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_U(h, n):
    L = 2 * n - 1
    U = np.zeros((n, L), dtype=float)
    for r in range(L):
        l = n - 1 - r
        a = 0 if l >= 0 else -l
        b = min(n, n - l)
        if b > a:
            U[a:b, r] += 1.0
            U[a:b, r] -= h[a + l:b + l]
            lo2 = max(0, a + l)
            hi2 = min(n, b + l)
            if hi2 > lo2:
                U[lo2:hi2, r] -= h[lo2 - l:hi2 - l]
    return U


def softmax_weights(c, beta):
    mc = float(np.max(c))
    e = np.exp(beta * (c - mc))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def projected_gradient(h, n, deadline, rng, beta=30.0, eta0=0.15):
    h = project_box_sum(h, n)
    best = compute_c5(h, n)
    eta = eta0
    stagnant = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        U = compute_U(h, n)
        p = softmax_weights(c, beta)
        grad = U.T @ p
        if not np.all(np.isfinite(grad)):
            break
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-12:
            break
        improved = False
        tried = eta
        while tried > 1e-5:
            v = project_box_sum(h - tried * grad, n)
            cv = compute_c5(v, n)
            if cv < best - 1e-13:
                h = v
                best = cv
                improved = True
                break
            tried *= 0.5
        if improved:
            stagnant = 0
            eta = min(eta0, max(tried * 1.3, 0.005))
        else:
            stagnant += 1
            if stagnant >= 4:
                break
            eta = max(1e-4, eta * 0.5)
    return h, best


def evaluate_pair_delta(c, A_buf, work, maxpos, maxneg, idxs, coefs, best_M):
    best_delta = 0.0
    steps = POS_STEPS
    fracs = FRACTIONS
    mlen = len(idxs)

    if maxpos > 1e-7:

            m = float(np.max(work))

                best_delta = delta
        for delta in steps:

def apply_pair(h, c, a, b, delta, A_buf, work, idxs, coefs):
    h[a] += delta
    h[b] -= delta
    if h[a] < 0.0 or h[a] > 1.0 or h[b] < 0.0 or h[b] > 1.0:
        h[a] = float(np.clip(h[a], 0.0, 1.0))
        h[b] = float(np.clip(h[b], 0.0, 1.0))
        c[:] = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(c))
    np.multiply(A_buf, delta, out=work)
    c += work
    dd = delta * delta
    for k in range(len(idxs)):
        c[idxs[k]] += coefs[k] * dd
    return float(np.max(c))


def pair_polish(h, n, deadline, rng):

            coefs.append(1.0)
            idxs.append(n - 1 + d)
            coefs.append(1.0)
        qidx.append(idxs)
        qcoef.append(coefs)

    A_buf = np.empty(L)
    best_A = np.empty(L)
    work = np.empty(L)

    beta = 20.0
    T = 20
    no_improve = 0
    move_count = 0

        p = softmax_weights(c, beta)

        best_delta = 0.0

def resample_values(src, n):
    m = len(src)
    if m == n:
        return src.copy()
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return project_box_sum(np.interp(xn, xo, src), n)
