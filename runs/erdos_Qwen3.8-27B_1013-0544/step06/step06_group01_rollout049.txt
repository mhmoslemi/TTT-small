def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coordinate_descent(h, n, deadline, grid=60, max_passes=300, penalty=5.0):

    current_max = float(np.max(c))

            if time.time() > deadline:

            k_hi = min(n2, n - 1 + p)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coordinate_descent(h, n, deadline, grid=80, max_passes=300, penalty=3.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    n2 = 2 * n - 2
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                return h, current_max * (2.0 / n)
            if h[p] <= 1e-12 or h[p] >= 1.0 - 1e-12:
                continue
            k_lo = max(0, n - 1 - p)
            k_hi = min(n2, n - 1 + p)
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < n2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            j1 = n - 1 - ks + p
            j2 = ks - n + 1 + p
            s = np.zeros(len(ks))
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            if np.any(v1):
                s[v1] += (1.0 - h[j1[v1]])
            if np.any(v2):
                s[v2] -= h[j2[v2]]
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            deltas = np.linspace(d_lo, d_hi, grid)
            base = c[ks][np.newaxis, :] + deltas[:, np.newaxis] * s[np.newaxis, :]
            if n - 1 >= k_lo and n - 1 <= k_hi:
                diag = n - 1 - k_lo
                base[:, diag] -= deltas[:, np.newaxis] ** 2
            vals = np.maximum(um, base.max(axis=1))
            vals += penalty * (sumh + deltas - n / 2.0) ** 2
            bi = int(np.argmin(vals))
            best_d = deltas[bi]
            if abs(best_d) > 1e-10:
                c[ks] += best_d * s
                if n - 1 >= k_lo:
                    c[n - 1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def fourier_moves(h, n, deadline, max_rounds=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        H = np.fft.fft(h)
        improved = False
        n_modes = min(n // 4, 15)
        for k in range(1, n_modes + 1):
            if time.time() > deadline:
                return h, best_c
            orig_k = H[k].copy()
            orig_nk = H[n - k].copy()
            base_amp = np.abs(orig_k)
            orig_phase = np.angle(orig_k) if base_amp > 1e-10 else 0.0
            scales = [0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0]
            phases = [0, np.pi/3, 2*np.pi/3, 4*np.pi/3, -np.pi/3, -2*np.pi/3]
            found = False
            for sc in scales:
                if found:
                    break
                for ps in phases:
                    if time.time() > deadline:
                        return h, best_c
                    H[k] = base_amp * sc * np.exp(1j * (orig_phase + ps))
                    H[n - k] = np.conj(H[k])
                    h_new = np.fft.ifft(H).real
                    h_new = project_feasible(h_new, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-10:
                        h = h_new
                        best_c = c_new
                        improved = True
                        found = True
                        break
        H[k] = orig_k
        H[n - k] = orig_nk
        if improved:
            H = np.fft.fft(h)
        if not improved:
            break
    return h, best_c

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    if time.time() < deadline - 50:
        for a in [20000, 80000, 200000]:
            if time.time() > deadline - 40:
                break
            h_s = slsqp_opt(h, n, alpha=a, maxiter=1200)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:

    if time.time() < deadline - 30:

        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    if time.time() < deadline - 25:
        h_f, c_f = fourier_moves(h, n, deadline=min(deadline - 10, time.time() + 80), max_rounds=25)
        if c_f < best_c:
            h, best_c = h_f, c_f
    if time.time() < deadline - 20:
        h_s = slsqp_opt(h, n, alpha=500000, maxiter=2000)
        c_s = compute_c5(h_s, n)

        c_cd = compute_c5(h_cd, n)
