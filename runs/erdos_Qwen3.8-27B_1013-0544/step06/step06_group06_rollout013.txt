def compute_grad(h, n, top_k=20, alpha=100):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Softmax weights
    shifted = corr - corr.max()
    weights = np.exp(alpha * shifted)
    total = weights.sum()
    if total > 1e-10:
        weights = weights / total
    # Now compute gradient
    grad = np.zeros(n)
    # Only use top-k lags for efficiency
    top_idx = np.argsort(corr)[-top_k:]
    for k in top_idx:
        s = k - (n-1)  # shift
        w = weights[k]
        if s == 0:
            grad += w * (1 - 2*h)
        elif s > 0:
            # (1-h[j+s]) for j from 0 to n-1-s
            grad[:n-s] += w * (1 - h[s:])
            # -h[j-s] for j from s to n-1
            grad[s:] -= w * h[:n-s]
        else:

            # (1-h[j-s]) for j from s to n-1
            grad[s:] += w * (1 - h[:-s])
            # -h[j+s] for j from 0 to s-1
            grad[:s] -= w * h[s:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def compute_effects(h, n):
    effects = np.zeros((n, 2 * n - 1))
    for s in range(1, n):
        k_pos = n - 1 + s
        k_neg = n - 1 - s
        effects[:n - s, k_pos] += (1.0 - h[s:])
        effects[s:, k_pos] -= h[:n - s]
        effects[s:, k_neg] += (1.0 - h[:n - s])
        effects[:n - s, k_neg] -= h[s:]
    return effects

def subgradient_descent(h_start, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h_start, n).copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    alpha = 40.0
    lr = 0.004
    top_k = 25
    momentum = 0.85
    stall = 0
    n_iter = 0

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = corr.max()
        weights = np.exp(alpha * (corr - c_max))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros_like(corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= w_sum

        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            w = weights[k]
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
            else:
                sp = -s
                grad[sp:] += w * (1.0 - h[:n - sp])
                grad[:n - sp] -= w * h[sp:]

        vel = momentum * vel - lr * grad
        h = project_feasible(h + vel, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        n_iter += 1
        alpha = min(600.0, alpha + 1.5)
        lr *= 0.9995

        if stall > 3000:
            h = project_feasible(best_h + 0.03 * rng.randn(n), n)
            vel = np.zeros(n)
            alpha = 40.0
            lr = 0.004
            stall = 0

    return best_h, best_c

def binary_local_search(h_start, n, deadline, seed=42, top_k=12):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    effects = compute_effects(h, n)
    stall = 0
    total_iters = 0

    while time.time() < deadline and total_iters < 2000000:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ez = effects[zeros]
        eo = effects[ones]
        top_idx = np.argsort(corr)[-top_k:]
        ct = corr[top_idx]
        D = np.full((len(zeros), len(ones)), -np.inf)
        for k_idx in range(len(top_idx)):
            D = np.maximum(D, (ct[k_idx] + ez[:, k_idx])[:, None] - eo[None, :, k_idx])

        flat_idx = np.argmin(D)
        ii = flat_idx // len(ones)
        jj = flat_idx % len(ones)
        new_c = D[ii, jj] * dx

        if new_c < cur_c - 1e-12:
            i = zeros[ii]
            j = ones[jj]
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + effects[i] - effects[j]
            effects = compute_effects(h, n)
            cur_c = corr.max() * dx
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                ri = rng.randint(len(zeros))
                rj = rng.randint(len(ones))
                i, j = zeros[ri], ones[rj]
                h[i], h[j] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                effects = compute_effects(h, n)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                stall = 0
        total_iters += 1

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
