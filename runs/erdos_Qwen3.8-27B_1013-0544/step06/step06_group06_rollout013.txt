            if m.any():
                C[m, t] = h[vo[m]]
        D = C[None, :, :] - B[:, None, :] + corr[top_idx][None, None, :]
        new_maxes = D.max(axis=2)
        flat = int(new_maxes.argmin())
        zi = flat // no_
        oi = flat % no_
        best_new = new_maxes[zi, oi]
        if best_new < cur_e - 1e-12:
            z = zeros[zi]
            o = ones[oi]
            h[z] = 1.0
            h[o] = 0.0
            zeros = np.delete(zeros, zi)
            ones = np.delete(ones, oi)
            zeros = np.append(zeros, o)
            ones = np.append(ones, z)
            corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            cur_e = corr.max()
            stall = 0
            if cur_e < best_e:
                best_e = cur_e
                best_h = h.copy()
        else:
            stall += 1
            if stall > 30:
                nk = min(4, nz, no_)
                for _ in range(nk):
                    ri = rng.randint(nz)
                    rj = rng.randint(no_)
                    h[zeros[ri]] = 1.0
                    h[ones[rj]] = 0.0
                    zeros = np.delete(zeros, ri)
                    ones = np.delete(ones, rj)
                    zeros = np.append(zeros, h == 1 and ones[-1] or o)
                    ones = np.append(ones, h == 0 and zeros[-1] or z)
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                cur_e = corr.max()
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
    return best_h, best_e * dx

def continuous_polish(h_bin, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = project_feasible(h_bin + 0.01 * rng.randn(n), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [50000, 200000, 1000000]:
        if time.time() > deadline - 5:
            break
        h_cont = best_h.copy()
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = g - m
            sm = shifted.max()
            if sm > 50:
                return m + 0.01
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        def grad(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            w = np.zeros_like(g)
            m = g.max()
            shifted = g - m
            sm = shifted.max()
            bw = np.exp(shifted - sm)
            bw /= bw.sum()
            dg = 2 * x - 1
            grad_full = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                shift = k - (n - 1)
                ig = np.arange(n) + shift
                valid = (ig >= 0) & (ig < n)
                if valid.any():
                    grad_full[k] = np.sum(dg[ig[valid]] * (1 - x[ig[valid]]) if shift >= 0 else (1 - x[ig[valid]]) * dg[ig[valid]] * 0 + dg[ig[valid]])
            gx = np.zeros(n)
            for i in range(n):
                for k in range(2 * n - 1):
                    shift = k - (n - 1)
                    j = i - shift
                    if 0 <= j < n:
                        gx[i] += bw[k] * (1 - x[j]) if shift < 0 else bw[k] * (-x[j]) if False else 0
            return gx * dx
        try:
            res = minimize(obj, h_cont, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c:
                best_c = c
                best_h = h_try.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: SA on initial n (35% time)
    p1 = t0 + 0.35 * (budget_s - 5)
    if time.time() < p1:
        hb, cb = sa_binary(h0, n0, p1, seed=seed)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n0
    # Phase 2: Greedy descent (15% time)
    p2 = t0 + 0.50 * (budget_s - 5)
    if time.time() < p2:
        hb, cb = greedy_descent(best_h, best_n, p2, top_k=15, seed=seed + 1)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), best_n
    # Phase 3: SA on n=400 with Thue-Morse (20% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3:
        n_alt = 400
        h_tm = thue_morse_init(n_alt, rng)
        hb, cb = sa_binary(h_tm, n_alt, p3, seed=seed + 200, t0_temp=2.5, t1_temp=0.02)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n_alt
    # Phase 4: SA on n=500 (15% time)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        n_alt2 = 500
        h_rand = np.zeros(n_alt2)
        h_rand[rng.choice(n_alt2, n_alt2 // 2, replace=False)] = 1.0
        hb, cb = sa_binary(h_rand, n_alt2, p4, seed=seed + 400, t0_temp=3.0, t1_temp=0.03)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n_alt2
    # Phase 5: Continuous polish (10% time)
    if time.time() < dl - 2:
        hc, cc = continuous_polish(best_h, best_n, dl - 2, seed=seed + 600)
        if cc < best_c:
            best_c, best_h, best_n = cc, hc.copy(), best_n
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
