**Idea: "Worst-pair" search**
1. Compute c = h * (1-h) * dx
2. Find the shift m* with maximum c[m*]
3. For that shift, compute contributions: contrib[i] = h[i] * (1-h[i+k*]) * dx for valid i
4. The sum of contribs is c[m*]
5. To reduce c[m*], we want to reduce the largest contributors
6. For the largest contributor at index i, try:

   - Moving mass from i to somewhere else: reduces contrib[i] but might increase other c[m]
   - Moving mass from i+k* to i: reduces contrib[i]

The key insight is that we need to be careful not to increase other c[m] values while reducing c[m*].

**Idea: "Simulated annealing" style local search**
- Try random pairs of coordinates
- Move mass between them
- Accept if it reduces the max (or with some probability if it increases slightly)
- Gradually decrease the "temperature"

            i, j = rng.randint(n, size=2)

            c = compute_c5_vec(h, n)
            m_star = np.argmax(c)
            k = m_star - (n-1)
            if k > 0:

                contrib = h[:nv] * (1-h[k:])
                i = int(np.argmax(contrib))
                j = i + k

                        h[i] += move
                        h[j] -= move
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return best_h, best_c

def active_shift_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    while time.time() < deadline - 2:
        c = compute_c5_vec(h, n)
        m = c.max()
        top_shifts = np.argsort(c)[-8:]
        improved = False
        for m_idx in top_shifts:
            k = m_idx - (n - 1)
            if k > 0:
                nv = n - k
                contrib = h[0:nv] * (1.0 - h[k:n])
                top_i = np.argsort(contrib)[-20:]
                for i in top_i:
                    if time.time() > deadline - 2:
                        return best_h, best_c
                    if contrib[i] < 1e-10:
                        continue
                    for delta in [0.003, 0.008, 0.015, 0.03]:
                        if h[i] >= delta and h[i + k] + delta <= 1.0:
                            h[i] -= delta
                            h[i + k] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] += delta
                                h[i + k] -= delta
            elif k < 0:
                kk = -k
                nv = n - kk
                contrib = h[kk:n] * (1.0 - h[0:nv])
                top_i = np.argsort(contrib)[-20:]
                for i in top_i:
                    if time.time() > deadline - 2:
                        return best_h, best_c
                    if contrib[i] < 1e-10:
                        continue
                    for delta in [0.003, 0.008, 0.015, 0.03]:
                        if h[i + kk] >= delta and h[i] + delta <= 1.0:
                            h[i + kk] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i + kk] += delta
                                h[i] -= delta
        if not improved:
            break
    return best_h, best_c

def random_pair_polish(h, n, deadline, rng, n_scan=100):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.001, 0.002, 0.004, 0.008, 0.016, 0.032]
    while time.time() < deadline - 2:
        improved = False
        for d in deltas:
            for _ in range(n_scan):
                if time.time() > deadline - 2:
                    return best_h, best_c
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                sign = 1 if rng.random() < 0.5 else -1
                move = sign * d
                if h[i] + move < 0 or h[i] + move > 1 or \
                   h[j] - move < 0 or h[j] - move > 1:
                    continue
                h[i] += move
                h[j] -= move
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= move
                    h[j] += move
        if not improved:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.0005, 0.001, 0.002, 0.004, 0.001, 0.0005]
    for d in deltas:
        while time.time() < deadline - 2:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_n = n

    # Phase 1: L-BFGS-B (~250s)
    t1 = min(t0 + 250, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, cur_n, t1, max_iters=300)

    # Phase 2: Gradient pair search (~200s)
    if time.time() < deadline - 350:
        t2 = min(time.time() + 200, deadline - 350)
        h2, c2 = gradient_pair_search(best_h, cur_n, t2, step_size=0.004, top_k=10)
        if c2 < best_c:
            best_c = c2; best_h = h2

    # Phase 3: Active shift search (~200s)
    if time.time() < deadline - 250:
        t3 = min(time.time() + 200, deadline - 250)
        h3, c3 = active_shift_search(best_h, cur_n, t3)
        if c3 < best_c:
            best_c = c3; best_h = h3

    # Phase 4: L-BFGS refinement with high alpha
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        h4, c4 = lbfgs_phase(best_h, cur_n, t4,
                             alphas=[500000, 2000000, 5000000], max_iters=150)
        if c4 < best_c:
            best_c = c4; best_h = h4

    # Phase 5: Fine pair search (~150s)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 150, deadline - 80)
        h5, c5 = fine_pair_search(best_h, cur_n, t5, radius=60)
        if c5 < best_c:
            best_c = c5; best_h = h5

    # Phase 6: Random pair polish (~80s)
    if time.time() < deadline - 30:
        t6 = min(time.time() + 80, deadline - 30)
        h6, c6 = random_pair_polish(best_h, cur_n, t6, rng, n_scan=50)
        if c6 < best_c:
            best_c = c6; best_h = h6

    # Phase 7: Try alternative n values
    if time.time() < deadline - 60:
        for n_try in [250, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, cur_n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = max(30, deadline - time.time() - 30)
            ht, ct = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5,
                                 alphas=[10000, 100000, 500000], max_iters=150)
            if time.time() < deadline - 20:
                ht, ct = fine_pair_search(ht, n_try, deadline - 20, radius=50)
            if ct < best_c:
                best_c = ct; best_h = ht; cur_n = n_try

    # Phase 8: Final polish
    if time.time() < deadline - 8:
        best_h, best_c = gradient_pair_search(best_h, cur_n, deadline - 8,
                                              step_size=0.002, top_k=6)
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_search(best_h, cur_n, deadline - 2, radius=40)

    best_h = project_feasible(best_h, cur_n)
    best_c = compute_c5(best_h, cur_n)
    return best_h, best_c, cur_n
