def local_search_fast(h, n, deadline, max_sweeps=50000):
    """Fast vectorized local search for binary h."""
    dx = 2.0 / n
    c = correlate_full(h, n)
    M = c.max()
    best_c = M * dx
    best_h = h.copy()
    
    L = np.arange(-(n-1), n)
    all_idx = np.arange(n)
    
    A = compute_A_matrix(h, n)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    sweep = 0
    while sweep < max_sweeps and time.time() < deadline:
        sweep += 1
        found = False
        # Shuffle zeros for diversity
        rng_order = np.random.permutation(len(zeros)) if sweep > 1 else np.arange(len(zeros))
        for zi in rng_order:
            if time.time() > deadline:
                return best_h, best_c
            i = zeros[zi]
            if h[i] != 0:
                continue
            B_i = A[i, :]
            D_i = c - B_i
            # For each j in ones, compute max(D_i + A[j, :])
            if len(ones) == 0:
                break
            candidates = A[ones, :] + D_i[None, :]
            new_maxes = candidates.max(axis=1)
            best_idx = new_maxes.argmin()
            best_max = new_maxes[best_idx]
            if best_max < M - 1e-10:
                j = ones[best_idx]
                # Perform swap
                h[i] = 1.0
                h[j] = 0.0
                c = correlate_full(h, n)
                M = c.max()
                A = compute_A_matrix(h, n)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                found = True
                if M * dx < best_c:
                    best_c = M * dx
                    best_h = h.copy()
                break  # Restart sweep with updated state
    return best_h, best_c

def simulated_annealing(h, n, deadline, t_start=10.0, t_end=0.1, cooldown=0.999):
    """Simulated annealing with swap moves."""
    dx = 2.0 / n
    rng = np.random.RandomState(42)
    c = correlate_full(h, n)
    M = c.max()
    best_c = M * dx
    best_h = h.copy()
    T = t_start
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and T > t_end:
        # Pick random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        if h[i] != 0 or h[j] != 1:
            continue
        
        # Compute new correlation
        h[i] = 1.0
        h[j] = 0.0
        c_new = correlate_full(h, n)
        M_new = c_new.max()
        
        delta = M_new - M
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            c = c_new
            M = M_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if M * dx < best_c:
                best_c = M * dx
                best_h = h.copy()
        else:
            # Reject
            h[i] = 0.0
            h[j] = 1.0
        
        T *= cooldown
    
    return best_h, best_c

def slsqp_polish(h, n, maxiter=2000, deadline=None):
    """SLSQP optimization for continuous values."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 1000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        k_max = g.argmax()
        # Gradient of c[k_max] = sum_i x[i](1-x[i+L])
        # dc/dx[j] = (1-x[j+L]) - x[j-L] (for valid indices)
        L = k_max - (n - 1)
        grad = np.zeros(n)
        # x[j] as first index: (1-x[j+L]) if valid
        jp = np.arange(n) + L
        vm = (jp >= 0) & (jp < n)
        grad[vm] += (1 - x[jp[vm]])
        # x[j] as second index: -x[j-L] if valid
        jm = np.arange(n) - L
        vp = (jm >= 0) & (jm < n)
        grad[vp] -= x[jm[vp]]
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0, 1)
        # Project to feasible
        target = n / 2.0
        for _ in range(1000):
            diff = target - h_new.sum()
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                room = 1.0 - h_new
                s = room.sum()
                if s < 1e-12:
                    break
                h_new = np.clip(h_new + diff/s * room, 0, 1)
            else:
                s = h_new.sum()
                if s < 1e-12:
                    break
                h_new = np.clip(h_new + diff/s * h_new, 0, 1)
        return h_new
    except Exception:
        return h

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin
