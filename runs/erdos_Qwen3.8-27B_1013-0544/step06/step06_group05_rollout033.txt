def compute_delta(cur, i, j, n):
    """Compute delta correlation vector when swapping i (0→1) and j (1→0)."""
    delta = np.zeros(2*n-1)
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    # General formula (no overlap): delta[l] = h[j+l] - h[i+l] + h[j-l] - h[i-l]
    # Use h=0 for out-of-range
    
    # h[i+l] for all lags
    idx_ip = i + lags  # i + l for each lag l
    idx_im = i - lags  # i - l for each lag
    idx_jp = j + lags
    idx_jm = j - lags
    
    h_ip = np.where((idx_ip >= 0) & (idx_ip < n), cur[idx_ip[(idx_ip >= 0) & (idx_ip < n)]], 0.0)
    h_im = np.where((idx_im >= 0) & (idx_im < n), cur[idx_im[(idx_im >= 0) & (idx_im < n)]], 0.0)
    h_jp = np.where((idx_jp >= 0) & (idx_jp < n), cur[idx_jp[(idx_jp >= 0) & (idx_jp < n)]], 0.0)
    h_jm = np.where((idx_jm >= 0) & (idx_jm < n), cur[idx_jm[(idx_jm >= 0) & (idx_jm < n)]], 0.0)
    
    delta = h_jp - h_ip + h_jm - h_im
    
    # Fix overlap: if i+l = j (l = j-i), the term h[i+l] should be h_new[j]=0 instead of h_old[j]=1
    l_overlap = j - i
    if -(n-1) <= l_overlap <= n-1:
        k = l_overlap + (n-1)  # index in delta array
        # In this case, h_ip[k] = cur[j] = 1, but should be 0 (h_new[j])
        # Also h_jm[k] = cur[i] = 0, but should be 1 (h_new[i])
        # So delta[k] should be increased by 1 (from -h_ip) and increased by 1 (from +h_jm)
        # Wait: delta = h_jp - h_ip + h_jm - h_im
        # h_ip[k] = cur[j] = 1 (should be h_new[j] = 0) → delta should be +1 more
        # h_jm[k] = cur[i] = 0 (should be h_new[i] = 1) → delta should be +1 more
        # Total correction: +2
        # But from our case 1 analysis: delta = 1 - h[2i-j] + h[2j-i]
        # General formula gives: h[j+l] - h[i+l] + h[j-l] - h[i-l] = h[2j-i] - h[j] + h[i] - h[2i-j]
        #   = h[2j-i] - 1 + 0 - h[2i-j] = h[2j-i] - 1 - h[2i-j]
        # Correct value: 1 - h[2i-j] + h[2j-i]
        # Difference: (1 - h[2i-j] + h[2j-i]) - (h[2j-i] - 1 - h[2i-j]) = 2
        # So correction is +2. ✓
        delta[k] += 2

                continue

            # A_i[k] = h[i+lag] + h[i-lag] (with 0 for out of range)

                l = lags_c[idx_c]

            # Need: for all critical k, corr[k] + delta[k] < cur_max
            # delta[k] = (h[j+lag] + h[j-lag]) - A[k] (approximately, ignoring overlap corrections)

            # Vectorized check over all j:

            for idx_c in range(len(critical)):
                l = lags_c[idx_c]
                idx_jp = ones + l
                idx_jm = ones - l
                B[:, idx_c] = np.where((idx_jp >= 0) & (idx_jp < n), cur[np.clip(idx_jp, 0, n-1)], 0.0) + \
                              np.where((idx_jm >= 0) & (idx_jm < n), cur[np.clip(idx_jm, 0, n-1)], 0.0)
            
            # Feasibility: corr[critical][None, :] + B - A[None, :] < cur_max

            rhs = cur_max - corr[critical][None, :] + A[None, :]

            if rng.random() < 0.15:
                for _ in range(rng.randint(2, 5)):
                    zi = np.where(cur == 0)[0]
                    oi = np.where(cur == 1)[0]
                    if len(zi) > 0 and len(oi) > 0:
                        a = zi[rng.randint(len(zi))]
                        b = oi[rng.randint(len(oi))]
                        cur[a], cur[b] = cur[b].copy(), cur[a].copy()
                corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
                cur_max = corr.max()
            else:
                break

    return best, float(best_max * dx)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def projected_gradient(h, n, deadline, alpha=50000):
    dx = 2.0 / n
    cur = h.copy()
    best = cur.copy()
    best_c = compute_c5(cur, n)
    eta = 0.005
    stall = 0
    j_arr = np.arange(n)
    lags = np.arange(2 * n - 1) - (n - 1)

    while time.time() < deadline:
        g = np.correlate(cur, 1.0 - cur, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()

        f_plus = np.zeros(n)
        f_minus = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            lag = lags[k]
            src_p = j_arr + lag
            mp = (src_p >= 0) & (src_p < n)
            if mp.any():
                f_plus[mp] += w[k] * cur[src_p[mp]]
            src_m = j_arr - lag
            mm = (src_m >= 0) & (src_m < n)
            if mm.any():
                f_minus[mm] += w[k] * cur[src_m[mm]]

        grad = dx * (1.0 - f_plus - f_minus)

        cur_new = cur - eta * grad
        cur_new = project_feasible(cur_new, n)
        c_new = compute_c5(cur_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best = cur_new.copy()
            cur = cur_new
            stall = 0
            eta = min(eta * 1.05, 0.1)
        else:
            stall += 1
            eta *= 0.7
            if stall > 100:
                break

    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search (45% of budget)
    p1 = t0 + 0.45 * (budget_s - 10)
    hb, cb = fast_local_search(best_h, n, p1, max_sweeps=200000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Multi-start local search from perturbations (20%)
    p2 = t0 + 0.65 * (budget_s - 10)
    for restart in range(8):
        if time.time() > p2:
            break
        h_pert = best_h.copy()
        n_swaps = rng.randint(2, 8)
        for _ in range(n_swaps):
            zi = np.where(h_pert == 0)[0]
            oi = np.where(h_pert == 1)[0]
            if len(zi) > 0 and len(oi) > 0:
                a = zi[rng.randint(len(zi))]
                b = oi[rng.randint(len(oi))]
                h_pert[a], h_pert[b] = h_pert[b].copy(), h_pert[a].copy()
        remaining = max(1.0, p2 - time.time()) / (8 - restart)
        hb2, cb2 = fast_local_search(h_pert, n, time.time() + remaining,
                                     max_sweeps=30000, seed=seed + 1000 + restart)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 3: Projected gradient (15%)
    p3 = t0 + 0.80 * (budget_s - 10)
    if time.time() < p3:
        h_pg = best_h + rng.normal(0, 0.02, n)
        h_pg = project_feasible(h_pg, n)
        hp, cp = projected_gradient(h_pg, n, p3, alpha=50000)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: SLSQP polishing (remaining time)
    alphas = [10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > dl - 5:
            break
        # From current best with small perturbation
        h_try = best_h + rng.normal(0, 0.005, n)
        h_try = project_feasible(h_try, n)
        h_res = slsqp_optimize(h_try, n, alpha, 3000)
        c_res = compute_c5(h_res, n)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()

        # From binary best directly
        if time.time() < dl - 5:
            h_res2 = slsqp_optimize(best_h.copy(), n, alpha, 3000)
            c_res2 = compute_c5(h_res2, n)
            if c_res2 < best_c:
                best_c = c_res2
                best_h = h_res2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
