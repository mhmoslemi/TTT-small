def build_sensitivity(h, n):
    n_corr = 2 * n - 1
    p_idx = np.arange(n)
    k_idx = np.arange(n_corr)
    # S[p,k] = (1-h[p+k-(n-1)]) if 0<=p+k-(n-1)<n else 0
    #         - h[p-(k-(n-1))] if 0<=p-(k-(n-1))<n else 0
    # First term: index = p+k-(n-1), valid if 0<=index<n
    pk = p_idx[:,None] + k_idx[None,:] - (n-1)  # (n, n_corr)
    valid1 = (pk >= 0) & (pk < n)
    term1 = np.where(valid1, 1.0 - h[pk[valid1]], 0.0)
    # Second term: index = p-(k-(n-1)) = p-k+n-1
    pk2 = p_idx[:,None] - k_idx[None,:] + (n-1)
    valid2 = (pk2 >= 0) & (pk2 < n)
    term2 = np.where(valid2, h[pk2[valid2]], 0.0)
    return term1 - term2

Actually, for the continuous case, I can use a projected gradient method:
1. Compute gradient of the log-sum-exp objective
2. Take a gradient step
3. Project onto {x ∈ [0,1]^n : sum(x) = n/2}

The projection onto the intersection of the box and the hyperplane can be done with bisection.

The gradient of the log-sum-exp objective:
- Let f(h) = max_k c_k + (1/alpha) log(sum exp(alpha * (c_k - max_j c_j)))
  where c_k = dx * sum_p h[p](1-h[p+k-(n-1)])
  
- dc_k/dh[j] = dx * [(1-h[j+k-(n-1)]) * I(valid) - h[j-(k-(n-1))] * I(valid)]
  (same as the sensitivity vector, times dx)

- Let m = max_k c_k, s = sum_k exp(alpha*(c_k - m))
- df/dh[j] = (1/alpha) * sum_k [alpha * exp(alpha*(c_k-m)) / s] * dc_k/dh[j]
             * (for k where c_k = m, there's also the derivative of max, but in log-sum-exp this is handled)
  
  Actually, the gradient of log-sum-exp is: softmax(alpha*(c-m)) * alpha
  So df/dh[j] = sum_k softmax_k * dc_k/dh[j]
  where softmax_k = exp(alpha*(c_k-m)) / sum_l exp(alpha*(c_l-m))

This is computable! Let me implement it.

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

        S = build_sensitivity(h, n)

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):

    if num_ones > target:

    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

    if time.time() < p4:
        h_tm = thue_morse_init(n, rng)
        hb, cb = binary_sa(h_tm, n, p4, top_k=25, seed=seed + 900)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: SLSQP polish (7% time)
    p5 = t0 + 0.92 * (budget_s - 5)
    if time.time() < p5:
        hp = project_feasible(best_h + 0.01 * rng.randn(n), n)
        for a in [100000, 500000]:
            if time.time() > p5 - 3:
                break
            hp = slsqp_smooth(hp, n, a, 800, p5 - 2)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 6: Final binary polish (3% time)
    if time.time() < dl - 2:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, n)
        hb, cb = binary_sa(h_bin2, n, dl - 1, top_k=30, seed=seed + 1200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final continuous micro-polish
    if time.time() < dl - 1:
        hf = project_box_simplex(best_h + 0.005 * rng.randn(n), n)
        hf, cf = projected_gradient(hf, n, 200000, dl - 0.5)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
