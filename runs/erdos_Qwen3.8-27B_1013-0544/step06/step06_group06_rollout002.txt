    # bisection
    for _ in range(80):
       mid=0.5*(lo+hi)
       s=np.clip(y-mid,0.0,1.0).sum()
       if s > target: lo=mid
       else: hi=mid
    x=np.clip(y-0.5*(lo+hi),0.0,1.0)

          if s<=1e-14: break
          x = x + (diff/s)*x
    return np.clip(x,0.0,1.0)

import numpy as np
import time
from scipy.optimize import minimize

G_N = 0
G_ALPHA = 1.0
TARGET_C5 = 0.3808


def explicit_c(h, n):
    if n <= 0:
        return np.zeros(0, dtype=float)
    m = 2 * n - 1
    b = n - 1
    c = np.empty(m, dtype=float)
    c[b] = float(np.dot(h, 1.0 - h))
    for d in range(1, n):
        L = n - d
        c[b + d] = float(np.dot(h[:L], 1.0 - h[d:]))
        c[b - d] = float(np.dot(h[d:], 1.0 - h[:L]))
    return c


def c5_grad_weights(x, n, w):
    g = np.zeros(n, dtype=float)
    if n <= 0:
        return g
    b = n - 1
    if w[b] != 0.0:
        g += w[b] * (1.0 - 2.0 * x)
    for d in range(1, n):
        L = n - d
        wp = w[b + d]
        if wp != 0.0:
            g[:L] += wp * (1.0 - x[d:])
            g[d:] -= wp * x[:L]
        wm = w[b - d]
        if wm != 0.0:
            g[d:] += wm * (1.0 - x[:L])
            g[:L] -= wm * x[d:]
    return g


def smooth_count_only(x, n, alpha):
    c = explicit_c(x, n)
    m = float(c.max())
    y = float(alpha) * (c - m)
    s = float(np.exp(y).sum())
    if s <= 0.0:
        return m
    return float(m + np.log(s) / float(alpha))


def smooth_grad_only(x, n, alpha):
    c = explicit_c(x, n)
    m = float(c.max())
    y = float(alpha) * (c - m)
    p = np.exp(y)
    s = float(p.sum())
    if s <= 0.0:
        return np.zeros_like(x, dtype=float)
    p /= s
    return c5_grad_weights(x, n, p)


def g_smooth_obj(x):
    return smooth_count_only(x, G_N, G_ALPHA)


def g_smooth_jac(x):
    return smooth_grad_only(x, G_N, G_ALPHA)

                continue
            topr = min(LOCAL_R, int(flatb.size))
            if topr == flatb.size:
                topb = np.arange(flatb.size)
            else:
                topb = np.argpartition(flatb, topr - 1)[:topr]
            for cf in topb:
                rel_ip = int(cf // lo)
                iq = int(cf % lo)
                ip = bstart + rel_ip
                i = ones[iq]
                j = zeros[ip]
                key = (i, j)
                if key in tabu:
                    continue
                newcorr = pair_new_corr(corr, h, n, i, j)
                nm = float(newcorr.max())
                if nm < best_nm:
                    best_nm = nm
                    best_newcorr = newcorr
                    best_pair = (i, j)

        if best_pair is None:
            for _ in range(5):
                i = ones[int(rng.randint(lo))]
                j = zeros[int(rng.randint(lz))]
                if (i, j) not in tabu:
                    break
            newcorr = pair_new_corr(corr, h, n, i, j)
            best_nm = float(newcorr.max())
            best_newcorr = newcorr
            best_pair = (i, j)

        i, j = best_pair
        nm = best_nm
        accept = nm < M - 1e-9
        if not accept and abs(nm - M) <= 1e-9:
            if stall > 20:
                accept = rng.random() < 0.15
        if not accept and nm > M:
            accept = rng.random() < float(np.exp(-(nm - M) / max(temp, 0.2)))

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr = best_newcorr
            M = nm

            zp = zero_pos[j]
            op = one_pos[i]

            lastz = zeros[-1]
            zeros[zp] = lastz
            zero_pos[lastz] = zp
            zeros.pop()
            zero_pos[j] = -1

            lasto = ones[-1]
            ones[op] = lasto
            one_pos[lasto] = op
            ones.pop()
            one_pos[i] = -1

            zeros.append(j)
            zero_pos[j] = len(zeros)
            ones.append(i)
            one_pos[i] = len(ones)

            tabu.clear()
            temp = max(0.05, temp * 0.999)
            if M < best_m - 1e-10:
                best_m = M
                best_h = h.copy()
                best_c = M * 2.0 / float(n)
                no_improve = 0
            else:
                no_improve += 1
            stall = 0
        else:
            tabu.add((i, j))
            if len(tabu) > 800:
                tabu.clear()
            stall += 1
            no_improve += 1
            temp = max(0.05, temp * 0.995)

            if stall > 80:
                kmax = min(4, len(zeros), len(ones))
                if kmax >= 2:
                    k = int(rng.randint(2, kmax + 1))
                    iz = rng.choice(len(zeros), k, replace=False)
                    io = rng.choice(len(ones), k, replace=False)
                    zp = np.asarray(zeros, dtype=int)[iz]
                    op = np.asarray(ones, dtype=int)[io]
                    h2 = h.copy()
                    h2[zp] = 0.0
                    h2[op] = 1.0
                    corr2 = explicit_c(h2, n)
                    nm2 = float(corr2.max())
                    if nm2 < M + max(1.0, 3.0 * temp):
                        h = h2
                        corr = corr2
                        M = nm2

                        zeros = [a for a, v in enumerate(h) if v < 0.5]
                        ones = [a for a, v in enumerate(h) if v >= 0.5]
                        zero_pos = [-1] * n
                        one_pos = [-1] * n
                        for a, p in enumerate(zeros):
                            zero_pos[p] = a
                        for a, p in enumerate(ones):
                            one_pos[p] = a

                        tabu.clear()
                        temp = max(0.05, temp * 0.7)
                        stall = 0
                        if M < best_m - 1e-10:
                            best_m = M
                            best_h = h.copy()
                            best_c = M * 2.0 / float(n)
                            no_improve = 0
                        else:
                            no_improve += 1

    return best_h, best_c


def slsqp_smooth(h, n, alpha, maxiter):
    global G_N, G_ALPHA
    h = project_feasible(h, n)
    G_N = int(n)
    G_ALPHA = float(alpha)
    maxiter = max(1, int(maxiter))
    bounds = [(0.0, 1.0)] * int(n)
    try:
        res = minimize(
            fun=g_smooth_obj,
            x0=h,
            jac=g_smooth_jac,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": g_sum_obj, "jac": g_sum_jac}],
            method="SLSQP",
            options={"maxiter": maxiter, "ftol": 1e-13, "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
        nit = int(getattr(res, "nit", maxiter))
        if not np.all(np.isfinite(x)):
            return h, maxiter
        return x, max(1, nit)
    except Exception:
        return h, maxiter


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    span = max(1.0, float(budget_s) - 12.0)
    deadline = t0 + span
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size == 0:
            raise ValueError
        h0 = h0.copy()
        n = int(h0.size)
    except Exception:
        n = 250
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = project_feasible(h0, n)
    if not np.all(np.isfinite(best_h)):
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5_np(best_h, n)

    if n > 1 and n % 2 == 0:
        p1 = t0 + 0.30 * span
        if p1 > time.time() + 5.0:
            hb = make_binary(best_h, n, 0, rng)
            hb, _ = binary_local_search(hb, n, p1, int(seed) + 1)
            hb = project_feasible(hb, n)
            if np.all(np.isfinite(hb)):
                cnp = compute_c5_np(hb, n)
                if cnp < best_c:
                    best_h = hb.copy()
                    best_c = cnp

        p2 = t0 + 0.48 * span
        if time.time() < p2 - 10.0:
            hb2 = make_binary(best_h, n, 1, rng)
            hb2, _ = binary_local_search(hb2, n, p2, int(seed) + 2)
            hb2 = project_feasible(hb2, n)
            if np.all(np.isfinite(hb2)):
                cnp = compute_c5_np(hb2, n)
                if cnp < best_c:
                    best_h = hb2.copy()
                    best_c = cnp

        p2b = t0 + 0.58 * span
        if time.time() < p2b - 10.0:
            hb3 = make_binary(best_h, n, 2, rng)
            hb3, _ = binary_local_search(hb3, n, p2b, int(seed) + 3)
            hb3 = project_feasible(hb3, n)
            if np.all(np.isfinite(hb3)):
                cnp = compute_c5_np(hb3, n)
                if cnp < best_c:
                    best_h = hb3.copy()
                    best_c = cnp

    p_sls = t0 + 0.88 * span
    G_N = int(n)
    last_dt = 0.0
    schedules = [
        (20.0, 0.0),
        (50.0, 0.02),
        (100.0, 0.05),
        (200.0, 0.01),
        (500.0, 0.0),
        (800.0, 0.004),
    ]

    for idx, (alpha, noise) in enumerate(schedules):
        stop = p_sls - 15.0
        if time.time() >= stop:
            break
        remaining = max(0.0, stop - time.time())
        if remaining < 30.0:
            break

        base_iter = 50 + 20 * idx
        if last_dt > 0.0:
            allow = int(0.70 * remaining / max(last_dt, 1e-4)) + 10
            maxiter = max(30, min(base_iter * 3, max(30, allow)))
        else:
            maxiter = base_iter

        if noise > 0.0:
            hs = project_feasible(best_h + float(noise) * rng.randn(n), n)
        else:
            hs = best_h.copy()

        t1 = time.time()
        hx, nit = slsqp_smooth(hs, n, float(alpha), int(maxiter))
        elapsed = max(1e-6, time.time() - t1)
        nit = max(1, int(nit))
        dt = elapsed / float(nit)
        if last_dt <= 0.0:
            last_dt = dt
        else:
            last_dt = 0.5 * last_dt + 0.5 * dt

        hx = project_feasible(hx, n)
        if np.all(np.isfinite(hx)):
            cnp = compute_c5_np(hx, n)
            if cnp < best_c:
                best_h = hx.copy()
                best_c = cnp

    p_fb = t0 + 0.94 * span
    if n > 1 and n % 2 == 0 and time.time() < p_fb - 5.0:
        hfb = make_binary(best_h, n, 0, rng)
        hfb, _ = binary_local_search(hfb, n, p_fb, int(seed) + 11)
        hfb = project_feasible(hfb, n)
        if np.all(np.isfinite(hfb)):
            cnp = compute_c5_np(hfb, n)
            if cnp < best_c:
                best_h = hfb.copy()
                best_c = cnp

    if time.time() < deadline - 10.0:
        hfin = project_feasible(best_h + 0.005 * rng.randn(n), n)
        hfin, _ = slsqp_smooth(hfin, n, 600.0, 80)
        hfin = project_feasible(hfin, n)
        if np.all(np.isfinite(hfin)):
            cnp = compute_c5_np(hfin, n)
            if cnp < best_c:
                best_h = hfin.copy()
                best_c = cnp

    best_h = project_feasible(best_h, n)
    if not np.all(np.isfinite(best_h)):
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5_np(best_h, n)
    return best_h, float(best_c), int(n)
