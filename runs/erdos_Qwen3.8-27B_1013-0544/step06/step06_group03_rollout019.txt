```python
                    a_idx = np.arange(n - m)
                    b_idx = a_idx + m
                else:
                    b_idx = np.arange(n - m)
                    a_idx = b_idx - m

                b = h[b_idx]
                diff = a - b
                worst = np.abs(diff).argmax()

                    src = a_idx[worst]
                    dst = b_idx[worst]
                else:
                    src = b_idx[worst]

                if delta < 1e-10:

                h[src] -= delta
                h[dst] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[src] += delta
                    h[dst] -= delta
                    h = project_feasible(h, n)
    
    return h, best_c
```

Wait, I realize the local search is still going to be slow because of the project_feasible call after each move. Let me optimize:

Instead of projecting after each move, I can:
1. Make the move
2. Check if it's still feasible (sum constraint)
3. If not feasible, adjust
4. Evaluate C₅
5. If improved, keep; else revert

The sum constraint: if I move δ from src to dst, the sum doesn't change! So I don't need to project for the sum. I just need to check bounds (0 ≤ h ≤ 1), which I already ensure by choosing delta appropriately.

So I can skip the project_feasible call in the inner loop! This will be much faster.

Let me also think about a **multi-pair move**: instead of moving mass between one pair, try moving between multiple pairs simultaneously. This can escape local minima that single-pair moves can't.

For the LP relaxation at point h₀:
minimize t
subject to:
  sum(h) = n/2
  0 ≤ h ≤ 1
  f_k(h₀) + ∇f_k(h₀)·(h - h₀) ≤ t for all k

This is a valid cutting plane (linear underestimator... wait, is it?)

For a quadratic f_k(h) = a + b·h + h^T Q h, the first-order Taylor approximation at h₀ is:
f_k(h₀) + ∇f_k(h₀)·(h - h₀)

This is an underestimate iff f_k is convex (Q ⪰ 0). But we showed Q is indefinite, so this is NOT a valid underestimate. It could be an overestimate for some directions.

So the LP cutting plane approach is not guaranteed to work. However, it can still be useful as a heuristic - the LP solution might happen to be good.

Actually, wait. Let me reconsider. The previous code uses:
```
c[k] + dx * (grad @ (h_var - h_curr)) <= t_var
```

This is a linearization of the quadratic at h_curr. Since the quadratic is not convex, this linearization is not a valid lower bound. But it's still a valid heuristic - the LP solution gives a point that should be at least as good as the current point in the linearized sense.

Actually, for a non-convex function, the linearization at h₀ satisfies:
f(h) ≥ f(h₀) + ∇f(h₀)·(h - h₀) if f is convex
f(h) ≤ f(h₀) + ∇f(h₀)·(h - h₀) if f is concave

Since our functions are neither convex nor concave, the linearization could go either way. But in practice, the LP solution often still improves the objective, so it's a useful heuristic.

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_slsqp(h, n, deadline, top_m=8, alphas=(20000, 100000)):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for _ in range(4):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(active_smooth_obj, h, args=(n, top_idx, alpha),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def lp_cutting_plane(h, n, deadline, max_iters=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    trust = 0.2
    for _ in range(max_iters):
        if time.time() > deadline - 3:
            break
        dx = 2.0 / n
        c = compute_all_lags(h, n)
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1,
                cp.abs(hv - h) <= trust]
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            grad = np.zeros(n)
            if lag == 0:
                grad[:] = 1.0 - 2.0 * h
            elif 0 < lag < n:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] += (-h[:n - lag])
            elif -n < lag < 0:
                m = -lag
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] += (-h[m:])
            cons.append(c[k] + dx * (grad @ (hv - h)) <= tv)
        prob = cp.Problem(cp.Minimize(tv), cons)
        solved = False
        for solver in [cp.GLOP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or hv.value is None:
            break
        h_new = project_feasible(hv.value, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            h = h_new
        step = np.max(np.abs(h_new - h))
        if step < 1e-9:
            break
        trust = max(0.01, trust * 0.85)
    return h, best_c

def fast_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:
        if time.time() > deadline:
            break
        stall = 0
        while stall < 80 and time.time() < deadline:
            c = compute_all_lags(h, n)
            k_star = c.argmax()
            lag = k_star - (n - 1)
            if lag == 0:
                mid_mask = (h > 0.02) & (h < 0.98)
                if not mid_mask.any():
                    break
                central = np.abs(h - 0.5).copy()
                central[~mid_mask] = 1e9
                src = central.argmin()
                if h[src] > 0.5:
                    dst = np.argmin(h)
                    delta = min(step, 1.0 - h[src], 1.0 - h[dst])
                else:
                    dst = np.argmax(h)
                    delta = min(step, h[src], h[dst])
                if delta < 1e-12:
                    break
                h[src] -= delta
                h[dst] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    stall = 0
                else:
                    h[src] += delta
                    h[dst] -= delta
                    stall += 1
            else:
                m = abs(lag)
                if m >= n:
                    break
                if lag > 0:
                    diffs = h[:n - m] - h[m:]
                else:
                    diffs = h[m:] - h[:n - m]
                worst = np.abs(diffs).argmax()
                if abs(diffs[worst]) < 1e-9:
                    break
                if lag > 0:
                    if diffs[worst] > 0:
                        src, dst = worst, worst + m
                    else:
                        src, dst = worst + m, worst
                else:
                    if diffs[worst] > 0:
                        src, dst = worst + m, worst
                    else:
                        src, dst = worst, worst + m
                delta = min(step, abs(diffs[worst]) * 0.4, h[src], 1.0 - h[dst])
                if delta < 1e-12:
                    break
                h[src] -= delta
                h[dst] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    stall = 0
                else:
                    h[src] += delta
                    h[dst] -= delta
                    stall += 1
    return h, best_c

def multi_pair_search(h, n, deadline, n_pairs=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    for _ in range(200):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = c.argmax()
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if m >= n:
            break
        if lag > 0:
            diffs = h[:n - m] - h[m:]
        else:
            diffs = h[m:] - h[:n - m]
        top_worst = np.argsort(np.abs(diffs))[-n_pairs:]
        total_delta = 0.0
        moves = []
        for w in top_worst:
            if abs(diffs[w]) < 1e-8:
                continue
            if lag > 0:
                if diffs[w] > 0:
                    src, dst = w, w + m
                else:
                    src, dst = w + m, w
            else:
                if diffs[w] > 0:
                    src, dst = w + m, w
                else:
                    src, dst = w, w + m
            delta = min(step / n_pairs, abs(diffs[w]) * 0.3, h[src], 1.0 - h[dst])
            if delta > 1e-10:
                moves.append((src, dst, delta))
        if not moves:
            break
        h_old = h.copy()
        for src, dst, delta in moves:
            h[src] -= delta
            h[dst] += delta
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            h = h_old
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # LP cutting plane
    lp_end = time.time() + min(60, (deadline - time.time()) * 0.3)
    h_lp, c_lp = lp_cutting_plane(best_h, n, lp_end)
    if c_lp < best_c:
        best_c = c_lp
        best_h = h_lp
    # SLSQP
    if time.time() < deadline - 20:
        for alpha in [50000, 200000]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Active set
    as_end = time.time() + min(40, (deadline - time.time()) * 0.4)
    h_as, c_as = active_set_slsqp(best_h, n, as_end)
    if c_as < best_c:
        best_c = c_as
        best_h = h_as
    # Multi-pair search
    if time.time() < deadline - 10:
        mp_end = time.time() + min(20, (deadline - time.time()) * 0.3)
        h_mp, c_mp = multi_pair_search(best_h, n, mp_end)
        if c_mp < best_c:

            best_h = h_mp
    # Fast local search
    if time.time() < deadline - 3:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```
