def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

            if tot < 1e-15:

            h += (diff / tot) * h

def thue_morse_init(n, shift=0):
    L = 1 << 12
    tm = np.zeros(L, dtype=np.int8)

        tm[i] = (tm[i >> 1] + (i & 1)) % 2
    idx = (np.arange(n) * L // n + shift) % L
    h = tm[idx].astype(float)

            bk2 = b - k_w
            if 0 <= bk2 < n:
                d -= (1.0 - h[bk2])
            d_zeros[ii] = d
        
        # Find best pairs (top_k combinations)
        best_improvement = -1e-10
        found = False
        top_i = np.argsort(d_ones)[:top_k]
        top_j = np.argsort(d_zeros)[:top_k]
        
        for ii in top_i:
            for jj in top_j:
                a = ones[ii]
                b = zeros[jj]
                h[a] = 0.0
                h[b] = 1.0
                c_new, c_new_vals = compute_c5_full(h, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    c_vals = c_new_vals
                    found = True
                    break
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            if found:
                break
        if not found:
            break
    return h, best_c

def continuous_refine(h, n, deadline, alpha=5000):
    """Gradient-based refinement with small steps."""
    dx = 2.0 / n
    best_c, _ = compute_c5_full(h, n)
    best_h = h.copy()
    lr = 0.02
    
    for _ in range(20000):
        if time.time() > deadline:
            break
        c_vals = np.correlate(h, 1 - h, mode='full') * dx
        m = c_vals.max()
        sh = alpha * (c_vals - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        
        g = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if w[m_idx] < 1e-12:
                continue
            k = m_idx - (n - 1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - h[k:n])
                g[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - h[0:nv])
                g[0:nv] += wm * (-h[kk:n])
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-13:
            break
        g /= g_norm
        
        step = lr
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - step * g, n)
            c_new, _ = compute_c5_full(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            if lr < 1e-12:
                break
    return best_h, best_c

def pair_finetune(h, n, deadline, deltas=None):
    """Fine pair search with gradient guidance."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    dx = 2.0 / n
    best_c, _ = compute_c5_full(h, n)
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c_vals = np.correlate(h, 1 - h, mode='full') * dx
            worst_m = int(np.argmax(c_vals))
            k_w = worst_m - (n - 1)
            
            # Compute per-position gradient contribution at worst shift
            g_worst = np.zeros(n)
            if k_w >= 0:
                nv = n - k_w
                g_worst[0:nv] = (1.0 - h[k_w:n])
                g_worst[k_w:] = (-h[0:nv])
            else:
                kk = -k_w
                nv = n - kk
                g_worst[kk:n] = (1.0 - h[0:nv])
                g_worst[0:nv] = (-h[kk:n])
            
            # Best to increase: most negative g_worst
            # Best to decrease: most positive g_worst
            inc_idx = np.argsort(g_worst)[:80]
            dec_idx = np.argsort(-g_worst)[:80]
            
            for i in inc_idx:
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0:
                    continue
                for j in dec_idx:
                    if i == j or h[j] - d < 0.0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new, _ = compute_c5_full(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved_any = True
                    else:
                        h[i] -= d
                        h[j] += d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_h = thue_morse_init(300, 0)
        best_n = 300
    
    best_h = project_feasible(best_h, best_n)
    best_c, _ = compute_c5_full(best_h, best_n)
    
    # Phase 1: Generate diverse initializations
    inits = [best_h.copy()]
    for s in [0, 37, 73, 128]:
        inits.append(thue_morse_init(best_n, s))
    for f in [3, 5, 7, 11, 15, 20]:
        inits.append(sinusoidal_init(best_n, f))
    for bs in [3, 5, 7, 11, 17]:
        inits.append(block_init(best_n, bs))
    for s in [100, 200, 300]:
        inits.append(random_binary_init(best_n, seed + s))
    
    # Phase 2: Quick screening - short L-BFGS on each
    candidates = []
    screen_time = max(10, min(25, int((deadline - t0 - 400) / len(inits))))
    for i, h_init in enumerate(inits):
        if time.time() > deadline - 350:
            break
        t_end = time.time() + screen_time
        if time.time() >= t_end:
            continue
        h_opt, c_opt = lbfgs_optimize(h_init, best_n, t_end, alpha=5000, mu=0.3, maxiter=100)
        candidates.append((c_opt, h_opt))
    
    candidates.sort(key=lambda x: x[0])
    
    # Phase 3: Deep optimization on top candidates
    top_k_cands = min(4, len(candidates))
    for ci in range(top_k_cands):
        if time.time() > deadline - 80:
            break
        c_cur, h_cur = candidates[ci]
        if c_cur >= best_c:
            continue
        
        # 3a: Binary attack
        t_bin = min(time.time() + 80, deadline - 40)
        if time.time() < t_bin:
            h_cur, c_cur = binary_attack(h_cur, best_n, t_bin)
        
        # 3b: L-BFGS with increasing alpha
        for alpha in [20000, 100000, 500000]:
            if time.time() > deadline - 40:
                break
            t_lb = min(time.time() + 35, deadline - 35)
            if time.time() >= t_lb:
                break
            h_new, c_new = lbfgs_optimize(h_cur, best_n, t_lb, alpha=alpha, mu=0.1, maxiter=200)
            if c_new < c_cur:
                c_cur = c_new
                h_cur = h_new
        
        # 3c: Continuous refinement
        if time.time() < deadline - 30:
            t_cont = min(time.time() + 30, deadline - 25)
            h_cur, c_cur = continuous_refine(h_cur, best_n, t_cont, alpha=3000)
        
        # 3d: Pair finetune
        if time.time() < deadline - 20:
            t_pf = min(time.time() + 30, deadline - 15)
            h_cur, c_cur = pair_finetune(h_cur, best_n, t_pf)
        
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
    
    # Phase 4: Try the original with deeper optimization
    if best_c > 0.38084:
        h_orig = project_feasible(best_h, best_n)
        # Binary attack on original
        if time.time() < deadline - 60:
            t_b2 = min(time.time() + 50, deadline - 50)
            h_b, c_b = binary_attack(h_orig, best_n, t_b2)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b
        
        # Fine pair search on best
        if time.time() < deadline - 15:
            best_h, best_c = pair_finetune(best_h, best_n, min(time.time() + 30, deadline - 10),
                                           deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    # Phase 5: Try different n values
    if time.time() < deadline - 60:
        for n_try in [200, 250, 350]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(35, deadline - time.time() - 20)
            if alloc < 15:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5, alpha=20000, mu=0.5, maxiter=150)
            if time.time() < deadline - 15:
                h_t, c_t = pair_finetune(h_t, n_try, min(time.time() + 15, deadline - 10))
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
    
    # Phase 6: Final squeeze
    n_final = len(best_h)
    if time.time() < deadline - 5:
        best_h, best_c = pair_finetune(best_h, n_final, deadline - 2,
                                       deltas=[0.001, 0.0005, 0.0002, 0.0001])
    
    best_h = project_feasible(best_h, len(best_h))
    final_c, _ = compute_c5_full(best_h, len(best_h))
    return best_h, final_c, len(best_h)
