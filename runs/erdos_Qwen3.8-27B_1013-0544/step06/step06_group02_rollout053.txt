import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _lbfgs_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = max(0.01, alpha * 0.005)
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def solve_1d_convex(c_cur, a, t_old):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        if hi - lo < 1e-14:
            break
        mid = (lo + hi) * 0.5
        vals = c_cur + a * (mid - t_old)
        max_val = vals.max()
        active = vals >= (max_val - 1e-12)
        if not active.any():
            active[int(np.argmax(vals))] = True
        a_min = a[active].min()
        a_max = a[active].max()
        if a_min >= -1e-14:
            hi = mid
        elif a_max <= 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, deadline, max_sweeps=500):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    m_total = 2 * n - 1
    stale = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline - 2:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline - 2:
                return h, best_c
            c_cur = compute_c5_vec(h, n)
            t_old = h[i]
            a = np.zeros(m_total)
            for k in range(-(n - 1), n):
                m_idx = k + n - 1
                hp = h[i + k] if 0 <= i + k < n else 0.0
                hm = h[i - k] if 0 <= i - k < n else 0.0
                a[m_idx] = dx * (1.0 - hp - hm)
            t_new = solve_1d_convex(c_cur, a, t_old)
            if abs(t_new - t_old) > 1e-13:
                h[i] = t_new
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    improved_any = True
                    stale = 0
                else:
                    stale += 1
        if not improved_any:
            stale += 1
            if stale > 5:
                break
    return h, best_c

def active_lag_descend(h, n, deadline, window=5):
    best_c = compute_c5(h, n)
    recent = []
    lr = 0.2
    stagnant = 0
    while time.time() < deadline - 2:
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c))
        recent.append(k_star)
        if len(recent) > window:
            recent.pop(0)
        grad = np.zeros(n)
        inv_w = 1.0 / len(recent)
        for k_idx in recent:
            k = k_idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * inv_w
                grad[k:] += (-h[:nv]) * inv_w
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * inv_w
                grad[:nv] += (-h[kk:]) * inv_w
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            h = h_new
            stagnant = 0
            lr = min(lr * 1.03, 1.0)
        else:
            lr *= 0.4
            stagnant += 1
            if stagnant > 100:
                break
    return h, best_c

def local_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in steps:
        if time.time() > deadline - 3:
            break
        while time.time() < deadline - 3:
            improved = False
            for i in range(n):
                if time.time() > deadline - 3:
                    return h, best_c
                for j in range(max(0, i - 25), min(n, i + 26)):
                    if i == j:
                        continue
                    for sign in (1, -1):
                        ni = h[i] + sign * d
                        nj = h[j] - sign * d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = h[i] - sign * d, h[j] + sign * d
            if not improved:
                break
    return h, best_c

    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Phase 4: Coordinate descent again
    t4 = min(time.time() + 150, deadline - 130)
    h4, c4 = coord_descent(best_h, n, t4, max_sweeps=3000)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: L-BFGS with very high alpha
    t5 = min(time.time() + 80, deadline - 80)
    h5, c5 = lbfgs_optimize(best_h, n, t5, max_iters=100)
    if c5 < best_c:

    t6 = min(time.time() + 70, deadline - 20)
    best_h, best_c = local_refine(best_h, n, t6)

    best_c = compute_c5(best_h, n)
