def project_box_sum(y, n):
    target = 0.5*n
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    if abs(float(y.sum()) - target) < 1e-13:
        return y
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0

def fft_len(n):
 L=1; need=2*n-1

def project_box_sum(y,n):
 target=0.5*n
 y=np.clip(np.asarray(y,float),0,1)
 if abs(y.sum()-target)<1e-13: return y
 lo=y.min()-1; hi=y.max()+1
 for _ in range(80):

def c_terms(h):
 n=h.size; S=h.sum(); L=fft_len(n)
 f=np.fft.rfft(h,L); auto=np.fft.irfft(f*np.conj(f),L)[:n]
 P=np.empty(n+1); P[0]=0; P[1:]=np.cumsum(h)
 c=np.empty(2*n-1)
 c[0]=S-auto[0]
 if n>1:
  c[1:n]=P[n-1:0:-1]-auto[1:]
  c[n:]=(S-P[1:n])-auto[1:]
 return c

def c5_exact(h):

def smooth_obj(x,beta,mu,n):
 x=np.clip(x,0,1); c=c_terms(x); m=c.max()
 z=beta*(c-m); e=np.exp(z); lse=m+np.log(e.sum())/beta

def smooth_grad(x,beta,mu,n):
 x=np.clip(x,0,1); c=c_terms(x); m=c.max()
 z=beta*(c-m); e=np.exp(z); w=e/e.sum()
 dx=2.0/n
 w0=w[0]; base=w0*(1-2*x)
 if n>1:
  p=w[1:n]; mvec=w[n:]; q=p+mvec
  rev=np.cumsum(p[::-1]); pos=np.zeros(n); pos[:n-1]=rev[n-2::-1]
  neg=np.zeros(n); neg[1:]=np.cumsum(mvec)
  base+=pos+neg
  qa=np.zeros(n); qa[1:]=q; L=fft_len(n)
  fx=np.fft.rfft(x,L); fq=np.fft.rfft(qa,L)
  px=np.fft.irfft(fx*np.conj(fq),L)[:n]
  fr=np.fft.rfft(x[::-1],L)
  pr=np.fft.irfft(fr*np.conj(fq),L)[:n]
  base-=px+pr[::-1]

import numpy as np
import time
from scipy.optimize import minimize

def fft_len(n):
    L = 1
    need = 2 * n - 1
    while L < need:
        L <<= 1
    return L

def project_box_sum(y, n):
    target = 0.5 * n
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    y = np.where(np.isfinite(y), y, 0.5)
    y = np.clip(y, 0.0, 1.0)
    if abs(float(y.sum()) - target) < 1e-13:
        return y
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = float(np.clip(y - tau, 0.0, 1.0).sum())
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def c_terms(h):
    n = h.size
    S = float(h.sum())
    L = fft_len(n)
    f = np.fft.rfft(h, L)
    auto = np.fft.irfft(f * np.conj(f), L)[:n]
    P = np.empty(n + 1)
    P[0] = 0.0
    P[1:] = np.cumsum(h)
    c = np.empty(2 * n - 1)
    c[0] = S - auto[0]
    if n > 1:
        c[1:n] = P[n - 1:0:-1] - auto[1:]
        c[n:] = (S - P[1:n]) - auto[1:]
    return c

def c5_exact(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def smooth_obj(x, beta, mu, n):
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    c = c_terms(x)
    m = float(c.max())
    z = beta * (c - m)
    e = np.exp(z)
    lse = (m + float(np.log(e.sum()))) / beta
    diff = float(x.sum() - 0.5 * n)
    return float((2.0 / n) * lse + 0.5 * mu * diff * diff)

def smooth_grad(x, beta, mu, n):
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    c = c_terms(x)
    m = float(c.max())
    z = beta * (c - m)
    e = np.exp(z)
    w = e / float(e.sum())
    dx = 2.0 / n
    w0 = w[0]
    base = w0 * (1.0 - 2.0 * x)
    if n > 1:
        p = w[1:n]
        mv = w[n:]
        q = p + mv
        rev = np.cumsum(p[::-1])
        pos = np.zeros(n)
        pos[:n - 1] = rev[n - 2::-1]
        neg = np.zeros(n)
        neg[1:] = np.cumsum(mv)
        base += pos + neg
        qa = np.zeros(n)
        qa[1:] = q
        L = fft_len(n)
        fx = np.fft.rfft(x, L)
        fq = np.fft.rfft(qa, L)
        px = np.fft.irfft(fx * np.conj(fq), L)[:n]
        fr = np.fft.rfft(x[::-1], L)
        pr = np.fft.irfft(fr * np.conj(fq), L)[:n]
        base -= px + pr[::-1]
    g = dx * base
    if mu:
        g = g + mu * (x.sum() - 0.5 * n)
    return g

def symmetrize(h):
    n = h.size
    return project_box_sum(0.5 * (h + h[::-1]), n)

def interpolate_h(h, n):
    n0 = h.size
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_box_sum(np.interp(x1, x0, h), n)

def perturb_h(h, rng, scale):
    n = h.size
    return project_box_sum(h + rng.normal(0.0, scale, n), n)

def sine_start(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    a = rng.uniform(0.10, 0.28)
    b = rng.uniform(0.04, 0.20)
    d = rng.uniform(0.0, 0.12)
    h = 0.5 + a * np.sin(2.0 * np.pi * x) + b * np.cos(4.0 * np.pi * x) + d * np.sin(6.0 * np.pi * x)
    return project_box_sum(h, n)

def lbfgs_polish(h, n, deadline, sym=False):
    h = project_box_sum(h, n)
    if sym:
        h = symmetrize(h)
    best_h = h.copy()
    best_c = c5_exact(h)
    bounds = [(0.0, 1.0)] * n
    if n > 700:
        maxiter = 220
    elif n > 450:
        maxiter = 320
    else:
        maxiter = 450
    betas = (1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 250.0, 600.0)
    for beta in betas:
        if time.time() > deadline - 5:
            break
        mu = 40.0
        try:
            res = minimize(
                smooth_obj, h, args=(beta, mu, n),
                method="L-BFGS-B", jac=smooth_grad, bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-13, "gtol": 1e-12,
                         "maxfun": maxiter * 12, "maxls": 20, "maxcor": 10,
                         "disp": False}
            )
            h2 = project_box_sum(res.x, n)
        except Exception:
            h2 = project_box_sum(h, n)
        if sym:
            h2 = symmetrize(h2)
        c2 = c5_exact(h2)
        if np.isfinite(c2) and c2 < best_c:
            best_c = c2
            best_h = h2.copy()
        if np.isfinite(c2) and (c2 - best_c) < 0.025:
            h = h2
        else:
            h = best_h
    return best_h, best_c

    h = project_box_sum(h, n)
    if alloc <= 8:

    return lbfgs_polish(h, n, time.time() + alloc, sym)
