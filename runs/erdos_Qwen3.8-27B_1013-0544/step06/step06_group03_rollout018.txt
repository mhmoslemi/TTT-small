import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_val(c, alpha)

def _obj_active(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_val(cc[idx], alpha)

def slsqp_run(h, n, alpha, top_idx=None, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    if top_idx is not None:
        obj, args = _obj_active, (n, top_idx, alpha)
    else:
        obj, args = _obj_full, (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_grad(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if lag > 0:
        g[:n-m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:n-m]
    else:
        g[m:] += dx * (1.0 - h[:n-m])
        g[:n-m] -= dx * h[m:]
    return g

def outer_lp(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_grad(h_curr, n, lag)
        cons.append(c[k] + g @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.CPLEX if hasattr(cp, 'CPLEX') else cp.GLOP]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), None
    val = h_var.value
    if val is None:
        return h_curr.copy(), None
    return project_feasible(val, n), prob.value
