        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            if tot < 1e-15:

            h += (diff / tot) * h

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h

        grad[m:] += 1.0 - h[:n - m]
    else:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] -= h[:n - m]
    return grad

def local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for d in [0.015, 0.007, 0.003, 0.001, 0.0004, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-3:]
            for k_star in top_idx:
                lag = int(k_star) - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                stride = max(1, (n - m) // 150)
                for i in range(0, n - m, stride):
                    if time.time() > deadline:
                        break
                    j = i + m
                    diff_h = h[i] - h[j]
                    if abs(diff_h) < 1e-7:
                        continue
                    delta = min(d, abs(diff_h) * 0.4) * (1 if diff_h > 0 else -1)
                    delta = max(-min(d, h[i]), min(d, 1-h[j])) if diff_h > 0 else min(min(d, 1-h[i]), max(-d, -h[j]))
                    # Simplified:
                    if diff_h > 0:
                        delta = min(d, diff_h * 0.4, h[i], 1.0 - h[j])
                    else:
                        delta = -min(d, (-diff_h) * 0.4, h[j], 1.0 - h[i])
                    if abs(delta) < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def binary_anneal(h_start, n, deadline, rng):
    h_bin = np.where(h_start >= 0.5, 1.0, 0.0)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        order = np.argsort(-h_start)
        for idx in order:
            if h_bin[int(idx)] == 1:
                h_bin[int(idx)] = 0
                n_ones -= 1
                if n_ones == target:
                    break
    elif n_ones < target:
        order = np.argsort(h_start)
        for idx in order:
            if h_bin[int(idx)] == 0:
                h_bin[int(idx)] = 1
                n_ones += 1
                if n_ones == target:
                    break
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    T = 0.05
    for it in range(5000):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h_bin[i], h_bin[j] = 0.0, 1.0
        c_new = compute_c5(h_bin, n)
        if c_new <= best_c or rng.random() < np.exp((best_c - c_new) / T):
            best_c = min(best_c, c_new)
            best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
        if it % 100 == 99:
            T *= 0.95
        if T < 0.005:
            T = 0.005
    return best_h, best_c

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Active set SLSQP
    if time.time() < deadline - 10:
        h_as, c_as = active_set_opt(best_h, n, min(deadline, time.time() + 25))
        if c_as < best_c:
            best_c, best_h = c_as, h_as
    
    # Phase 2: Full SLSQP
    if time.time() < deadline - 8:
        for alpha in [20000, 80000, 200000]:
            if time.time() > deadline - 4:
                break
            h_new = slsqp(best_h, n, alpha, 2500, 1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
    
    # Phase 3: Multi-gradient
    if time.time() < deadline - 8:
        h_sg, c_sg = multigrad_search(best_h, n, min(deadline, time.time() + 12))
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg
    
    # Phase 4: Binary annealing
    if time.time() < deadline - 6:
        h_bin, c_bin = binary_anneal(best_h, n, min(deadline, time.time() + 8), rng)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin
        # Also try to optimize from the binary solution
        if c_bin < best_c + 0.005 and time.time() < deadline - 5:
            h_new = slsqp(h_bin, n, 50000, 2000, 1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
    
    # Phase 5: Local search
    if time.time() < deadline - 1:
        h_ls, c_ls = local_search(best_h, n, deadline - 0.5)
        if c_ls < best_c:
            best_c, best_h = c_ls, h_ls
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        alloc = min(280, (deadline - t0) * 0.22)
        h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    n_candidates = [200, 250, 300, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 45:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 20
        if remaining < 45:
            break
        alloc = min(50, remaining * 0.15)
        if alloc < 30:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
