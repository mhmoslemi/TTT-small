```python
def project_box_sum(y, n, target):
    """Project y onto {0<=x<=1, sum(x)=target}."""
    # Bisection on tau: sum(clip(y-tau, 0, 1)) = target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)
```

                       n_iter=200000, momentum=0.9):

    h = h.copy()
    v = h.copy()  # momentum variable
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Adaptive alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iter)
        
        # Compute gradient
        grad = compute_grad(h, n, alpha)
        
        # Nesterov step
        v = momentum * v + grad
        h_new = h - lr * v  # need to determine lr

        # Check improvement
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cs = alpha * (c - c.max())
    w = np.exp(cs)
    w /= w.sum()
    # Gradient computation via matrix
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(-(n-1), n)[None, :]
    ipm = i_idx + m_idx
    mask1 = (ipm >= 0) & (ipm < n)
    term1 = np.where(mask1, 1.0 - h[np.clip(ipm, 0, n-1)], 0.0)
    imn = i_idx - m_idx
    mask2 = (imn >= 0) & (imn < n)
    term2 = np.where(mask2, h[np.clip(imn, 0, n-1)], 0.0)
    return dx * (term1 - term2) @ w

def project_box_sum(y, target):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0, 1)

def nesterov_hdescend(h, n, deadline, alpha_start=50, alpha_end=500000, n_max=500000):

    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = 0.9
    lr = 1.0
    prev_c = best_c
    
    for it in range(n_max):
        if time.time() > deadline:
            break
        t = it / n_max
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        
        if it % 20 == 0:  # periodic evaluation
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        
        grad = compute_grad(h, n, alpha)
        v = mu * v + grad
        # Adaptive step size
        h_trial = h - lr * v
        h_trial = project_box_sum(h_trial, target)
        h = h_trial
        # Decay momentum slightly
        mu = 0.9 + 0.05 * (1 - t)  # start with 0.95, end with 0.9
        # Decay learning rate
        lr *= 0.9999
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cs = alpha * (c - c.max())
    w = np.exp(cs)
    w /= w.sum()
    i_idx = np.arange(n)[:, None]
