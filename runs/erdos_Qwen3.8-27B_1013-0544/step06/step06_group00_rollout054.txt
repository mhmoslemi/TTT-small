```python
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx

            kp = -k
```

```python
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            s = mask.sum()
            if s == 0: break
            h[mask] += diff / s
        else:
            mask = h > 1e-15
            s = mask.sum()
            if s == 0: break
            h[mask] += diff / s
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_grad_fast(h, n, alpha, K=30):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()

def binary_delta(h, n, i, j):
    """Compute delta in correlation when swapping h[i]=0→1, h[j]=1→0."""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    k1 = n-1 + (j-i)
    delta[k1] += 1.0
    k2 = n-1 + (i-j)
    delta[k2] += 1.0
    return delta

    best_c = cur_c
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False

                    cur_c = nc
                    ones.remove(j); zeros.remove(i)
                    ones.add(i); zeros.add(j)
                    improved = True
                    break
            if improved:

        if not improved:

        if cur_c < best_c:

            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=2000000):

    h = to_binary(h_init, n).copy()

    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    zeros_arr = np.where(h == 0)[0].copy()
    ones_arr = np.where(h == 1)[0].copy()
    T = 0.02
    cool = 0.99997
    scale = 5000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai]); j = int(ones_arr[bi])
        d = binary_delta(h, n, i, j)
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale*(cur_c-nc)/T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[ai] = j; ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
```

```python
                grad[kp:] += w * (1.0 - h[:nk])
                grad[:nk] -= w * h[kp:]
    return grad
```

For n=400, this loop has 799 iterations. Each iteration does O(n) work. Total: 799*400 = 320K operations. In Python (not vectorized across the loop), this might take 1-5ms per call. With 10000 steps, that's 10-50 seconds per alpha level. With 9 alpha levels, that's 90-450 seconds total. Tight but feasible.

Actually, I just realized there might be an issue with the projection. When I do h - lr * g and then project, the projection might distort the direction of descent. A better approach is to use the projected gradient method where the projection is the Euclidean projection onto the feasible set.

The exact Euclidean projection onto {h: 0≤h≤1, sum(h)=n/2} is:
1. Find λ such that sum_i clip(x_i - λ, 0, 1) = n/2
2. h_i = clip(x_i - λ, 0, 1)

Finding λ is a 1D root-finding problem. I can use bisection.

```python
def project_exact(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix any numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0, 1)
    return h
```

This is O(n) per call (with 100 bisection iterations, it's O(100n) = O(40000) which is fine).

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_exact(x, n):
    target = n / 2.0
    lo, hi = x.min() - 2.0, x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-11:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += w * (1.0 - h[k:])
                if nk < n:
                    grad[k:] -= w * h[:nk]
        else:
            kp = -k
            nk = n - kp
            if nk > 0:
                grad[kp:] += w * (1.0 - h[:nk])
                if nk < n:
                    grad[:nk] -= w * h[kp:]
    return grad

def smooth_optimize(h, n, deadline, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [15, 40, 100, 250, 600, 1500, 4000, 10000, 30000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        lr = 0.5
        h_cur = h.copy()
        c_cur = compute_c5(h_cur, n)
        stall = 0
        for step in range(30000):
            if time.time() > deadline - 2:
                break
            g = smooth_grad(h_cur, n, alpha)
            h_new = project_exact(h_cur - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur - 1e-15:
                h_cur = h_new
                c_cur = c_new
                lr = min(lr * 1.03, 5.0)
                stall = 0
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_new.copy()
            else:
                lr *= 0.85
                stall += 1
                if stall > 80:
                    break
        h = best_h.copy()
    return best_h, best_c

def binary_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def greedy_binary(h_init, n, deadline, max_sweeps=150):
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0].astype(int))
    zeros = list(np.where(h == 0)[0].astype(int))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx_i, i in enumerate(zeros[:]):
            if time.time() > deadline:
                break
            for j in ones[:]:
                d = binary_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    jv = ones.index(j)
                    del ones[jv]
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=3000000):
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    zeros_arr = np.where(h == 0)[0].copy()
    ones_arr = np.where(h == 1)[0].copy()
    T = 0.02
    cool = 0.99996
    scale = 5000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        d = binary_delta(h, n, i, j)
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

    t0 = time.time()
```
