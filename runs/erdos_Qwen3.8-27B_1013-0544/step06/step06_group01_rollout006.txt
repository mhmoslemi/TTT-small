        k_hi = min(2 * n - 2, 2 * n - 2 - j + n - 1)  # need j+k-n+1 < n => k < n-1-j+n = 2n-1-j

        # j+k-n+1 <= n-1 => k <= 2n-2-j
        k_lo_c = max(0, n - 1 - j)
        k_hi_c = min(2 * n - 2, 2 * n - 2 - j)
        if k_lo_c <= k_hi_c:
            ks = np.arange(k_lo_c, k_hi_c + 1)
            m = ks + j - n + 1  # index into h
            grad[j] += np.dot(w[ks], 1.0 - h[m])
        # Second part
        # j-k+n-1 >= 0 => k <= j+n-1
        # j-k+n-1 <= n-1 => k >= j
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + n - 1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            m2 = j - ks2 + n - 1  # index into h
            grad[j] -= np.dot(w[ks2], h[m2])
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-15: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (d / t) * h
    return np.clip(h, 0, 1)

def fourier_to_h(a, b, n):
    """a and b have length K, h has length n over [0,2)."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, len(a) + 1):
        h += a[k-1] * np.cos(k * np.pi * x) + b[k-1] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, K, n):
    x = np.linspace(0, 2, n, endpoint=False)
    a = np.zeros(K); b = np.zeros(K)
    for k in range(1, K + 1):
        a[k-1] = 2.0 / n * np.sum(h * np.cos(k * np.pi * x))
        b[k-1] = 2.0 / n * np.sum(h * np.sin(k * np.pi * x))
    return a, b

def fourier_obj(params, K, n, penalty=100.0):
    a = params[:K]; b = params[K:]
    h = fourier_to_h(a, b, n)
    c5 = compute_c5(h, n)
    viol = np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h - 1)**2)
    return c5 + penalty * viol

def optimize_fourier(a, b, K, n, deadline, rng):
    params0 = np.concatenate([a, b])
    best_params = params0.copy()
    best_val = fourier_obj(params0, K, n)
    # Try multiple methods and initial points
    methods = ['Nelder-Mead', 'Powell']
    for m in methods:
        if time.time() > deadline - 20: break
        for restart in range(3):
            if time.time() > deadline - 10: break
            p0 = params0 + rng.normal(0, 0.05, 2*K) if restart > 0 else params0.copy()
            try:
                res = minimize(fourier_obj, p0, args=(K, n), method=m,
                             options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
                if res.fun < best_val:
                    best_val = res.fun; best_params = res.x
            except: pass
    h_best = fourier_to_h(best_params[:K], best_params[K:], n)
    return project_feasible(h_best, n)

def smooth_max_val(h, n, alpha):
    c = np.correlate(h, 1-h, mode='full')
    cs = c - c.max()
    if alpha * cs.max() > 500: return c.max() * 2.0/n
    return (c.max() + np.log(np.sum(np.exp(alpha * cs))) / alpha) * 2.0/n

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    cs = c - c.max()
    if alpha * cs.max() > 500:
        w = np.zeros(len(c)); w[c == c.max()] = 1.0; w /= w.sum()
    else:
        w = np.exp(alpha * cs); w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_k w[k] * (1-h[j+k-n+1]) for valid indices
        lo = max(0, n-1-j); hi = min(2*n-2, 2*n-2-j)
        if lo <= hi:
            ks = np.arange(lo, hi+1)
            m = ks + j - n + 1
            grad[j] += np.dot(w[lo:hi+1], 1.0 - h[m])
        # Second term: sum_k w[k] * h[j-k+n-1] for valid indices
        lo2 = max(0, j); hi2 = min(2*n-2, j+n-1)
        if lo2 <= hi2:
            ks2 = np.arange(lo2, hi2+1)
            m2 = j - ks2 + n - 1
            grad[j] -= np.dot(w[lo2:hi2+1], h[m2])
    return grad * dx

def projected_gradient(h, n, alpha, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for it in range(500):
        if time.time() > deadline: break
        g = smooth_max_grad(h, n, alpha)
        # Add constraint penalty gradient
        g += 0.1 * (h.sum() - n/2.0) * np.ones(n)
        # Adaptive step size (Barzilai-Borwein style)
        step = 1.0 / (alpha * 10.0 + 1.0)
        # Backtracking
        f_cur = smooth_max_val(h, n, alpha)
        for _ in range(20):
            h_try = project_feasible(h - step * g, n)
            f_try = smooth_max_val(h_try, n, alpha)
            if f_try < f_cur:
                break
            step *= 0.5
        h = h_try
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
    return h, best_c
