        if diff > 0:

        else:

            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pair_cd_sweep(h, n, order=None, max_rounds=5):

    h = h.copy()
    c = h.shape[0]  # just for reference

    if order is None:
        order = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    for rnd in range(max_rounds):
        improved = False
        for (i, j) in order:
            if hi := min(1.0 - h[i], h[j]) <= max(-h[i], h[j] - 1.0):
                continue
            lo = max(-h[i], h[j] - 1.0)
            
            # Compute slopes for this pair
            m_arr = np.arange(n_corr)
            valid_mi = m_arr - i
            mask_mi = (valid_mi >= 0) & (valid_mi < n)
            s_mi = np.where(mask_mi, 1.0 - h[np.clip(valid_mi, 0, n-1)], 1.0)
            # Fix: where not valid, coefficient is 1 (the h[i] term just adds h[i] to c[m])
            # Actually where mask is False, the term h[i] doesn't appear in the product part,
            # so c[m] just gets +h[i] added (from sum_j h[j] part)
            # So coefficient of h[i] in c[m] is 1 (from linear part) - 0 (from product part) = 1
            # And when valid, coefficient is 1 - h[m-i]
            
            valid_mj = m_arr - j
            mask_mj = (valid_mj >= 0) & (valid_mj < n)
            s_mj = np.where(mask_mj, 1.0 - h[np.clip(valid_mj, 0, n-1)], 1.0)
            
            slopes = s_mi - s_mj
            
            # Current c values
            c_cur = compute_all_lags_nodx(h, n)
            
            # Binary search for optimal delta
            f_lo = np.max(c_cur + lo * slopes)
            f_hi = np.max(c_cur + hi * slopes)
            L_lo = min(f_lo, f_hi)
            L_hi = max(f_lo, f_hi, np.max(c_cur))
            
            best_delta = 0.0
            for _ in range(40):
                L = (L_lo + L_hi) / 2.0
                # Check if feasible
                zero_mask = np.abs(slopes) < 1e-15
                if np.any(zero_mask) and np.max(c_cur[zero_mask]) > L:
                    L_lo = L
                    continue
                pos_mask = slopes > 1e-15
                neg_mask = slopes < -1e-15
                ub = np.min((L - c_cur[pos_mask]) / slopes[pos_mask]) if np.any(pos_mask) else np.inf
                lb = np.max((L - c_cur[neg_mask]) / slopes[neg_mask]) if np.any(neg_mask) else -np.inf
                lb = max(lb, lo)
                ub = min(ub, hi)
                if lb <= ub:
                    L_hi = L
                    best_delta = (lb + ub) / 2.0
                else:
                    L_lo = L
            
            new_val = np.max(c_cur + best_delta * slopes)
            old_val = np.max(c_cur)
            if new_val < old_val - 1e-12:
                h[i] = np.clip(h[i] + best_delta, 0, 1)
                h[j] = np.clip(h[j] - best_delta, 0, 1)
                improved = True
        
        if not improved:
            break
    
    return h

If h[i] > h[j], we should decrease h[i] (and increase h[j]).

This is exactly what the pair CD does! The binary search finds the optimal δ that balances increasing R(k*) against potentially decreasing other R(k).

2. Focus on pairs (i, i+|k*|) for i = 0, ..., n-1-|k*|

def bottleneck_pair_cd(h, n, max_rounds=50):
    """Pair CD focusing on the bottleneck lag."""
    h = h.copy()
    for rnd in range(max_rounds):
        # Compute all lags
        c = get_all_lags(h)  # c[m] for m=0..2n-2
        # R(k) = c[k+n-1] (the autocorrelation at lag k)
        # Actually, let me reclarify the indexing.
        # c[m] = sum_i h[i]*(1-h[m-i])
        # R(k) = sum_i h[i]*h[i+k] = c[2(n-1)-k] ... 
        # Hmm, let me just work with c directly.
        # C5 = max_m c[m] * dx
        # We want to minimize max_m c[m]
        
        m_star = np.argmax(c)
        # Focus on increasing c[other] or decreasing c[m_star]
        # The pairs that affect c[m_star] most are those (i,j) where 
        # the move decreases c[m_star].
        
        # For c[m_star], changing h[i]+=delta, h[j]-=delta:
        # delta_c = delta * (s_mi[m_star] - s_mj[m_star])
        # where s_mi[m] = 1 if m-i out of [0,n-1], else 1-h[m-i]
        # To decrease c[m_star], we want delta_c < 0
        # i.e., delta * (s_mi - s_mj) < 0
        
        # Simplest: focus on pairs (i, j) where j-i ≈ m_star - (n-1)
        # (the lag that achieves the max)
        
        lag = m_star - (n - 1)
        m_abs = abs(lag)
        
        # Pairs to try: (i, i+m_abs) for valid i
        pairs = []
        for i in range(n - m_abs):
            j = i + m_abs
            pairs.append((i, j))
        
        # Also try the opposite lag
        for i in range(n - m_abs):
            j = i + m_abs
            pairs.append((j, i))  # opposite direction
        
        improved = 0
        for (i, j) in pairs:
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi <= lo + 1e-15:
                continue
            
            # Compute slopes for this pair
            m_arr = np.arange(2*n-1)
            vm = m_arr - i
            mi = (vm >= 0) & (vm < n)
            smi = np.where(mi, 1.0 - h[np.clip(vm, 0, n-1)], 1.0)
            vm = m_arr - j
            mj = (vm >= 0) & (vm < n)
            smj = np.where(mj, 1.0 - h[np.clip(vm, 0, n-1)], 1.0)
            slopes = smi - smj
            
            # Binary search
            f_lo = np.max(c + lo * slopes)
            f_hi = np.max(c + hi * slopes)
            L_lo = min(f_lo, f_hi)
            L_hi = max(f_lo, f_hi)
            best_d = 0.0
            for _ in range(30):
                L = (L_lo + L_hi) / 2.0
                zm = np.abs(slopes) < 1e-15
                if np.any(zm) and np.max(c[zm]) > L:
                    L_lo = L; continue
                pm = slopes > 1e-15
                nm = slopes < -1e-15
                ub = np.min((L - c[pm]) / slopes[pm]) if np.any(pm) else np.inf
                lb = np.max((L - c[nm]) / slopes[nm]) if np.any(nm) else -np.inf
                lb, ub = max(lb, lo), min(ub, hi)
                if lb <= ub:
                    L_hi = L; best_d = (lb+ub)/2.0
                else:
                    L_lo = L
            
            if np.max(c + best_d*slopes) < np.max(c) - 1e-12:
                h[i] = np.clip(h[i]+best_d, 0, 1)
                h[j] = np.clip(h[j]-best_d, 0, 1)
                c = get_all_lags(h)
                improved += 1
        
        if improved == 0:
            break
    
    return project_feasible(h, n)
