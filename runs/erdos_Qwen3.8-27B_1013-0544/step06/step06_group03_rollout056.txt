def lag_gradient(h, n, k):
    """Gradient of c_k = dx * sum h[n+k]*(1-h[n]) w.r.t. h, for lag k (can be negative)."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
        return g * dx
    if k > 0:
        # h[j] as first factor (h[n+k] where n+k=j, so n=j-k): valid for j>=k
        valid1 = np.arange(k, n)
        g[valid1] += (1.0 - h[valid1 - k])
        # h[j] as second factor (1-h[n] where n=j): valid for j+k < n, i.e., j < n-k
        valid2 = np.arange(0, n - k)
        g[valid2] -= h[valid2 + k]
    else:
        m = -k

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h += (diff / room.sum()) * room
        else:
            if h.sum() < 1e-14:
                break
            h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def _sf(x, n):
    return float(np.sum(x) - n / 2.0)

def _sj(x, n):
    return np.ones(n)

def _lse(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp_run(h, n, alpha, maxiter=5000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n,)}]
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return _lse(c, alpha)
    try:
        res = minimize(f, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        return (1.0 - 2.0 * h) * dx
    if k > 0:
        v1 = np.arange(k, n)
        g[v1] += (1.0 - h[v1 - k])
        v2 = np.arange(0, n - k)
        g[v2] -= h[v2 + k]
    else:
        m = -k
        v1 = np.arange(0, n - m)
        g[v1] += (1.0 - h[v1 + m])
        v2 = np.arange(m, n)
        g[v2] -= h[v2 - m]
    return g * dx

def topk_local_search(h, n, deadline, K=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    stall = 0
    max_stall = 200
    while time.time() < deadline and stall < max_stall:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-K:]
        best_move = None
        best_gain = 1e-15
        for ki in top_idx:
            lag = ki - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            g = lag_gradient(h, n, lag)
            if lag > 0:
                gains = g[:n - m] - g[m:]
                best_a = int(np.argmax(gains))
                if gains[best_a] > best_gain:
                    best_gain = gains[best_a]
                    best_move = (best_a, best_a + m, 1)
            else:
                gains = g[n - m:] - g[:m]
                best_a = int(np.argmax(gains))
                if gains[best_a] > best_gain:
                    best_gain = gains[best_a]
                    best_move = (n - m + best_a, best_a, 1)
        if best_move is None:
            break
        a, b, direction = best_move
        if direction == 1:
            d = min(step, h[a] * 0.4, (1.0 - h[b]) * 0.4)
        else:
            d = min(step, h[b] * 0.4, (1.0 - h[a]) * 0.4)
        if d < 1e-9:
            break
        h_old = h.copy()
        if direction == 1:
            h[a] -= d
            h[b] += d
        else:
            h[a] += d
            h[b] -= d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            stall = 0
        else:
            h = h_old
            stall += 1
            step *= 0.7
    return h, best_c

def cutting_plane(h, n, deadline, max_iters=60):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top_count = 15
        top_idx = np.argsort(c)[-top_count:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
                cp.abs(h_var - h_curr) <= trust]
        for ki in top_idx:
            lag = ki - (n - 1)
            g = lag_gradient(h_curr, n, lag)
            delta = (g @ (h_var - h_curr))
            cons.append(c[ki] + delta <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        if c_new > best_c * 1.001:
            trust *= 0.7
        else:
            trust *= 0.9
        trust = max(0.003, trust)
    return best_h, best_c

def structured_restart(h, n, rng, mode):
    hp = h.copy()
    if mode == 0:
        perm = np.random.permutation(n)
        hp[perm] += rng.normal(0, 0.02, n)
    elif mode == 1:
        block = max(5, n // 10)
        s = rng.randint(0, n - block)
        hp[s:s+block] += rgp = rng.uniform(-0.03, 0.03, block)
    elif mode == 2:
        freq = rng.randint(1, 5)
        phase = rng.rand() * 2 * np.pi
        x = np.linspace(0, 2 * np.pi * freq, n)
        hp += 0.01 * np.sin(x + phase)
    else:
        amp = 0.015
        hp += amp * rng.randn(n)
    return project_feasible(hp, n)

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t_end1 = min(deadline, time.time() + 40)
    for alpha in [5000, 50000, 200000]:
        if time.time() > t_end1:
            break
        h_new = slsqp_run(best_h, n, alpha, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 20:
        ls_end = min(deadline - 15, time.time() + 120)
        h_new, c_new = topk_local_search(best_h, n, ls_end, K=5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 20:
        cp_end = min(deadline - 10, time.time() + 100)
        h_new, c_new = cutting_plane(best_h, n, cp_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 10:
        h_new = slsqp_run(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)

            best_c = c_new
