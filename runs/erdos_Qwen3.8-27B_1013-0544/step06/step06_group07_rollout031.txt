```python
def c5_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Log-sum-exp smoothing
    m = corr.max()
    shifted = (corr - m) / alpha
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient of log-sum-exp
    # d/dh_j of log(sum_k exp(O_k/alpha)) = (1/alpha) * sum_k weights_k * dO_k/dh_j
    # dO_k/dh_j = [(1-h_{j+k}) - h_{j-k}] * dx (for valid indices)
    
    grad = np.zeros(n)
    for k_offset in range(-(n-1), n):
        # For shift k_offset (lag in correlate output is n-1+k_offset)
        w = weights[n-1+k_offset]
        if w < 1e-10:
            continue
        # dO/dh_j for this shift
        # h[i] * (1-h[i+k_offset]) where i ranges over valid indices
        # d/dh_j: (1-h[j+k_offset]) if j+k_offset < n, minus h[j-k_offset] if j-k_offset >= 0
        j = np.arange(n)
        pos = j + k_offset
        neg = j - k_offset
        valid_pos = (pos >= 0) & (pos < n)
        valid_neg = (neg >= 0) & (neg < n)
        contrib = np.zeros(n)
        contrib[valid_pos] += (1 - h[pos[valid_pos]])
        contrib[valid_neg] -= h[neg[valid_neg]]
        grad += w * contrib * dx / alpha
    return grad
```
