    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _min_max_convex(D, C, iters=50):
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if float(np.max(D + m1 * C)) < float(np.max(D + m2 * C)):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _min_max_convex(D, C, iters=35):
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if float(np.max(D + m1 * C)) < float(np.max(D + m2 * C)):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def coord_desc(h, n, deadline, max_sweeps=400, rng=None):
    h = project_feasible(h.copy(), n)
    all_lags = compute_all_lags(h, n)
    best_h = h.copy()
    best_c = float(np.max(all_lags))
    dx = 2.0 / n
    n_lags = 2 * n - 1
    L_arr = np.arange(n_lags) - (n - 1)
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 30 == 29 and rng is not None:
            rng.shuffle(order)
        for j in order:
            jp = j + L_arr
            jm = j - L_arr
            mask_p = (jp >= 0) & (jp < n)
            mask_m = (jm >= 0) & (jm < n)
            C = np.zeros(n_lags)
            C[mask_p] = dx * (1.0 - h[jp[mask_p]])
            C[mask_m] -= dx * h[jm[mask_m]]
            hj = h[j]
            D = all_lags - hj * C
            v = _min_max_convex(D, C, iters=30)
            delta = v - hj
            if abs(delta) > 1e-13:
                h[j] = v
                all_lags += delta * C
        h = project_feasible(h, n)
        all_lags = compute_all_lags(h, n)
        c = float(np.max(all_lags))
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_snap_and_search(h, n, deadline):
    h_bin = (h > 0.5).astype(float)
    deficit = n / 2.0 - h_bin.sum()
    if deficit > 0:
        vals = sorted(range(n), key=lambda i: h[i], reverse=True)

                h_bin[vals[i]] = 0.0
    c_bin = compute_c5(h_bin, n)

    return h, compute_c5(h, n)
