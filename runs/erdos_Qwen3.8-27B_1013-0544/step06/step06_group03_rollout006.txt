    c = np.correlate(h, 1 - h, mode='full')

        if diff > 0:

            if tot < 1e-14:

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    c_full = np.correlate(x, 1 - x, mode='full') * dx
    c_top = c_full[top_idx]
    m = c_top.max()
    shifted = alpha * (c_top - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)

    grad = np.zeros(n)

        ml = k - (n - 1)
        if ml == 0:

        else:

            grad[mm:] += wi * (1 - x[:n - mm])
            grad[:n - mm] += wi * (-x[mm:])

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ct = c[top_idx]
    m = ct.max()
    sh = alpha * (ct - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _smooth_grad(x, n, alpha, top_idx):
    dx = 2.0 / n
    c_full = np.correlate(x, 1 - x, mode='full') * dx
    ct = c_full[top_idx]
    m = ct.max()
    sh = alpha * (ct - m)
    mx = sh.max()
    w = np.exp(sh - mx) if mx > 500 else np.exp(sh)
    w /= w.sum()
    g = np.zeros(n)
    for idx, k in enumerate(top_idx):
        wi = w[idx] * dx
        ml = k - (n - 1)
        if ml == 0:
            g += wi * (1 - 2 * x)
        elif ml > 0:
            g[:n-ml] += wi * (1 - x[ml:])
            g[ml:] += wi * (-x[:n-ml])
        else:
            mm = -ml
            g[mm:] += wi * (1 - x[:n-mm])
            g[:n-mm] += wi * (-x[mm:])
    return g

def cont_opt(h, n, deadline, alpha=10000, gamma=0.1):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    c = compute_all_lags(h, n)
    top = np.argsort(c)[-6:]
    lr = 0.02
    for it in range(200000):
        if time.time() > deadline: break
        g = _smooth_grad(h, n, alpha, top) + gamma * (1 - 2 * h)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if it % 200 == 0:
            c = compute_all_lags(h_new, n)
            top = np.argsort(c)[-6:]
            lr = min(0.05, max(0.002, lr * (0.9 if c_new > best_c * 1.005 else 1.05)))
        h = h_new
    return best_h, best_c

def binary_hill_climb(h, n, max_iters=5000):
    h = h.copy()
    c_curr = compute_c5(h, n)
    for _ in range(max_iters):
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        m_star = k_star - (n - 1)
        if m_star == 0: break
        # Find all swaps that reduce c_{m_star}
        if m_star > 0:
            idxs = np.arange(n - m_star)
            good = idxs[(h[idxs] == 0) & (h[idxs + m_star] == 1)]
        else:
            mm = -m_star
            idxs = np.arange(mm, n)
            good = idxs[(h[idxs] == 1) & (h[idxs - mm] == 0)]
        if len(good) == 0: break
        best_delta = 0
        best_swap = None
        for i in good:
            if m_star > 0:
                j = i + m_star
            else:
                j = i - mm
            h[i], h[j] = h[j], h[i]
            c_new = compute_c5(h, n)
            h[i], h[j] = h[j], h[i]
            delta = c_new - c_curr
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)
        if best_swap is None: break
        i, j = best_swap
        h[i], h[j] = h[j], h[i]
        c_curr += best_delta
    return h, c_curr

def ils_binary(h0, n, deadline, rng):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 10 == 1:
            # Perturb: random k swaps
            h = best_h.copy()
            k = rng.randint(3, 10)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(k):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                # Update sets
                ones[ones == i] = j
                zeros[zeros == j] = i
        h_new, c_new = binary_hill_climb(h, n, max_iters=2000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new  # continue from current
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=3000):
    target = n / 2.0
    def obj(x):
        return _smooth_obj(x, n, alpha, None)
    def obj_jac(x):
        # Use numerical or the gradient for all lags - expensive
        # Just return the gradient for top lags as approximation
        c = compute_all_lags(x, n)
        top = np.argsort(c)[-5:]
        return _smooth_grad(x, n, alpha, top)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
