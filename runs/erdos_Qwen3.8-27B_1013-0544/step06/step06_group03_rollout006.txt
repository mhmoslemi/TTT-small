```
def project_box_sum(v,n):
    v=np.asarray(v,dtype=float)

    target=0.5*n
    lo=float(v.min())-1.0

    for _ in range(42):
        mid=0.5*(lo+hi)
```

```
def lag_gradient(h,s,n):
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def project_box_sum(v, n):
    v = np.asarray(v, dtype=float)
    if v.size != n:
        v = np.resize(v, n)
    t = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(42):
        m = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - m, 0.0, 1.0)))
        if s > t:
            lo = m
        else:
            hi = m
    w = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        d = t - float(w.sum())
        if abs(d) < 1e-13:
            break
        idx = (w > 1e-12) & (w < 1.0 - 1e-12)
        cnt = int(idx.sum())
        if cnt == 0:
            break
        w = w.copy()
        w[idx] += d / cnt
        w = np.clip(w, 0.0, 1.0)
    return w

def all_lags(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_of(h, n):
    return float(np.max(all_lags(h, n)))

def lag_value(h, s, n):
    dx = 2.0 / float(n)
    s = int(s)
    if s == 0:
        return float(np.sum(h * (1.0 - h)) * dx)
    m = abs(s)
    if s > 0:
        return float(np.sum(h[m:] * (1.0 - h[:-m])) * dx)
    return float(np.sum(h[:-m] * (1.0 - h[m:])) * dx)

def lag_gradient(h, s, n):
    dx = 2.0 / float(n)
    s = int(s)
    m = abs(s)
    g = np.zeros(n, dtype=float)
    if m >= n:
        return g
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    if s > 0:
        g[m:] += dx * (1.0 - h[:-m])
        g[:-m] -= dx * h[m:]
    else:
        g[:-m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:-m]
    return g

def weighted_gradient(h, n, w):
    dx = 2.0 / float(n)
    off = n - 1
    L = 2 * n - 1
    if w.size != L:
        w = np.resize(w, L)
    a = np.convolve(1.0 - h, w, mode='full')[off:off + n]
    b = np.convolve(h, w[::-1], mode='full')[off:off + n]
    return dx * (a - b)

def smooth_core(h, n, alpha):
    c = all_lags(h, n)
    cm = float(np.max(c))
    shifted = alpha * (c - cm)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        f = cm
    else:
        w = e / s
        f = cm + float(np.log(s)) / alpha
    return f, w

def _sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)

def _sum_con_jac(x, n):
    return np.ones(n, dtype=float)

def _smooth_fn(x, n, alpha):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    f, w = smooth_core(h, n, alpha)
    return f * (0.5 * n)

def _smooth_jac(x, n, alpha):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    f, w = smooth_core(h, n, alpha)
    return weighted_gradient(h, n, w) * (0.5 * n)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    x0 = project_box_sum(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        r = minimize(_smooth_fn, x0, jac=_smooth_jac, args=(n, float(alpha)),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': int(maxiter), 'ftol': 1e-10, 'disp': False})
    except Exception:
        return x0.copy()
    if r.x is None:
        return x0.copy()
    return project_box_sum(r.x, n)

def _active_obj(x, n):
    return float(x[-1]) * (0.5 * n)

def _active_obj_jac(x, n):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 0.5 * n
    return j

def _active_sum(x, n):
    return float(np.sum(x[:n]) - 0.5 * n)

def _active_sum_jac(x, n):
    j = np.zeros(n + 1, dtype=float)
    j[:n] = 1.0
    return j

def _active_ineq(x, n, s):
    h = x[:n]
    return (float(x[-1]) - lag_value(h, s, n)) * (0.5 * n)

def _active_ineq_jac(x, n, s):
    h = x[:n]
    scale = 0.5 * n
    j = np.zeros(n + 1, dtype=float)
    j[:n] = -lag_gradient(h, s, n) * scale
    j[-1] = scale
    return j

def active_slsqp(h, n, deadline):
    hc = project_box_sum(h, n)
    bh = hc.copy()
    bc = c5_of(hc, n)
    stale = 0
    for r in range(4):
        if time.time() > deadline - 12:
            break
        top = min(2 + 2 * r, 10)
        c = all_lags(hc, n)
        cm = float(np.max(c))
        tol = max(1e-7, 4e-5 * (1.0 + 0.5 * r))
        act = np.where(c >= cm - tol)[0]
        if act.size > top:
            act = act[np.argsort(c[act])[-top:]]
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        cons = [{'type': 'eq', 'fun': _active_sum, 'jac': _active_sum_jac, 'args': (n,)}]
        for idx in act:
            s = int(idx - (n - 1))
            cons.append({'type': 'ineq', 'fun': _active_ineq, 'jac': _active_ineq_jac, 'args': (n, s)})
        x0 = np.concatenate([hc, [min(1.0, cm + 1e-8)]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        try:
            res = minimize(_active_obj, x0, jac=_active_obj_jac, args=(n,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 220, 'ftol': 1e-10, 'disp': False})
        except Exception:
            break
        if res.x is None:
            break
        hn = project_box_sum(res.x[:n], n)
        cn = c5_of(hn, n)
        if cn < bc - 1e-12:
            bc = cn
            bh = hn.copy()
            hc = hn
            stale = 0
        else:
            stale += 1
            if stale >= 3:
                break
            if cn <= bc + 5e-5:
                hc = hn
            else:
                mx = project_box_sum(0.8 * hc + 0.2 * hn, n)
                if c5_of(mx, n) <= bc + 1e-5:
                    hc = mx
                else:
                    break
    return bh, bc

def active_grad(h, n, deadline):
    h = project_box_sum(h, n)
    best = c5_of(h, n)
    target = 0.05
    while time.time() < deadline - 2:
        c = all_lags(h, n)
        cm = float(np.max(c))
        act = np.where(c >= cm - 1e-5)[0]
        if act.size > 10:
            act = act[np.argsort(c[act])[-10:]]
        g = np.zeros(n, dtype=float)
        for idx in act:
            g += lag_gradient(h, int(idx - (n - 1)), n)
        g /= float(act.size)
        mx = float(np.max(np.abs(g)))
        if mx <= 1e-12:
            break
        step = target / mx
        ok = False
        for _ in range(12):
            tr = project_box_sum(h - step * g, n)
            cn = c5_of(tr, n)
            if cn < best - 1e-14:
                h = tr
                best = cn
                ok = True
                break
            step *= 0.5
        if not ok:
            break
        target = min(target * 1.15, 0.18)
    return h, best

def coordinate_polish(h, n, deadline):
    h = project_box_sum(h, n)
    best = c5_of(h, n)
    evals = 0
    while time.time() < deadline - 1 and evals < 3000:
        c = all_lags(h, n)
        idx = int(np.argmax(c))
        s = int(idx - (n - 1))
        if s == 0:
            break
        m = abs(s)
        if s > 0:
            sr = np.arange(m, n)
            ds = np.arange(n - m)
        else:
            sr = np.arange(n - m)
            ds = np.arange(m, n)
        hs = h[sr]
        hd = h[ds]
        room = np.minimum(hs, 1.0 - hd)
        score = (1.0 + hs - hd) * room
        if float(score.max()) <= 1e-9:
            break
        order = np.argsort(score)[::-1][:32]
        moved = False
        for o in order:
            if time.time() > deadline - 1:
                break
            a = int(sr[o])
            b = int(ds[o])
            md = min(0.02, 0.6 * h[a], 1.0 - h[b])
            for f in (1.0, 0.5, 0.25):
                d = md * f
                if d < 1e-7:
                    break
                oa = h[a]
                ob = h[b]
                h[a] = oa - d
                h[b] = ob + d
                evals += 1
                cn = c5_of(h, n)
                if cn < best - 1e-14:
                    best = cn
                    moved = True
                    break
                h[a] = oa
                h[b] = ob
            if moved:
                break
        if not moved:
            break
    return h, best

def optimize_at_n(h_start, n, deadline):
    h = project_box_sum(h_start, n)
    bh = h.copy()
    bc = c5_of(h, n)
    for alpha, iters in ((50, 100), (500, 100), (2500, 100)):
        if time.time() > deadline - 25:
            break
        h2 = slsqp_smooth(h, n, alpha, iters, deadline)
        c2 = c5_of(h2, n)
        if c2 < bc - 1e-13:
            bc = c2
            bh = h2
            h = h2
        elif c2 <= bc + 5e-5:
            h = h2
    if time.time() < deadline - 25:
        h2, c2 = active_slsqp(h, n, deadline - 15)
        if c2 < bc - 1e-13:
            bc = c2
            bh = h2
            h = h2
        elif c2 <= bc + 5e-5:
            h = h2
    if time.time() < deadline - 18:
        h2, c2 = active_grad(h, n, deadline - 8)
        if c2 < bc - 1e-13:
            bc = c2
            bh = h2
            h = h2
    if time.time() < deadline - 12:
        h2, c2 = coordinate_polish(h, n, deadline - 4)
        if c2 < bc - 1e-13:
            bc = c2
            bh = h2
    if time.time() < deadline - 18:
        h2 = slsqp_smooth(h, n, 9000, 50, deadline)
        c2 = c5_of(h2, n)
        if c2 < bc - 1e-13:
            bc = c2
            bh = h2
    bh = project_box_sum(bh, n)
    return bh, float(c5_of(bh, n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s) - 8.0)
    rng = np.random.RandomState(int(seed))
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None or len(h0) < 20:
        h0 = np.full(300, 0.5)
    base = len(h0)
    best_h = None
    best_c = 1e100
    cand = [base]
    for d in (60, 120, 180, -60, -120):
        nn = base + d
        if 120 <= nn <= 800 and nn not in cand:
            cand.append(nn)
    first = True
    for n in cand:
        rem = deadline - time.time()
        if rem < 120:
            break
        if not first and best_h is not None and n == len(best_h):
            continue
        if first:
            alloc = min(480.0, rem * 0.55)
        else:
            alloc = min(220.0, rem * 0.35)
        if alloc < 90:
            break
        if best_h is not None:
            st = np.interp(np.linspace(0.0, 2.0, n),
                           np.linspace(0.0, 2.0, len(best_h)), best_h)
        elif n == base:
            st = h0.copy()
        else:
            st = np.interp(np.linspace(0.0, 2.0, n),
                           np.linspace(0.0, 2.0, base), h0)
        ho, co = optimize_at_n(st, n, time.time() + alloc)
        if co < best_c:
            best_c = co
            best_h = ho.copy()
        first = False
    if best_h is None:
        best_h = project_box_sum(h0, base)
        best_c = c5_of(best_h, base)
    if deadline - time.time() > 160:
        n = len(best_h)
        for sc in (0.015, 0.03):
            if deadline - time.time() < 130:
                break
            hp = project_box_sum(best_h + rng.normal(0.0, sc, n), n)
            alloc = min(130.0, deadline - time.time() - 25.0)
            if alloc < 80:
                break
            ho, co = optimize_at_n(hp, n, time.time() + alloc)
            if co < best_c:
                best_c = co
                best_h = ho.copy()
    if deadline - time.time() > 120:
        n = len(best_h)
        p = max(1, n // 50)
        alloc = min(100.0, deadline - time.time() - 20.0)
        if alloc >= 80:
            ho, co = optimize_at_n(np.roll(best_h, p), n, time.time() + alloc)
            if co < best_c:
                best_c = co
                best_h = ho.copy()
    bh = best_h
    bc = float(best_c)
    if deadline - time.time() > 30:
        n = len(bh)
        h2, c2 = active_grad(bh, n, deadline - 3)
        if c2 < bc:
            bc = c2
            bh = h2
        if deadline - time.time() > 20:
            h2 = slsqp_smooth(bh, n, 20000, 35, deadline - 1)
            c2 = c5_of(h2, n)
            if c2 < bc:
                bc = c2
                bh = h2
        if deadline - time.time() > 6:
            h2, c2 = coordinate_polish(bh, n, deadline - 1)
            if c2 < bc:
                bc = c2
                bh = h2
    bh = project_box_sum(bh, len(bh))
    bc = float(c5_of(bh, len(bh)))
    return bh, bc, int(len(bh))
```
