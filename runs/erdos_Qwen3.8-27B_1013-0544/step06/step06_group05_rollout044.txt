def swap_delta(corr, h, i, j, n):
    """Compute the change in correlation when swapping h[i]:1->0 and h[j]:0->1."""
    # delta_c[k] = h[i-k] - h[j-k] for valid k
    # k ranges from -(n-1) to (n-1)
    # h[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
    # h[j-k] is valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
    
    delta = np.zeros(2*n-1)
    # For h[i-k]: k in [i-n+1, i], index in corr is n-1+k
    k_min1 = max(i - n + 1, -(n-1))
    k_max1 = min(i, n-1)
    for k in range(k_min1, k_max1 + 1):
        delta[n-1+k] -= (1 - h[i-k])

    """Convert to binary with exactly n//2 ones."""
    n_ones = n // 2
    h = np.asarray(h, dtype=float)
    # Use partition for speed
    thresh = np.partition(h, n - n_ones)[n - n_ones]

        zeros = np.where(h == 0)[0]
        h[zeros[:n//2 - n_ones]] = 1.0
    elif n_ones > n // 2:
        ones = np.where(h == 1)[0]
        h[ones[n_ones - n//2:]] = 0.0
    return h

def sa_optimize(h_init, n, deadline, seed=42, T_start=0.03, T_end=0.0005):

    dx = 2.0 / n

    cur_val = cur_int * dx
    
    best_h = cur.copy()
    best_int = cur_int
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    # Precompute for fast delta
    # When we swap i (1->0) and j (0->1):
    # delta at position p in corr (p = n-1+k, k = lag):
    #   if i-k >= 0 and i-k < n: corr[p] -= (1 - cur[i-k])
    #   if j-k >= 0 and j-k < n: corr[p] += (1 - cur[j-k])
    
    iter_count = 0
    no_improve = 0
    T = T_start
    # Estimate total iterations
    t_check = time.time()
    
    while time.time() < deadline:
        if iter_count % 10000 == 0 and iter_count > 0:
            # Adaptive temperature
            if no_improve > 50000:
                T = min(T * 1.5, T_start)  # reheat
                no_improve = 0
            
        # Pick random swap
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        
        # Compute delta
        # For position p in corr array (0 to 2n-2):
        # k = p - (n-1)
        # cur[i-k] valid when 0 <= i-k < n => i-n+1 <= k <= i
        # cur[j-k] valid when 0 <= j-k < n => j-n+1 <= k <= j
        
        # delta[p] = (1-cur[j-k]) - (1-cur[i-k]) where k = p-(n-1)
        # = cur[i-k] - cur[j-k]
        # But only where the corresponding index is valid
        
        # Vectorized computation:
        # For i: k ranges from max(i-n+1, -(n-1)) to min(i, n-1)
        #   p ranges from n-1 + k_min1 to n-1 + k_max1
        #   cur[i-k] = cur[i - (p - (n-1))] = cur[i + n - 1 - p]
        
        k_min_i = max(i - n + 1, -(n-1))
        k_max_i = min(i, n-1)
        p_min_i = n - 1 + k_min_i
        p_max_i = n - 1 + k_max_i
        # indices into cur: i - k = i - (p - n + 1) = i + n - 1 - p
        idx_i = i + n - 1 - np.arange(p_min_i, p_max_i + 1)
        
        k_min_j = max(j - n + 1, -(n-1))
        k_max_j = min(j, n-1)
        p_min_j = n - 1 + k_min_j
        p_max_j = n - 1 + k_max_j
        idx_j = j + n - 1 - np.arange(p_min_j, p_max_j + 1)
        
        new_corr = corr.copy()
        new_corr[p_min_i:p_max_i+1] -= (1.0 - cur[idx_i])
        new_corr[p_min_j:p_max_j+1] += (1.0 - cur[idx_j])
        
        new_int = int(np.max(new_corr))
        new_val = new_int * dx
        delta = new_val - cur_val
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # Accept
            cur[i] = 0.0
            cur[j] = 1.0
            corr = new_corr
            cur_int = new_int
            cur_val = new_val
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            iter_count += 1
            if new_int < best_int:
                best_int = new_int
                best_h = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject (no change to cur)
            iter_count += 1
            no_improve += 1
        
        # Cool down
        if iter_count % 1000 == 0:
            T = max(T * 0.995, T_end)
    
    return best_h, best_int * dx

def make_binary(h, n, seed=0):
    n_ones = n // 2
    h = np.asarray(h, dtype=float)
    thresh = np.partition(h, n - n_ones)[n - n_ones]
    h_bin = (h > thresh).astype(float)
    cnt = int(h_bin.sum())
    rng = np.random.RandomState(seed)
    if cnt < n_ones:
        pool = np.where(h == thresh)[0]
        rng.shuffle(pool)
        h_bin[pool[:n_ones - cnt]] = 1.0
    elif cnt > n_ones:
        pool = np.where(h_bin == 1)[0]
        rng.shuffle(pool)
        h_bin[pool[cnt - n_ones:]] = 0.0
        # if still not enough, take from above thresh
        cnt2 = int(h_bin.sum())
        if cnt2 < n_ones:
            pool2 = np.where(h > thresh)[0]
            # need to flip some
            extra = n_ones - cnt2
            pool2 = np.where(h_bin == 1)[0]
            h_bin[pool2[:extra]] = 0.0
            h_bin[np.where(h_bin == 0)[0][:extra]] = 1.0
    return h_bin
