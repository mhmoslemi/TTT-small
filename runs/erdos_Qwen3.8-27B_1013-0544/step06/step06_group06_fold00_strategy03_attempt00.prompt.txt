def project_feasible(h, target):
    """Ensure sum(h) = target using continuous projection."""
    diff = target - np.sum(h)
    if abs(diff) < 1e-8:
        return h
    if diff > 0:
        # Need to add mass
        room = np.clip(1 - h, 0, 1)
        h += diff * (room / np.sum(room))
    else:
        # Need to remove mass
        mass = np.clip(h, 0, 1)
        h += diff * (mass - np.clip(mass + diff, 0, None)) / np.sum(mass + diff)
    return np.clip(h, 0, 1)

    """Penalized objective function for optimization."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx, c
