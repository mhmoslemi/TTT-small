def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = np.minimum(diff / tot, 1.0)
            h += frac * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, c, n, p, grid_size=200):
    """Try to improve coordinate p. Returns (new_h_p, new_c_segment, ks, improved)."""
    d_lo = -h[p]
    d_hi = 1.0 - h[p]
    if d_hi - d_lo < 1e-14:
        return h[p], None, None, False
    
    total_len = 2 * n - 1
    k_lo = max(0, n - 1 - p)
    k_hi = min(total_len - 1, 2 * n - 2 - p)
    
    um = 0.0
    if k_lo > 0:
        um = max(um, float(np.max(c[:k_lo])))
    if k_hi < total_len - 1:
        um = max(um, float(np.max(c[k_hi + 1:])))
    
    current_max = float(np.max(c))
    if current_max <= um + 1e-15:

    c_aff = c[ks]
    has_self = (n - 1) >= k_lo and (n - 1) <= k_hi

import numpy as np

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_update(h, c, n, p, grid_size=200):

    if d_hi - d_lo < 1e-14:
        return None, None, None

    k_hi = min(total_len - 1, 2 * n - 2 - p)
    um = 0.0

    if k_hi < total_len - 1:
        um = max(um, float(np.max(c[k_hi + 1:])))
    if float(np.max(c)) <= um + 1e-15:

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    if mx > 500:
        lse = mx + float(np.log(np.sum(np.exp(shifted - mx))))
    else:
        lse = float(np.log(np.sum(np.exp(shifted))))
    return (m + lse / alpha) * dx

    return x.sum() - n / 2.0

    return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
