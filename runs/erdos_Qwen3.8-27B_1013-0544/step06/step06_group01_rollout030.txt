        if abs(diff) < 1e-14:

            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    if alpha < 1e6:
        s = alpha * (c - cm)
        s = np.clip(s, -500, 500)
        val = cm + np.log(np.sum(np.exp(s))) / alpha
    else:
        val = cm
    return float(val * dx)

    exp_s = np.exp(s)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    s = alpha * (c - cm) * dx
    s = np.clip(s, -500, 500)
    return float((cm + np.log(np.sum(np.exp(s))) / alpha) * dx)

def sm_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    s = alpha * (c - cm) * dx
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = (cm + np.log(sum_exp) / alpha) * dx
    term1 = np.correlate(1.0 - x, w, mode='full')[n-1:n-1+n]
    term2 = np.correlate(x, w[::-1], mode='full')[n-1:n-1+n]
    grad = dx * (term1 - term2)
    return obj, grad

def _con_fun(x, n):
    return x.sum() - n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    s = alpha * (c - cm) * dx
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = (cm + np.log(sum_exp) / alpha) * dx
    term1 = np.correlate(1.0 - x, w, mode='full')[n-1:n-1+n]
    term2 = np.correlate(x, w[::-1], mode='full')[n-1:n-1+n]
    grad = dx * (term1 - term2)
    return obj, grad

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp(h0, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj_grad, h0, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def anneal_slsqp(h0, n, deadline):
    h = h0.copy()
    alphas = [3000, 10000, 40000, 150000, 600000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        remaining = deadline - time.time() - 5
        maxiter = max(200, int(remaining * 3))
        h_new = slsqp(h, n, alpha, maxiter=maxiter)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new <= c_old:
            h = h_new
    return h

def coord_descent(h, n, max_passes=80, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            m_min = min(-p, p - n + 1)
            m_max = max(n - 1 - p, p)
            k_min = max(0, n - 1 + m_min)
            k_max = min(2*n-2, n - 1 + m_max)
            um = 0.0
            if k_min > 0:
                um = max(um, float(np.max(c[:k_min])))
            if k_max < 2*n-2:
                um = max(um, float(np.max(c[k_max+1:])))
            ks = np.arange(k_min, k_max+1)
            ms = ks - (n-1)
            pm = p + ms
            valid_pm = (pm >= 0) & (pm < n)
            pm2 = p - ms
            valid_pm2 = (pm2 >= 0) & (pm2 < n)
            s_fwd = np.zeros(len(ks))
            s_fwd[valid_pm] = 1.0 - h[pm[valid_pm]]
            s_bwd = np.zeros(len(ks))
            s_bwd[valid_pm2] = h[pm2[valid_pm2]]
            s_vec = s_fwd - s_bwd
            scale = min(d_hi - d_lo, 0.03)
            deltas = np.linspace(-scale, scale, 13)
            deltas = np.clip(deltas, d_lo, d_hi)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-13:
                    continue
                new_vals = c[ks] + d * s_vec
                idx_c = (ks == n-1)
                if np.any(idx_c):
                    new_vals[idx_c] = c[n-1] + d*(1-2*h[p]) - d*d
                val = max(um, float(np.max(new_vals)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-11:
                d = best_d
                c[ks] += d * s_vec
                c[n-1] -= d*d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[k_min:k_max+1])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True; improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True; improved_any = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
