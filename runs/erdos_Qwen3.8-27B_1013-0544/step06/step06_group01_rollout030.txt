```python
                      bounds=bounds, method='highs')

                    best_c = c_try
                    break
    
    return h, best_c
```

- Find which k achieves the max at t=0 and t=1

- Otherwise, find the breakpoint between the two relevant lines

3. If f(1) ≤ f(0) and the slope at 1 is negative, min is at 1
4. Otherwise, do a bisection/binary search for where the subgradient changes sign

```python
    idx2 = p + ms[valid2]

    # m < 0 part: k = n-1+m for m = -(n-1), ..., -1
    mps = np.arange(1, n)  # |m| = 1, 2, ..., n-1
    ks_neg = n-1 - mps  # k indices
    # Condition: p+|m| < n
    valid3 = (p + mps) < n
    idx3 = p + mps[valid3]
    a[ks_neg[valid3]] += 1.0 - h[idx3]
    # Condition: p >= |m|
    valid4 = (p >= mps)
    idx4 = p - mps[valid4]
    a[ks_neg[valid4]] -= h[idx4]
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.minimum(np.maximum(h.copy(), 0.0), 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h = np.minimum(h + (diff / tot) * room, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h = np.maximum(h * (1.0 + diff / tot), 0.0)
    return np.minimum(np.maximum(h, 0.0), 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    m = ac.max()
    e = np.exp(ac - m)
    S = e.sum()
    w = e / S
    obj = float((m + np.log(S)) / alpha * dx + 500.0 * (h.sum() - n / 2.0) ** 2)
    grad = np.zeros(n)
    nc = 2 * n - 1
    a = np.zeros(nc)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-12:
            continue
        ml = k - (n - 1)
        if ml >= 0:
            a[ml:nc - n + ml + 1] = 0.0
            js = ml
            a[js:js + n - ml] = 1.0 - h[:n - ml]
            if ml > 0:
                a[:n - ml] = a[:n - ml] - h[ml: n]
                a[n - ml:] = 0.0
            else:
                a[:n] = a[:n] - h[:n]
        else:
            mp = -ml
            a[:n - mp] = a[:n - mp] + (1.0 - h[mp:n])
            a[mp:] = a[mp:] - h[:n - mp]
        grad += wk * a
        a = np.zeros(nc)
    grad *= dx
    grad += 1000.0 * (h.sum() - n / 2.0) * np.ones(n)
    return obj, grad

def lbfgs_opt(h, n, alphas, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def _obj(x):
            xc = project_feasible(x, n)
            o, _ = smooth_obj_grad(xc, n, alpha)
            return o
        def _grad(x):
            xc = project_feasible(x, n)
            _, g = smooth_obj_grad(xc, n, alpha)
            return g
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                           bounds=[(0, 1)] * n,
                           options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
        except:
            h_new = project_feasible(h, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        h = h_new
    return best_h, best_c

def compute_a_vec(h, p, n):
    a = np.zeros(2 * n - 1)
    ms = np.arange(n)
    v1 = ms <= p
    a[n - 1 + ms[v1]] += 1.0 - h[p - ms[v1]]
    v2 = (ms > 0) & (p + ms < n)
    a[n - 1 + ms[v2]] -= h[p + ms[v2]]
    mps = np.arange(1, n)
    kn = n - 1 - mps
    v3 = (p + mps) < n
    a[kn[v3]] += 1.0 - h[p + mps[v3]]
    v4 = p >= mps
    a[kn[v4]] -= h[p - mps[v4]]
    return a

                nc = c.max() * dx
                if nc + 50.0 * (h.sum() - n / 2.0) ** 2 < best_c:
                    best_c = nc
                    improved = True
                else:
                    h[p] = h[p]  # revert handled by recompute
                    h[p] = grid[bev] if False else h[p]
                    # revert
                    h[p] = h[p]
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def active_lp(h, n, deadline, max_iter=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        tol = max(0.002, 0.05 * cmax / float(n))
        active = np.where(c >= cmax - tol)[0]
        if len(active) > 30:
            vals = c[active]
            active = active[np.argsort(-vals)[:30]]
        A_ub = []
        b_ub = []
        for k in active:
            ml = k - (n - 1)
            gk = np.zeros(n)
            if ml >= 0:
                gk[ml:] = 1.0 - h[:n - ml]
                if ml > 0:
                    gk[:n - ml] -= h[ml:]
                else:
                    gk -= h
            else:
                mp = -ml
                gk[:n - mp] += 1.0 - h[mp:]
                gk[mp:] -= h[:n - mp]
            row = np.concatenate([gk, [-1.0]])
            A_ub.append(row)
            b_ub.append(c[k] - gk @ h)
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(-1, 1)]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        try:
            res = linprog(c_obj, A_ub, b_ub, A_eq, b_eq, bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except:
            break
        if not res.success:
            break
        h_new = res.x[:n]
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
        elif c_new < best_c:
            for alpha_s in [0.5, 0.25, 0.1]:
                hs = project_feasible(h + alpha_s * (h_new - h), n)
                cs = compute_c5(hs, n)
                if cs < best_c:
                    h, best_c = hs, cs
                    break
        else:
            break
    return h, best_c

def pair_search(h, n, deadline, steps, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= step and h[j] <= 1.0 - step:
                        h[i] -= step
                        h[j] += step
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    if h[j] >= step and h[i] <= 1.0 - step:
                        h[j] -= step
                        h[i] += step
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def optimize_full(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase 1: L-BFGS-B
    if time.time() < deadline - 60:
        h, c = lbfgs_opt(h, n, [100, 300, 1000], deadline=time.time() + min(120, (deadline - time.time()) * 0.3))
        if c < best_c:
            best_c, h = c, h
    # Phase 2: Smart coordinate descent
    if time.time() < deadline - 40:
        h2, c2 = smart_cd(h, n, deadline=min(time.time() + 80, deadline - 40), grid_size=80)
        if c2 < best_c:
            best_c, h = c2, h2
    # Phase 3: Active-set LP
    if time.time() < deadline - 30:
        h3, c3 = active_lp(h, n, deadline=min(time.time() + 60, deadline - 30))
        if c3 < best_c:
            best_c, h = c3, h3
    # Phase 4: More coordinate descent
    if time.time() < deadline - 20:
        h4, c4 = smart_cd(h, n, deadline=time.time() + 40, grid_size=120)
        if c4 < best_c:
            best_c, h = c4, h4
    # Phase 5: Pair search
    if time.time() < deadline - 5:
        h5, c5 = pair_search(h, n, deadline - 2, [0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=40)
        if c5 < best_c:
            best_c, h = c5, h5
    # Phase 6: Final LP + pair
    if time.time() < deadline - 3:
        h6, c6 = active_lp(h, n, deadline=time.time() + 20, max_iter=30)
        if c6 < best_c:
            best_c, h = c6, h6
    if time.time() < deadline - 1:
        h7, c7 = pair_search(h, n, deadline - 0.5, [0.0001, 0.00005], radius=30)
        if c7 < best_c:
            best_c, h = c7, h7
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except:
        h_init = np.full(200, 0.5)
    n_init = len(h_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Optimize at initial n
    alloc1 = min(500, (deadline - t0) * 0.45)
    h1, c1 = optimize_full(best_h, n_init, t0 + alloc1)
    if c1 < best_c:
        best_c, best_h = c1, h1
    # Try other n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in [200, 150, 250]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_init, best_h), n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 50:
            break
        h_t, c_t = optimize_full(h_interp, n_try, time.time() + alloc)
        if c_t < best_c:
            best_c, best_h = c_t, h_t
    # Multi-restart from best
    n = len(best_h)
    if time.time() < deadline - 80:
        n_restarts = max(2, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 + 0.03 * r
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(50, (deadline - time.time() - 20) * 0.5)
            if alloc < 30:
                break
            h_r, c_r = optimize_full(hp, n, time.time() + alloc)
            if c_r < best_c:
                best_c, best_h = c_r, h_r
    # Final polish
    if time.time() < deadline - 10:
        h_f, c_f = active_lp(best_h, n, deadline - 15, max_iter=50)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 2:
        h_p, c_p = pair_search(best_h, n, deadline - 1, [0.0001, 0.00005, 0.00002], radius=40)
        if c_p < best_c:
            best_c, best_h = c_p, h_p
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
