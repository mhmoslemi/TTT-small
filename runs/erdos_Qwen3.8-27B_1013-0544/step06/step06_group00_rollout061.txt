def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room

            s = h.sum()
            if s < 1e-14:
                break

    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_grad_fn(n):
    # Returns a function that computes objective and gradient
    # Precompute index arrays
    jj = np.arange(n)[:, None]
    dd = np.arange(-(n-1), n)[None, :]
    idx_plus = jj + dd
    idx_minus = jj - dd
    vp = (idx_plus >= 0) & (idx_plus < n)
    vm = (idx_minus >= 0) & (idx_minus < n)
    
    def obj(x, alpha):
        dx = 2.0 / n
        c_raw = np.correlate(x, 1.0 - x, mode='full')
        C = c_raw * dx
        M = C.max()
        shift = alpha * (C - M)
        shift = shift - shift.max()  # numerical stability
        exp_sum = np.exp(shift).sum()
        w = np.exp(shift) / exp_sum
        val = M + np.log(exp_sum) / alpha

        clipped_minus = np.clip(idx_minus, 0, n-1)

        h_minus = np.where(vm, x[clipped_minus], 0.0)
        grad = dx * ((w[None, :] * (1.0 - h_plus)).sum(axis=1) - (w[None, :] * h_minus).sum(axis=1))
        return val, grad

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return project_feasible(res.x, n)

        return h

                delta[n-1] -= 2.0

                nc = (corr + delta).max() * dx
