import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.permutation(n)
    b[idx[:n//2]] = 1.0
    return b

def local_search_binary(h, n, deadline, max_iters=200000, seed=42, first_improve=True):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(new_corr.max())
        if new_max < cur_max:
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_max * dx

def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        # Soft max
        shifted = alpha * (g - m)
        smax = shifted.max()
        return m + (smax + np.log(np.exp(shifted - smax).sum())) / alpha
    def obj_grad(x):
        # Numerical gradient (expensive but works)
        h_grad = 1e-7
        g = np.zeros(n)
        for i in range(n):
            xp = x.copy()
            xp[i] += h_grad
            g[i] = obj(xp)
            xm = x.copy()
            xm[i] -= h_grad
            g[i] = (g[i] - obj(xm)) / (2 * h_grad)
            if time.time() > deadline - 5:
                break
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
