    c = np.correlate(h, 1.0 - h, mode='full')
    fc = c * dx
    sc = alpha * (fc - fc.max())
    p = np.exp(sc)
    p_sum = p.sum()
    p /= p_sum
    obj = fc.max() + np.log(p_sum) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-12:
            continue
        k = m - (n - 1)
        # Role 1: j = i+k, contribution (1-h[j-k])
        j_lo = max(k, 0)
        j_hi = min(n - 1, n - 1 + k)  # j <= n-1 and j-k <= n-1-k means j <= n-1
        # Actually: i ranges from max(0,-k) to min(n-1, n-1-k)
        # j = i+k ranges from max(k, 0) to min(n-1+k, n-1)
        j_lo1 = max(k, 0)
        j_hi1 = min(n - 1 + k, n - 1)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            grad[js] += p[m] * (1.0 - h[js - k])
        # Role 2: j = i, contribution -h[j+k]
        # i ranges from max(0,-k) to min(n-1, n-1-k)
        j_lo2 = max(0, -k)
        j_hi2 = min(n - 1, n - 1 - k)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            grad[js] += p[m] * (-h[js + k])
    
    grad *= dx
    return obj, grad
