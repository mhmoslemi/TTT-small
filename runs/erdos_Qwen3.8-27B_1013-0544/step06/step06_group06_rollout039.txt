    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)

        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        p_lo = max(0, k - n + 1)
        p_hi = min(n - 1, k)
        if p_lo > p_hi:
            continue
        idx = np.arange(p_lo, p_hi + 1)
        q = k - idx
        g[idx, k] = 1.0 - 2.0 * h[q]
        if k % 2 == 0:
            dp = k // 2
            if p_lo <= dp <= p_hi:
                g[dp, k] = 0.0
    return g

def do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj):
    g_i = g[i].copy()
    g_j = g[j].copy()
    corr = corr + g_i - g_j
    p_idx = np.arange(n)
    mi = (p_idx != i) & (p_idx + i < 2 * n - 1)
    g[p_idx[mi], p_idx[mi] + i] -= 2.0
    mj = (p_idx != j) & (p_idx + j < 2 * n - 1)
    g[p_idx[mj], p_idx[mj] + j] += 2.0
    h[i], h[j] = 1.0, 0.0
    zeros = np.delete(zeros, ii)
    ones = np.delete(ones, jj)
    zeros = np.append(zeros, j)
    ones = np.append(ones, i)
    return g, h, corr, zeros, ones

def binary_search(h_start, n, deadline, top_k=15, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    if np.all((h_start == 0) | (h_start == 1)) and int(h_start.sum()) == target:
        h = h_start.astype(np.float64).copy()
    else:
        idx = np.argsort(h_start)[::-1]
        h = np.zeros(n)
        h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    temp = 0.004
    no_improve = 0
    last_kick = 0

    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        g_act = g[:, top_idx]
        M = corr[top_idx][None, :] + g_act[zeros]
        D = M[:, None, :] - g_act[ones][None, :, :]
        flat = np.argmin(D)
        ii = flat // len(ones)
        jj = flat % len(ones)
        i = zeros[ii]
        j = ones[jj]
        new_corr = corr + g[i] - g[j]
        new_c = new_corr.max() * dx

        if new_c < cur_c - 1e-14:
            g, h, corr, zeros, ones = do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj)
            cur_c = new_c
            stall = 0
            temp = max(0.0005, temp * 0.9999)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        elif rng.rand() < np.exp(-(new_c - cur_c) / max(temp, 0.001)):
            g, h, corr, zeros, ones = do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj)
            cur_c = new_c
            stall += 1
            no_improve += 1
        else:
            stall += 1
            no_improve += 1

        if stall > 40 and time.time() < deadline - 5 and no_improve > 30:
            nk = min(2, len(zeros), len(ones))
            ri = rng.choice(len(zeros), nk, replace=False)
            rj = rng.choice(len(ones), nk, replace=False)
            g_delta = np.zeros(2 * n - 1)
            for a in range(nk):
                g_delta += g[zeros[ri[a]]] - g[ones[rj[a]]]
            new_corr_k = corr + g_delta
            new_c_k = new_corr_k.max() * dx
            if np.random.rand() < np.exp(-(new_c_k - cur_c) / max(temp * 3, 0.003)):
                for a in range(nk):
                    ii2 = np.where(zeros == zeros[ri[a]])[0]
                    jj2 = np.where(ones == ones[rj[a]])[0]
                    if len(ii2) > 0 and len(jj2) > 0:
                        g, h, corr, zeros, ones = do_swap(
                            g, h, corr, zeros[ri[a]], ones[rj[a]],
                            n, zeros, ones, ii2[0], jj2[0])
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                g = build_g(h, n)
                cur_c = corr.max() * dx
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
            temp = max(0.0005, temp * 0.92)
            stall = 0
            no_improve = 0

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
