def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sobj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sgrad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def adaptive_lbfgs(h, n, deadline, rng, max_total_iters=3000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    mus = [1.0, 2.0, 5.0, 10.0, 50.0, 100.0, 500.0, 2000.0, 5000.0, 10000.0]
    stall_count = 0
    for idx in range(len(alphas)):
        if time.time() > deadline - 3:
            break
        alpha = alphas[idx]
        mu = mus[idx]
        remaining = deadline - time.time() - 3
        max_it = max(20, min(400, int(remaining * 1.5)))
        try:
            res = minimize(sobj, h, jac=sgrad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_it, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_it * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                stall_count = 0
            else:
                stall_count += 1
                if stall_count >= 2:
                    h = project_feasible(h + rng.normal(0, 0.005, n), n)
                    stall_count = 0
                else:
                    h = project_feasible(res.x, n)
        except Exception:
            h = project_feasible(h, n)
    return best_h, best_c

def pair_search(h, n, deadline, steps, radius=50):
    best_c = compute_c5(h, n)
    for step in steps:
        any_imp = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            any_imp = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            any_imp = True
                        else:
                            h[i] += step
                            h[j] -= step
            h = project_feasible(h, n)
            if not improved:
                break
        if not any_imp:
            continue
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Initial adaptive L-BFGS
    t1 = t0 + 280
    best_h, best_c = adaptive_lbfgs(best_h, n, t1, rng)

    # Phase 2: Even higher alpha refinement
    if time.time() < deadline - 200:
        t2 = min(time.time() + 200, deadline - 200)
        h2 = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        for alpha, mu in [(5000000, 5000.0), (20000000, 10000.0), (100000000, 20000.0)]:
            if time.time() > t2 - 5:
                break
            try:
                res = minimize(sobj, h2, jac=sgrad, args=(n, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 200, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 4000})
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2, n)
                if c2 < best_c - 1e-15:
                    best_c = c2
                    best_h = h2.copy()
            except Exception:
                pass

    # Phase 3: Pair search with decreasing steps
    if time.time() < deadline - 80:
        t3 = min(time.time() + 250, deadline - 80)
        best_h, best_c = pair_search(best_h, n, t3,
            [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5],
            radius=40)

    # Phase 4: Try symmetry complement (1-h)
    if time.time() < deadline - 120:
        h_sym = project_feasible(1.0 - best_h, n)
        c_sym = compute_c5(h_sym, n)
        if c_sym < best_c:
            best_c = c_sym
            best_h = h_sym.copy()
            if time.time() < deadline - 100:
                best_h, best_c = adaptive_lbfgs(best_h, n, deadline - 100, rng, max_total_iters=1000)
            if time.time() < deadline - 50:
                best_h, best_c = pair_search(best_h, n, deadline - 30,
                    [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5], radius=40)

    # Phase 5: Try different n values
    if time.time() < deadline - 60:
        for n_try in [250, 200, 350]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 25)
            if alloc < 20:
                break
            h_t, c_t = adaptive_lbfgs(h_interp, n_try, time.time() + alloc * 0.7, rng, max_total_iters=500)
            if time.time() < deadline - 20:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 15, deadline - 20),
                    [0.01, 0.005, 0.002, 0.001, 0.0005], radius=30)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        final_steps = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        remaining = deadline - time.time() - 3
        best_h, best_c = pair_search(best_h, n, deadline - 2, final_steps, radius=30)

    # Extra L-BFGS if time permits
    if time.time() < deadline - 10:
        h_final = best_h.copy()
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(sobj, h_final, jac=sgrad, args=(n, 50000000, 10000.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14})
            h_final = project_feasible(res.x, n)
            c_f = compute_c5(h_final, n)
            if c_f < best_c:
                best_c = c_f
                best_h = h_final
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
