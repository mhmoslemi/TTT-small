    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def _smooth_obj(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def exact_pair_search(h, n, deadline, max_passes=10):
    """Exact line search pair optimization."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try pairs based on gradient
        g = grad_smooth_max(h, n, 1000.0)
        # Sort indices by gradient value
        order = np.argsort(g)
        # Try moving from high gradient to low gradient
        for ai in range(min(50, n)):
            for aj in range(n - 1 - min(50, n), n):
                if time.time() > deadline:
                    return h, best_c
                i = order[aj]  # high gradient (want to decrease)
                j = order[ai]  # low gradient (want to increase)
                if i == j:
                    continue
                
                # Compute direction: d[m] = partial c_m / partial h[i] - partial c_m / partial h[j]
                # For speed, compute this vector
                d = np.zeros(2 * n - 1)
                for m in range(2 * n - 1):
                    k = m - (n - 1)
                    term_i = 0.0
                    if 0 <= i + k < n:
                        term_i += (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        term_i -= h[i - k]
                    term_j = 0.0
                    if 0 <= j + k < n:
                        term_j += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        term_j -= h[j - k]
                    d[m] = dx * (term_i - term_j)
                
                # Feasible range for delta (h[i] += delta, h[j] -= delta)
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_lo >= delta_hi - 1e-15:
                    continue
                
                # Minimize f(delta) = max_m (c_vec[m] + delta * d[m])
                # Convex PL function, use ternary search
                a, b = delta_lo, delta_hi
                for _ in range(50):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = np.max(c_vec + m1 * d)
                    f2 = np.max(c_vec + m2 * d)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                best_delta = (a + b) / 2.0
                
                # Evaluate at candidate points
                candidates = [best_delta, 0.0]
                if delta_lo > -1e-15:
                    candidates.append(0.0)
                for dc in candidates:
                    dc = max(delta_lo, min(delta_hi, dc))
                    f_val = np.max(c_vec + dc * d)
                    if f_val < best_c - 1e-15:
                        h[i] += dc
                        h[j] -= dc
                        h = np.clip(h, 0, 1)
                        # Re-normalize
                        h = project_feasible(h, n)
                        # Recompute c_vec
                        c_vec = compute_c5_vec(h, n)
                        best_c = compute_c5(h, n)
                        improved = True
                        break
        
        if not improved:
            break
    return h, best_c
