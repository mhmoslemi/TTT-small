import numpy as np

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    target = 0.5 * float(n)
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = np.clip(y - tau, 0.0, 1.0)
    for _ in range(20):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    return project_box_sum(np.clip(np.asarray(h, dtype=float), 0.0, 1.0), n)

def build_e(h, n):
    z = n - 1
    e = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        r = np.arange(p, p + n)
        e[p, r] += 1.0 - h[p + z - r]
        r = np.arange(z - p, z + n - p)
        e[p, r] -= h[p + r - z]
    return e

def apply_single(h, c, e, n, p, delta):
    if abs(delta) < 1e-16:
        return
    z = n - 1
    c += delta * e[p]
    c[z] -= delta * delta
    h[p] += delta
    if h[p] < 0.0:
        h[p] = 0.0
    elif h[p] > 1.0:
        h[p] = 1.0
    q = np.arange(n)
    r1 = z + q - p
    r2 = z + p - q
    e[:, r1] -= delta
    e[:, r2] -= delta

def try_pair_update(h, c, e, n, a, b, frac, delta, tol, cur, z):
    tlo = -h[b]
    thi = 1.0 - h[a]
    if tlo < -delta:

def process_pair_pass(h, c, e, n, mode, step, frac, delta, deadline, rng, z, tol):

    if mode == 'adj':

def pair_descent(h, n, deadline, grid=36, max_delta=0.25, rng=None, polish=False):

def make_mats(n):
    q = np.arange(n)
    idx = q[:, None] - q[None, :] + (n - 1)
    idxr = 2 * (n - 1) - idx
    return idx, idxr

def smooth_value_c(c, alpha, n):
    m = float(c.max())
    z = alpha * (c - m)
    ez = np.exp(z)
    return float((m + np.log(float(ez.sum())) / alpha) * (2.0 / float(n)))

def smooth_fg(x, n, alpha, mats):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    ez = np.exp(z)
    w = ez / ez.sum()
    idx, idxr = mats
    dx = 2.0 / float(n)
    A = ((1.0 - x)[None, :] * w[idx]).sum(axis=1)
    B = (x[None, :] * w[idxr]).sum(axis=1)
    f = float((m + np.log(float(ez.sum())) / alpha) * dx)
    g = (A - B) * dx
    return f, g

    if pair_dl > time.time() + 6.0:

        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if use_gradient and time.time() < deadline - 20.0:
        rem = deadline - time.time()
        if rem > 25.0:
            for a in (30.0, 150.0, 700.0):
                if time.time() > deadline - 15.0:
                    break
                rem = deadline - time.time()
                if rem < 15.0:
                    break
                stage_dl = min(deadline - 8.0, time.time() + max(8.0, 0.22 * rem))
                h2, c2 = gradient_polish(best_h, n, stage_dl, alpha=a, step=0.015, rng=rng)
                if c2 < best_c:
                    best_h, best_c = h2, c2
    if time.time() < deadline - 10.0:
        rem = deadline - time.time()
        pair_dl = min(deadline - 8.0, time.time() + max(6.0, min(60.0, 0.35 * rem)))
        if pair_dl > time.time() + 4.0:
            h2, _ = pair_descent(best_h, n, deadline=pair_dl, grid=48, max_delta=0.03, rng=rng, polish=True)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
    if use_gradient and time.time() < deadline - 12.0:
        rem = deadline - time.time()
        if rem > 10.0:
            stage_dl = min(deadline - 4.0, time.time() + max(5.0, min(35.0, 0.4 * rem)))
            h2, c2 = gradient_polish(best_h, n, stage_dl, alpha=1800.0, step=0.01, rng=rng)
            if c2 < best_c:
                best_h, best_c = h2, c2
    if time.time() < deadline - 4.0:
        rem = deadline - time.time()
        pair_dl = min(deadline - 2.0, time.time() + max(4.0, min(25.0, 0.7 * rem)))
        if pair_dl > time.time() + 3.0:
            h2, _ = pair_descent(best_h, n, deadline=pair_dl, grid=64, max_delta=0.002, rng=rng, polish=True)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interp_to_n(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    xold = np.linspace(0.0, 2.0, old_n)
    xnew = np.linspace(0.0, 2.0, new_n)
    return np.interp(xnew, xold, h)

def pattern_start(n, kind, rng):
    x = np.linspace(0.0, 2.0, n)
    k = int(kind) % 6
    if k == 0:
        h = 0.5 + 0.45 * np.sin(0.5 * np.pi * x)
    elif k == 1:
        h = 0.5 + 0.45 * np.sin(np.pi * x + 0.7)
    elif k == 2:
        h = 0.5 + 0.35 * np.sin(1.5 * np.pi * x) + 0.15 * np.cos(0.5 * np.pi * x)
    elif k == 3:
        h = 0.5 + 0.4 * np.cos(0.5 * np.pi * x)
    elif k == 4:
        h = 0.5 + 0.12 * rng.normal(size=n)
    else:
        h = np.zeros(n)
        total = 0.5 * float(n)
        lengths = rng.dirichlet(np.ones(4)) * total
        starts = rng.uniform(0.0, float(n) - 1.0, 4)
        for s, l in zip(starts, lengths):
            i0 = int(max(0.0, s))
            i1 = int(min(float(n), s + l))
            h[i0:i1] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    total = deadline - t0
    main_alloc = min(total, max(10.0, 0.55 * total))
    main_alloc = min(560.0, main_alloc)
    dl = t0 + main_alloc
    h2, c2 = optimize_once(h_init, n_init, rng, dl, use_gradient=(n_init <= 450))
    if c2 < best_c:
        best_h, best_c = h2, c2

    tried = set()
    tried.add(len(best_h))
    nb = len(best_h)
    alt_ns = (nb + 1, nb - 1, 250, 350, 400, 200, 300, 450, 150, 120)
    for n_alt in alt_ns:
        if time.time() > deadline - 110.0:
            break
        if n_alt in tried or n_alt < 80 or n_alt > 600:
            continue
        rem = deadline - time.time()
        alloc = min(80.0, max(20.0, 0.18 * rem))
        if alloc < 20.0:
            break
        h_start = project_feasible(interp_to_n(best_h, n_alt), n_alt)
        h2, c2 = optimize_once(h_start, n_alt, rng, time.time() + alloc, use_gradient=(n_alt <= 350))
        if c2 < best_c:
            best_h, best_c = h2, c2
            tried.add(n_alt)

    n_pat = len(best_h)
    if n_pat < 120 or n_pat > 500:
        n_pat = 300
    pat = 0
    while time.time() < deadline - 160.0 and pat < 6:
        rem = deadline - time.time()
        alloc = min(35.0, max(15.0, 0.10 * rem))
        if alloc < 15.0:
            break
        h_start = pattern_start(n_pat, pat, rng)
        h2, c2 = optimize_once(h_start, n_pat, rng, time.time() + alloc,
                               use_gradient=(n_pat <= 350 and alloc > 30.0))
        if c2 < best_c:
            best_h, best_c = h2, c2
        pat += 1

    rs = 0
    while time.time() < deadline - 130.0 and rs < 4:
        rem = deadline - time.time()
        alloc = min(30.0, max(15.0, 0.09 * rem))
        if alloc < 15.0:
            break
        scale = 0.012 + 0.03 * rs
        n_cur = len(best_h)
        h_start = project_feasible(best_h + scale * rng.normal(size=n_cur), n_cur)
        h2, c2 = optimize_once(h_start, n_cur, rng, time.time() + alloc,
                               use_gradient=(n_cur <= 350 and alloc > 30.0))
        if c2 < best_c:
            best_h, best_c = h2, c2
        rs += 1

    if time.time() < deadline - 12.0:
        h2, c2 = optimize_once(best_h, len(best_h), rng, deadline, use_gradient=(len(best_h) <= 400))
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
