import numpy as np
import time

def c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def project_box_sum(y, t):
    y = np.asarray(y, dtype=float).copy()
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if float(np.clip(y + m, 0.0, 1.0).sum()) < t:
            lo = m
        else:
            hi = m
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        d = t - float(x.sum())
        if abs(d) < 1e-13:
            break
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        cnt = int(mask.sum())
        if cnt:
            x[mask] += d / cnt
        else:
            x += d / x.size
    return np.clip(x, 0.0, 1.0)

def project_sym(h, t):
    h = 0.5 * (h + h[::-1])
    x = project_box_sum(h, t)
    x = 0.5 * (x + x[::-1])
    return project_box_sum(x, t)

def interp_n(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, m), h)

def smooth_eval(h, alpha):
    n = len(h)
    v = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
    m = float(v.max())
    e = np.exp(alpha * (v - m))
    return float(m + np.log(e.sum()) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, 'full')
    v = cr * dx
    m = float(v.max())
    e = np.exp(alpha * (v - m))
    w = e / e.sum()
    g = (1.0 - 2.0 * h) * w[n - 1]
    for s in range(1, n):
        wp = w[n - 1 + s]
        wn = w[n - 1 - s]
        if wp + wn <= 1e-15:
            continue
        u = h[:-s]
        x = h[s:]
        g[:-s] += wn * (1.0 - x) - wp * x
        g[s:] += wp * (1.0 - u) - wn * u
    g *= dx
    return float(m + np.log(e.sum()) / alpha), g

def pg_optimize(h, n, deadline, sym=False, alphas=None, max_iter=400, step0=0.03):
    if alphas is None:
        alphas = (20, 50, 100, 200, 500, 1000)
    t = n / 2.0
    h = project_box_sum(h, t)
    if sym:
        h = project_sym(h, t)
    bh = h.copy()
    bc = c5(h)
    st = step0
    for al in alphas:
        if time.time() > deadline - 8:
            break
        st = max(st, 0.4 * step0)
        stuck = 0
        for _ in range(max_iter):
            if time.time() > deadline - 8 or st < 1e-5:
                break
            f, g = smooth_grad(h, al)
            if sym:
                g = 0.5 * (g + g[::-1])
            gn = float(np.linalg.norm(g))
            if gn < 1e-10:
                break
            d = g / gn
            tt = st
            ok = False
            for _ in range(4):
                y = h - tt * d
                if sym:
                    y = 0.5 * (y + y[::-1])
                x = project_box_sum(y, t)
                if sym:
                    x = project_sym(x, t)
                if smooth_eval(x, al) < f - 1e-12:
                    h = x
                    cc = c5(h)
                    if cc < bc:
                        bc = cc
                        bh = h.copy()
                    ok = True
                    break
                tt *= 0.5
            if ok:
                if tt < 0.5 * st:
                    st = max(1.2 * tt, 1e-5)
                else:
                    st = min(0.35, 1.08 * st)
                stuck = 0
            else:
                st *= 0.5
                stuck += 1
                if stuck > 6:
                    break
    return bh, bc

def sym_c(h):
    n = len(h)
    c = np.empty(n)
    s = float(h.sum())
    c[0] = s - float(np.dot(h, h))
    p = np.concatenate(([0.0], np.cumsum(h)))
    for k in range(1, n):
        c[k] = s - p[k] - float(np.dot(h[k:], h[:-k]))
    return c

def sym_coeff(h, q):
    n = len(h)
    a = np.zeros(n)
    for x in (q, n - 1 - q):
        if x > 0:
            a[1:x + 1] += 1.0 - h[x - 1::-1]
        r = n - x - 1
        if r > 0:
            a[1:r + 1] -= h[x + 1:x + r + 1]
    return a

def sym_cd(h, deadline, grid=20, lam=10.0, max_passes=90, seed=0):
    n = len(h)
    m = n // 2
    t = n / 2.0
    dx = 2.0 / n
    h = project_sym(h, t)
    c = sym_c(h)
    bh = h.copy()
    bc = float(c.max()) * dx
    curr = float(c.max())
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        imp = False
        for i in rng.permutation(m):
            v = h[i]
            a = sym_coeff(h, i)
            d = np.linspace(-v, 1.0 - v, grid)
            d = np.concatenate((d, [0.0]))
            new0 = c[0] + 2.0 * d * (1.0 - 2.0 * v) - 2.0 * d * d
            vals = np.maximum(np.max(c[1:][None, :] + d[:, None] * a[1:][None, :], axis=1), new0)
            vals += lam * (2.0 * d) ** 2
            bi = int(np.argmin(vals))
            step = float(d[bi])
            if vals[bi] < curr - 1e-12 and abs(step) > 1e-13:
                h[i] += step
                h[n - 1 - i] += step
                c[1:] += step * a[1:]
                c[0] = new0[bi]
                curr = float(c.max())
                bc = min(bc, curr * dx)
                imp = True
        if imp:
            h = project_sym(h, t)
            c = sym_c(h)
            curr = float(c.max())
            hc = c5(h)
            if hc < bc:
                bc = hc
                bh = h.copy()
        else:
            break
    return bh, bc

def sym_pair_cd(h, deadline, grid=16, max_passes=40, seed=1):

            curr = float(c.max())

def full_cd(h, deadline, grid=12, lam=10.0, max_passes=60, seed=2):
    n = len(h)
    L = 2 * n - 1
    t = n / 2.0
    dx = 2.0 / n
    h = project_box_sum(h, t)
    c = np.correlate(h, 1.0 - h, 'full')
    bh = h.copy()
    bc = float(c.max()) * dx
    curr = float(c.max())
    sumh = float(h.sum())
    rng = np.random.RandomState(seed)
    idx = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        imp = False
        for p in rng.permutation(n):
            v = h[p]
            a = np.zeros(L)
            a[n - 1 + p - idx] += 1.0 - h[idx]
            k2 = np.arange(-p, n - p)
            a[n - 1 + k2] -= h[p + k2]
            d = np.linspace(-v, 1.0 - v, grid)
            d = np.concatenate((d, [0.0]))
            ne = c[None, :] + d[:, None] * a[None, :]
            ne[:, n - 1] = c[n - 1] + d * (1.0 - 2.0 * v) - d * d
            vals = ne.max(axis=1) + lam * (sumh + d - t) ** 2
            bi = int(np.argmin(vals))
            step = float(d[bi])
            if vals[bi] < curr - 1e-12 and abs(step) > 1e-13:

                c += step * a

                imp = True
                if time.time() > deadline:

            curr = float(c.max())

    if deadline - time.time() > 120:
        alloc = min(620.0, 0.55 * (deadline - time.time()))
        ho, co = optimize_candidate(interp_n(bh, n_init), n_init, min(deadline, time.time() + alloc), rng, True)
        if co < bc:
            bh, bc = ho, co

    for n in (200, 250, 300, 400, 150):
        if n % 2 or n == n_init:
            continue
        if n < 80 or n > 700:
            continue
        if time.time() > deadline - 110:
            break
        alloc = min(150.0, 0.25 * (deadline - time.time()))
        if alloc < 45:
            break
        ho, co = optimize_candidate(interp_n(bh, n), n, min(deadline, time.time() + alloc), rng, n <= 300)
        if co < bc:
            bh, bc = ho, co

    for sc in (0.005, 0.01, 0.02, 0.05):
        if time.time() > deadline - 95:
            break
        n = len(bh)
        z = bh + rng.normal(0.0, sc, n)
        if n % 2 == 0:
            hz = project_sym(z, n / 2.0)
        else:
            hz = project_box_sum(z, n / 2.0)
        alloc = min(55.0, deadline - time.time() - 45)
        if alloc < 25:
            break
        hz, cz = pg_optimize(hz, n, min(deadline, time.time() + alloc), sym=(n % 2 == 0),
                             alphas=(80, 200, 500), max_iter=250)
        if cz < bc and n % 2 == 0 and time.time() < deadline - 25:
            h2, c2 = sym_cd(hz, min(deadline - 5, time.time() + 35), grid=16, seed=int(rng.randint(1000000)))
            if c2 < cz:
                hz, cz = h2, c2
        if cz < bc:
            bh, bc = hz, cz

    if time.time() < deadline - 12 and len(bh) % 2 == 0:
        hf, cf = sym_pair_cd(bh, min(deadline - 2, time.time() + 35), grid=14,
                             max_passes=18, seed=int(rng.randint(1000000)))
        if cf < bc:
            bh, bc = hf, cf
    if time.time() < deadline - 10 and len(bh) <= 320:
        hf, cf = full_cd(bh, min(deadline - 2, time.time() + 45), grid=12,
                         max_passes=30, seed=int(rng.randint(1000000)))
        if cf < bc:
            bh, bc = hf, cf

    bh = project_box_sum(bh, len(bh) / 2.0)
    return bh, c5(bh), len(bh)
