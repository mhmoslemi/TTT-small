import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _pen_obj(x, n, alpha, mu):
    return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _pen_grad(x, n, alpha, mu):
    return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    return smooth_max_value(x, n, alpha)

def run_lbfgs(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 10.0), (10000, 50.0),
               (50000, 200.0), (200000, 1000.0), (1000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                          args=(n, alpha, mu), bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:

                      bounds=bounds, constraints=cons,

        c_new = compute_c5(h_new, n)
        if c_new < best_c:

    except Exception:
        pass
    return h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def _pen_obj(x, n, alpha, mu):
    return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _pen_grad(x, n, alpha, mu):
    return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha):
    return smooth_max_value(x, n, alpha)

def run_lbfgs(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 10.0), (10000, 50.0),
               (50000, 200.0), (200000, 1000.0), (1000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                          args=(n, alpha, mu), bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_slsqp(h, n, deadline, alpha=2000000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def nesterov_pg(h, n, deadline, lr_init=0.03, alpha_init=200, momentum=0.85):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    lr = lr_init
    stall = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        y = project_feasible(h + momentum * v, n)
        g = grad_smooth_max(y, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:

            improved = False
            for f in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h + f * v, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-15:

                    c_new = c_try
                    best_c = c_try

def smart_local_search(h, n, deadline, max_iters=100000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 1000
    lr = 0.01
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        improved = False
        neg_sorted = np.argsort(g)[:15]
        pos_sorted = np.argsort(g)[-15:][::-1]
        for i in neg_sorted:
            if improved:
                break
            for j in pos_sorted:
                if i == j:

                    if 0 <= hi <= 1 and 0 <= hj <= 1:

def active_set_lp_polish(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    tol = 1e-4
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        active = np.where(c > c_max - tol)[0]
        if len(active) == 0 or len(active) > 20:
            active = np.argsort(c)[-10:]
        n_act = len(active)
        J = np.zeros((n_act, n))
        for idx, m in enumerate(active):
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                J[idx, 0:nv] += (1.0 - h[k:n])
                J[idx, k:n] += (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                J[idx, kk:n] += (1.0 - h[0:nv])
                J[idx, 0:nv] += (-h[kk:n])
        J *= dx
        c_lp = np.zeros(n + 1)
        c_lp[n] = 1.0
        A_ub = np.zeros((n_act, n + 1))
        A_ub[:, :n] = J
        A_ub[:, n] = -1.0
        b_ub = -c[active]
        A_eq = np.ones((1, n + 1))

        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(0, None)]

                    best_h = h_try.copy()
