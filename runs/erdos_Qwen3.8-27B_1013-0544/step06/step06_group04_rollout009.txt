def slsqp_smooth(h, n, alpha, maxiter=500):

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        return m + np.log(np.exp(sh).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    d_lo1 = max(-(n-1), i - n + 1)
    d_hi1 = min(n-1, i)
    if d_lo1 <= d_hi1:
        d_vals = np.arange(d_lo1, d_hi1 + 1)
        k_vals = d_vals + (n - 1)
        m_vals = i - d_vals
        delta[k_vals] += (1.0 - h[m_vals])
    d_lo2 = max(-(n-1), -i)
    d_hi2 = min(n-1, n-1-i)
    if d_lo2 <= d_hi2:
        d_vals = np.arange(d_lo2, d_hi2 + 1)
        k_vals = d_vals + (n - 1)
        m_vals = i + d_vals
        delta[k_vals] -= h[m_vals]
    d_lo1 = max(-(n-1), j - n + 1)
    d_hi1 = min(n-1, j)
    if d_lo1 <= d_hi1:
        d_vals = np.arange(d_lo1, d_hi1 + 1)
        k_vals = d_vals + (n - 1)
        m_vals = j - d_vals
        delta[k_vals] -= (1.0 - h[m_vals])
    d_lo2 = max(-(n-1), -j)

        d_vals = np.arange(d_lo2, d_hi2 + 1)
        k_vals = d_vals + (n - 1)
        m_vals = j + d_vals
        delta[k_vals] += h[m_vals]
    return delta

def local_search(h, n, deadline, max_sweeps=5000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

            break

        j = int(rng.choice(ones))

        nc = float((corr + d).max()) * dx
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-12)):
            h[i], h[j] = 1.0, 0.0
            corr = corr + d
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        return m + np.log(np.exp(sh).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgs_penalty(h, n, alpha, mu, maxiter=3000):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        return m + np.log(np.exp(sh).sum()) / alpha + mu * (x.sum() - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'maxfun': 5000})
        return project_feasible(res.x, n)
    except Exception:
        return h
