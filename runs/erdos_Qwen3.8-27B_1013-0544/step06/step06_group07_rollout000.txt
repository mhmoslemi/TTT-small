OPT_N = 0
OPT_ALPHA = 1.0
OPT_DEADLINE = 0.0
OPT_STOP = False
def opt_obj(x):
    h = np.asarray(x, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    a = OPT_ALPHA
    shifted = a * (c - m)
    # for stability if a huge, exp underflow okay
    return m + float(np.log(np.sum(np.exp(shifted)))) / a
def opt_jac(x):
    h = np.asarray(x, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    a=OPT_ALPHA
    shifted = a*(c-m)
    w=np.exp(shifted)
    s=w.sum()
    if s ==0: w=np.zeros_like(c); w[int(np.argmax(c))]=1.0
    else: w/=s
    n=OPT_N; L=n-1
    g1 = np.convolve(1.0 - h, w, mode='full')[L:L+n]
    g2 = np.convolve(h, w[::-1], mode='full')[L:L+n]
    return g1 - g2
def opt_con(x): return float(np.sum(x) - OPT_N*0.5)
def opt_callback(x):
    if time.time() > OPT_DEADLINE or OPT_STOP:
        return False
    return True

def slsqp_optimize(h_start,n,alpha,maxiter,deadline):
    global OPT_N, OPT_ALPHA, OPT_DEADLINE, OPT_STOP
    h = project_feasible(h_start,n)
    OPT_N=n; OPT_ALPHA=float(alpha); OPT_DEADLINE=deadline - 2.0?; OPT_STOP=False
    bounds=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':opt_con}]
    try:
       res = minimize(opt_obj, h, method='SLSQP', jac=opt_jac, bounds=bounds, constraints=cons, options={'maxiter':maxiter,'ftol':1e-12,'maxls':20}, callback=opt_callback)
       h = project_feasible(res.x,n)
    except Exception:
       h = project_feasible(h,n)

def project_feasible(v,n):
    v=np.asarray(v,dtype=float)
    t=0.5*float(n)
    lo=float(v.min())-1.0; hi=float(v.max())+1.0
    for _ in range(50):
        m=0.5*(lo+hi); s=float(np.clip(v-m,0.0,1.0).sum())
        if s>t: lo=m
        else: hi=m
    x=np.clip(v-0.5*(lo+hi),0.0,1.0)
    for _ in range(5):
        d=t-x.sum()
        if abs(d)<1e-13: break
        if d>0:
            r=1.0-x; sr=float(r.sum())
            if sr>1e-14: x=np.clip(x+(d/sr)*r,0.0,1.0)
        else:
            sh=float(x.sum())
            if sh>1e-14: x=np.clip(x+(d/sh)*x,0.0,1.0)
    return x

import time
import math
import numpy as np
from scipy.optimize import minimize

OPT_N = 0
OPT_ALPHA = 1.0
OPT_DEADLINE = 0.0


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(v, n):
    v = np.asarray(v, dtype=float)
    target = 0.5 * float(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(6):
        diff = target - float(x.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - x
            sr = float(room.sum())
            if sr > 1e-14:
                x = np.clip(x + (diff / sr) * room, 0.0, 1.0)
        else:
            sh = float(x.sum())
            if sh > 1e-14:
                x = np.clip(x + (diff / sh) * x, 0.0, 1.0)
    return x


def round_to_binary(h, n):
    h = np.asarray(h, dtype=float)
    order = np.argsort(-h)
    out = np.zeros(n, dtype=float)
    out[order[:n // 2]] = 1.0
    return out


def resample_h(h0, n0, n1):
    h0 = np.asarray(h0, dtype=float)
    if n0 == n1:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n1) + 0.5) * (2.0 / float(n1))
    return np.interp(x1, x0, h0)


def low_disc_start(n, alpha):
    out = np.zeros(n, dtype=float)
    t = np.arange(n) * float(alpha)
    frac = t - np.floor(t)
    idx = np.argsort(frac)[:n // 2]
    out[idx] = 1.0
    return out


def delta_fill(delta, p, h, n, i, j):
    delta.fill(0.0)
    L = n - 1
    delta[L - i:L - i + n] += p
    delta[L - j:L - j + n] -= p
    delta[L - i:L - i + n] -= h
    delta[L - j:L - j + n] += h
    delta[L] -= 2.0
    delta[L + j - i] += 1.0
    delta[L + i - j] += 1.0


def active_pair(h, lag, rng, tries):
    n = h.size
    if lag > 0:
        a = 0
        b = n - lag
    elif lag < 0:
        a = -lag
        b = n
    else:
        return None
    if b <= a:
        return None
    for _ in range(tries):
        i = int(rng.randint(a, b))
        j = i + int(lag)
        if h[i] == 0.0 and h[j] == 1.0:
            return i, j
    return None


def binary_search(h_start, n, deadline, seed):
    h = round_to_binary(h_start, n)
    p = 1.0 - h
    c1 = np.correlate(h, p, mode="full").astype(float).copy()
    c2 = np.empty_like(c1)
    delta = np.zeros(2 * n - 1)
    L = n - 1
    curM = float(c1.max())
    curS = float(c1.sum())
    bestM = curM
    bestS = curS
    best_h = h.copy()
    zeros = np.where(h == 0.0)[0]
    ones = np.where(h == 1.0)[0]
    rng = np.random.RandomState(seed)
    T = 0.35
    Tend = 0.0015
    cool = (Tend / T) ** (1.0 / 250000.0)
    top_lags = np.argsort(c1)[-40:]
    noimp = 0
    step = 0

    while time.time() < deadline:
        if len(zeros) == 0 or len(ones) == 0:
            break
        if (step & 255) == 0:
            top_lags = np.argsort(c1)[-40:]
        pair = None
        if len(top_lags) > 0 and rng.random() < 0.7:
            lag = int(top_lags[rng.randint(len(top_lags))]) - L
            pair = active_pair(h, lag, rng, 12)
        if pair is None:
            i = int(zeros[rng.randint(len(zeros))])
            j = int(ones[rng.randint(len(ones))])
        else:

            delta_fill(delta, p, h, n, i, j)

            nm = float(c2.max())

def opt_obj(x):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    a = OPT_ALPHA
    sh = a * (c - m)
    return float(m + math.log(float(np.sum(np.exp(sh)))) / a)


def opt_jac(x):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    a = OPT_ALPHA
    w = np.exp(a * (c - m))
    s = float(w.sum())
    if s > 0.0:
        w = w / s
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    n = OPT_N
    L = n - 1
    grad = np.zeros(n, dtype=float)
    nz = np.where(w > 1e-12)[0]
    if len(nz) < max(10, (2 * n - 1) // 8):
        p = 1.0 - x
        for l in nz:
            wd = float(w[l])
            d = int(l) - L
            if d == 0:
                grad += wd * (1.0 - 2.0 * x)
            elif d > 0:
                grad[d:] += wd * p[:n - d]
                grad[:n - d] -= wd * x[d:]
            else:
                e = -d
                grad[:n - e] += wd * p[e:]
                grad[e:] -= wd * x[:n - e]
    else:
        g1 = np.convolve(1.0 - x, w, mode="full")[L:L + n]
        g2 = np.convolve(x, w[::-1], mode="full")[L:L + n]
        grad = g1 - g2
    return grad


def opt_con(x):
    return float(np.sum(x) - 0.5 * float(OPT_N))


def opt_callback(x):
    return time.time() > OPT_DEADLINE


def slsqp_one(h_start, n, alpha, maxiter, deadline):
    global OPT_N, OPT_ALPHA, OPT_DEADLINE
    n = int(n)
    h = project_feasible(h_start, n)
    OPT_N = n
    OPT_ALPHA = float(alpha)
    OPT_DEADLINE = float(deadline)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": opt_con}]
        opts = {"maxiter": int(max(max(10, maxiter))), "ftol": 1e-10, "disp": False}
        res = minimize(
            opt_obj,
            h,
            method="SLSQP",
            jac=opt_jac,
            bounds=bounds,
            constraints=cons,
            options=opts,
            callback=opt_callback,
        )
        if hasattr(res, "x"):
            h = res.x
    except Exception:
        pass
    return project_feasible(h, n)


def slsqp_sequence(h_start, n, t_start, t_end):

    h = project_feasible(h_start, n)

    start = 1 if (n > 700 and t_end - time.time() < 120.0) else 0
    for idx in range(start, len(alphas)):
        if time.time() >= t_end:

        rem = (t_end - time.time()) / float(len(alphas) - idx)
        if rem < 8.0:
            break
        call = min(rem, 40.0)

            call = 6.0
        if time.time() + call >= t_end:

        if call < 2.0:

        h1 = slsqp_one(best_h, n, alphas[idx], maxiter, time.time() + call)
        c1 = compute_c5(h1, n)
        if c1 < best_c:

            best_c = c1
