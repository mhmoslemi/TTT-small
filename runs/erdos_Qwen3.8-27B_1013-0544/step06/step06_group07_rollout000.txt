def swap_delta(h,n,i,j):
    d=np.zeros(2*n-1)
    omh=1.0-h
    d[n-1-i:2*n-i]+=omh
    d[i:n+i]-=h[::-1]

import numpy as np
import time


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def c_natural(h, n):
    return np.correlate(h, 1.0 - h, mode="full")[::-1]


def project_sum_box(h, S):
    h = np.asarray(h, dtype=float)
    out = np.clip(h, 0.0, 1.0)
    diff = S - out.sum()
    if abs(diff) < 1e-10:
        out[out < 1e-14] = 0.0
        out[out > 1.0 - 1e-14] = 1.0
        if abs(diff) > 1e-13:
            m = (out > 1e-12) & (out < 1.0 - 1e-12)
            if m.any():
                i = int(np.flatnonzero(m)[0])
                room = 1.0 - out[i] if diff > 0 else out[i]
                out[i] += max(-room, min(room, diff))
        return out

    lo = min(-1.0, float(h.min()) - 1.0)
    hi = max(1.0, float(h.max()) + 1.0)
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid

    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - out.sum()
    m = (out > 1e-12) & (out < 1.0 - 1e-12)
    if m.any():
        i = int(np.flatnonzero(m)[0])
        room = 1.0 - out[i] if diff > 0 else out[i]
        out[i] += max(-room, min(room, diff))
        diff = S - out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            flat = np.flatnonzero(out < 1.0)
            if flat.size:
                out[flat[0]] = min(1.0, out[flat[0]] + diff)
        else:
            flat = np.flatnonzero(out > 0.0)
            if flat.size:
                out[flat[0]] = max(0.0, out[flat[0]] + diff)
    return out


def round_to_binary(h, n):
    target = n // 2
    out = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1][:target]
    out[idx] = 1.0
    return out


def resample_array(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    if n_old == 1:
        return np.full(n_new, float(h[0]))
    x_new = np.linspace(0.0, float(n_old - 1), n_new)
    return np.interp(x_new, np.arange(n_old), h)


def grad_w(h, W, n):
    return (
        np.convolve(1.0 - h, W[::-1], mode="full")[n - 1:n - 1 + n]
        - np.convolve(h, W, mode="full")[n - 1:n - 1 + n]
    )


def make_weights(c, C, n, mode):
    W = np.zeros_like(c, dtype=float)
    dist = np.maximum(0.0, C - c)
    if mode == 0:
        W.fill(1.0)
    elif mode == 1:
        W = 1.0 / (1.0 + dist)
        W += 0.02
        W[c >= C] = 1.0
    elif mode == 2:
        W[c >= C - 1.0] = 1.0
        W[(c >= C - 2.0) & (c < C - 1.0)] = 0.3
    elif mode == 3:
        W = np.exp(-0.9 * dist) + 0.05
    else:
        W = np.exp(-2.0 * dist) + 0.1
    W[n - 1] = 0.0
    s = W.sum()
    if s <= 1e-15:
        W.fill(1.0)
        W[n - 1] = 0.0
        s = W.sum()
    if s > 0:
        W /= s
    return W


def swap_delta(h, n, i, j):
    d = np.zeros(2 * n - 1, dtype=float)
    omh = 1.0 - h
    hr = h[::-1]
    d[n - 1 - i:2 * n - i] += omh
    d[i:n + i] -= hr
    d[n - 1 - j:2 * n - j] -= omh
    d[j:n + j] += hr
    d[n - 1] = 0.0
    dd = j - i
    if dd:
        d[n - 1 + dd] += 1.0
        d[n - 1 - dd] += 1.0
    return d


def random_walk_binary(h, c, C, n, rng, deadline):
    T = 1.0
    step = 0
    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(zeros[rng.randint(zeros.size)])
        j = int(ones[rng.randint(ones.size)])
        d = swap_delta(h, n, i, j)
        nc = float((c + d).max())
        diff = nc - C
        accept = False
        if diff < -1e-12:
            accept = True
        elif abs(diff) <= 1e-12:
            accept = rng.random() < 0.55
        elif diff <= 3.0 and T > 1e-8:
            accept = rng.random() < float(np.exp(-diff / T))
        if accept:
            h[i] = 1.0
            h[j] = 0.0
            c += d
            C = nc
            step += 1
            T *= 0.999
            if step % 500 == 0:
                c = c_natural(h, n)
                C = float(c.max())
        else:
            T *= 0.9995
    c = c_natural(h, n)
    C = float(c.max())
    return h, c, C


def local_binary(h, n, deadline, seed):

    c = c_natural(h, n)
    C = float(c.max())
    best_h = h.copy()
    best_C = C
    rng = np.random.RandomState(seed)
    no_improve = 0
    equal_streak = 0
    random_count = 0
    step = 0
    modes = (1, 3, 2, 4, 0)

    while time.time() < deadline:
        strict_pair = None
        strict_delta = None
        strict_next = C
        equal_pair = None
        equal_delta = None
        equal_score = 1e100

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        for m in modes:
            if time.time() >= deadline:
                break
            W = make_weights(c, C, n, m)
            if W.sum() <= 1e-15:
                continue
            G = grad_w(h, W, n)
            O = W + W[::-1]
            no_o = ones.size
            diff = zeros[:, None] - ones[None, :]
            scores = G[zeros][:, None] - G[ones][None, :] + O[diff + (n - 1)]

            K = 80 if n <= 900 else 50

                order = np.argsort(scores.ravel())
            else:

            h[i] = 1.0

            best_h = h.copy()
        no_improve = 0
        equal_streak = 0
        random_count += 1
        if random_count > 60:

    return best_h, float(best_C) * (2.0 / n), float(best_C)


def smooth_improve(h, n, deadline, seed):

            if s <= 1e-15:

            if gs < 1e-12:

                    C = Ctry

                    if C < best_C - 1e-12:
                        best_C = C
                        best_h = h.copy()
                    break

            if improved:
                lr = min(0.08, lr * 1.25)
                fail = 0
            else:
                lr *= 0.5
                fail += 1

                    h = project_sum_box(old_h + 0.01 * rng.standard_normal(n), n / 2.0)
                    c = c_natural(h, n)
                    C = float(c.max())
                    if C < best_C - 1e-12:
                        best_C = C
                        best_h = h.copy()
                    lr = lr0 * 0.7
                    fail = 0
                    stuck += 1
                    if stuck > 8:
                        break

    return best_h, float(best_C) * (2.0 / n)

    best_c = compute_c5(best_h, best_n)

    usable = max(10.0, float(budget_s) - 8.0)

    qend = min(dl, t0 + min(60.0, 0.08 * usable))
    if qend > time.time() + 5.0:
        hs, cs = smooth_improve(best_h, best_n, qend, seed)
        if cs < best_c - 1e-12:
            best_c = cs
            best_h = hs.copy()

    if best_n < 450:
        n1 = 500
    elif best_n < 850:
        n1 = 750
    else:
        n1 = best_n

    p1 = t0 + 0.48 * usable
    if p1 > time.time() + 20.0:
        if n1 != best_n:
            hstart = resample_array(best_h, best_n, n1)
        else:
            hstart = best_h.copy()
        hb, cb = local_binary(hstart, n1, min(dl, p1), seed)
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb.copy()
            best_n = n1

    if best_n < 750:
        n2 = 750
    elif best_n < 900:
        n2 = 900
    else:
        n2 = best_n

    p2 = t0 + 0.74 * usable
    if p2 > time.time() + 60.0:
        if n2 != best_n:
            hstart = resample_array(best_h, best_n, n2)
        else:
            hstart = best_h.copy()
        rng = np.random.RandomState(seed + 11)
        hstart = project_sum_box(hstart + 0.018 * rng.standard_normal(n2), n2 / 2.0)
        hb, cb = local_binary(hstart, n2, min(dl, p2), seed + 11)
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb.copy()
            best_n = n2

    p3 = t0 + 0.90 * usable
    if p3 > time.time() + 30.0:
        hs, cs = smooth_improve(best_h, best_n, min(dl, p3), seed + 31)
        if cs < best_c - 1e-12:
            best_c = cs
            best_h = hs.copy()

    if dl > time.time() + 25.0:
        hb, cb = local_binary(best_h, best_n, dl - 2.0, seed + 991)
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb.copy()

    best_h = project_sum_box(best_h, best_n / 2.0)
    if np.all((best_h < 5e-12) | (best_h > 1.0 - 5e-12)):
        best_h = np.where(best_h < 0.5, 0.0, 1.0)

    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
