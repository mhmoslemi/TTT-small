import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=np.int64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    count = t.sum()
    if count > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:count - target]] = 0
    elif count < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - count]] = 1
    return t.astype(float)

def exhaustive_best_swap(h, n, corr):
    cur_max = int(np.max(corr))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    best_new_max = cur_max
    best_new_corr = None
    for i in zeros:
        for j in ones:
            h[i], h[j] = 1.0, 0.0
            nc = full_corr(h)
            nm = int(np.max(nc))
            h[i], h[j] = 0.0, 1.0
            if nm < best_new_max:
                best_new_max = nm
                best_i, best_j = int(i), int(j)
                best_new_corr = nc.copy()
    return best_i, best_j, best_new_corr, best_new_max

        h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
    return None, cur_max

def slsqp_smooth(h, n, deadline, alpha=200000):
    dx = 2.0 / n
    def slsqp_obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        mshift = shifted.max()
        return m + (mshift + np.log(np.exp(shifted - mshift).sum())) / alpha
    def slsqp_cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons}]
    try:
        res = minimize(slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def project_constraint(h, n):
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    return np.clip(h, 0, 1)
