import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Small correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def make_p1_matrix(idx1, p, n):
    """Build P1 matrix from precomputed indices."""
    mask1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
    p1 = np.zeros((n, n))
    np.copyto(p1, p[np.clip(idx1, 0, 2*n-2)], where=mask1)
    return p1

def grad_lse(h, n, alpha, idx1):
    """Compute gradient of log-sum-exp objective."""
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / exp_g.sum()
    
    # Build P1 matrix
    mask1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
    P1 = np.zeros((n, n))
    np.copyto(P1, p[np.clip(idx1, 0, 2*n-2)], where=mask1)
    
    C1 = P1 @ h
    C2 = P1.T @ h
    S1 = P1.sum(axis=1)
    
    return dx * (S1 - C1 - C2)

    """Adam optimizer with projection onto feasible set."""
    h = h0.copy()

    v = np.zeros(n)
    t = 1
    m1 = 1 - beta1**t
    m2 = 1 - beta2**t

            best_h = h_new.copy()
        
        h = h_new
        t += 1
        m1 = 1 - beta1**t
        m2 = 1 - beta2**t
    
    return best_h, best_c

def compute_c5(h, n):

        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_lse_fft(h, n, alpha, N_fft, idx_c1, idx_c2, ones_rfft):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = c.max()
    exp_c = np.exp(alpha * (c - gm))
    p = exp_c / exp_c.sum()
    
    P = np.fft.rfft(p, N_fft)
    H = np.fft.rfft(h, N_fft)
    cross = np.fft.irfft(P * np.conj(H), N_fft)
    cross_s = np.fft.irfft(P * np.conj(ones_rfft), N_fft)
    
    C1 = cross[idx_c1]
    C2 = cross[idx_c2]
    S1 = cross_s[idx_c1]
    
    return dx * (S1 - C1 - C2)

def adam_optimize(h0, n, deadline, alpha, lr, max_check):
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft *= 2
    idx_c1 = (np.arange(n) - (n - 1)) % N_fft
    idx_c2 = (n - 1 - np.arange(n)) % N_fft
    ones_rfft = np.fft.rfft(np.ones(n), N_fft)
    
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    while time.time() < deadline:
        g = grad_lse_fft(h, n, alpha, N_fft, idx_c1, idx_c2, ones_rfft)
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        
        if t % max_check == 0:
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                break
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c
