4. Update: corr = corr + D[i,:] - D[j,:], update D, update h
5. Repeat

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_D(h, n):
    D = np.zeros((n, 2*n-1))
    lags = np.arange(2*n-1) - (n-1)
    for p in range(n):
        idx1 = p + lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        if valid1.any():
            D[p, valid1] += 1.0 - h[idx1[valid1]]
        idx2 = p - lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        if valid2.any():
            D[p, valid2] -= h[idx2[valid2]]
    return D

def update_D(D, h, n, i, j):
    """Update D after h[i]: 0->1 and h[j]: 1->0. Must be called BEFORE updating h."""
    lags = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):
        lag = lags[k]
        # i: 0->1
        p1 = i - lag
        if 0 <= p1 < n:
            D[p1, k] -= 1.0
        p2 = i + lag
        if 0 <= p2 < n:
            D[p2, k] -= 1.0
        # j: 1->0
        p3 = j - lag
        if 0 <= p3 < n:
            D[p3, k] += 1.0
        p4 = j + lag
        if 0 <= p4 < n:
            D[p4, k] += 1.0

def greedy_sa(h, n, deadline, seed=42, top_k=20, temp_init=0.01, temp_min=0.001, temp_decay=0.999):
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1.0-h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    D = compute_D(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    temp = temp_init
    stall = 0
    
    while time.time() < deadline:
        # Compute all swap values
        # new_max[i,j] = max_k (corr[k] + D[zeros[i],k] - D[ones[j],k])
        Dz = D[zeros]  # (nz, 2n-1)
        Do = D[ones]   # (no, 2n-1)
        # For each (i,j): max_k(corr + Dz[i] - Do[j])
        # = max_k((corr + Dz[i]) - Do[j])
        # Let's compute: A[i,k] = corr[k] + Dz[i,k]  (nz, 2n-1)
        # Then new_max[i,j] = max_k(A[i,k] - Do[j,k])
        # This is a "max minus" operation... not easily vectorized as a single op.
        # Alternative: new_max[i,j] = max_k(A[i,:,None] - Do[None,j,:]).max(axis=1)
        # Shape: (nz, no, 2n-1) -> max over last axis -> (nz, no)
        
        # For memory: nz=125, no=125, 2n-1=499 -> 125*125*499*8 bytes ≈ 62 MB. OK.
        
        A = corr[None, :] + Dz  # (nz, 2n-1)
        # new_max[i,j] = max_k (A[i,k] - Do[j,k])
        # = max over k of (A[i,:,None] - Do[None,j,:])
        diff_3d = A[:, :, None] - Do[None, :, :]  # (nz, no, 2n-1)
        new_maxes = diff_3d.max(axis=2)  # (nz, no)
        
        flat_idx = np.argmin(new_maxes)
        ii = flat_idx // len(ones)
        jj = flat_idx % len(ones)
        best_val = new_maxes[ii, jj]
        i_idx = zeros[ii]
        j_idx = ones[jj]
        
        if best_val < cur_c - 1e-12:
            # Accept the move
            corr = corr + D[i_idx, :] - D[j_idx, :]
            update_D(D, n, i_idx, j_idx)
            h[i_idx] = 1.0
            h[j_idx] = 0.0
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j_idx)
            ones = np.append(ones, i_idx)
            cur_c = corr.max()
            stall = 0
            temp = max(temp_min, temp * temp_decay)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # No improving move; try SA acceptance or kick
            stall += 1
            if stall > 50:
                # Random kick: swap 2-3 random pairs
                nk = min(3, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    i_idx = zeros[ri[a]]
                    j_idx = ones[rj[a]]
                    corr = corr + D[i_idx, :] - D[j_idx, :]
                    update_D(D, n, i_idx, j_idx)
                    h[i_idx] = 1.0
                    h[j_idx] = 0.0
                # Update zeros/ones
                rii = np.sort(ri)
                rjj = np.sort(rj)
                zeros = np.delete(zeros, rii)
                ones = np.delete(ones, rjj)
                zeros = np.append(zeros, rjj[lambda x: ones[rjj]] ... )
                # This is getting messy. Let me just recompute zeros and ones.
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                cur_c = corr.max()
                stall = 0
                temp = max(temp_min, temp * 0.8)
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_D(h, n):
    D = np.zeros((n, 2*n-1))
    lags = np.arange(2*n-1) - (n-1)
    for p in range(n):
        idx1 = p + lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        if valid1.any():
            D[p, valid1] += 1.0 - h[idx1[valid1]]
        idx2 = p - lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        if valid2.any():
            D[p, valid2] -= h[idx2[valid2]]
    D[:, n-1] = 0.0  # lag=0 special case
    return D
