import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    m = corr.max()
    s = alpha * (corr - m)
    s_max = s.max()
    logsum = s_max + np.log(np.exp(s - s_max).sum())
    obj = m + logsum / alpha
    sv = h.sum() - n / 2.0
    return obj + penalty * sv * sv

def smooth_max_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    g = corr * dx
    m = g.max()
    s = alpha * (g - m)
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    w = exp_s / exp_s.sum()
    
    # Compute gradient of smooth max
    grad_sm = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-18:
            continue
        lag = k - (n-1)
        # (1-h[i+lag]) for valid i
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo < i_hi:
            grad_sm[i_lo:i_hi] += w[k] * (1.0 - h[i_lo+lag:i_hi+lag])
        # -h[i-lag] for valid i
        i_lo2 = max(0, lag)
        i_hi2 = min(n, n + lag)
        if i_lo2 < i_hi2:
            grad_sm[i_lo2:i_hi2] -= w[k] * h[i_lo2-lag:i_hi2-lag]
    grad_sm *= dx
    
    # Add penalty gradient
    sv = h.sum() - n / 2.0
    grad = grad_sm + 2.0 * penalty * sv * np.ones(n)
    return grad

def lbfgs_smooth(h, n, alpha, maxiter, penalty, ftol):
    res = minimize(
        smooth_max_obj, h,
        jac=smooth_max_grad,
        args=(n, alpha, penalty),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12}
    )
    return project_feasible(res.x, n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h

def smooth_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    m = g.max()
    s = alpha * (g - m)
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    w = exp_s / exp_s.sum()
    obj = m + (s_max + np.log(exp_s.sum())) / alpha
    sv = x.sum() - n / 2.0
    obj += penalty * sv * sv
    
    # Gradient
    grad_sm = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo < i_hi:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    m = g.max()
    s = alpha * (g - m)
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    w = exp_s / exp_s.sum()
    obj = m + (s_max + np.log(exp_s.sum())) / alpha
    sv = x.sum() - n / 2.0
    obj += penalty * sv * sv
    grad_sm = np.zeros(n)
    threshold = 1e-14
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        i_lo = max(0, lag)
        i_hi = min(n, n + lag)
        if i_lo < i_hi:
            grad_sm[i_lo:i_hi] += wk * (1.0 - x[i_lo - lag:i_hi - lag])
        i_lo2 = max(0, -lag)
        i_hi2 = min(n, n - lag)
        if i_lo2 < i_hi2:
            grad_sm[i_lo2:i_hi2] -= wk * x[i_lo2 + lag:i_hi2 + lag]
    grad_sm *= dx
    grad = grad_sm + 2.0 * penalty * sv
    return obj, grad

def lbfgs_optimize(h, n, alpha, maxiter=500, penalty=5000):
    def fun(x):
        o, _ = smooth_obj_grad(x, n, alpha, penalty)
        return o
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha, penalty)
        return g
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_search(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=1000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.02
    cool = 0.99997
    scale = 2000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
