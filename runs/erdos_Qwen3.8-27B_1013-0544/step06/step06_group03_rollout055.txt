    dx = 2.0 / n

    return float(np.max(c) * dx)

def compute_all_lags(h, n):

            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

def lag_grad(h, n, lag):
    """Gradient of g_lag = dx * sum h[j](1-h[j+lag]) w.r.t. h."""
