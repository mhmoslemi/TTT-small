import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_lag(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def _smooth_max(x, alpha):
    m = x.max()
    shifted = alpha * (x - m)
    mx = shifted.max()
    log_sum = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + log_sum / alpha)

def _smooth_max_grad(x, alpha):
    m = x.max()
    shifted = alpha * (x - m)
    mx = shifted.max()
    e = np.exp(shifted - mx)
    return e / e.sum()

def slsqp_with_grad(h, n, top_idx, alpha, maxiter=3000):
    dx = 2.0 / n

    def obj(x):
        return _smooth_max(compute_all_lags(x, n)[top_idx], alpha)

    def obj_grad(x):
        c_act = compute_all_lags(x, n)[top_idx]
        w = _smooth_max_grad(c_act, alpha)
        grad = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            lag = k_idx - (n - 1)
            grad += dx * w[i] * _grad_lag(x, n, lag)
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sca_step(h_curr, n, top_m):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-top_m:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    for k_idx in top_idx:
        lag = k_idx - (n - 1)
        grad = _grad_lag(h_curr, n, lag)
        lin_val = c[k_idx] + dx * (grad @ (h_var - h_curr))
        cons.append(lin_val <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

def subgradient_polish(h, n, deadline, iters=2000):

    best_c = compute_c5(h, n)

        if c_new < best_c - 1e-15:

            h = h_new
    return best_h, best_c
