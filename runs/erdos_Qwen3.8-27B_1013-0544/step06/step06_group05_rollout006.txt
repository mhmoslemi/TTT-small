def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

            break

import numpy as np
from scipy.optimize import minimize
import time

_slsqp = {}

def _slsqp_obj(x):
    n = _slsqp['n']
    alpha = _slsqp['alpha']
    g = np.correlate(x, 1.0 - x, mode='full')
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def _slsqp_cons(x):
    return x.sum() - _slsqp['n'] / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter):
    _slsqp['n'] = n
    _slsqp['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_flip_vectors(h, n):
    k_range = np.arange(-(n - 1), n)
    F = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pk = p + k_range
        valid = (pk >= 0) & (pk < n)
        F[p, valid] += (1.0 - h[pk[valid]])
        pk2 = p - k_range
        valid2 = (pk2 >= 0) & (pk2 < n)
        F[p, valid2] -= h[pk2[valid2]]
    return F

def systematic_search(h, n, deadline, max_passes=500):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    F = compute_flip_vectors(h, n)
    best_h = h.copy()
    best_max = cur_max
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        for a in ones:
            if time.time() > deadline:
                return best_h, float(best_max * dx)
            base = corr - F[a]
            candidates = base[None, :] + F[zeros, :]
            new_maxes = candidates.max(axis=1)
            sorted_idx = np.argsort(new_maxes)
            n_try = min(5, len(sorted_idx))
            for si in range(n_try):
                b = int(zeros[sorted_idx[si]])
                h[a] = 0.0
                h[b] = 1.0
                new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                exact_max = new_corr.max()
                if exact_max < cur_max - 1e-10:
                    corr = new_corr
                    cur_max = exact_max
                    F = compute_flip_vectors(h, n)
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                    found = True
                    break
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            if found:
                break
        if not found:
            break
    return best_h, float(best_max * dx)

def sa_search(h_init, n, deadline, seed=42, n_iter=400000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    T0 = 4.0
    Tend = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (it / n_iter)
        if T < 0.01:
            T = 0.01
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        delta = np.zeros(2 * n - 1)
        ak = a + k_range
        v1 = (ak >= 0) & (ak < n)
        delta[v1] -= (1.0 - h[ak[v1]])
        ak2 = a - k_range
        v2 = (ak2 >= 0) & (ak2 < n)
        delta[v2] += h[ak2[v2]]
        bk = b + k_range
        v3 = (bk >= 0) & (bk < n)
        delta[v3] += (1.0 - h[bk[v3]])
        bk2 = b - k_range
        v4 = (bk2 >= 0) & (bk2 < n)
        delta[v4] -= h[bk2[v4]]
        new_max = (corr + delta).max()
        d = new_max - cur_max
        if d < 0 or rng.random() < np.exp(-d / T):
            h[a] = 0.0
            h[b] = 1.0
            corr += delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, float(best_max * dx)

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin
