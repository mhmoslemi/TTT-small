```python
import numpy as np
import time
import scipy.optimize as sopt
import cvxpy as cp

RUN_DEADLINE = 0.0


def _cb(x):
    return time.time() > RUN_DEADLINE


def compute_all_lags(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))


def _clip_sum(z, lam):
    return float(np.sum(np.clip(z - lam, 0.0, 1.0)))


def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    target = n / 2.0
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if _clip_sum(z, mid) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    err = target - float(h.sum())
    if abs(err) > 1e-13:
        if err > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot > 1e-14:
                h += (err / tot) * room
        else:
            tot = float(h.sum())
            if tot > 1e-14:
                h += (err / tot) * h
    return np.clip(h, 0.0, 1.0)


def resample(h, n):
    h = np.asarray(h, dtype=float)
    old = len(h)
    if old == n:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, old), h)


def make_binary(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        h = resample(h, n)
    order = np.argsort(-h, kind="stable")
    x = np.zeros(n, dtype=np.int32)
    x[order[:n // 2]] = 1
    return x


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def grad_lag_unscaled(h, n, lag):
    g = np.zeros(n, dtype=float)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g


def _smooth_core(x, n, alpha, active):
    C = compute_all_lags(x, n)
    vals = C[active]
    mx = float(vals.max())
    sh = alpha * (vals - mx)
    ew = np.exp(sh)
    den = float(ew.sum())
    if den <= 0.0:
        den = 1.0
    f = mx + float(np.log(den)) / alpha
    return f, ew / den


def _smooth_f_top(x, n, alpha, active):
    f, _ = _smooth_core(x, n, alpha, active)
    return f


def _smooth_g_top(x, n, alpha, active):
    _, w = _smooth_core(x, n, alpha, active)
    g = np.zeros(n, dtype=float)
    off = n - 1
    dx = 2.0 / n
    for idx, wi in zip(active, w):
        wi = float(wi)
        if wi < 1e-12:
            continue
        lag = int(idx) - off
        if lag == 0:
            g += wi * (1.0 - 2.0 * x)
        elif lag > 0:
            m = lag
            g[:n - m] -= wi * x[m:]
            g[m:] += wi * (1.0 - x[:n - m])
        else:
            m = -lag
            g[:n - m] += wi * (1.0 - x[m:])
            g[m:] -= wi * x[:n - m]
    return dx * g


def _f_g_top_pen(x, n, alpha, active, mu):
    f0, w = _smooth_core(x, n, alpha, active)
    g = np.zeros(n, dtype=float)
    off = n - 1
    dx = 2.0 / n
    for idx, wi in zip(active, w):
        wi = float(wi)
        if wi < 1e-12:
            continue
        lag = int(idx) - off
        if lag == 0:
            g += wi * (1.0 - 2.0 * x)
        elif lag > 0:
            m = lag
            g[:n - m] -= wi * x[m:]
            g[m:] += wi * (1.0 - x[:n - m])
        else:
            m = -lag
            g[:n - m] += wi * (1.0 - x[m:])
            g[m:] -= wi * x[:n - m]
    g *= dx
    err = float(np.sum(x) - n / 2.0)
    scaled = err * (2.0 / n)
    f = f0 + mu * scaled * scaled
    if mu != 0.0:
        g += (2.0 * mu * scaled * (2.0 / n)) * np.ones(n)
    return f, g


def lbfgs_refine(h, n, alpha, active, mu, maxiter, end):
    global RUN_DEADLINE
    h0 = project_feasible(h, n)
    RUN_DEADLINE = float(end)
    try:
        res = sopt.minimize(
            _f_g_top_pen,
            h0,
            args=(n, alpha, active, mu),
            jac=True,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0)] * n,
            callback=_cb,
            options={"maxiter": int(maxiter), "ftol": 1e-15, "gtol": 1e-10, "maxls": 30},
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def slsqp_top(h, n, alpha, active, maxiter):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_con, "jac": _sum_con_jac, "args": (n,)}]
    try:
        res = sopt.minimize(
            _smooth_f_top,
            h0,
            args=(n, alpha, active),
            jac=_smooth_g_top,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-15},
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def linear_top_step(h, n, active, trust):
    try:
        h0 = np.asarray(h, dtype=float).copy()
        dx = 2.0 / n
        C = compute_all_lags(h0, n)
        off = n - 1
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [
            cp.sum(hv) == n / 2.0,
            hv >= 0.0,
            hv <= 1.0,
            hv >= h0 - trust,
            hv <= h0 + trust,
        ]
        for idx in active:
            lag = int(idx) - off
            g = grad_lag_unscaled(h0, n, lag)
            cons.append(C[idx] + dx * (g @ (hv - h0)) <= tv)
        prob = cp.Problem(cp.Minimize(tv), cons)
        for solver in (cp.GLOP, cp.ECOS, cp.CBC):
            try:
                prob.solve(solver=solver)
                if prob.status in ("optimal", "optimal_inaccurate") and hv.value is not None:
                    return project_feasible(np.asarray(hv.value, dtype=float), n)
            except Exception:
                continue
        return h0
    except Exception:
        return np.asarray(h, dtype=float).copy()


def binary_local(x, n, deadline, rng):
    x = np.asarray(x, dtype=np.int32).copy()
    target = n // 2
    if int(x.sum()) != target:
        if x.sum() < target:
            zeros = np.flatnonzero(x == 0)
            while x.sum() < target and zeros.size:
                x[rng.randint(zeros.size)] = 1
                zeros = np.flatnonzero(x == 0)
        else:
            ones = np.flatnonzero(x == 1)
            while x.sum() > target and ones.size:
                x[rng.randint(ones.size)] = 0
                ones = np.flatnonzero(x == 1)

    C = np.correlate(x, 1 - x, "full")
    M = int(C.max())
    top = int(np.sort(C)[-5:].sum())
    off = n - 1

    for _ in range(400):
        if time.time() > deadline:
            break
        active = np.flatnonzero(C >= M - 1)
        if active.size > 10:
            active = active[:10]
        cands = []
        for idx in active:
            lag = int(idx) - off
            m = abs(lag)
            L = n - m
            if m == 0:
                continue
            if lag > 0:
                ii = np.where((x[:L] == 0) & (x[m:] == 1))[0]
            else:
                ii = np.where((x[:L] == 1) & (x[m:] == 0))[0]
            if ii.size:
                if ii.size > 8:
                    ii = ii[rng.choice(ii.size, size=8, replace=False)]
                for p in ii:
                    cands.append((int(p), int(p + m)))
        ones = np.flatnonzero(x == 1)
        zeros = np.flatnonzero(x == 0)
        if ones.size and zeros.size:
            for _ in range(10):
                cands.append((int(ones[rng.randint(ones.size)]), int(zeros[rng.randint(zeros.size)])))

        best_cand = None
        best_Cn = None
        best_M = M
        best_top = top
        seen = set()
        for a, b in cands:
            if a == b or x[a] == x[b]:
                continue
            key = (a, b) if a < b else (b, a)
            if key in seen:
                continue
            seen.add(key)
            y = x.copy()
            y[a], y[b] = x[b], x[a]
            Cn = np.correlate(y, 1 - y, "full")
            Mn = int(Cn.max())
            Tn = int(np.sort(Cn)[-5:].sum())
            if Mn < best_M or (Mn == best_M and Tn < best_top):
                best_M = Mn
                best_top = Tn
                best_cand = y
                best_Cn = Cn
        if best_cand is None:
            break
        x = best_cand
        C = best_Cn
        M = best_M
        top = best_top

    return x.astype(float), compute_c5(x.astype(float), n)


def refine_continuous(h, n, end, quick=False):
    end = float(end)
    h = project_feasible(h, n)
    if time.time() + 5 < 0 or time.time() >= end:
        return h, compute_c5(h, n)
    best = h.copy()
    best_c = compute_c5(h, n)
    all_k = 2 * n - 1

    klin = min(40, all_k)
    for trust in (0.20, 0.10, 0.05, 0.025, 0.012):
        if time.time() > end - 8:
            break
        C = compute_all_lags(best, n)
        lin_top = np.argsort(C)[-klin:]
        cand = linear_top_step(best, n, lin_top, trust)
        c = compute_c5(cand, n)
        if c < best_c - 1e-13:
            best = cand
            best_c = c

    if quick:
        stages = [(1000, min(30, all_k), 1000, 100), (8000, min(80, all_k), 2000, 140)]
    else:
        stages = [
            (200, all_k, 1000, 80),
            (800, min(50, all_k), 1000, 140),
            (3000, min(80, all_k), 1000, 160),
            (12000, min(120, all_k), 2000, 200),
            (40000, min(200, all_k), 3000, 240),
        ]

    for alpha, K, mu, it in stages:
        if time.time() > end - 10:
            break
        C = compute_all_lags(best, n)
        top = np.argsort(C)[-K:]
        alloc = min(45.0, max(10.0, 0.28 * (end - time.time())))
        cand = lbfgs_refine(best, n, alpha, top, mu, it, time.time() + alloc)
        c = compute_c5(cand, n)
        if c < best_c - 1e-13:
            best = cand
            best_c = c
        if time.time() < end - 8 and c < best_c + 1e-11:
            C = compute_all_lags(best, n)
            lin_top = np.argsort(C)[-klin:]
            cand2 = linear_top_step(best, n, lin_top, 0.02)
            c2 = compute_c5(cand2, n)
            if c2 < best_c - 1e-13:
                best = cand2
                best_c = c2

    if not quick and time.time() < end - 30:
        for alpha, K, mi in [(3000, min(30, all_k), 25), (12000, min(60, all_k), 30), (50000, min(120, all_k), 35)]:
            if time.time() > end - 35:
                break
            C = compute_all_lags(best, n)
            top = np.argsort(C)[-K:]
            cand = slsqp_top(best, n, alpha, top, mi)
            c = compute_c5(cand, n)
            if c < best_c - 1e-13:
                best = cand
                best_c = c

    best = project_feasible(best, n)
    return best, compute_c5(best, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 300
        h0 = np.full(n0, 0.5)

    n0 = int(n0)
    if n0 < 50:
        n0 = 300
        h0 = np.full(n0, 0.5)
    if n0 % 2 != 0:
        n0 += 1
        h0 = resample(h0, n0)
    h0 = project_feasible(h0, n0)

    best = h0.copy()
    best_c = compute_c5(h0, n0)

    for nb in (n0, 250, 200, 150, 100, 75, 50):
        if time.time() > deadline - 220 or nb < 50 or nb % 2:
            continue
        xb = make_binary(best, nb)
        c = compute_c5(xb.astype(float), nb)
        if c < best_c:
            best = xb.astype(float)
            best_c = c

    for nb in (n0, 200, 150, 100):
        if time.time() > deadline - 220 or nb < 50 or nb % 2:
            continue
        xs = np.linspace(0.0, 2.0, nb)
        for f in (1.0, 2.0, 3.0, 5.0):
            xb = make_binary(np.sin(2.0 * np.pi * f * xs), nb)
            c = compute_c5(xb.astype(float), nb)
            if c < best_c:
                best = xb.astype(float)
                best_c = c

    if deadline - t0 < 300:
        main_end = deadline - 20
    else:
        main_end = min(deadline - 170, t0 + 470)
    if time.time() < main_end - 12:
        best, best_c = refine_continuous(best, len(best), main_end, quick=False)

    for nb in (250, 200, 150, 120, 100, 80, 60, 50):
        if time.time() > deadline - 70 or nb < 50 or nb % 2 or nb == len(best):
            continue
        alloc = min(85.0, max(25.0, 0.25 * (deadline - time.time())))
        if time.time() + alloc > deadline - 35:
            alloc = deadline - 35 - time.time()
        if alloc < 20:
            break
        hs = project_feasible(resample(best, nb), nb)
        xb = make_binary(hs, nb)
        c = compute_c5(xb.astype(float), nb)
        if c < best_c:
            hs = xb.astype(float)
            best_c = c
        cand, cand_c = refine_continuous(hs, nb, time.time() + alloc, quick=True)
        if cand_c < best_c:
            best = cand
            best_c = cand_c

    for scale in (0.005, 0.01, 0.02, 0.035):
        if time.time() > deadline - 90:
            break
        n = len(best)
        hp = project_feasible(best + rng.normal(0.0, scale, n), n)
        alloc = min(55.0, max(20.0, 0.35 * (deadline - time.time())))
        if time.time() + alloc > deadline - 50:
            alloc = deadline - 50 - time.time()
        if alloc < 18:
            break
        cand, cand_c = refine_continuous(hp, n, time.time() + alloc, quick=True)
        if cand_c < best_c:
            best = cand
            best_c = cand_c

    n = len(best)
    if time.time() < deadline - 25:
        C = compute_all_lags(best, n)
        top = np.argsort(C)[-min(200, C.size):]
        cand = lbfgs_refine(best, n, 25000, top, 2000, 130, min(deadline - 5, time.time() + 30))
        c = compute_c5(cand, n)
        if c < best_c:
            best = cand
            best_c = c

    n = len(best)
    if time.time() < deadline - 35:
        C = compute_all_lags(best, n)
        top = np.argsort(C)[-min(50, C.size):]
        for alpha, mi in [(12000, 25), (60000, 30)]:
            if time.time() > deadline - 25:
                break
            cand = slsqp_top(best, n, alpha, top, mi)
            c = compute_c5(cand, n)
            if c < best_c:
                best = cand
                best_c = c

    if time.time() < deadline - 35:
        n = len(best)
        xb = make_binary(best, n)
        xb, cx = binary_local(xb, n, min(deadline - 8, time.time() + 35), rng)
        if cx < best_c:
            best = xb
            best_c = cx

    if time.time() < deadline - 18:
        best, best_c = refine_continuous(best, len(best), deadline - 2, quick=True)

    best = project_feasible(best, int(len(best)))
    best_c = compute_c5(best, len(best))
    return best.astype(float), float(best_c), int(len(best))
```
