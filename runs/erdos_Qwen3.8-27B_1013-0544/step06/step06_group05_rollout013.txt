def fast_swap_check(h, corr, i, j, critical_lags, n, current_max):
    """Check if swap (i:0->1, j:1->0) keeps all critical lags ≤ current_max.
    Returns True if feasible, False otherwise."""
    for k in critical_lags:
        L = k - (n - 1)
        # Compute delta for this lag
        delta = 0.0
        # old_i: h[i-L] if valid
        if 0 <= i - L < n:
            delta -= h[i - L]
        # old_j: (1-h[j+L]) if valid
        if 0 <= j + L < n:
            delta -= (1.0 - h[j + L])
        # new_i: (1-h'[i+L]) if valid
        if 0 <= i + L < n:
            pos = i + L
            if pos == j:
                delta += 1.0  # 1 - 0
            elif L == 0:
                delta += 0.0  # 1 - 1
            else:
                delta += (1.0 - h[pos])
        # new_j: h'[j-L] if valid
        if 0 <= j - L < n:
            pos = j - L
            if pos == i:
                delta += 1.0
            elif L == 0:
                delta += 0.0
            else:
                delta += h[pos]
        if corr[k] + delta > current_max + 1e-12:
            return False
    return True

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def swap_delta(h, i, j, L, n):
    d = 0.0
    if 0 <= i - L < n:
        d -= h[i - L]
    if 0 <= j + L < n:
        d -= 1.0 - h[j + L]
    if 0 <= i + L < n:
        p = i + L
        if p == j:
            d += 1.0
        elif L != 0:
            d += 1.0 - h[p]
    if 0 <= j - L < n:
        p = j - L
        if p == i:
            d += 1.0
        elif L != 0:
            d += h[p]
    return d

def ilds(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    corr = get_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    last_improve = time.time()
    stall = 0

    while time.time() < deadline:
        if time.time() - last_improve > 15.0:
            h = best_h.copy()
            corr = get_corr(h, n)
            cur_max = corr.max()
            stall = 0
            nz = np.where(h == 0)[0]
            no = np.where(h == 1)[0]
            npert = min(rng.randint(8, 16), len(nz), len(no))
            if npert > 0:
                pz = rng.choice(nz, npert, replace=False)
                po = rng.choice(no, npert, replace=False)
                for a, b in zip(pz, po):
                    h[a] = 1.0
                    h[b] = 0.0
                corr = get_corr(h, n)
                cur_max = corr.max()
            continue

        thresh = cur_max - 4.0
        crit = np.where(corr >= thresh)[0]
        improved = False
        zeros = list(np.where(h == 0)[0])
        ones = list(np.where(h == 1)[0])
        rng.shuffle(zeros)
        rng.shuffle(ones)

        for i in zeros:
            if h[i] != 0:
                continue
            if time.time() > deadline:
                return best_h, best_max
            for j in ones:
                if h[j] != 1:
                    continue
                ok = True
                for k in crit:
                    L = k - (n - 1)
                    if corr[k] + swap_delta(h, i, j, L, n) > cur_max + 1e-10:
                        ok = False
                        break
                if not ok:
                    continue
                h[i] = 1.0
                h[j] = 0.0
                nc = get_corr(h, n)
                nm = float(nc.max())
                if nm < cur_max - 1e-10:
                    corr = nc
                    cur_max = nm
                    improved = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                        last_improve = time.time()
                        stall = 0
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0

        if not improved:
            stall += 1
            nz = np.where(h == 0)[0]
            no = np.where(h == 1)[0]
            npert = min(rng.randint(3, 7) + stall, len(nz), len(no), 20)
            if npert > 0:

                po = rng.choice(no, npert, replace=False)
                for a, b in zip(pz, po):
                    h[a] = 1.0
                    h[b] = 0.0
                corr = get_corr(h, n)
                cur_max = corr.max()

    return best_h, best_max

def golden_init(n, seed=42):
    n_ones = n // 2
    rng = np.random.RandomState(seed)
    alpha = rng.uniform(0.3, 0.7)
    h = np.zeros(n)
    seen = set()
    k = 0
    while len(seen) < n_ones and k < n * 20:
        p = int((alpha * k) % n)
        if p not in seen:
            seen.add(p)
            h[p] = 1.0
        k += 1
    return h

def random_init(n, seed=42):
    h = np.zeros(n)
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h
