from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def full_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def local_search_strong(h, n, deadline, seed=42, max_rounds=200):
    """Strong local search using strategic 1-swaps."""
    dx = 2.0 / n
    corr = full_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Identify critical lags (within 1 of max)
        crit_threshold = cur_max - 1.0
        crit_lags = np.where(corr >= crit_threshold)[0]
        
        # For each critical lag, find "conflict" positions
        # Try to resolve conflicts by swapping
        for lag_idx in crit_lags:
            if time.time() > deadline:
                break
            s = lag_idx - (n - 1)  # actual shift
            # Find positions where h[i]=1 and h[i+s]=0
            if s >= 0:
                valid_i = np.arange(0, n - s)
            else:
                valid_i = np.arange(-s, n)
            
            h_i = h[valid_i]
            h_is = h[valid_i + s]
            conflicts = valid_i[(h_i == 1) & (h_is == 0)]
            
            for ci in conflicts:
                if time.time() > deadline:
                    break
                # Try to find a position j where h[j]=0 and moving 1 from ci to j
                # would reduce the critical lags
                # Simple heuristic: try random zero positions
                zeros = np.where(h == 0)[0]
                rng.shuffle(zeros)
                for j in zeros[:50]:
                    if j == ci:
                        continue
                    h[ci] = 0.0
                    h[j] = 1.0
                    new_corr = full_corr(h, n)
                    new_max = new_corr.max()
                    if new_max < cur_max:
                        cur_max = new_max
                        corr = new_corr
                        improved = True
                        if cur_max < best_max:
                            best_max = cur_max
                            best_h = h.copy()
                        break
                    else:
                        h[ci] = 1.0
                        h[j] = 0.0
                if improved:
                    break
        
        if not improved:
            # No improvement, try random perturbation + restart
            for _ in range(rng.randint(3, 10)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = rng.choice(ones)
                    b = rng.choice(zeros)
                    h[a], h[b] = h[b], h[a]
            corr = full_corr(h, n)
            cur_max = corr.max()
            # If worse than best, revert to best
            if cur_max > best_max:
                h = best_h.copy()
                corr = full_corr(h, n)
                cur_max = best_max
            else:
                # Accept and continue
                pass
    
    return best_h, float(best_max * dx)

    # h[pos+s] term: s ranges from -pos to n-1-pos
    s_min = -pos
    s_max = n - 1 - pos
    if s_min < s_max:
        idx = np.arange(s_min, s_max+1) + (n-1)
        vals = h[pos + np.arange(s_min, s_max+1)]
        vec[idx] += vals
    # h[pos-s] term: s ranges from pos-(n-1) to pos
    s_min2 = pos - (n-1)
    s_max2 = pos
    if s_min2 < s_max2:
        idx = np.arange(s_min2, s_max2+1) + (n-1)
        vals = h[pos - np.arange(s_min2, s_max2+1)]
        vec[idx] += vals
    return vec

```python
def efficient_local_search(h, n, deadline, seed=42):
    dx = 2.0 / n
    corr = full_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Precompute vec for all ones and zeros
        # vec_ones has shape (n_ones, 2n-1)
        # vec_zeros has shape (n_zeros, 2n-1)
        
        # For each one position a, compute vec_a
        # For each zero position b, compute vec_b
        
        # For swap (a, b): delta_c = vec_a - vec_b
        # new_c = corr + vec_a - vec_b
        # new_max = max(new_c)
        
        # We want min over (a,b) of new_max
        
        # This is a min-max problem. For a fixed a, we want:
        # min_b max_k (corr[k] + vec_a[k] - vec_b[k])
        
        # For a fixed a, this is equivalent to:
        # min_b max_k (corr[k] + vec_a[k] - vec_b[k])
        
        # The max is achieved at some k*. At k*, we want to minimize:
        # corr[k*] + vec_a[k*] - vec_b[k*]
        # So we want vec_b[k*] to be as large as possible.
        
        # But k* depends on b! So this is tricky.
        
        # Alternative: for each a, compute the "best" b by checking a few candidates.
        
        found_improvement = False
        for a in ones:
            if time.time() > deadline:
                break
            vec_a = compute_vec(a, h, n)
            base = corr + vec_a  # This is c + vec_a, we subtract vec_b
            for b in zeros:
                if time.time() > deadline:
                    break
                vec_b = compute_vec(b, h, n)
                new_max = np.max(base - vec_b)
                if new_max < cur_max:
                    h[a] = 0.0
                    h[b] = 1.0
                    corr = full_corr(h, n)
                    cur_max = corr.max()
                    found_improvement = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                    break
            if found_improvement:
                break
        
        if not found_improvement:
            # Stuck! Try perturbation
            n_pert = rng.randint(2, 8)
            for _ in range(n_pert):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                a = rng.choice(ones)
                b = rng.choice(zeros)
                h[a], h[b] = h[b], h[a]
            corr = full_corr(h, n)
            cur_max = corr.max()
            if cur_max > best_max * 1.01:  # Too bad, revert
                h = best_h.copy()
                corr = full_corr(h, n)
                cur_max = best_max
    
    return best_h, float(best_max * dx)
```

def efficient_local_search(h, n, deadline, seed=42):
    dx = 2.0 / n
    corr = full_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()

        zeros = np.where(h == 0)[0]

        n_zeros = len(zeros)

        # V[b_idx, k] = h[b+s] + h[b-s] for valid indices
        # Build V as a matrix
        V = np.zeros((n_zeros, 2*n-1))

            s_min = -b
            s_max = n - 1 - b

            # h[b-s] term

            V[b_idx, idx2] += h[b - np.arange(s_min2, s_max2+1)]

                break

            s_min = -a
            s_max = n - 1 - a

                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break  # Restart the sweep
        
        if not found:

                    corr = full_corr(h, n)
                    cur_max = best_max
                stagnant = 0
    
    return best_h, float(best_max * dx)

    zeros = np.where(h == 0)[0]

        k = s + (n-1)
        # (1-h[i+s]) term: valid when 0 <= i+s < n, i.e., max(0,-s) <= i < min(n, n-s)
        i_min = max(0, -s)
        i_max = min(n, n-s)
        if i_min < i_max:
            idx = np.arange(i_min, i_max)
            grad[k, idx] += (1 - h[idx + s])
        # -h[i-s] term: valid when 0 <= i-s < n, i.e., max(0,s) <= i < min(n, n+s)
        i_min2 = max(0, s)
        i_max2 = min(n, n+s)
        if i_min2 < i_max2:
            idx2 = np.arange(i_min2, i_max2)
            grad[k, idx2] -= h[idx2 - s]
    return grad

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()

        # grad of c w.r.t. x
        gc = np.zeros((L, n))

            i_max = min(n, n - s)

        return np.clip(res.x, 0.0, 1.0)
