import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slp_optimize(h, n, max_iter=50, deadline=None):
    """Successive Linear Programming"""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Build LP: min t s.t. g_m · h - t <= rhs_m, 0<=h<=1, sum h = n/2
        # g_m[j] = (1-h[j+d])*I(j in S) - h[j-d]*I(j-d in S) where d = m-(n-1), S = valid indices
        # rhs_m = g_m · h0 - c[m]
        
        m_count = 2 * n - 1
        A = np.zeros((m_count, n + 1))  # constraints: g_m · h - t <= rhs_m
        b = np.zeros(m_count)
        
        for m in range(m_count):
            d = m - (n - 1)
            # S = valid first indices: max(0,-d) <= i <= min(n-1, n-1-d)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue  # no valid pairs, c[m] = 0
            # g_m[j] for j in range(n):
            gm = np.zeros(n)
            # First part: (1-h[j+d]) for j in S
            for j in range(i_lo, i_hi + 1):
                gd = j + d
                if 0 <= gd < n:
                    gm[j] += (1.0 - h[gd])
            # Second part: -h[j-d] for j-d in S, i.e., j in S+d = [i_lo+d, i_hi+d]
            j_lo2 = max(0, i_lo + d)
            j_hi2 = min(n - 1, i_hi + d)
            for j in range(j_lo2, j_hi2 + 1):
                gd = j - d
                if 0 <= gd < n and i_lo <= gd <= i_hi:
                    gm[j] -= h[gd]
            A[m, :n] = gm
            A[m, n] = -1.0  # -t
            b[m] = gm @ h - c[m]
        
        # Add sum constraint: sum h = n/2 (equality)
        # linprog handles equality constraints via A_eq, b_eq
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(None, None)]  # h bounds, t unbounded
        
        try:
            res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                         A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = project_feasible(res.x[:n], n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    h = h_new
                    best_c = c_new
            else:
                break
        except Exception:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, n, max_passes=200, grid=200, deadline=None, rng=None, penalty=50.0):

    c = np.correlate(h, 1.0 - h, mode='full')
    sumh = h.sum()
    target = n / 2.0
    best_c = float(c.max())
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d = h[p]
            if d < 1e-12 or d > 1.0 - 1e-12:
                # Still try to move (to interior)
                pass
            
            d_lo = p - n + 1
            d_hi = n - 1 - p

def slp_optimize(h, n, max_iter=30, deadline=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Build LP
        m_count = 2 * n - 1
        A = np.zeros((m_count, n + 1))
        b = np.zeros(m_count)
        
        for m in range(m_count):
            dl = m - (n - 1)
            i_lo = max(0, -dl)
            i_hi = min(n - 1, n - 1 - dl)
            if i_lo > i_hi:
                continue
            gm = np.zeros(n)
            # First part: (1-h[j+dl]) for j in [i_lo, i_hi]
            if dl >= 0:
                # j+dl ranges from i_lo+dl to i_hi+dl
                gm[i_lo:i_hi+1] += (1.0 - h[i_lo+dl:i_hi+1+dl])
                # Second part: -h[j-dl] for j-dl in [i_lo, i_hi] → j in [i_lo+dl, i_hi+dl]
                j_lo2 = i_lo + dl
                j_hi2 = i_hi + dl
                if j_lo2 < n:
                    gm[j_lo2:min(j_hi2+1, n)] -= h[j_lo2-dl:min(j_hi2+1, n)-dl]
            else:
                # dl < 0, let dl_abs = -dl
                # j+dl = j-dl_abs, ranges from i_lo-dl_abs to i_hi-dl_abs
                gm[i_lo:i_hi+1] += (1.0 - h[i_lo-dl:i_hi+1-dl])
                # Second part: -h[j-dl] = -h[j+dl_abs] for j-dl in [i_lo, i_hi] → j in [i_lo+dl, i_hi+dl]
                j_lo2 = i_lo + dl  # negative
                j_hi2 = i_hi + dl  # = n-1+dl = n-1-dl_abs
                if j_hi2 >= 0:
                    j_lo2 = max(0, j_lo2)
                    gm[j_lo2:j_hi2+1] -= h[j_lo2-dl:j_hi2+1-dl]
            
            A[m, :n] = gm
            A[m, n] = -1.0
            b[m] = gm @ h - c[m]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        obj = np.zeros(n + 1)
        obj[n] = 1.0
        
        bounds = [(0, 1)] * n + [(None, None)]
        
        try:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = project_feasible(res.x[:n], n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-10:
                    h = h_new
                    best_c = c_new
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsumexp = np.log(np.sum(np.exp(shifted)))
    val = (m + logsumexp / alpha) * (2.0 / n)
    # Add constraint violation penalty
    penalty = 1000.0 * (x.sum() - n / 2.0)**2
    return val + penalty

    grad = np.zeros(n)
    for mm in range(2*n-1):
        dl = mm - (n-1)
        w = weights[mm]
        if w < 1e-10:
            continue
        i_lo = max(0, -dl)
        i_hi = min(n-1, n-1-dl)
        if i_lo > i_hi:
            continue
        # First part
        if dl >= 0:
            grad[i_lo:i_hi+1] += w * (1.0 - x[i_lo+dl:i_hi+1+dl])
            j_lo2 = i_lo + dl
            j_hi2 = i_hi + dl
            if j_lo2 < n:
                grad[j_lo2:min(j_hi2+1, n)] -= w * x[j_lo2-dl:min(j_hi2+1, n)-dl]
        else:
            grad[i_lo:i_hi+1] += w * (1.0 - x[i_lo-dl:i_hi+1-dl])
            j_lo2 = max(0, i_lo + dl)
            j_hi2 = i_hi + dl
            if j_hi2 >= 0:
                grad[j_lo2:j_hi2+1] -= w * x[j_lo2-dl:j_hi2+1-dl]
    grad *= (2.0 / n) / alpha
    # Constraint gradient
    grad += 2 * 1000.0 * (x.sum() - n/2.0)
    return grad

def lbfgs_polish(h, n, deadline=None, alpha=50000):
    h = project_feasible(h.copy(), n)
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha),
                      method='L-BFGS-B', bounds=[(0,1)]*n,
                      options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-8})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h
