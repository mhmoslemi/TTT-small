```python
def subgradient_method(h, n, max_iter, deadline, alpha0=0.01):
    dx = 2.0 / n
    for t in range(max_iter):
        if time.time() > deadline:
            break
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Find active sets (within tolerance of max)
        m = corr.max()
        active = np.where(corr >= m - 0.001)[0]
        # Compute subgradient (average over active)
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            # ∂g_k/∂h_i = dx * [(1 - h[i+lag]) - h[i-lag]]
            for i in range(n):
                val = 0.0
                if 0 <= i + lag < n:
                    val += (1.0 - h[i + lag])
                if 0 <= i - lag < n:
                    val -= h[i - lag]
                grad[i] += val
        grad /= len(active)
        # Update
        alpha = alpha0 / (1 + t * 0.01)
        h = h - alpha * grad
        h = project_feasible(h, n)
    return h
```

```python
def slsqp_refine(h, n, deadline, maxiter=5000):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 10000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 10000.0
        sh = alpha * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        softmax = exp_sh / exp_sh.sum()
        # Gradient of the smooth max
        grad = np.zeros_like(x)
        for k_idx, w in enumerate(softmax):
            if w < 1e-10:
                continue
            lag = k_idx - (n - 1)
            j_range = np.arange(n)
            idx1 = j_range + lag
            idx2 = j_range - lag
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            if v1.any():
                grad[v1] += w * (1.0 - x[idx1[v1]]) * dx
            if v2.any():
                grad[v2] -= w * x[idx2[v2]] * dx
        return grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

- l+k = i, i.e., l = i-k: contributes h[i-k]*(1-h[i]) → changes from h[i-k]*1 to h[i-k]*0 = 0

So the change when h[i] goes 0→1 is:
Δcorr_k = (1-h[i+k]) - h[i-k] (for valid indices)

Similarly, when h[j] goes 1→0:
Δcorr_k = -(1-h[j+k]) + h[j-k] = -(1-h[j+k]) + h[j-k]

So total change for swap (i: 0→1, j: 1→0):
Δcorr_k = [(1-h[i+k]) - h[i-k]] + [-(1-h[j+k]) + h[j-k]]

This is O(1) per lag! So I can compute the change in all correlations in O(n) time (one pass over all lags).

Then I need to check if the new max is less than the current max. The new max is:
new_max = max over k of (corr_k + Δcorr_k)

But I can't just look at the lags that changed; the max might be at a lag that didn't change. However, the new max is at most max(old values at unchanged lags, new values at changed lags).

The old values at unchanged lags are at most cur_max. The new values at changed lags are corr_k + Δcorr_k.

So: new_max = max(cur_max, max over changed lags of (corr_k + Δcorr_k))

Wait, that's not right either. The lags that change are those where i±k or j±k is in range. For a given swap, the lags that change are: k = i-l for all l, and k = j-l for all l... hmm, this is getting complicated.

Actually, for a given pair (i, j), the lags that are affected are limited. Let me think again.

corr_k = sum_l h[l]*(1-h[l+k])

The change Δcorr_k for the swap (i: 0→1, j: 1→0) is:
Δcorr_k = [(1-h[i+k]) if i+k valid] - [h[i-k] if i-k valid] + [-(1-h[j+k]) if j+k valid] + [h[j-k] if j-k valid]

This is non-zero only when i+k, i-k, j+k, or j-k is in [0, n-1]. For a given k, this is determined by the positions of i and j relative to k.

For all k from -(n-1) to (n-1) (which is 2n-1 total lags), Δcorr_k is computed in O(1). So the total cost to evaluate one swap is O(n) (to compute all Δcorr_k) + O(n) (to find the new max) = O(n).

```python
def eval_swap(h, i, j, n, corr, cur_max):
    """Evaluate the effect of swapping h[i] (0→1) and h[j] (1→0).
    Returns (new_max, changes) where changes can be used to update corr."""
    # h[i] is currently 0, h[j] is currently 1
    # After swap: h[i] = 1, h[j] = 0
    new_corr = corr.copy()
    for k_idx in range(2*n - 1):
        lag = k_idx - (n - 1)
        delta = 0.0
        if 0 <= i + lag < n:
            delta += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            delta -= h[i - lag]
        if 0 <= j + lag < n:
            delta -= (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            delta += h[j - lag]
        if delta != 0:
            new_corr[k_idx] += delta
    
    new_max = new_corr.max()
    return new_max, new_corr
```

```python
    v_j1 = (idx_j1 >= 0) & (idx_j1 < n)

    delta[v_j1] -= 1.0 - h[idx_j1[v_j1]]
    delta[v_j2] += h[idx_j2[v_j2]]
    
    return delta
```

```python
            if cur_max < best_max:
                best_max = cur_max
                best = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur_corr -= delta
            no_improve += 1
        T = max(T * 0.99999, 0.02)
    return best, float(best_max * dx)

def slsqp_refine(h, n, deadline, alpha_smooth=5000.0, maxiter=5000):
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    idx_range = np.arange(n)

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha_smooth * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha_smooth

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha_smooth * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        softmax = exp_sh / exp_sh.sum()
        grad = np.zeros(n)
        for k_idx in range(len(softmax)):
            w = softmax[k_idx]
            if w < 1e-6:
                continue
            lag = lags[k_idx]
            ii1 = idx_range + lag
            ii2 = idx_range - lag
            v1 = (ii1 >= 0) & (ii1 < n)
            v2 = (ii2 >= 0) & (ii2 < n)
            if v1.any():
                grad[v1] += w * (1.0 - x[ii1[v1]]) * dx
            if v2.any():
                grad[v2] -= w * x[ii2[v2]] * dx
        return grad

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_refine(h, n, deadline):
    dx = 2.0 / n
    h = h.copy().astype(float)
    lags = np.arange(-(n - 1), n)
    idx_range = np.arange(n)
    for t in range(300000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        g_max = corr.max()
        active = np.where(corr >= g_max - 0.005)[0]
        grad = np.zeros(n)
        for k_idx in active:
            lag = lags[k_idx]
            ii1 = idx_range + lag
            ii2 = idx_range - lag
            v1 = (ii1 >= 0) & (ii1 < n)
            v2 = (ii2 >= 0) & (ii2 < n)
            if v1.any():
                grad[v1] += (1.0 - h[ii1[v1]])
            if v2.any():
                grad[v2] -= h[ii2[v2]]
        grad = grad * dx / max(len(active), 1)
        alpha = 0.02 / (1.0 + 0.0005 * t)
        h = h - alpha * grad
        h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0_full = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0_full)
    except Exception:
        h0_full = None
        n0 = 0

    best_c = float('inf')
    best_h = None
    best_n = 250

    n_candidates = []
    if n0 > 0:
        n_candidates.append(n0)
    for nc in [250, 200, 300]:
        if nc not in n_candidates:
            n_candidates.append(nc)

    total_time = dl - t0 - 30
    time_per_n = total_time / len(n_candidates)

    for ni, n in enumerate(n_candidates):
        phase_start = t0 + 30 + ni * time_per_n
        phase_end = phase_start + time_per_n
        if time.time() > phase_end - 10:
            continue

        if n == n0 and h0_full is not None:
            h_init = h0_full.copy()
        elif h0_full is not None:
            h_init = np.interp(np.linspace(0, n0 - 1, n), np.arange(n0), h0_full)
            h_init = project_feasible(h_init, n)
        else:
            h_init = np.full(n, 0.5)

        h_bin = binarize(h_init, n)
        c_init = compute_c5(h_bin, n)
        if c_init < best_c:
            best_c = c_init
            best_h = h_bin.copy()
            best_n = n

        # SA phase
        sa_end = phase_start + time_per_n * 0.45
        h_sa, c_sa = sim_anneal(h_bin, n, min(sa_end, dl - 20), seed + ni * 7)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
            best_n = n

        # SLSQP with multiple alpha values
        slq_end = phase_start + time_per_n * 0.75
        for alpha_s in [2000.0, 10000.0, 50000.0]:
            if time.time() > slq_end:
                break
            h_slq = slsqp_refine(h_sa, n, min(slq_end, dl - 15), alpha_smooth=alpha_s, maxiter=4000)
            c_slq = compute_c5(h_slq, n)
            if c_slq < best_c:
                best_c = c_slq
                best_h = h_slq.copy()
                best_n = n

        # Subgradient refinement
        if time.time() < phase_end - 5:
            h_sg = subgradient_refine(best_h if best_n == n else h_slq, n, min(phase_end - 2, dl - 10))
            c_sg = compute_c5(h_sg, n)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg.copy()
                best_n = n

        # Re-binarize + quick SA
        if time.time() < phase_end - 1:
            h_bin2 = binarize(best_h if best_n == n else h_sv, n)
            h_sa2, c_sa2 = sim_anneal(h_bin2, n, min(phase_end, dl - 5), seed + 50000 + ni)
            if c_sa2 < best_c:
                best_c = c_sa2
                best_h = h_sa2.copy()
                best_n = n

    # Final polish
    if best_h is not None and time.time() < dl - 8:
        h_f = slsqp_refine(best_h, best_n, dl - 3, alpha_smooth=100000.0, maxiter=5000)
        c_f = compute_c5(h_f, best_n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
