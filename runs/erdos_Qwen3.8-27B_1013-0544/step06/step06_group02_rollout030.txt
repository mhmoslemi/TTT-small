def grad_from_weights(h, n, weights):
    dx = 2.0 / float(n)
    center = n - 1
    wp = np.zeros(n, dtype=float)
    wn = np.zeros(n, dtype=float)
    if n > 1:
        wp[1:] = weights[:center][::-1]
        wn[1:] = weights[center+1:]
    w0 = float(weights[center]) if center < len(weights) else 0.0
    if n == 1:
        return np.array([w0 * (1.0 - 2.0*h[0]) * dx])
    A = wp + wn
    Sf = np.convolve(A[::-1], h, mode='full')
    Sf = Sf[center:center+n]
    D = wn - wp
    Sb = np.convolve(D, h, mode='full')
    Sb = Sb[:n]
    cumwp = np.cumsum(wp)
    pre = cumwp[center::-1]  # length n
    suf = np.cumsum(wn)
    grad = w0 * (1.0 - 2.0*h) + (pre + suf) - Sf + Sb
    return grad * dx
```

import numpy as np
import time, math
from scipy.optimize import minimize, minimize_scalar

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy(); target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - float(h.sum())
        if abs(d) < 1e-14: break
        if d > 0:
            m = h < 1.0 - 1e-15; c = int(m.sum())
            if not c: break
            h[m] = np.minimum(1.0, h[m] + d / c)
        else:
            m = h > 1e-15; c = int(m.sum())
            if not c: break
            h[m] = np.maximum(0.0, h[m] + d / c)
    for _ in range(50):
        d = target - float(h.sum())
        if abs(d) < 1e-15: break
        m = (h > 1e-13) & (h < 1.0 - 1e-13); c = int(m.sum())
        if c:
            h[m] += d / c
        else:
            idx = int(np.argmin(1.0 - h) if d > 0 else np.argmax(h))
            if 0.0 <= h[idx] + d <= 1.0: h[idx] += d
            else: break
        h = np.clip(h, 0.0, 1.0)
    return h

def c5vec(h, n):
    dx = 2.0 / float(n)
    b = 1.0 - h
    return np.correlate(h, b, mode='full') * dx

def c5(h, n):
    return float(np.max(c5vec(h, n)))

def smooth(c, alpha):
    cm = float(np.max(c))
    if alpha <= 1e-12:
        w = np.zeros(c.shape); w[int(np.argmax(c))] = 1.0
        return cm, w
    v = np.clip(alpha * (c - cm), -745.0, 0.0)
    e = np.exp(v); s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(c.shape); w[int(np.argmax(c))] = 1.0; return cm, w
    return cm + float(np.log(s)) / alpha, e / s

def grad_w(h, n, w):
    dx = 2.0 / float(n); c0 = n - 1
    wp = np.zeros(n); wn = np.zeros(n)
    if n > 1:
        wp[1:] = w[:c0][::-1]
        wn[1:] = w[c0+1:]
    w0 = float(w[c0])
    A = wp + wn
    Sf = np.convolve(A[::-1], h, mode='full')[c0:c0+n]
    D = wn - wp
    Sb = np.convolve(D, h, mode='full')[:n]
    pre = np.cumsum(wp)[c0::-1]
    suff = np.cumsum(wn)
    g = w0 * (1.0 - 2.0 * h) + (pre + suff) - Sf + Sb
    return g * dx

def f_lb(x, n, a, mu):
    v, _ = smooth(c5vec(x,n), a)
    d = float(x.sum()) - 0.5*n
    return float(v + mu*d*d)

def g_lb(x, n, a, mu):
    _, w = smooth(c5vec(x,n), a)
    g = grad_w(x,n,w)
    g += 2.0*mu*(float(x.sum())-0.5*n)
    return g

def lbfgs(h, n, deadline, alpha, mu):
    h = project_feasible(h,n); best = c5(h,n)
    if n <= 300: base=500
    elif n <= 450: base=350
    elif n <= 650: base=250
    else: base=150
    bounds=[(0.0,1.0)]*n; no=0
    while time.time() < deadline - 2.0:
        rem = deadline - time.time() - 2.0
        if rem <= 0: break
        per = 0.006 + 1.2e-5*float(n)
        mi = int(min(base, max(40, rem/per)))
        try:
            r = minimize(f_lb, h, jac=g_lb, args=(n,alpha,mu), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter':mi,'ftol':1e-12,'gtol':1e-7,'maxls':50})
            h2 = project_feasible(r.x,n); c2 = c5(h2,n)
        except Exception:
            break
        if c2 < best - 1e-14:
            h, best, no = h2, c2, 0
        else:
            no += 1
            if no >= 3: break
        if r.status == 1 and no > 0: break
    return h, best

def f_sl(x, n, a):
    return float(smooth(c5vec(x,n), a)[0])

def g_sl(x, n, a):
    _, w = smooth(c5vec(x,n), a)
    return grad_w(x,n,w)

def eqf(x,n): return float(x.sum()-0.5*n)
def eqj(x,n): return np.ones((1,n), dtype=float)

def slsqp(h,n,deadline,alpha):
    h=project_feasible(h,n); best=c5(h,n)
    bounds=[(0.0,1.0)]*n
    con=[{'type':'eq','fun':eqf,'jac':eqj,'args':(n,)}]
    base = 80 if n <= 350 else 50
    no=0
    while time.time() < deadline - 2.0:
        rem=deadline-time.time()-2.0
        if rem <= 0: break
        per=0.035 + 1e-5*float(n)
        mi=int(min(base, max(10, rem/per)))
        try:
            r=minimize(f_sl,h,jac=g_sl,args=(n,alpha),method='SLSQP',bounds=bounds,constraints=con,options={'maxiter':mi,'ftol':1e-12})
            h2=project_feasible(r.x,n); c2=c5(h2,n)
        except Exception:
            break
        if c2 < best - 1e-14:
            h,best,no=h2,c2,0
        else:
            no+=1
            if no>=3: break
        if r.status == 1 and no>0: break
    return h,best

def movef(a,h,n,s,t):
    v=h.copy(); v[s]-=a; v[t]+=a
    return c5(np.clip(v,0.0,1.0), n)

def fw(h,n,deadline,k=24):
    h=project_feasible(h,n); best=c5(h,n); no=0
    while time.time() < deadline - 2.0 and no < 4:
        cv=c5vec(h,n); cm=cv.max()
        if cv.size > 1:
            sm=float(np.partition(cv.copy(), -2)[-2]); gap=max(cm-sm,1e-15)
        else: gap=1e-15
        alpha=max(1000.0, min(5000000.0, 50.0/gap))
        _, w = smooth(cv, alpha)
        grad=grad_w(h,n,w)
        order=np.argsort(grad); targets=[]; sources=[]
        for i in order:
            if h[i] < 1.0 - 1e-8:
                targets.append(int(i))
                if len(targets)>=k: break
        for i in order[::-1]:
            if h[i] > 1e-8:
                sources.append(int(i))
                if len(sources)>=k: break
        improved=False; tested=0
        for s in sources:
            if time.time() > deadline - 3.0: break
            gs=float(grad[s])
            for t in targets:
                if time.time() > deadline - 3.0: break
                if s==t or gs <= float(grad[t]) + 1e-10: continue
                hi=min(float(h[s]), 1.0-float(h[t]))
                if hi <= 1e-10: continue
                tested += 1
                la=0.0; lc=best
                for a in (0.5*hi, hi):
                    v=h.copy(); v[s]-=a; v[t]+=a; v=np.clip(v,0.0,1.0)
                    cc=c5(v,n)
                    if cc < lc - 1e-13:
                        lc=cc; la=a
                if hi > 1e-5:
                    try:
                        r=minimize_scalar(movef, bounds=(0.0,hi), args=(h,n,s,t), method='bounded', options={'xatol':1e-10,'maxiter':25})
                        a=float(r.x)
                        if 0.0 <= a <= hi:
                            v=h.copy(); v[s]-=a; v[t]+=a; v=np.clip(v,0.0,1.0)
                            cc=c5(v,n)
                            if cc < lc - 1e-13:
                                lc=cc; la=a
                    except Exception:
                        pass
                if lc < best - 1e-12:
                    h[s]-=la; h[t]+=la
                    h2=project_feasible(h,n); cc=c5(h2,n)
                    if cc < best - 1e-12:
                        h=h2; best=cc; improved=True; break
                    else:
                        h[s]+=la; h[t]-=la
                if tested >= 900: break
            if improved: break
        if improved: no=0
        else:
            no += 1
            if no >= 3: break
    return h,best

def resample(h,no,nn):
    if no == nn: return h.copy()
    x0=np.linspace(0,2,no,endpoint=False); x1=np.linspace(0,2,nn,endpoint=False)
    return np.interp(x1,x0,h)

def default_initial(n):
    x=np.linspace(0,2,n,endpoint=False)
    h=0.5+0.5*np.cos(2.0*np.pi*3.0*x/2.0)
    return project_feasible(h,n)

def sweep(h,n,deadline,alphas=None,fw_frac=0.25,k=24):
    h=project_feasible(h,n); best=c5(h,n)
    now=time.time()
    if deadline <= now+5.0: return h,best
    if alphas is None: alphas=[100.0,1000.0,10000.0,100000.0,1000000.0,5000000.0]
    total=deadline-now; fwt=0.0
    if total > 60.0:
        fwt=min(total*fw_frac,120.0)
        if total-fwt < 30.0: fwt=max(0.0,total-30.0)
    tstop=deadline-fwt; t0=time.time(); base=max(0.0,tstop-t0); num=len(alphas)
    for i in range(num):
        now=time.time()
        if tstop <= now+2.0: break
        if i < num-1:
            se=t0+base*((i+1)/float(num))
            if se <= now+3.0: continue
        else:
            se=tstop
        a=alphas[i]; mu=float(5.0 + math.sqrt(a))
        if mu > 20000.0: mu=20000.0
        h,best=lbfgs(h,n,se,a,mu)
    if fwt > 10.0 and time.time() < deadline - 5.0:
        h,best=fw(h,n,deadline-2.0,k)
    return h,best
```

```python
def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    bad = ~np.isfinite(h)
    if bad.any():
        h[bad] = 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - float(h.sum())
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            m = h < 1.0 - 1e-15
            cnt = int(m.sum())
            if not cnt:
                break
            h[m] = np.minimum(1.0, h[m] + d / cnt)
        else:
            m = h > 1e-15
            cnt = int(m.sum())
            if not cnt:
                break
            h[m] = np.maximum(0.0, h[m] + d / cnt)
    for _ in range(50):
        d = target - float(h.sum())
        if abs(d) < 1e-15:
            break
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(m.sum())
        if cnt:
            h[m] += d / cnt
        else:
            idx = int(np.argmin(1.0 - h) if d > 0.0 else np.argmax(h))
            if 0.0 <= h[idx] + d <= 1.0:
                h[idx] += d
            else:
                break
        h = np.clip(h, 0.0, 1.0)
    return h


def c5vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def c5val(h, n):
    return float(np.max(c5vec(h, n)))


def smooth(c, alpha):
    cm = float(np.max(c))
    if alpha <= 1e-12:
        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return cm, w

        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    return cm + float(np.log(s)) / alpha, e / s


def grad_w(h, n, w):
    dx = 2.0 / float(n)
    c0 = n - 1
    wp = np.zeros(n, dtype=float)
    wn = np.zeros(n, dtype=float)
    if n > 1:
        wp[1:] = w[:c0][::-1]
        wn[1:] = w[c0 + 1:]
    w0 = float(w[c0])
    A = wp + wn
    Sf = np.convolve(A[::-1], h, mode='full')[c0:c0 + n]
    D = wn - wp
    Sb = np.convolve(D, h, mode='full')[:n]
    pre = np.cumsum(wp)[c0::-1]
    suf = np.cumsum(wn)
    g = w0 * (1.0 - 2.0 * h) + (pre + suf) - Sf + Sb
    return g * dx


def f_lb(x, n, a, mu):
    v, _ = smooth(c5vec(x, n), a)
    d = float(x.sum()) - 0.5 * float(n)
    return float(v + mu * d * d)


def g_lb(x, n, a, mu):
    _, w = smooth(c5vec(x, n), a)
    g = grad_w(x, n, w)
    g += 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return g

def resample(h, no, nn):
    if no == nn:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, no, endpoint=False)
    x1 = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(x1, x0, h)
```
