from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-15: break
            h += d / r * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-15: break
            h += d / s * h
    return np.clip(h, 0.0, 1.2)  # allow slight overshoot, clip later

def project_feasible_strict(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            sr = r.sum()
            if sr < 1e-15: break
            h += d / sr * r
        else:
            s = h.sum()
            if s < 1e-15: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def smooth_max_lse(c, alpha):
    m = c.max()
    sh = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(sh).sum()) / alpha

def obj_pen(h, n, alpha, pen_w):
    dx = 2.0 / n
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    sm = smooth_max_lse(c, alpha) * dx
    pen = pen_w * (h.sum() - n/2.0)**2
    return sm + pen

def grad_ann(h, n, alpha, pen_w):
    """Analytical gradient of the penalized smooth max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    m = c.max()
    sh = np.clip(alpha * (c - m), -400, 400)
    ew = np.exp(sh)
    w = ew / ew.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wj = w[k]
        if wj < 1e-10:
            continue
        j = k - (n-1)
        # term1: (1-h[m-j]) for valid m-j
        if j >= 0:
            grad[j:] += wj * (1.0 - h[:n-j])
        else:
            grad[:n+j] += wj * (1.0 - h[-j:])
        # term2: -h[m+j] for valid m+j
        if j >= 0:
            grad[:n-j] -= wj * h[j:]
        else:
            grad[-j:] -= wj * h[:n+j]
    
    grad *= dx
    grad += 2.0 * pen_w * (h.sum() - n/2.0)
    return grad

def do_lbfgs(h, n, alpha, maxiter, pen_w, deadline):
    def f(x):
        return obj_pen(x, n, alpha, pen_w)
    def g(x):
        return grad_ann(x, n, alpha, pen_w)
    
    res = minimize(f, h, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                            'maxls': 20, 'maxfun': maxiter*3, 'iprint': -1})
    if time.time() > deadline:
        return h
    return project_feasible_strict(res.x, n)

def binary_ls(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    cr = np.correlate(h, 1-h, mode='full').astype(float)
    cur = cr.max() * dx
    bh, bc = h.copy(), cur
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        imp = False
        zeros = np.where(h==0)[0]
        ones = np.where(h==1)[0]
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = np.zeros(2*n-1)
                im = i + m_range
                mk = (im>=0) & (im<n)
                d[mk] += 1.0 - h[im[mk]]
                im2 = i - m_range
                mk = (im2>=0) & (im2<n)
                d[mk] -= h[im2[mk]]
                jm = j + m_range
                mk = (jm>=0) & (jm<n)
                d[mk] -= 1.0 - h[jm[mk]]
                jm2 = j - m_range
                mk = (jm2>=0) & (jm2<n)
                d[mk] += h[jm2[mk]]
                d[n-1] -= 2.0
                d[n-1+(j-i)] += 1.0
                d[n-1+(i-j)] += 1.0
                nc = (cr + d).max() * dx
                if nc < cur - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    cr += d
                    cur = nc
                    imp = True
                    break
            if imp:
                break
        if not imp:
            break
        if cur < bc:
            bc, bh = cur, h.copy()
    return bh, bc
