def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        st = 1e-15
        if abs(diff) < st:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < st:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < st:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    bounds = [(0.0, 1.0)] * n

            break

                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                   'maxfun': maxiter * 20})
            h_new = project_feasible(res.x, n)

            if c_new < best_c - 1e-15:

def coordinate_descent(h, n, deadline):

Let me try a fundamentally different approach:

**Approach: Direct minimax with active set**

1. Start with initial_h_values
2. Compute all shift values c_k
3. Identify the "active set" A = {k : c_k is within ε of max}
4. Solve the subproblem: minimize max_{k∈A} c_k subject to constraints
   - This can be done with a QP or even a simple averaging approach
5. Update A and repeat

For step 4, a simple approach:
- Compute the average of the active shifts: c_avg = mean(c_k for k in A)
- Compute the gradient of c_avg (which is the average of individual gradients)
- Take a step in the negative gradient direction
- This tends to reduce all active constraints simultaneously
