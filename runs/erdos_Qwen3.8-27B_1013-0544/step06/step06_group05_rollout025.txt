def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    dx = 2.0 / n

        top_idx = np.argsort(p)[-top_k:]
        p_top = p[top_idx]
        p_top = p_top / p_top.sum()
        use_idx = top_idx
        use_p = p_top
    else:
        use_idx = np.arange(len(p))
        use_p = p
    
    grad = np.zeros(n)
    for k, w in zip(use_idx, use_p):
        if w < 1e-10:

        offset = k - (n - 1)

        # For i in range(max(0, -offset), min(n, n-offset)):

        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:

        # Compute h[i-offset] term
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-offset:hi2-offset]
    return grad

    """Projected gradient descent with backtracking line search."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = smooth_max_grad(cur, n, alpha, top_k)
        
        # Line search (backtracking)
        if lr is None:
            step = 1.0 / alpha
        else:
            step = lr
        
        improved = False
        for _ in range(30):
            new_h = project_feasible(cur - step * grad, n)
            new_c = compute_c5(new_h, n)
            # Check if smooth objective decreased
            old_obj = smooth_max_obj(cur, n, alpha)
            new_obj = smooth_max_obj(new_h, n, alpha)
            if new_obj < old_obj - 1e-12 * abs(old_obj):
                cur = new_h
                improved = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur.copy()
                break
            step *= 0.5
        
        if not improved:
            break
    
    return best_h, best_c

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = 1.0 / (alpha * 3.0 * n + 1.0)
        ok = False
        for _ in range(50):
            nh = project_feasible(cur - step * grad, n)
            no = obj_and_grad(nh, n, alpha, top_k)[0]
            if no <= prev_obj + 1e-15:
                cur, prev_obj = nh, no
                nc = compute_c5(cur, n)
                if nc < best_c:
                    best_c, best_h = nc, cur.copy()
                    stall = 0
                else:
                    stall += 1
                ok = True
                break
            step *= 0.5
        if not ok:
            break
        if stall > 200:
            break
    return best_h, best_c

def binary_refine(h, n, deadline, seed=42):
    dx = 2.0 / n
    srt = np.argsort(h)[::-1]
    cur = np.zeros(n)
    cur[srt[:n // 2]] = 1.0
    corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cm = corr.max()
    best_c, best_h = cm * dx, cur.copy()
    rng = np.random.RandomState(seed)
    stall = 0
    while stall < 500:
        if time.time() > deadline:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        for _ in range(100):
            if time.time() > deadline:
                break
            i, j = rng.choice(zeros), rng.choice(ones)
            cur[i], cur[j] = 1.0, 0.0
            nc = np.correlate(cur, 1.0 - cur, mode='full').max()
            if nc < cm - 0.5:
                cm = float(nc)
                if cm * dx < best_c:
                    best_c, best_h = cm * dx, cur.copy()
                found = True
                stall = 0
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                break
            cur[i], cur[j] = 0.0, 1.0
        if not found:
            stall += 1
    return best_h, best_c

def perturb(h, n, n_flips, seed):
    rng = np.random.RandomState(seed)
    cur = h.copy()
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur < 0.5)[0]
    for _ in range(n_flips):
        if len(ones) == 0 or len(zeros) == 0:
            break
        cur[rng.choice(ones)] = 0.0
        cur[rng.choice(zeros)] = 1.0
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
    return cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Multi-restart PGD (50% time)
    p1 = t0 + 0.50 * (budget_s - 5)
    cur_h = best_h.copy()
    for restart in range(6):
        if time.time() > p1:
            break
        if restart == 0:
            start = cur_h.copy()
        elif restart == 1:
            start = perturb(best_h, n, 5, seed + 1)
        elif restart == 2:
            start = perturb(best_h, n, 10, seed + 2)
        elif restart == 3:
            srt = np.argsort(best_h)[::-1]
            start = np.zeros(n)
            start[srt[:n // 2]] = 1.0
        else:
            start = perturb(best_h, n, 3, seed + 3 + restart)
        for a in [20, 50, 100, 300, 1000, 3000, 10000]:
            if time.time() > p1:
                break
            hn, cn = pgd(start, n, a, 5000, p1, top_k=25)
            if cn < best_c:
                best_c, best_h = cn, hn
            start = hn
    # Phase 2: Binary local search (30% time)
    p2 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p2:
        hb, cb = binary_refine(best_h, n, p2, seed=seed)
        if cb < best_c:
            best_c, best_h = cb, hb
    # Phase 3: Final high-alpha PGD polish (20% time)
    if time.time() < dl - 2:
        for a in [5000, 15000, 50000]:
            if time.time() > dl - 2:
                break
            hn, cn = pgd(best_h.copy(), n, a, 4000, dl - 2, top_k=15)
            if cn < best_c:
                best_c, best_h = cn, hn
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
