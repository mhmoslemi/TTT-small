        # For efficiency, recompute each time
        d_vals = np.arange(-(n-1), n)  # shape (2n-1,)
        # A[p, m] where m corresponds to d_vals[m]
        # source term: (1-h[p+d]) if 0<=p+d<n
        # target term: h[p-d] if 0<=p-d<n

        # p is 0..n-1, d is -(n-1)..(n-1)
        # p+d ranges from -(n-1) to 2n-2
        # We need 0 <= p+d < n

        d_row = d_vals[None, :]  # (1, 2n-1)

        pmd = p_col - d_row  # (n, 2n-1) = p - d

        valid_pmd = (pmd >= 0) & (pmd < n)
        
        # h[p+d] where valid, else 0
        h_pd = np.where(valid_pd, h[pd.clip(0, n-1)], 0.0)
        h_pmd = np.where(valid_pmd, h[pmd.clip(0, n-1)], 0.0)
        
        A = (1.0 - h_pd) * valid_pd - h_pmd * valid_pmd  # (n, 2n-1)

        # Δc = A[i, :] - A[j, :]
        # new_max = (c + Δc).max()
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_new_M = M
        best_i = -1
        best_j = -1
        
        for i in zeros:
            # Δc for all j at once: shape (n_ones, 2n-1)
            delta_c = A[ones, :] - A[i, :][None, :]  # (n_ones, 2n-1)
            new_maxes = (c[None, :] + delta_c).max(axis=1)
            idx = new_maxes.argmin()
            if new_maxes[idx] < best_new_M:
                best_new_M = new_maxes[idx]
                best_i = i
                best_j = ones[idx]
        
        if best_new_M >= M:
            break  # No improvement possible
        
        # Perform the swap
        h[best_i] = 1.0
        h[best_j] = 0.0
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        M = c.max()
        
        if M < best_M:
            best_M = M
            best_h = h.copy()
    
    return best_h, best_M * dx

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def fast_binary_search(h, n, deadline, max_sweeps=50000):
    """Fast greedy local search for binary h."""
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    best_h = h.copy()
    best_M = M
    stagnant = 0
    
    # Precompute d_vals
    d_vals = np.arange(-(n-1), n)  # (2n-1,)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Compute A matrix: A[p, m] = effect of h[p] going 0->1 on c[m]
        p_col = np.arange(n)[:, None]
        d_row = d_vals[None, :]
        pd = p_col + d_row
        pmd = p_col - d_row
        valid_pd = (pd >= 0) & (pd < n)
        valid_pmd = (pmd >= 0) & (pmd < n)
        
        # Clip for safe indexing
        pd_safe = pd.clip(0, n-1)
        pmd_safe = pmd.clip(0, n-1)
        
        h_pd = np.where(valid_pd, h[pd_safe], 0.0)
        h_pmd = np.where(valid_pmd, h[pmd_safe], 0.0)
        
        A = (1.0 - h_pd) * valid_pd.astype(np.float64) - h_pmd * valid_pmd.astype(np.float64)
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_new_M = M
        best_i = -1
        best_j = -1
        
        # For each zero position, find the best one position to swap with
        for i in zeros:
            delta_c = A[ones, :] - A[i, :][None, :]
            new_maxes = (c[None, :] + delta_c).max(axis=1)
            idx = new_maxes.argmin()
            if new_maxes[idx] < best_new_M:
                best_new_M = new_maxes[idx]
                best_i = i
                best_j = ones[idx]
        
        if best_new_M >= M:
            stagnant += 1
            if stagnant > 100:
                # Try random perturbation
                if len(zeros) > 0 and len(ones) > 0:
                    ri = np.random.randint(len(zeros))
                    rj = np.random.randint(len(ones))
                    h[zeros[ri]], h[ones[rj]] = h[ones[rj]], h[zeros[ri]]
                    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                    M = c.max()
                    stagnant = 0
                else:
                    break
            continue
        
        # Perform swap
        h[best_i] = 1.0
        h[best_j] = 0.0
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        M = c.max()
        stagnant = 0
        
        if M < best_M:
            best_M = M
            best_h = h.copy()
    
    return best_h, best_M * dx

def smooth_max_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation * dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = c.max()
    
    # Softmax weights
    z = alpha * (c - M)
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    
    # Gradient: G[p, m] * w[m]
    d_vals = np.arange(-(n-1), n)
    p_col = np.arange(n)[:, None]
    d_row = d_vals[None, :]

    pmd_safe = pmd.clip(0, n-1)
    h_pd = np.where(valid_pd, h[pd_safe], 0.0)
    h_pmd = np.where(valid_pmd, h[pmd_safe], 0.0)
    G = (1.0 - h_pd) * valid_pd.astype(np.float64) - h_pmd * valid_pmd.astype(np.float64)
    
    grad = G @ w
    return grad, M

        if c_new < best_c:

            best_h = h_new.copy()

        alpha *= 1.005
        # Decay learning rate
        lr *= 0.9999
    
    return best_h, best_c
