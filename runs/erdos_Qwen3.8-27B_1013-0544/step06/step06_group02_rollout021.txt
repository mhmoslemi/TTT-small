   For the binary case (h ∈ {0,1}), the problem becomes: find a binary sequence of length 2n with exactly n ones, such that max over all shifts k of (number of positions where h[j]=1 and h[j+k]=0) is minimized.

   This is related to the autocorrelation of the sequence.

   For the continuous case, we can "soften" the binary solution to get a potentially better value.

   4. Repeat
   5. Also try starting from different structured initializations

        if abs(diff) < 1e-14:

            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def coord_descent_step(h, n, i):
    """Find optimal value of h[i] given all other coordinates fixed."""
    c = compute_c5_vals(h, n)
    dx = 2.0 / n
    
    # Compute alpha_m = dc[m]/dh[i] (before multiplying by dx)
    alphas = np.zeros(2*n-1)
    s = np.arange(-(n-1), n)  # shift values
    
    for idx, shift in enumerate(s):
        if shift >= 0:
            a = 0.0
            if shift <= i <= n-1:
                a += (1.0 - h[i - shift])
            if 0 <= i <= n-1-shift:
                a -= h[i + shift]
            alphas[idx] = a
        else:
            abs_shift = -shift
            a = 0.0
            if 0 <= i <= n-1-abs_shift:
                a += (1.0 - h[i + abs_shift])
            if abs_shift <= i <= n-1:
                a -= h[i - abs_shift]
            alphas[idx] = a
    
    # c[m] is already the overlap (includes dx)
    # overlap_m = alphas[m] * h[i] * dx + (c[m] - alphas[m] * h[i] * dx)
    # Let T = overlap_m as function of t = h[i]:
    # T(t) = alphas[m] * dx * t + (c[m] - alphas[m] * dx * h[i])
    
    # Simplify: f(t) = max_m (A_m * t + B_m)
    # where A_m = alphas[m] * dx, B_m = c[m] - alphas[m] * dx * h[i]
    
    A = alphas * dx
    B = c - alphas * dx * h[i]
    
    # Find min of f(t) = max_m (A_m * t + B_m) for t ∈ [0,1]
    # f is convex piecewise-linear
    
    # Evaluate at endpoints and find minimum via the structure
    # Simple approach: ternary-like search exploiting convexity
    # Or: binary search on the subdifferential
    
    # Let's use a practical approach:
    # 1. Evaluate at t=0, t=1
    # 2. The minimum is where the "slope" changes sign
    # Use binary search on the right derivative
    
    def f(t):
        return np.max(A * t + B)
    
    def right_deriv(t):
        val = f(t)
        active = np.abs(A * t + B - val) < 1e-10
        if active.any():
            return A[active].max()
        return A[np.argmax(A * t + B)]
    
    def left_deriv(t):
        val = f(t)
        active = np.abs(A * t + B - val) < 1e-10
        if active.any():
            return A[active].min()
        return A[np.argmax(A * t + B)]
    
    # Check endpoints
    f0 = f(0.0)
    f1 = f(1.0)
    
    # Binary search for where 0 enters the subdifferential
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        ld = left_deriv(mid)
        rd = right_deriv(mid)
        if ld <= 0 and rd >= 0:
            return mid
        elif rd < 0:
            lo = mid
        else:  # ld > 0
            hi = mid
    
    best_t = min(0, 1, (lo+hi)/2, key=lambda t: f(t))
    return best_t, f(best_t)

            if tot < 1e-14:

        h3, c3 = coord_descent(best_h, n, t3, max_sweeps=1500, perturb_every=10)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    # Phase 4: Pair search
    if time.time() < deadline - 30:
        h4, c4 = pair_search(best_h, n, deadline - 5, radius=35)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    # Phase 5: Try other n
    if time.time() < deadline - 50:
        for n_try in [200, 400, 500, 250]:
            if time.time() > deadline - 35:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            hi = np.interp(x_new, x_old, best_h)
            hi = project_feasible(hi, n_try)
            alloc = min(35, deadline - time.time() - 15)
            if alloc < 12:
                break
            ht, ct = coord_descent(hi, n_try, time.time() + alloc, max_sweeps=300, perturb_every=10)
            if ct < best_c:
                best_c = ct
                best_h = ht
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
