def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

    h_b[idx[:n // 2]] = 1.0

def slsqp_smooth(h, n, alpha, maxiter, ftol):

    def obj(x):

        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

        return h

    hb, cb = greedy_binary(best_h, n, p3 - 30, max_sweeps=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=1500000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Phase 4: Re-smooth from best binary
    p4 = t0 + 0.82 * (dl - t0)
    h = best_h.copy()
    for a in [10000, 100000, 1000000, 10000000]:
        if time.time() > p4 - 3:
            break
        h = slsqp_smooth(h, n, a, 1000, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 5: Try other resolutions
    p5 = dl - 3
    for nh in [n // 2, n * 3 // 2, n + 100]:
        if time.time() > p5 - 20:
            break
        if nh < 100:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for a in [50000, 500000, 5000000]:
            if time.time() > p5 - 15:
                break
            hh = slsqp_smooth(hh, nh, a, 800, 1e-14)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
