    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        else:

    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def build_influence(h, n):
    """Build n x (2n-1) influence matrix"""
    K = np.arange(2 * n - 1)
    I = np.arange(n)
    A = I[:, None] + K[None, :] - n + 1  # (n, 2n-1)
    B = I[:, None] - K[None, :] + n - 1
    valid_a = (A >= 0) & (A < n)
    valid_b = (B >= 0) & (B < n)
    infl = np.zeros((n, 2 * n - 1))
    ia, ka = np.where(valid_a)
    jb, kb = np.where(valid_b)
    infl[ia, ka] = 1.0 - h[A[ia, ka]]
    infl[jb, kb] -= h[B[jb, kb]]
    return infl

def lse_objective(h, alpha, infl):
    """Log-sum-exp objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = max_c + np.log(sum_exp) / alpha
    w = exp_shifted / sum_exp
    grad = dx * (infl.T @ w)
    return obj, grad, corr

    h = h0.copy()

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    mc = corr.max()
    sh = alpha * (corr - mc)
    es = np.exp(sh)
    w = es / es.sum()

    # infl[i,k] = (1-h[a])*[va] - h[b]*[vb] where a=i+k-n+1, b=i-k+n-1

    vb = (B >= 0) & (B < n)
