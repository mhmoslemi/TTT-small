import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def corr_values(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def gradient_topk(h, n, k):
    """Gradient using top-k correlation values."""
    C = corr_values(h, n)
    m = C.max()
    # Get top k indices
    if k >= len(C):
        idx = np.arange(len(C))
        weights = np.ones(len(C)) / len(C)
    else:
        # Soft top-k: use sigmoid weighting
        shifted = 50.0 * (C - m)
        exps = np.exp(shifted - shifted.max())
        # Take top k
        top_idx = np.argsort(exps)[::-1][:k]
        weights = np.zeros(len(C))
        weights[top_idx] = exps[top_idx]
        weights /= weights.sum()
    grad = np.zeros(n)
    for ki in range(len(C)):
        w = weights[ki]
        if w < 1e-15:
            continue
        s = ki - (n - 1)
        jlo1 = max(0, s)
        jhi1 = min(n, n + s)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += w * (1.0 - h[jlo1 - s:jhi1 - s])
        jlo2 = max(0, -s)
        jhi2 = min(n, n - s)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= w * h[jlo2 + s:jhi2 + s]
    return grad * (2.0 / n), m

def gradient_lse(h, n, alpha):
    """Gradient using LSE smoothing."""
    C = corr_values(h, n)
    m = C.max()
    shifted = alpha * (C - m)
    max_s = shifted.max()
    exps = np.exp(shifted - max_s)
    P = exps / exps.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        w = P[ki]
        if w < 1e-15:
            continue
        s = ki - (n - 1)
        jlo1 = max(0, s)
        jhi1 = min(n, n + s)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += w * (1.0 - h[jlo1 - s:jhi1 - s])
        jlo2 = max(0, -s)
        jhi2 = min(n, n - s)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= w * h[jlo2 + s:jhi2 + s]
    return grad * (2.0 / n), m

def pgd_step(h, n, lr, max_ls=20):
    """One projected gradient descent step with backtracking line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for attempt in range(max_ls):
        grad, _ = gradient_lse(h, n, 500)
        step = lr * (0.5 ** attempt)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            return h_new, c_new
        best_h = h_new
        best_c = c_new
    return best_h, best_c
