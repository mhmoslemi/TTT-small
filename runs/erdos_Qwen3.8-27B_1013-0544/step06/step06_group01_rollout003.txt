def coordinate_descent(h, n, max_passes=500, grid=80, deadline=None, rng=None, penalty=5.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    target = n / 2.0
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            hv = h[p]
            d_lo = -hv
            d_hi = 1.0 - hv
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            
            # Use a smarter grid: focus on the direction that reduces max
            # Determine the sign that would reduce the max
            peak_idx = np.argmax(c[ks])
            if s[peak_idx] > 0:
                # Increasing h[p] would increase this peak, so decrease
                sign = -1
            else:
                sign = 1
            
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - target) ** 2
            
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks].copy() + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * hv) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n - 1)]
                if k_lo <= (n - 1) <= k_hi:
                    c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    a = sh.max()
    lse = m + (a + np.log(np.sum(np.exp(sh - a)))) / alpha
    viol = x.sum() - n / 2.0
    return float(lse * dx + 500.0 * viol * viol)

def slsqp_opt(h, n, alpha=50000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def cfun(x, n):
        return x.sum() - n / 2.0
    def cjac(x, n):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:

                        else:

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp(h, n, alpha=50000, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        r = minimize(so_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h, n)

def ps(h, n, dl, dts, rad=30):
    bc = compute_c5(h, n)
    for d in dts:
        while time.time() < dl:
            imp = False
            for i in range(n):
                if time.time() > dl:
                    return h, bc
                lo = max(0, i-rad)
                hi = min(n, i+rad+1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i]+d <= 1.0 and h[j]-d >= 0.0:
                        h[i]+=d; h[j]-=d
                        cv = compute_c5(h, n)
                        if cv < bc - 1e-14:
                            bc = cv; imp = True
                        else:
                            h[i]-=d; h[j]+=d
                    if h[i]-d >= 0.0 and h[j]+d <= 1.0:
                        h[i]-=d; h[j]+=d
                        cv = compute_c5(h, n)
                        if cv < bc - 1e-14:
                            bc = cv; imp = True
                        else:
                            h[i]+=d; h[j]-=d
            if not imp:
                break
    return h, bc
