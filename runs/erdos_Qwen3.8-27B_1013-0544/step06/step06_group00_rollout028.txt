def project_feasible(h, n):

    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        free = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            h_p[free] += diff / nf
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def build_offset(n):
    jj = np.arange(n)
    return (n - 1) + jj[:, None] - jj[None, :]

def build_mask(n):
    jj = np.arange(n)
    return (jj[None, :] >= jj[:, None]).astype(np.float64)

def lse_obj_grad(h, alpha, offset, mask):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / exp_c.sum()
    obj = (c_max + np.log(exp_c.sum())) / alpha
    P = p[offset]
    A = (P * mask) @ (1.0 - h)
    B = P @ h
    return obj, A - B

def lse_obj_only(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    return (c_max + np.log(exp_c.sum())) / alpha

def grad_desc(h_init, n, alpha, deadline, offset, mask, max_iter=50000, lr=None):
    if lr is None:
        lr = 0.8 / (alpha * n * 0.1 + 1.0)
    lr = min(lr, 0.5)
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()

    mu = 0.4

            break
        obj, grad = lse_obj_grad(h, alpha, offset, mask)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stag = 0

            h = h_trial

                lr_cur = min(lr, lr_cur * 1.1)
        else:
            lr_cur *= 0.5
            if lr_cur < 1e-14:

    return best_h, best_obj

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_init, n, deadline, max_sweeps=150):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    offset = build_offset(n)
    mask = build_mask(n)
    # Phase 1: gradient descent with alpha annealing (main workhorse)
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    p1_end = t0 + 0.45 * total
    h = best_h.copy()
    for idx_a, alpha in enumerate(alphas):
        frac = (idx_a + 1) / len(alphas)
        phase_end = t0 + (p1_end - t0) * frac
        if time.time() > phase_end - 3:
            phase_end = min(phase_end, dl - 2)
        h, _ = grad_desc(h, n, alpha, phase_end - 1, offset, mask,
                         max_iter=80000, lr=None)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 2: binary local search
    p2_end = t0 + 0.55 * total
    hb, cb = binary_local_search(best_h, n, p2_end, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 3: gradient descent from (possibly binary) solution
    p3_end = t0 + 0.75 * total
    h = best_h.copy()
    for alpha in [5000, 50000, 500000, 5000000]:
        if time.time() > p3_end - 5:
            break
        h, _ = grad_desc(h, n, alpha, p3_end - 3, offset, mask, max_iter=50000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 4: noise restarts
    p4_end = t0 + 0.88 * total
    for trial in range(6):
        if time.time() > p4_end - 8:
            break
        scale = 0.03 * (1.0 - trial / 6.0) + 0.005
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [20000, 200000, 2000000]:
            if time.time() > p4_end - 4:
                break
            h, _ = grad_desc(h, n, alpha, p4_end - 2, offset, mask, max_iter=40000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # Phase 5: try other resolutions
    p5_end = dl - 3
    for nh in [300, 500, 600]:
        if time.time() > p5_end - 40:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        off_h = build_offset(nh)
        msk_h = build_mask(nh)
        for alpha in [5000, 100000, 1000000]:
            if time.time() > p5_end - 10:
                break
            hh, _ = grad_desc(hh, nh, alpha, p5_end - 5, off_h, msk_h, max_iter=30000)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
    # Final polish
    if time.time() < dl - 2:
        offset2 = build_offset(n)
        mask2 = build_mask(n)
        h = best_h + 0.005 * rng.randn(n)
        h = project_feasible(h, n)
        h, _ = grad_desc(h, n, 5000000, dl - 1, offset2, mask2, max_iter=30000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
