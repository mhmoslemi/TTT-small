def nesterov_pgd(h, n, deadline, alpha=100, lr=None, seed=42):
    """Nesterov accelerated projected gradient descent."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    if lr is None:
        lr = 2.0 / (alpha * dx)  # rough estimate
    
    h = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    beta = 0.9  # momentum
    
    for it in range(1000000):
        if time.time() > deadline:
            break
        
        # Nesterov step
        h_mom = h + beta * (h - h_prev)
        h_mom = project_feasible(h_mom, n)
        
        # Compute gradient of softmax(corr)
        corr = np.correlate(h_mom, 1-h_mom, mode='full').astype(float)
        scaled = alpha * corr
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        p = exp_s / exp_s.sum()
        
        # Gradient: for each i, grad[i] = sum_s p[s] * d(c_s)/d(h[i])
        # d(c_s)/d(h[i]) where c_s = sum_x h[x]*(1-h[x-shift_s])
        # = (1-h[i-shift_s]) - h[i+shift_s] (with boundary conditions)
        
        shifts = np.arange(-(n-1), n)
        grad = np.zeros(n)
        # Only use top-K shifts for efficiency
        top_k = np.argsort(p)[::-1][:30]
        for s in top_k:
            ps = p[s]
            if ps < 1e-8:
                break
            sh = shifts[s]
            # term 1: (1-h[i-sh]) for valid i-sh
            lo = max(0, sh)
            hi = min(n-1, n-1+sh)
            if lo <= hi:
                grad[lo:hi+1] += ps * (1 - h_mom[lo-sh:hi+1-sh])
            # term 2: -h[i+sh] for valid i+sh
            lo2 = max(0, -sh)
            hi2 = min(n-1, n-1-sh)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= ps * h_mom[lo2+sh:hi2+1+sh]
        
        grad *= dx
        
        # Update
        h_new = project_feasible(h_mom - lr * grad, n)
        h_prev = h
        h = h_new
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Adaptive lr
        # (simplified: reduce lr periodically)
        if it % 1000 == 999:
            lr *= 0.95
            beta = 1.0 - 1.0 / np.sqrt(it + 1)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def nesterov_pgd(h, n, deadline, alpha=200, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    shifts = np.arange(-(n - 1), n)
    lr = 0.5 / (alpha * dx)
    beta = 0.9
    h = project_feasible(h, n)
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(500000):
        if time.time() > deadline:
            break
        h_mom = h + beta * (h - h_prev)
        h_mom = project_feasible(h_mom, n)
        corr = np.correlate(h_mom, 1 - h_mom, mode='full').astype(float)
        scaled = alpha * corr
        mx = scaled.max()
        exp_s = np.exp(scaled - mx)
        p = exp_s / exp_s.sum()
        grad = np.zeros(n)
        top_k = np.argsort(p)[::-1][:40]
        for s in top_k:
            ps = p[s]
            if ps < 1e-8:
                break
            sh = shifts[s]
            lo = max(0, sh)
            hi = min(n - 1, n - 1 + sh)
            if lo <= hi:
                grad[lo:hi+1] += ps * (1.0 - h_mom[lo-sh:hi+1-sh])
            lo2 = max(0, -sh)
            hi2 = min(n - 1, n - 1 - sh)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= ps * h_mom[lo2+sh:hi2+1+sh]
        grad *= dx
        h_new = project_feasible(h_mom - lr * grad, n)
        h_prev = h
        h = h_new
        if it % 50 == 49:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr *= 0.98
            beta = 1.0 - 1.0 / np.sqrt(it + 1.0)
        if it % 5000 == 4999:
            h = project_feasible(h + 0.005 * rng.randn(n), n)
            h_prev = h.copy()
            lr = 0.5 / (alpha * dx)
    return best_h, best_c

def gen_starts(n, seed, count):
    rng = np.random.RandomState(seed)
    target = n // 2
    starts = []
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:target]] = 1.0
    starts.append(h.copy())
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    for phase in [0.0, 0.1, 0.3]:
        h = np.zeros(n)
        positions = np.floor(np.arange(target) * golden + phase * n) % n
        positions = positions.astype(int)
        h[positions] = 1.0
        cnt = int(h.sum())
        if cnt > target:
            zi = np.where(h == 0)[0]
            ones_i = np.where(h == 1)[0]
            rng.shuffle(ones_i)
            for k in range(target, len(ones_i)):
                h[ones_i[k]] = 0.0
        elif cnt < target:
            zi = np.where(h == 0)[0]
            to_one = rng.choice(zi, target - cnt, replace=False)
            h[to_one] = 1.0
        starts.append(h.copy())
    for _ in range(max(0, count - 4)):
        h = np.zeros(n)
        pos = 0
        while pos < n:
            blk = rng.randint(2, 7)
            h[pos:min(pos + blk, n)] = 1.0 if (pos // max(blk,1)) % 2 == 0 else 0.0
            pos += blk
        cnt = int(h.sum())
        if cnt > target:
            oi = np.where(h == 1)[0]
            to_zero = rng.choice(oi, cnt - target, replace=False)
            h[to_zero] = 0
        elif cnt < target:
            zi = np.where(h == 0)[0]
            to_one = rng.choice(zi, target - cnt, replace=False)
            h[to_one] = 1
        starts.append(h)
    return starts[:count]

def slsqp_smooth(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
