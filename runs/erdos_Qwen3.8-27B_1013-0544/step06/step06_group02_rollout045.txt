            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (200, 1.0), (1000, 5.0), (5000, 20.0),
        (20000, 100.0), (100000, 500.0),
        (500000, 2000.0), (2000000, 10000.0),
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': max_iters * 15,
                                    'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def simulated_annealing(h, n, deadline, seed=99, t_init=0.015, t_end=0.00005):
    rng = np.random.RandomState(seed)
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_c = best_c
    move_sizes = np.array([0.001, 0.002, 0.003, 0.005, 0.001, 0.002, 0.001, 0.0005])
    total_iters = 0
    no_improve = 0
    while time.time() < deadline:
        for _ in range(200):
            if time.time() > deadline:
                return best_h, best_c
            total_iters += 1
            progress = min(1.0, total_iters / 300000.0)
            t = t_init * (t_end / t_init) ** progress
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = move_sizes[rng.randint(len(move_sizes))]
            if rng.rand() < 0.5:
                d = -d
            if cur_h[i] + d < -0.01 or cur_h[i] + d > 1.01:
                continue
            if cur_h[j] - d < -0.01 or cur_h[j] - d > 1.01:
                continue
            cur_h[i] += d
            cur_h[j] -= d
            new_c = compute_c5(cur_h, n)
            if new_c < cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            elif rng.rand() < np.exp(-((new_c - cur_c) / max(t, 1e-10))):
                cur_c = new_c
                no_improve += 1
            else:
                cur_h[i] -= d
                cur_h[j] += d
                no_improve += 1
            if no_improve > 50000:
                cur_h = best_h.copy() + rng.randn(n) * 0.01
                cur_h = project_feasible(cur_h, n)
                cur_c = compute_c5(cur_h, n)
                no_improve = 0
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    while time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                return best_h, best_c
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius) if radius is not None else 0
                hi = min(n, i + radius + 1) if radius is not None else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
        if not improved:
            break
    return best_h, best_c

def active_shift_search(h, n, deadline, top_k=3):
    """Target moves that specifically reduce the worst shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_pass = 0
    while time.time() < deadline and n_pass < 20:
        n_pass += 1
        c = compute_c5_vec(h, n)
        idx = np.argsort(c)[-top_k:]
        active_shifts = [int(i) - (n - 1) for i in idx]
        improved = False
        for _ in range(5):
            if time.time() > deadline:
                return best_h, best_c
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for k in active_shifts:
                    if abs(k) >= n:
                        continue
                    j = i + k
                    if j < 0 or j >= n:
                        continue
                    if i == j:
                        continue
                    # Try to reduce h[i]*(1-h[j]) + h[j]*(1-h[i]) at this pair
                    for d in [0.005, 0.002, 0.001, 0.0005]:
                        # Make i and j more similar
                        if h[i] > h[j]:
                            if h[i] - d >= 0 and h[j] + d <= 1:
                                h[i] -= d; h[j] += d
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-15:
                                    best_c = c_new; best_h = h.copy(); improved = True
                                else:
                                    h[i] += d; h[j] -= d
                        elif h[j] > h[i]:
                            if h[i] + d <= 1 and h[j] - d >= 0:
                                h[i] += d; h[j] -= d
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-15:
                                    best_c = c_new; best_h = h.copy(); improved = True
                                else:
                                    h[i] -= d; h[j] += d
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=300)

    # Phase 2: Active-shift focused search
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        h2, c2 = active_shift_search(best_h, n, t2, top_k=5)
        if c2 < best_c:
            best_c = c2; best_h = h2

    # Phase 3: Simulated annealing (key novel component)
    if time.time() < deadline - 300:
        t3 = min(time.time() + 250, deadline - 300)
        h3, c3 = simulated_annealing(best_h, n, t3, seed=seed, t_init=0.012, t_end=0.00003)
        if c3 < best_c:
            best_c = c3; best_h = h3

    # Phase 4: L-BFGS-B after SA
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        h4, c4 = lbfgs_phase(best_h, n, t4, max_iters=250)
        if c4 < best_c:
            best_c = c4; best_h = h4

    # Phase 5: Second SA round (cooler)
    if time.time() < deadline - 120:
        t5 = min(time.time() + 100, deadline - 120)
        h5, c5 = simulated_annealing(best_h, n, t5, seed=seed + 7, t_init=0.005, t_end=0.00001)
        if c5 < best_c:
            best_c = c5; best_h = h5

    # Phase 6: Fine pair search
    if time.time() < deadline - 40:
        t6 = min(time.time() + 60, deadline - 40)
        best_h, best_c = fine_pair_search(best_h, n, t6,
                                          [0.001, 0.0005, 0.0002, 0.0001], radius=50)

    # Phase 7: Try other resolutions
    if time.time() < deadline - 60:
        for n_try in [200, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 30)
            if alloc < 25:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=200)
            if time.time() < deadline - 15:
                h_t, c_t = fine_pair_search(h_t, n_try,
                                            time.time() + min(20, deadline - time.time() - 10),
                                            [0.001, 0.0005, 0.0001], radius=30)
            if c_t < best_c:
                best_c = c_t; best_h = h_t

    # Phase 8: Final polish
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_search(best_h, len(best_h), deadline - 1,
                                           [0.0005, 0.0002, 0.0001, 0.00005], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
