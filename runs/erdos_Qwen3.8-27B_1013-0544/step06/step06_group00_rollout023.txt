    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    logsum = shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())
    obj = m + logsum / alpha
    # Weights: w_k = delta_{k,k*} + exp(alpha*c_k) / sum(exp(alpha*c_k))

    w = exp_c / exp_c.sum()
    # Add the hard max component
    kmax = np.argmax(c)
    w[kmax] += 1.0
    return obj, w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    logsum = shifted_max + np.log(exp_shifted.sum())
    obj = m + logsum / alpha
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    w = w + (1.0 / len(c)) * 0  # no, just add the delta
    kmax = int(np.argmax(c))
    w = w.copy()
    w[kmax] += 1.0
    
    # Gradient: dF/dh[j] = dx * sum_k w_k * ((1-h[j+k]) - h[j-k])
    m_range = np.arange(-(n-1), n)
    idx = np.arange(n)[:, None] + m_range[None, :]
    mask = (idx >= 0) & (idx < n)
    h_safe = np.where(mask, h[np.clip(idx, 0, n-1)], 0.0)
    
    idx2 = np.arange(n)[:, None] - m_range[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    h_safe2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    w_row = w[None, :]
    term1 = (w_row * mask.astype(np.float64)).sum(axis=1)
    term2 = (w_row * h_safe).sum(axis=1)
    term3 = (w_row * h_safe2).sum(axis=1)
    
    grad = dx * (term1 - term2 - term3)
    return obj, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    def obj(x):
        v, _ = smooth_obj_and_grad(x, n, alpha)
        return v
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def binary_local_search(h, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        done = False
        for i in zeros[:]:
            if time.time() > deadline:
                done = True
                break
            for j in ones[:]:
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:

                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    done = True
                    break
            if done:
                break
        if not done:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
