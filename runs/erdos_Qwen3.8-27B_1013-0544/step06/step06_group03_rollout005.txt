def quad_obj(h, n, lam):
    """Weighted sum of correlations: sum_k lam[k] * c_k(h)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.dot(lam, c))

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, beta, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def run_slsqp_quad(h, n, lam, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(quad_obj, h, args=(n, lam),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def dual_decompose(h, n, deadline, n_iter=50):
    """Iterate: solve QP for current weights, update weights toward active lags."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    nlags = 2 * n - 1
    lam = np.full(nlags, 1.0 / nlags)
    for it in range(n_iter):
        if time.time() > deadline - 5:
            break
        h_new = run_slsqp_quad(h_curr, n, lam, maxiter=1000)
        c = compute_all_lags(h_new, n)
        c5 = float(np.max(c))
        if c5 < best_c - 1e-12:
            best_c = c5
            best_h = h_new.copy()
        # Update lam: concentrate on active lags
        threshold = c.max() * 0.95
        active = np.where(c >= threshold)[0]
        lam = np.zeros(nlags)
        lam[active] = 1.0 / len(active)
        h_curr = h_new
    return best_h, best_c

def greedy_lag_search(h, n, deadline, max_iters=10000):
    """Target the single worst contributing pair at the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        c = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            # Push toward binary at lag 0
            devs = np.abs(h - 0.5)
            i = int(np.argmin(devs))
            step = 0.05
            if h[i] >= 0.5:
                h[i] = min(1.0, h[i] + step)
            else:
                h[i] = max(0.0, h[i] - step)
        else:
            m = abs(lag)
            if lag > 0:
                contribs = h[:n-m] * (1.0 - h[m:])
                i = int(np.argmax(contribs))
                transfer = min(0.03, h[i] * 0.4, (1.0 - h[i+m]) * 0.4)
            else:
                contribs = h[m:] * (1.0 - h[:n-m])
                i = int(np.argmax(contribs))
                transfer = min(0.03, h[i+m] * 0.4, (1.0 - h[i]) * 0.4)
            if transfer < 1e-7:
                stuck += 1
                if stuck > 50:
                    break
                # Random perturbation to escape
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    tmp = h[i]
                    h[i] = h[j]
                    h[j] = tmp
            else:
                if lag > 0:
                    h[i] -= transfer
                    h[i + m] += transfer
                else:
                    h[i + m] -= transfer
                    h[i] += transfer
                stuck = 0
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
    
    return h, best_c

def subgrad_descent(h, n, deadline, n_steps=3000):
    """Subgradient method with Polyak step size."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    lb = 0.35  # lower bound estimate
    for step in range(n_steps):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        c_max = float(c.max())
        # Weight active lags
        threshold = c_max * 0.98
        active = np.where(c >= threshold)[0]
        weights = c[active] - c_max + 0.01
        weights = np.maximum(weights, 0.01)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k_idx in zip(weights, active):
            lag = k_idx - (n - 1)
            grad += w * lag_gradient(h_curr, n, lag)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        step_size = (c_max - lb) / (grad_norm * grad_norm) * dx
        step_size = min(step_size, 0.5)
        h_new = h_curr - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n = n_init
        h_curr = project_feasible(h_init, n)
        best_h = h_curr.copy()
        best_c = compute_c5(h_curr, n)

        # Phase 1: SLSQP with binary penalty sweep
        for alpha, beta in [(2000, 0.0), (5000, 0.0), (10000, 0.005),
                           (10000, 0.01), (20000, 0.005), (50000, 0.0)]:
            if time.time() > deadline - 150:
                break
            h_new = run_slsqp(h_curr, n, alpha, beta, maxiter=2500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new

        # Phase 2: Dual decomposition
        if time.time() < deadline - 100:
            h_new, c_new = dual_decompose(h_curr, n, deadline - 60)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new

        # Phase 3: Subgradient descent
        if time.time() < deadline - 60:
            h_new, c_new = subgrad_descent(h_curr, n, deadline - 30, n_steps=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new

        # Phase 4: Greedy lag search (fast, targeted)
        if time.time() < deadline - 20:
            h_new, c_new = greedy_lag_search(h_curr, n, deadline - 5)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

        # Phase 5: Final SLSQP + greedy polish
        if best_h is not None and time.time() < deadline - 10:
            h_new = run_slsqp(best_h, n, 50000, 0.0, maxiter=2000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_new, c_new = greedy_lag_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Try different n values
    if best_h is not None and time.time() < deadline - 60:
        for n_try in [200, 150]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
            h_start = project_feasible(h_start, n_try)
            h_opt = h_start
            c_opt = compute_c5(h_start, n_try)
            for alpha, beta in [(5000, 0.0), (10000, 0.01)]:
                if time.time() > deadline - 25:
                    break
                h_new = run_slsqp(h_opt, n_try, alpha, beta, maxiter=1500)
                c_new = compute_c5(h_new, n_try)
                if c_new < c_opt:
                    c_opt = c_new
                    h_opt = h_new
            if time.time() < deadline - 15:
                h_opt, c_opt = greedy_lag_search(h_opt, n_try, deadline - 5)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
