    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

            s = room.sum()

            if s < 1e-14:

            h += diff / s * h

def shift_vec(x, s, n):
    """y[i] = x[i+s] if 0<=i+s<n else 0"""
    y = np.zeros(n)
    if s >= 0:
        y[:n-s] = x[s:]
    else:

        exp_S = np.exp(D * (S - Smax))
