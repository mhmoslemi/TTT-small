def subgradient_equalize(h, n, deadline, n_iter=100000):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute correlation
        c = np.correlate(h, 1-h, mode='full') * dx
        # Find top values
        top_k = np.argsort(c)[-5:]  # indices of top 5 c values
        top_c = c[top_k].max()
        
        # Subgradient: for each top k, gradient of c(k) w.r.t. h
        # d c(k) / d h[i] = (1-h[i+k_valid]) for the forward part
        #                       - h[i-k_valid] for the backward part
        # More precisely: c(k) = sum_i h[i]*(1-h[i+k])
        # dc(k)/dh[i] = (1-h[i+k]) - h[i-k] (where indices are valid)
        
        grad = np.zeros(n)
        for k_idx in top_k:
            k = k_idx - (n-1)  # actual shift
            # Forward: h[i] appears as h[i]*(1-h[i+k]) for valid i

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_swap_delta(i, j, h, m_range, n):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greedy_full(h_init, n, deadline, max_sweeps=200):
    """Steepest descent greedy: for each 0 position, find the best 1 to swap with."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = compute_swap_delta(i, j, h, m_range, n)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                delta = compute_swap_delta(i, best_j, h, m_range, n)
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

    """Iterated Local Search for binary optimization."""
    rng = np.random.RandomState(seed)
    h, c = greedy_full(h_init, n, min(deadline, time.time() + 60))

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
