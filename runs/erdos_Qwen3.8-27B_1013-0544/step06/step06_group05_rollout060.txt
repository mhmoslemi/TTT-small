Since the objective is (approximately) the max of affine functions (when we linearize), we can use:

The subgradient of corr[k] with respect to h[i] is:
∂corr[k]/∂h[i] = (1-h[i-k]) * 1_{valid} - h[i+k] * 1_{valid}

For the max, the subgradient is a convex combination of the subgradients of the active lags.

```python
def subgradient_h(h, n, deadline, lr_schedule='decreasing'):
    """Subgradient method for minimizing max correlation."""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    lr = 0.01
    for it in range(100000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = corr.max()
        
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
        
        # Find active lags (within tolerance of max)
        tol = 0.5
        active = np.where(corr >= cur_max - tol)[0]
        
        # Subgradient: average of gradients of active lags
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            for i in range(n):
                # ∂corr[m]/∂h[i]
                val = 0.0
                if 0 <= i - k < n:
                    val += (1.0 - h[i - k])
                if 0 <= i + k < n:
                    val += -h[i + k]
                grad[i] += val
        grad /= max(1, len(active))
        
        # Gradient step
        h_new = h - lr * grad
        h = project_feasible(h_new, n)
        
        # Decrease learning rate
        lr = 0.01 / (1 + 0.001 * it)
    
    return best_h, best_c
```

def compute_subgrad(h, n, active_idx):
    """Compute subgradient of max correlation at active lags."""
    n = len(h)
    grad = np.zeros(n)
    for m in active_idx:
        k = m - (n - 1)
        # For each i, the contribution to grad[i] is:
        # (1-h[i-k]) * 1_{0≤i-k<n} - h[i+k] * 1_{0≤i+k<n}
        i = np.arange(n)

        ipk = i + k

        g[valid_k] += (1.0 - h[ik[valid_k]])
        g[valid_p] += -h[ipk[valid_p]]
        grad += g
    grad /= max(1, len(active_idx))
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, deadline, T=10, max_iter=50000):
    """Vectorized binary local search focusing on top T critical lags."""
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    # Ensure exactly n//2 ones
    n_ones = int(h.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        idx = np.where(h == 1)[0]
        h[idx[-(n_ones - target_ones):]] = 0.0
    elif n_ones < target_ones:
        idx = np.where(h == 0)[0]
        h[idx[:target_ones - n_ones]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    dx = 2.0 / n
    best_h = h.copy()
    best_max = cur_max
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        top_idx = np.argsort(corr)[-T:]
        ks = top_idx - (n - 1)
        corr_top = corr[top_idx].copy()
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_improve = 1e-10
        best_pair = None
        
        for i in zeros:
            # Change if h[i] goes 0->1
            ci = np.zeros(T)
            for t in range(T):
                k = ks[t]
                if 0 <= i - k < n:
                    ci[t] += (1.0 - h[i - k])
                if 0 <= i + k < n:
                    ci[t] += -h[i + k]
            
            # Vectorized over all j in ones
            cj = np.zeros((len(ones), T))
            for t in range(T):
                k = ks[t]
                j_minus = ones - k
                j_plus = ones + k
                vm = (j_minus >= 0) & (j_minus < n)
                vp = (j_plus >= 0) & (j_plus < n)
                # When h[j] goes 1->0:
                # j as h[n+k] side (n=j-k): change = -(1-h[j-k])
                # j as (1-h[n]) side (n=j): change = +h[j+k]
                cj[vm, t] += -(1.0 - h[j_minus[vm]])
                cj[vp, t] += h[j_plus[vp]]
            
            new_vals = corr_top[None, :] + ci[None, :] + cj
            new_maxes = new_vals.max(axis=1)
            
            idx_best = np.argmin(new_maxes)
            improve = cur_max - new_maxes[idx_best]

                best_improve = improve
                best_pair = (i, ones[idx_best])
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = corr.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:

    return best_h, float(best_max * dx)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        exp_shifted = np.exp(shifted - sm)
        weights = exp_shifted / exp_shifted.sum()
        # Gradient of corr[m] w.r.t. x[i]
        grad = np.zeros(n)
        for mi in np.where(weights > 1e-8)[0]:
            k = mi - (n - 1)
            i_arr = np.arange(n)
            ik = i_arr - k
            ipk = i_arr + k
            vm = (ik >= 0) & (ik < n)
            vp = (ipk >= 0) & (ipk < n)
            g = np.zeros(n)
            g[vm] += (1.0 - x[ik[vm]])
            g[vp] += -x[ipk[vp]]
            grad += weights[mi] * g
        grad *= dx
        return grad
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

from scipy.optimize import minimize
import time

_VAR = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def _obj_lse(x):
    n = _VAR['n']
    alpha = _VAR['alpha']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def _obj_bal(x):
    n = _VAR['n']
    active = _VAR['active']
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(corr[active] ** 2))

def _con(x):
    n = _VAR['n']
    return x.sum() - n / 2.0

def slsqp_lse(h, n, alpha, maxiter, deadline):
    _VAR['n'] = n
    _VAR['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con}]

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)

        return h

            improve = cur_max - new_maxes[idx_best]
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, ones[idx_best])
        if best_pair is not None:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = corr.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break
    return best_h, float(best_max * dx)
