import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta_fast(h, n, i, j):
    s = 1.0 - 2.0 * h
    mi = n - 1 - i
    mj = n - 1 - j
    delta = np.zeros(2 * n - 1)
    delta[mi:mi+n] += s
    delta[mj:mj+n] -= s
    return delta

def sa_binary(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.00001, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start
    n_zero = len(zeros_idx)
    n_one = len(ones_idx)
    
    while time.time() < deadline:
        i = zeros_idx[rng.randint(n_zero)]
        j = ones_idx[rng.randint(n_one)]
        delta = swap_delta_fast(h, n, i, j)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            n_zero = len(zeros_idx)
            n_one = len(ones_idx)
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 5 + rng.randint(10)
            for _ in range(nk):
                if n_zero > 0 and n_one > 0:
                    ii = zeros_idx[rng.randint(n_zero)]
                    jj = ones_idx[rng.randint(n_one)]
                    h[ii] = 1.0
                    h[jj] = 0.0
            corr = get_corr(h, n)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            n_zero = len(zeros_idx)
            n_one = len(ones_idx)
            no_improve = 0
            T = min(t_start, T * 5.0)
    
    return best_h, best_c

def sa_targeted(h_bin, n, deadline, seed=42, t_start=0.03, patience=20000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    T = t_start
    cool_steps = 50000
    cool_rate = (0.00001 / t_start) ** (1.0 / cool_steps)
    
    while time.time() < deadline:
        # Identify peak shift
        k_peak = np.argmax(corr)
        offset = k_peak - (n - 1)
        
        # With probability 0.3, do a targeted move
        if rng.random() < 0.3:
            # Find contributing pairs at k_peak
            # c[k_peak] = sum_i h[i+offset] * (1-h[i]) for valid i
            # Valid: 0 <= i < n and 0 <= i+offset < n
            if offset >= 0:
                i_min = 0
                i_max = min(n-1, n-1-offset)
            else:
                i_min = -offset
                i_max = n-1
            # Pairs where h[i+offset]=1 and h[i]=0
            i_candidates = np.arange(i_min, i_max+1)
            contributing = (h[i_candidates + offset] == 1) & (h[i_candidates] == 0)
            if contributing.any():
                ci = i_candidates[contributing]
                i = ci[rng.randint(len(ci))]
                j = i + offset  # this is a 1, swap with a random 0
                # Actually swap h[j] (which is 1) with a random 0
                j_swap = np.where(h == 1)[0][rng.randint(n//2)]
                if j_swap != i:
                    i, j = j_swap, i  # i is the 1, j is the 0
                    # Wait, we need i to be a 0 and j to be a 1 for our swap
                    # Let me redo: we want to swap a 0 and a 1
                    # The 0 we want to change is at position i (to reduce the pair)
                    # The 1 we want to change is at position j = i+offset
                    # So: swap positions i (currently 0) and j (currently 1)
                    # That means: i is the zero to become 1, j is the one to become 0
                    pass
            # If no contributing pairs, fall through to random
            if not (contributing.any() if 'contributing' in dir() else False):
                i = zeros_idx[rng.randint(len(zeros_idx))]
                j = ones_idx[rng.randint(len(ones_idx))]
        else:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
        
        # Evaluate swap
        s = 1.0 - 2.0 * h
        mi = n - 1 - i
        mj = n - 1 - j
        new_corr = corr + s[mi:mi+n] - s[mj:mj+n]
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool_rate
        if no_improve > patience:
            # Reheat with random perturbations
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                if len(zeros_idx) > 0 and len(ones_idx) > 0:
                    ii = zeros_idx[rng.randint(len(zeros_idx))]
                    jj = ones_idx[rng.randint(len(ones_idx))]
                    h[ii] = 1.0
                    h[jj] = 0.0
            corr = get_corr(h, n)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = t_start * (0.5 + 0.5 * rng.random())
    
    return best_h, best_c

def greedy_construct(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    corr = np.zeros(2*n-1)  # all zero initially
    
    positions = list(range(n))
    rng.shuffle(positions)
    
    s = np.ones(n)  # 1-2*h = 1 initially
    
    for _ in range(n_ones):
        # Try each remaining position, pick the one that minimizes max(corr + delta)
        best_pos = -1
        best_max = np.inf
        for pos in positions:
            if h[pos] == 0:
                mi = n - 1 - pos
                new_max = (corr + s[mi:mi+n]).max()
                if new_max < best_max:
                    best_max = new_max
                    best_pos = pos
        h[best_pos] = 1.0
        mi = n - 1 - best_pos
        corr += s[mi:mi+n]
        s[best_pos] = -1.0  # update s = 1-2h
    
    return h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def greedy_construct(n, seed=42, top_k=5):

    h = np.zeros(n)
    n_ones = n // 2
    corr = np.zeros(2 * n - 1)

        corr += s[mi:mi + n]
        s[pos] = -1.0
        available.remove(pos)
    
    return h

def sa_fast(h_bin, n, deadline, seed=42, t_start=0.02, patience=15000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = get_corr(h, n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    T = t_start
    cool_rate = (0.000005 / t_start) ** (1.0 / 100000.0)
    s = 1.0 - 2.0 * h
    n_zero = len(zeros_idx)
    n_one = len(ones_idx)
    
    while time.time() < deadline:
        if n_zero == 0 or n_one == 0:
            break
        i = zeros_idx[rng.randint(n_zero)]
        j = ones_idx[rng.randint(n_one)]
        mi = n - 1 - i
        mj = n - 1 - j
        new_corr = corr + s[mi:mi + n] - s[mj:mj + n]
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            s[i] = -1.0
            s[j] = 1.0
            corr = new_corr
            cur_c = nc
            # Update indices
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            n_zero = len(zeros_idx)
            n_one = len(ones_idx)
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(10)
            for _ in range(nk):
                if n_zero > 0 and n_one > 0:
                    ii = zeros_idx[rng.randint(n_zero)]
                    jj = ones_idx[rng.randint(n_one)]
                    h[ii] = 1.0
                    h[jj] = 0.0
                    s[ii] = -1.0
                    s[jj] = 1.0
            corr = get_corr(h, n)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            n_zero = len(zeros_idx)
            n_one = len(ones_idx)
            no_improve = 0
            T = t_start * (0.3 + 0.7 * rng.random())
    
    return best_h, best_c

def local_search(h_bin, n, max_sweeps=100, deadline=None):
    """Greedy local search: find best improving swap."""
    dx = 2.0 / n
    h = h_bin.copy()
    corr = get_corr(h, n)
    cur_max = corr.max()
    s = 1.0 - 2.0 * h
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # For efficiency, check a subset of swaps per sweep
        # Focus on swaps that affect the current max shifts
        peak_k = np.where(corr == cur_max)[0]
        rng = np.random.RandomState(sweep)
        
        # Try random swaps, biased towards those affecting peak shifts
        n_tries = min(5000, len(zeros) * len(ones))
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            mi = n - 1 - i
            mj = n - 1 - j
            new_max = (corr + s[mi:mi + n] - s[mj:mj + n]).max()
            if new_max < cur_max - 1e-10:
                h[i] = 1.0
                h[j] = 0.0
                s[i] = -1.0
                s[j] = 1.0
                corr += s[mi:mi + n] - s[mj:mj + n]
                cur_max = new_max
                improved = True
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
        if not improved:
            break
    
    return h, cur_max * dx
