    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def top1_grad(h, n):
    """Gradient of the single worst shift's correlation value."""
    c = compute_c5_vec(h, n)
    k_star = np.argmax(c)
    k = k_star - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += dx * (1.0 - h[k:n])
        grad[k:n] += dx * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += dx * (1.0 - h[0:nv])
        grad[0:nv] += dx * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alpha, maxiter=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def top1_descent(h, n, deadline, lr_init=0.01):
    """Projected gradient descent on the single worst shift."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for it in range(50000):
        if time.time() > deadline:
            break
        g = top1_grad(h, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=80):
    """Pair search with small steps and local radius."""
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_descent_smooth(h, n, deadline, alpha=5000.0):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.02
    for it in range(30000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c

def try_other_n(h_best, n_best, best_c, deadline):
    """Try different resolutions."""
    for n_try in [250, 400, 500]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 30)
        if alloc < 20:
            break
        h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.7, alpha=500000, maxiter=200)
        if time.time() < deadline - 20:
            h_t, c_t = pair_search_fine(h_t, n_try, min(time.time() + 25, deadline - 15),
                                         [0.005, 0.002, 0.001, 0.0005], radius=40)
        if c_t < best_c:
            best_c = c_t
            h_best = h_t
            n_best = n_try
    return h_best, best_c, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B with moderate alpha
    if time.time() < deadline - 400:
        t1 = min(t0 + 200, deadline - 400)
        h, best_c = lbfgs_optimize(h, n, t1, alpha=100000, maxiter=400)
        best_h = h.copy()

    # Phase 2: L-BFGS-B with high alpha (sharper focus)
    if time.time() < deadline - 300:
        t2 = min(time.time() + 200, deadline - 300)
        h2, c2 = lbfgs_optimize(h, n, t2, alpha=500000, maxiter=300)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            h = h2
        else:
            h = best_h

    # Phase 3: Top-1 projected gradient descent
    if time.time() < deadline - 250:
        t3 = min(time.time() + 120, deadline - 250)
        h3, c3 = top1_descent(h, n, t3, lr_init=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            h = h3

    # Phase 4: Smooth gradient descent
    if time.time() < deadline - 200:
        t4 = min(time.time() + 100, deadline - 200)
        h4, c4 = gradient_descent_smooth(h, n, t4, alpha=10000.0)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            h = h4

    # Phase 5: Very fine pair search
    if time.time() < deadline - 100:
        t5 = min(time.time() + 180, deadline - 100)
        h5, c5 = pair_search_fine(h, n, t5,
                                   [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                   radius=80)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            h = h5

    # Phase 6: Try other resolutions
    if time.time() < deadline - 80:
        h, best_c, n = try_other_n(best_h, n, best_c, deadline - 40)
        best_h = h.copy()

    # Phase 7: Final refinement on best solution
    if time.time() < deadline - 10:
        nf = len(best_h)
        if time.time() < deadline - 15:
            t7 = min(time.time() + 40, deadline - 15)
            h7, c7 = lbfgs_optimize(best_h, nf, t7, alpha=1000000, maxiter=200)
            if c7 < best_c:
                best_c = c7
                best_h = h7
        if time.time() < deadline - 5:
            h8, c8 = pair_search_fine(best_h, nf, deadline - 2,
                                       [0.001, 0.0005, 0.0002, 0.0001], radius=50)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Phase 8: Final top-1 polish
    if time.time() < deadline - 2:
        h9, c9 = top1_descent(best_h, len(best_h), deadline - 1, lr_init=0.002)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
