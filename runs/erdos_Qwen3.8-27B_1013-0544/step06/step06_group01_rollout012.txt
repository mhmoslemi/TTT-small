            grad[p2_s:p2_e] -= w * h[p2_s-m:p2_e-m]
    return grad, c, c_max

def optimize_subgrad(h, n, budget_s, rng, n_active=30):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_start = time.time()
    
    # Adaptive parameters
    alpha = 0.005  # initial step size
    beta = 0.9     # momentum
    momentum = np.zeros(n)
    no_improve = 0
    best_no_improve_c = best_c
    
    while time.time() < t_start + budget_s:
        grad, c, c_max = subgradient_step(h, n, n_active)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad = grad / gnorm * max(c_max * 2.0 / n * 0.1, 0.01)
        
        # Momentum step
        momentum = beta * momentum - alpha * grad
        h_new = h + momentum
        
        # Project
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-8:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            # Occasionally increase alpha
            if rng.random() < 0.3:
                alpha *= 1.1
        else:
            no_improve += 1
            h = best_h.copy()  # backtrack
            if no_improve > 50:
                alpha *= 0.8
                momentum *= 0.5
                if no_improve > 200:
                    alpha = max(alpha, 0.0001)
                    if alpha < 0.0001:
                        break
                    no_improve = 0

        if no_improve > 30 and rng.random() < 0.1:
            freq = rng.randint(1, n//2)

            pert = amp * np.sin(2*np.pi*freq*x/2.0 + phase)
            h_try = project_feasible(h + pert, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                no_improve = 0
                momentum = np.zeros(n)

def coordinate_descent(h, n, budget_s, rng, grid=50):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_start = time.time()
    c = compute_corr(h, n)
    
    while time.time() < t_start + budget_s:
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for p in indices:
            if time.time() > t_start + budget_s:
                break
            d_lo = -h[p] + 1e-10
            d_hi = 1.0 - h[p] - 1e-10
            if d_hi - d_lo < 1e-12:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2*n-2, max(p + n - 1, 2*n-2-p))
            um = 0.0
            if k_lo > 0:
                um = max(um, c[:k_lo].max())
            if k_hi < 2*n-2:
                um = max(um, c[k_hi+1:].max())
            
            ks = np.arange(k_lo, k_hi+1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p+n-1)
            bwd = (ks >= n-1-p) & (ks <= 2*n-2-p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n-1)]
            
            # Golden section or grid search
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = um  # current value at d=0 is max(um, c[ks].max())
            cur_max_ks = c[ks].max()
            best_val = max(um, cur_max_ks)
            
            for d in deltas:
                if abs(d) < 1e-12:
                    continue
                new_c_ks = c[ks] + d * s
                ci = (ks == (n-1))
                if np.any(ci):
                    new_c_ks[ci] = c[n-1] + d*(1.0 - 2.0*h[p]) - d*d
                val = max(um, new_c_ks.max())
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-10:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n-1)]
                c[n-1] -= d*d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                cur_max = max(um, c[ks].max())
                if cur_max < best_c * n / 2.0 - 1e-10:
                    best_c = cur_max * 2.0 / n
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    
    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    return best_h, best_c

def fourier_restart(h, n, rng, n_freqs=5):
    """Create a perturbed version using Fourier modes."""
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.randint(1, n//2)
        amp = 0.02 + 0.08 * rng.random()
        ph = rng.uniform(0, 2*np.pi)
        pert += amp * np.sin(2*np.pi*f*x/2.0 + ph)
    return project_feasible(h + pert, n)

def binary_relaxation(h, n, rng):
    """Push towards binary and then relax."""
    # Threshold
    h_bin = (h > 0.5).astype(float)
    # Fix the count
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones
        excess = n_ones - target
        ones_idx = np.where(h_bin == 1)[0]
        # Remove the ones with h value closest to 0.5
        scores = h[ones_idx]
        to_remove = ones_idx[np.argsort(scores)[:excess]]
        h_bin[to_remove] = 0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(h_bin == 0)[0]
        scores = h[zeros_idx]
        to_add = zeros_idx[np.argsort(scores)[-deficit:]]
        h_bin[to_add] = 1
    return project_feasible(h_bin + 0.1 * (h - 0.5), n)

def smooth_objective(x, n, alpha=50000):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        soft = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        soft = np.log(np.sum(np.exp(sh))) / alpha
    return float((m + soft) * dx)

def slsqp_optimize(h, n, alpha=100000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(smooth_objective, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (abs(diff) / tot) * h

            p1s = max(0, -m); p1e = min(n, n - m)
            grad[p1s:p1e] += active_weights[k] * (1.0 - h[p1s+m:p1e+m])
            p2s = max(0, m); p2e = min(n, m + n)
            grad[p2s:p2e] -= active_weights[k] * h[p2s-m:p2e-m]
    return grad

def subgrad_optimize(h, n, budget_s, rng, n_active=25, use_anti=False):
    h = project_feasible(h.copy(), n)
    if use_anti:
        h = project_anti(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_start = time.time()
    alpha = 0.008
    momentum = np.zeros(n)
    no_improve = 0
    stall_count = 0

    while time.time() < t_start + budget_s:
        c = compute_corr(h, n)
        c_max = c.max()
        sorted_c = np.sort(c)[::-1]
        thresh = sorted_c[min(n_active, 2*n-2)]
        spread = max(c_max - thresh, 1e-10)
        weights = np.where(c >= thresh, (c - thresh) / spread, 0.0)
        
        grad = grad_of_corr(h, n, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
 
        momentum = 0.85 * momentum - alpha * grad
        h_new = h + momentum
        if use_anti:
            h_new = project_anti(h_new, n)
        else:
            h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-9:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rng.random() < 0.4:
                alpha = min(alpha * 1.05, 0.05)
        else:
            no_improve += 1
            h = best_h.copy()
            if no_improve > 30:
                alpha *= 0.92
                momentum *= 0.6
            if no_improve > 100:
                stall_count += 1
                alpha = max(0.0005, alpha * 0.7)
                if use_anti:
                    h = project_anti(h + rng.normal(0, 0.03, n), n)
                else:
                    h = project_feasible(h + rng.normal(0, 0.03, n), n)
                momentum = np.zeros(n)
                no_improve = 0
                if stall_count > 8:
                    break
        if alpha < 0.0003:
            break
    return best_h, best_c

def coord_descent(h, n, budget_s, rng, grid=50):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_start = time.time()
    c = compute_corr(h, n)
    stall = 0

    while time.time() < t_start + budget_s and stall < 3:
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for p in indices:
            if time.time() > t_start + budget_s:
                break
            d_lo = -h[p] + 1e-10
            d_hi = 1.0 - h[p] - 1e-10
            if d_hi - d_lo < 1e-12:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2*n-2, max(p+n-1, 2*n-2-p))
            um = 0.0
            if k_lo > 0:
                um = max(um, c[:k_lo].max())
            if k_hi < 2*n-2:
                um = max(um, c[k_hi+1:].max())
            ks = np.arange(k_lo, k_hi+1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p+n-1)
            bwd = (ks >= n-1-p) & (ks <= 2*n-2-p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n-1)]
            cur_max = max(um, c[ks].max())
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = cur_max
            for d in deltas:
                if abs(d) < 1e-12:
                    continue
                new_c_ks = c[ks] + d * s
                ci = (ks == (n-1))
                if np.any(ci):
                    new_c_ks[ci] = c[n-1] + d*(1.0-2.0*h[p]) - d*d
                val = max(um, new_c_ks.max())
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-10:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n-1)]
                c[n-1] -= d*d
                h[p] = np.clip(h[p]+d, 0.0, 1.0)
                cur_max = max(um, c[ks].max())
                if cur_max * 2.0 / n < best_c - 1e-10:
                    best_c = cur_max * 2.0 / n
                    best_h = h.copy()
                improved = True
        if not improved:
            stall += 1
    h = project_feasible(h, n)
    fc = compute_c5(h, n)
    if fc < best_c:
        best_c, best_h = fc, h.copy()
    return best_h, best_c

def _sobj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        soft = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        soft = np.log(np.sum(np.exp(sh))) / alpha
    return float((m + soft) * dx)

def _cfun(x, n):
    return x.sum() - n / 2.0

def _cjac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def binary_local_search(h, n, budget_s, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[np.argsort(h[idx])[:n_ones-target]]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[np.argsort(-h[idx])[:target-n_ones]]] = 1
    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best = h_bin.copy()
    t_start = time.time()
    c_best = compute_corr(best, n)
    while time.time() < t_start + budget_s:
        improved = False
        for _ in range(200):
            if time.time() > t_start + budget_s:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or best[i] == best[j]:
                continue
            best[i], best[j] = best[j], best[i]
            c = compute_corr(best, n)
            cval = c.max() * 2.0 / n
            if cval < best_c - 1e-12:
                best_c = cval
                c_best = c
                improved = True
            else:
                best[i], best[j] = best[j], best[i]
        if not improved:
            break
    return best, best_c

def fourier_perturb(h, n, rng, n_freqs=3, amp_range=(0.02, 0.1)):
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.randint(1, n // 2)
        amp = amp_range[0] + (amp_range[1] - amp_range[0]) * rng.random()
        ph = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * x / 2.0 + ph)
    return project_feasible(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Subgradient on full space
    t_alloc = min(180, (deadline - t0) * 0.25)
    if t_alloc > 20:
        h_sg, c_sg = subgrad_optimize(best_h, n, t_alloc, rng, n_active=25, use_anti=False)
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg

    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        t_alloc = min(120, (deadline - time.time()) * 0.2)
        h_cd, c_cd = coord_descent(best_h, n, t_alloc, rng, grid=60)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd

    # Phase 3: Anti-periodic optimization
    if time.time() < deadline - 180:
        h_anti = project_anti(best_h, n)
        c_anti = compute_c5(h_anti, n)
        t_alloc = min(120, (deadline - time.time()) * 0.2)
        h_a, c_a = subgrad_optimize(h_anti, n, t_alloc, rng, n_active=25, use_anti=True)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
        else:
            # Try from random anti-periodic start
            a_rand = np.clip(0.5 + rng.normal(0, 0.15, n // 2), 0, 1)
            h_r = np.empty(n)
            h_r[:n//2] = a_rand
            h_r[n//2:] = 1.0 - a_rand
            t_alloc2 = min(80, (deadline - time.time()) * 0.15)
            if t_alloc2 > 20:
                h_r2, c_r2 = subgrad_optimize(h_r, n, t_alloc2, rng, n_active=25, use_anti=True)
                h_r2, c_r2 = coord_descent(h_r2, n, min(30, deadline - time.time() - 50), rng, grid=40)
                if c_r2 < best_c:
                    best_c, best_h = c_r2, h_r2

    # Phase 4: SLSQP
    if time.time() < deadline - 100:
        for a in [80000, 300000]:
            if time.time() > deadline - 70:
                break
            h_s = slsqp_opt(best_h, n, alpha=a, maxiter=800)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c, best_h = c_s, h_s

    # Phase 5: Multi-restart with Fourier perturbations
    if time.time() < deadline - 80:
        for r in range(5):
            if time.time() > deadline - 50:
                break
            h_p = fourier_perturb(best_h, n, rng, n_freqs=rng.randint(2, 5),
                                   amp_range=(0.03, 0.12))
            t_sub = min(40, (deadline - time.time() - 30) * 0.4)
            if t_sub < 15:
                break
            h_new, c_new = subgrad_optimize(h_p, n, t_sub * 0.5, rng, n_active=20)
            h_new, c_new = coord_descent(h_new, n, min(20, t_sub * 0.4), rng, grid=40)
            if c_new < best_c:
                best_c, best_h = c_new, h_new

    # Phase 6: Binary local search + relaxation
    if time.time() < deadline - 60:
        h_bin, c_bin = binary_local_search(best_h, n, min(30, deadline - time.time() - 30), rng)
        h_rel = project_feasible(0.3 * h_bin + 0.7 * best_h, n)
        h_rel, c_rel = coord_descent(h_rel, n, min(25, deadline - time.time() - 20), rng, grid=50)
        if c_rel < best_c:
            best_c, best_h = c_rel, h_rel

    # Phase 7: Try smaller n
    for n_try in [200, 150]:
        if n_try == n or time.time() > deadline - 60:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        t_sub = min(35, (deadline - time.time()) * 0.4)
        if t_sub < 20:
            break
        h_new, c_new = coord_descent(h_int, n_try, t_sub * 0.5, rng, grid=50)
        if time.time() < deadline - 20:
            h_s = slsqp_opt(h_new, n_try, alpha=200000, maxiter=500)
            c_s = compute_c5(h_s, n_try)
            if c_s < c_new:
                h_new, c_new = h_s, c_s
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Final polish
    if time.time() < deadline - 15:
        n_f = len(best_h)
        h_f = slsqp_opt(best_h, n_f, alpha=500000, maxiter=1500)
        c_f = compute_c5(h_f, n_f)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        n_f = len(best_h)
        best_h, best_c = coord_descent(best_h, n_f, min(10, deadline - time.time() - 2), rng, grid=80)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
