    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:

            if tot < 1e-15:
                break
            h += (diff / tot) * h

def soft_max_obj(x, n, alpha, lam=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cd = c * dx
    m = cd.max()
    z = alpha * (cd - m)
    z_max = z.max()
    logsum = z_max + np.log(np.sum(np.exp(z - z_max)))
    obj = (m + logsum / alpha)
    if lam > 0:
        s = x.sum() - n / 2.0
        obj += lam * s * s
    return obj

def soft_max_grad(x, n, alpha, lam=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cd = c * dx
    m = cd.max()
    z = alpha * (cd - m)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    p = exp_z / exp_z.sum()
    
    grad = np.zeros(n)
    w = dx * p  # weights for each correlation value
    
    # Zero lag: c[n-1], dc/dx[p] = 1-2x[p]
    grad += w[n-1] * (1 - 2*x)
    
    # For efficiency, precompute the weights
    # Positive lags: k=1..n-1, c[n-1+k]
    # dc/dx[p] = (1-x[p-k]) for p>=k, minus x[p+k] for p<=n-1-k
    for k in range(1, n):

        if wk < 1e-18:
            continue
        grad[k:] += wk * (1 - x[:n-k])
        if n-k > 0:

    # dc/dx[p] = (1-x[p+k]) for p<=n-1-k, minus x[p-k] for p>=k
    for k in range(1, n):
        wk = w[n-1-k]
        if wk < 1e-18:
            continue
        if n-k > 0:
            grad[:n-k] += wk * (1 - x[k:])
        grad[k:] -= wk * x[:n-k]
    
    if lam > 0:
        s = x.sum() - n / 2.0
        grad += 2 * lam * s * np.ones(n)
    
    return grad

def lbfgs_optimize(h, n, alpha, lam, maxiter, deadline):
    def obj(x):
        return soft_max_obj(x, n, alpha, lam)
    def grad(x):
        return soft_max_grad(x, n, alpha, lam)
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if time.time() > deadline:
            return project_feasible(h, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

- h[p] in (1-h[j]) when j=p, valid if 0≤p≤n-1-ℓ
  → -h[p+ℓ] for 0≤p≤n-1-ℓ

dc_{-ℓ}/dh[p]:
- h[p] in h[j-ℓ] when j-ℓ=p, j=p+ℓ, valid if ℓ≤p+ℓ≤n-1, i.e., 0≤p≤n-1-ℓ
  → (1-h[p+ℓ]) for 0≤p≤n-1-ℓ
- h[p] in (1-h[j]) when j=p, valid if ℓ≤p≤n-1
  → -h[p-ℓ] for ℓ≤p≤n-1

So in terms of output indices:
- output[n-1+ℓ] = c_ℓ for ℓ≥0
- output[n-1-ℓ] = c_{-ℓ} for ℓ≥0

        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        alloc = min(100, (deadline - time.time() - 100) * 0.3)
        if alloc < 30:
            break
        for alpha in [100, 500]:
            if time.time() > time.time() + alloc:
                break
            h_opt = lbfgs_optimize(h_interp, n_try, alpha, 1e5, 600, time.time() + alloc)
            c_opt = compute_c5(h_opt, n_try)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
    
    # Phase 4: Fourier-based exploration
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        h_four = fourier_optimize(best_h, n_cur, time.time() + 60, rng, K=20)
        c_four = compute_c5(h_four, n_cur)
        if c_four < best_c:
            best_h = h_four
            best_c = c_four
            # Polish the fourier result
            if time.time() < deadline - 60:
                h_opt = lbfgs_optimize(best_h, n_cur, 500, 1e5, 500, deadline - 30)
                c_opt = compute_c5(h_opt, n_cur)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
    
    # Phase 5: Final polish with coordinate descent
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        h_cd, c_cd = coord_descent_fast(best_h, n_cur, deadline - 10, grid=100, rng=rng)
        h_cd = project_feasible(h_cd, n_cur)
        c_cd = compute_c5(h_cd, n_cur)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    
    # Phase 6: One more L-BFGS polish with high alpha
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h_final = lbfgs_optimize(best_h, n_cur, 2000, 1e6, 500, deadline - 5)
        c_final = compute_c5(h_final, n_cur)
        if c_final < best_c:
            best_h, best_c = h_final, c_final
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
