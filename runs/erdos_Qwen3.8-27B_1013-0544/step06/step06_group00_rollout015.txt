import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            if diff > 0:
                i = np.argmin(h)
                h[i] = min(1.0, h[i] + abs(diff) / n * 2)
            else:
                i = np.argmax(h)
                h[i] = max(0.0, h[i] - abs(diff) / n * 2)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    exp_s = np.exp(s)
    log_sum = np.log(exp_s.sum())
    obj = (m + log_sum / alpha) * dx
    p = exp_s / exp_s.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        m_shift = k - (n - 1)
        # vec1[j] = 1 - h[j+m_shift] if 0<=j+m_shift<n else 1
        # vec2[j] = h[j-m_shift] if 0<=j-m_shift<n else 0
        if m_shift == 0:
            v1 = 1.0 - h
            v2 = h.copy()
        elif m_shift > 0:
            v1 = np.ones(n)
            v1[m_shift:] = 1.0 - h[:n-m_shift]
            v2 = np.zeros(n)
            v2[m_shift:] = h[:n-m_shift]
        else:

    h = h.copy()

            if obj_new <= obj or it % 50 == 0:

            else:

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = binary_h(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    m_range = np.arange(-(n-1), n)
    T = 0.01
    cool = 0.9999
    it = 0
    while time.time() < deadline:
        it += 1
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        if i == j:
            continue
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(-(nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
