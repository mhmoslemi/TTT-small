                vals = np.maximum(um, np.max(c_slice + outer, axis=1))
                if has_quad:
                    vals = np.maximum(vals, c_n1 + grid * s_n1 - grid * grid)
                idx_b = int(np.argmin(vals))
                best_d = grid[idx_b]
                if _ref > 0:
                    lo = grid[max(0, idx_b - 1)]
                    hi = grid[min(len(grid) - 1, idx_b + 1)]
                    if hi - lo < 1e-13:
                        break
                    grid = np.linspace(lo, hi, 8)
            if abs(best_d) > 1e-13:
                c[j_lo:j_hi + 1] = c_slice + best_d * s
                if has_quad:
                    c[n - 1] = c_n1 + best_d * s_n1 - best_d * best_d
                h[p] = np.clip(hp + best_d, 0.0, 1.0)
                improved = True
        if not improved:
            break
    return h, float(np.max(c)) * dx

def obj_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n))
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * (2.0 / n))

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def bimodal_push(h, n, gamma=3.0):
    t = 2.0 * h - 1.0
    pushed = 0.5 + 0.5 * np.sign(t) * np.abs(t) ** gamma
    return project_feasible(pushed, n)

def pair_search_fast(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_c = cv
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    t_sub = min(120, deadline - time.time() - 80)
    if t_sub > 20:
        h2, c2 = precise_cd(h, n, max_passes=300, deadline=time.time() + t_sub, rng=rng)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c2
    if time.time() < deadline - 60:
        h3 = slsqp_opt(h, n, alpha=100000, maxiter=1000)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            h, best_c = h3, c3
    if time.time() < deadline - 40:
        h4, c4 = precise_cd(h, n, max_passes=200, deadline=deadline - 20, rng=rng)
        h4 = project_feasible(h4, n)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            h, best_c = h4, c4
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Optimize initial at its native n
    h1, c1 = optimize_candidate(h_init, n_init, t0 + 250, rng)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Try other n values via interpolation
    x_best = np.linspace(0, 2, len(best_h))
    n_tries = [n_init]
    for delta_n in [-50, 50, -100, 100, 150]:
        nt = n_init + delta_n
        if 100 <= nt <= 800 and nt != n_init:
            n_tries.append(nt)
    for n_try in n_tries:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        alloc = min(140, (deadline - time.time()) * 0.3)
        if alloc < 60:
            break
        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_best = np.linspace(0, 2, len(best_h))
    # Phase 3: Bimodal push + re-optimize
    if time.time() < deadline - 150:
        n = len(best_h)
        for gamma in [2.5, 4.0, 6.0]:
            if time.time() > deadline - 80:
                break
            h_push = bimodal_push(best_h, n, gamma)
            alloc = min(80, (deadline - time.time()) * 0.3)
            if alloc < 40:
                break
            h_opt, c_opt = optimize_candidate(h_push, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
    # Phase 4: Gaussian restarts
    if time.time() < deadline - 120:
        n = len(best_h)
        for scale in [0.02, 0.05, 0.1, 0.15]:
            if time.time() > deadline - 60:
                break
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(70, (deadline - time.time()) * 0.3)
            if alloc < 35:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
    # Phase 5: SLSQP high-alpha polish
    if time.time() < deadline - 50:
        n = len(best_h)
        h_s = slsqp_opt(best_h, n, alpha=500000, maxiter=2000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
    # Phase 6: Final precise CD
    if time.time() < deadline - 15:
        n = len(best_h)
        h_f, c_f = precise_cd(best_h, n, max_passes=300, deadline=deadline - 5, rng=rng)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Phase 7: Pair search for fine tuning
    if time.time() < deadline - 10:
        n = len(best_h)
        best_h, best_c = pair_search_fast(best_h, n, deadline - 3, radius=25)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
