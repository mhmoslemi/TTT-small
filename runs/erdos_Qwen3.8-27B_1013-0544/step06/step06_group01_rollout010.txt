import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_grad_weighted(h, n, w):
    """Weighted sum of d(c_k)/d(h[i]) for all lags k."""
    grad = np.zeros(n)
    omh = 1.0 - h
    len_w = len(w)
    for k in range(len_w):
        wk = w[k]
        if wk < 1e-10:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * omh[idx - k + n - 1]
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= wk * h[idx + k - n + 1]
    return grad

def lse_grad(h, n, alpha):
    """Gradient of LSE-smoothed max correlation, scaled by dx."""
    dx = 2.0 / n
    c = compute_corr(h, n)
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh = np.clip(sh, -80, 80)
    exp_sh = np.exp(sh)
    s = exp_sh.sum()
    w = exp_sh / s
    obj = (cmax + np.log(s) / alpha) * dx
    grad = corr_grad_weighted(h, n, w) * dx
    return obj, grad

def topk_grad(h, n, k_top):
    """Gradient from top-k lags only."""
    dx = 2.0 / n
    c = compute_corr(h, n)
    idx = np.argsort(c)[-k_top:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k_top
    grad = corr_grad_weighted(h, n, w) * dx
    return grad

def smooth_grad_descent(h, n, deadline, alphas, step_init=0.05):
    """LSE-smoothed gradient descent with adaptive step."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = step_init
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        t_phase_end = deadline - 10
        for it in range(50000):
            if time.time() > t_phase_end:
                break
            obj, grad = lse_grad(h, n, alpha)
            gnorm = np.sqrt(np.sum(grad * grad))
            if gnorm < 1e-12:
                break
            direction = -grad / gnorm
            # Adaptive step
            h_new = project_feasible(h + step * direction, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                if c_new < best_c - 1e-6:
                    step = min(step * 1.1, 0.5)
                else:
                    step *= 0.95
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                h_new = project_feasible(h + step * 0.5 * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < compute_c5(h, n):
                    h = h_new
                    step *= 0.85
                else:
                    step *= 0.6
                    if step < 1e-7:
                        break
    return best_h, best_c

def topk_gradient_descent(h, n, deadline, k_top=15):
    """Top-k active set gradient descent."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.03
    stall = 0
    while time.time() < deadline:
        grad = topk_grad(h, n, k_top)
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-12:
            break
        direction = -grad / gnorm
        h_new = project_feasible(h + step * direction, n)
        c_new = compute_c5(h_new, n)
        current_c = compute_c5(h, n)
        if c_new < current_c:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            step = min(step * 1.02, 0.15)
        else:
            step *= 0.7
            stall += 1
            if stall > 80:
                break
    return best_h, best_c

def coord_descent(h, n, deadline, grid=80, max_passes=200):
    """Fast coordinate descent."""
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    dx = 2.0 / n
    sumh = float(h.sum())
    rng = np.random.RandomState(123)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for p in indices:
            if h[p] <= 1e-10 or h[p] >= 1.0 - 1e-10:
                continue
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n - 1)]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def slsqp_refine(h, n, alpha=200000, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        sh = np.clip(sh, -60, 60)
        return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Phase 1: LSE gradient descent (smooth to sharp)
    t1 = min(deadline - 60, (deadline - time.time()) * 0.3)
    if time.time() + 10 < deadline:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
        h_gd, c_gd = smooth_grad_descent(h, n, time.time() + t1, alphas)
        if c_gd < best_c:
            best_c, best_h = c_gd, h_gd
    # Phase 2: Top-k gradient descent
    t2 = min(deadline - 40, (deadline - time.time()) * 0.2)
    if time.time() + 10 < deadline:
        h_tk, c_tk = topk_gradient_descent(best_h, n, time.time() + t2, k_top=10)
        if c_tk < best_c:
            best_c, best_h = c_tk, h_tk
    # Phase 3: Coordinate descent
    t3 = min(deadline - 30, (deadline - time.time()) * 0.25)
    if time.time() + 10 < deadline:
        h_cd, c_cd = coord_descent(best_h, n, time.time() + t3, grid=60, max_passes=150)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd
    # Phase 4: SLSQP polish
    if time.time() < deadline - 20:
        h_s = slsqp_refine(best_h, n, alpha=300000, maxiter=1500)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_c, best_h = c_s, h_s
    # Phase 5: Final top-k + CD
    if time.time() < deadline - 10:
        h_tk2, c_tk2 = topk_gradient_descent(best_h, n, time.time() + min(20, deadline - time.time() - 5), k_top=5)
        if c_tk2 < best_c:
            best_c, best_h = c_tk2, h_tk2
    if time.time() < deadline - 5:
        h_cd2, c_cd2 = coord_descent(best_h, n, time.time() + min(15, deadline - time.time() - 2), grid=100, max_passes=50)
        h_cd2 = project_feasible(h_cd2, n)
        c_cd2 = compute_c5(h_cd2, n)
        if c_cd2 < best_c:
            best_c, best_h = c_cd2, h_cd2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Optimize parent at its native resolution
    alloc1 = min(250, deadline - t0 - 200)
    h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try other resolutions by interpolation
    for n_try in [200, 250, 400, 500]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 50:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Multi-restart from best
    if time.time() < deadline - 100:
        n = len(best_h)
        n_restarts = max(2, int((deadline - time.time() - 50) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            scale = 0.03 + 0.05 * (r / max(1, n_restarts - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(70, (deadline - time.time() - 40) * 0.35)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 15:
        n = len(best_h)
        h_f = slsqp_refine(best_h, n, alpha=1000000, maxiter=3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        h_final, c_final = topk_gradient_descent(best_h, len(best_h), deadline - 2, k_top=5)
        if c_final < best_c:
            best_c, best_h = c_final, h_final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
