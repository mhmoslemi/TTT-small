    best_h = h.copy()
    n2 = 2 * n - 1
    momentum = np.zeros(n)
    mu = 0.7
    for it in range(500000):
        if time.time() > deadline:
            break
        frac = min(1.0, it / 30000.0)
        alpha = 5.0 * (20000.0) ** frac
        lr = 2.0 / (alpha + 1.0) * (1.0 / (1.0 + it * 0.0001))
        corr = compute_corr(h, n)
        c_max = float(corr.max())
        shifted = corr - c_max
        exps = np.exp(alpha * shifted)
        wsum = exps.sum()
        if wsum < 1e-30:
            weights = np.ones(n2) / n2
        else:
            weights = exps / wsum
        grad = np.zeros(n)
        for k in range(n2):
            w = weights[k]
            if w < 1e-10:
                continue
            d = k - (n - 1)
            lo1 = max(0, d)
            hi1 = min(n, n + d)
            if lo1 < hi1:
                grad[lo1:hi1] += w * (1.0 - h[lo1 - d:hi1 - d])
            lo2 = max(0, -d)
            hi2 = min(n, n - d)
            if lo2 < hi2:
                grad[lo2:hi2] -= w * h[lo2 + d:hi2 + d]
        momentum = mu * momentum + (1 - mu) * grad
        h_new = project_feasible(h - lr * momentum, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def sa_continuous(h0, n, deadline, seed=42, t0=0.05, t1=0.0005):
    h = h0.copy()
    dx = 2.0 / n
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    n_iter = max(1, int((deadline - time.time()) * 300))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            i = int(rng.randint(n))
            h[i] = 1.0 - h[i]
        else:
            i = int(rng.choice(zeros_idx))
            j = int(rng.choice(ones_idx))
            h[i], h[j] = h[j], h[i]
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / max(T, 1e-12)):
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
        else:
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, n)
            else:
                h[i] = 1.0 - h[i]
                h = project_feasible(h, n)
    return best_h, best_c

def greedy_polish(h, n, deadline):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = compute_corr(h, n)
    cur_max = float(corr.max())
    best_c = cur_max
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        k_star = int(corr.argmax())
        lag = k_star - (n - 1)
        if lag >= 0:
            valid_i = np.arange(0, n - lag)
        else:
            valid_i = np.arange(-lag, n)
        if len(valid_i) == 0:
            break
        mask = (h[valid_i] == 1) & (h[valid_i + lag] == 0)
        contributing = valid_i[mask]
        zeros_idx = np.where(h == 0)[0]
        for i in contributing:
            best_j = -1
            best_new = cur_max
            for j in zeros_idx:
                if j == i + lag:
                    continue
                h2 = h.copy()
                h2[i], h2[j] = 0.0, 1.0
                nc = float(compute_corr(h2, n).max())
                if nc < best_new:
                    best_new = nc
                    best_j = j
            if best_j >= 0 and best_new < cur_max:
                h[i], h[best_j] = 0.0, 1.0
                corr = compute_corr(h, n)
                cur_max = best_new
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
                improved = True
                zeros_idx = np.where(h == 0)[0]
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    p1 = t0 + 0.15 * (budget_s - 5)
    hb, cb = targeted_binary_search(best_h, n, p1, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    p2 = t0 + 0.30 * (budget_s - 5)
    if time.time() < p2:
        hb2, cb2 = two_swap_search(best_h, n, p2, seed=seed + 100)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
    
    p3 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3:
        for attempt in range(3):
            if time.time() > p3 - 30:
                break
            sub_dl = min(p3 - 10, time.time() + (p3 - time.time()) / (3 - attempt))
            h_init = best_h + 0.02 * rng.randn(n)
            h_init = project_feasible(h_init, n)
            hg, cg = gradient_minimax(h_init, n, sub_dl, seed=seed + attempt * 50)
            if cg < best_c:
                best_c, best_h = cg, hg.copy()
            hb_b, cb_b = targeted_binary_search(hg, n, min(p3 - 5, time.time() + 20), seed=seed + attempt * 37)
            if cb_b < best_c:
                best_c, best_h = cb_b, hb_b.copy()
    
    p4 = t0 + 0.75 * (budget_s - 5)
    for restart in range(8):
        if time.time() > p4 - 15:
            break
        h_p = best_h.copy()
        n_swap = rng.randint(3, 8)
        ones = np.where(h_p >= 0.5)[0]
        zeros = np.where(h_p < 0.5)[0]
        for _ in range(n_swap):
            if len(ones) > 0 and len(zeros) > 0:
                a, b = int(rng.choice(ones)), int(rng.choice(zeros))
                h_p[a], h_p[b] = h_p[b], h_p[a]
                ones = np.where(h_p >= 0.5)[0]
                zeros = np.where(h_p < 0.5)[0]
        sub_dl = min(p4 - 5, time.time() + 25)
        hs, cs = sa_continuous(h_p, n, sub_dl, seed=seed + restart * 113 + 5)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()
        hb, cb = targeted_binary_search(hs, n, min(p4 - 3, time.time() + 15), seed=seed + restart * 71 + 3)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    
    p5 = t0 + 0.90 * (budget_s - 5)
    if time.time() < p5:
        hg, cg = gradient_minimax(best_h.copy(), n, p5, seed=seed + 999)
        if cg < best_c:
            best_c, best_h = cg, hg.copy()
    
    if time.time() < deadline - 10:
        hb, cb = greedy_polish(best_h, n, deadline - 5)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
