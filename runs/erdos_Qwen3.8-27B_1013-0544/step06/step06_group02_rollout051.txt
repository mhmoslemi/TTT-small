def slsqp_optimize(h, n, alpha, deadline, maxit=2000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac_eq, 'args': (n,)}]
    try:
        res = minimize(_slsqp_f, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def fine_pair_search(h, n, deadline, max_rounds=20000):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    alphas = [200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        stalled = 0
        for _ in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            c = compute_c5_vec(h, n)
            g = smooth_max_grad(h, n, alpha)
            gc = g - g.mean()
            # Try top-K candidate pairs
            order_i = np.argsort(gc)[:8]
            order_j = np.argsort(gc)[::-1][:8]
            improved = False
            best_delta = 0.0
            best_pair = None
            best_c_new = best_c
            for ii in order_i:
                for jj in order_j:
                    if ii == jj:
                        continue
                    lo = max(-h[ii], h[jj] - 1.0)
                    hi = min(1.0 - h[ii], h[jj])
                    if hi - lo < 1e-12:
                        continue
                    # Binary search for best delta (convex objective)
                    blo, bhi = lo, hi
                    for _ in range(25):
                        mid = (blo + bhi) / 2.0
                        h[ii] += mid
                        h[jj] -= mid
                        c_new = compute_c5(h, n)
                        if c_new < best_c_new:
                            best_c_new = c_new
                            best_delta = mid
                            best_pair = (ii, jj)
                            blo = mid
                        else:
                            bhi = mid
                        h[ii] -= mid
                        h[jj] += mid
                    if best_c_new < best_c - 1e-15:
                        i2, j2 = best_pair
                        h[i2] += best_delta
                        h[j2] -= best_delta
                        best_c = best_c_new
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                stalled += 1
                if stalled > 50:
                    break
    return h, best_c

def gradient_polish(h, n, deadline, lr=0.005):
    best_c = compute_c5(h, n)
    alpha = 50000
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        step = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            alpha *= 1.3
            if lr < 1e-14:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: L-BFGS-B with increasing alpha (0-150s)
    p1_end = min(t0 + 150, deadline - 400)
    configs = [(50, 0.1, 300), (200, 0.5, 300), (1000, 1.0, 300),
               (5000, 1.0, 200), (20000, 1.0, 200), (100000, 5.0, 100)]
    for alpha, mu, it in configs:
        if time.time() > p1_end:
            break
        alloc = min(35, p1_end - time.time() - 1)
        if alloc < 5:
            break
        h2, c2 = lbfgs_optimize(h, n, alpha, mu, time.time() + alloc, maxit=it)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2

    # Phase 2: SLSQP refinement (150-280s)
    p2_end = min(t0 + 280, deadline - 300)
    for alpha in [20000, 100000, 500000]:
        if time.time() > p2_end:
            break
        alloc = min(45, p2_end - time.time() - 1)
        if alloc < 10:
            break
        h2, c2 = slsqp_optimize(h, n, alpha, time.time() + alloc, maxit=1500)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2

    # Phase 3: Try different n values (280-500s)
    p3_end = min(t0 + 500, deadline - 350)
    n_tries = [280, 320, 350, 260]
    for n_try in n_tries:
        if time.time() > p3_end:
            break
        alloc = p3_end - time.time()
        if alloc < 40:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.clip(np.interp(x_new, x_old, best_h), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        # Optimize at new n
        t_sub = time.time()
        h_t, c_t = lbfgs_optimize(h_interp, n_try, 5000, 1.0, t_sub + min(25, alloc * 0.5), maxit=200)
        if time.time() < p3_end - 20:
            h_t, c_t = slsqp_optimize(h_t, n_try, 100000, time.time() + min(25, p3_end - time.time() - 10), maxit=1000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
            best_n = n_try
            h = h_t
            n = n_try

    # Phase 4: Gradient polish (500-700s)
    p4_end = min(t0 + 700, deadline - 300)
    if time.time() < p4_end:
        h, best_c = gradient_polish(h, n, p4_end, lr=0.003)
        best_h = h.copy()

    # Phase 5: Fine pair search (700-1050s)
    p5_end = min(t0 + 1050, deadline - 30)
    if time.time() < p5_end:
        h, best_c = fine_pair_search(h, n, p5_end)
        best_h = h.copy()

    # Phase 6: Final gradient polish
    if time.time() < deadline - 3:
        h, best_c = gradient_polish(h, n, deadline - 2, lr=0.001)
        best_h = h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
