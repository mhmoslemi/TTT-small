    p2 = t0 + 0.50 * total
    h = best_h.copy()
    alphas = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 30000000]
    for a in alphas:
        if time.time() > p2 - 8:
            break
        h = optimize_lbfgs(h, n, a, 4000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Binary search from smoothed, then re-smooth
    p3 = t0 + 0.62 * total
    hb, cb = binary_local_search(best_h, n, p3, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    p3b = t0 + 0.76 * total
    h = best_h.copy()
    for a in [5000, 50000, 500000, 5000000, 50000000]:
        if time.time() > p3b - 5:
            break
        h = optimize_lbfgs(h, n, a, 3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Multi-resolution exploration
    p4 = t0 + 0.90 * total
    xo = np.linspace(0, 2, n)
    tried_ns = set()
    for nh in [n + 50, n + 100, 500, 600, 300]:
        if time.time() > p4 - 12:
            break
        if nh == n or nh in tried_ns or nh < 100:
            continue
        tried_ns.add(nh)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h, assume_sorted=True), nh)
        bh, bc = binary_local_search(hh, nh, min(time.time() + 15, p4 - 8), max_sweeps=80)
        if bc < compute_c5(hh, nh):
            hh = bh
        for a in [5000, 50000, 500000, 5000000]:
            if time.time() > p4 - 6:
                break
            hh = optimize_lbfgs(hh, nh, a, 2000)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
                xo = np.linspace(0, 2, n)
                break

    # Phase 5: Final polish with high-precision optimization
    p5 = dl - 2
    if time.time() < p5:
        for trial in range(5):
            if time.time() > p5 - 1:
                break
            noise_scale = 0.003 * (trial + 1)
            h = best_h + noise_scale * rng.randn(n)
            h = project_feasible(h, n)
            for a in [10000000, 100000000, 1000000000]:
                if time.time() > p5 - 0.5:
                    break
                h = optimize_lbfgs(h, n, a, 5000)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
                    break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
