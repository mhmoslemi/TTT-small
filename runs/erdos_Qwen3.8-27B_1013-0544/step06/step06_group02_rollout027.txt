def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    # final small correction
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            add = min(diff, room[idx])
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            h[idx] -= sub
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            add = min(diff, room[idx])
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            h[idx] -= sub
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c_raw(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def make_best(h, n):
    h = project_feasible(h, n)
    c = compute_c_raw(h, n)
    return h, float(c.max()) * (2.0 / n)


def update_best(best, h, n):
    h, c = make_best(h, n)
    if c < best[1]:
        return (h, c, n)
    return best


def smooth_weights(c, alpha):
    if alpha <= 0.0:
        return np.full(len(c), 1.0 / len(c), dtype=float)
    s = alpha * c
    s = s - s.max()
    e = np.exp(s)
    e /= e.sum()
    return e


def gradient_from_weights(w, h, n):
    g = np.zeros(n, dtype=float)
    m = 2 * n - 1
    for p in range(m):
        wp = w[p]
        if wp <= 1e-14:
            continue
        t = n - 1 - p
        if t >= 0:
            nv = n - t
            g[:nv] += wp * (1.0 - h[t:])
            g[t:] += -wp * h[:nv]
        else:
            d = -t
            nv = n - d
            g[:nv] += -wp * h[d:]
            g[d:] += wp * (1.0 - h[:nv])
    return g


def smooth_obj(x, n, alpha, mu):
    c = compute_c_raw(x, n)
    s = alpha * c
    sm = s.max()
    val = (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = compute_c_raw(x, n)
    s = alpha * c
    sm = s.max()
    e = np.exp(s - sm)
    e /= e.sum()
    g = gradient_from_weights(e, x, n)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g


def lbfgs_phase(h, n, deadline, maxiter=60, alphas=(30, 100, 300), mu=5.0):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c_raw(h, n).max() * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(
                smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-15,
                         "gtol": 1e-12, "maxfun": max(50, maxiter * 10)}
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c_raw(h2, n).max() * (2.0 / n)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                h = h2
        except Exception:
            pass
    return best_h, best_c


def derivative_for_index(r, h, n):
    m = 2 * n - 1
    d = np.zeros(m, dtype=float)
    p1 = np.arange(r, n + r)
    q1 = n - 1 + r - p1
    d[p1] += 1.0 - h[q1]
    p2 = np.arange(n - 1 - r, 2 * n - 1 - r)
    q2 = r - n + 1 + p2
    d[p2] += -h[q2]
    return d


def pair_eval(t, c_eff, a_eff, b_eff, uncons):
    v = c_eff + a_eff * t
    v = v + b_eff * t * t
    mv = float(v.max())
    return mv if mv > uncons else uncons


def pair_line_search(h, c, i, j, n):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo <= 1e-13:
        return None
    di = derivative_for_index(i, h, n)
    dj = derivative_for_index(j, h, n)
    a = di - dj
    m = 2 * n - 1
    b = np.zeros(m, dtype=float)
    b[n - 1 - (j - i)] = 1.0
    b[n - 1 - (i - j)] = 1.0
    cmax = float(c.max())
    active = c >= cmax - 1e-9
    if not np.any(active):
        return None
    sl = a[active]
    right = float(sl.max())
    left = float(sl.min())
    stol = 1e-8
    if right < -stol:
        aa, bb = 0.0, hi
    elif left > stol:
        aa, bb = lo, 0.0
    else:
        return None
    if bb - aa <= 1e-12:
        return None
    mask = (a != 0.0) | (b != 0.0)
    ce = c[mask]
    ae = a[mask]
    be = b[mask]
    if np.any(~mask):
        un = float(c[~mask].max())
    else:
        un = -np.inf
    gr = 0.6180339887498949
    t1 = bb - gr * (bb - aa)
    t2 = aa + gr * (bb - aa)
    f1 = pair_eval(t1, ce, ae, be, un)
    f2 = pair_eval(t2, ce, ae, be, un)
    for _ in range(32):
        if f1 < f2:
            bb = t2
            t2 = t1
            f2 = f1
            t1 = bb - gr * (bb - aa)
            f1 = pair_eval(t1, ce, ae, be, un)
        else:
            aa = t1
            t1 = t2
            f1 = f2
            t2 = aa + gr * (bb - aa)
            f2 = pair_eval(t2, ce, ae, be, un)
    cand_t = (aa, bb, 0.5 * (aa + bb), 0.0)
    fbest = cmax
    tb = 0.0
    for tt in cand_t:
        ft = pair_eval(tt, ce, ae, be, un)
        if ft < fbest:
            fbest = ft
            tb = tt
    if abs(tb) <= 1e-14:
        return None
    if fbest < cmax - 1e-8 * (cmax + 1.0):
        return tb, c + a * tb + b * tb * tb
    return None


def active_derivative_matrix(h, n, active_p):
    p = int(active_p.size)
    d = np.zeros((p, n), dtype=float)
    for r, idx_p in enumerate(active_p):
        pidx = int(idx_p)
        t = n - 1 - pidx
        g = np.zeros(n, dtype=float)
        if t >= 0:
            nv = n - t
            g[:nv] += 1.0 - h[t:]
            g[t:] += -h[:nv]
        else:
            dd = -t
            nv = n - dd
            g[:nv] += -h[dd:]
            g[dd:] += 1.0 - h[:nv]
        d[r, :] = g
    return d


def pair_descent(h, n, dx, deadline, rng, K=12, active_tol=1e-5, max_active=30, stage0=0):
    h = project_feasible(h, n)
    c = compute_c_raw(h, n)
    stage = stage0
    noimp = 0
    accepts = 0
    while time.time() < deadline:
        kk = min(K + 4 * stage, 28)
        alpha = 32.0 * (2.0 ** min(stage, 4))
        w = smooth_weights(c, alpha)
        g = gradient_from_weights(w, h, n)
        order = np.argsort(g + 1e-8 * rng.uniform(size=n))
        lows = order[:kk]
        highs = order[-kk:][::-1]
        improved = False
        for ii in range(kk):
            i = int(lows[ii])
            for jj in range(kk):
                if time.time() > deadline:
                    break
                j = int(highs[jj])
                if i == j:
                    continue
                res = pair_line_search(h, c, i, j, n)
                if res is not None:
                    t, new_c = res
                    h[i] += t
                    h[j] -= t
                    if h[i] < 0.0:
                        h[i] = 0.0
                    elif h[i] > 1.0:
                        h[i] = 1.0
                    if h[j] < 0.0:
                        h[j] = 0.0
                    elif h[j] > 1.0:
                        h[j] = 1.0
                    c = new_c
                    accepts += 1
                    improved = True
                    if accepts % 25 == 0:
                        if abs(h.sum() - n * 0.5) > 1e-9:
                            h = project_feasible(h, n)
                        c2 = compute_c_raw(h, n)
                        if float(np.max(np.abs(c2 - c))) > 1e-4:
                            c = c2
            if time.time() > deadline:
                break
        if time.time() < deadline and stage < 4:
            cmax = float(c.max())
            act = np.where(c >= cmax - active_tol)[0]
            if act.size:
                if act.size > max_active:
                    act = act[np.argsort(c[act])[-max_active:]]
                D = active_derivative_matrix(h, n, act)
                limit = 450 + 250 * stage
                count = 0
                wraps = 0
                for j in range(n):
                    if count >= limit or wraps >= 35 or time.time() > deadline:
                        break
                    a = D - D[:, j][:, None]
                    pos = np.nonzero(np.all(a < -1e-6, axis=0))[0][:30]
                    neg = np.nonzero(np.all(a > 1e-6, axis=0))[0][:30]
                    for i in pos:
                        i = int(i)
                        count += 1
                        if count >= limit or time.time() > deadline:
                            break
                        res = pair_line_search(h, c, i, j, n)
                        if res is not None:
                            t, new_c = res
                            h[i] += t
                            h[j] -= t
                            if h[i] < 0.0:
                                h[i] = 0.0
                            elif h[i] > 1.0:
                                h[i] = 1.0
                            if h[j] < 0.0:
                                h[j] = 0.0
                            elif h[j] > 1.0:
                                h[j] = 1.0
                            c = new_c
                            accepts += 1
                            improved = True
                            wraps += 1
                            if wraps >= 35:
                                break
                    if time.time() > deadline:
                        break
                    for i in neg:
                        i = int(i)
                        count += 1
                        if count >= limit or time.time() > deadline:
                            break
                        res = pair_line_search(h, c, i, j, n)
                        if res is not None:
                            t, new_c = res
                            h[i] += t
                            h[j] -= t
                            if h[i] < 0.0:
                                h[i] = 0.0
                            elif h[i] > 1.0:
                                h[i] = 1.0
                            if h[j] < 0.0:
                                h[j] = 0.0
                            elif h[j] > 1.0:
                                h[j] = 1.0
                            c = new_c
                            accepts += 1
                            improved = True
                            wraps += 1
                            if wraps >= 35:
                                break
                if wraps >= 35:
                    if abs(h.sum() - n * 0.5) > 1e-9:
                        h = project_feasible(h, n)
                    c = compute_c_raw(h, n)
        if time.time() > deadline:
            break
        if improved:
            noimp = 0
        else:
            noimp += 1
            if noimp >= 3:
                break
            stage += 1
    h = project_feasible(h, n)
    c = compute_c_raw(h, n)
    return h, c, float(c.max()) * dx


def random_h(n, rng, kind):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if kind == 0:
        val = np.full(n, 0.5)
        for _ in range(4):
            m = int(rng.randint(1, 8))
            ph = rng.uniform(0.0, 2.0 * np.pi)
            amp = rng.uniform(0.10, 0.45)
            val += amp * np.cos(m * np.pi * x + ph)
        h = 0.5 + 0.5 * np.tanh(1.5 * (val - 0.5))
        h = np.clip(h, 0.0, 1.0)
    elif kind == 1:
        m = int(rng.randint(1, 6))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.70, 1.20)
        h = np.clip(0.5 + amp * np.cos(m * np.pi * x + ph), 0.0, 1.0)
    else:
        base = (rng.rand(n) > 0.5).astype(float)
        k = int(rng.randint(3, 7))
        base = np.convolve(base, np.ones(k) / float(k), mode="same")
        h = np.clip(0.15 + 0.70 * base, 0.0, 1.0)
    return project_feasible(h, n)


def optimize_candidate(h0, n, deadline, rng, do_lbfgs=True, lb_iters=25, pair_K=8):
    dx = 2.0 / n
    h = project_feasible(h0, n)
    c = compute_c_raw(h, n)
    C = float(c.max()) * dx
    if do_lbfgs and deadline - time.time() > 25:
        alloc = min(45.0, (deadline - time.time()) * 0.4)
        if alloc > 18:
            h2, C2 = lbfgs_phase(h, n, time.time() + alloc, maxiter=lb_iters, alphas=(30, 120), mu=5.0)
            if C2 < C:
                h = h2
                c = compute_c_raw(h, n)
                C = float(c.max()) * dx
    if deadline - time.time() > 12:
        h, c, C = pair_descent(h, n, dx, deadline - 2, rng, K=pair_K)
    return h, C


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        hpar = np.array(initial_h_values, dtype=float).copy()
        npar = int(len(hpar))
    except Exception:
        hpar = np.full(300, 0.5)
        npar = 300
    if npar < 60:
        hpar = np.full(300, 0.5)
        npar = 300
    best = update_best((None, float("inf"), 0), hpar, npar)

    if time.time() < deadline - 260:
        ndx = 2.0 / best[2]
        h, c, C = pair_descent(best[0].copy(), best[2], ndx, deadline - 260, rng, K=12)
        best = update_best(best, h, best[2])

    if time.time() < deadline - 170:
        n = best[2]
        h = best[0].copy()
        alloc = min(70.0, deadline - time.time() - 170)
        if alloc > 18:
            h, C = lbfgs_phase(h, n, time.time() + alloc, maxiter=80, alphas=(60, 180, 450), mu=5.0)
            best = update_best(best, h, n)

    if time.time() < deadline - 75:
        n = best[2]
        ndx = 2.0 / n
        h, c, C = pair_descent(best[0].copy(), n, ndx, deadline - 75, rng, K=20)
        best = update_best(best, h, n)

    if time.time() < deadline - 55:
        nbase = best[2]
        for n2 in (256, 320, 400, 500):
            if time.time() > deadline - 45:
                break
            if n2 == nbase:
                continue
            rem = deadline - time.time()
            alloc = min(75.0, rem * 0.25)
            if alloc < 28:
                break
            x_old = np.linspace(0.0, 2.0, nbase, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h0 = np.interp(x_new, x_old, best[0], left=best[0][0], right=best[0][-1])
            h, C = optimize_candidate(h0, n2, time.time() + alloc, rng, True, 22, 8)
            best = update_best(best, h, n2)

    if time.time() < deadline - 38:
        n = best[2]
        h0 = best[0].copy()
        h0 = project_feasible(h0 + 0.08 * (rng.rand(n) - 0.5), n)
        rem = deadline - time.time()
        alloc = min(65.0, rem * 0.45)
        if alloc > 20:
            h, C = optimize_candidate(h0, n, time.time() + alloc, rng, False, 0, 14)
            best = update_best(best, h, n)

    if time.time() < deadline - 28:
        n = best[2] if 150 <= best[2] <= 700 else 300
        for kind in range(4):
            if time.time() > deadline - 22:
                break
            rem = deadline - time.time()
            alloc = min(48.0, rem * 0.55)
            if alloc < 20:
                break
            h0 = random_h(n, rng, kind)
            h, C = optimize_candidate(h0, n, time.time() + alloc, rng, True, 22, 8)
            best = update_best(best, h, n)

    if time.time() < deadline - 2:
        n = best[2]
        ndx = 2.0 / n
        h, c, C = pair_descent(best[0].copy(), n, ndx, deadline - 1, rng, K=16)
        best = update_best(best, h, n)

    return (best[0], float(best[1]), int(best[2]))
