def project_box_sum(x,T):
  y=x.copy()
  for _ in range(20):
    if y.min()<0 or y.max()>1:
      y=np.clip(y,0,1)
    diff=T-y.sum()
    if abs(diff)<1e-14: return y
    # adjust only non-saturated
    active=(y>1e-12)&(y<1-1e-12)
    if active.sum()==0: return y
    # If need add, can add equally to active; but adding may saturate; if subtract similarly.
    y[active]+=diff/active.sum()
  return np.clip(y,0,1)

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return x
    s = np.sort(x)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(s)
    lo = s[0] - 1.0
    hi = s[-1] + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        l = int(np.searchsorted(s, mid, side='right'))
        r = int(np.searchsorted(s, mid + 1.0, side='left'))
        val = float(n - r) + float(pref[r] - pref[l]) - float(r - l) * mid
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(x - lam, 0.0, 1.0)
    for _ in range(4):
        diff = target - float(y.sum())
        if abs(diff) < 1e-13:
            break
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt == 0:
            break
        y[free] += diff / float(cnt)
    y = np.clip(y, 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-15:
        free_idx = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if free_idx.size > 0:
            for idx in free_idx:
                add = np.clip(diff, -y[idx], 1.0 - y[idx])
                y[idx] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
    return y

def grad_matrix(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    L = 2 * n - 1
    dx = 2.0 / float(n)
    G = np.zeros((L, n), dtype=float)
    for r in range(L):
        k = r - (n - 1)
        row = G[r]
        lo = k if k > 0 else 0
        hi = n + k
        if hi > n:
            hi = n
        if hi > lo:
            row[lo:hi] += 1.0
            row[lo:hi] -= h[lo - k: hi - k]
        lo2 = -k if k < 0 else 0
        hi2 = n - k
        if hi2 > n:
            hi2 = n
        if hi2 > lo2:
            row[lo2:hi2] -= h[lo2 + k: hi2 + k]
        row *= dx
    return G

def lp_polish(h, t_end, max_steps=30):

    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = c5_of(h)
    c = best_c
    stalled = 0
    for _ in range(max_steps):
        if time.time() > t_end:
            break
        g = lags_c5(h)
        cm = float(np.max(g))

            best_c = cm
            best_h = h.copy()
            c = cm

        L = G.shape[0]

        Aeq = np.zeros((1, n + 1), dtype=float)

        cvec = np.zeros(n + 1, dtype=float)
        cvec[n] = 1.0

        res = None

            res = linprog(cvec, A, b, Aeq, np.array([target]), bounds=bounds, options={'presolve': True})

def active_subgrad(h, g, topB=32, beta=10.0):
    n = int(h.size)
    dx = 2.0 / float(n)
    if g.size <= topB:
        idx = np.arange(g.size)
    else:
        idx = np.argpartition(g, -topB)[-topB:]
    top = g[idx]
    mx = float(top.max())
    sh = np.clip(beta * (top - mx), -50.0, 50.0)
    w = np.exp(sh)
    ws = float(w.sum())
    if ws <= 0.0:
        w = np.full(idx.size, 1.0 / idx.size)
    else:
        w = w / ws
    grad = np.zeros(n, dtype=float)
    for r, wt in zip(idx, w):
        k = int(r) - (n - 1)
        lo = k if k > 0 else 0
        hi = n + k
        if hi > n:
            hi = n
        if hi > lo:
            grad[lo:hi] += wt
            grad[lo:hi] -= wt * h[lo - k: hi - k]
        lo2 = -k if k < 0 else 0

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def lags_c5(h):
    n = int(len(h))
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full").astype(float) * dx


def c5_of(h):
    return float(np.max(lags_c5(h)))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return x
    s = np.sort(x)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(s)

    lo = float(s[0] - 1.0)
    hi = float(s[-1] + 1.0)
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        l = int(np.searchsorted(s, mid, side="right"))
        r = int(np.searchsorted(s, mid + 1.0, side="left"))
        val = float(n - r) + float(pref[r] - pref[l]) - float(r - l) * mid
        if val > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    y = np.clip(x - lam, 0.0, 1.0)

    for _ in range(4):
        diff = target - float(y.sum())
        if abs(diff) < 1e-13:
            break
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt == 0:
            break
        y[free] += diff / float(cnt)

    y = np.clip(y, 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-15:
        free_idx = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        for idx in free_idx:
            add = np.clip(diff, -y[idx], 1.0 - y[idx])
            y[idx] += add
            diff -= add
            if abs(diff) < 1e-15:
                break
    return y


def grad_matrix(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    L = 2 * n - 1
    dx = 2.0 / float(n)
    G = np.zeros((L, n), dtype=float)
    for r in range(L):
        k = r - (n - 1)
        row = G[r]

        lo = k if k > 0 else 0
        hi = n + k
        if hi > n:
            hi = n
        if hi > lo:
            row[lo:hi] += 1.0
            row[lo:hi] -= h[lo - k: hi - k]

        lo2 = -k if k < 0 else 0
        hi2 = n - k
        if hi2 > n:
            hi2 = n
        if hi2 > lo2:
            row[lo2:hi2] -= h[lo2 + k: hi2 + k]

        row *= dx
    return G


def lp_polish(h, t_end, max_steps=30):
    if linprog is None:
        target = 0.5 * float(len(h))
        h = project_box_sum(h, target)
        return h, c5_of(h)

    n = int(len(h))
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = c5_of(h)
    c = best_c
    stalled = 0

    for _ in range(max_steps):
        if time.time() > t_end:
            break

        g = lags_c5(h)
        cm = float(np.max(g))
        if cm < best_c:
            best_c = cm
            best_h = h.copy()
            c = cm

        G = grad_matrix(h)
        b = G @ h - g
        L = G.shape[0]

        A = np.empty((L, n + 1), dtype=float)
        A[:, :n] = G
        A[:, n] = -1.0

        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        b_eq = np.array([target], dtype=float)

        cvec = np.zeros(n + 1, dtype=float)
        cvec[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(float(max(-2.0, cm - 2.0)), float(cm + 2.0))]

        res = None
        rem = max(1.0, t_end - time.time() - 1.0)
        try:
            res = linprog(
                cvec, A, b, Aeq, b_eq, bounds=bounds,
                options={"presolve": True, "time_limit": float(rem)}
            )
        except Exception:
            try:
                res = linprog(cvec, A, b, Aeq, b_eq, bounds=bounds, options={"presolve": True})
            except Exception:
                try:
                    res = linprog(cvec, A, b, Aeq, b_eq, bounds=bounds)
                except Exception:
                    res = None

        if res is not None and hasattr(res, "x") and res.x is not None:
            hn = np.asarray(res.x[:n], dtype=float)
            if np.all(np.isfinite(hn)):
                hn = np.clip(hn, 0.0, 1.0)
                d = hn - h
                if np.linalg.norm(d) > 1e-8:
                    best_lam = 0.0
                    best_hh = None
                    best_cand = c
                    for lam in (1.0, 0.85, 0.7, 0.55, 0.45,
                                0.35, 0.25, 0.15, 0.08, 0.04, 0.02):
                        if time.time() > t_end:
                            break
                        hh = project_box_sum(h + lam * d, target)
                        cc = c5_of(hh)
                        if cc < best_cand - 1e-14:
                            best_cand = cc
                            best_lam = lam
                            best_hh = hh
                    if best_lam > 0.0 and best_hh is not None:
                        h = best_hh
                        c = best_cand
                        if c < best_c:
                            best_c = c
                            best_h = h.copy()
                        stalled = 0
                        continue

        stalled += 1
        if stalled >= 4:
            break

    return best_h, best_c


def active_subgrad(h, g, topB=32, beta=10.0):
    n = int(h.size)
    dx = 2.0 / float(n)
    topB = max(1, int(topB))

    if g.size <= topB:
        idx = np.arange(g.size)
    else:
        idx = np.argpartition(g, -topB)[-topB:]

    top = g[idx]
    mx = float(top.max())
    sh = np.clip(beta * (top - mx), -50.0, 50.0)
    w = np.exp(sh)
    ws = float(w.sum())
    if ws <= 0.0:
        w = np.full(idx.size, 1.0 / float(idx.size))
    else:
        w = w / ws

    grad = np.zeros(n, dtype=float)
    for r, wt in zip(idx, w):
        k = int(r) - (n - 1)

        lo = k if k > 0 else 0
        hi = n + k
        if hi > n:
            hi = n
        if hi > lo:
            grad[lo:hi] += wt
            grad[lo:hi] -= wt * h[lo - k: hi - k]

        lo2 = -k if k < 0 else 0
        hi2 = n - k
        if hi2 > n:
            hi2 = n
        if hi2 > lo2:
            grad[lo2:hi2] -= wt * h[lo2 + k: hi2 + k]

    return grad * dx


def smooth_polish(h, t_end, max_iter=3000, topB=32, beta=10.0):
    n = int(len(h))
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = c5_of(h)
    c = best_c
    scale = 1.0

    for _ in range(max_iter):
        if time.time() > t_end:
            break

        g = lags_c5(h)
        cm = float(np.max(g))
        if cm < best_c:
            best_c = cm
            best_h = h.copy()
            c = cm

        grad = active_subgrad(h, g, topB, beta)
        grad -= float(np.mean(grad))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12:
            break

        base = 0.05 / max(gn, 1e-12)
        step = min(base * scale, 600.0)

        hh = project_box_sum(h - step * grad, target)
        cc = c5_of(hh)
        if cc < c - 1e-14:
            h = hh
            c = cc
            if c < best_c:
                best_c = c
                best_h = h.copy()
            scale = min(1.5, 1.12 * scale)
            continue

        step2 = step * 0.6
        hh = project_box_sum(h - step2 * grad, target)
        cc = c5_of(hh)
        if cc < c - 1e-14:
            h = hh
            c = cc
            if c < best_c:
                best_c = c
                best_h = h.copy()
            scale = min(1.5, step2 / max(step, 1e-12) * scale)
        else:
            scale *= 0.65
            if scale < 1e-4:
                break

    return best_h, best_c


def make_start(n, rng, style):
    n = int(n)
    target = 0.5 * float(n)
    st = int(style) % 4

    if st == 0:
        h = np.zeros(n, dtype=float)
        order = np.argsort(rng.rand(n))
        h[order[: n // 2]] = 1.0
    elif st == 1:
        q = int(rng.randint(3, 11))
        p = int(rng.randint(1, q))
        h = ((np.arange(n) % q) < p).astype(float)
    elif st == 2:
        x = np.linspace(0.0, 2.0, n)
        h = np.full(n, 0.5)
        for _ in range(5):
            fr = int(rng.randint(1, 13))
            amp = float(rng.uniform(0.06, 0.25))
            if rng.rand() < 0.5:
                amp = -amp
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * fr * x / 2.0 + ph)
        h = np.clip(h, 0.0, 1.0)
    else:
        h = np.full(n, 0.5)
        h = h + 0.25 * rng.randn(n)
        h = np.clip(h, 0.0, 1.0)

    return project_box_sum(h, target)
