def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fast_cd(h, n, max_passes=800, grid=50, deadline=None, rng=None, penalty=5.0):
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    n_corr = 2 * n - 1
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - p, p))
            m_hi = min(n_corr - 1, max(n - 1 + p, p + n - 1))
            um = 0.0
            if m_lo > 0:
                um = float(np.max(c[:m_lo]))
            if m_hi < n_corr - 1:
                um = max(um, float(np.max(c[m_hi + 1:])))
            ks = np.arange(m_lo, m_hi + 1)
            coeff = np.zeros(len(ks))
            quad = np.zeros(len(ks))
            for idx, m in enumerate(ks):
                q = p + m - (n - 1)
                if 0 <= q < n:
                    if q == p:
                        coeff[idx] += 1.0 - 2.0 * h[p]
                        quad[idx] += 1.0
                    else:
                        coeff[idx] += 1.0 - h[q]
                i2 = p - m + (n - 1)
                if 0 <= i2 < n and i2 != p:
                    coeff[idx] -= h[i2]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n / 2.0) ** 2
            for d in deltas:
                if abs(d) < 1e-15:
                    continue
                new_c = c[ks] + d * coeff - d * d * quad
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - n / 2.0) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                c[ks] += best_d * coeff - best_d * best_d * quad
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def smooth_blend(h, n, sigma=1.5):
    """Apply a small Gaussian blur and re-project to escape local optima."""
    from scipy.ndimage import gaussian_filter1d
    hs = gaussian_filter1d(h, sigma)
    return project_feasible(hs, n)

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase A: coordinate descent
    t_sub = min(deadline - time.time(), 200)
    if t_sub > 30:
        h_cd, c_cd = fast_cd(h, n, max_passes=600, grid=50,
                             deadline=time.time() + t_sub, rng=rng, penalty=5.0)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase B: SLSQP with annealing
    if time.time() < deadline - 80:
        for a in [20000, 80000, 300000]:
            if time.time() > deadline - 50:
                break
            h_s = slsqp_opt(h, n, alpha=a, maxiter=2000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                h, best_c = h_s, c_s
    # Phase C: smoothing + re-optimize
    if time.time() < deadline - 80:
        h_sm = smooth_blend(h, n, sigma=2.0)
        h_cd2, c_cd2 = fast_cd(h_sm, n, max_passes=400, grid=50,
                               deadline=time.time() + 60, rng=rng, penalty=2.0)
        h_cd2 = project_feasible(h_cd2, n)
        c_cd2 = compute_c5(h_cd2, n)
        if c_cd2 < best_c:
            h, best_c = h_cd2, c_cd2
            if time.time() < deadline - 40:
                h_s = slsqp_opt(h, n, alpha=200000, maxiter=1500)
                c_s = compute_c5(h_s, n)
                if c_s < best_c:
                    h, best_c = h_s, c_s
    # Phase D: fine coordinate descent
    if time.time() < deadline - 30:
        h_cd3, c_cd3 = fast_cd(h, n, max_passes=400, grid=80,
                               deadline=time.time() + 80, rng=rng, penalty=0.5)
        h_cd3 = project_feasible(h_cd3, n)
        c_cd3 = compute_c5(h_cd3, n)
        if c_cd3 < best_c:
            h, best_c = h_cd3, c_cd3
    # Phase E: pair search
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline - 3,
                                [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Optimize initial construction with generous budget
    alloc1 = min(300, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_candidate(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    for n_try in [250, 350]:
        if time.time() > deadline - 200:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_init, best_h), n_try)
        alloc = min(180, (deadline - time.time()) * 0.3)
        if alloc < 60:
            break
        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Multi-restart from best with diverse perturbations
    if time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restart = max(2, int(remaining / 100))
        for r in range(n_restart):
            if time.time() > deadline - 80:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.03, n), n)
            elif r % 3 == 1:
                # Smooth blend then perturb
                hp = smooth_blend(best_h, n, sigma=2.0 + r * 0.5)
                hp = project_feasible(hp + rng.normal(0, 0.02, n), n)
            else:
                # Binary-ish perturbation
                mask = rng.random(n) < 0.1
                hp = best_h.copy()
                hp[mask] += rng.uniform(-0.1, 0.1, mask.sum())
                hp = project_feasible(hp, n)
            alloc = min(80, (deadline - time.time() - 40) * 0.5)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final SLSQP polish on best
    if time.time() < deadline - 30:
        n = len(best_h)
        h_f = slsqp_opt(best_h, n, alpha=500000, maxiter=3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    # Final fine pair search
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline - 2,
                                    [0.001, 0.0005, 0.0002, 0.0001], radius=50)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
