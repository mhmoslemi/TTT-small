```python
def subgradient_method(h, n, deadline, n_iters=5000):
    t = 0.05  # initial step size
    h_avg = np.zeros(n)
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    
    for k in range(n_iters):
        if time.time() > deadline:
            break
        
        dx = 2.0 / n
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        
        # Find active lags (within tolerance of max)
        tol = 0.01 * dx  # small tolerance
        active = np.where(c >= c_max - tol)[0]
        
        # Compute subgradient (average of active lags)
        g = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            g += _lag_grad(h_curr, n, lag)  # without dx
        g /= len(active)
        g *= dx  # full gradient
        
        # Take step
        h_new = h_curr - t * g
        h_new = project_feasible(h_new, n)
        
        # Update Polyak average
        h_avg = (k * h_avg + h_new) / (k + 1)
        
        # Also try the average
        c_avg = compute_c5(h_avg, n)
        if c_avg < best_c:
            best_c = c_avg
            best_h = h_avg.copy()
        
        h_curr = h_new
        
        # Decay step size
        t = 0.05 / np.sqrt(k + 1)
    
    return best_h, best_c
```

```python
        if n_elite > pop_size * 0.6:
            sigma *= 1.03
        elif n_elite < pop_size * 0.15:
            sigma *= 0.9
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i], 1.0 - h[j])
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j], 1.0 - h[i])
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() < deadline - 40:
        t1 = min(120, (deadline - time.time()) * 0.35)
        h1, c1 = subgrad_descent(h, n, t1, top_m=5, init_t=0.04)
        if c1 < best_c:
            best_c, best_h = c1, h1
    if time.time() < deadline - 30:
        t2 = min(100, (deadline - time.time()) * 0.3)
        h2, c2 = mirror_descent(h, n, t2, top_m=3, init_t=0.025)
        if c2 < best_c:
            best_c, best_h = c2, h2
    if time.time() < deadline - 20:
        t3 = min(80, (deadline - time.time()) * 0.3)
        h3, c3 = evolution_strategy(best_h, n, t3, rng, pop_size=25)
        if c3 < best_c:
            best_c, best_h = c3, h3
    if time.time() < deadline - 15:
        for alpha in [20000, 100000]:
            if time.time() > deadline - 15:
                break
            hr = slsqp_refine(best_h, n, alpha, 2500)
            cr = compute_c5(hr, n)
            if cr < best_c:
                best_c, best_h = cr, hr
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    n_candidates = [150, 200, 250, 300, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 90:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 70:
            break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(55, (deadline - time.time() - 30) * 0.45)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        hr = slsqp_refine(best_h, n, 500000, 4000)
        cr = compute_c5(hr, n)
        if cr < best_c:
            best_c, best_h = cr, hr
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    return best_h, best_c, len(best_h)
```
