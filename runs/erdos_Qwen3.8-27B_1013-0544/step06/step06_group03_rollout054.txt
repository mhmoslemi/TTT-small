def slsqp_opt(h, n, alpha, maxiter=4000, ftol=1e-16):
    def fg(x):
        return obj_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(fg, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif m < n:
        grad[m:] += (1.0 - h[:n - m]) * dx
        grad[:n - m] -= h[m:] * dx
    return grad

def bundle_method(h_init, n, deadline, max_iters=300):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cuts = []
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        c = compute_all_lags(h, n)
        top_m = min(2 * n - 1, max(30, 2 * n // 5))
        top_idx = np.argsort(c)[-top_m:]
        for k in top_idx:
            lag = k - (n - 1)
            g = lag_gradient(h, n, lag)
            cuts.append((h.copy(), c[k], g))
        if len(cuts) > 300:
            cuts = cuts[-250:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for (h_pt, c_val, g) in cuts:
            cons.append(c_val + g @ (h_var - h_pt) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h))
        h = h_new
        if step < 1e-10:
            break
    return best_h, best_c

def freeze_optimize(h, n, deadline, threshold=0.1):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(15):
        if time.time() > deadline - 10:
            break
        free = (h > threshold) & (h < 1.0 - threshold)
        n_free = int(free.sum())
        if n_free < 3:
            break
        free_idx = np.where(free)[0]
        target_sum_free = n / 2.0 - h[~free].sum()

        def red_obj(x_f):
            h_full = h.copy()
            h_full[free_idx] = x_f
            h_full = np.clip(h_full, 0, 1)
            dx = 2.0 / n
            c = np.correlate(h_full, 1 - h_full, mode='full') * dx
            m = c.max()
            shifted = 200000 * (c - m)
            smx = shifted.max()
            es = np.exp(shifted - smx)
            return m + (smx + np.log(es.sum())) / 200000

        def red_con(x_f):
            return x_f.sum() - target_sum_free

        def red_con_jac(x_f):
            return np.ones(n_free)
        try:
            res = minimize(red_obj, h[free_idx].copy(), method='SLSQP',
                          bounds=[(0, 1)] * n_free,
                          constraints=[{'type': 'eq', 'fun': red_con, 'jac': red_con_jac}],
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = h.copy()
            h_new[free_idx] = np.clip(res.x, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for ds in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            step = max(1, (n - m) // 300)
            for i in range(0, n - m, step):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-10:
                    delta = min(ds, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-9:
                        continue
                    oi, oj = h[i], h[j]
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i], h[j] = oi, oj
                elif h[j] > h[i] + 1e-10:
                    delta = min(ds, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-9:
                        continue
                    oi, oj = h[i], h[j]
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i], h[j] = oi, oj
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm-up
    for alpha in [5000, 20000]:
        if time.time() > deadline - 200:
            break
        h_new = slsqp_opt(best_h, n, alpha, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 2: Bundle method (accumulated cuts)
    if time.time() < deadline - 150:
        b_end = time.time() + min(250, (deadline - time.time()) * 0.45)
        h_new, c_new = bundle_method(best_h, n, b_end)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 3: High-alpha SLSQP
    for alpha in [50000, 150000, 400000]:
        if time.time() > deadline - 80:
            break
        h_new = slsqp_opt(best_h, n, alpha, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 4: Freezing
    if time.time() < deadline - 40:
        f_end = time.time() + min(80, (deadline - time.time()) * 0.5)
        h_new, c_new = freeze_optimize(best_h, n, f_end)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 5: Another bundle round on refined solution
    if time.time() < deadline - 30:
        b_end = time.time() + min(40, (deadline - time.time()) * 0.5)
        h_new, c_new = bundle_method(best_h, n, b_end)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 10:
        h_new = slsqp_opt(best_h, n, 1000000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

    # Phase 7: Fine local search
    if time.time() < deadline - 2:
        best_h, best_c = fine_local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, n
