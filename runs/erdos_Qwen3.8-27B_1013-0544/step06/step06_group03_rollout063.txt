```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h

    vals = c[top_idx]
    m = vals.max()
```

The gradient of the overlap at lag k with respect to h[i] is:
- For lag k > 0: d(overlap_k)/dh[i] = -h[i+k] (if i < n-k) + (1-h[i-k]) (if i >= k)
- For lag k = 0: d(overlap_0)/dh[i] = 1 - 2h[i]
- For lag k < 0: similar to positive case with m = |k|

So the gradient of the max is determined by which lag is active.

For a swap (moving 1 from position i to position j), the change in overlap at lag k is approximately:
Δ_k ≈ gradient_k[i] * (-1) + gradient_k[j] * (+1) = gradient_k[j] - gradient_k[i]

So the swap that most reduces the maximum overlap is the one where gradient_active[j] - gradient_active[i] is most negative.

This gives an O(n²) algorithm per step (try all pairs), which for n=300 is 90000 operations per step. Very fast!

```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

    c = np.correlate(x, 1 - x, mode='full') * dx
```

```python
def project_onto_simplex_box(h, n):
    """Project h onto {x: 0 ≤ x[i] ≤ 1, sum(x) = n/2}."""
    target = n / 2.0
    lo, hi = -h.max(), 1 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
def steepest_descent_one_lag(h, n, lag, n_iter=100):
    """Minimize g_lag(h) using steepest descent with exact line search."""
    h = h.copy()
    dx = 2.0 / n
    for _ in range(n_iter):
        # Gradient
        grad = np.zeros(n)
        if lag == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[:n-lag] = -h[lag:] * dx
            grad[lag:] = (1.0 - h[:n-lag]) * dx
        else:
            m = -lag
            grad[:n-m] = (1.0 - h[m:]) * dx
            grad[m:] = -h[:n-m] * dx
        
        # Line search: minimize g(h + t*(-grad)) for t ≥ 0
        # Since g is quadratic, this is a 1D quadratic min
        # g(h + t*d) = g(h) + t * (grad^T d) + t²/2 * (d^T H d)
        # where d = -grad, H is the Hessian
        # We want: t* = -(grad^T d) / (d^T H d) = (grad^T grad) / (d^T H d)
        
        d = -grad
        # g(h+td) = g(h) + t*grad·d + t²/2 * d·H·d
        # grad·d = -|grad|²
        # d·H·d = grad^T H grad (since d = -grad)
        # t* = |grad|² / (grad^T H grad)
        
        # Compute d^T H d
        # H has H[i,j] = -dx*(1_{j=i+lag} + 1_{j=i-lag})
        # d^T H d = -dx * sum_i (d[i]*d[i+lag] + d[i]*d[i-lag])
        #           = -2*dx * sum_{i < i+lag} d[i]*d[i+lag] (for lag > 0)
        
        if lag > 0:
            dHd = -2.0 * dx * np.dot(d[:n-lag], d[lag:])
        elif lag < 0:
            m = -lag
            dHd = -2.0 * dx * np.dot(d[m:], d[:n-m])
        else:
            dHd = 0.0  # H = 0 for lag 0 (g_0 is linear! Wait, no...)
            # g_0(h) = sum h[i]*(1-h[i]) * dx = (sum h - sum h²) * dx
            # ∂g_0/∂h[i] = (1-2h[i]) * dx
            # ∂²g_0/∂h[i]² = -2*dx
            # So H = -2*dx * I
            dHd = -2.0 * dx * np.dot(d, d)
        
        if abs(dHd) < 1e-12:
            break
        t = np.dot(d, d) / abs(dHd)
        t = np.clip(t, 0.01, 10.0)  # Clip step size
        
        # But we also need to satisfy 0 ≤ h+td ≤ 1 and sum(h+td) = n/2
        # For the sum constraint: sum(h+td) = sum(h) + t*sum(d) = n/2 + t*sum(d)
        # So we need sum(d) = 0 for the constraint to be maintained
        # sum(d) = -sum(grad) = -(sum of all partial derivatives)
        # For lag ≠ 0: sum(grad) = dx * sum((1-h[i+lag])-h[i-lag]) which is not necessarily 0
        
        # This is getting complicated. Let me just use a simple step.
        h_new = h + t * d
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            h = h_new
        else:
            break
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_lse(x, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    vals = c[top_idx]
    m = vals.max()
    shifted = alpha * (vals - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_f(x, n):
    return float(np.sum(x) - n * 0.5)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, top_m, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(_obj_lse, h, args=(n, alpha, top_m), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _grad_lag(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] = -h[lag:]
        g[lag:] = (1.0 - h[:n - lag])
    else:
        m = -lag
        g[:n - m] = (1.0 - h[m:])
        g[m:] = -h[:n - m]
    return g

def local_polish(h, n, deadline, steps=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            idx = int(np.argmin(np.minimum(h, 1.0 - h)))
            if h[idx] >= 0.5:
                h[idx] = min(1.0, h[idx] + 0.08)
            else:
                h[idx] = max(0.0, h[idx] - 0.08)
        else:
            m = abs(lag)
            if lag > 0:
                prod = h[:n - m] * (1.0 - h[m:])
            else:
                prod = h[m:] * (1.0 - h[:n - m])
            i = int(np.argmax(prod))
            if lag > 0:
                if h[i] > h[i + m]:
                    h[i] = max(0.0, h[i] - 0.04)
                else:
                    h[i + m] = min(1.0, h[i + m] + 0.04)
            else:
                if h[i + m] > h[i]:
                    h[i + m] = max(0.0, h[i + m] - 0.04)
                else:
                    h[i] = min(1.0, h[i] + 0.04)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

def swap_polish(h, n, deadline, max_iters=3000):
    h = (h >= 0.5).astype(float)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    stall = 0
    rng = np.random.RandomState(12345)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = _grad_lag(h, n, lag)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        scores = grad[ones][:, None] - grad[zeros][None, :]
        fi = int(np.argmax(scores))
        ii = fi // len(zeros)
        jj = fi % len(zeros)
        i = ones[ii]
        j = zeros[jj]
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
            if stall > 50:
                ri = rng.randint(len(ones))
                rj = rng.randint(len(zeros))
                h[ones[ri]], h[zeros[rj]] = h[zeros[rj]], h[ones[ri]]
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    stall = 0
                else:
                    h[ones[ri]], h[zeros[rj]] = h[zeros[rj]], h[ones[ri]]
                    break
    return h, best_c
```
