```python
def projected_gradient(h, n, alpha, lr, max_iter, deadline):
    h = h.copy().astype(float)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute correlation
        c = np.correlate(h, 1 - h, mode='full') * dx
        # Soft-max weights
        c_max = c.max()
        w = np.exp(alpha * (c - c_max))
        w /= w.sum()
        # Compute gradient
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            k = m - (n - 1)
            wm = w[m]
            if wm < 1e-15:
                continue
            if k >= 0:
                grad[:n-k] += wm * (1 - h[k:])
                grad[k:] -= wm * h[:n-k]
            else:
                kp = -k
                grad[kp:] += wm * (1 - h[:n-kp])
                grad[:n-kp] -= wm * h[kp:]
        grad *= dx
        # Gradient step
        h_new = h - lr * grad
        # Project
        h_new = project_feasible(h_new, n)
        # Check improvement
        c_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if c_new < c_max + 1e-15:
            h = h_new
        else:
            # Halve step size
            lr *= 0.5
            if lr < 1e-8:
                break
    return h
```
