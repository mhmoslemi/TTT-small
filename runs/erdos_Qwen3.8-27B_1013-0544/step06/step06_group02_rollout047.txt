```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def penalty_obj(x, alpha, mu):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def penalty_grad(x, alpha, mu):
        g = grad_smooth_max(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0), (10000, 100.0), (50000, 500.0), (200000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': max_iters * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def cutting_plane_lp(h, n, deadline, tol=1e-8):
    """Iterative LP-based cutting plane method."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    h_var = cp.Variable(n)
    
    for it in range(200):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        C = c.max()
        
        if C < best_c - tol:
            best_c = C
            best_h = h.copy()
        
        # Active set: lags within 5% of max
        threshold = C - max(0.005, 0.02 * C)
        active = np.where(c >= threshold)[0]
        
        if len(active) < 2:
            active = np.where(c >= C - 0.01)[0]
        
        # Compute gradient of each active correlation
        # c[m] = sum over valid j of h[j]*(1-h[j+m-(n-1)]) * dx
        # dc[m]/dh[i] = dx * [(1-h[i+k]) if i in [0, n-k) else 0] 
        #              + dx * [-h[i-k]] if i-k in [0,n) ... 
        # Let me just compute it properly
        
        grad_active = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad_active[idx, 0:nv] = dx * (1.0 - h[k:n])
                grad_active[idx, k:n] = dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad_active[idx, kk:n] = dx * (1.0 - h[0:nv])
                grad_active[idx, 0:nv] = dx * (-h[kk:n])
        
        # Build LP
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        
        for idx, m in enumerate(active):
            lin = c[m] + grad_active[idx] @ (h_var - h)
            constraints.append(lin <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = np.array(h_var.value).flatten()
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        
        if c_new >= C - 1e-10:
            # No improvement, try smaller active set or break
            if len(active) > 3:
                threshold = C - max(0.002, 0.01 * C)
                active = np.where(c >= threshold)[0]
                continue
            break
        
        h = h_new
    
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 0.5), (200, 3.0), (1000, 10.0), (5000, 50.0),
               (20000, 200.0), (100000, 1000.0), (500000, 3000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': max_iters * 8})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def cutting_plane_lp(h, n, deadline, max_rounds=100):
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    prev_c = best_c
    
    for it in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        C = c.max()
        if C < best_c - 1e-12:
            best_c = C
            best_h = h.copy()
        if C >= prev_c - 1e-11:
            break
        prev_c = C
        # Active set
        gap = max(0.003, 0.015 * C)
        active = np.where(c >= C - gap)[0]
        if len(active) < 2:
            active = np.argsort(c)[-5:]
        if len(active) > 200:
            active = np.argsort(c)[-100:]
        # Gradients for active lags
        ga = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                ga[idx, 0:nv] = dx * (1.0 - h[k:n])
                ga[idx, k:n] = dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                ga[idx, kk:n] = dx * (1.0 - h[0:nv])
                ga[idx, 0:nv] = dx * (-h[kk:n])
        # LP formulation
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for idx in range(len(active)):
            lin = c[active[idx]] + ga[idx] @ (h_var - h)
            cons.append(lin <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_lp = np.array(h_var.value).flatten()
        h_lp = np.clip(h_lp, 0, 1)
        # Line search
        found = False
        for lam in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            h_try = (1 - lam) * h + lam * h_lp
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < C - 1e-13:
                h = h_try
                found = True
                break
        if not found:
            break
    return h, best_c

def fast_pair_search(h, n, deadline, radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```
