```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * alpha * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += -wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += -wm * h[kk:n]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_arr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += -wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[kk:]
```

```python
def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    return project_sum(h, n / 2.0)

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_arr(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _block_obj(x, prefix, suffix, n, alpha):
    h_full = np.concatenate([prefix, x, suffix])
    c = compute_c5_arr(h_full, n)
    return float(smooth_max_val(c, alpha))

def _block_jac(x, prefix, suffix, n, alpha):
    h_full = np.concatenate([prefix, x, suffix])
    g = smooth_max_grad(h_full, n, alpha)
    s = len(prefix)
    return g[s:s + len(x)]

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(300, 5.0), (1000, 10.0), (5000, 50.0), (20000, 100.0), (100000, 500.0), (500000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def block_cd(h, n, deadline, block_sizes=None):
    best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    if block_sizes is None:
        block_sizes = [30, 20, 15, 10]
    alphas = [200, 1000, 5000, 20000, 100000, 500000]
    rng = np.random.RandomState(42)
    
    for bs in block_sizes:
        for alpha in alphas:
            while time.time() < deadline - 5:
                pass_improved = False
                # Randomize block order
                starts = list(range(0, n - 2, bs))
                rng.shuffle(starts)
                for start in starts:
                    if time.time() > deadline - 3:
                        return h, best_c
                    m = min(bs, n - start)
                    if m < 3:
                        continue
                    prefix = h[:start].copy()
                    suffix = h[start + m:].copy()
                    target_sum = h[start:start + m].sum()
                    x0 = h[start:start + m].copy()
                    try:
                        res = minimize(_block_obj, x0, jac=_block_jac,
                                      args=(prefix, suffix, n, alpha),
                                      method='L-BFGS-B',
                                      bounds=[(0.0, 1.0)] * m,
                                      options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-10})
                        x_new = project_sum(res.x, target_sum)
                        h[start:start + m] = x_new
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            pass_improved = True
                    except Exception:
                        pass
                if not pass_improved:
                    break
    return h, best_c

def smart_pair_search(h, n, deadline, alpha=500):
    best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline - 3:
            break
        while time.time() < deadline - 2:
            pass_improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                gi = g[i]
                if gi > 0:
                    if h[i] < step * 1.1:
                        continue
                    # Move mass from i to j with lowest gradient
                    target_j = np.argmin(g)
                    if target_j == i or h[target_j] > 1.0 - step:
                        # Try second best
                        g_copy = g.copy()
                        g_copy[i] = np.inf
                        target_j = np.argmin(g_copy)
                        if h[target_j] > 1.0 - step:
                            continue
                    h[i] -= step; h[target_j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; pass_improved = True
                    else:
                        h[i] += step; h[target_j] -= step
                elif gi < 0:
                    if h[i] > 1.0 - step * 1.1:
                        continue
                    target_j = np.argmax(g)
                    if target_j == i or h[target_j] < step * 1.1:
                        g_copy = g.copy()
                        g_copy[i] = -np.inf
                        target_j = np.argmax(g_copy)
                        if h[target_j] < step * 1.1:
                            continue
                    h[i] += step; h[target_j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; pass_improved = True
                    else:
                        h[i] -= step; h[target_j] += step
            if not pass_improved:
                break
    return h, best_c
```
