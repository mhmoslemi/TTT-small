def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.000005, patience=60000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 600000.0)
    T = t_start
    tabu_arr = np.zeros(n * n, dtype=np.int32)
    tabu_counter = 0
    tabu_len = 100

    while time.time() < deadline:
        i = int(zeros_idx[rng.randint(len(zeros_idx))])
        j = int(ones_idx[rng.randint(len(ones_idx))])
        key = i * n + j
        if tabu_arr[key] > tabu_counter:
            continue
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            tabu_arr[key] = tabu_counter + tabu_len
            tabu_arr[j * n + i] = tabu_counter + tabu_len
            tabu_counter += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(10)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                ii = int(zi[rng.randint(len(zi))])
                jj = int(oi[rng.randint(len(oi))])
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 6.0)
            tabu_arr.fill(0)
            tabu_counter = 0
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def targeted_smooth(h, n, deadline):
    """Targeted perturbation: reduce h at positions contributing to peak correlations."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    for _ in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        peak_idx = np.argmax(corr)
        peak_val = corr[peak_idx]
        k = peak_idx - (n - 1)
        if k >= 0:
            src = np.arange(n - k)
            dst = np.arange(k, n)
        else:
            src = np.arange(-k, n)
            dst = np.arange(0, -k)
        contribute = h[src] * (1 - h[dst])
        top_mask = contribute > contribute.mean()
        n_red = max(1, int(np.sum(top_mask) * 0.3))
        red_idx = np.where(top_mask)[0][:n_red]
        if len(red_idx) == 0:
            red_idx = np.zeros(1, dtype=int)
        for ri in red_idx:
            si = src[ri]
            h[si] = max(0, h[si] - step)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            h = best_h.copy()
            step *= 0.95
    return best_h, best_c

def greedy_local(h_bin, n, deadline, max_sweeps=30):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_dc = -1e-10
        best_i, best_j = -1, -1
        for i in zeros:
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_dc:
                    best_dc = dc
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    # Phase 1: Quick SLSQP
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [1000, 5000, 20000, 100000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n0, alpha, 1500, 1e-15)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Binary SA across multiple n values
    n_list = [n0]
    for candidate in [248, 252, 254, 256]:
        if candidate != n0 and candidate >= 240 and candidate <= 260:
            n_list.append(candidate)
    sa_budget = 0.52 * (budget_s - 5)
    sa_per_n = sa_budget / len(n_list)
    for ni in n_list:
        t_start_sa = time.time()
        t_end_sa = min(t_start_sa + sa_per_n, dl - 80)
        if time.time() > t_end_sa:
            continue
        if ni == len(best_h):
            h_init = round_to_binary(best_h, ni)
        else:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, ni)
            h_interp = np.interp(xn, xo, best_h)
            h_init = round_to_binary(h_interp, ni)
        for attempt in range(3):
            if time.time() > t_end_sa:
                break
            s = seed + attempt * 7919 + ni * 31
            if attempt == 0:
                start = h_init
            elif attempt == 1:
                start = round_to_binary(h_init + 0.12 * rng.randn(ni), ni)
            else:
                start = round_to_binary(h_init + 0.25 * rng.randn(ni), ni)
            hb, cb = binary_sa(start, ni, t_end_sa, seed=s,
                              t_start=0.04, t_end=0.000005, patience=50000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
        # Quick greedy after SA for this n
        if time.time() < t_end_sa:
            hb2 = round_to_binary(best_h if len(best_h) == ni else np.interp(np.linspace(0,2,ni), np.linspace(0,2,len(best_h)), best_h), ni)
            hg, cg = greedy_local(hb2, ni, t_end_sa, max_sweeps=20)
            if cg < best_c:
                best_c = cg
                best_h = hg.copy()
    # Phase 3: SLSQP smoothing on best
    p3 = dl - 25
    if time.time() < p3:
        nb = len(best_h)
        hp = project_feasible(best_h.copy(), nb)
        for alpha in [5000, 20000, 100000, 500000, 2000000]:
            if time.time() > p3:
                break
            hp = slsqp_smooth(hp, nb, alpha, 2000, 1e-15)
            c = compute_c5(hp, nb)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    # Phase 4: Targeted smooth
    if time.time() < dl - 10:
        ht, ct = targeted_smooth(best_h, len(best_h), dl - 8)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    # Phase 5: Final greedy
    if time.time() < dl - 3:
        nb = len(best_h)
        hb = round_to_binary(best_h, nb)
        hg, cg = greedy_local(hb, nb, dl - 2, max_sweeps=10)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
