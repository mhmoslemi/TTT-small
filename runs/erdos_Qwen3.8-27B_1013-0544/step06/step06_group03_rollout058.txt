import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

# Helper functions (top-level, no closures)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return _smooth_max(c, alpha)

def _active_obj(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full') * dx
    return _smooth_max(cc[idx], alpha)

def _eq_fun(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_full_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_m, alpha, maxiter=5000, ftol=1e-16):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_active_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_descent(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    npass = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(100):
        if time.time() > deadline:
            break
        sweep_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Bisection to find optimal h[i]
            lo, hi = 0.0, 1.0
            # Evaluate at endpoints
            h[i] = 0.0
            c_lo = compute_c5(h, n)
            h[i] = 1.0
            c_hi = compute_c5(h, n)
            # Since C5(t) is convex in t, use bisection on the subgradient
            # Or just golden section
            a, b = 0.0, 1.0
            t1 = b - npass * (b - a)
            t2 = a + npass * (b - a)
            h[i] = t1
            f1 = compute_c5(h, n)
            h[i] = t2
            f2 = compute_c5(h, n)
            for _ in range(35):
                if f1 < f2:
                    b = t2
                    t2 = t1
                    f2 = f1
                    t1 = b - npass * (b - a)
                    h[i] = t1
                    f1 = compute_c5(h, n)
                else:
                    a = t1
                    t1 = t2
                    f1 = f2
                    t2 = a + npass * (b - a)
                    h[i] = t2
                    f2 = compute_c5(h, n)
            t_opt = (a + b) / 2.0
            h[i] = t_opt
            f_opt = compute_c5(h, n)
            if f_opt < best_c - 1e-12:
                best_c = f_opt
                sweep_improved = True
            # Keep the better of old and new
            h[i] = t_opt
        if not sweep_improved:

    return project_feasible(h, n), best_c

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += 1.0 - h[:n-m]
        else:
            grad[:n-m] += 1.0 - h[m:]
            grad[m:] -= h[:n-m]
    return grad

def outer_lp(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

def outer_approx(h, n, deadline, max_iter=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        h_new = outer_lp(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.002, trust * 0.85)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return _smooth_max(c, alpha)

def _active_obj(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full') * dx
    return _smooth_max(cc[idx], alpha)

def _eq_fun(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_full_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_m, alpha, maxiter=5000, ftol=1e-16):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_active_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += 1.0 - h[:n-m]
        else:
            grad[:n-m] += 1.0 - h[m:]
            grad[m:] -= h[:n-m]
    return grad

def outer_lp(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

def outer_approx(h, n, deadline, max_iter=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        h_new = outer_lp(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.002, trust * 0.85)
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    npass = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] < 0.005 or h[i] > 0.995:
                continue
            old_val = h[i]
            a, b = 0.0, 1.0
            t1 = b - npass * (b - a)
            t2 = a + npass * (b - a)
            h[i] = t1
            f1 = compute_c5(h, n)
            h[i] = t2
            f2 = compute_c5(h, n)
            for _ in range(30):
                if f1 < f2:
                    b = t2; t2 = t1; f2 = f1
                    t1 = b - npass * (b - a)
                    h[i] = t1; f1 = compute_c5(h, n)
                else:
                    a = t1; t1 = t2; f1 = f2
                    t2 = a + npass * (b - a)
                    h[i] = t2; f2 = compute_c5(h, n)
            t_opt = (a + b) / 2.0
            h[i] = t_opt
            f_opt = compute_c5(h, n)
            if f_opt < best_c - 1e-12:
                best_c = f_opt
                sweep_improved = True
            else:
                h[i] = old_val
        if not sweep_improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-10:
                    delta = min(d, (h[i] - h[j]) * 0.5)
                    if delta < 1e-10:
                        continue
                    h[i] -= delta; h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new; improved = True
                    else:
                        h[i] += delta; h[j] -= delta
                elif h[j] > h[i] + 1e-10:
                    delta = min(d, (h[j] - h[i]) * 0.5)
                    if delta < 1e-10:
                        continue
                    h[i] += delta; h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new; improved = True
                    else:
                        h[i] -= delta; h[j] += delta
    return h, best_c
