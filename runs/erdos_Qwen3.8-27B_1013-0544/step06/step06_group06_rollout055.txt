def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(h, n, alpha, top_k=20):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.argsort(corr)[-top_k:]
    ft = corr[top]
    w = np.exp(alpha * (ft - ft.max()))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(len(top)):
        m = top[idx]
        s = n - 1 - m
        wi = w[idx]
        if s >= 0:
            g[:n-s] += wi * (1.0 - h[s:])
            g[s:] -= wi * h[:n-s]
        else:
            sc = -s
            g[sc:] += wi * (1.0 - h[:n-sc])
            g[:n-sc] -= wi * h[sc:]
    return g
