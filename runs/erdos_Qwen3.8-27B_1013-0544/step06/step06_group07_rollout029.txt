```
def resample_to_n(h_in, n_in, n_out):
    if n_in==n_out: return np.asarray(h_in,float).copy()

    x_out=(np.arange(n_out)+0.5)*(2.0/n_out)
    return np.interp(x_out,x_in,np.asarray(h_in,float))
```

```
def build_batch_feats(indices, h, d, n, sign, complement):
    q = indices[:, None] + sign * d[None, :]
    valid = (q >= 0) & (q < n)
    qc = np.clip(q, 0, n - 1)
    vals = h[qc]
    if complement:
        vals = 1 - vals
    return np.where(valid, vals, np.int8(0)).astype(np.int8)
```

```
def eval_swap_batch(h, corr, I, J, n):
    B = I.shape[0]
    if B == 0:
        return np.empty(0, dtype=np.int16)
    d = get_d(n)
    e0 = n - 1
    h8 = h.astype(np.int8, copy=False)
    outI = build_batch_feats(I, h8, d, n, 1, True)
    outJ = build_batch_feats(J, h8, d, n, 1, True)
    inI = build_batch_feats(I, h8, d, n, -1, False)
    inJ = build_batch_feats(J, h8, d, n, -1, False)
    delta = outI.astype(np.int16) - outJ.astype(np.int16) + inJ.astype(np.int16) - inI.astype(np.int16)
    delta[:, e0] -= 2
    rows = np.arange(B, dtype=np.intp)
    idx1 = (e0 + (J - I)).astype(np.intp)
    idx2 = (e0 + (I - J)).astype(np.intp)
    delta[rows, idx1] += 1
    delta[rows, idx2] += 1
    return (corr + delta).max(axis=1).astype(np.int16)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_D_CACHE = {}

def get_d(n):
    d = _D_CACHE.get(n)
    if d is None:
        d = np.arange(-(n - 1), n, dtype=np.int32)
        _D_CACHE[n] = d
    return d

def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    a, b = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (a + b)
        f = float(np.clip(h - m, 0.0, 1.0).sum())
        if f > target:
            a = m
        else:
            b = m
    out = np.clip(h - 0.5 * (a + b), 0.0, 1.0)
    d = float(target - out.sum())
    if d > 0:
        rs = float((1.0 - out).sum())
        if rs > 1e-14:
            out = out + (d / rs) * (1.0 - out)
    elif d < 0:
        rs = float(out.sum())
        if rs > 1e-14:
            out = out + (d / rs) * out
    return np.clip(out, 0.0, 1.0)

def round_to_binary(h, n):
    h = np.asarray(h, dtype=float)
    t = n // 2
    out = np.zeros(n, dtype=np.int8)
    if t > 0:
        out[np.argpartition(-h, t - 1)[:t]] = 1
    return out

def direct_corr(h):
    h = np.asarray(h)
    if np.issubdtype(h.dtype, np.floating):
        return np.correlate(h, 1.0 - h, mode='full')[::-1]
    h = h.astype(np.int16)
    return np.correlate(h, 1 - h, mode='full')[::-1].astype(np.int16)

def weighted_gradient(h, n, weights, order, dx):
    grad = np.zeros(n, dtype=float)
    for k in range(len(order)):
        w = float(weights[k])
        if w <= 1e-9:
            continue
        d = int(order[k]) - (n - 1)
        if d >= 0:
            cnt = n - d
            if cnt > 0:
                grad[:cnt] += w * (1.0 - h[d:])
                grad[d:] += -w * h[:cnt]
        else:
            dd = -d
            cnt = n - dd
            if cnt > 0:
                grad[dd:] += w * (1.0 - h[:cnt])
                grad[:cnt] += -w * h[dd:]
    return grad * dx

def swap_delta_single(h, n, i, j):
    d = get_d(n)
    e0 = n - 1
    delta = np.zeros(2 * n - 1, dtype=np.int16)
    vi = (i + d >= 0) & (i + d < n)
    if np.any(vi):
        delta[vi] += 1 - h[i + d[vi]]
    vj = (j + d >= 0) & (j + d < n)
    if np.any(vj):
        delta[vj] -= 1 - h[j + d[vj]]
    wi = (i - d >= 0) & (i - d < n)
    if np.any(wi):
        delta[wi] -= h[i - d[wi]]
    wj = (j - d >= 0) & (j - d < n)
    if np.any(wj):
        delta[wj] += h[j - d[wj]]
    delta[e0] -= 2
    delta[e0 + j - i] += 1
    delta[e0 + i - j] += 1
    return delta

def build_batch_feats(indices, h, d, n, sign, complement):
    q = indices[:, None] + sign * d[None, :]
    valid = (q >= 0) & (q < n)
    qc = np.clip(q, 0, n - 1)
    vals = h[qc]
    if complement:
        vals = 1 - vals
    return np.where(valid, vals, np.int8(0)).astype(np.int8)

def eval_swap_batch(h, corr, I, J, n):
    B = I.shape[0]
    if B == 0:
        return np.empty(0, dtype=np.int16)
    d = get_d(n)
    e0 = n - 1
    h8 = h.astype(np.int8, copy=False)
    outI = build_batch_feats(I, h8, d, n, 1, True)
    outJ = build_batch_feats(J, h8, d, n, 1, True)
    inI = build_batch_feats(I, h8, d, n, -1, False)
    inJ = build_batch_feats(J, h8, d, n, -1, False)
    delta = (outI.astype(np.int16) - outJ.astype(np.int16) +
             inJ.astype(np.int16) - inI.astype(np.int16))
    delta[:, e0] -= 2
    rows = np.arange(B, dtype=np.intp)
    idx1 = (e0 + (J - I)).astype(np.intp)
    idx2 = (e0 + (I - J)).astype(np.intp)
    delta[rows, idx1] += 1
    delta[rows, idx2] += 1
    return (corr + delta).max(axis=1).astype(np.int16)

def make_bounds(n):
    return [(0.0, 1.0)] * n

def soft_top_obj(x, n, order, alpha, pen):
    dx = 2.0 / n
    R = np.correlate(x, 1.0 - x, mode='full')[::-1] * dx
    rt = R[order]
    m = rt.max()
    z = alpha * (rt - m)
    z = z - z.max()
    lse = z.max() + np.log(np.exp(z).sum())
    s = x.sum() - n / 2.0
    return float(m + lse / alpha + pen * s * s)

def soft_top_grad(x, n, order, alpha, pen):
    dx = 2.0 / n
    R = np.correlate(x, 1.0 - x, mode='full')[::-1] * dx
    rt = R[order]
    m = rt.max()
    z = alpha * (rt - m)
    z = z - z.max()
    p = np.exp(z)
    p /= p.sum()
    g = weighted_gradient(x, n, p, order, dx)
    s = x.sum() - n / 2.0
    return g + 2.0 * pen * s * np.ones_like(x)

        hb, cb = binary_greedy_all(start2, n2, p3, seed + 11)
        if cb < c2:
            start2 = hb
        if cb < best_c - 1e-12:
            best_h = hb
            best_c = cb
            best_n = n2
    p4 = t0 + 0.78 * B
    if time.time() + 30 < p4:
        st = project_feasible(start2, n2 / 2.0)
        hc, cc = continuous_lbfgsb(st, n2, p4, seed + 21)
        if cc < best_c - 1e-12:
            best_h = hc
            best_c = cc
            best_n = n2
    if time.time() + 15 < dl:
        hn = best_n if best_n <= 500 else 500
        st = resample_to_n(best_h, best_n, hn)
        st = project_feasible(st, hn / 2.0)
        hc, cc = continuous_lbfgsb(st, hn, dl, seed + 31)
        if cc < best_c - 1e-12:
            best_h = hc
            best_c = cc
            best_n = hn
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
