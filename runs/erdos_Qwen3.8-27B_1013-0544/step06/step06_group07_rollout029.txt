    # Let me verify with a small example...
    
    # Actually, let me just compute it with a loop over k (only 2n-1 iterations, each O(n)):
    grad = np.zeros(n)
    for ki in range(len(k_vals)):
        k = k_vals[ki]
        pk = p[ki]
        if abs(pk) < 1e-15:
            continue
        # First sum contribution: (1-h[j-k]) for valid j (0<=j-k<n means max(0,k)<=j<=min(n-1,n-1+k))
        j_lo = max(0, k)
        j_hi = min(n-1, n-1+k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += pk * (1 - h[j_lo-k:j_hi+1-k])

        j_lo2 = max(0, -k)
        j_hi2 = min(n-1, n-1-k)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= pk * h[j_lo2+k:j_hi2+1+k]
    grad *= dx
    return grad, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g, corr = softmax_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta_fast(h, n, i, j, corr):
    """Compute new correlation after swapping h[i] and h[j] (assuming h[i]=0, h[j]=1)."""
    # After swap: h[i] becomes 1, h[j] becomes 0
    # delta to corr[k] for each lag k:
    # corr[k] = sum_m h[m+k]*(1-h[m])
    # Change at position i (0->1): 
    #   For lag k: contribution is (1-h[i-k]) if i-k valid [from h[i]=h[(i-k)+k]]
    #              minus h[i+k] if i+k valid [from h[i]=h[m] where m=i, i.e., h[m+k]=h[i+k]]
    # Change at position j (1->0):
    #   For lag k: contribution is -(1-h[j-k]) if j-k valid
    #              plus h[j+k] if j+k valid
    
    # Building the full delta is O(n^2) which is too slow.
    # Instead, just recompute the correlation after the swap.
    # For n=250, correlate is O(n^2) = 62500 ops, which is fast.
    # But in SA we do millions of iterations, so we need the O(n) delta.
    
    # The O(n) delta approach:
    # delta[k] = (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
    #           -(1-h[j-k])*I(0<=j-k<n) + h[j+k]*I(0<=j+k<n)
    # where k ranges from -(n-1) to (n-1)
    
    delta = np.zeros(2*n-1)
    # i goes 0->1:
    for k in range(-(n-1), n):
        ik = i - k
        if 0 <= ik < n:
            delta[k+n-1] += (1 - h[ik])
        ik2 = i + k
        if 0 <= ik2 < n:
            delta[k+n-1] -= h[ik2]
    # j goes 1->0:
    for k in range(-(n-1), n):
        jk = j - k
        if 0 <= jk < n:
            delta[k+n-1] -= (1 - h[jk])
        jk2 = j + k
        if 0 <= jk2 < n:
            delta[k+n-1] += h[jk2]
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_objective(x, n, alpha):
    corr = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def softmax_gradient(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    p /= p.sum()
    k_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki in range(len(k_vals)):
        pk = p[ki]
        if abs(pk) < 1e-14:
            continue
        k = k_vals[ki]
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += pk * (1 - x[lo1 - k:hi1 + 1 - k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= pk * x[lo2 + k:hi2 + 1 + k]
    grad *= dx
    return grad

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:

                       jac=lambda x: softmax_gradient(x, n, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta
