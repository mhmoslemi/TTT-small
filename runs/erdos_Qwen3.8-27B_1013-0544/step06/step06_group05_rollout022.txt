def local_search_fast(h, n, deadline, max_iters=500000, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    best_M = M
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stalls = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stalls > 5000:
            break
        # Pick random i from zeros, j from ones
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        if i == j:
            continue
        delta = compute_delta(i, j, h, n)
        new_c = c + delta
        new_M = new_c.max()
        if new_M < M - 1e-12:
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            c = new_c
            M = new_M
            # Update ones and zeros
            # Remove i from zeros, add to ones
            # Remove j from ones, add to zeros
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            stalls = 0
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            stalls += 1
            # Occasionally accept worse moves (simulated annealing)
            if rng.random() < 0.01 and new_M < M + 2:
                h[i] = 1.0
                h[j] = 0.0
                c = new_c
                M = new_M
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
    
    return best_h, best_M * dx

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def swap_delta(i, j, h, n):
    """Compute change in correlation vector for swap: h[i] 0->1, h[j] 1->0."""
    delta = np.zeros(2*n-1, dtype=np.float64)
    off = n - 1
    # h[i] 0->1, as h[m] term: (1-h[i-lag]) for valid i-lag
    t_lo = max(0, i - off)
    t_hi = min(i, n-1)
    if t_lo <= t_hi:
        t = np.arange(t_lo, t_hi+1)
        delta[i - t + off] += (1.0 - h[t])
    # h[i] 0->1, as (1-h[m-lag]) term: -h[i+lag] for valid i+lag
    t_lo = max(0, i - off)
    t_hi = min(i + off, n-1)
    if t_lo <= t_hi:
        t = np.arange(t_lo, t_hi+1)
        delta[t - i + off] -= h[t]
    # h[j] 1->0, as h[m] term: -(1-h[j-lag]) for valid j-lag
    t_lo = max(0, j - off)
    t_hi = min(j, n-1)
    if t_lo <= t_hi:
        t = np.arange(t_lo, t_hi+1)
        delta[j - t + off] -= (1.0 - h[t])
    # h[j] 1->0, as (1-h[m-lag]) term: +h[j+lag] for valid j+lag
    t_lo = max(0, j - off)
    t_hi = min(j + off, n-1)
    if t_lo <= t_hi:
        t = np.arange(t_lo, t_hi+1)
        delta[t - j + off] += h[t]
    return delta

def binary_search_phase(h, n, deadline, seed=42):
    """Fast binary local search with simulated annealing."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    best_M = M
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    no_improve = 0
    T = 2.0  # annealing temperature
    T_min = 0.1
    cooling = 0.9995
    
    while time.time() < deadline:
        if no_improve > 50000:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        d = swap_delta(i, j, h, n)
        new_M = (c + d).max()
        if new_M <= M + 1e-12:
            h[i] = 1.0; h[j] = 0.0
            c += d
            M = new_M
            zeros.remove(i); ones.append(i)
            ones.remove(j); zeros.append(j)
            no_improve = 0
            if M < best_M:
                best_M = M; best_h = h.copy()
        elif rng.random() < np.exp(-(new_M - M) / T):
            h[i] = 1.0; h[j] = 0.0
            c += d
            M = new_M
            zeros.remove(i); ones.append(i)
            ones.remove(j); zeros.append(j)
            T *= cooling
        else:
            no_improve += 1
            T *= cooling
    return best_h, best_M * dx

def continuous_optimize(h, n, deadline, maxiter=5000):
    """SLSQP continuous optimization with smooth max objective."""
    dx = 2.0 / n
    alpha = 500.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        s = alpha * (corr - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    
    def con(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
