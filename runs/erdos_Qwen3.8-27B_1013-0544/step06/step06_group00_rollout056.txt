    """Project y onto {h: 0<=h<=1, sum(h)=target}."""
    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Adjust for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_smoothed(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    sm = shifted.max()
    val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return val

def c5_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    
    sum1 = np.correlate(1.0 - h, p, mode='full')[2*n-2 : 3*n-2][::-1]
    sum2 = np.correlate(h[::-1], p, mode='full')[2*n-2 : 3*n-2]
    
    grad = (sum1 - sum2) * dx
    return grad

def fista(h0, n, alpha, max_iter, lr=None, deadline=None):
    """FISTA for the smoothed problem with projection."""
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_val = c5_smoothed(h, n, alpha)
    
    # Compute initial gradient norm for step size
    g0 = c5_grad(z, n, alpha)
    grad_norm = np.linalg.norm(g0)
    if grad_norm < 1e-10:
        grad_norm = 1e-10
    if lr is None:
        lr = 1.0 / (grad_norm * 2.0)
        # Actually, for Lipschitz constant, we need L such that ||g(x)-g(y)|| <= L||x-y||
        # For our problem, L ~ O(alpha * n * dx) roughly. Let's use a safe value.
        lr = 1.0 / (1.0 + alpha * n * (2.0/n))  # rough estimate
        # Better: use backtracking
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        g = c5_grad(z, n, alpha)
        
        # Gradient step
        y = z - lr * g
        
        # Project
        h = project_feasible(y, n)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h + ((t - 1.0) / t_new) * (h - z)
        z = project_feasible(z, n)  # Keep z feasible
        t = t_new
        
        # Track best
        val = c5(h, n)  # Use actual C5, not smoothed
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        # Backtracking line search (every 10 iterations)
        if it % 10 == 0 and it > 0:
            # Check descent
            pass
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_smoothed_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def c5_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    sum1 = np.correlate(1.0 - h, p, mode='full')[2 * n - 2:3 * n - 2][::-1]
    sum2 = np.correlate(h[::-1], p, mode='full')[2 * n - 2:3 * n - 2]
    return (sum1 - sum2) * dx

def projected_fista(h0, n, alpha, max_iter, deadline):
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    lr = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    prev_val = c5_smoothed_val(h, n, alpha)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = c5_grad(z, n, alpha)
        # Backtracking line search
        for _ in range(20):
            h_new = project_feasible(z - lr * g, n)
            v_new = c5_smoothed_val(h_new, n, alpha)
            if v_new < prev_val - 1e-12 * abs(prev_val):
                break
            lr *= 0.5
        else:
            h_new = project_feasible(z - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h_new + ((t - 1.0) / t_new) * (h_new - z)
        z = project_feasible(z, n)
        t = t_new
        h = h_new
        prev_val = v_new
        if it % 50 == 0:
            c_actual = compute_c5(h, n)
            if c_actual < best_c:
                best_c = c_actual
                best_h = h.copy()
    c_actual = compute_c5(h, n)
    if c_actual < best_c:
        best_c = c_actual
        best_h = h.copy()
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def fourier_perturb(h, n, seed, amplitude=0.05):
    rng = np.random.RandomState(seed)
    freqs = np.arange(1, n // 2 + 1)
    coeffs = rng.randn(n // 2) * amplitude
    phases = rng.uniform(0, 2 * np.pi, n // 2)
    x = np.arange(n)
    pert = np.zeros(n)
    for f, c, p in zip(freqs, coeffs, phases):
        pert += c * np.cos(2 * np.pi * f * x / n + p)
    return project_feasible(h + pert, n)
