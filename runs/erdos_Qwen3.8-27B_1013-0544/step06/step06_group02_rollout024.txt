def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_weighted(h, n, weights):
    dx = 2.0 / n
    m = 2 * n - 1
    grad = np.zeros(n)
    for k_shift in range(m):
        w = weights[k_shift]
        if w < 1e-13:
            continue
        k = k_shift - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _get_weights_from_h(h, n, alpha):
    c = compute_c5_vec(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    sh_mx = sh.max()
    w = np.exp(sh - sh_mx)
    w /= w.sum()
    return w

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    mx = c.max()
    sh = alpha * (c - mx)
    sh_mx = sh.max()
    val = mx + (sh_mx + np.log(np.sum(np.exp(sh - sh_mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    w = _get_weights_from_h(x, n, alpha)
    g = grad_weighted(x, n, w)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    sh = alpha * (c - mx)
    sh_mx = sh.max()

def _slsqp_grad(x, n, alpha):

    return x.sum() - n / 2.0

    if alphas is None:

    bounds = [(0.0, 1.0)] * n

            break
        mu = 0.01
        for attempt in range(4):
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(_pen_obj, h, jac=_pen_grad,
                             args=(n, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 150, 'ftol': 1e-15,
                                      'gtol': 1e-12, 'maxfun': 300})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                break
            except Exception:
                mu *= 10.0
    return h, best_c

def slsqp_phase(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50000, 500000, 5000000, 50000000]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 4:
            break
        try:
            res = minimize(_slq_obj, h, jac=_slq_grad,
                         args=(n, alpha), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fictitious_play(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h_curr = h.copy()
    tau = 0.05
    lr = 0.1 / n
    while time.time() < deadline - 2:
        c = compute_c5_vec(h_curr, n)
        w = np.exp(c / tau)
        w /= w.sum()
        lr_local = lr
        for _ in range(25):
            if time.time() > deadline - 2:
                break
            g = grad_weighted(h_curr, n, w)
            h_new = np.clip(h_curr - lr_local * g, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5_vec(h_new, n)
            if np.dot(w, c_new) <= np.dot(w, c) + 1e-13:
                h_curr = h_new
            else:
                lr_local *= 0.5
                if lr_local < 1e-20:
                    break
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        tau *= 0.96
        if tau < 0.004:
            tau = 0.004
        lr = max(lr * 0.99, 0.01 / n)
    return best_h, best_c

def subgrad_method(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h_curr = h.copy()
    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.9
    while time.time() < deadline - 1:
        c = compute_c5_vec(h_curr, n)
        k_star = np.argmax(c)
        k = k_star - (n - 1)
        g = np.zeros(n)
        dx = 2.0 / n
        if k >= 0:
            nv = n - k
            g[:nv] += dx * (1.0 - h_curr[k:])
            g[k:] += dx * (-h_curr[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += dx * (1.0 - h_curr[:nv])
            g[:nv] += dx * (-h_curr[kk:])
        momentum = beta * momentum + (1 - beta) * g
        h_new = np.clip(h_curr - lr * momentum, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        lr *= 0.9995
        if lr < 1e-10:
            break
    return best_h, best_c

def pair_search(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.005, 0.001, 0.0005, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def resample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    h = project_feasible(h_init, n_init)
    best_c = compute_c5(h, n_init)
    best_h = h.copy()
    n_best = n_init

    # Phase 1: L-BFGS-B with penalty (200s)
    t1 = min(t0 + 180, deadline - 500)
    if t1 > t0 + 30:
        h, c = lbfgs_phase(h, n_init, t1)
        if c < best_c:
            best_c, best_h, n_best = c, h.copy(), n_init
        else:
            h = best_h.copy()

    # Phase 2: SLSQP high-alpha (180s)
    t2 = min(time.time() + 180, deadline - 400)
    if time.time() < deadline - 400:
        h2, c2 = slsqp_phase(h, n_init, t2)
        if c2 < best_c:
            best_c, best_h, n_best = c2, h2.copy(), n_init
        h = best_h.copy() if c2 < best_c else h

    # Phase 3: Fictitious play (120s)
    t3 = min(time.time() + 120, deadline - 250)
    if time.time() < deadline - 250:
        h3, c3 = fictitious_play(best_h, n_best, t3)
        if c3 < best_c:
            best_c, best_h, n_best = c3, h3.copy(), n_best

    # Phase 4: Subgradient with momentum (80s)
    t4 = min(time.time() + 80, deadline - 180)
    if time.time() < deadline - 180:
        h4, c4 = subgrad_method(best_h, n_best, t4)
        if c4 < best_c:
            best_c, best_h, n_best = c4, h4.copy(), n_best

    # Phase 5: Multi-resolution (250s)
    t5_end = deadline - 80
    n_tries = [200, 250, 350, 400, 500, 450]
    for n_try in n_tries:
        if time.time() > t5_end - 15:
            break
        if n_try == n_best:
            continue
        h_interp = resample(best_h, n_best, n_try)
        h_interp = project_feasible(h_interp, n_try)
        alloc_lbfgs = min(25, t5_end - time.time() - 20)
        if alloc_lbfgs < 10:
            break
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc_lbfgs,
                               alphas=[10000, 100000, 1000000])
        if time.time() < t5_end - 10:
            h_t, c_t = slsqp_phase(h_t, n_try, t5_end - 5,
                                   alphas=[500000, 5000000])
        if c_t < best_c:
            best_c, best_h, n_best = c_t, h_t.copy(), n_try

    # Phase 6: Final SLSQP on best (50s)
    if time.time() < deadline - 15:
        h_f, c_f = slsqp_phase(best_h, n_best, min(time.time() + 50, deadline - 8),
                               alphas=[5000000, 50000000])
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    # Phase 7: Pair search (remaining)
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n_best, deadline - 1, radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
