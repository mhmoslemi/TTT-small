            if s < 1e-14:

def project_feasible(h, n):

            if s < 1e-14:

def grad_smooth_max(h, ov, T, dx):
    """Gradient of (1/T)log(sum exp(T*ov)) with respect to h."""
    n = len(h)

    w = np.exp(shifted)

    grad = np.zeros(n)
