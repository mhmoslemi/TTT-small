    m = len(h)
    
    # Adam state

    v_hat = np.zeros(m)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_h = h.copy()
    best_c = compute_c5(h, n)

            break

        # Log-sum-exp
        max_c = c.max()
        exp_c = np.exp(alpha * (c - max_c))
        logsum = max_c + np.log(exp_c.sum()) / alpha
        obj = logsum + max_c  # Actually just logsum

        p = exp_c / exp_c.sum()
        
        # Gradient
        grad = np.zeros(m)
        m0 = n - 1
        for k in range(2 * n - 1):
            j = k - m0  # shift
            if p[k] < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def fft_gradient(c, p, h, n, dx):
    """Compute gradient of log-sum-exp objective using FFT."""
    b = 1.0 - h
    N = n
    Lp = 2 * N - 1  # length of p (correlation)
    size = 1
    while size < N + Lp - 1:
        size *= 2
    # A[i] = (b * p)[i+N-1]
    B_fft = np.fft.rfft(b, size)
    P_fft = np.fft.rfft(p, size)
    conv1 = np.fft.irfft(B_fft * P_fft, size)
    A = conv1[N-1 : N-1+N]
    # B[i] = (h * pff)[N-1+i] where pff[j] = p[2N-2-j]
    pff = p[::-1]
    H_fft = np.fft.rfft(h, size)
    PFF_fft = np.fft.rfft(pff, size)
    conv2 = np.fft.irfft(H_fft * PFF_fft, size)
    B = conv2[N-1 : N-1+N]
    return dx * (A - B)

def adam_optimize(h, n, deadline, alpha_init=100, alpha_max=10000000, lr=0.5, n_ep=100, seed=42):
    """Adam optimization with adaptive alpha."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    N = n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alpha = alpha_init
    for ep in range(n_ep):
        if time.time() > deadline:
            break
        # Adam state
        m_hat = np.zeros(N)
        v_hat = np.zeros(N)
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        t_adam = 0
        
        # Inner iterations per epoch
        n_inner = 2000
        for it in range(n_inner):
            if time.time() > deadline:
                break
            t_adam += 1
            
            # Compute correlation and softmax
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            max_c = c.max()
            shifted = alpha * (c - max_c)
            exp_c = np.exp(shifted)
            p = exp_c / exp_c.sum()
            
            # Gradient via FFT
            grad = fft_gradient(c, p, h, N, dx)
            
            # Adam update
            m_hat = beta1 * m_hat + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * grad**2
            m_hat_b = m_hat / (1 - beta1**t_adam)
            v_hat_b = v_hat / (1 - beta2**t_adam)
            h = h - lr * m_hat_b / (np.sqrt(v_hat_b) + eps)
            h = project_feasible(h, n)
            
        # Evaluate and update best
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Increase alpha
        alpha = min(alpha * 2, alpha_max)
    
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    """Greedy binary swap optimization."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Random order for diversity
        rng_order = np.random.permutation(len(zeros_list))
        for idx in rng_order:
            i = zeros_list[idx]
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def fft_corr(h, b, n):
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(np.conj(H) * B, size)[:2 * n - 1]

def fft_gradient(p, h, n):
    dx = 2.0 / n
    b = 1.0 - h
    N = n
    Lp = 2 * N - 1
    size = 1
    while size < N + Lp - 1:
        size *= 2
    Bf = np.fft.rfft(b, size)
    Pf = np.fft.rfft(p, size)
    conv1 = np.fft.irfft(Bf * Pf, size)
    A = conv1[N - 1:N - 1 + N]
    pff = p[::-1]
    Hf = np.fft.rfft(h, size)
    Pff = np.fft.rfft(pff, size)
    conv2 = np.fft.irfft(Hf * Pff, size)
    B = conv2[N - 1:N - 1 + N]
    return dx * (A - B)

def adam_optimize(h_init, n, deadline, alpha_init=100, alpha_max=50000000, lr=0.05, n_ep=30, seed=42, inner_iter=3000):
    rng = np.random.RandomState(seed)
    N = n
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    stall = 0
    for ep in range(n_ep):
        if time.time() > deadline:
            break
        m_hat = np.zeros(N)
        v_hat = np.zeros(N)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t_adam = 0
        ep_best = best_c
        for it in range(inner_iter):
            if time.time() > deadline:
                break
            t_adam += 1
            c = fft_corr(h, 1.0 - h, N) * (2.0 / N)
            max_c = c.max()
            shifted = alpha * (c - max_c)
            shifted = np.clip(shifted, -500, 500)
            exp_c = np.exp(shifted)
            p = exp_c / exp_c.sum()
            grad = fft_gradient(p, h, N)
            m_hat = beta1 * m_hat + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)
            mb = m_hat / (1.0 - beta1 ** t_adam)
            vb = v_hat / (1.0 - beta2 ** t_adam)
            h_new = h - lr * mb / (np.sqrt(vb) + eps)
            h_new = project_feasible(h_new, n)
            h = h_new
            if it % 500 == 499:
                cv = compute_c5(h, n)
                if cv < ep_best:
                    ep_best = cv
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 4:
                h = best_h + 0.01 * rng.randn(N)
                h = project_feasible(h, n)
                stall = 0
        alpha = min(alpha * 3.0, alpha_max + 1.0)
        lr = max(lr * 0.85, 0.005)
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=300, seed=42):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(len(zeros_list))
        for idx in order:
            i = zeros_list[idx]
            for j in ones_list:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                jm = j + m_range
                msk2 = (jm >= 0) & (jm < n)
                imn = i - m_range
                msk3 = (imn >= 0) & (imn < n)
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                d = 0.0
                d += (1.0 - h[im[msk]]).sum()
                d -= h[imn[msk3]].sum()
                d -= (1.0 - h[jm[msk2]]).sum()
                d += h[jmn[msk4]].sum()
                d -= 2.0
                k1 = n - 1 + (j - i)
                k2 = n - 1 + (i - j)
                if k1 >= 0 and k1 < 2 * n - 1:
                    d += 1.0
                if k2 >= 0 and k2 < 2 * n - 1:
                    d += 1.0
                nc_rough = (max(corr) + max(0, d)) * dx
                if nc_rough >= cur_c + 0.01 * dx:
                    continue
                delta = np.zeros(2 * n - 1)
                delta[msk] += 1.0 - h[im[msk]]
                delta[msk3] -= h[imn[msk3]]
                delta[msk2] -= 1.0 - h[jm[msk2]]
                delta[msk4] += h[jmn[msk4]]
                delta[n - 1] -= 2.0
                if k1 >= 0 and k1 < 2 * n - 1:
                    delta[k1] += 1.0
                if k2 >= 0 and k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
