def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0:
        return x
    lo, hi = float(np.min(x)) - 1.0, float(np.max(x)) + 1.0
    for _ in range(70):
        mid = 0.5*(lo+hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - p.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            cap = 1.0 - p
        else:
            cap = p
        tot = cap.sum()
        if tot <= 1e-15:
            break
        p = np.clip(p + diff/tot*cap, 0.0, 1.0)
    p = np.clip(p, 0.0, 1.0)
    r = target - p.sum()
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.where(p < 1.0)[0]
        else:
            idx = np.where(p > 0.0)[0]
        if idx.size:
            i = idx[0]
            p[i] = float(np.clip(p[i] + r, 0.0, 1.0))
            # if clip failed, distribute residual through remaining
            r2 = target - p.sum()
            if abs(r2) > 1e-14 and idx.size > 1:
                j = idx[1]

def set_one_matrix(x):

    A = np.zeros((n,L), dtype=float)

def corr_w(x, alpha, k=None):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if alpha <= 1e-12:
       return m,c,np.ones_like(c)/len(c)
    d = alpha*(c - m)
    e = np.exp(d)
    s = float(e.sum())
    w = e / s
    if k is not None and k < c.size:
       order = np.argsort(w)[::-1][:k]
       wt = np.zeros_like(w)
       wt[order] = w[order]
       ss = wt.sum()
       if ss > 0:
          w = wt / ss
    return m + np.log(s)/alpha, c, w

def exact_c(h):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))


def exact_counts_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def binarize_top(x, target):
    n = x.size
    k = int(round(float(target)))
    order = np.argsort(x)[::-1]
    b = np.zeros(n, dtype=float)
    b[order[:k]] = 1.0
    return b


def make_binary_target(x, target):
    k = int(round(float(target)))
    b = (x >= 0.5).astype(float)
    s = int(b.sum())
    if s == k:
        return b
    if s > k:
        ones = np.where(b == 1.0)[0]
        b[ones[-(s - k):]] = 0.0
    else:
        zeros = np.where(b == 0.0)[0]
        b[zeros[:k - s]] = 1.0
    return b


def make_pattern(code, n, target):
    arr = np.empty(0, dtype=float)
    for ch in code:
        v = 1.0 if ch == '1' else 0.0
        arr = np.append(arr, v)
    b = np.tile(arr, n // max(1, arr.size) + 1)[:n]
    return make_binary_target(b, target)


def set_one_matrix(x):
    n = x.size
    L = 2 * n - 1
    A = np.zeros((n, L), dtype=float)
    for q in range(1, n):
        p1 = n - 1 - q
        p2 = n - 1 + q
        m = n - q
        rv = x[q:]
        A[:m, p1] += 1.0 - rv
        A[:m, p2] -= rv
        lv = x[:m]
        A[q:, p1] += -lv
        A[q:, p2] += 1.0 - lv
    return A

        else:
            no_impr += 1
        Z = np.where(x < 0.5)[0]
        O = np.where(x > 0.5)[0]
        if Z.size == 0 or O.size == 0:
            break
        temp = temp0 * np.exp(-sweep / temp_decay)
        if no_impr > 900 and temp < 1e-2:
            break
        if no_impr > 70 and temp > 2e-3:
            for _ in range(rng.randint(3, 9)):
                Zi = np.where(x < 0.5)[0]
                Oi = np.where(x > 0.5)[0]
                if Zi.size and Oi.size:
                    x[rng.choice(Zi)] = 1.0
                    x[rng.choice(Oi)] = 0.0
            no_impr = 0
            continue
        A = set_one_matrix(x)
        addZ = A[Z]
        remO = -A[O]
        best_val = cm + 1000.0
        ba = bb = 0
        for a in range(Z.size):
            i = Z[a]
            base = c + addZ[a]
            vals = remO + base
            darr = np.abs(O - i)
            idx1 = n - 1 - darr
            idx2 = n - 1 + darr
            rowmax = vals.max(axis=1)
            m1 = np.diag(vals[:, idx1]) + 1.0
            m2 = np.diag(vals[:, idx2]) + 1.0
            nm = np.maximum(rowmax, m1)
            nm = np.maximum(nm, m2)
            li = int(np.argmin(nm))
            lv = float(nm[li])
            if lv < best_val - 1e-12:
                best_val = lv
                ba = a
                bb = li
            elif abs(lv - best_val) <= 1e-12 and rng.random() < 0.3:
                ba = a
                bb = li
        if best_val > cm + 1e-9:
            if temp < 1e-5:
                break
            if rng.random() < np.exp(-(best_val - cm) / max(temp, 1e-12)):
                x[Z[ba]] = 1.0
                x[O[bb]] = 0.0
            else:
                if no_impr > 50:
                    for _ in range(rng.randint(3, 9)):
                        Zi = np.where(x < 0.5)[0]
                        Oi = np.where(x > 0.5)[0]
                        if Zi.size and Oi.size:
                            x[rng.choice(Zi)] = 1.0
                            x[rng.choice(Oi)] = 0.0
                    no_impr = 0
                    continue
                break
        else:
            x[Z[ba]] = 1.0
            x[O[bb]] = 0.0
    best_cm = min(best_cm, exact_counts_max(best_x))
    return best_x, float(best_cm * 2.0 / n)


def corr_w(x, alpha, k=None):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    if alpha <= 1e-12:
        w = np.ones_like(c) / c.size
        return m, c, w
    d = alpha * (c - m)
    e = np.exp(d)
    s = float(e.sum())
    w = e / s
    if k is not None and k < c.size:
        order = np.argsort(w)[::-1][:k]
        wt = np.zeros_like(w)
        wt[order] = w[order]
        ss = wt.sum()
        if ss > 0:
            w = wt / ss
    val = m + np.log(s) / alpha
    return val, c, w


def soft_grad_w(x, w):
    n = x.size
    g = np.zeros(n, dtype=float)
    w0 = w[n - 1]
    if abs(w0) > 1e-300:
        g += w0 * (1.0 - 2.0 * x)
    wp = w[:n - 1][::-1]
    wn = w[n:]
    for t in range(1, n):
        a = wp[t - 1]
        b = wn[t - 1]
        s = a + b
        if s <= 1e-18:
            continue
        if a:
            g[:n - t] += a
        if b:
            g[t:] += b
        if s:
            g[:n - t] -= s * x[t:]
            g[t:] -= s * x[:n - t]
    return g


def smooth_fun(x):
    val, c, w = corr_w(x, _g_alpha)
    s = x.sum() - _g_target
    return val + 0.5 * _g_mu * s * s


def smooth_jac(x):
    val, c, w = corr_w(x, _g_alpha)
    g = soft_grad_w(x, w)
    s = x.sum() - _g_target
    return g + _g_mu * s


def smooth_callback(x):
    global _g_last_x
    _g_last_x = np.array(x, copy=True)
    if time.time() >= _g_deadline:
        raise Exception('timeout')


def lbfgs_smooth(x, alpha, mu, deadline):
    global _g_alpha, _g_mu, _g_target, _g_deadline, _g_last_x
    n = x.size
    target = 0.5 * n
    x = project_box_sum(x, target)
    _g_alpha = float(alpha)
    _g_mu = float(mu)
    _g_target = float(target)
    _g_deadline = float(min(deadline, time.time() + 3600.0))
    _g_last_x = x.copy()
    bounds = [(0.0, 1.0)] * n
    options = {'maxiter': 20000, 'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 40}
    cand = []
    try:
        res = minimize(smooth_fun, x, method='L-BFGS-B', jac=smooth_jac, bounds=bounds,
                       callback=smooth_callback, options=options)
        if getattr(res, 'x', None) is not None:
            cand.append(res.x)
    except Exception:
        pass
    if _g_last_x is not None:
        cand.append(_g_last_x)
    best_y = x
    best_val = exact_c(x)
    for cy in cand:
        if cy is None:
            continue
        py = project_box_sum(cy, target)
        val = exact_c(py)
        if val < best_val:
            best_val = val
            best_y = py
    return best_y


def projected_smooth(x, deadline, target, alphas=None):
    if alphas is None:
        alphas = [4, 12, 40, 150, 500]
    x = project_box_sum(x, target)
    best_x = x.copy()
    best_c = exact_c(x)
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        x = project_box_sum(x, target)
        step = 0.4
        val, c, w = corr_w(x, alpha, 40)
        for it in range(5000):
            if time.time() > deadline - 2:
                break
            g = soft_grad_w(x, w)
            d = g - np.mean(g)
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-12:
                break
            step = min(step, 1.0 / (nrm + 1e-12))
            accepted = False
            for _ in range(50):
                y = project_box_sum(x - step * d, target)
                if np.max(np.abs(y - x)) < 1e-14:
                    break
                valy, cy, wy = corr_w(y, alpha, 40)
                if valy < val - 1e-14:
                    x = y
                    val = valy
                    c, w = cy, wy
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                break
            if it % 25 == 0:
                cc = exact_c(x)
                if cc < best_c:
                    best_c = cc
                    best_x = x.copy()
            step = min(step * 1.2, 2.0 / (nrm + 1e-12))
    cc = exact_c(project_box_sum(x, target))
    if cc < best_c:
        best_c = cc
        best_x = x.copy()
    return best_x, best_c


def continuous_lbfgs(x, deadline, seed=0):
    n = x.size
    target = 0.5 * n
    x = project_box_sum(x, target)
    best_x = x.copy()
    best_c = exact_c(x)
    if time.time() < deadline - 35:
        try:
            y, yc = projected_smooth(x, min(deadline, time.time() + 30), target)
            if yc < best_c:
                best_c = yc
                best_x = y.copy()
        except Exception:
            pass
    combos = [(10, 50), (40, 80), (150, 100), (600, 150), (2500, 200)]
    for alpha, mu in combos:
        if time.time() > deadline - 3:
            break
        per = min(18.0, max(6.0, 0.12 * (deadline - time.time())))
        try:
            y = lbfgs_smooth(best_x, alpha, mu, time.time() + per)
            c = exact_c(y)
            if c < best_c - 1e-14:
                best_c = c
                best_x = y.copy()
        except Exception:
            pass
    if time.time() < deadline - 8:
        try:
            y, yc = projected_smooth(best_x, deadline, target)
            if yc < best_c:
                best_c = yc
                best_x = y.copy()
        except Exception:
            pass
    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + float(budget_s) - 5.0
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = int(h0.size)
    except Exception:
        n = 250
        h0 = make_pattern('1100', n, 0.5 * n)
    if n < 2:
        n = 2
        h0 = np.empty(2, dtype=float)
        h0[0] = 1.0
        h0[1] = 0.0
    if n % 2 != 0:
        n -= 1
        h0 = h0[:n]
    target = 0.5 * n
    use_bin = int(round(target)) == int(target)
    best_h = project_box_sum(h0, target)
    best_c = exact_c(best_h)

    if use_bin and time.time() < hard_dl:
        binary_end = t0 + 0.34 * float(budget_s)
        cands = []
        cands.append(binarize_top(best_h, target))
        cands.append(make_pattern('1100', n, target))
        if budget_s > 450:
            cands.append(make_pattern('111000', n, target))
        if budget_s > 750:
            cands.append(make_pattern('11110000', n, target))
        for idx, cand in enumerate(cands):
            if time.time() >= binary_end:
                break
            dl = min(binary_end, time.time() + 70.0 + 15.0 * idx)
            try:
                xb, cb = binary_local_sa(cand, dl, seed=seed + 7 + idx)
                if cb < best_c:
                    best_c = cb
                    best_h = xb.copy()
            except Exception:
                pass
        if time.time() < binary_end - 5:
            try:
                xb, cb = binary_local_sa(binarize_top(best_h, target), binary_end,
                                         seed=seed + 101)
                if cb < best_c:
                    best_c = cb
                    best_h = xb.copy()
            except Exception:
                pass

    cont_end = t0 + 0.87 * float(budget_s)
    starts = []
    if use_bin:
        starts.append(binarize_top(best_h, target))
    starts.append(best_h.copy())
    rng = np.random.RandomState(seed + 999)
    starts.append(project_box_sum(best_h + 0.035 * rng.randn(n), target))
    if use_bin:
        b = binarize_top(best_h, target)
        r2 = np.random.RandomState(seed + 123)
        for _ in range(8):
            Zi = np.where(b < 0.5)[0]
            Oi = np.where(b > 0.5)[0]
            if Zi.size and Oi.size:
                b[r2.choice(Zi)] = 1.0
                b[r2.choice(Oi)] = 0.0
        starts.append(b)
    for idx, st in enumerate(starts):
        if time.time() >= cont_end:
            break
        dl = min(cont_end, time.time() + 110.0)
        try:
            y, yc = continuous_lbfgs(project_box_sum(st, target), dl, seed=seed + 90 + idx)
            y = project_box_sum(y, target)
            yc = exact_c(y)
            if yc < best_c:
                best_c = yc
                best_h = y.copy()
        except Exception:
            pass

    if time.time() < hard_dl - 8:
        try:
            b = binarize_top(best_h, target) if use_bin else best_h
            y, yc = continuous_lbfgs(project_box_sum(b, target), hard_dl - 3.0,
                                     seed=seed + 2000)
            y = project_box_sum(y, target)
            yc = exact_c(y)
            if yc < best_c:
                best_c = yc
                best_h = y.copy()
        except Exception:
            pass

    if use_bin and time.time() < hard_dl - 3:
        try:
            b = binarize_top(best_h, target)
            xb, cb = binary_local_sa(b, hard_dl - 1.5, seed=seed + 3000)
            if cb < best_c:
                best_c = cb
                best_h = xb.copy()
        except Exception:
            pass

    best_h = project_box_sum(best_h, target)
    best_c = exact_c(best_h)
    return best_h, float(best_c), int(n)
