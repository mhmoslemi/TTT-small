def soft_value_grad(h, alpha, mu):

```python
import numpy as np
import time
import math
from scipy.optimize import minimize
try:
    import cvxpy as cp
except Exception:
    cp = None

N_G = 0
DX_G = 1.0
TARGET_G = 0.0
LAG_COLS_G = []
EQ_JAC_G = None
EQ_JAC_N_G = None
OBJ_JAC_G = None
ACTIVE_IDX_G = np.array([], dtype=int)
ALPHA_LBFGS_G = 100.0
MU_LBFGS_G = 100.0
ALPHA_SOFT_G = 1000.0


def set_size(n):
    global N_G, DX_G, TARGET_G, LAG_COLS_G, EQ_JAC_G, EQ_JAC_N_G, OBJ_JAC_G
    n = int(n)
    N_G = n
    DX_G = 2.0 / n
    TARGET_G = n / 2.0
    LAG_COLS_G = []
    for idx in range(2 * n - 1):
        s = n - 1 - idx
        LAG_COLS_G.append(np.arange(max(0, -s), min(n, n - s)))
    EQ_JAC_G = np.ones((1, n + 1))
    EQ_JAC_G[0, n] = 0.0
    EQ_JAC_N_G = np.ones((1, n))
    OBJ_JAC_G = np.zeros(n + 1)
    OBJ_JAC_G[n] = 1.0


def project_feasible(v, n):
    v = np.asarray(v, dtype=float).copy()
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if float(np.clip(v - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-15:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                step = min(d, 1.0 - h[i])
            else:
                step = max(d, -h[i])
            h[i] += step
            d = target - float(h.sum())
            if abs(d) < 1e-15:
                break
    h = np.clip(h, 0.0, 1.0)
    if abs(target - float(h.sum())) > 1e-13:
        i = int(np.argmax(np.minimum(h, 1.0 - h)))
        val = target - float(h.sum() - h[i])
        h[i] = min(1.0, max(0.0, float(val)))
    return h


def c_values(h):
    return np.correlate(h, 1.0 - h, mode="full") * DX_G


def c5(h):
    return float(np.max(c_values(h)))


def lag_value(h, idx):
    s = N_G - 1 - idx
    cols = LAG_COLS_G[idx]
    return DX_G * float(np.dot(h[cols], 1.0 - h[cols + s]))


def lag_grad(h, idx):
    n = N_G
    s = n - 1 - idx
    g = np.zeros(n)
    if s == 0:
        return DX_G * (1.0 - 2.0 * h)
    cols = LAG_COLS_G[idx]
    m = len(cols)
    g[cols] = 1.0
    if s > 0:
        g[s:s + m] -= h[cols]
    else:
        g[:m] -= h[cols]
    return DX_G * g


def active_fun(y):
    h = y[:N_G]
    t = y[N_G]
    idxs = ACTIVE_IDX_G
    out = np.empty(len(idxs))
    for r, i in enumerate(idxs):
        out[r] = lag_value(h, i) - t
    return out


def active_jac(y):
    h = y[:N_G]
    idxs = ACTIVE_IDX_G
    m = len(idxs)
    J = np.empty((m, N_G + 1))
    for r, i in enumerate(idxs):
        J[r, :N_G] = lag_grad(h, i)
        J[r, N_G] = -1.0
    return J


def eq_fun(y):
    return float(np.sum(y[:N_G]) - TARGET_G)


def eq_jac(y):
    return EQ_JAC_G


def eq_n_fun(x):
    return float(np.sum(x[:N_G]) - TARGET_G)


def eq_n_jac(x):
    return EQ_JAC_N_G


def active_obj(y):
    return float(y[N_G])


def active_obj_jac(y):
    return OBJ_JAC_G


def active_slsqp_once(h, cmax, active_idx, maxiter):
    global ACTIVE_IDX_G
    n = N_G
    ACTIVE_IDX_G = np.asarray(active_idx, dtype=int)
    if len(ACTIVE_IDX_G) == 0:
        return h
    y0 = np.concatenate((h, np.array([cmax + 1e-7], dtype=float)))
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {"type": "eq", "fun": eq_fun, "jac": eq_jac},
        {"type": "ineq", "fun": active_fun, "jac": active_jac},
    ]
    try:
        res = minimize(
            active_obj,
            y0,
            jac=active_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11},
        )
    except Exception:
        return None
    if res.x is None:
        return None
    return project_feasible(res.x[:n], n)


def active_polish(h, best_c, deadline):
    n = N_G
    h = project_feasible(h, n)
    cur = c5(h)
    if cur < best_c:
        best_c = cur
    best_h = h.copy()
    for _ in range(15):
        if time.time() > deadline:
            break
        cv = c_values(h)
        cm = float(np.max(cv))
        if cm < best_c:
            best_c = cm
            best_h = h.copy()
        near = int(np.sum(cv >= cm - 1e-5))
        arg = np.argsort(cv)
        caps = (10, 24) if near <= 15 else (10, 24, 60)
        improved = False
        for cap in caps:
            if time.time() > deadline - 5:
                break
            size = min(80, max(cap, near if near <= 80 else cap))
            idx = arg[-size:]
            mi = 80 if size <= 15 else (60 if size <= 35 else 45)
            h2 = active_slsqp_once(h, cm, idx, mi)
            if h2 is not None:
                c2 = c5(h2)
                if c2 < best_c - 1e-14 or c2 < cm - 1e-13:
                    h = h2
                    if c2 < best_c:
                        best_c = c2
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c


def soft_value_grad(h, alpha, mu):
    n = N_G
    cv = c_values(h)
    m = float(np.max(cv))
    u = alpha * (cv - m)
    u -= u.max()
    w = np.exp(u)
    ws = float(w.sum())
    w /= ws
    V = w[::-1]
    W0 = float(V[n - 1])
    Vn = V.copy()
    Vn[n - 1] = 0.0
    pref = np.concatenate(([0.0], np.cumsum(Vn)))
    i = np.arange(n)
    A = pref[2 * n - 1 - i] - pref[n - 1 - i]
    C = np.convolve(h, Vn, mode="full")[n - 1:n - 1 + n]
    g = DX_G * (A - C + W0 * (1.0 - 2.0 * h))
    s = float(np.sum(h) - TARGET_G)
    g += mu * s
    val = m + math.log(ws) / alpha + 0.5 * mu * s * s
    return val, g


def lbfgs_obj(x):
    return soft_value_grad(x, ALPHA_LBFGS_G, MU_LBFGS_G)[0]


def lbfgs_grad(x):
    return soft_value_grad(x, ALPHA_LBFGS_G, MU_LBFGS_G)[1]


def lbfgs_polish(h, best_c, deadline):
    global ALPHA_LBFGS_G, MU_LBFGS_G
    n = N_G
    h = project_feasible(h, n)
    best = min(best_c, c5(h))
    bounds = [(0.0, 1.0)] * n
    for alpha in (25.0, 100.0, 400.0, 1500.0):
        if time.time() > deadline:
            break
        for mu in (10.0, 100.0):
            if time.time() > deadline - 5:
                break
            ALPHA_LBFGS_G = alpha
            MU_LBFGS_G = mu
            try:
                res = minimize(
                    lbfgs_obj,
                    h,
                    jac=lbfgs_grad,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": 350, "ftol": 1e-11, "gtol": 1e-7},
                )
            except Exception:
                continue
            h2 = project_feasible(res.x, n)
            c2 = c5(h2)
            if c2 < best:
                best = c2
                h = h2
    return h, best


def slsqp_soft_obj(x):
    return soft_value_grad(x, ALPHA_SOFT_G, 0.0)[0]


def slsqp_soft_grad(x):
    return soft_value_grad(x, ALPHA_SOFT_G, 0.0)[1]


def slsqp_soft(h, alpha, maxiter):
    global ALPHA_SOFT_G
    n = N_G
    h = project_feasible(h, n)
    ALPHA_SOFT_G = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_n_fun, "jac": eq_n_jac}]
    try:
        res = minimize(
            slsqp_soft_obj,
            h,
            jac=slsqp_soft_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11},
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def lp_linear(h0, idxs, radius):
    if cp is None:
        return None
    try:
        x = cp.Variable(N_G)
        t = cp.Variable()
        cons = [x >= 0, x <= 1, cp.sum(x) == TARGET_G, t >= 0]
        if radius is not None:
            cons.append(cp.norm1(x - h0) <= float(radius))
        for i in idxs:
            val = lag_value(h0, i)
            g = lag_grad(h0, i)
            cons.append(cp.dot(g, x) <= t + float(np.dot(g, h0)) - val)
        prob = cp.Problem(cp.Minimize(t), cons)
        solved = False
        for name in ("PDLP", "GLOP", "ECOS", "CVCXOPT"):
            solver = getattr(cp, name, None)
            if solver is None:
                continue
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(verbose=False)
            except Exception:
                pass
        if str(prob.status).startswith("optimal") and x.value is not None:
            return project_feasible(np.asarray(x.value), N_G)
    except Exception:
        return None
    return None


def perturb_feasible(h, scale, rng):
    return project_feasible(h + rng.normal(0.0, scale, h.size), N_G)


def topk_binary(h, rng):
    n = N_G
    k = n // 2
    v = h + rng.uniform(0.0, 1e-3, n)
    order = np.argsort(v)[::-1]
    b = np.zeros(n)
    b[order[:k]] = 1.0
    return project_feasible(b, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    usable = max(20.0, float(budget_s) - 15.0)
    deadline = t0 + usable

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        if n0 < 2:
            raise ValueError
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    set_size(n0)
    best_h = project_feasible(h0, N_G)
    best_c = c5(best_h)
    rng = np.random.RandomState(seed)

    d = t0 + 0.22 * usable
    if time.time() < d:
        h1, c1 = active_polish(best_h, best_c, d)
        if c1 < best_c:
            best_h, best_c = h1, c1

    d = t0 + 0.35 * usable
    if time.time() < d:
        cv = c_values(best_h)
        cv_max = float(np.max(cv))
        near = int(np.sum(cv >= cv_max - 1e-5))
        idx = np.argsort(cv)[-min(40, max(20, near)):]
        for rad in (0.5, 2.0, 6.0, None):
            if time.time() > d:
                break
            h2 = lp_linear(best_h, idx, rad)
            if h2 is not None:
                c2 = c5(h2)
                if c2 < best_c:
                    best_h, best_c = h2, c2

    d = t0 + 0.58 * usable
    if time.time() < d:
        h3, c3 = lbfgs_polish(best_h, best_c, d)
        if c3 < best_c:
            best_h, best_c = h3, c3

    d = t0 + 0.72 * usable
    if time.time() < d:
        for alpha in (1000.0, 5000.0):
            if time.time() > d:
                break
            h4 = slsqp_soft(best_h, alpha, 120)
            c4 = c5(h4)
            if c4 < best_c:
                best_h, best_c = h4, c4

    d = t0 + 0.90 * usable
    if time.time() < d:
        h5, c5v = active_polish(best_h, best_c, d)
        if c5v < best_c:
            best_h, best_c = h5, c5v

    if time.time() < deadline - 25:
        h2 = topk_binary(best_h, rng)
        d = min(deadline - 5, time.time() + 45)
        h3, c3 = active_polish(h2, best_c, d)
        if c3 < best_c:
            best_h, best_c = h3, c3

    if time.time() < deadline - 25:
        for sc in (0.05, 0.2, 0.5):
            rem = deadline - time.time()
            if rem < 20:
                break
            d = min(deadline - 5, time.time() + min(45.0, 0.6 * rem))
            h2 = perturb_feasible(best_h, sc, rng)
            h3, c3 = active_polish(h2, best_c, d)
            if c3 < best_c:
                best_h, best_c = h3, c3

    best_h = project_feasible(best_h, N_G)
    best_c = c5(best_h)
    return best_h, float(best_c), int(N_G)
```
