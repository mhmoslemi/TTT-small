```python
    """Compute change in correlation when h[i] 0->1 and h[j] 1->0."""
    # i is currently 0, j is currently 1
    delta = np.zeros(2*n-1)
    # lag m corresponds to index m + (n-1) in the correlate output
    # C[m] = sum_p h[p] * (1-h[p-m])
    
    # Affected terms:
    # 1. p = i: h[i]*(1-h[i-m]) goes from 0 to 1*(1-h_new[i-m])
    #    h_new[i-m]: if i-m == j, h_new[j]=0; if i-m == i (m=0), h_new[i]=1; else h_old[i-m]
    for m in range(-(n-1), n):
        idx = m + (n-1)
        dc = 0.0
        # p = i
        if 0 <= i - m < n:
            if i - m == j:
                dc += 1.0  # was 0, now 1*(1-0) = 1
            elif m == 0:
                dc += 0.0  # was 0, now 1*(1-1) = 0
            else:
                dc += 1.0 - h[i-m]
        # p = i + m (where p-m = i)
        if m != 0 and 0 <= i + m < n:
            old_val = h[i+m] * (1 - 0)  # was h[i+m]*(1-0) since h[i]=0
            if i + m == j:
                new_val = 0.0  # h_new[j] = 0
            elif i + m == i:  # impossible since m != 0
                new_val = 0.0
            else:
                new_val = h[i+m] * (1 - 1)  # h_new[i] = 1
            dc += new_val - old_val
        # p = j
        if 0 <= j - m < n:
            old_val = 1.0 * (1 - h[j-m])  # h[j] was 1
            if j - m == i:
                new_val = 0.0  # h_new[i] = 1, so 1-1=0... wait h_new[j-m] = h_new[i] = 1

                new_val = 0.0  # h_new[j] = 0
            else:
                new_val = 0.0  # h_new[j] = 0
            dc += new_val - old_val
        # p = j + m (where p - m = j)
        if m != 0 and 0 <= j + m < n:

            dc += new_val - old_val

    return delta
```

```python
def compute_delta(h, i, j, n):
    """Vectorized computation of delta_C for swap i:0->1, j:1->0."""
    m = np.arange(-(n-1), n)  # all possible lags
    delta = np.zeros(2*n-1)
    
    # p=i term: +1 if h_new[i-m]=0
    im = i - m
    valid_im = (im >= 0) & (im < n)
    # h_new[im]: if im==j, it's 0; if im==i, it's 1; else h[im]
    contrib1 = np.zeros(2*n-1)
    v = valid_im
    # For valid indices:
    h_im = h[np.clip(im, 0, n-1)]  # h[i-m] for valid i-m
    # If im == j, h_new = 0 (was 1, now 0): contribution is 1 (since 1-0=1)
    # If im == i (m=0), h_new = 1: contribution is 0 (since 1-1=0)
    # Otherwise: contribution is 1 - h[im]
    contrib1[v] = 1.0 - h_im[v]
    contrib1[v & (im == j)] = 1.0  # h_new[j] = 0
    contrib1[v & (im == i)] = 0.0  # h_new[i] = 1, m=0 case
    # Wait, im == i means i-m == i means m == 0. And im == j means i-m == j.
    # These can't both be true (i≠j).
    delta += contrib1
    
    # p=i+m term: -h[i+m] (for m≠0, valid i+m)
    ipm = i + m
    valid_ipm = (ipm >= 0) & (ipm < n) & (m != 0)
    contrib2 = np.zeros(2*n-1)
    contrib2[valid_ipm] = -h[np.clip(ipm, 0, n-1)][valid_ipm]
    delta += contrib2
    
    # p=j term: h[j-m] - 1 (for valid j-m)
    jm = j - m
    valid_jm = (jm >= 0) & (jm < n)
    contrib3 = np.zeros(2*n-1)
    contrib3[valid_jm] = h[np.clip(jm, 0, n-1)][valid_jm] - 1.0
    delta += contrib3
    
    # p=j+m term: +h_new[j+m] (for m≠0, valid j+m)
    jpm = j + m
    valid_jpm = (jpm >= 0) & (jpm < n) & (m != 0)
    contrib4 = np.zeros(2*n-1)
    h_jpm = h[np.clip(jpm, 0, n-1)]
    # If jpm == i, h_new = 1
    contrib4[valid_jpm] = h_jpm[valid_jpm]
    contrib4[valid_jpm & (jpm == i)] = 1.0
    delta += contrib4
    
    return delta
```

```python
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        def cons_fun(x):
            return x.sum() - target

        try:
            res = minimize(obj, h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': cons_fun}],
                           options={'maxiter': 500, 'ftol': 1e-14})
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
        except Exception:
            pass

    return best_h, best_c

def structured_start(n, seed):
    rng = np.random.RandomState(seed)
    # Try a few structured patterns
    patterns = []
    # Periodic block pattern
    for period in [7, 11, 13, 17, 19, 23, 29, 31]:
        freq = period / n
        x = np.arange(n)
        pat = ((x % period) < period // 2).astype(float)
        # Adjust to have exactly n/2 ones
        h = binarize(pat + rng.randn(n) * 0.1, n)
        patterns.append(h)
    # Random
    patterns.append(binarize(np.full(n, 0.5) + rng.randn(n) * 0.3, n))
    # Thue-Morse inspired
    tm = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    patterns.append(binarize(tm + rng.randn(n) * 0.2, n))

    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    # Binarize initial
    h_init = binarize(h0, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Phase 1: Local search from initial (50% budget)
    p1 = t0 + 0.50 * (budget_s - 5)
    p1 = min(p1, dl - 60)
    hb, cb = fast_local_search(h_init.copy(), n, p1, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Try structured starts (20% budget)
    if time.time() < dl - 40:
        p2 = min(t0 + 0.70 * (budget_s - 5), dl - 40)
        patterns = structured_start(n, seed + 1)
        for pat in patterns:
            if time.time() > p2:
                break
            hp, cp = fast_local_search(pat.copy(), n, min(time.time() + 20, p2), seed=seed + 7)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()

    # Phase 3: SLSQP polish (15% budget)
    p3 = min(t0 + 0.85 * (budget_s - 5), dl - 20)
    if time.time() < p3:
        hp, cp = slsqp_polish(best_h.copy(), n, p3)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Re-binarize and search again (remaining)
    if time.time() < dl - 10:
        h_bin = binarize(best_h, n)
        hb2, cb2 = fast_local_search(h_bin.copy(), n, dl - 5, seed=seed + 100)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Final SLSQP if time
    if time.time() < dl - 3:
        hp, cp = slsqp_polish(best_h.copy(), n, dl - 2)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
