        xn = np.linspace(0, 2, n2)
        hh = project_feasible(np.interp(xn, xo, best_h), n2)
        hbin2 = round_to_binary(hh, n2)
        hb3, cb3 = binary_sa(hbin2, n2, p3, seed=seed + 900,
                             t_start=0.04, t_end=0.000005, patience=45000)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
            best_n = n2

    # Phase 4: Final polish (remaining time)
    if time.time() < dl - 2:
        n_cur = len(best_h)
        h_fin = round_to_binary(best_h, n_cur)
        hb4, cb4 = binary_sa(h_fin, n_cur, dl - 2, seed=seed + 777,
                             t_start=0.025, t_end=0.000008, patience=80000)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
            best_n = n_cur

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
