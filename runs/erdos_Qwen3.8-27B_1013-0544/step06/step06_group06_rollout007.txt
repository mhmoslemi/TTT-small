    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def to_binary(h, n, rng):
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > target:
        excess = rng.choice(ones, len(ones) - target, replace=False)
        h_b[excess] = 0
    elif len(ones) < target:
        deficit = rng.choice(zeros, target - len(ones), replace=False)
        h_b[deficit] = 1
    return h_b

def tabu_search(h0, n, deadline, seed=42, tk_start=15, tk_end=5, tabu_sz=60):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    h = to_binary(h0, n, rng)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    tabu_expiry = np.zeros((n, n), dtype=np.int32)
    stall = 0
    no_improve = 0
    it = 0
    kick_cd = 0
    while time.time() < deadline:
        it += 1
        if it > 200000:
            break
        progress = min(1.0, it / 6000.0)
        tk = max(tk_end, int(tk_start - (tk_start - tk_end) * progress))
        top_idx = np.argsort(corr)[-tk:]
        g_z = g[zeros][:, top_idx]
        g_o = g[ones][:, top_idx]
        D = corr[top_idx][None, None, :] + g_z[:, None, :] - g_o[None, :, :]
        new_maxes = D.max(axis=2)
        active_tabu = tabu_expiry[zeros][:, ones] > it
        if active_tabu.any():
            new_maxes = new_maxes.copy()
            new_maxes[active_tabu] = 1e10
        if not np.isfinite(new_maxes).any():
            tabu_expiry[:] = 0
            continue
        flat = int(np.argmin(new_maxes))
        ii = flat // len(ones)
        jj = flat % len(ones)
        i = zeros[ii]
        j = ones[jj]
        new_c_est = new_maxes[ii, jj] * dx
        if new_c_est < cur_c - 1e-14:
            g, h, corr, zeros, ones = do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj)
            cur_c = corr.max() * dx
            stall = 0
            no_improve = 0
            tabu_expiry[i, j] = it + tabu_sz
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
        else:
            stall += 1
            no_improve += 1
            if (stall > 80 or no_improve > 400) and time.time() < deadline - 2:
                nk = min(3, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    ii2 = int(np.where(zeros == zeros[ri[a]])[0][0])
                    jj2 = int(np.where(ones == ones[rj[a]])[0][0])
                    g, h, corr, zeros, ones = do_swap(g, h, corr, zeros[ri[a]], ones[rj[a]], n, zeros, ones, ii2, jj2)
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                g = build_g(h, n)
                cur_c = corr.max() * dx
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
                no_improve = min(no_improve, 100)
                kick_cd += 1
                if kick_cd > 30:
                    break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def soft_round(h_bin, n, rng, softness):
    h = h_bin.copy()
    mask = rng.random(n) < 0.3
    h[mask] += softness * (rng.random(int(mask.sum())) - 0.5) * 2
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = thue_morse_init(n, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Tabu search from parent (42%)
    p1 = t0 + 0.42 * (budget_s - 5)
    if time.time() < p1 - 5:
        hb, cb = tabu_search(h0, n, p1, seed=seed, tk_start=15, tk_end=5, tabu_sz=60)
        if cb < best_c:
            best_c = cb
            best_h = hb
    # Phase 2: Tabu from Thue-Morse (18%)
    p2 = t0 + 0.60 * (budget_s - 5)
    if time.time() < p2 - 5:
        h_tm = thue_morse_init(n, rng)
        hb, cb = tabu_search(h_tm, n, p2, seed=seed + 77, tk_start=12, tk_end=5, tabu_sz=50)
        if cb < best_c:
            best_c = cb
            best_h = hb
    # Phase 3: SLSQP polish (20%)
    p3 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p3 - 3:
        h_soft = soft_round(best_h, n, rng, 0.4)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p3 - 3:
                break
            h_soft = slsqp_smooth(h_soft, n, alpha, 2000, p3 - 2)
            c = compute_c5(h_soft, n)
            if c < best_c:
                best_c = c
                best_h = h_soft.copy()
    # Phase 4: Tabu on best binary (15%)
    p4 = dl - 8
    if time.time() < p4 - 3:
        h_bin = to_binary(best_h, n, rng)
        hb, cb = tabu_search(h_bin, n, p4, seed=seed + 200, tk_start=10, tk_end=4, tabu_sz=40)
        if cb < best_c:
            best_c = cb
            best_h = hb
    # Phase 5: Final SLSQP micro-polish
    if time.time() < dl - 2:
        hf = soft_round(best_h, n, rng, 0.2)
        hf = slsqp_smooth(hf, n, 2000000, 800, dl - 1)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
