                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-12:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_pen(h, n, alpha, mu):
    return smooth_max_val(h, n, alpha) + mu * (h.sum() - n / 2.0) ** 2

def grad_pen(h, n, alpha, mu):
    return smooth_max_grad(h, n, alpha) + 2.0 * mu * (h.sum() - n / 2.0)

def lbfgs_step(h, n, deadline, alpha, mu, maxiter=200):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 15, 'maxls': 80})
        return project_feasible(res.x, n)
    except Exception:
        return h

def try_transfer(h, n, i, j, cur_best):
    hi = min(h[i], 1.0 - h[j])
    if hi < 1e-10:
        return h, cur_best
    oi, oj = h[i], h[j]
    best_c = cur_best
    best_t = 0.0
    fracs = (0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0)
    for f in fracs:
        t = hi * f
        h[i] = oi - t
        h[j] = oj + t
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_t = t
    h[i] = oi - best_t
    h[j] = oj + best_t
    return h, best_c

def grad_pair_search(h, n, deadline, alpha=5000):
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        order = np.argsort(g)
        pre = best_c
        step = max(1, n // 40)
        for idx in range(0, n // 2, step):
            if time.time() > deadline:
                break
            i = order[n - 1 - idx]
            j = order[idx]
            if h[i] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            h, best_c = try_transfer(h, n, i, j, best_c)
        if best_c >= pre - 1e-15:
            break
    return h, best_c

def random_pair_search(h, n, deadline, rng, n_pairs=150):
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        pre = best_c
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h, best_c = try_transfer(h, n, i, j, best_c)
        if best_c >= pre - 1e-15:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS with progressive alpha sharpening
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for ai, alpha in enumerate(alphas):
        if time.time() > deadline - 200:
            break
        remaining = (deadline - 200 - time.time()) / (len(alphas) - ai)
        alloc = min(50, max(15, remaining))
        mu = max(1.0, alpha * 3e-5)
        h = lbfgs_step(h, n, time.time() + alloc, alpha, mu, maxiter=200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Gradient-guided pair search (multiple alpha levels)
    h = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 100:
            break
        alloc = min(60, max(20, (deadline - 100 - time.time()) / 4.0))
        if alloc < 10:
            break
        h, c = grad_pair_search(h, n, time.time() + alloc, alpha)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Random pair search for exploration
    if time.time() < deadline - 50:
        h = best_h.copy()
        alloc = min(40, deadline - 50 - time.time())
        if alloc > 10:
            h, c = random_pair_search(h, n, time.time() + alloc, rng, n_pairs=200)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 4: Perturbation + re-optimization to escape local minima
    for attempt in range(4):
        if time.time() > deadline - 30:
            break
        h_p = best_h + rng.randn(n) * 0.02
        h_p = project_feasible(h_p, n)
        t_rem = deadline - 30 - time.time()
        if t_rem < 15:
            break
        h_p = lbfgs_step(h_p, n, time.time() + 15, 100000, 10, maxiter=120)
        h_p, c_p = grad_pair_search(h_p, n, time.time() + min(10, t_rem - 16), 20000)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()

    # Phase 5: Try smaller n via interpolation
    if time.time() < deadline - 40:
        for n_try in [150, 200]:
            if time.time() > deadline - 20:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, n_try)
            alloc1 = min(25, (deadline - 20 - time.time()) // 2)
            if alloc1 < 10:
                break
            h_t = lbfgs_step(h_t, n_try, time.time() + alloc1, 50000, 5, maxiter=120)
            h_t, c_t = grad_pair_search(h_t, n_try, time.time() + min(15, deadline - 18), 20000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Phase 6: Final fine-tuning with high alpha
    if time.time() < deadline - 3:
        h_f = best_h.copy()
        h_f, c_f = grad_pair_search(h_f, n, deadline - 2, alpha=500000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
