def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:

            if s < 1e-14:

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

    time_per_n = time_avail / len(n_list)
    for n_try in n_list:
        if time.time() > p2 - 10:
            break
        p_n = min(time.time() + time_per_n, p2 - 5)
        if n_try == len(best_h):
            h_interp = best_h.copy()
        else:
            xo = np.linspace(0, 2.0, len(best_h))
            xn = np.linspace(0, 2.0, n_try)
            h_interp = np.interp(xn, xo, best_h)
        h_bin = round_to_binary(h_interp, n_try)
        for att in range(3):
            if time.time() > p_n:
                break
            if att == 0:
                start_h = h_bin
                s = seed * 100 + n_try
            elif att == 1:
                start_h = round_to_binary(h_interp + 0.15 * rng.randn(n_try), n_try)
                s = seed * 100 + n_try + 1
            else:
                base = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, len(best_h)), best_h)
                start_h = round_to_binary(base + 0.1 * rng.randn(n_try), n_try)
                s = seed * 100 + n_try + 2
            hb, cb = binary_sa(start_h, n_try, p_n, seed=s)
            actual = compute_c5(hb, n_try)
            if actual < best_c:
                best_c = actual; best_h = hb.copy(); best_n = n_try
    # Phase 3: Greedy polish (7%)
    p3 = t0 + 0.87 * (budget_s - 5)
    hg, cg = greedy_binary(best_h, best_n, p3, max_sweeps=60, seed=seed + 5)
    if cg < best_c:
        best_c = cg; best_h = hg.copy()
    # Phase 4: Final SLSQP refinement (8%)
    p4 = dl
    if time.time() < p4 - 15:
        hp = project_feasible(best_h + 0.02 * rng.randn(best_n), best_n)
        for alpha in [100000, 1000000, 5000000]:
            if time.time() > p4 - 5:
                break
            hp = slsqp_smooth(hp, best_n, alpha, 2000, 1e-15)
            c = compute_c5(hp, best_n)
            if c < best_c:
                best_c = c; best_h = hp.copy()
    # Phase 5: Higher-resolution attempt if time permits
    if time.time() < dl - 25 and best_n < 500:
        n2 = min(500, best_n + 50)
        xo = np.linspace(0, 2.0, len(best_h))
        xn = np.linspace(0, 2.0, n2)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, n2)
        hb5, cb5 = binary_sa(round_to_binary(hh, n2), n2, dl - 15, seed=seed + 500)
        c5a = compute_c5(hb5, n2)
        hp5 = slsqp_smooth(project_feasible(hb5 + 0.02 * rng.randn(n2), n2), n2, 500000, 1000, 1e-14)
        c5b = compute_c5(hp5, n2)
        cc = min(c5a, c5b)
        if c5a <= c5b and c5a < best_c:
            best_c = c5a; best_h = hb5.copy(); best_n = n2
        elif c5b < best_c:
            best_c = c5b; best_h = hp5.copy(); best_n = n2
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
