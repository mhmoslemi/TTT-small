import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_improve(h, corr, n, m_range, dx, rng, max_rounds=200, deadline=None):
    cur_c = corr.max() * dx
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_dc = -1e-15
        best_i = -1
        best_j = -1
        best_d = None
        for i in zeros:
            for j in ones:
                d = swap_delta(h, n, i, j, m_range)
                nc = (corr + d).max() * dx
                dc = nc - cur_c
                if dc < best_dc:

                    best_i = i
                    best_j = j
                    best_d = d
        if best_i < 0:
            break
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr = corr + best_d
        cur_c += best_dc
    return h, corr, cur_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        ew = np.exp(sh - sm)
        sw = ew.sum()
        grad_c = alpha * ew / sw
        grad_c = np.correlate(grad_c, np.ones(2 * n - 1), mode='full')[:n] * dx
        return -grad_c
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h

def structured_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    base = np.arange(target) * (n // target)
    jitter = rng.randint(-2, 3, size=target)
    pos = np.clip(base + jitter, 0, n - 1)
    h[np.unique(pos)] = 1.0
    if h.sum() < target:
        remaining = np.where(h == 0)[0]
        extra = rng.choice(remaining, size=target - int(h.sum()), replace=False)
        h[extra] = 1.0
    elif h.sum() > target:
        ones = np.where(h == 1)[0]
        remove = rng.choice(ones, size=int(h.sum()) - target, replace=False)
        h[remove] = 0.0
    return h
