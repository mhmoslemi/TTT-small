def softmax_obj(h, n, alpha):
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full')
    # Softmax of c*dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c) * dx)
    return (max_c + np.log(exp_c.sum()) / alpha) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=200, grid=150, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo, d_hi = -h[p], 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - p, p))
            m_hi = min(2 * n - 2, max(2 * n - 2 - p, n - 1 + p))
            um = 0.0
            if m_lo > 0:
                um = max(um, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                um = max(um, float(c[m_hi + 1:].max()))
            ks = np.arange(m_lo, m_hi + 1)
            ksh = ks - (n - 1)
            i1, i2 = p + ksh, p - ksh
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            s = np.zeros(len(ks))
            s[v1] += 1.0 - h[i1[v1]]
            s[v2] -= h[i2[v2]]
            if np.max(np.abs(s)) < 1e-15 and abs(1 - 2 * h[p]) < 1e-15:
                continue
            deltas = np.linspace(d_lo, d_hi, grid)
            aff = c[ks][None, :] + deltas[:, None] * s[None, :]
            ci = np.where(ks == n - 1)[0]
            if len(ci) > 0:
                aff[:, ci[0]] = c[n - 1] + deltas * (1 - 2 * h[p]) - deltas ** 2
            vals = np.maximum(um, aff.max(axis=1))
            bi = vals.argmin()
            if vals[bi] < current_max - 1e-15:
                d = deltas[bi]
                c[ks] += d * s
                c[n - 1] -= d * d
                h[p] = np.clip(h + 0, 0, 1)  # placeholder
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(c[ks].max()))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def softmax_lbfgs(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cd = c * dx
        mc = cd.max()
        return mc + np.log(np.sum(np.exp(alpha * (cd - mc)))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cd = c * dx
        mc = cd.max()
        exp_cd = np.exp(alpha * (cd - mc))
        w = exp_cd / exp_cd.sum()
        g = np.zeros(n)
        for p in range(n):
            k = np.arange(-(n - 1), n)
            m_idx = (n - 1) + k
            i1 = p + k
            i2 = p - k
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            t1 = np.zeros(len(k))
            t1[v1] = 1.0 - x[i1[v1]]
            t2 = np.zeros(len(k))
            t2[v2] = x[i2[v2]]
            g[p] = dx * (w * (t1 - t2)).sum()
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo, hi = max(0, i - radius), min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:

                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

1. Precompute the correlation c

                i2 = i - ksh_i

                s_i[ks_i[v2]] -= h[i2[v2]]

                        continue

                    m_lo_j = max(0, min(n-1-j, j))

                    m_lo = min(m_lo_i, m_lo_j)

                        um = max(um, float(c[:m_lo].max()))

                        um = max(um, float(c[m_hi+1:].max()))

                    new_c[ks == n-1] -= 2 * d * d  # from both h[i]+d and h[j]-d
                    # Wait, I need to be more careful.
                    # The s_i already includes the linear part. The quadratic part is:
                    # -d² for h[i] and -d² for h[j], total -2d²
                    # But s_i at the center = 1-2h[i], and the linear change is d*(1-2h[i])
                    # Actual change: d*(1-2h[i]) - d²
                    # Similarly for j: -d*(1-2h[j]) - d²
                    # Total actual: d*(1-2h[i]) - d² - d*(1-2h[j]) - d² = d*(s_i[center] - s_j[center]) - 2d²
                    # My formula gives: d*(s_i[center] - s_j[center]) - 2d² ✓
                    
                    val = max(um, float(new_c.max()))
                    if val < best_c - 1e-14:
                        # Accept
                        c[ks] = new_c
                        h[i] += d
                        h[j] -= d
                        best_c = val
                        improved = True
            if not improved:
                break
    return h, best_c * (2.0/n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=300, grid=150, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    nfull = 2 * n - 1
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo, d_hi = -h[p], 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - p, p))
            m_hi = min(nfull - 1, max(nfull - 1 - p, n - 1 + p))
            um = 0.0
            if m_lo > 0:
                um = max(um, float(c[:m_lo].max()))
            if m_hi < nfull - 1:
                um = max(um, float(c[m_hi + 1:].max()))
            ks = np.arange(m_lo, m_hi + 1)
            ksh = ks - (n - 1)
            i1, i2 = p + ksh, p - ksh
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            s = np.zeros(len(ks))
            s[v1] += 1.0 - h[i1[v1]]
            s[v2] -= h[i2[v2]]
            if np.max(np.abs(s)) < 1e-15 and abs(1 - 2 * h[p]) < 1e-15:
                continue
            deltas = np.linspace(d_lo, d_hi, grid)
            aff = c[ks][None, :] + deltas[:, None] * s[None, :]
            ci = np.where(ks == n - 1)[0]
            if len(ci) > 0:
                aff[:, ci[0]] = c[n - 1] + deltas * (1 - 2 * h[p]) - deltas ** 2
            vals = np.maximum(um, aff.max(axis=1))
            bi = vals.argmin()
            if vals[bi] < current_max - 1e-15:
                d = deltas[bi]
                c[ks] += d * s
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                current_max = max(um, float(c[ks].max()))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def softmax_lbfgs(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cd = c * dx
        mc = cd.max()
        return mc + np.log(np.sum(np.exp(alpha * (cd - mc)))) / alpha

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cd = c * dx
        mc = cd.max()
        exp_cd = np.exp(alpha * (cd - mc))
        w = exp_cd / exp_cd.sum()
        g = np.zeros(n)
        for p in range(n):
            k = np.arange(-(n - 1), n)
            i1 = p + k
            i2 = p - k
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            t1 = np.zeros(len(k))
            t1[v1] = 1.0 - x[i1[v1]]
            t2 = np.zeros(len(k))
            t2[v2] = x[i2[v2]]
            g[p] = dx * (w * (t1 - t2)).sum()
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
