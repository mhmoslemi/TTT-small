```python
def subgradient_step(h, n, step_size):
    """One step of subgradient method for min max_k f_k(h)."""
    dx = 2.0 / n
    # Compute all f_k
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = corr.argmax() - (n-1)  # the lag that achieves the max
    
    # Compute gradient at k_max
    k = k_max
    grad = np.zeros(n)
    if k >= 0:
        # f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[j] = (1-h[j+k]) if j < n-k, plus -h[j-k] if k <= j < n
        for j in range(n - k):
            grad[j] += (1 - h[j+k])
        for j in range(k, n):
            grad[j] -= h[j-k]
    else:
        # k < 0, let k' = -k
        # f_{-k'} = sum_{i=k'}^{n-1} h[i]*(1-h[i-k'])
        # Let j = i: f = sum_{j=k'}^{n-1} h[j]*(1-h[j-k'])
        # d/dh[j] = (1-h[j-k']) if j >= k', plus -h[j+k'] if j+k' < n
        kp = -k
        for j in range(kp, n):
            grad[j] += (1 - h[j-kp])
        for j in range(n - kp):
            grad[j] -= h[j+kp]
    
    # Step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

This is O(n) per step, which is very efficient! We can do many steps.

Alternatively, I can use a **proximal gradient** approach:
- Maintain a "smooth" estimate of the max using logsumexp
- Use the analytical gradient of the logsumexp objective
- Project onto the constraint set at each step

The gradient of the logsumexp objective:
g(x) = alpha * (corr - max(corr))
weights = exp(g) / sum(exp(g))
objective = max(corr) + log(sum(exp(g))) / alpha

d(objective)/dx = d(max(corr))/dx + (1/alpha) * sum_k weights[k] * d(corr[k])/dx

The first term is the gradient at the argmax. The second term is a weighted sum of gradients.

This is still O(n) per gradient evaluation (since each corr[k] has a sparse gradient).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_grad(h, n, alpha):
    """Fast gradient computation for LSE objective using sparse significant weights."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    center = n - 1
    sig = np.where(weights > 1e-12)[0]
    for idx in sig:
        w = weights[idx]
        k = idx - center
        if k >= 0:
            if k < n:
                grad[:n-k] += w * dx * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w * dx * h[:n-k]
            else:
                grad[:] -= w * dx * h[:]
        else:
            kp = -k
            if kp < n:
                grad[kp:] += w * dx * (1.0 - h[:n-kp]) if kp > 0 else w * dx * (1.0 - h[:])
            if kp > 0:
                grad[:n-kp] -= w * dx * h[kp:]
            else:
                grad[:] -= w * dx * h[:]
    return grad

def project_gradient_step(h, n, alpha, lr, deadline, n_steps=50000):
    """Project gradient descent with LSE smooth max."""
    h = h.copy()
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        grad = lse_grad(h, n, alpha)
        h = h - lr * grad
        h = project_feasible(h, n)
    return h

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        return m + np.log(np.exp(shifted).sum()) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n-1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                delta[k1] += 1.0
                k2 = n-1 + (i-j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=10000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n

    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        delta[k1] += 1.0
        k2 = n-1 + (i-j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
```
