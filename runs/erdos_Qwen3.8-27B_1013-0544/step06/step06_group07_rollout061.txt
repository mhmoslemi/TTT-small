import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greedy_local_search(h, n, deadline, max_sweeps=200):
    """Greedy: accept first improving swap found."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Try all pairs in random order
        pairs = [(i, j) for i in zeros for j in ones]
        np.random.shuffle(pairs)
        for i, j in pairs:
            delta = swap_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            nm = new_corr.max()
            if nm < cur_max - 1e-15:
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_max = nm
                improved = True
                break
        if not improved:
            break
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    return best_h, best_max * dx

def slsqp_smooth(h, n, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        M = g.max()
        u = alpha * (g - M)
        um = u.max()
        return M + (um + np.log(np.exp(u - um).sum())) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        M = g.max()
        k_star = np.argmax(g)
        u = alpha * (g - M)
        um = u.max()
        w = np.exp(u - um)
        ws = w.sum()
        wn = w / ws
        val = M + (um + np.log(ws)) / alpha
        c1 = np.correlate(wn, 1.0 - x, mode='full')
        c2 = np.correlate(wn, x, mode='full')
        A = c1[n - 1: 3 * n - 2]
        B = c2[2 * n - 2: 3 * n - 2][::-1]
        ls = lags[k_star]
        idx = np.arange(n)
        ip = idx + ls
        im2 = idx - ls
        sub = np.zeros(n)
        mp = (ip >= 0) & (ip < n)
        sub[mp] += 1.0 - x[ip[mp]]
        mm = (im2 >= 0) & (im2 < n)
        sub[mm] -= x[im2[mm]]
        grad = dx * (sub + A - B)
        return val, grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,

            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                zi = zeros[rng.randint(len(zeros))]
                oj = ones[rng.randint(len(ones))]
                h[zi] = 1.0
                h[oj] = 0.0
                zeros.remove(zi)
                zeros.append(oj)
                ones.remove(oj)
                ones.append(zi)
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max()
            no_improve = 0
            T = t_start * 0.6
    return best_h, best_c * dx


def greedy_polish(h, n, deadline, max_iter=200000, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0].tolist()
    ones = np.where(h == 1)[0].tolist()
    iters = 0
    while time.time() < deadline and iters < max_iter:
        improved = False
        n_try = min(200, len(zeros) * len(ones))
        for _ in range(n_try):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = swap_delta(h, n, i, j, m_range)
            nc = (corr + delta).max()
            if nc < cur_c - 1e-15:
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta
                cur_c = nc
                zeros.remove(i)
                zeros.append(j)
                ones.remove(j)
                ones.append(i)
                improved = True
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                break
            iters += 1
        if not improved:
            break
    return best_h, best_c * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP smooth (25%)
    p1 = t0 + 0.25 * (budget_s - 3)
    h = project_feasible(h0.copy(), n)
    for alpha in [200, 500, 1000, 3000, 10000, 50000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 3000, 1e-16, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary SA (30%)
    p2 = t0 + 0.55 * (budget_s - 3)
    h_bin = round_to_binary(best_h, n)
    for attempt in range(4):
        if time.time() > p2:
            break
        if attempt == 0:
            start = h_bin.copy()
            s = seed
        elif attempt == 1:
            start = round_to_binary(best_h + 0.08 * rng.randn(n), n)
            s = seed + 10
        elif attempt == 2:
            start = round_to_binary(best_h + 0.15 * rng.randn(n), n)
            s = seed + 20
        else:
            start = round_to_binary(0.5 * best_h + 0.5 * rng.rand(n), n)
            s = seed + 30
        end = p2 if attempt < 3 else p2
        hb, cb = binary_sa(start, n, end, seed=s, t_start=0.03, t_end=0.00001, patience=30000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy polish (15%)
    p3 = t0 + 0.70 * (budget_s - 3)
    hg, cg = greedy_polish(best_h, n, p3, max_iter=300000, seed=seed + 50)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 4: SLSQP from best binary (20%)
    p4 = t0 + 0.90 * (budget_s - 3)
    if time.time() < p4 - 5:
        h4 = project_feasible(best_h.copy() + 0.02 * rng.randn(n), n)
        for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
            if time.time() > p4:
                break
            h4 = slsqp_smooth(h4, n, alpha, 3000, 1e-16, p4)
            c = compute_c5(h4, n)
            if c < best_c:
                best_c = c
                best_h = h4.copy()

    # Phase 5: Final SA + greedy (remaining)
    if time.time() < deadline - 15:
        h_bin2 = round_to_binary(best_h, n)
        hb2, cb2 = binary_sa(h_bin2, n, deadline - 15, seed=seed + 100,
                             t_start=0.015, t_end=0.00002, patience=25000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    if time.time() < deadline - 3:
        hg2, cg2 = greedy_polish(best_h, n, deadline - 1, max_iter=100000, seed=seed + 200)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
