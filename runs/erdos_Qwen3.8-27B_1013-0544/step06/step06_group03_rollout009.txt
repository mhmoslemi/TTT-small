def project_box_sum(v, target):
    v = np.array(v, dtype=float, copy=True)
    n = v.size
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    # if all v same, bisection okay.
    for _ in range(80):
        mid = 0.5*(lo+hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = float(h.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
        # fallback
        for _ in range(200):
            diff = target - float(h.sum())
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                room = 1.0 - h
                tot = float(room.sum())
                if tot < 1e-12:
                    break
                h += diff * room / tot
            else:
                tot = float(h.sum())
                if tot < 1e-12:
                    break
                h += diff * h / tot
    return np.clip(h, 0.0, 1.0)

def resample_h(h, n_new):
    n=len(h)
    dx=2.0/n
    old_x = np.linspace(0.0, 2.0, n+1)

def _add_lag_grad(g, h, L, wt):
    n = len(h)
    p = abs(int(L))

    order = np.argsort(w)[::-1]
    out = []
    cum = 0.0
    for idx in order:
        idx = int(idx)
        out.append(idx)
        cum += float(w[idx])
        if cum >= max_mass or len(out) >= max_lags:
            break
    return out


def _smooth_obj(x, n, alpha):
    C = _get_C(x)
    mx = float(C.max())
    z = alpha * (C - mx)
    s = float(np.sum(np.exp(z)))
    return float((mx + np.log(s) / alpha) * (n / 2.0))


def _smooth_jac(x, n, alpha):
    C = _get_C(x)
    mx = float(C.max())
    z = alpha * (C - mx)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return np.zeros(n)
    w = e / s
    g = np.zeros(n)
    base = n - 1
    for idx, wt in enumerate(w):
        if wt > 1e-12:
            _add_lag_grad(g, x, base - idx, float(wt))
    return g * (n / 2.0)


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def slsqp_all(h, n, alpha, stop_time):
    if time.time() > stop_time - 1.0:
        return project_box_sum(np.asarray(h, dtype=float), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_con, "jac": _sum_con_jac, "args": (n,)}]
    maxiter = int(min(1500, max(20, (stop_time - time.time()) * 3.0)))
    _STOP_TIME[0] = stop_time - 0.4
    try:
        res = minimize(
            _smooth_obj,
            project_box_sum(np.asarray(h, dtype=float), n / 2.0),
            args=(n, alpha),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=_smooth_jac,
            callback=_time_stop_callback,
            options={"maxiter": maxiter, "ftol": 1e-13},
        )
    except Exception:
        res = None
    finally:
        _STOP_TIME[0] = None
    if res is not None and hasattr(res, "x"):
        return project_box_sum(np.asarray(res.x, dtype=float), n / 2.0)
    return project_box_sum(np.asarray(h, dtype=float), n / 2.0)


def optimize_slsqp(hstart, n, deadline):
    h_work = project_box_sum(np.asarray(hstart, dtype=float).copy(), n / 2.0)
    c_work = compute_c5(h_work)
    best_h = h_work.copy()
    best_c = c_work
    stages = [(10.0, 0.45), (30.0, 0.25), (100.0, 0.15), (300.0, 0.10), (1000.0, 0.05)]
    for alpha, frac in stages:
        if time.time() > deadline - 4.0:
            break
        left = deadline - time.time()
        stage_end = min(deadline, time.time() + max(6.0, left * frac))
        h_new = slsqp_all(h_work, n, alpha, stage_end)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < best_c + 0.01:
            h_work = project_box_sum(h_new, n / 2.0)
            c_work = c_new
        else:
            h_work = best_h.copy()
            c_work = best_c
    return best_h, best_c


def fw_line_search(hstart, n, stop_time):
    h = project_box_sum(np.asarray(hstart, dtype=float).copy(), n / 2.0)
    c = compute_c5(h)
    target = n / 2.0
    alpha = 350.0
    for _ in range(5000):
        if time.time() > stop_time:
            break
        C = np.correlate(h, 1.0 - h, mode="full")
        z = alpha * (C - C.max())
        e = np.exp(z)
        s = float(e.sum())
        w = e / s
        idxs = _active_indices(w, 0.99, 180)
        g = np.zeros(n)
        base = n - 1
        for idx in idxs:
            _add_lag_grad(g, h, base - idx, float(w[idx]))
        order = np.argsort(g)
        zdir = np.zeros(n)
        k = int(np.floor(target))
        frac = target - k
        if k > 0:
            zdir[order[:k]] = 1.0
        if frac > 1e-12 and k < n:
            zdir[order[k]] = frac
        if np.max(np.abs(h - zdir)) < 1e-7:
            break
        best_h = h
        best_c = c
        best_t = 0.0
        for t in (0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
            cand = (1.0 - t) * h + t * zdir
            if cand.min() < -1e-10 or cand.max() > 1.0 + 1e-10 or abs(target - cand.sum()) > 1e-10:
                cand = project_box_sum(cand, target)
            else:
                cand = np.clip(cand, 0.0, 1.0)
            cn = compute_c5(cand)
            if cn < best_c - 1e-13:
                best_c = cn
                best_h = cand
                best_t = t
        if best_t == 0.0:
            break
        h = best_h
        c = best_c
    return project_box_sum(h, target), compute_c5(h)


def cvx_linearized(hstart, n, stop_time):
    h = project_box_sum(np.asarray(hstart, dtype=float).copy(), n / 2.0)
    c = compute_c5(h)
    try:
        import cvxpy as cp
    except Exception:
        return h, c
    for _ in range(2):
        if time.time() > stop_time:
            break
        C = np.correlate(h, 1.0 - h, mode="full")
        order = np.argsort(C)[::-1][:100]
        x = cp.Variable(n)
        t = cp.Variable()
        cons = [x >= 0.0, x <= 1.0, cp.sum(x) == n / 2.0, t >= 0.0]
        dx = 2.0 / n
        for idx in order:
            g = np.zeros(n)
            _add_lag_grad(g, h, n - 1 - int(idx), 1.0)
            cons.append(C[idx] + dx * (g @ (x - h)) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        solved = False
        for sol in (cp.GLOP, cp.CBC, cp.ECOS):
            try:
                prob.solve(solver=sol)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        xval = x.value
        if xval is None:
            break
        hnew = project_box_sum(np.asarray(xval, dtype=float), n / 2.0)
        cnew = compute_c5(hnew)
        if cnew < c - 1e-12:
            h = hnew
            c = cnew
        else:
            break
    return project_box_sum(h, n / 2.0), compute_c5(h)


def pair_descent(hstart, n, stop_time, rng):
    h = project_box_sum(np.asarray(hstart, dtype=float).copy(), n / 2.0)
    c = compute_c5(h)
    step = 0.06
    bad = 0
    alpha = 450.0
    for _ in range(30000):
        if time.time() > stop_time:
            break
        C = np.correlate(h, 1.0 - h, mode="full")
        z = alpha * (C - C.max())
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0:
            break
        w = e / s
        idxs = _active_indices(w, 0.99, 180)
        g = np.zeros(n)
        base = n - 1
        for idx in idxs:
            _add_lag_grad(g, h, base - idx, float(w[idx]))
        dec_mask = h > 1e-9
        inc_mask = h < 1.0 - 1e-9
        if not dec_mask.any() or not inc_mask.any():
            break
        gd = np.where(dec_mask, g, -np.inf)
        gi = np.where(inc_mask, g, np.inf)
        i = int(np.argmax(gd))
        j = int(np.argmin(gi))
        if i == j:
            dec_idx = np.where(dec_mask)[0]
            inc_idx = np.where(inc_mask)[0]
            i = int(dec_idx[rng.randint(len(dec_idx))])
            j = int(inc_idx[rng.randint(len(inc_idx))])
        moves = []
        if h[i] > 1e-9 and h[j] < 1.0 - 1e-9:
            moves.append((i, j))
        if h[j] > 1e-9 and h[i] < 1.0 - 1e-9:
            moves.append((j, i))
        best_h = None
        best_c = c
        improved = False
        for a, b in moves:
            full = min(h[a], 1.0 - h[b])
            if full <= 1e-8:
                continue
            d0 = min(step, full)
            for d in (d0, full, 0.5 * full, 0.25 * full):
                if d <= 1e-8:
                    continue
                cand = h.copy()
                cand[a] -= d
                cand[b] += d
                if cand[a] < -1e-9 or cand[b] > 1.0 + 1e-9:
                    continue
                cn = compute_c5(cand)
                if cn < best_c - 1e-13:
                    best_c = cn
                    best_h = cand
                    improved = True
        if improved:
            h = best_h
            c = best_c
            step = min(0.35, max(0.002, step * 1.12))
            bad = 0
        else:
            step *= 0.75
            bad += 1
            if bad > 35 or step < 5e-5:
                break
    return project_box_sum(h, n / 2.0), compute_c5(h)


def resample_h(h, n_new):
    n = len(h)
    dx = 2.0 / n
    old_x = np.linspace(0.0, 2.0, n + 1)
    old_c = np.concatenate(([0.0], np.cumsum(h) * dx))
    new_dx = 2.0 / n_new
    new_edges = np.linspace(0.0, 2.0, n_new + 1)
    new_areas = np.interp(new_edges, old_x, old_c) - np.interp(
        new_edges - new_dx, old_x, old_c, left=0.0, right=0.0
    )
    h_new = new_areas / new_dx
    return project_box_sum(h_new, n_new / 2.0)


def default_start(n, rng):
    x = 2.0 * np.arange(n, dtype=float) / float(n)
    h = 0.5 + 0.35 * np.cos(np.pi * x) + 0.1 * np.sin(2.0 * np.pi * x)
    h += 0.02 * rng.normal(size=n)
    return project_box_sum(h, n / 2.0)


def optimize_one(hstart, n, deadline, rng):
    h = project_box_sum(np.asarray(hstart, dtype=float).copy(), n / 2.0)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    if time.time() < deadline - 12.0:
        dl = min(deadline, time.time() + max(10.0, (deadline - time.time()) * 0.50))
        h2, c2 = optimize_slsqp(h, n, dl)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
    if time.time() < deadline - 10.0:
        dl = min(deadline, time.time() + max(8.0, (deadline - time.time()) * 0.25))
        h2, c2 = cvx_linearized(best_h, n, dl)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
    if time.time() < deadline - 8.0:
        dl = min(deadline, time.time() + max(6.0, (deadline - time.time()) * 0.30))
        h2, c2 = fw_line_search(best_h, n, dl)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
    if time.time() < deadline - 5.0:
        dl = min(deadline, time.time() + max(5.0, (deadline - time.time()) * 0.40))
        h2, c2 = pair_descent(best_h, n, dl, rng)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
    if time.time() < deadline - 4.0:
        dl = min(deadline, time.time() + max(4.0, (deadline - time.time()) * 0.35))
        h2 = slsqp_all(best_h, n, 10000.0, dl)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2
    best_h = project_box_sum(best_h, n / 2.0)
    best_c = compute_c5(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(int(seed))
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 10:
            raise ValueError
    except Exception:
        n0 = 300
        h0 = default_start(n0, rng)
    if n0 > 700:
        h0 = resample_h(h0, 400)
        n0 = 400
    best_h = project_box_sum(h0, n0 / 2.0)
    best_c = compute_c5(best_h)
    if time.time() < deadline - 20.0:
        main_end = min(deadline, t0 + max(90.0, (deadline - t0) * 0.38))
        h_opt, c_opt = optimize_one(best_h, len(best_h), main_end, rng)
        if c_opt < best_c - 1e-12:
            best_h, best_c = h_opt, c_opt
    for n_try in (240, 360, 220, 280, 320, 200, 260):
        if time.time() > deadline - 70.0:
            break
        if n_try == len(best_h):
            continue
        remaining = deadline - time.time()
        if remaining < 80.0:
            break
        h_start = resample_h(best_h, n_try)
        dl = min(deadline, time.time() + min(130.0, max(45.0, remaining * 0.22)))
        h_opt, c_opt = optimize_one(h_start, n_try, dl, rng)
        if c_opt < best_c - 1e-12:
            best_h, best_c = h_opt, c_opt
    if time.time() < deadline - 100.0:
        for r in range(2):
            if time.time() > deadline - 50.0:
                break
            lb = len(best_h)
            scale = 0.015 * (0.5 + r)
            hp = best_h + rng.normal(0.0, scale, lb)
            hp = project_box_sum(hp, lb / 2.0)
            dl = min(deadline, time.time() + min(90.0, max(30.0, (deadline - time.time()) * 0.25)))
            h_opt, c_opt = optimize_one(hp, lb, dl, rng)
            if c_opt < best_c - 1e-12:
                best_h, best_c = h_opt, c_opt
    if time.time() < deadline - 15.0:
        dl = deadline - 1.0
        h_opt, c_opt = optimize_one(best_h, len(best_h), dl, rng)
        if c_opt < best_c - 1e-12:
            best_h, best_c = h_opt, c_opt
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
