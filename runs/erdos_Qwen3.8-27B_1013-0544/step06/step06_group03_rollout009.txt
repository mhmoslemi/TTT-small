def weighted_gradient(h, n, alpha):
    dx = 2.0 / n
    lags = compute_all_lags(h, n)  # shape (2n-1,)
    # softmax weights
    shifted = alpha * (lags - lags.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for k in range(2*n - 1):
        m = abs(k - (n-1))
        if m == 0:
            g = dx * (1 - 2*h)
        elif k - (n-1) > 0:
            # f_m(h) = dx * sum_{i=0}^{n-m-1} h[i]*(1-h[i+m])
            g = np.zeros(n)
            g[:n-m] += dx * (1 - h[m:])
            g[m:] -= dx * h[:n-m]
        else:
            # f_{-m}(h) = dx * sum_{i=0}^{n-m-1} h[i+m]*(1-h[i])
            g = np.zeros(n)
            g[m:] += dx * (1 - h[:n-m])
            g[:n-m] -= dx * h[m:]
        grad += w[k] * g
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, m):
    """Gradient of f_m(h) = dx * sum h[i](1-h[i+m]) or h[i+m](1-h[i])"""
    dx = 2.0 / n
    g = np.zeros(n)
    if m == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif m > 0:
        g[:n-m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:n-m]
    else:
        m = -m
        g[m:] += dx * (1.0 - h[:n-m])
        g[:n-m] -= dx * h[m:]
    return g

def weighted_grad(h, n, alpha):

    dx = 2.0 / n
    lags = compute_all_lags(h, n)

        grad += w[k] * lag_gradient(h, n, m)

def nesterov_optimize(h, n, deadline, alpha=1000, n_iter=2000):
    """Nesterov accelerated gradient with projection."""
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.1
    momentum = 0.9
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Nesterov step
        h_y = h_curr + momentum * (h_curr - h_prev)
        h_y = project_feasible(h_y, n)
        grad, _ = weighted_grad(h_y, n, alpha)
        h_new = h_y - lr * grad
        h_prev = h_curr
        h_curr = project_feasible(h_new, n)
        # Adaptive lr
        c = compute_c5(h_curr, n)
        if c < best_c:
            best_c = c
            best_h = h_curr.copy()
        # Simple lr adaptation
        if it % 100 == 99:
            lr *= 0.95
    return best_h, best_c

    top_idx = np.argsort(lags)[-top_k:]

        c = compute_all_lags(x, n)
        a = c[top_idx]
        m = a.max()
        s = alpha * (a - m)
        return float(m + np.log(np.sum(np.exp(s))) / alpha)

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, m):
    dx = 2.0 / n
    g = np.zeros(n)
    if m == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif m > 0:
        g[:n - m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:n - m]
    else:
        m = -m
        g[m:] += dx * (1.0 - h[:n - m])
        g[:n - m] -= dx * h[m:]
    return g

def softmax_grad(h, n, alpha):
    lags = compute_all_lags(h, n)
    s = alpha * (lags - lags.max())
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * lag_gradient(h, n, k - (n - 1))
    return grad

def nesterov_run(h, n, deadline, alpha=5000, n_iter=3000):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.5
    beta = 0.9

        if time.time() > deadline:
            break
        h_y = h_curr + beta * (h_curr - h_prev)
        h_y = project_feasible(h_y, n)
        grad = softmax_grad(h_y, n, alpha)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-12:

        h_new = project_feasible(h_y - lr * grad / gnorm * 0.01, n)
        h_prev = h_curr
        h_curr = h_new
        c = compute_c5(h_curr, n)

            best_h = h_curr.copy()
        t_k = (it + 1) / (it + 1 + 1.5)
        beta = t_k
    return best_h, best_c

    idx = np.argsort(lags)[-top_k:]

    def obj(x):

        a = c[idx]

        s = alpha * (a - m)
        mx = s.max()
        if mx > 100:
            return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(s))) / alpha)

    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]

def coord_descent(h, n, deadline, n_sweeps=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    v_grid = np.linspace(0, 1, 201)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            lags_now = compute_all_lags(h, n)
            a = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                m = k - (n - 1)
                if m == 0:
                    a[k] = 1.0 - 2.0 * h[i]  # approx linear (ignoring quadratic)
                elif m > 0:
                    if i < n - m:
                        a[k] = 1.0 - h[i + m]
                    if i >= m:
                        a[k] -= h[i - m]
                else:
                    ma = -m
                    if i >= ma:
                        a[k] = 1.0 - h[i - ma]
                    if i < n - ma:
                        a[k] -= h[i + ma]
            dx = 2.0 / n
            a *= dx
            b = lags_now - a * h[i]

        if time.time() < deadline - 60:
            h_t = slsqp_smooth(h_t, n_try, 50000, 5, 2000)
            c_t = compute_c5(h_t, n_try)
        if c_t < best_c:
            best_c, best_h = c_t, h_t
            n = n_try

    # Phase 5: Iterated restarts with coordinate descent
    n = len(best_h)
    n_restarts = max(1, int((deadline - time.time() - 40) / 50))
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        scale = 0.008 * (1.0 + 0.5 * r)
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        if time.time() < deadline - 20:
            hp, c_try = coord_descent(hp, n, time.time() + 20, n_sweeps=10)
        if time.time() < deadline - 8:
            hp = slsqp_smooth(hp, n, 50000, 5, 2000)
            c_try = compute_c5(hp, n)
        if c_try < best_c:
            best_c, best_h = c_try, hp

    # Phase 6: Final SLSQP with high alpha
    if time.time() < deadline - 5:
        best_h = slsqp_smooth(best_h, n, 500000, 10, 5000)
        best_c = compute_c5(best_h, n)

    # Phase 7: Final coordinate descent polish
    if time.time() < deadline - 2:
        best_h, best_c = coord_descent(best_h, n, deadline - 1, n_sweeps=5)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
