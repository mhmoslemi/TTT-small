import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _f_val(d, c_ks, s, um, has_self, si):
    vals = c_ks + d * s
    if has_self:
        vals = vals.copy()
        vals[si] -= d * d
    return max(um, float(np.max(vals)))

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n))
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * (2.0 / n))

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n = n_init
    # Phase 1: Primary optimization from initial
    alloc1 = min(350, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize_full(best_h, n, t0 + alloc1, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt
    # Phase 2: Try interpolating to different n
    x_orig = np.linspace(0, 2, n)
    for n_try in [200, 400, 500]:
        if time.time() > deadline - 200:
            break
        if n_try == n:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        alloc = min(150, (deadline - time.time()) * 0.25)
        if alloc < 60:
            break
        h_o, c_o = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_o < best_c:
            best_h, best_c, n = h_o, c_o, n_try
    # Phase 3: Multi-restart from best
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 60
        n_restart = max(2, int(remaining / 100))
        for r in range(n_restart):
            if time.time() > deadline - 70:
                break
            scale = 0.01 + 0.04 * ((r + 1) / max(1, n_restart))
            hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            alloc = min(80, (deadline - time.time() - 40) * 0.5)
            if alloc < 30:
                break
            h_o, c_o = optimize_full(hp, n_cur, time.time() + alloc, rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 4: Final high-alpha SLSQP + CD polish
    if time.time() < deadline - 25:
        n_final = len(best_h)
        h_f = slsqp_opt(best_h, n_final, alpha=5000000, maxiter=5000)
        c_f = compute_c5(h_f, n_final)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 8:
        n_final = len(best_h)
        h_cd, c_cd = cd_golden(best_h, n_final, max_passes=300, deadline=deadline - 3, rng=rng)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
