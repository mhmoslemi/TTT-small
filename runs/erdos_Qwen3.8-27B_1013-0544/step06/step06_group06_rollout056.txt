def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_bounded_simplex(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        k_start = max(0, p)
        k_end = min(2 * n - 2, p + n - 1)
        for k in range(k_start, k_end + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def apply_swap(g, h, corr, i, j, n):
    """Apply a single swap: h[i] 0->1, h[j] 1->0. Returns updated corr and g."""
    corr = corr + g[i] - g[j]
    p_idx = np.arange(n)
    mi = (p_idx != i) & (p_idx + i < 2 * n - 1)
    g[p_idx[mi], p_idx[mi] + i] -= 2.0
    mj = (p_idx != j) & (p_idx + j < 2 * n - 1)
    g[p_idx[mj], p_idx[mj] + j] += 2.0
    h[i], h[j] = 1.0, 0.0
    return g, h, corr
