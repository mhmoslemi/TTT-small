            if s < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    t = np.zeros(n, dtype=np.int64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def binary_snap(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def local_bin_search(h, n, deadline, top_k=15, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = binary_snap(h, n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur = corr.max() * dx
    best_c, best_h = cur, h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    temp = 0.005
    stall = 0
    while time.time() < deadline:
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        ti = np.argsort(corr)[-top_k:]
        sh = ti - (n - 1)
        ez = np.zeros((nz, top_k))
        for a in range(nz):
            i = zeros[a]
            for b in range(top_k):
                s = sh[b]
                v = 0.0
                if 0 <= i - s < n:
                    v += 1.0 - h[i - s]
                if 0 <= i + s < n:
                    v -= h[i + s]
                ez[a, b] = v
        eo = np.zeros((no, top_k))
        for a in range(no):
            j = ones[a]
            for b in range(top_k):
                s = sh[b]
                v = 0.0
                if 0 <= j - s < n:
                    v -= 1.0 - h[j - s]
                if 0 <= j + s < n:
                    v += h[j + s]
                eo[a, b] = v
        D = corr[ti][None, None, :] + ez[:, None, :] + eo[None, :, :]
        nm = D.max(axis=2)
        fl = nm.argmin()
        ai, cj = fl // no, fl % no
        i, j = zeros[ai], ones[cj]
        nc = nm[ai, cj] * dx
        if nc < cur - 1e-14:
            h[i], h[j] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur = corr.max() * dx
            stall = 0
            if cur < best_c:
                best_c, best_h = cur, h.copy()
            ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
        else:
            stall += 1
            if stall > 12:
                nk = min(2, nz, no)
                ri = rng.choice(nz, nk, replace=False)
                rj = rng.choice(no, nk, replace=False)
                sz, so = zeros[ri].copy(), ones[rj].copy()
                for a in range(nk):
                    h[zeros[ri[a]]] = 1.0
                    h[ones[rj[a]]] = 0.0
                c2 = np.correlate(h, 1 - h, mode='full').astype(float)
                nc2 = c2.max() * dx
                if rng.rand() < np.exp(-(nc2 - cur) / max(temp, 1e-4)):
                    cur = nc2
                    ones, zeros = np.where(h == 1)[0], np.where(h == 0)[0]
                else:
                    h[sz] = 0.0
                    h[so] = 1.0
                    corr = np.correlate(h, 1 - h, mode='full').astype(float)
                temp = max(1e-4, temp * 0.96)
                stall = 0
    return best_h, best_c

def slsqp_softmax(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sms = sh.max()
        return m + (sms + np.log(np.exp(sh - sms).sum())) / alpha
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_direct(h, n, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return float(c.max())
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def try_slsqp_chain(start, n, deadline, rng):
    """Run SLSQP with increasing alpha, chaining results."""
    best_h, best_c = start.copy(), compute_c5(start, n)
    alphas = [20000, 50000, 100000, 300000, 800000, 2000000]
    cur = start.copy()
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        rem = max(10, int((deadline - time.time()) / (len(alphas) + 1)))
        cur = slsqp_softmax(cur, n, alpha, max(200, rem * 150), deadline)
        c = compute_c5(cur, n)
        if c < best_c:
            best_c, best_h = c, cur.copy()
    # Also try direct max
    if time.time() < deadline - 5:
        cur2 = slsqp_direct(best_h, n, 2000, deadline)
        c2 = compute_c5(cur2, n)
        if c2 < best_c:
            best_c, best_h = c2, cur2.copy()
    return best_h, best_c
