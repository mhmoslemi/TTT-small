def active_objective(x, n, K, alpha):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1.0e100, np.zeros(n)
    # allow tiny out-of-bounds in objective? clip to [0,1] to keep f nonnegative but gradient uses clipped x.
    if x.min() < -1e-8 or x.max() > 1.0+1e-8:
        x = np.clip(x, 0.0, 1.0)
    fvals = f_values(x,n)
    m_all = fvals.size
    if K >= m_all:
       idx = np.arange(m_all)
    else:
       # top K by value; use argpartition
       idx = np.argpartition(fvals, m_all - K)[-K:]
    fsub=fvals[idx]
    m=fsub.max()
    z=alpha*(fsub-m)
    z-=z.max()
    e=np.exp(z)
    ssum=e.sum()
    if ssum <=0 or not finite:
       e=np.ones_like(e)/len(e); ssum=1.0
    ws=e/ssum
    F=m + np.log(ssum)/alpha
    Wplus=np.zeros(n); Wminus=np.zeros(n); w0=0.0
    # update weights; loop over K (max 499) fine

    if h.size != n: h=h[:n]
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    s=h.sum()
    if not np.isfinite(s) or s<0:
       return np.full(n,0.5)
    if abs(s-target) <= 1e-13:
       return h
    if s > target:
       if s > 0: h *= target/s
       else: h = np.full(n,0.5)
       h=np.clip(h,0.0,1.0)
       # after scaling may be exact, but fix residual
    else:
       for _ in range(30):
          s=h.sum(); diff=target-s
          if abs(diff) <= 1e-13: break
          room=1.0-h
          r=room.sum()
          if r < 1e-14:
             h*= target/s if s>0 else 1; h=np.clip(h,0,1); break
          h = h + (diff/r)*room
          h = np.clip(h,0.0,1.0)
    # residual fix
    diff=target - h.sum()
    if abs(diff) > 1e-12:

import time
import numpy as np
from scipy.optimize import minimize

_OPT_CACHE = {'valid': False, 'key': None, 'x': None, 'f': None, 'g': None}


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / float(n)))


def f_values_counts(x, n):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = x[:n]
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(x, out=pref[1:])
    autoc = np.correlate(x, x, mode='full')
    B = autoc[n - 1:n - 1 + n]
    s = np.arange(n, dtype=int)
    fpos = (pref[n - s] - B)
    fneg = (pref[n] - pref[s] - B)
    fvals = np.empty(2 * n - 1, dtype=float)
    fvals[0] = fpos[0]
    if n > 1:
        fvals[1::2] = fpos[1:]
        fvals[2::2] = fneg[1:]
    fvals = np.maximum(fvals, 0.0)
    return fvals


def active_objective(x, n, K, alpha):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = x[:n]
    if not np.all(np.isfinite(x)):
        return 1.0e100, np.zeros(n, dtype=float)
    if x.min() < -1.0e-10 or x.max() > 1.0 + 1.0e-10:
        x = np.clip(x, 0.0, 1.0)

    fvals = f_values_counts(x, n)
    m = fvals.size
    K = int(K)
    if K <= 0 or K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(fvals, m - K)[-K:]

    fsub = fvals[idx]
    mx = float(fsub.max())
    z = alpha * (fsub - mx)
    z -= float(z.max())
    e = np.exp(z)
    ssum = float(e.sum())
    if not np.isfinite(ssum) or ssum <= 0.0:
        e = np.ones_like(e)
        ssum = float(e.size)
    ws = e / ssum
    F = mx + np.log(ssum) / float(alpha)

    Wplus = np.zeros(n, dtype=float)
    Wminus = np.zeros(n, dtype=float)
    w0 = 0.0
    for ii, w in zip(idx, ws):
        w = float(w)
        if w < 1.0e-16:
            continue
        if ii == 0:
            w0 += w
        elif (ii & 1):
            Wplus[ii // 2 + 1] += w
        else:
            Wminus[ii // 2] += w

    g = np.zeros(n, dtype=float)
    if w0 != 0.0:
        g += w0 * (1.0 - 2.0 * x)

    g += np.cumsum(Wplus)[::-1]
    g += np.cumsum(Wminus)

    Wsum = Wplus + Wminus
    if float(Wsum.max()) > 1.0e-16:
        g -= np.convolve(x, Wsum, mode='full')[:n]
        g -= np.convolve(x[::-1], Wsum, mode='full')[:n][::-1]

    return float(F), g


def loss_cached(x, n, K, alpha):
    F, g = active_objective(x, n, K, alpha)
    _OPT_CACHE['valid'] = True
    _OPT_CACHE['key'] = (int(n), int(K), float(alpha))
    _OPT_CACHE['x'] = np.array(x, dtype=float, copy=True)
    _OPT_CACHE['f'] = F
    _OPT_CACHE['g'] = g
    return float(F)


def grad_cached(x, n, K, alpha):
    xarr = np.asarray(x, dtype=float)
    key = (int(n), int(K), float(alpha))
    if (_OPT_CACHE['valid'] and _OPT_CACHE['key'] == key and
            _OPT_CACHE['x'] is not None and xarr.shape == _OPT_CACHE['x'].shape):
        if np.array_equal(xarr, _OPT_CACHE['x']):
            return _OPT_CACHE['g']
    F, g = active_objective(xarr, n, K, alpha)
    _OPT_CACHE['valid'] = True
    _OPT_CACHE['key'] = key
    _OPT_CACHE['x'] = np.array(xarr, dtype=float, copy=True)
    _OPT_CACHE['f'] = F
    _OPT_CACHE['g'] = g
    return g


def sum_constraint(x, n, target):
    return float(np.sum(x) - float(target))


def sum_constraint_jac(x, n, target):
    return np.ones(n, dtype=float)


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        h = h[:n]
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if not np.isfinite(s) or s < 0.0:
        return np.full(n, 0.5, dtype=float)

    if abs(s - target) <= 1.0e-13:
        return h

    if s > target:
        if s > 0.0:
            h *= target / s
        else:
            h = np.full(n, 0.5, dtype=float)
        h = np.clip(h, 0.0, 1.0)
    else:
        for _ in range(30):
            s = float(h.sum())
            diff = target - s
            if abs(diff) <= 1.0e-13:
                break
            room = 1.0 - h
            r = float(room.sum())
            if r < 1.0e-14:
                if s > 0.0:
                    h *= target / s
                h = np.clip(h, 0.0, 1.0)
                break
            h = h + (diff / r) * room
            h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1.0e-12:
        if diff > 0.0:
            idxs = np.where(h < 1.0)[0]
        else:
            idxs = np.where(h > 0.0)[0]
        if idxs.size:
            add = diff / float(idxs.size)
            h[idxs] = np.clip(h[idxs] + add, 0.0, 1.0)
        diff = target - float(h.sum())
        if abs(diff) > 1.0e-12:
            if diff > 0.0:
                cand = np.where(h < 1.0)[0]
            else:
                cand = np.where(h > 0.0)[0]
            if cand.size:
                h[cand[0]] = float(np.clip(h[cand[0]] + diff, 0.0, 1.0))

    return h


def slsqp_refine(h0, n, K, alpha, maxiter):
    h0 = project_feasible(h0, n)
    target = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': sum_constraint,
             'jac': sum_constraint_jac,
             'args': (n, target)}]
    try:
        res = minimize(
            loss_cached,
            h0,
            args=(n, K, float(alpha)),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=grad_cached,
            options={'maxiter': int(maxiter), 'ftol': 1.0e-10, 'iprint': 0}
        )
    except Exception:
        return h0
    if res is not None and hasattr(res, 'x') and res.x is not None:
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    return h0


def default_h(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.15 * np.sin(3.0 * np.pi * x) + 0.07 * np.sin(7.0 * np.pi * x)
    return project_feasible(h, n)


def random_binary_h(n, rng):
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[:n // 2]] = 1.0
    return h


def smooth_random_h(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5, dtype=float)
    for _ in range(6):
        freq = float(rng.randint(1, 9))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.03, 0.22))
        h += amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)


def binarize_exact(v, n):
    v = np.asarray(v, dtype=float)
    if v.size != n:
        v = v[:n]
    order = np.argsort(v)[::-1]
    h = np.zeros(n, dtype=float)
    h[order[:n // 2]] = 1.0
    return h


def optimize_chain(start, n, deadline, alphas):
    cur = project_feasible(start, n)
    best = cur.copy()
    best_c = compute_c5(best, n)
    K = 2 * n - 1

    for alpha, base_it in alphas:
        now = time.time()
        if now > deadline - 75.0:
            break
        rem = deadline - now
        if rem < 130.0:
            break
        if rem < 300.0:
            it = min(int(base_it), 50)
        elif rem < 650.0:
            it = min(int(base_it), 130)
        else:
            it = int(base_it)
        cur = slsqp_refine(cur, n, K, float(alpha), max(10, it))
        if not np.all(np.isfinite(cur)):
            cur = best.copy()
        c = compute_c5(cur, n)
        if c < best_c:
            best = cur.copy()
            best_c = c
        if c > best_c + 0.01:
            cur = best.copy()

    return best, best_c
