import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, active_lags, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, 2) <= trust]
    for lag in active_lags:
        k = lag + (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy(), c
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), c
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy(), c
    return project_feasible(h_new, n), c

    if time.time() < deadline - 20:
        h_snap = binary_snap(best_h, n)
        h_new = slsqp_optimize(h_snap, n, 50000, None, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Main optimization at initial n
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_block(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    # Phase 2: Try alternative n values
    for n_try in [250, 200, 400]:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.2)
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_block(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: Perturbation restarts at best n
    n = len(best_h)
    restart_count = 0
    while time.time() < deadline - 70:
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        restart_count += 1
        scale = 0.008 + 0.004 * min(restart_count, 10)
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(80, remaining * 0.3)
        if alloc < 25:
            break
        h_opt, c_opt = optimize_block(hp, n, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 4: Final polish
    if time.time() < deadline - 5:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        h_new = slsqp_optimize(best_h, n, 1000000, top_idx, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
