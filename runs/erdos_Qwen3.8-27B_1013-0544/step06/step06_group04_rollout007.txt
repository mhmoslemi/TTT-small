def smooth_gradient(h,beta,n):
    c=np.correlate(h,1.0-h, mode='full')
    cm=c.max()
    if beta <=0: return c? no
    u=beta*(c-cm)
    expu=np.exp(u)  # max 1
    z=expu.sum()
    if z <=0: w=zeros; w[np.argmax(c)]=1
    else: w=expu/z
    # weights sum 1
    L=len(c)
    # no need normalize? weights sum1.
    A = np.convolve(h, w, mode='full')[n-1:n-1+n]
    Cr = np.convolve(h, w[::-1], mode='full')[n-1:n-1+n]
    pref=np.concatenate(([0.0], np.cumsum(w)))
    B = pref[n:n+n] - pref[:n]
    grad = B - A - Cr
    f = cm + np.log(z)/beta
    return grad, f
```

def continuous_refine(h,n,deadline,betas=None,max_iter_total=20000):
    h=project_box_sum(h,0.5*n)
    best_h=h.copy(); best_raw=raw_max(h); best_c=best_raw*2/n
    if betas is None: betas=[10,30,100,300,1000,3000]
    it_total=0
    for beta in betas:
       step=0.05
       stall=0
       while time.time()<deadline and it_total<max_iter_total and stall<80:
          grad,f0=smooth_gradient(h,beta,n)
          if np.linalg.norm(grad) < 1e-9: break
          accepted=False
          for mult in (1.0,0.5,0.25,0.1,0.05,0.02,0.01,0.005,0.002,0.001,0.0005):
             step_try=step*mult
             hnew=project_box_sum(h-step_try*grad,n*0.5)
             fnew=smooth_max_raw(hnew,beta)
             if fnew < f0 - 1e-12:
                 h=hnew; accepted=True; break
          if accepted:
             stall=0
             step=min(0.2, step*1.05)
             raw=raw_max(h)
             if raw*2/n < best_c - 1e-12:
                 best_c=raw*2/n; best_h=h.copy()
          else:
             stall+=1; step*=0.5
             if step<1e-7: break
          it_total+=1
    return best_h,best_c
```

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = float(target)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(6):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if mask.any():
            x[mask] += diff / float(mask.sum())
        x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        if len(idx):
            x[idx[0]] += diff
            x = np.clip(x, 0.0, 1.0)

def random_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def binarize(h, n):
    target = n // 2
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def single_delta(h, i, n, omh=None):
    if omh is None:
        omh = 1.0 - h
    L = 2 * n - 1
    d = np.zeros(L)
    d[i:i + n] += omh[::-1]
    d[n - 1 - i:n - 1 - i + n] -= h
    d[n - 1] = 0.0
    return d

def swap_delta_binary(h, i, j, n, omh=None):
    if omh is None:
        omh = 1.0 - h
    L = 2 * n - 1
    d = np.zeros(L)
    d[i:i + n] += omh[::-1]
    d[n - 1 - i:n - 1 - i + n] -= h
    d[j:j + n] -= omh[::-1]
    d[n - 1 - j:n - 1 - j + n] += h
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    d[n - 1] = 0.0
    return d

def greedy_binary(n):
    h = np.zeros(n)
    corr = np.zeros(2 * n - 1)
    target = n // 2
    for _ in range(target):
        omh = 1.0 - h
        best_i = -1
        best_d = None
        best_c = np.inf
        best_s = np.inf
        for i in np.where(h == 0.0)[0]:
            d = single_delta(h, int(i), n, omh)
            nc = float((corr + d).max())
            ns = float((corr + d).sum())
            if nc < best_c - 1e-12 or (abs(nc - best_c) < 1e-12 and ns < best_s - 1e-10):
                best_c = nc
                best_s = ns
                best_i = int(i)
                best_d = d
        if best_i < 0:
            break
        h[best_i] = 1.0
        corr = corr + best_d
    return h

def binary_sa(h, n, deadline, seed=0, t0=0.32, t1=0.012, max_iter=40000):
    h = binarize(np.asarray(h, dtype=float), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur = float(corr.max())
    best_c = cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    it = 0
    denom = max(1, max_iter - 1)
    while time.time() < deadline and it < max_iter:
        T = t0 * (t1 / t0) ** (it / denom)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        omh = 1.0 - h
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        d = swap_delta_binary(h, i, j, n, omh)
        nc = float((corr + d).max())
        acc = False
        if nc < cur - 1e-12:
            acc = True
        else:
            gap = nc - cur
            if 0.0 < gap < 2.5 and T > 1e-9 and rng.random() < np.exp(-gap / T):
                acc = True
        if acc:
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + d
            cur = nc
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        it += 1
    return best_h, best_c * (2.0 / float(n))

def binary_local_search(h, n, deadline, first_improve=True, max_sweeps=120, seed=0):
    h = binarize(np.asarray(h, dtype=float), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur = float(corr.max())
    best_c = cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_sweeps):
        if time.time() >= deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        omh = 1.0 - h
        improved = False
        best_d = None
        best_i = best_j = -1
        best_m = cur
        if first_improve:
            for ii in zeros:
                found = False
                i = int(ii)
                for jj in ones:
                    j = int(jj)
                    d = swap_delta_binary(h, i, j, n, omh)
                    nc = float((corr + d).max())
                    if nc < cur - 1e-12:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = corr + d
                        cur = nc
                        improved = True
                        found = True
                        if cur < best_c:
                            best_c = cur
                            best_h = h.copy()
                        break
                if found:
                    break
        else:
            for ii in zeros:
                i = int(ii)
                for jj in ones:
                    j = int(jj)
                    d = swap_delta_binary(h, i, j, n, omh)
                    nc = float((corr + d).max())
                    if nc < best_m - 1e-12:
                        best_m = nc
                        best_d = d
                        best_i = i
                        best_j = j
            if best_d is not None:
                h[best_i] = 1.0
                h[best_j] = 0.0
                corr = corr + best_d
                cur = best_m
                improved = True
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c * (2.0 / float(n))

def smooth_max_raw(h, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    if beta <= 1e-8:
        return cm
    u = beta * (c - cm)
    e = np.exp(u)
    z = float(e.sum())
    if z <= 0.0:
        return cm
    return cm + math.log(z) / beta

def smooth_gradient(h, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    u = beta * (c - cm)
    e = np.exp(u)
    z = float(e.sum())
    if z <= 0.0:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / z
    n = len(h)
    A = np.convolve(h, w, mode='full')[n - 1:n - 1 + n]
    Cr = np.convolve(h, w[::-1], mode='full')[n - 1:n - 1 + n]
    pref = np.concatenate(([0.0], np.cumsum(w)))
    B = pref[n:n + n] - pref[:n]
    grad = B - A - Cr
    f = cm + math.log(max(z, 1e-300)) / beta
    return grad, f

def continuous_refine(h, n, deadline, betas=None, max_iter=8000):
    h = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    best_c = c5_exact(h, n)
    best_h = h.copy()
    if betas is None:
        betas = [12, 40, 150, 600, 2500]
    it = 0
    for beta in betas:
        if time.time() >= deadline or it >= max_iter:
            break
        step = 0.06
        stall = 0
        while time.time() < deadline and it < max_iter and stall < 120:
            grad, f0 = smooth_gradient(h, beta)
            if float(np.sqrt(max(0.0, float(np.dot(grad, grad))))) < 1e-10:
                break
            accepted = False
            for mult in (1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001):
                st = step * mult
                hnew = project_box_sum(h - st * grad, 0.5 * n)
                fnew = smooth_max_raw(hnew, beta)
                if fnew < f0 - 1e-12:
                    h = hnew
                    accepted = True
                    break
            if accepted:
                it += 1
                stall = 0
                step = min(0.18, step * 1.08)
                c = c5_exact(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
            else:
                stall += 1
                step *= 0.5
                if step < 1e-8:
                    break
    return best_h, best_c

def candidate_ns(max_n=130, limit=14, goal=0.3808):
    out = []
    for n in range(48, max_n + 1, 2):
        T = int(math.floor(goal * n * 0.5 - 1e-9))
        if T > 0:
            c = 2.0 * T / n
            if c < goal - 1e-12:
                out.append((c, n, T))
    out.sort()
    return out[:limit]

def milp_feasible_binary(n, T, time_limit=40):
    if not HAVE_MILP:
        return None
    if T <= 0:
        return None
    req = []
    for k in range(-(n - 1), n):
        cnt = n - abs(k)
        if cnt > T:
            req.append((k, cnt))
    P = sum(c for _, c in req)
    if P == 0:
        h = random_balanced(n, np.random.RandomState(1234))
        return h if raw_max(h) <= T + 1e-6 else None
    z0 = n
    rows = len(req) + 3 * P
    num_vars = n + P
    A = lil_matrix((rows, num_vars), dtype=float)
    b = np.empty(rows)
    pairs = []
    z = z0
    r = 0
    for k, cnt in req:
        start = z
        if k >= 0:
            for j in range(cnt):
                i = j + k
                pairs.append((z, i, j))
                z += 1
        else:
            m = -k
            for j in range(m, n):
                i = j - m
                pairs.append((z, i, j))
                z += 1
        for cc in range(start, z):
            A[r, cc] = 1.0
        b[r] = float(T)
        r += 1
    for zz, i, j in pairs:
        A[r, zz] = 1.0
        A[r, i] = -1.0
        b[r] = 0.0
        r += 1
        A[r, zz] = 1.0
        A[r, j] = 1.0
        b[r] = 1.0
        r += 1
        A[r, zz] = -1.0
        A[r, i] = -1.0
        A[r, j] = 1.0
        b[r] = 0.0
        r += 1
    A_eq = lil_matrix((1, num_vars), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    try:
        cons = [LinearConstraint(csc_matrix(A), -np.inf, b),
                LinearConstraint(csc_matrix(A_eq), b_eq, b_eq)]
        integ = np.zeros(num_vars)
        integ[:n] = 1
        bd = Bounds(np.zeros(num_vars), np.ones(num_vars))
        res = milp(np.zeros(num_vars), constraints=cons, integrality=integ,
                   bounds=bd, options={'time_limit': int(max(1, time_limit))})
    except Exception:
        try:
            cons = [LinearConstraint(csc_matrix(A), -np.inf, b),
                    LinearConstraint(csc_matrix(A_eq), b_eq, b_eq)]
            integ = np.zeros(num_vars)
            integ[:n] = 1
            bd = Bounds(np.zeros(num_vars), np.ones(num_vars))
            res = milp(np.zeros(num_vars), constraints=cons, integrality=integ, bounds=bd)
        except Exception:
            return None
    if res is None or getattr(res, "x", None) is None:
        return None
    x = np.asarray(res.x, dtype=float)[:n]
    h = (x >= 0.5).astype(float)
    target = n // 2
    if h.sum() != target:
        if h.sum() > target:
            for idx in np.where(h == 1.0)[0]:
                h[idx] = 0.0
                if h.sum() == target:
                    break
        else:
            for idx in np.where(h == 0.0)[0]:
                h[idx] = 1.0
                if h.sum() == target:
                    break
    return h if raw_max(h) <= T + 1e-5 else None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(20.0, float(budget_s)) - 5.0
    rng = np.random.RandomState(seed)
    try:
        parr = np.asarray(initial_h_values, dtype=float)
        pn = len(parr)
        if pn < 4:
            raise ValueError
    except Exception:
        pn = 250
        parr = random_balanced(pn, rng)
    ph = project_box_sum(parr, 0.5 * pn)
    best_h = ph.copy()
    best_c = c5_exact(ph, pn)
    best_n = pn
    goal = 0.3808
    cands = candidate_ns(130, 16, goal)
    small_ns = []
    for _, n, _ in cands[:7]:
        if n % 2 == 0 and n not in small_ns:
            small_ns.append(n)
    p1 = t0 + 0.18 * budget_s
    for n in small_ns:
        if time.time() > p1 - 60:
            break
        seeds = [greedy_binary(n), random_balanced(n, rng)]
        for si, hs in enumerate(seeds):
            if time.time() > p1 - 20:
                break
            d1 = min(p1 - 20, time.time() + 4.5)
            hb, c1 = binary_sa(hs, n, d1, seed=seed + si * 17 + 3)
            d2 = min(p1 - 14, time.time() + 7.0)
            hb2, c2 = binary_local_search(hb, n, d2, first_improve=(n > 110), max_sweeps=40,
                                          seed=seed + si * 31 + 9)
            if c2 < best_c:
                best_c, best_h, best_n = c2, hb2.copy(), n
            if c2 <= goal + 0.025 and time.time() < p1 - 8:
                d3 = min(p1 - 5, time.time() + 7.0)
                hc, c3 = continuous_refine(hb2, n, d3, betas=[25, 120, 700], max_iter=2500)
                if c3 < best_c:
                    best_c, best_h, best_n = c3, hc.copy(), n
    p2 = t0 + 0.46 * budget_s
    if not HAVE_MILP:
        p2 = max(p1 + 1.0, p2 * 0.4)
    if HAVE_MILP:
        done = set([best_n])
        for cgoal, n, T in cands:
            if time.time() > p2 - 45:
                break
            if n in done:
                continue
            tl = min(48, max(9, int(p2 - time.time() - 18)))
            hm = milp_feasible_binary(n, T, tl)
            done.add(n)
            if hm is None:
                continue
            d1 = min(p2 - 16, time.time() + 16.0)
            hb, c1 = binary_local_search(hm, n, d1, first_improve=(n > 110), max_sweeps=60,
                                         seed=seed + 1013 + n)
            if c1 < best_c:
                best_c, best_h, best_n = c1, hb.copy(), n
            d2 = min(p2 - 5, time.time() + 22.0)
            hc, c2 = continuous_refine(hb, n, d2, betas=[50, 250, 1400], max_iter=4500)
            if c2 < best_c:
                best_c, best_h, best_n = c2, hc.copy(), n
    p3 = t0 + 0.80 * budget_s
    if time.time() < p3 and pn % 2 == 0:
        d = min(p3, time.time() + 75.0)
        hc, c = continuous_refine(ph.copy(), pn, d, betas=[18, 75, 350, 2000], max_iter=6000)
        if c < best_c:
            best_c, best_h, best_n = c, hc.copy(), pn
    if time.time() < p3 and pn % 2 == 0:
        d = min(p3 - 20, time.time() + 45.0)
        hb, c = binary_local_search(binarize(ph, pn), pn, d, first_improve=True, max_sweeps=180,
                                    seed=seed + 2027)
        if c < best_c:
            best_c, best_h, best_n = c, hb.copy(), pn
    if time.time() < p3 and pn % 2 == 0:
        d = min(p3 - 8, time.time() + 35.0)
        hc, c = continuous_refine(hb, pn, d, betas=[80, 500, 2500], max_iter=4000)
        if c < best_c:
            best_c, best_h, best_n = c, hc.copy(), pn
    if time.time() < p3:
        hr = ph + 0.018 * rng.randn(pn)
        hr = project_box_sum(hr, 0.5 * pn)
        d = min(p3 - 5, time.time() + 30.0)
        hc, c = continuous_refine(hr, pn, d, betas=[70, 400, 2500], max_iter=3500)
        if c < best_c:
            best_c, best_h, best_n = c, hc.copy(), pn
    if best_n % 2 == 0 and time.time() < deadline - 70 and best_c > goal - 1e-8:
        d1 = min(deadline - 50, time.time() + 45.0)
        hb, c = binary_local_search(binarize(best_h, best_n), best_n, d1, first_improve=True,
                                    max_sweeps=120, seed=seed + 3331)
        if c < best_c:
            best_c, best_h, best_n = c, hb.copy(), best_n
    if time.time() < deadline - 15:
        d2 = deadline - 5
        hc, c = continuous_refine(best_h, best_n, d2, betas=[150, 900, 4000], max_iter=8000)
        if c < best_c:
            best_c, best_h, best_n = c, hc.copy(), best_n
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = c5_exact(best_h, best_n)
    return best_h, best_c, best_n
