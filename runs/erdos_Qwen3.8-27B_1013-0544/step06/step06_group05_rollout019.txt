def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {h : 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def binary_hill_climb(h, n, deadline, max_iters=20000):
    """Fast binary local search using random + greedy swaps."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a batch of random swaps
        n_try = min(100, len(ones) * len(zeros))
        best_delta = 0
        best_i, best_j = -1, -1
        
        for _ in range(n_try):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])

            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')

            delta = new_max - cur_max

            ones = np.where(h == 1)[0]

            no_improve = 0
        else:

                break

def slsqp_refine(h, n, deadline):

    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)

            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + float(sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        try:

            if new_c < best_c:
                best_c = new_c
                best_h = new_h

            pass
    
    return best_h, best_c
