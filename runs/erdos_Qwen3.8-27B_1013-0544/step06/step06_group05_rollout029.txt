def simulated_annealing(h, n, deadline, t_start=10.0, t_end=0.01, max_iter=1000000):
    corr = compute_corr(h, n)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    n_zeros = len(zeros)
    n_ones = len(ones)
    
    lags = np.arange(2*n-1) - (n-1)
    rng = np.random.RandomState(42)
    
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Pick random swap
        i = zeros[rng.randint(n_zeros)]
        j = ones[rng.randint(n_ones)]
        
        # Compute delta in corr (O(n))
        # Δcorr[k] = (1-h'[i+lag]) - (1-h'[j+lag]) - h[i-lag] + h[j-lag]
        # where h' is after swap
        
        idx_ip = i + lags
        idx_jp = j + lags
        idx_im = i - lags
        idx_jm = j - lags
        
        # 1 - h'[i+lag]: if i+lag == j, h'[j]=0 so 1-h'=1. Otherwise 1-h[i+lag] (if in bounds) or 1 (if out)
        one_minus_hip = np.ones(2*n-1)
        vp = (idx_ip >= 0) & (idx_ip < n)
        one_minus_hip[vp] = 1.0 - h[idx_ip[vp]]
        if i + (j - i) == j:  # always true, this is just i+lag=j when lag=j-i
            pass
        # Handle interaction: if idx_ip[k] == j, then h'[j] = 0, so 1-h' = 1
        inter_ip = (idx_ip == j)
        one_minus_hip[inter_ip] = 1.0
        
        # 1 - h'[j+lag]: if j+lag == i, h'[i]=1 so 1-h' = 0. Otherwise 1-h[j+lag] or 1
        one_minus_hjp = np.ones(2*n-1)
        vt = (idx_jp >= 0) & (idx_jp < n)
        one_minus_hjp[vt] = 1.0 - h[idx_jp[vt]]
        inter_jp = (idx_jp == i)
        one_minus_hjp[inter_jp] = 0.0
        
        h_im = np.zeros(2*n-1)
        vm = (idx_im >= 0) & (idx_im < n)
        h_im[vm] = h[idx_im[vm]]
        
        h_jm = np.zeros(2*n-1)
        vm2 = (idx_jm >= 0) & (idx_jm < n)
        h_jm[vm2] = h[idx_jm[vm2]]
        
        delta = one_minus_hip - one_minus_hjp - h_im + h_jm
        new_corr = corr + delta
        new_max = new_corr.max()
        
        # Accept/reject
        if new_max < cur_max or rng.rand() < np.exp(-(new_max - cur_max) / t):
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            # Update zeros and ones
            zeros[list(zeros).index(i)] = j  # replace i with j in zeros
            ones[list(ones).index(j)] = i    # replace j with i in ones
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        
        t *= cooling
    
    return best_h, float(best_max * (2.0/n))

**Key insight for continuous optimization:**

The objective is max_k corr_k where corr_k = sum_i h[i]*(1-h[i+k])*dx.

For a given h, let L be the set of "critical" lags where corr_k is close to the maximum. We want to reduce corr_k for all k in L simultaneously.

The gradient of corr_k with respect to h[i] is:
∂corr_k/∂h[i] = (1-h[i+k]) - h[i-k]  (with appropriate boundary handling)

The gradient of the smooth max (log-sum-exp) is:
∂S/∂h[i] = sum_k w_k * ((1-h[i+k]) - h[i-k])
where w_k = exp(alpha*corr_k) / sum_j exp(alpha*corr_j)

This is a weighted average of the individual gradients, weighted by how "critical" each lag is.

The projected gradient step is:
h_new = h - lr * grad
h_projected = project_simplex(h_new, n/2)

The key parameters are:
- alpha: controls how "peaked" the softmax is (higher = more focused on critical lags)
- lr: learning rate
- Number of iterations

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h, n)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def project_simplex(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h, n)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def project_simplex(h, target_sum):
    lo, hi = -1.0, 2.0

def projected_gradient(h, n, deadline, alpha_start=100, alpha_end=50000, max_iter=5000):

    best_h = h.copy()

def continuous_optimize(h, n, deadline, alpha=1000):
    from scipy.optimize import minimize
    
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def grad(x):
        return grad_smooth_max(x, n, alpha)
    
    def constraint_val(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint_val, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-18})
        return res.x, compute_c5(res.x, n)
    except Exception:
        return h, compute_c5(h, n)

import numpy as np
import time

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def project_simplex(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, min_weight=1e-10):
    dx = 2.0 / n
    corr = compute_corr(h, n)
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    
    active = weights >= min_weight
    if not active.any():
        return grad
    
    for k in np.where(active)[0]:
        w = weights[k]
        lag = lags[k]
        idx_p = np.arange(n) + lag
        idx_m = np.arange(n) - lag
        vp = (idx_p >= 0) & (idx_p < n)
        vm = (idx_m >= 0) & (idx_m < n)
        val_p = np.zeros(n)
        val_p[vp] = 1.0 - h[idx_p[vp]]
        val_m = np.zeros(n)
        val_m[vm] = h[idx_m[vm]]
        grad += w * (val_p - val_m)
    
    return grad * dx

def obj_value(h, n, alpha):
    dx = 2.0 / n
    corr = compute_corr(h, n) * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    log_alpha_start = np.log(max(alpha_start, 1.0))

    stale = 0

        if it % 200 == 0 and it > 0:

        frac = min(it / max(max_iters, 1), 1.0)

        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
