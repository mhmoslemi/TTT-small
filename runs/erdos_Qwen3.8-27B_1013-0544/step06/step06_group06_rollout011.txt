def exact_swap_step(h, F, n, recompute):
    off=n-1; L=2*n-1; cur=int(np.max(F))
    ones=np.flatnonzero(h==1)
    zeros=np.flatnonzero(h==0)

    B=compute_B_binary(h,n)
    n1=ones.size; nz=zeros.size
    best_new=cur; best_i=-1; best_j=-1
    # dynamic block
    denom=max(1, n1*L)
    block=int(2000000 // denom)
    block=max(4, min(64, block))
    rows=None; cols=None
    for s in range(0,nz,block):
       J=zeros[s:s+block]; nb=J.size
       if nb==0: continue
       D=F[None,None,:] + B[J,None,:] - B[ones,None,:]
       if D.dtype != np.int16: D=D.astype(np.int16, copy=False)
       d=J[:,None] - ones[None,:]
       idxp=off + d
       idxn=off - d
       rows=np.arange(nb)[:,None]
       cols=np.arange(n1)[None,:]

import numpy as np
import time


def compute_c5(h, n):
    if n <= 0:
        return float("inf")
    dx = 2.0 / n
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_F_binary(h, n):
    a = h.astype(np.int16)
    b = (1 - a).astype(np.int16)
    return np.correlate(a, b, mode="full").astype(np.int16)


def binarize(values, n):
    y = np.asarray(values, dtype=float)
    if y.size != n:
        x_old = np.linspace(0.0, 1.0, y.size)
        x_new = np.linspace(0.0, 1.0, n)
        y = np.interp(x_new, x_old, y)
    target = n // 2
    h = (y >= 0.5).astype(np.int8)
    s = int(h.sum())
    if s > target:
        ones = np.flatnonzero(h == 1)
        order = np.argsort(y[ones])
        h[ones[order[:s - target]]] = 0
    elif s < target:
        zeros = np.flatnonzero(h == 0)
        order = np.argsort(-y[zeros])
        h[zeros[order[:target - s]]] = 1
    return h


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-14:
                break
            h = h + diff * room / s
        else:
            s = h.sum()
            if s <= 1e-14:
                break
            h = h + diff * h / s
        h = np.clip(h, 0.0, 1.0)
    return h


def project_polytope(y, target):
    lo = -float(np.max(y))
    hi = 1.0 - float(np.min(y))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)


def random_perm_init(n, rng):
    h = np.zeros(n, dtype=np.int8)
    p = rng.permutation(n)
    h[p[:n // 2]] = 1
    return h


def random_block_init(n, rng):
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    pos = 0
    placed_ones = 0
    val = int(rng.randint(0, 2))
    max_run = min(6, max(2, n // 8))
    while pos < n:
        if val == 1 and placed_ones >= target:
            val = 0
        zeros_placed = pos - placed_ones
        if val == 0 and zeros_placed >= target:
            val = 1
        l = int(rng.choice([1, 2, 2, 3, 3, 4, 5, max_run]))
        if l > n - pos:
            l = n - pos
        if val == 1:
            l = min(l, target - placed_ones)
        else:
            l = min(l, target - (pos - placed_ones))
        if l <= 0:
            val = 1 - val
            continue
        h[pos:pos + l] = val
        pos += l
        if val == 1:
            placed_ones += l
        val = 1 - val
    return h


def thue_morse_init(n):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t


def compute_B_binary(h, n):
    L = 2 * n - 1
    off = n - 1
    B = np.zeros((n, L), dtype=np.int16)
    for q in range(1, n):
        sl = n - q
        hp = off + q
        hn = off - q
        B[:sl, hp] = -h[q:n].astype(np.int16)
        B[q:n, hp] += (1 - h[:sl]).astype(np.int16)
        B[:sl, hn] = (1 - h[q:n]).astype(np.int16)
        B[q:n, hn] += -h[:sl].astype(np.int16)
    return B


def exact_swap_step(h, F, n, recompute):
    off = n - 1
    cur = int(np.max(F))
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    if ones.size == 0 or zeros.size == 0:
        return h, F, False

    B = compute_B_binary(h, n)
    n1 = ones.size
    nz = zeros.size
    L = 2 * n - 1

    best_new = cur
    best_i = -1
    best_j = -1

    denom = max(1, n1 * L)
    block = int(2000000 // denom)
    block = max(4, min(64, block))

    for s in range(0, nz, block):
        J = zeros[s:s + block]
        nb = J.size
        if nb == 0:
            continue
        D = F[None, None, :] + B[J, None, :] - B[ones, None, :]
        d = J[:, None] - ones[None, :]
        idxp = off + d
        idxn = off - d
        rows = np.arange(nb)[:, None]
        cols = np.arange(n1)[None, :]
        np.add.at(D, (rows, cols, idxp), 1)
        np.add.at(D, (rows, cols, idxn), 1)
        M = D.max(axis=2)
        flat = int(M.argmin())
        val = int(M.flat[flat])
        if val < best_new:
            best_new = val
            r = flat // n1
            c = flat % n1
            best_j = int(J[r])
            best_i = int(ones[c])

    if best_i < 0 or best_new >= cur:
        return h, F, False

    F_old = F
    i = best_i
    j = best_j
    h[i] = 0
    h[j] = 1

    if recompute:
        F_new = compute_F_binary(h, n)
    else:
        F_new = F_old + B[j] - B[i]
        F_new[off + j - i] += 1
        F_new[off + i - j] += 1
        if int(np.max(F_new)) >= cur or int(np.min(F_new)) < 0 or int(np.max(F_new)) > n:
            F_new = compute_F_binary(h, n)

    if int(np.max(F_new)) >= cur:
        h[i] = 1
        h[j] = 0
        return h, F_old, False

    return h, F_new, True


def random_multi_swap(h, n, k, rng):
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    if k == 0 or ones.size < k or zeros.size < k:
        return False
    ri = rng.choice(ones.size, k, replace=False)
    rj = rng.choice(zeros.size, k, replace=False)
    h[zeros[rj]] = 1
    h[ones[ri]] = 0
    return True


def binary_search(h_start, n, deadline, rng):

    F = compute_F_binary(h, n)
    c = float(np.max(F) * (2.0 / n))
    best_h = h.copy()
    best_c = c
    moves = 0

    if n < 128:

def lse_value(x, n, alpha):
    dx = 2.0 / n
    G = np.correlate(x, 1.0 - x, mode="full").astype(float) * dx
    m = float(G.max())
    z = alpha * (G - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / alpha


def lse_value_and_grad(x, n, alpha):
    dx = 2.0 / n
    off = n - 1
    G = np.correlate(x, 1.0 - x, mode="full").astype(float) * dx
    m = float(G.max())
    z = alpha * (G - m)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(G)
        w[np.argmax(G)] = 1.0
    else:
        w = e / s
    val = m + float(np.log(s)) / alpha

    grad = np.zeros(n, dtype=float)
    w0 = float(w[off]) * dx
    if abs(w0) > 1e-18:
        grad += (1.0 - 2.0 * x) * w0

    for q in range(1, n):
        sl = n - q
        wp = float(w[off + q]) * dx
        if abs(wp) > 1e-18:
            grad[:sl] -= x[q:] * wp
            grad[q:n] += (1.0 - x[:sl]) * wp
        wn = float(w[off - q]) * dx
        if abs(wn) > 1e-18:
            grad[:sl] += (1.0 - x[q:]) * wn
            grad[q:n] -= x[:sl] * wn

    return val, grad

            break
        h = project_polytope(h, target)

            cv = compute_c5(h, n)
            if cv < best_c:
                best_c = cv
                best_h = h.copy()
                if n % 2 == 0:
                    hb = binarize(h, n)
                    cb = compute_c5(hb.astype(float), n)
                    if cb < best_c:
                        best_c = cb
                        best_h = hb.astype(float)

    return best_h, best_c


def make_starts(n, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent)
        pp = np.asarray(parent, dtype=float)
        if pp.size == n:
            y = pp
        else:
            y = np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, pp.size), pp)
        starts.append(np.clip(y + 0.08 * rng.randn(n), 0.0, 1.0))
    starts.append(random_block_init(n, rng))
    if n <= 256:
        starts.append(random_perm_init(n, rng))
    starts.append(thue_morse_init(n))
    starts.append(random_block_init(n, rng))
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(60.0, float(budget_s) - 8.0)
    deadline = t0 + B
    rng = np.random.RandomState(seed)

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float)
        if pv.ndim == 1 and pv.size >= 8:
            orig = pv.size
            if orig > 512:
                pv = np.interp(np.linspace(0.0, 1.0, 512), np.linspace(0.0, 1.0, orig), pv)
            parent = project_feasible(pv, pv.size)
    except Exception:
        parent = None

    if parent is None:
        pn = 250
        parent = project_feasible(np.full(pn, 0.5), pn)

    best_h = parent.copy()
    best_c = compute_c5(best_h, best_h.size)

    quick_deadline = t0 + 0.035 * B
    if time.time() < quick_deadline and best_h.size <= 512:
        hb, cb = continuous_polish(best_h, best_h.size, quick_deadline, [60, 250], rng)
        if cb < best_c:
            best_c = cb
            best_h = hb

    base_ns = (64, 96, 128, 160, 192, 224, 256, 320, 384)
    ns = []
    for v in base_ns:
        if v not in ns:
            ns.append(v)
    pn = int(parent.size)
    if pn % 2 == 0 and 32 <= pn <= 384:
        if pn not in ns:
            ns.append(pn)
    ns.sort()

    total_bin = 0.60 * B
    binary_deadline = t0 + total_bin
    weights = [1.0 + 128.0 / float(v) for v in ns]
    wsum = float(sum(weights))
    cur = t0

    for n, w in zip(ns, weights):
        if time.time() >= binary_deadline:
            break
        slice_len = total_bin * (w / wsum)
        n_deadline = min(binary_deadline, cur + max(4.0, slice_len))
        if time.time() >= n_deadline:
            cur = max(cur, n_deadline)
            continue

        starts = make_starts(n, parent, rng)
        for idx, st in enumerate(starts):
            now = time.time()
            if now >= n_deadline:
                break
            remaining = max(1, len(starts) - idx)
            chunk = max(3.0, (n_deadline - now) / float(remaining))
            start_deadline = min(n_deadline, now + chunk)
            hb, cb = binary_search(st, n, start_deadline, rng)
            if cb < best_c:
                best_c = cb
                best_h = hb
        cur = max(cur, n_deadline)

    cont_deadline = t0 + 0.82 * B
    if time.time() < cont_deadline and best_h.size <= 512:
        hb, cb = continuous_polish(best_h, best_h.size, cont_deadline, [120, 500, 2000], rng)
        if cb < best_c:
            best_c = cb
            best_h = hb

    final_binary_deadline = min(deadline - 2.0, time.time() + 0.10 * B)
    if best_h.size % 2 == 0 and time.time() < final_binary_deadline - 2.0:
        hb, cb = binary_search(best_h, best_h.size, final_binary_deadline, rng)
        if cb < best_c:
            best_c = cb
            best_h = hb

    final_cont_deadline = deadline - 2.0
    if time.time() < final_cont_deadline - 2.0 and best_h.size <= 512:
        hb, cb = continuous_polish(best_h, best_h.size, final_cont_deadline, [3000], rng)
        if cb < best_c:
            best_c = cb
            best_h = hb

    best_h = project_feasible(best_h, best_h.size)
    best_c = compute_c5(best_h, best_h.size)
    return best_h.astype(float), float(best_c), int(best_h.size)
