def thue_morse(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    if t.sum() > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:t.sum() - target]] = 0
    elif t.sum() < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - t.sum()]] = 1
    return t.astype(float)

def soft_max_obj_grad(h, n, alpha):

    grad *= dx
    return obj, grad
