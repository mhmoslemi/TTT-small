    """Compute new correlation after swapping h[i]=1->0 and h[j]=0->1.
    Returns the change in corr (array of length 2n-1)."""

    # j is the position turning on (was 0, now 1)
    delta = np.zeros(2*n-1)
    # For each lag k (0 to 2n-2, where lag = k-(n-1)):
    # Change from turning off i:
    #   -(1-h[i-k]) at position where i-k is valid (i.e., 0<=i-k<n)
    #   +h[i+k] at position where i+k is valid

    #   +(1-h[j-k]) at position where j-k is valid
    #   -h[j+k] at position where j+k is valid

    # k ranges from -(n-1) to n-1
    
    # Turning off i:
    # For lag k: contribution is at index n-1+k
    # -(1-h[i-k]) when 0<=i-k<n, i.e., i-n+1<=k<=i, i.e., k in [i-(n-1), i]

        delta[idx] += h[i+k]

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def steepest_descent(h, n, deadline, max_sweeps=100):
    """Try all single-swap moves, accept the best improvement."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = get_corr(h, n)
    cur_max = corr.max()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_improvement = 0
        best_pair = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                # Try swap
                h[i] = 0
                h[j] = 1
                new_corr = get_corr(h, n)
                new_max = new_corr.max()
                if cur_max - new_max > best_improvement:
                    best_improvement = cur_max - new_max
                    best_pair = (i, j)
                    best_corr = new_corr.copy()
                h[i] = 1
                h[j] = 0
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            corr = best_corr
            cur_max = corr.max()
        else:
            break
    
    return h, float(cur_max)

def simulated_annealing(h, n, deadline, T_start=5.0, T_end=0.01, n_iter=None):
    """SA with swap moves."""
    corr = get_corr(h, n)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    if n_iter is None:
        # Estimate how many iterations we can do
        # Each iteration: ~100μs (correlation + max + random pick)
        # Available time: deadline - now
        avail = deadline - time.time()
        n_iter = int(avail * 5000)  # ~5000 iterations per second
    
    rng = np.random.RandomState(42)
    T = T_start
    cool_rate = (T_end / T_start) ** (1.0 / n_iter)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0
        h[j] = 1
        new_corr = get_corr(h, n)
        new_max = new_corr.max()
        
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
            corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[i] = 1
            h[j] = 0
        
        T *= cool_rate
    
    return best_h, float(best_max)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-11:

        return None  # Let SLSQP use numerical gradients

Let me verify: 
- Position i was 1, now 0. As the "first" element h[p] in corr, it was contributing h[i]*(1-h[i-k]) = 1-h[i-k], now contributes 0. Change: -(1-h[i-k]).
- Position i was 1, now 0. As the "second" element (1-h[q]) where q=i, i.e., p-k=i, p=i+k. It was contributing h[i+k]*(1-h[i]) = h[i+k]*0 = 0, now contributes h[i+k]*1 = h[i+k]. Change: +h[i+k].
- Position j was 0, now 1. As the "first" element h[p], it was contributing 0*(1-h[j-k]) = 0, now contributes 1*(1-h[j-k]) = 1-h[j-k]. Change: +(1-h[j-k]).

    """Compute change in corr array when swapping h[i]:1->0 and h[j]:0->1.
    h is the OLD (before swap) values."""
    # We need to compute the change at each lag
    # Δcorr[n-1+k] for all k
    delta = np.zeros(2*n-1)
    
    # For each k, compute the four terms
    # k ranges from -(n-1) to n-1
    # Index in corr: n-1+k
    
    # Term 1: -(1-h[i-k]) for valid i-k (0<=i-k<n)
    # k range: i-(n-1) <= k <= i
    k_lo = max(-(n-1), -(i) + 0)  # i-k >= 0 => k <= i
    k_hi = min(n-1, i)
    k_lo2 = max(-(n-1), 0 - i + i - i)  # i-k <= n-1 => k >= i-(n-1)
    # Actually: 0 <= i-k <= n-1 => i-(n-1) <= k <= i
    k_lo = max(-(n-1), i - (n-1))
    k_hi = min(n-1, i)
    for k in range(k_lo, k_hi + 1):
        idx = n - 1 + k
        delta[idx] -= (1 - h[i-k])
    
    # Term 2: +h[i+k] for valid i+k (0<=i+k<n)
    # 0 <= i+k <= n-1 => -i <= k <= n-1-i
    k_lo = max(-(n-1), -i)
    k_hi = min(n-1, n-1-i)
    for k in range(k_lo, k_hi + 1):
        idx = n - 1 + k
        delta[idx] += h[i+k]
    
    # Term 3: +(1-h[j-k]) for valid j-k (0<=j-k<n)
    k_lo = max(-(n-1), j-(n-1))
    k_hi = min(n-1, j)
    for k in range(k_lo, k_hi + 1):
        idx = n - 1 + k
        delta[idx] += (1 - h[j-k])
    
    # Term 4: -h[j+k] for valid j+k (0<=j+k<n)
    k_lo = max(-(n-1), -j)
    k_hi = min(n-1, n-1-j)
    for k in range(k_lo, k_hi + 1):
        idx = n - 1 + k
        delta[idx] -= h[j+k]
    
    return delta

        best_i = best_j = -1
        best_c = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = np.correlate(h, 1.0-h, mode='full')
                nm = float(c.max())
                imp = cur_max - nm
                if imp > best_imp:
                    best_imp = imp
                    best_i, best_j, best_c = i, j, c
                h[i], h[j] = 1.0, 0.0
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            cur_max = float(best_c.max())
        else:
            break
    return h, cur_max
