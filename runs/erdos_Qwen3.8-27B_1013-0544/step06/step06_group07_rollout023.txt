    best_h = h0.copy()
    best_n = n_init
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Optimize at n_init
    p1_end = t0 + 0.50 * (budget_s - 10)
    if best_n % 2 == 0 and best_n <= 500 and best_n >= 50:
        h_work = best_h.copy()
        p1a = t0 + 0.12 * (budget_s - 10)
        for alpha in [500, 2000, 10000, 50000, 200000]:
            if time.time() > p1a:
                break
            h_work = slsqp_smooth(h_work, best_n, alpha, 800, 1e-14)
            c = compute_c5(h_work, best_n)
            if c < best_c:
                best_c = c
                best_h = h_work.copy()
        h_bin = round_to_binary(h_work, best_n)
        sa_end = p1_end - 15
        if time.time() < sa_end - 5:
            for att in range(4):
                if time.time() > sa_end:
                    break
                ts, te, pat = [0.03, 0.05, 0.02, 0.04][att], [0.00001, 0.00003, 0.000005, 0.00002][att], [40000, 50000, 60000, 45000][att]
                start_h = h_bin if att == 0 else round_to_binary(h_bin + 0.3 * rng.randn(best_n), best_n)
                hb, cb = binary_sa(start_h, best_n, sa_end, seed=seed + att * 100, t_start=ts, t_end=te, patience=pat)
                if cb < best_c:
                    best_c = cb
                    best_h = hb.copy()
                    h_bin = hb.copy()
        if time.time() < p1_end - 2:
            hg, cg = greedy_polish(best_h, best_n, p1_end - 2, seed=seed + 500)
            if cg < best_c:
                best_c = cg
                best_h = hg.copy()

    # Phase 2: Try other n values
    n_candidates = [120, 160, 200, 240, 280, 320, 400]
    n_candidates = [nc for nc in n_candidates if nc != best_n and nc % 2 == 0]
    if n_candidates and time.time() < dl - 50:
        time_left = dl - time.time() - 40
        time_per_n = max(30, min(120, time_left / max(len(n_candidates), 1)))
        for nc in n_candidates:
            n_deadline = min(time.time() + time_per_n, dl - 40)
            if n_deadline - time.time() < 15:
                continue
            best_here_h = None
            best_here_c = float('inf')
            for init_type in range(3):
                if time.time() > n_deadline - 5:
                    break
                if init_type == 0:
                    np.random.seed(seed + nc)
                    tm = thue_morse(nc)
                    h_start = make_balanced(tm, nc, rng)
                elif init_type == 1:
                    x_old = np.linspace(0, 2, best_n)
                    x_new = np.linspace(0, 2, nc)
                    h_interp = np.interp(x_new, x_old, best_h)
                    h_start = round_to_binary(h_interp, nc)
                else:
                    h_start = make_balanced(rng.randint(2, size=nc), nc, rng)
                t_sa = n_deadline - 3
                if time.time() < t_sa - 5:
                    hb, cb = binary_sa(h_start, nc, t_sa, seed=seed + nc * 11 + init_type * 7,
                                       t_start=0.03, t_end=0.00002, patience=35000)
                    if time.time() < t_sa - 1:
                        hg, cg = greedy_polish(hb, nc, t_sa, seed=seed + nc * 13 + init_type)
                        if cg < cb:
                            hb, cb = hg, cg
                    if cb < best_here_c:
                        best_here_c = cb
                        best_here_h = hb.copy()
            if best_here_h is not None and best_here_c < best_c:
                best_c = best_here_c
                best_h = best_here_h.copy()
                best_n = nc

    # Phase 3: Final polish
    if time.time() < dl - 8:
        h_bin_final = round_to_binary(best_h, best_n)
        hb, cb = binary_sa(h_bin_final, best_n, dl - 5, seed=seed + 5000,
                          t_start=0.02, t_end=0.000005, patience=60000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    if time.time() < dl - 2:
        hg, cg = greedy_polish(best_h, best_n, dl - 1, seed=seed + 6000)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
