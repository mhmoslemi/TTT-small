```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max_of_subset(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lag_gradient(h, n, lag):
    """Gradient of the lag value w.r.t. h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        grad[:] = 0
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def subgradient_minmax(h, n, deadline, n_iter=2000, lr_init=0.001):

    h_cur = h.copy()

    lr = lr_init

        if time.time() > deadline:

def outer_approx_one_step(h_curr, n, trust):

    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * float(grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or h_var.value is None:
        return None
    return project_feasible(h_var.value, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max_subset(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n - m] -= h[m:]
            grad[m:] += (1.0 - h[:n - m])
        else:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def gradient_descent(h, n, deadline, top_k=5, max_outer=300):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_cur, n)
        top_idx = np.argsort(c)[-top_k:]
        g_avg = np.zeros(n)
        for idx in top_idx:
            lag = int(idx - (n - 1))
            g_avg += lag_gradient(h_cur, n, lag)
        g_avg /= top_k
        g_avg -= g_avg.mean()
        best_step = 0
        best_c_trial = best_c
        best_trial = h_cur
        for lr in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            trial = h_cur - lr * g_avg
            trial = project_feasible(trial, n)
            c_trial = compute_c5(trial, n)
            if c_trial < best_c_trial:
                best_c_trial = c_trial
                best_step = lr
                best_trial = trial
        if best_step == 0:
            break
        h_cur = best_trial
        if best_c_trial < best_c:
            best_c = best_c_trial
            best_h = h_cur.copy()
    return best_h, best_c

def active_set_loop(h, n, deadline, top_m=5, max_rounds=20):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        c = compute_all_lags(h_cur, n)
        improved = False
        for tm in [top_m, min(10, 2 * top_m)]:
            top_idx = np.argsort(c)[-tm:]
            for alpha in [5000, 20000, 100000, 500000]:
                if time.time() > deadline - 3:
                    break
                bounds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
                try:
                    res = minimize(_smooth_max_subset, h_cur,
                                  args=(n, top_idx, alpha),
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-12:
                        best_c = c_new
                        best_h = h_new.copy()
                        h_cur = h_new
                        improved = True
                except Exception:
                    pass
        if not improved:
            break
    return best_h, best_c

def local_search_multi(h, n, deadline, top_k=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_lags = np.argsort(c)[-top_k:]
            for lag_idx in top_lags:
                if time.time() > deadline:
                    break
                lag = int(lag_idx - (n - 1))
                m = abs(lag)
                if m == 0 or m >= n:
                    continue
                best_improve = 0
                best_move = None
                stride = max(1, (n - m) // 150)
                for i in range(0, n - m, stride):
                    j = i + m
                    if lag > 0:
                        if h[j] > h[i]:
                            red = (h[j] - h[i]) * d
                            if red > best_improve:
                                best_improve = red
                                best_move = (j, i, -1)
                    else:
                        if h[i] > h[j]:
                            red = (h[i] - h[j]) * d
                            if red > best_improve:
                                best_improve = red
                                best_move = (i, j, -1)
                if best_move is None:
                    continue
                src, dst, sign = best_move
                delta = min(d, 0.5 * abs(h[src] - h[dst]))
                if delta < 1e-10:
                    continue
                h_old = h.copy()
                h[src] -= delta
                h[dst] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h = h_old
    return h, best_c

def outer_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * float(grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solved = False
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    if not solved or h_var.value is None:
        return None
    return project_feasible(npv(h_var.value), n)

def npv(x):
    return np.asarray(x).flatten()

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Outer approximation
    if time.time() < deadline - 80:
        trust = 0.2
        oo_deadline = min(time.time() + 35, deadline - 80)
        for it in range(40):

            h_new = outer_step(best_h, n, trust)

                if c_new < best_c - 1e-12:

                trust *= 0.8
            else:
                break
    # Phase 2: Active set optimization
    if time.time() < deadline - 50:

        if c_as < best_c:

    # Phase 3: Gradient descent on active set
    if time.time() < deadline - 40:

        h_gd, c_gd = gradient_descent(best_h, n, gd_dl, top_k=5)
        if c_gd < best_c:

            best_h = h_gd
    # Phase 4: Fine active set
    if time.time() < deadline - 30:

        h_as, c_as = active_set_loop(best_h, n, as_dl, top_m=3, max_rounds=12)
        if c_as < best_c:

    # Phase 5: Gradient descent fine
    if time.time() < deadline - 20:
        gd_dl = min(time.time() + 25, deadline - 20)

        if c_gd < best_c:
            best_c = c_gd

    # Phase 6: Local search

        top_idx = np.argsort(c)[-4:]
        bounds = [(0.0, 1.0)] * n

    best_h = project_feasible(best_h, n)
```
