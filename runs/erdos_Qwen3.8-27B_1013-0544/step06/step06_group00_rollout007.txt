from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on shift for sum constraint
    lo, hi = h.min() - 1.0, h.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def pgd_optimize(h, n, alpha, n_iters, step=0.8, deadline=None):
    """Projected gradient descent on smoothed max objective."""
    h = h.copy()
    for it in range(n_iters):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_shifted = alpha * (c - c.max())
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        # Form circular weight
        w_circ = np.zeros(n)
        # w is indexed 0..2n-2, corresponding to shift m = k-(n-1) for k=0..2n-2
        # So w[k] corresponds to shift m = k - (n-1)
        # w_circ[m % n] += w[m + (n-1)]
        for m in range(n):
            # m corresponds to shift m (positive)
            w_circ[m] += w[m + n - 1]
            if m > 0:
                w_circ[m] += w[n - 1 - m]  # shift -m = shift (n-1-m) in w indexing
        # Wait, this double-counts m=0. Let me be more careful.
        # Shift m ranges from -(n-1) to (n-1).
        # w index for shift m is: k = m + (n-1), so w[k] = w[m+n-1]
        # For circular index: m mod n
        # m=0: w[n-1], circular index 0
        # m=1: w[n], circular index 1
        # ...
        # m=n-1: w[2n-2], circular index n-1
        # m=-1: w[n-2], circular index n-1
        # m=-2: w[n-3], circular index n-2
        # ...
        # m=-(n-1): w[0], circular index 1
        
        # So:
        # circular index 0: w[n-1] (shift 0)
        # circular index 1: w[n] + w[0] (shift 1 and shift -(n-1))
        # circular index 2: w[n+1] + w[1] (shift 2 and shift -(n-2))
        # ...
        # circular index k: w[n+k] + w[k-1] for k=1,...,n-1
        
        w_circ = np.zeros(n)
        w_circ[0] = w[n-1]
        for k in range(1, n):
            w_circ[k] = w[n+k] + w[k-1]
        
        W_fft = np.fft.fft(w_circ)
        H_fft = np.fft.fft(h)

        # Check improvement

            break
        h = h_new
    return h

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    """Binary local search with greedy swaps."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_set = set(int(x) for x in np.where(h == 0)[0])
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                if i == j:
                    continue
                # Compute delta for swapping: h[i] 0→1, h[j] 1→0
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    zeros_list.remove(i)
                    ones_list.remove(j)
                    ones_list.append(i)
                    zeros_list.append(j)
                    zeros_set.remove(i)
                    ones_set.remove(j)
                    zeros_set.add(j)
                    ones_set.add(i)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
