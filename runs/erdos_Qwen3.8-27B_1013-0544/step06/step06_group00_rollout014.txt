```python
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda for sum constraint
    lo, hi = -2.0, 2.0

        if s > target:

        else:

    """Full correlation of a and b using FFT."""

    # Pad to power of 2 for efficiency

    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, c):
    dx = 2.0 / n
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    two_n_minus_1 = 2 * n - 1
    for m in range(two_n_minus_1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n-k] += wm * (1.0 - h[k:])
                grad[k:] -= wm * h[:n-k]
        else:
            ka = -k
            if ka < n:
                grad[ka:] += wm * (1.0 - h[:n-ka])
                grad[:n-ka] -= wm * h[ka:]
    return grad

def fista_optimize(h, n, alpha, max_iters=10000, tol=1e-10):
    dx = 2.0 / n
    h = project_feasible(h, n)
    y = h.copy()
    t = 1.0
    step = 1.0 / (alpha + 1.0)
    prev_val = None
    prev_y = None
    
    for it in range(max_iters):
        c = np.correlate(y, 1 - y, mode='full') * dx
        val = c.max()
        if prev_val is not None and abs(prev_val - val) < tol:
            break
        prev_val = val
        
        grad = smooth_max_grad(y, n, alpha, c)
        
        # Backtracking line search
        x = y - step * grad
        for _ in range(20):
            x_proj = project_feasible(x, n)
            c_x = np.correlate(x_proj, 1 - x_proj, mode='full') * dx
            # Check Armijo condition (simplified)
            if c_x.max() <= val - 0.5 * step * np.dot(grad, x_proj - y) + 1e-14:
                break
            step *= 0.5
            x = y - step * grad
        else:
            x_proj = project_feasible(y - step * grad, n)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_proj + ((t - 1.0) / t_new) * (x_proj - h)
        
        h = x_proj
        y = y_new
        t = t_new
    
    return h, c.max()

def direct_polish(h, n, deadline, max_iters=50000):
    """Polish using direct gradient at argmax."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = c.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Gradient at argmax
        k_star = np.argmax(c) - (n - 1)  # actual shift
        grad = np.zeros(n)
        k = k_star
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            ka = -k
            grad[ka:] += (1.0 - h[:n-ka])
            grad[:n-ka] -= h[ka:]
        
        if np.abs(grad).max() < 1e-10:
            break
        
        # Try the step
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < cur_c:
            h = h_new
            # Adaptive step increase
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial values
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: FISTA with increasing alpha
    alphas = [5, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    
    for alpha in alphas:
        if time.time() > deadline - 120:
            break
        iters = max(1000, min(20000, int(5000000 / (alpha + 1))))
        h_out, c_out = fista_optimize(best_h, n, alpha, max_iters=iters)
        c_final = compute_c5(h_out, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_out.copy()
    
    # Phase 2: Direct gradient polish
    if time.time() < deadline - 60:
        h_pol, c_pol = direct_polish(best_h, n, deadline - 60, max_iters=100000)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()
    
    # Phase 3: Try different resolutions
    if time.time() < deadline - 30:
        for nh in [200, 300, 500, 600]:
            if time.time() > deadline - 20:
                break
            try:
                xo = np.linspace(0, 2, n + 1)[:-1] + 1.0/n
                xn = np.linspace(0, 2, nh + 1)[:-1] + 1.0/nh
                h_interp = np.interp(xn, xo, best_h)
                h_interp = project_feasible(h_interp, nh)
                for alpha in [100, 1000, 10000, 100000]:
                    if time.time() > deadline - 15:
                        break
                    h_out, _ = fista_optimize(h_interp, nh, alpha, max_iters=3000)
                    c_val = compute_c5(h_out, nh)
                    if c_val < best_c:
                        best_c = c_val
                        best_h = h_out.copy()
                        n = nh
            except Exception:
                pass
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```

```python
import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, c):
    c_max = c.max()
    ec = np.exp(alpha * (c - c_max))
    s = ec.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w = ec / s
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]

        else:
            ka = -k
            if ka < n:
                grad[ka:] += wm * (1.0 - h[:n - ka])

    return grad
```
