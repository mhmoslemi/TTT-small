def compute_gradient(h, alpha):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1-h, mode='full') * dx
    # Compute weights
    m = c.max()
    shifted = alpha * (c - m)
    exp_vals = np.exp(shifted - shifted.max())
    w = exp_vals / exp_vals.sum()  # length 2N-1
    
    # c is indexed 0 to 2N-2, corresponding to lags -(N-1) to N-1
    w_pos = w[N-1:]    # w for lags 0 to N-1, index 0 = lag 0
    w_neg = w[N-2::-1]  # w for lags -1 to -(N-1), index 0 = lag -1
    
    grad = np.zeros(N)
    f = 1 - h
    
    for j in range(N):
        # S1 from positive lags: sum_{l=0}^{j} w_pos[l] * f[j-l]
        s1_pos = np.dot(w_pos[:j+1], f[j::-1][:j+1])
        # Wait, f[j-l] for l=0..j is f[j], f[j-1], ..., f[0] = f[:j+1][::-1]
        s1_pos = np.dot(w_pos[:j+1], f[:j+1][::-1])
        
        # S1 from negative lags: sum_{m=1}^{N-1-j} w_neg[m-1] * f[j+m]
        # w_neg[m-1] corresponds to lag -m
        if j < N-1:
            s1_neg = np.dot(w_neg[:N-1-j], f[j+1:])
        else:
            s1_neg = 0.0
        
        # S2 from positive lags: sum_{l=0}^{N-1-j} w_pos[l] * h[j+l]
        s2_pos = np.dot(w_pos[:N-j], h[j:])
        
        # S2 from negative lags: sum_{m=1}^{j} w_neg[m-1] * h[j-m]
        if j > 0:
            s2_neg = np.dot(w_neg[:j], h[j-1::-1])
        else:
            s2_neg = 0.0
        
        grad[j] = dx * (s1_pos + s1_neg - s2_pos - s2_neg)
    
    return grad

def compute_gradient(h, alpha):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m) - alpha * c.max()  # for numerical stability
    # Actually: shifted = alpha * (c - c.max())
    shifted = alpha * (c - c.max())
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    w_pos = w[N-1:]       # length N
    w_neg = w[N-2::-1]    # length N-1
    
    f = 1 - h
    s1_pos = np.convolve(w_pos, f, 'full')[:N]
    s1_neg = np.concatenate([np.correlate(f, w_neg, 'full')[N-1:2*N-2], [0.0]])
    s2_pos = np.correlate(h, w_pos, 'full')[N-1:2*N-1]
    s2_neg = np.zeros(N)
    s2_neg[1:] = np.convolve(w_neg, h, 'full')[:N-1]
    
    grad = dx * (s1_pos + s1_neg - s2_pos - s2_neg)
