```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def get_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def sa_multiswap(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00002,
                 patience=25000, p2swap=0.15, p3swap=0.05):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start

    while time.time() < deadline:
        r1 = rng.random()
        if r1 < p3swap:
            kswaps = 3
        elif r1 < p3swap + p2swap:
            kswaps = 2
        else:
            kswaps = 1

        lz, lo = len(zeros_idx), len(ones_idx)
        if lz < kswaps or lo < kswaps:
            kswaps = min(lz, lo)
            if kswaps == 0:
                break

        iz = rng.choice(lz, size=kswaps, replace=False)
        io = rng.choice(lo, size=kswaps, replace=False)

        total_delta = np.zeros(2 * n - 1)
        positions = []
        for ks in range(kswaps):
            i = zeros_idx[iz[ks]]
            j = ones_idx[io[ks]]
            positions.append((i, j))
            total_delta += get_swap_delta(h, n, i, j, m_range)

        new_corr = corr + total_delta
        nc = float(new_corr.max()) * dx
        delta_c = nc - cur_c

        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            for (i, j) in positions:
                h[i] = 1.0
                h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cool_rate
        if T < t_end:
            T = t_end

        if no_improve > patience:
            nk = 3 + rng.randint(5)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                if len(zi) == 0 or len(oi) == 0:
                    break
                ii = zi[rng.randint(len(zi))]
                jj = oi[rng.randint(len(oi))]
                h[ii], h[jj] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = max(T * 4.0, 0.5 * t_start)

    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_iters=5000, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    for rnd in range(4):
        if time.time() > deadline:
            break
        for it in range(max_iters):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            n_try = min(10, len(zeros))
            tidx = rng.choice(len(zeros), size=n_try, replace=False)
            found = False
            for ti in tidx:
                i = zeros[ti]
                bj = -1
                bnc = cur_c
                for j in ones:
                    d = get_swap_delta(h, n, i, j, m_range)
                    nc = float((corr + d).max()) * dx
                    if nc < bnc - 1e-15:
                        bnc = nc
                        bj = j
                if bj >= 0:
                    d = get_swap_delta(h, n, i, bj, m_range)
                    h[i], h[bj] = 1.0, 0.0
                    corr = corr + d
                    cur_c = bnc
                    found = True
                    break
            if not found:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) > 0 and len(zi) > 0:
            npert = 4 + rng.randint(4)
            for _ in range(npert):
                a = oi[rng.randint(len(oi))]
                b = zi[rng.randint(len(zi))]
                h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx

    return best_h, best_c
```
