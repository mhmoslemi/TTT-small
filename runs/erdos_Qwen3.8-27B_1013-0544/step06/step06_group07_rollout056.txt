def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            step = diff / s
            h = h + step * room
            h = np.clip(h, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            step = diff / s
            h = h + step * h
            h = np.clip(h, 0, 1)
    return h

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target} via bisection."""
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.025, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = list(np.where(h == 0)[0])
    ones_idx = list(np.where(h == 1)[0])
    no_improve = 0
    t_end = t_start * 1e-4
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start

    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx.remove(i)
            ones_idx.remove(j)
            zeros_idx.append(j)
            ones_idx.append(i)
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(7)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
                zeros_idx.remove(ii)
                ones_idx.remove(jj)
                zeros_idx.append(jj)
                ones_idx.append(ii)
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def projected_gradient(h, n, deadline, alpha=500.0, lr=0.01, seed=42):
    """Projected gradient descent on the smooth C5 objective."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    d = np.arange(-(n-1), n)  # (2n-1,)
    
    # Precompute index matrix
    J = np.arange(n)[:, None]
    D = d[None, :]
    idx_plus = J + D   # j + d_k
    idx_minus = J - D  # j - d_k
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    idx_plus_c = np.clip(idx_plus, 0, n-1)
    idx_minus_c = np.clip(idx_minus, 0, n-1)
    
    step = 0
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        Z = exps.sum()
        w = exps / Z  # (2n-1,)
        
        # Gradient
        h_at_plus = np.where(valid_plus, h[idx_plus_c], 0.0)
        h_at_minus = np.where(valid_minus, h[idx_minus_c], 0.0)
        term1 = np.where(valid_minus, 1.0 - h_at_minus, 0.0)
        term2 = np.where(valid_plus, h_at_plus, 0.0)
        grad = dx * ((term1 - term2) * w[None, :]).sum(axis=1)
        
        # Adaptive learning rate
        h_new = project_box_sum(h - lr * grad, target)
        c_new = compute_c5(h_new, n)
        
        # Backtracking if needed
        if c_new > best_c + 0.01:
            lr *= 0.5
            if lr < 1e-8:
                lr = 0.01
                break
        else:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        
        step += 1
        # Periodic perturbation
        if step % 5000 == 0:
            h = project_box_sum(h + 0.05 * rng.randn(n), target)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, target):
    lo, hi = h.min() - 1.5, h.max() + 1.5
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def projected_gradient(h, n, deadline, alpha, lr_init, seed):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    d = np.arange(-(n - 1), n)
    J = np.arange(n)[:, None]
    D = d[None, :]
    idx_p = J + D
    idx_m = J - D
    v_p = (idx_p >= 0) & (idx_p < n)
    v_m = (idx_m >= 0) & (idx_m < n)
    ipc = np.clip(idx_p, 0, n - 1)
    imc = np.clip(idx_m, 0, n - 1)
    lr = lr_init
    step = 0
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        exps = np.exp(sh - sm)
        w = exps / exps.sum()
        hp_ = np.where(v_p, h[ipc], 0.0)
        hm_ = np.where(v_m, h[imc], 0.0)
        t1 = np.where(v_m, 1.0 - hm_, 0.0)
        t2 = np.where(v_p, hp_, 0.0)
        grad = dx * ((t1 - t2) * w[None, :]).sum(axis=1)
        h_new = project_box_sum(h - lr * grad, target)
        c_new = compute_c5(h_new, n)
        if c_new > best_c + 0.005:
            lr *= 0.5
            if lr < 1e-9:
                break
        else:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            else:
                lr = min(lr * 1.05, 0.1)
        step += 1
        if step % 3000 == 0:
            h = project_box_sum(h + 0.03 * rng.randn(n), target)
            lr = lr_init * 0.5
    return best_h, best_c

def binary_sa(h_bin, n, deadline, seed, t_start, patience):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zset = list(np.where(h == 0)[0])
    oset = list(np.where(h == 1)[0])
    no_imp = 0
    T = t_start
    t_end = t_start * 1e-5
    cr = (t_end / t_start) ** (1.0 / 150000.0)
    while time.time() < deadline:
        if no_imp > patience:
            nk = 4 + rng.randint(8)
            for _ in range(nk):
                ii = zset[rng.randint(len(zset))]
                jj = oset[rng.randint(len(oset))]
                h[ii], h[jj] = 1.0, 0.0
                zset.remove(ii); oset.remove(jj)
                zset.append(jj); oset.append(ii)
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_imp = 0
            T = min(t_start, T * 4.0)
            continue
        i = zset[rng.randint(len(zset))]
        j = oset[rng.randint(len(oset))]
        delta = swap_delta(h, n, i, j)
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-13 and rng.random() < np.exp(-dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zset.remove(i); oset.remove(j)
            zset.append(j); oset.append(i)
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cr
    return best_h, best_c
