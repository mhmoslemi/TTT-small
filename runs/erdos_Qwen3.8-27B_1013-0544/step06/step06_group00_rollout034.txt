If A is "spread out" (like a fat Cantor set), the autocorrelation decays more slowly.

def sm_obj_grad(x, n, alpha, pen):
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    sh = alpha * (corr - mc)
    sh -= sh.max()
    es = np.exp(sh)
    w = es / es.sum()
    obj = mc + np.log(es.sum()) / alpha
    sv = h.sum() - n / 2.0
    obj += pen * sv * sv
    grad = np.zeros(n)
    n2 = n - 1
    for m in range(-n2, n2 + 1):
        wm = w[m + n2]
        if wm < 1e-14:
            continue
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if hi1 > lo1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, m + n)
        if hi2 > lo2:
            grad[lo2:hi2] -= wm * h[lo2 - m:hi2 - m]
    grad += 2.0 * pen * sv
    return obj, grad

                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxfun': 5000})
    return res.x

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                if i == j:
                    continue
                # Compute new corr after swapping i and j
                # h[i]: 0->1, h[j]: 1->0
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                # Update corr
                # Recompute delta for the actual swap
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]  # h[i] is now 1, so 1-h[im] is wrong...
                # Actually, let me just recompute corr
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sm_obj_grad(x, n, alpha, pen):
    dx = 2.0 / n
    h = x
    L = 2 * n
    H = np.fft.rfft(h, L)
    Oh = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(Oh), L)[:2 * n - 1] * dx
    mc = corr.max()
    sh = alpha * (corr - mc)
    sh -= sh.max()
    es = np.exp(sh)
    w = es / es.sum()
    obj = mc + np.log(es.sum()) / alpha
    sv = h.sum() - n / 2.0
    obj += pen * sv * sv
    L2 = 3 * n + 4
    Wf = np.fft.rfft(w, L2)
    f = 1.0 - 2.0 * h
    ff = np.fft.rfft(f, L2)
    conv = np.fft.irfft(Wf * ff, L2)
    grad = conv[n - 1:2 * n - 1].copy()
    grad += 2.0 * pen * sv
    return obj, grad

def run_lbfgs(h_init, n, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(sm_obj_grad, h_init, args=(n, alpha, pen),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxfun': 8000})
    return res.x

def greedy_binary(h_init, n, deadline, max_sweeps=150):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_nc = cur_c + 1e-9
            best_j = -1
            for j in ones_list:
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-14:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
