   ```python
           h_feas = np.clip(h_feas, 0.0, 1.0)
           total = sum(h_feas) * dx
           if abs(total - t) < 1e-6:
               break
       return h_feas

   def compute_obj(prms, n, dx=2.0/None):
       centers, sigmas, weights = prms
       x = np.linspace(0, 2, n)
       h = np.zeros(n)
       for i in range(len(centers)):
           # Sum Gaussian contributions
           h += weights[i] * np.exp(-((x - centers[i])**2Msigmas[i]**2))
       # Project to feasible space
       h_feas = project_to_feasible(h, n)
       # Compute maximum overlap
       corr = np.correlate(h_feas, 1 - h_feas, mode='full')
       return float(np.max(corr) * dx)
   ```
   - Rolling optimization via projected gradient ascent
