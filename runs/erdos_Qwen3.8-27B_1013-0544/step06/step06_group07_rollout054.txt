```python
class BinaryState:
    def __init__(self, h, n):
        self.h = h.copy()
        self.n = n
        self.zeros = list(np.where(h == 0)[0])
        self.ones = list(np.where(h == 1)[0])
        self.zero_pos = {v: i for i, v in enumerate(self.zeros)}
        self.one_pos = {v: i for i, v in enumerate(self.ones)}
    
    def swap(self, i, j):
        """Swap: h[i] 0->1, h[j] 1->0"""
        # Remove i from zeros
        zi = self.zero_pos[i]
        last_z = self.zeros[-1]
        self.zeros[zi] = last_z
        self.zero_pos[last_z] = zi
        self.zeros.pop()
        del self.zero_pos[i]
        # Remove j from ones
        oj = self.one_pos[j]
        last_o = self.ones[-1]
        self.ones[oj] = last_o
        self.one_pos[last_o] = oj
        self.ones.pop()
        del self.one_pos[j]
        # Add j to zeros
        self.zeros.append(j)
        self.zero_pos[j] = len(self.zeros) - 1
        # Add i to ones
        self.ones.append(i)
        self.one_pos[i] = len(self.ones) - 1
        # Update h
        self.h[i] = 1.0
        self.h[j] = 0.0
```

```python
def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2

    out[idx[:target]] = 1.0

def swap_delta(h, n, i, j, mr):
    d = np.zeros(2*n-1)
    im = i + mr
    msk = (im >= 0) & (im < n)
    d[msk] += 1.0 - h[im[msk]]
    imn = i - mr
    msk = (imn >= 0) & (imn < n)
    d[msk] -= h[imn[msk]]
    jm = j + mr
    msk = (jm >= 0) & (jm < n)
    d[msk] -= 1.0 - h[jm[msk]]
    jmn = j - mr
    msk = (jmn >= 0) & (jmn < n)
    d[msk] += h[jmn[msk]]
    d[n-1] -= 2.0
    d[n-1+j-i] += 1.0
    d[n-1+i-j] += 1.0
    return d
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, mr):
    d = np.zeros(2 * n - 1)
    im = i + mr
    msk = (im >= 0) & (im < n)
    d[msk] += 1.0 - h[im[msk]]
    imn = i - mr
    msk = (imn >= 0) & (imn < n)
    d[msk] -= h[imn[msk]]
    jm = j + mr
    msk = (jm >= 0) & (jm < n)
    d[msk] -= 1.0 - h[jm[msk]]
    jmn = j - mr
    msk = (jmn >= 0) & (jmn < n)
    d[msk] += h[jmn[msk]]
    d[n - 1] -= 2.0
    d[n - 1 + j - i] += 1.0
    d[n - 1 + i - j] += 1.0
    return d

def make_starts(n, seed, count=8):
    rng = np.random.RandomState(seed)
    target = n // 2
    starts = []
    h = np.zeros(n)
    h[rng.choice(n, target, replace=False)] = 1.0
    starts.append(h.copy())
    for bs in [2, 3, 4, 5, 7]:
        h = np.zeros(n)
        for i in range(n):
            if (i // bs) % 2 == 0:
                h[i] = 1.0
        s = h.sum()
        if s > target:
            oi = np.where(h == 1)[0]
            h[rng.choice(oi, int(s - target), replace=False)] = 0.0
        elif s < target:
            zi = np.where(h == 0)[0]
            h[rng.choice(zi, int(target - s), replace=False)] = 1.0
        starts.append(h.copy())
    for p in [13, 17, 23, 31]:
        h = np.zeros(n)
        h[:] = ((np.arange(n) % p) < p // 2).astype(float)
        s = h.sum()
        if s > target:
            oi = np.where(h == 1)[0]
            h[rng.choice(oi, int(s - target), replace=False)] = 0.0
        elif s < target:
            zi = np.where(h == 0)[0]
            h[rng.choice(zi, int(target - s), replace=False)] = 1.0
        starts.append(h.copy())
    return starts[:count]

def sa_fast(h_init, n, deadline, seed, t_init=0.02, t_min=1e-5, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    mr = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur = corr.max() * dx
    best = cur
    best_h = h.copy()
    zi = list(np.where(h == 0)[0])
    oi = list(np.where(h == 1)[0])
    zmap = {v: i for i, v in enumerate(zi)}
    omap = {v: i for i, v in enumerate(oi)}
    T = t_init
    cool = (t_min / t_init) ** (1.0 / 300000.0)
    noimp = 0
    acc = 0
    step = 0
    while time.time() < deadline:
        i = zi[rng.randint(len(zi))]
        j = oi[rng.randint(len(oi))]
        d = swap_delta(h, n, i, j, mr)
        nc = (corr + d).max() * dx
        dc = nc - cur
        step += 1
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr += d
            cur = nc
            ip = zmap[i]
            zp = zi[-1]
            zi[ip] = zp
            zmap[zp] = ip
            zi.pop()
            del zmap[i]
            jp = omap[j]
            op = oi[-1]
            oi[jp] = op
            omap[op] = jp
            oi.pop()
            del omap[j]
            zi.append(j)
            zmap[j] = len(zi) - 1
            oi.append(i)
            omap[i] = len(oi) - 1
            acc += 1
            if nc < best:
                best = nc
                best_h = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
        if step % 800 == 0:
            rate = acc / 800.0
            if rate < 0.05:
                T = max(T * 0.7, t_min)
            elif rate > 0.85:
                T = min(T * 1.3, t_init)
            else:
                T = max(T * cool, t_min)
            acc = 0
        if noimp > patience:
            nk = 4 + rng.randint(10)
            for _ in range(nk):
                a = zi[rng.randint(len(zi))]
                b = oi[rng.randint(len(oi))]
                h[a] = 1.0
                h[b] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur = corr.max() * dx
            zi = list(np.where(h == 0)[0])
            oi = list(np.where(h == 1)[0])
            zmap = {v: i for i, v in enumerate(zi)}
            omap = {v: i for i, v in enumerate(oi)}
            noimp = 0
            T = max(T * 2.5, t_init * 0.4)
    return best_h, best

def greedy_polish(h_init, n, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    mr = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur = corr.max() * dx
    best = cur
    best_h = h.copy()
    zi = list(np.where(h == 0)[0])
    oi = list(np.where(h == 1)[0])
    zmap = {v: i for i, v in enumerate(zi)}
    omap = {v: i for i, v in enumerate(oi)}
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zi)):
            i = zi[ii]
            for jj in range(len(oi)):
                j = oi[jj]
                d = swap_delta(h, n, i, j, mr)
                nc = (corr + d).max() * dx
                if nc < cur - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += d
                    cur = nc
                    zp = zi[-1]
                    zi[ii] = zp
                    zmap[zp] = ii
                    zi.pop()
                    del zmap[i]
                    op = oi[-1]
                    oi[jj] = op
                    omap[op] = jj
                    oi.pop()
                    del omap[j]
                    zi.append(j)
                    zmap[j] = len(zi) - 1
                    oi.append(i)
                    omap[i] = len(oi) - 1
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur < best:
            best = cur
            best_h = h.copy()
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
```
