def compute_g(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.astype(np.float64)

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    threshold = np.partition(h, n/2 - 1)[n/2 - 1] if n % 2 == 0 else np.partition(h, n//2)[n//2]
    h_bin = (h >= threshold).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        # Set some 1s to 0s (the ones closest to threshold)
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by h value (ascending) - those closest to threshold first
        order = np.argsort(h[ones_idx])
        for idx in ones_idx[order[:n_ones - n//2]]:
            h_bin[idx] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in zeros_idx[order[:n//2 - n_ones]]:
            h_bin[idx] = 1.0
    return h_bin

def batch_greedy(cur, g, n, deadline, max_rounds=1000):
    """Try all flip pairs simultaneously, pick the best one each round."""
    dx = 2.0 / n
    N = 2 * n - 1
    d_range = np.arange(-(n-1), n).astype(np.int64)
    
    M = g.max()
    best_c5 = M * dx
    best_h = cur.copy()
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        
        # Compute effect vectors for all ones positions (to flip to 0)
        # E_i[d] = -(1-cur[i+d])*[valid] + cur[i-d]*[valid]
        # Shape: (n_ones, N)
        E = np.zeros((len(ones), N))
        for idx, i in enumerate(ones):
            d_pos = d_range + i
            v1 = (d_pos >= 0) & (d_pos < n)
            E[idx, v1] -= (1.0 - cur[d_pos[v1]])
            d_neg = i - d_range
            v2 = (d_neg >= 0) & (d_neg < n)
            E[idx, v2] += cur[d_neg[v2]]
        
        # Compute effect vectors for all zeros positions (to flip to 1)
        # F_j[d] = (1-cur[j+d])*[valid] - cur[j-d]*[valid]
        F = np.zeros((len(zeros), N))
        for idx, j in enumerate(zeros):
            d_pos = d_range + j
            v3 = (d_pos >= 0) & (d_pos < n)
            F[idx, v3] += (1.0 - cur[d_pos[v3]])
            d_neg = j - d_round
            v4 = (d_neg >= 0) & (d_neg < n)
            F[idx, v4] -= cur[d_neg[v4]]
        
        # For each pair (i, j), new_max = max(g + E[i] + F[j])
        # This is (n_ones, n_zeros, N) -> max over N -> (n_ones, n_zeros)
        # Combined effect: (n_ones, 1, N) + (1, n_zeros, N) = (n_ones, n_zeros, N)
        combined = g[None, None, :] + E[:, None, :] + F[None, :, :]
        new_maxes = combined.max(axis=2)  # shape (n_ones, n_zeros)
        
        # Find the best pair
        min_new_max = new_maxes.min()
        if min_new_max < M:
            bi, bj = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
            i = ones[bi]
            j = zeros[bj]
            cur[i] = 0.0
            cur[j] = 1.0
            g += E[bi] + F[bj]
            M = min_new_max
            if M * dx < best_c5:
                best_c5 = M * dx
                best_h = cur.copy()
        else:
            break  # No improvement possible
    
    return best_h, best_c5, cur
