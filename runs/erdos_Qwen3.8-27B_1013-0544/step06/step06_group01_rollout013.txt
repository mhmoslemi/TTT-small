    dx = 2.0 / n

    w = np.exp(c_shifted)
    w /= w.sum()

        if wk < 1e-15:

                continue
            k_aff = shift
            fwd = (p - k_aff >= 0) & (p - k_aff < n)
            bwd = (p + k_aff >= 0) & (p + k_aff < n)
            affected = fwd | bwd
            if not affected.any():
                continue
            idx_a = np.where(affected)[0]
            s = np.zeros(len(idx_a))
            ka = k_aff[idx_a]
            s[fwd[idx_a]] += (1.0 - h[p - ka[fwd[idx_a]]])
            s[bwd[idx_a]] -= h[p + ka[bwd[idx_a]]]
            um = 0.0
            if not affected.all():
                um = float(np.max(c[~affected]))
            c_a = c[idx_a].copy()
            zero_pos = np.where(idx_a == (n - 1))[0]
            has_zero = len(zero_pos) > 0
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (h.sum() - n / 2.0) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                nc = c_a + d * s
                if has_zero:
                    nc[zero_pos[0]] = c[n-1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(nc)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                c[idx_a] += best_d * s
                if has_zero:
                    c[n-1] = c[n-1] - best_d * (1.0 - 2.0 * h[p]) + best_d * s[zero_pos[0]] - best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[idx_a])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    w = np.exp(alpha * c - alpha * cmax)
    w /= w.sum()
    grad = np.zeros(n)
    shift = np.arange(2 * n - 1) - (n - 1)
    for idx in range(2 * n - 1):
        k = shift[idx]
        wk = w[idx]
        if wk < 1e-13:
            continue
        if k == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            kp = -k
            grad[:n-kp] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:n-kp]
    return grad / alpha

def pg_optimize(h, n, deadline, alpha_start=20, alpha_end=80000, max_iter=3000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stalling = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        progress = min(1.0, it / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        grad = lse_grad(h, n, alpha)
        lr = 2.0 / alpha
        for _ in range(40):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + lr * 0.01:
                break
            lr *= 0.5
        h = h_new
        c_curr = compute_c5(h, n)
        if c_curr < best_c - 1e-8:
            best_c = c_curr
            best_h = h.copy()
            stalling = 0
        else:
            stalling += 1
            if stalling > 20 and alpha > 100:
                h = best_h.copy()
                stalling = 0
    return best_h, best_c

def block_moves(h, n, deadline, rng, block_size=5, n_iters=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    temp = 0.01
    for it in range(n_iters):
        if time.time() > deadline:
            break
        temp = max(0.0001, 0.01 * 0.995 ** it)
        p1 = rng.randint(0, n - block_size)
        block = h[p1:p1+block_size].copy()
        delta = rng.normal(0, 0.05, block_size)
        h[p1:p1+block_size] += delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
        else:
            accept = c_new < best_c + temp
            if not accept:
                h[p1:p1+block_size] = block
                h = project_feasible(h, n)
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=25):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def slsqp_polish(h, n, alpha=200000, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 400:
            return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
        return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase A: PG
    t_a = min(deadline - time.time(), 80)
    if t_a > 20:
        h_pg, c_pg = pg_optimize(h, n, time.time() + t_a)
        if c_pg < best_c:
            h, best_c = h_pg, c_pg
    # Phase B: CD
    if time.time() < deadline - 20:
        h_cd, c_cd = cd_fine(h, n, max_passes=600, grid=200,
                             deadline=time.time() + min(120, deadline - time.time() - 30),
                             rng=rng, penalty=5.0)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase C: SLSQP
    if time.time() < deadline - 30:
        h_s = slsqp_polish(h, n, alpha=300000, maxiter=1000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    # Phase D: Final CD + pair
    if time.time() < deadline - 10:
        h_cd2, c_cd2 = cd_fine(h, n, max_passes=400, grid=200,
                               deadline=time.time() + min(60, deadline - time.time() - 5),
                               rng=rng, penalty=0.1)
        h_cd2 = project_feasible(h_cd2, n)
        c_cd2 = compute_c5(h_cd2, n)
        if c_cd2 < best_c:
            h, best_c = h_cd2, c_cd2
    if time.time() < deadline - 5:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.001, 0.0005, 0.0002, 0.0001], radius=20)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Optimize initial with n=300
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_o, c_o = optimize_full(best_h, n_init, t0 + alloc1, rng)
    if c_o < best_c:
        best_c, best_h = c_o, h_o

    # Try n=200
    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, len(best_h))
        n2 = 200
        x_new = np.linspace(0, 2, n2)
        h2 = project_feasible(np.interp(x_new, x_old, best_h), n2)
        alloc2 = min(120, (deadline - time.time()) * 0.15)
        if alloc2 > 40:
            h_o2, c_o2 = optimize_full(h2, n2, time.time() + alloc2, rng)
            if c_o2 < best_c:
                best_c, best_h = c_o2, h_o2

    # Try n=150
    if time.time() < deadline - 150:
        x_old = np.linspace(0, 2, len(best_h))
        n3 = 150
        x_new = np.linspace(0, 2, n3)
        h3 = project_feasible(np.interp(x_new, x_old, best_h), n3)
        alloc3 = min(100, (deadline - time.time()) * 0.1)
        if alloc3 > 35:
            h_o3, c_o3 = optimize_full(h3, n3, time.time() + alloc3, rng)
            if c_o3 < best_c:
                best_c, best_h = c_o3, h_o3

    # Multi-restart on best
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restart = max(3, int(remaining / 60))
        for r in range(n_restart):
            if time.time() > deadline - 50:
                break
            scale = 0.01 + 0.04 * (r / max(1, n_restart - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n_best), n_best)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_or, c_or = optimize_full(hp, n_best, time.time() + alloc, rng)
            if c_or < best_c:
                best_c, best_h = c_or, h_or

    # Block moves on best
    if time.time() < deadline - 30:
        h_bm, c_bm = block_moves(best_h, len(best_h), time.time() + min(40, deadline - time.time() - 10),
                                 rng, block_size=4, n_iters=3000)
        if c_bm < best_c:
            best_c, best_h = c_bm, h_bm

    # Final polish
    if time.time() < deadline - 15:
        nf = len(best_h)
        h_f = slsqp_polish(best_h, nf, alpha=500000, maxiter=2000)
        c_f = compute_c5(h_f, nf)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        nf = len(best_h)
        h_cd, c_cd = cd_fine(best_h, nf, max_passes=300, grid=250,
                             deadline=deadline - 3, rng=rng, penalty=0.01)
        h_cd = project_feasible(h_cd, nf)
        c_cd = compute_c5(h_cd, nf)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd
    if time.time() < deadline - 2:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005], radius=15)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
