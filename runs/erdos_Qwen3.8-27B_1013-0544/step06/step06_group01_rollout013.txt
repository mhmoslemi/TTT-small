                grad[nl:] -= wm * h[:n - nl]
    return grad * dx

def fista_opt(h0, n, alphas, deadline, top_k=30):
    target = n / 2.0
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        y = h.copy()
        t = 1.0
        step = 1.0 / (alpha * 3.0)
        for it in range(4000):
            if time.time() > deadline:
                break
            g = grad_smooth(y, n, alpha, K=top_k)
            gn = np.max(np.abs(g))
            if gn < 1e-12:
                break
            x_new = project_fast(y - step * g, target)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - h)
            y = np.clip(y, -0.5, 1.5)
            h = x_new
            t = t_new
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def fast_cd(h, n, max_passes, grid, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(c))
    sumh = float(h.sum())
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(1, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = cur_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[bwd - ids_b + (len(ids_b)-len(ids_b))] - (n-1)]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                cur_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, cur_max * (2.0 / n)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_search(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(max(1, i - radius), min(n - 1, i + radius)):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                            c = np.correlate(h, 1.0 - h, mode='full')
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                            c = np.correlate(h, 1.0 - h, mode='full')
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_golden_start(n, seed):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n)
    h = np.where((x * phi) % 1.0 < 0.5, 1.0, 0.0)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = make_golden_start(300, seed)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: FISTA on initial (n=300)
    alloc1 = min(180, (deadline - t0) * 0.18)
    if alloc1 > 30:
        h_f, c_f = fista_opt(h_init, n_init, [50, 200, 1000, 5000], t0 + alloc1)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        h_cd, c_cd = fast_cd(best_h, n_init, 600, 100, time.time() + 120, rng)
        h_cd = project_feasible(h_cd, n_init)
        c_cd = compute_c5(h_cd, n_init)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd

    # Phase 3: SLSQP
    if time.time() < deadline - 150:
        h_s = slsqp_opt(best_h, n_init, 50000, 2000)
        c_s = compute_c5(h_s, n_init)
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 4: Try n=200
    if time.time() < deadline - 150:
        n2 = 200
        x_init = np.linspace(0, 2, n_init)
        x2 = np.linspace(0, 2, n2)
        h2 = project_feasible(np.interp(x2, x_init, best_h), n2)
        h2f, c2f = fista_opt(h2, n2, [100, 500, 2000], time.time() + 100)
        if c2f < best_c:
            best_c, best_h = c2f, h2f
        if time.time() < deadline - 100:
            h2c, c2c = fast_cd(best_h, len(best_h), 400, 100, time.time() + 60, rng)
            h2c = project_feasible(h2c, len(h2c))
            c2c = compute_c5(h2c, len(h2c))
            if c2c < best_c:
                best_c, best_h = c2c, h2c
        if time.time() < deadline - 80:
            h2s = slsqp_opt(best_h, len(best_h), 100000, 2000)
            c2s = compute_c5(h2s, len(h2s))
            if c2s < best_c:
                best_c, best_h = c2s, h2s

    # Phase 5: More CD + SLSQP on best
    if time.time() < deadline - 80:
        nn = len(best_h)
        h_cd2, c_cd2 = fast_cd(best_h, nn, 500, 120, time.time() + 80, rng)
        h_cd2 = project_feasible(h_cd2, nn)
        c_cd2 = compute_c5(h_cd2, nn)
        if c_cd2 < best_c:
            best_c, best_h = c_cd2, h_cd2
    if time.time() < deadline - 50:
        nn = len(best_h)
        h_s2 = slsqp_opt(best_h, nn, 200000, 3000)
        c_s2 = compute_c5(h_s2, nn)
        if c_s2 < best_c:
            best_c, best_h = c_s2, h_s2

    # Phase 6: Restart with perturbation
    if time.time() < deadline - 60:
        nn = len(best_h)
        hp = project_feasible(best_h + rng.normal(0, 0.03, nn), nn)
        h_r, c_r = fista_opt(hp, nn, [200, 1000], time.time() + 50)
        if c_r < best_c:
            best_c, best_h = c_r, h_r
        if time.time() < deadline - 30:
            h_rc, c_rc = fast_cd(best_h, nn, 300, 100, time.time() + 30, rng)
            h_rc = project_feasible(h_rc, nn)
            c_rc = compute_c5(h_rc, nn)
            if c_rc < best_c:
                best_c, best_h = c_rc, h_rc

    # Phase 7: Pair search (fine)
    if time.time() < deadline - 15:
        nn = len(best_h)
        best_h, best_c = pair_search(best_h, nn, deadline - 5,
                                     [0.005, 0.002, 0.001, 0.0005], radius=40)

    # Phase 8: Final SLSQP + pair
    if time.time() < deadline - 10:
        nn = len(best_h)
        h_f = slsqp_opt(best_h, nn, 500000, 2000)
        c_f = compute_c5(h_f, nn)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        nn = len(best_h)
        best_h, best_c = pair_search(best_h, nn, deadline - 2,
                                     [0.0005, 0.0002, 0.0001], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
