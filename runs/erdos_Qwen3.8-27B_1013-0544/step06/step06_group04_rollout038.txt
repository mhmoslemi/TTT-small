import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_initial_h(n):
    """Create a good initial binary h with n/2 ones."""
    h = np.zeros(n)
    h[:n//2] = 1.0
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cd = c * dx
    m = cd.max()
    exp_cd = np.exp(alpha * (cd - m))
    p = exp_cd / exp_cd.sum()
    f = m + np.log(exp_cd.sum()) / alpha
    
    # Vectorized gradient
    j_idx = np.arange(n)[:, None].astype(np.int64)
    taus = (np.arange(2*n-1)[None, :] - (n-1)).astype(np.int64)
    idx1 = j_idx + taus
    idx2 = j_idx - taus
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    h_ext = np.zeros(3*n - 1)
    h_ext[n-1 : 2*n-1] = h
    
    sh1 = idx1 + (n-1)
    sh2 = idx2 + (n-1)
    term1 = np.where(mask1, 1.0 - h_ext[sh1], 0.0)
    term2 = np.where(mask2, h_ext[sh2], 0.0)
    grad = dx * ((term1 - term2) @ p)
    
    return f, grad

def smooth_optimize(h, n, alpha, maxiter=200, tol=1e-10):

    # Add integral penalty
    def obj_with_penalty(x):
        f, g = smooth_obj_and_grad(x, n, alpha)
        integral = x.sum() * dx
        f += 1e6 * (integral - 1.0)**2
        g += 2e6 * (integral - 1.0) * dx
        return f, g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_with_penalty, h, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': tol, 'ftol': 1e-12})
    return res.x

    """Simple first-improvement local search on binary solutions."""
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                # Compute change in correlation from swapping h[i] and h[j]
                # h[i] goes 0->1, h[j] goes 1->0
                delta = _swap_delta_vec(h, int(i), int(j), n)

                if nc < cur_c - 1e-15:

                    corr = corr + delta
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
    return best_h, best_c

def _swap_delta_vec(h, i, j, n):

    delta = np.zeros(2 * n - 1)

    # Wait, the formula is: dc[k]/dh[i] = (1-h[i+tau_k])*[valid] - h[i-tau_k]*[valid]
    # where tau_k = k - (n-1)
    # But we're going from 0 to 1, so the change is:
    # new_c[k] - old_c[k] = dc[k]/dh[i] * 1 + dc[k]/dh[j] * (-1)
    # = [(1-h[i+tau_k])*[valid_i_plus] - h[i-tau_k]*[valid_i_minus]]
    #   - [(1-h[j+tau_k])*[valid_j_plus] - h[j-tau_k]*[valid_j_minus]]
    # But we need to evaluate at the CURRENT h (before the swap)
    
    taus = np.arange(2*n-1) - (n-1)

    idx_i_plus = i + taus
    mask_ip = (idx_i_plus >= 0) & (idx_i_plus < n)
    idx_i_minus = i - taus
    mask_im = (idx_i_minus >= 0) & (idx_i_minus < n)
    
    delta[mask_ip] += (1.0 - h[idx_i_plus[mask_ip]])
    delta[mask_im] -= h[idx_i_minus[mask_im]]

    mask_jp = (idx_j_plus >= 0) & (idx_j_plus < n)
    idx_j_minus = j - taus
    mask_jm = (idx_j_minus >= 0) & (idx_j_minus < n)
    
    delta[mask_jp] -= (1.0 - h[idx_j_plus[mask_jp]])
    delta[mask_jm] += h[idx_j_minus[mask_jm]]
    
    return delta
