        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') & 1
    return h

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.00001, patience=25000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 600000.0)
    T = t_start
    steps = 0
    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        steps += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 4 + rng.randint(10)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def greedy_binary(h_bin, n, deadline, max_sweeps=300, seed=42):
    idx = np.argsort(h_bin)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def ils_search(h_start, n, deadline, seed=42):
    """Iterated local search: greedy polish + random kick."""
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h_start.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    while time.time() < deadline:
        # Greedy polish
        sweeps = 0
        while sweeps < 100:
            improved = False
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            rng.shuffle(zeros)
            rng.shuffle(ones)
            for i in zeros:
                for j in ones:
                    delta = swap_delta(h, n, i, j, m_range)
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = corr + delta
                        cur_c = nc
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
            sweeps += 1
            if time.time() > deadline:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # Kick
        nk = 3 + rng.randint(8)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for _ in range(nk):
            if len(zeros) == 0 or len(ones) == 0:
                break
            ii = zeros[rng.randint(len(zeros))]
            jj = ones[rng.randint(len(ones))]
            h[ii] = 1.0
            h[jj] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_c = corr.max() * dx
    return best_h, best_c

def make_structures(n, seed):
    rng = np.random.RandomState(seed)
    target = n // 2
    results = []
    # Thue-Morse
    tm = thue_morse(n).astype(float)
    if tm.sum() != target:
        if tm.sum() > target:
            ones = np.where(tm == 1)[0]
            tm[rng.choice(ones, int(tm.sum() - target), replace=False)] = 0
        else:
            zeros = np.where(tm == 0)[0]
            tm[rng.choice(zeros, int(target - tm.sum()), replace=False)] = 1
    results.append(tm)
    # Alternating
    alt = np.zeros(n)
    alt[::2] = 1.0
    if alt.sum() != target:
        d = int(target - alt.sum())
        if d > 0:
            z = np.where(alt == 0)[0]
            alt[rng.choice(z, d, replace=False)] = 1
        else:
            o = np.where(alt == 1)[0]
            alt[rng.choice(o, -d, replace=False)] = 0
    results.append(alt)
    # Quadratic residues
    qr = np.zeros(n)
    for i in range(n):
        if (i * i) % n < n // 2:
            qr[i] = 1.0
    if qr.sum() != target:
        d = int(target - qr.sum())
        if d > 0:
            z = np.where(qr == 0)[0]
            qr[rng.choice(z, d, replace=False)] = 1
        else:
            o = np.where(qr == 1)[0]
            qr[rng.choice(o, -d, replace=False)] = 0
    results.append(qr)
    # Random
    r = np.zeros(n)
    r[rng.choice(n, target, replace=False)] = 1.0
    results.append(r)
    return results

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Build initial structures
    structs = make_structures(n, seed)
    structs.append(round_to_binary(h0, n))
    for s in structs:
        c = compute_c5(s, n)
        if c < best_c:
            best_c = c
            best_h = s.copy()
    # Phase 1: Quick SLSQP (10% budget)
    p1 = t0 + 0.10 * (budget_s - 5)
    hc = project_feasible(best_h + 0.05 * rng.randn(n), n)
    for alpha in [2000, 10000, 50000]:
        if time.time() > p1:
            break
        hc = slsqp_smooth(hc, n, alpha, 800, 1e-14)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c = c
            best_h = hc.copy()
    # Phase 2: SA from diverse starts (55% budget)
    p2 = t0 + 0.65 * (budget_s - 5)
    for k in range(4):
        seg_start = t0 + 0.10 * (budget_s - 5) + k * 0.11 * (budget_s - 5)
        seg_end = t0 + (0.10 + (k + 1) * 0.11) * (budget_s - 5)
        if time.time() > seg_end:
            break
        start_h = structs[k] if k < len(structs) else best_h
        hb, cb = binary_sa(start_h, n, seg_end, seed=seed + 1000 * k,
                           t_start=0.03, t_end=0.00002, patience=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Extended SA from best
    if time.time() < p2:
        hb, cb = binary_sa(best_h, n, p2, seed=seed + 7777,
                           t_start=0.02, t_end=0.00001, patience=35000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: ILS (20% budget)
    p3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p3:
        hi, ci = ils_search(best_h, n, p3, seed=seed + 3000)
        if ci < best_c:
            best_c = ci
            best_h = hi.copy()
    # Phase 4: SLSQP continuous refinement
    p4 = dl - 20
    if time.time() < p4:
        hp = project_feasible(best_h.astype(float), n)
        for alpha in [10000, 50000, 200000]:
            if time.time() > p4:
                break
            hp = slsqp_smooth(hp, n, alpha, 1500, 1e-14)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    # Phase 5: Final SA + greedy
    if time.time() < dl - 25:
        hb2, cb2 = binary_sa(round_to_binary(best_h, n), n, dl - 25,
                             seed=seed + 9000, t_start=0.01, t_end=0.00005, patience=15000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    if time.time() < dl - 5:
        hg2, cg2 = greedy_binary(best_h, n, dl - 3, max_sweeps=150, seed=seed + 9500)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
