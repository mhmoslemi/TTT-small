import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_weights(overlaps, alpha):
    shifted = alpha * (overlaps - overlaps.max())
    w = np.exp(shifted)
    w /= w.sum()
    return w

def compute_gradient(h, n, w):
    dx = 2.0 / n
    w_pos = w[n-1:2*n-1]
    w_neg_full = np.zeros(n)
    w_neg_full[1:] = w[n-2::-1]
    h_rev = h[::-1]
    omh = 1.0 - h
    omh_rev = omh[::-1]
    A = np.convolve(w_pos, omh_rev)[:n][::-1]
    B = np.convolve(w_pos, h)[:n]
    C = np.convolve(w_neg_full, omh)[:n]
    D = np.convolve(w_neg_full, h_rev)[:n][::-1]
    return dx * (A - B + C - D)

def pgd_step(h, n, w, lr, v, momentum):
    grad = compute_gradient(h, n, w)
    v = momentum * v + lr * grad
    h_new = project_feasible(h - v, n)
    return h_new, v, grad

def pgd_optimize(h_init, n, deadline, alpha_start=100, alpha_end=50000, 
                 lr_start=1.0, momentum=0.9, n_alpha_steps=50):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_start
    total_iters = 200000
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        
        # Adaptive alpha
        frac = min(1.0, it / n_alpha_steps)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        overlaps = compute_overlaps(h, n)
        w = smooth_max_weights(overlaps, alpha)
        h_new, v, _ = pgd_step(h, n, w, lr, v, momentum)
        
        # Periodic C5 evaluation
        if it % 500 == 0:
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            # Adaptive lr: if stuck, reduce lr
            if it > 1000 and not (best_c < compute_c5(h, n) - 1e-10):
                lr *= 0.9
                v *= 0.5
        
        h = h_new
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h
    return best_h, best_c

def binary_swaps(h_init, n, deadline, max_sweeps=200):
    """Greedy binary swap local search."""
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                # Compute delta in correlation
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1 + (j-i)] += 1.0
                delta[n-1 + (i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
