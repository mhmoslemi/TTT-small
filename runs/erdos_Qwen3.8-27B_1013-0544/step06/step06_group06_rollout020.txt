def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def build_g_fast(h, n):

        for _ in range(n_pert):
            oi = rng.randint(len(ones_idx))
            zi = rng.randint(len(zeros_idx))
            h_pert[ones_idx[oi]], h_pert[zeros_idx[zi]] = 0.0, 1.0
            ones_idx = np.delete(ones_idx, oi)
            zeros_idx = np.delete(zeros_idx, zi)
            ones_idx = np.append(ones_idx, zeros_idx[zi])
            zeros_idx = np.append(zeros_idx, ones_idx[np.argmax(h_pert == 1)])
            # Fix: properly track
        ones_idx = np.where(h_pert == 1)[0]
        zeros_idx = np.where(h_pert == 0)[0]
        hb, cb = vectorized_local_search(h_pert, n, p2, seed=seed + 500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Try Thue-Morse init if time permits (8% time)
    p3 = t0 + 0.82 * (budget_s - 10)
    if time.time() < p3 - 10 and n >= 200:
        h_tm = _thue_morse_init(n)
        hb, cb = vectorized_local_search(h_tm, n, p3, seed=seed + 1000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 4: SLSQP continuous polish (5% time)
    p4 = t0 + 0.89 * (budget_s - 10)
    if time.time() < p4 - 5:
        h_cont = project_feasible(best_h + 0.02 * rng.randn(n), n)
        h_cont = slsqp_polish(h_cont, n, p4 - 3)
        c = compute_c5(h_cont, n)
        if c < best_c:
            best_c = c
            best_h = h_cont.copy()

    # Phase 5: Final local search on best (remaining)
    if time.time() < dl - 5:
        hb, cb = vectorized_local_search(best_h, n, dl - 3, seed=seed + 3000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = np.round(best_h)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
