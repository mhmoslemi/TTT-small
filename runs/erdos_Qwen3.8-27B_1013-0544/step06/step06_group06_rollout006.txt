import numpy as np
from scipy.optimize import minimize
import time, math

_OPT = {}

def c_counts_max(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return c, float(np.max(c))

def evaluate_c5(h, n):
    _, mc = c_counts_max(h, n)
    return mc * (2.0 / float(n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).ravel().copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(70):
        y = 0.5 * (lo + hi)
        x = np.clip(h - y, 0.0, 1.0)
        sx = float(x.sum())
        if sx > target:
            lo = y
        else:
            hi = y
    y = 0.5 * (lo + hi)
    x = np.clip(h - y, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-15:
        if r > 0.0:
            room = 1.0 - x
            sr = float(room.sum())
            if sr > 1e-15:
                x = x + (r / sr) * room
        else:
            ss = float(x.sum())
            if ss > 1e-15:
                x = x + (r / ss) * x
    return np.clip(x, 0.0, 1.0)

def L_row(p, h, n):
    L = np.zeros(2 * n - 1)
    s1 = np.arange(p - n + 1, p + 1, dtype=int)
    L[s1 + n - 1] += 1.0 - h[p - s1]
    s2 = np.arange(-p, n - p, dtype=int)
    L[s2 + n - 1] -= h[p + s2]
    return L

def smooth_obj_and_grad(h, n, alpha):
    alpha = float(alpha)
    if alpha < 1.0:
        alpha = 1.0
    dx = 2.0 / float(n)
    c_counts = np.correlate(h, 1.0 - h, mode="full")
    C = c_counts * dx
    cm = float(np.max(C))
    z = alpha * (C - cm)
    ez = np.exp(z)
    zs = float(ez.sum())
    w = ez / zs
    F = cm + math.log(zs) / alpha
    idx = np.where(w > 1e-15)[0]
    g = np.zeros(n)
    for t in idx:
        wt = float(w[t])
        s = int(t) - (n - 1)
        if s >= 0:
            r = s
            m = n - r
            if m > 0:
                g[r:n] += wt * (1.0 - h[:m])
                g[:m] -= wt * h[r:n]
        else:
            r = -s
            m = n - r
            if m > 0:
                g[:m] += wt * (1.0 - h[r:n])
                g[r:n] -= wt * h[:m]
    return float(F), g * dx

def obj_proxy(x):
    h = np.asarray(x, dtype=float)
    F, _ = smooth_obj_and_grad(h, _OPT["n"], _OPT["alpha"])
    mu = float(_OPT["mu"])
    if mu != 0.0:
        s = float(h.sum()) - float(_OPT["target"])
        F += 0.5 * mu * s * s
    return float(F)

def jac_proxy(x):
    h = np.asarray(x, dtype=float)
    _, g = smooth_obj_and_grad(h, _OPT["n"], _OPT["alpha"])
    mu = float(_OPT["mu"])
    if mu != 0.0:
        s = float(h.sum()) - float(_OPT["target"])
        g = g + mu * s * np.ones(_OPT["n"])
    return g

                    best_h = x.copy()
        except Exception:
            pass
    out = project_feasible(best_h, n)
    return out, evaluate_c5(out, n)

def pair_search(h, n, t_end, seed):
    h = project_feasible(h.copy(), n)
    rng = np.random.RandomState(seed)
    dx = 2.0 / float(n)
    c_counts, max_count = c_counts_max(h, n)
    best_h = h.copy()
    best_c = float(max_count * dx)

    temp = 0.0008
    min_temp = 1e-9
    B = 14
    P = 6
    K = 22
    no_imp = 0

    while time.time() < t_end:
        c_counts, max_count = c_counts_max(h, n)
        cur_c = float(max_count * dx)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        order = np.argsort(c_counts)[::-1]
        Ku = min(K, int(order.size))
        act = order[:Ku]
        c_act = c_counts[act]
        spread = float(c_act[0] - c_act[-1])
        tau = max(0.25, spread / 4.0)
        aw = np.exp((c_act - c_act[0]) / tau)
        aw = aw / float(aw.sum())

        g = np.zeros(n)
        L_act = np.empty((n, Ku))
        center = n - 1
        for a in range(Ku):
            t = int(act[a])
            s = t - center
            col = np.zeros(n)
            if s >= 0:
                r = s
                m = n - r
                if m > 0:
                    col[r:n] += 1.0 - h[:m]
                    col[:m] -= h[r:n]
            else:
                r = -s
                m = n - r
                if m > 0:
                    col[:m] += 1.0 - h[r:n]
                    col[r:n] -= h[:m]
            L_act[:, a] = col
            g += aw[a] * col

        inc = np.where(h < 1.0 - 1e-9)[0]
        dec = np.where(h > 1e-9)[0]
        if inc.size == 0 or dec.size == 0:
            break

        inc_sorted = inc[np.argsort(g[inc])][:B]
        dec_sorted = dec[np.argsort(-g[dec])][:B]
        cands = []

        for ii in range(len(inc_sorted)):
            i = int(inc_sorted[ii])
            for jj in range(len(dec_sorted)):
                j = int(dec_sorted[jj])
                if i == j:
                    continue
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo <= 1e-11:
                    continue
                basis = rank_basis(lo, hi)
                if len(basis) == 0:
                    continue
                arr = np.asarray(basis, dtype=float)
                arr = arr[(arr >= lo - 1e-12) & (arr <= hi + 1e-12)]
                if arr.size == 0:
                    continue
                dabs = abs(i - j)
                quad_act = np.zeros(Ku)
                if dabs > 0:
                    quad_act[act == (center + dabs)] += 1.0
                    quad_act[act == (center - dabs)] += 1.0
                quad_act[act == center] -= 2.0
                Dv = L_act[i] - L_act[j]
                sq = arr * arr
                vals = np.max(c_act[None, :] + Dv[None, :] * arr[:, None] + quad_act[None, :] * sq[:, None], axis=1)
                cands.append((float(np.min(vals)), i, j))

        if not cands:
            i = int(rng.choice(inc))
            j = int(rng.choice(dec))
            cands.append((1.0e100, i, j))

        cands.sort()
        exact = []
        seen = set()
        for _, i, j in cands:
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            exact.append(key)
            if len(exact) >= P:
                break

        rows = {}
        for i, j in exact:
            if i not in rows:
                rows[i] = L_row(i, h, n)
            if j not in rows:
                rows[j] = L_row(j, h, n)

        best_move = None
        best_cnt = np.inf
        Lc = 2 * n - 1

        for i, j in exact:
            D = rows[i] - rows[j]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-9:
                continue
            dabs = abs(i - j)
            quad = np.zeros(Lc)
            quad[center] = -2.0
            if dabs > 0:
                quad[center + dabs] += 1.0
                quad[center - dabs] += 1.0
            delta, cnt = search_delta_pair(c_counts, D, quad, lo, hi)
            if cnt < best_cnt - 1e-13:
                best_cnt = float(cnt)
                best_move = (i, j, float(delta), float(cnt))

        if (best_move is None or best_move[3] >= max_count - 1e-12) and (no_imp > 8 or rng.rand() < 0.12):
            i = int(rng.choice(inc))
            j = int(rng.choice(dec))
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo > 1e-8:
                delta = lo + (0.2 + 0.6 * rng.rand()) * (hi - lo)
                if abs(delta) < 0.005:
                    delta = 0.25 * hi if hi > 0 else 0.25 * lo
                dabs = abs(i - j)
                quad = np.zeros(Lc)
                quad[center] = -2.0
                if dabs > 0:
                    quad[center + dabs] += 1.0
                    quad[center - dabs] += 1.0
                D = L_row(i, h, n) - L_row(j, h, n)
                cnt = eval_delta_full(c_counts, D, quad, delta)
                if best_move is None or cnt < best_cnt - 1e-13:
                    best_cnt = float(cnt)
                    best_move = (i, j, float(delta), float(cnt))

        if best_move is not None:
            i, j, delta, cnt = best_move
            new_c = float(cnt * dx)
            if abs(delta) > 1e-10 and new_c <= cur_c + 10.0 * temp:
                if new_c < cur_c - 1e-12:
                    accept = True
                else:
                    accept = False
                    if temp > min_temp and new_c > cur_c:
                        z = min(60.0, (new_c - cur_c) / temp)
                        accept = (rng.rand() < math.exp(-z))
                if accept:
                    h[i] += delta
                    h[j] -= delta
                    np.clip(h, 0.0, 1.0, out=h)
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                    no_imp = 0
                    temp = max(min_temp, temp * 0.965)
                else:
                    no_imp += 1
            else:
                no_imp += 1

            if no_imp > 35:
                h = random_perturb(h, n, rng, 0.02)
                no_imp = 0
                temp = min(0.005, temp * 1.7)
        else:
            no_imp += 1
            if no_imp > 35:
                h = random_perturb(h, n, rng, 0.02)
                no_imp = 0

        temp = max(min_temp, temp * 0.998)

    h = project_feasible(h, n)
    c1 = evaluate_c5(h, n)
    b2 = project_feasible(best_h, n)
    c2 = evaluate_c5(b2, n)
    if c1 <= c2:
        return h, float(c1)
    return b2, float(c2)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(20.0, float(budget_s) - 10.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel().copy()
        n = int(h0.size)
        if n < 2:
            raise ValueError
    except Exception:
        n = 250
        h0 = make_random_balanced(n, rng)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h, n)

    t1 = t0 + 0.38 * total
    if time.time() < t1:
        hb, cb = pair_search(h0, n, t1, seed)
        if cb < best_c:
            best_c = float(cb)
            best_h = hb.copy()

    if best_c > 0.3808:
        t2 = t0 + 0.55 * total
        if time.time() < t2 - 50.0:
            hb, cb = nlp_optimize(best_h, n, t2 - 5.0, seed + 10)
            if cb < best_c:
                best_c = float(cb)
                best_h = hb.copy()

    if best_c > 0.3808:
        t3 = t0 + 0.72 * total
        if time.time() < t3 - 70.0:
            if n % 2 == 0:
                hb_bin = balanced_binary(best_h, n, rng)
                hb, cb = pair_search(hb_bin, n, time.time() - 1.0 + min(70.0, 0.10 * total), seed + 20)
                if cb < best_c:
                    best_c = float(cb)
                    best_h = hb.copy()
            if time.time() < t3 - 45.0:
                hb_tm = thue_morse_init(n, rng)
                hb, cb = pair_search(hb_tm, n, t3 - 5.0, seed + 30)
                if cb < best_c:
                    best_c = float(cb)
                    best_h = hb.copy()
            if time.time() < t3 - 25.0:
                hp = random_perturb(best_h, n, rng, 0.05)
                hb, cb = pair_search(hp, n, t3 - 2.0, seed + 40)
                if cb < best_c:
                    best_c = float(cb)
                    best_h = hb.copy()

    if best_c > 0.3808:
        t4 = t0 + 0.88 * total
        if time.time() < t4 - 90.0:
            hb, cb = nlp_optimize(best_h, n, t4 - 5.0, seed + 50)
            if cb < best_c:
                best_c = float(cb)
                best_h = hb.copy()

    if time.time() < deadline - 8.0:
        hb, cb = pair_search(best_h, n, deadline - 4.0, seed + 60)
        if cb < best_c:
            best_c = float(cb)
            best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = evaluate_c5(best_h, n)
    return best_h, float(best_c), int(n)
