def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_delta(h, n):
    """Precompute delta_pos: effect of setting h[i]=1 (from 0) on corr."""
    n2 = 2 * n - 1
    delta = np.zeros((n, n2))
    # For position i, corr[p] changes by (1-h[i+n-1-p]) if 0 <= i+n-1-p < n
    # i.e., p in [i, i+n-1] (clipped to [0, n2-1])
    for i in range(n):
        lo = max(0, i)
        hi = min(n2 - 1, i + n - 1)
        q_range = np.arange(lo, hi + 1)
        q_idx = i + n - 1 - q_range
        delta[i, q_range] = 1.0 - h[q_idx]
    return delta

def binary_optimize(h_start, n, deadline, seed=42, top_k=15):
    """Fast binary optimization with 1-for-1 swaps."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    
    delta = build_delta(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    no_improve = 0
    kicks_left = 100
    
    n2 = 2 * n - 1
    
    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        
        # Vectorized swap evaluation
        d_top = delta[:, top_idx]  # (n, top_k)
        c_top = corr[top_idx]  # (top_k,)
        
        base = c_top[None, :] + d_top[zeros]  # (nz, top_k)
        reds = d_top[ones]  # (no, top_k)
        
        # new_max[i,j] = max_p(base[i,p] - reds[j,p])
        diffs = base[:, :, None] - reds[None, :, :]
        new_max = diffs.max(axis=2)  # (nz, no)
        
        flat_min = new_max.argmin()
        ii = flat_min // len(ones)
        jj = flat_min % len(ones)
        best_val = new_max[ii, jj]
        
        i = zeros[ii]
        j = ones[jj]
        new_c = best_val * dx
        
        if new_c < cur_c - 1e-12:
            # Execute swap
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta[i] - delta[j]
            
            # Update delta incrementally
            for k in range(n):
                p1 = k + n - 1 - i
                if 0 <= p1 < n2:
                    delta[k, p1] = 1.0 - h[i]  # h[i]=1 now, so 0
                p2 = k + n - 1 - j
                if 0 <= p2 < n2:
                    delta[k, p2] = 1.0 - h[j]  # h[j]=0 now, so 1
            
            zeros = np.concatenate([zeros[:ii], zeros[ii+1:], [j]])
            ones = np.concatenate([ones[:jj], ones[jj+1:], [i]])
            
            cur_c = float(corr.max()) * dx
            stall = 0
            no_improve = 0
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        else:
            stall += 1
            no_improve += 1
            if (stall > 30 or no_improve > 500) and kicks_left > 0:
                kicks_left -= 1
                # Random kick
                ri = rng.randint(len(zeros))
                rj = rng.randint(len(ones))
                i_k = zeros[ri]
                j_k = ones[rj]
                h[i_k] = 1.0
                h[j_k] = 0.0
                corr = corr + delta[i_k] - delta[j_k]
                for k in range(n):
                    p1 = k + n - 1 - i_k
                    if 0 <= p1 < n2:
                        delta[k, p1] = 1.0 - h[i_k]
                    p2 = k + n - 1 - j_k
                    if 0 <= p2 < n2:
                        delta[k, p2] = 1.0 - h[j_k]
                zeros = np.concatenate([zeros[:ri], zeros[ri+1:], [j_k]])
                ones = np.concatenate([ones[:rj], ones[rj+1:], [i_k]])
                cur_c = float(corr.max()) * dx
                stall = 0
    
    return best_h, best_c

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

2. Continuous polish with SLSQP (multiple restarts, high alpha)
3. Return the best found

Let me also add one more idea: **the "fix the worst lag" heuristic**. After finding the best binary solution, identify the lag(s) with maximum correlation. Then find a small perturbation (in continuous space) that specifically reduces those lags while keeping others from increasing.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def build_delta(h, n):
    n2 = 2 * n - 1
    delta = np.zeros((n, n2))
    for i in range(n):
        lo = i
        hi = min(n2 - 1, i + n - 1)
        q = i + n - 1 - np.arange(lo, hi + 1)
        delta[i, lo:hi+1] = 1.0 - h[q]
    return delta
