import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def subgradient_descent(h, n, deadline, lr_init=0.01, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    rng = np.random.RandomState(123)
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        # Subgradient of max: pick the active lag(s)
        # Use a random subset of top lags for robustness
        top_k = 5
        idx = np.argsort(c)[-top_k:]
        weights = np.ones(len(idx)) / len(idx)
        # Compute subgradient
        grad = np.zeros(n)
        for w, k_idx in zip(weights, idx):
            lag = k_idx - (n - 1)
            if lag == 0:

            elif lag > 0:

                grad[lag:] += w * (1 - h[:n-lag])

    # f(t) = max_k (a_k * t + b_k) for t in [0,1]
    # where a_k = dx * (1 - h_{j+k}) [if j+k<n] + dx * (-h_{j-k}) [if j>=k]
    # and b_k = c_k - a_k * h[j] (current value)
    
    dx = 2.0 / n
    cur = h[j]

    # Lag L = k - (n-1) where k is the index in the full correlation array

    #   It contributes to lag L as h[j] in the "h[i]" role when i=j, i.e., the shifted index is j+L
    #   It contributes to lag L as h[j] in the "h[i+L]" role when i+L=j, i.e., i=j-L
    
    # Let's just compute f(t) directly for efficiency

        h_copy[j] = t

        return float(np.max(c))  # without dx since we're just comparing

For lag L > 0:
c[L] = sum_{i=0}^{n-1-L} h[i] * (1 - h[i+L])

∂c[L]/∂h[j] for a specific j:
- j appears as h[i] when i=j (valid if j < n-L): contributes (1 - h[j+L])
- j appears as h[i+L] when i=j-L (valid if j >= L and j-L < n-L, i.e., j >= L and j < n): contributes -h[j-L]

So ∂c[L]/∂h[j] = (1-h[j+L]) * 1[j<n-L] + (-h[j-L]) * 1[j>=L, j<n]

For j < L: only the first term (if j < n-L): ∂c[L]/∂h[j] = 1-h[j+L]
For L <= j < n-L: both terms: ∂c[L]/∂h[j] = (1-h[j+L]) - h[j-L]
For j >= n-L: only the second term (if j >= L): ∂c[L]/∂h[j] = -h[j-L]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

        grad[m:] += (-h[:n - m])
