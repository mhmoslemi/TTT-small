    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        return float(c.max() + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha)
    return float(c.max() + np.log(np.sum(np.exp(c_shift))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w /= w.sum()
    g = np.zeros(n)

        k = m - (n - 1)

            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])

            if tot < 1e-14:

def admm_optimize(h, n, deadline, rho=100, max_outer=1000):
    h = project_feasible(h, n)
    z = h.copy()
    u = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        # h-update: minimize smooth_max(h) + (rho/2)||h - z + u||^2
        # This is a smooth unconstrained problem
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
            cs = alpha * (c - c.max())
            mx = cs.max()
            if mx > 500:
                val = c.max() + (mx + np.log(np.sum(np.exp(cs - mx)))) / alpha
            else:
                val = c.max() + np.log(np.sum(np.exp(cs)))
            val += (rho/2) * np.sum((x - z + u)**2)
            return float(val)
        
        def grad(x):

            g += rho * (x - z + u)
            return g

For the local search, the most effective technique for min-max problems is:
- Identify the top-K active constraints (shifts with correlation close to max)
- Find a direction that decreases all active constraints simultaneously
- This is a **Chebyshev center** type problem

The direction that most decreases the maximum is given by solving:
minimize max_m (g_m · d)
subject to ||d|| ≤ 1
where g_m is the gradient of the m-th constraint.

The solution is d = -∑ w_m g_m / ||∑ w_m g_m|| where w_m are chosen to equalize the active constraints.

In practice, this is what the smooth max gradient does (with weights w_m from the softmax). So the smooth max gradient IS the right direction.

The issue is that after projecting onto the feasible set, the direction might not be as effective. And the line search might not find the right step size.

Let me try a different approach: **Frank-Wolfe (conditional gradient)** method.

The Frank-Wolfe method for:
minimize f(x) subject to x ∈ P

iterates:
1. s = argmin_{y ∈ P} ∇f(x) · y  (linear minimization oracle)
2. x ← (1-t)x + t*s

For our problem, P = {x : 0≤x[i]≤1, sum(x) = n/2}.

The linear minimization oracle: minimize ∇f(x) · y subject to 0≤y[i]≤1, sum(y) = n/2.

This can be solved by:
1. Sort the components of ∇f(x)
2. The optimal y has y[i] = 1 for the smallest gradient components, y[i] = 0 for the largest, and one component is fractional to satisfy the sum constraint.

This is much faster than a full optimization step!

def frank_wolfe(h, n, deadline, alpha=10000, max_iter=5000):

        # If n is odd: we need one fractional component

    penalty = 1e-4
