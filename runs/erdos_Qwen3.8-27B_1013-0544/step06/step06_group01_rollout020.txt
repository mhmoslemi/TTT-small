P(h) = clip(h - τ, 0, 1) where τ is chosen so that sum(P(h)) = c.

This is a standard result in convex optimization (projection onto a box-constrained simplex-like set).

Bisection on τ:
- If τ is very negative, clip(h-τ, 0, 1) → clip(∞, 0, 1) = 1, sum = n
- If τ is very positive, clip(h-τ, 0, 1) → clip(-∞, 0, 1) = 0, sum = 0
- We want sum = n/2, so τ is somewhere in between.

```python
def project_simplex(h, c):
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = c
    lo, hi = -1.0, 2.0  # bounds for tau
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lse_grad(h, alpha):
    n = len(h)
    c = compute_corr(h, n)
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n), c_max
    w /= w_sum
    grad = np.zeros(n)
    w_prefix = np.concatenate([[0.0], np.cumsum(w)])
    for i in range(n):
        lo_i = n - 1 - i
        hi_i = 2 * n - 1 - i
        A_i = w_prefix[hi_i] - w_prefix[lo_i]
        B_i = float(np.dot(w[lo_i:hi_i], h[:n - i])) if i < n - 1 else 0.0
        C_i = float(np.dot(w[i:i + n][::-1], h))
        grad[i] = A_i - B_i - C_i
    return grad, c_max

def lse_optimize(h_init, n, deadline, max_iter=300):
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    it = 0
    patience = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        progress = min(1.0, it / max(1, max_iter))
        alpha = 5.0 * (5000.0 / 5.0) ** progress
        grad, c_max = lse_grad(h, alpha)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        lr = 3.0 / (alpha * gnorm + 1e-12)
        accepted = False
        for _ in range(30):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_max * (2.0 / n) - 1e-15:
                h = h_new
                accepted = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                break
            lr *= 0.5
        if not accepted:
            patience += 1
        if patience > 40:
            break
    return best_h, best_c

def targeted_cd(h, n, deadline, max_passes=200):
    h = h.copy()
    c = compute_corr(h, n)
    best_c = float(np.max(c))
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Focus on lags near the maximum
        thresh = best_c - 0.02 * best_c
        active = np.where(c >= thresh)[0]
        for i in range(n):
            if time.time() > deadline:
                break
            # Sensitivity vector for changing h[i]
            d_lo = -h[i]
            d_hi = 1.0 - h[i]
            if d_hi - d_lo < 1e-14:
                continue
            # Compute sensitivity s[k] = dc[k]/dh[i]
            s = np.zeros(2 * n - 1)
            m_lo = max(0, -(i))
            m_hi = min(n - 1, n - 1 - i)
            for m in range(m_lo, m_hi + 1):
                k = m + n - 1
                s[k] += (1.0 - h[i + m])
            m2_lo = max(0, i - (n - 1))
            m2_hi = min(n - 1, i)
            for m in range(m2_lo, m2_hi + 1):
                k = m + n - 1
                s[k] -= h[i - m]
            # Find best d via grid search on active lags
            deltas = np.linspace(d_lo, d_hi, 30)
            best_d = 0.0
            best_val = best_c
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                # Quick check: max over active lags
                new_c_act = c[active] + d * s[active]
                val = float(np.max(new_c_act))
                # Also check non-active lags could become active
                new_c_full = c + d * s
                val = max(val, float(np.max(new_c_full[~np.isin(np.arange(2*n-1), active)])) if len(active) < 2*n-2 else val)
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                c += best_d * s
                # Correct the diagonal term at k=n-1
                c[n-1] -= best_d * best_d
                best_c = max(best_c, float(np.max(c)))
                # Recompute exact c to avoid drift
                if _pass % 10 == 0:
                    c = compute_corr(h, n)
                    best_c = float(np.max(c))
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def _obj_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha) * (2.0 / n))
    return float((m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n))

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
