```python
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    n_corr = 2 * n - 1
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            m_lo1 = max(0, n - 1 - p)
            m_hi1 = min(n_corr - 1, n - 1 - p + n - 1)
            m_lo2 = max(0, p - (n - 1))
            m_hi2 = min(n_corr - 1, p)
            m_lo = min(m_lo1, m_lo2)
            m_hi = max(m_hi1, m_hi2)
            
            um = 0.0
            if m_lo > 0:
                um = float(np.max(c[:m_lo]))
            if m_hi < n_corr - 1:
                um = max(um, float(np.max(c[m_hi+1:])))
            
            ks = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ks))
            for i, m in enumerate(ks):
                k = m - (n - 1)
                if 0 <= p + k < n:
                    a[i] += (1.0 - h[p + k])
                if 0 <= p - k < n:
                    a[i] -= h[p - k]
            is_center = (ks == n - 1).astype(float)
            
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            
            deltas = np.linspace(d_lo, d_hi, grid)
            # Vectorized evaluation: (grid, len(ks))
            vals = c[ks][np.newaxis, :] + deltas[:, np.newaxis] * a[np.newaxis, :] - (deltas[:, np.newaxis]**2) * is_center[np.newaxis, :]
            max_vals = np.maximum(um, vals.max(axis=1))
            best_idx = np.argmin(max_vals)
            best_d = deltas[best_idx]
            best_val = max_vals[best_idx]
            
            if best_val < current_max - 1e-12 and abs(best_d) > 1e-12:
                c[ks] += best_d * a - best_d**2 * is_center
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                current_max = float(max(um, np.max(c[ks])))
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
```

```python
def subgradient_fast(h, n, c=None, eps=0.001):
    if c is None:
        c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    mask = c >= cmax - eps
    active_idx = np.where(mask)[0]
    n_active = len(active_idx)
    
    k_vals = active_idx - (n - 1)  # lag values
    
    # For each j in 0..n-1:
    # sg[j] = mean over k in k_vals of:
    #   (1-h[j+k]) * [0≤j+k<n] - h[j-k] * [0≤j-k<n]
    
    # Vectorize: create j_col (n,1) and k_row (1, n_active)
    j_col = np.arange(n)[:, np.newaxis]  # (n, 1)
    k_row = k_vals[np.newaxis, :]  # (1, n_active)
    
    # First term: (1-h[j+k]) * [0≤j+k<n]
    idx1 = j_col + k_row  # (n, n_active)
    valid1 = (idx1 >= 0) & (idx1 < n)
    vals1 = np.zeros_like(idx1, dtype=float)
    np.subtract(1.0, h[idx1[valid1]], out=vals1[valid1])
    
    # Second term: h[j-k] * [0≤j-k<n]
    idx2 = j_col - k_row  # (n, n_active)
    valid2 = (idx2 >= 0) & (idx2 < n)
    vals2 = np.zeros_like(idx2, dtype=float)
    np.copyto(vals2[valid2], h[idx2[valid2]])
    
    sg = (vals1.sum(axis=1) - vals2.sum(axis=1)) / n_active
    return sg, cmax * (2.0/n)
```
