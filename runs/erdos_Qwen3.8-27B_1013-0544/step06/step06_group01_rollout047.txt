def try_construction_block(n, n_blocks, fill_ratio):

def compute_corr(h):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_val(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h) * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(lse)

def slsqp_with_grad(h, n, alpha, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]

    h = h.copy()

    # Phase D: final coordinate descent
    if time.time() < deadline - 3:
        h_f, c_f = coordinate_descent_fast(h, n, deadline - 2, grid=400)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            h, best_c = h_f, c_f
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def make_sine_init(n, freq, phase, amp=0.15):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return project_feasible(h, n)

def make_block_init(n, n_blocks, fill=0.5):
    h = np.zeros(n)
    blen = n // n_blocks
    for i in range(n_blocks):
        start = i * blen
        end = min(start + blen, n)
        fill_n = int((end - start) * fill)
        h[start:start + fill_n] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Main optimization on initial (40% of budget)
    alloc_main = (budget_s - 10) * 0.40
    h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc_main, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt
    
    # Try different n values (30% of budget)
    t_phase2_end = t0 + (budget_s - 10) * 0.70
    for n_try in [250, 200, 350, 400]:
        if time.time() > t_phase2_end:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h, period=2), n_try)
        alloc = min(60, t_phase2_end - time.time())
        if alloc < 30:
            break
        h_t, c_t = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
    
    # Multi-restart (20% of budget)
    t_phase3_end = t0 + (budget_s - 10) * 0.90
    n_best = len(best_h)
    remaining = t_phase3_end - time.time()
    n_restarts = max(3, int(remaining / 50))
    for r in range(n_restarts):
        if time.time() > t_phase3_end - 30:
            break
        # Diverse perturbations
        scale = 0.02 + 0.04 * (r + 1) / n_restarts
        if r % 4 == 0:
            pert = rng.normal(0, scale, n_best)
        elif r % 4 == 1:
            x = np.linspace(0, 2, n_best, endpoint=False)
            pert = 0.06 * np.sin(2 * np.pi * (1 + r) * x)
        elif r % 4 == 2:
            x = np.linspace(0, 2, n_best, endpoint=False)
            pert = 0.06 * np.cos(2 * np.pi * (2 + r) * x + r * 0.5)
        else:
            pert = rng.uniform(-scale, scale, n_best)
        hp = project_feasible(best_h + pert, n_best)
        alloc = min(45, t_phase3_end - time.time() - 20)
        if alloc < 20:
            break
        h_r, c_r = full_optimize(hp, n_best, time.time() + alloc, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    
    # Final polish (10% of budget)
    if time.time() < deadline - 10:
        n = len(best_h)
        # Final SLSQP with high alpha
        h_f = slsqp_with_grad(best_h, n, alpha=500000, maxiter=3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        # Final coordinate descent
        if time.time() < deadline - 3:
            best_h, best_c = coordinate_descent_fast(best_h, n, deadline - 2, grid=500)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
