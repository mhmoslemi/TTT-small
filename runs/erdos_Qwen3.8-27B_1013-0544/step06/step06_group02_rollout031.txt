import numpy as np
import time
from scipy.optimize import minimize

def lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5(h, n):
    return float(np.max(lags(h, n)))

def sum_fix(z, t):
    z = np.asarray(z, dtype=float)
    for _ in range(200):
        d = t - float(z.sum())
        if abs(d) < 1e-15:
            break
        if d > 0:
            i = int(np.argmax(1.0 - z))
            z[i] = min(1.0, z[i] + d)
        else:
            i = int(np.argmax(z))
            z[i] = max(0.0, z[i] + d)
    return z

def project_bisect(y, n):
    y = np.asarray(y, dtype=float)
    t = 0.5 * n
    z = np.clip(y, 0.0, 1.0)
    if abs(float(z.sum()) - t) < 1e-12:
        return sum_fix(z, t)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(45):
        m = 0.5 * (lo + hi)
        if float(np.clip(y - m, 0.0, 1.0).sum()) > t:
            lo = m
        else:
            hi = m
    return sum_fix(np.clip(y - 0.5 * (lo + hi), 0.0, 1.0), t)

def project_fast(y, n):
    z = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(24):
        d = t - float(z.sum())
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = z < 1.0
            if not m.any():
                break
            z[m] = np.minimum(1.0, z[m] + d / float(m.sum()))
        else:
            m = z > 0.0
            if not m.any():
                break
            z[m] = np.maximum(0.0, z[m] + d / float(m.sum()))
    z = np.clip(z, 0.0, 1.0)
    if abs(float(z.sum()) - t) > 1e-8:
        return project_bisect(y, n)
    return sum_fix(z, t)

def weights(L, mode, strength):
    m = float(np.max(L))
    i = int(np.argmax(L))
    if strength <= 0.0 or not np.isfinite(strength):
        w = np.zeros(L.shape, dtype=float)
        w[i] = 1.0
        return w
    if mode == 1:
        mask = L >= (m - strength)
        if not mask.any():
            mask[i] = True
        w = mask.astype(float)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros(L.shape, dtype=float)
            w[i] = 1.0
        else:
            w /= s
        return w
    z = (L - m) / strength
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300:
        w = np.zeros(L.shape, dtype=float)
        w[i] = 1.0
        return w
    return e / s

def soft_value_weight(L, beta):
    m = float(np.max(L))
    i = int(np.argmax(L))
    if beta <= 0.0:
        w = np.zeros(L.shape, dtype=float)
        w[i] = 1.0
        return m, w
    z = beta * (L - m)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300:
        w = np.zeros(L.shape, dtype=float)
        w[i] = 1.0
        return m, w
    return m + float(np.log(s)) / beta, e / s

def grad_weighted(h, n, w):
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=float)
    wp = np.empty(n, dtype=float)
    wm = np.empty(n, dtype=float)
    wp[0] = w[n - 1]
    if n > 1:
        wp[1:] = w[:n - 1][::-1]
    wm[0] = 0.0
    if n > 1:
        wm[1:] = w[n:]
    pp = np.cumsum(wp)
    pm = np.cumsum(wm)
    g = (pp[::-1]
         - np.convolve(h, wp[::-1], mode='full')[n - 1:]
         - np.convolve(h, wp, mode='full')[:n]
         + pm
         - np.convolve(h, wm, mode='full')[:n]
         - np.convolve(h, wm[::-1], mode='full')[n - 1:])
    return (2.0 / (n * s)) * g

def lbfgs_obj(x, n, beta, mu):
    if not np.all(np.isfinite(x)):
        return 1e100
    L = lags(x, n)
    val, _ = soft_value_weight(L, beta)
    if mu != 0.0:
        r = float(x.sum() - 0.5 * n)
        val += 0.5 * mu * r * r
    return float(val)

def lbfgs_grad(x, n, beta, mu):
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    L = lags(x, n)
    _, w = soft_value_weight(L, beta)
    g = grad_weighted(x, n, w)
    if mu != 0.0:
        g = g + mu * float(x.sum() - 0.5 * n)
    return g

def lbfgs_run(h, n, beta, mu, maxiter):
    h = np.asarray(h, dtype=float).copy()
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, beta, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-8, 'maxfun': maxiter * 10, 'maxls': 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_bisect(res.x, n)
            return h2, c5(h2, n)
    except Exception:
        pass
    return h, c5(h, n)

def slsqp_obj(x, n, beta):
    if not np.all(np.isfinite(x)):
        return 1e100
    val, _ = soft_value_weight(lags(x, n), beta)
    return float(val)

def slsqp_grad(x, n, beta):
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    _, w = soft_value_weight(lags(x, n), beta)
    return grad_weighted(x, n, w)

def slsqp_eq(x, n):
    return float(x.sum() - 0.5 * n)

def slsqp_eq_jac(x, n):
    return np.ones((1, n), dtype=float)

def slsqp_run(h, n, beta, maxiter):
    h = np.asarray(h, dtype=float).copy()
    try:
        bounds = [(0.0, 1.0)] * n
        cons = ({'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},)
        res = minimize(slsqp_obj, h, jac=slsqp_grad, args=(n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_bisect(res.x, n)
            return h2, c5(h2, n)
    except Exception:
        pass
    return h, c5(h, n)

def pdescent(h, n, dl, strength0, strengthmin, max_it, mode, seed):

    rng = np.random.RandomState(seed)
    L = lags(h, n)
    bc = float(np.max(L))
    bh = h.copy()

    strength = strength0
    no = 0

                break

def ppairs(h, n, dl, top, strength, seed):

                    cv = float(np.max(np.correlate(h, inv, mode='full')) * dx)

                    else:

            inv[src] -= m
            h[dst] -= m
            inv[dst] += m
            stall += 1
    h = project_bisect(h, n)
    return h, c5(h, n)

def resample_h(h_old, n_old, n_new):
    if n_old == n_new:
        return project_bisect(h_old, n_new)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_bisect(np.interp(xn, xo, h_old), n_new)
