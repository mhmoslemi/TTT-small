    mx = s.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    return float(val * dx)

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def full_optimize(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    c = compute_c5(h, n)
    # CD moderate
    t1 = min(120, (deadline - time.time()) * 0.3)
    if t1 > 20:
        h2 = cd_opt(h, n, 5000, 200, time.time() + t1, 50.0)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < c:
            h, c = h2, c2
    # SLSQP
    if time.time() < deadline - 60:
        hs = slsqp_opt(h, n, 300000, 3000)
        cs = compute_c5(hs, n)
        if cs < c:
            h, c = hs, cs
    # CD fine
    t2 = min(120, (deadline - time.time()) * 0.3)
    if t2 > 20:
        h3 = cd_opt(h, n, 5000, 350, time.time() + t2, 200.0)
        h3 = project_feasible(h3, n)
        c3 = compute_c5(h3, n)
        if c3 < c:
            h, c = h3, c3
    # SLSQP again
    if time.time() < deadline - 40:
        hs = slsqp_opt(h, n, 800000, 3000)
        cs = compute_c5(hs, n)
        if cs < c:
            h, c = hs, cs
    # CD very fine
    t3 = min(80, (deadline - time.time()) * 0.4)
    if t3 > 15:
        h4 = cd_opt(h, n, 5000, 500, time.time() + t3, 500.0)
        h4 = project_feasible(h4, n)
        c4 = compute_c5(h4, n)
        if c4 < c:
            h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    h_best = project_feasible(h_init, n)
    c_best = compute_c5(h_best, n)
    # Main optimization on initial
    h_opt, c_opt = full_optimize(h_init, n, time.time() + 500, rng)
    if c_opt < c_best:
        h_best, c_best = h_opt, c_opt
    # Try different n
    for n_try in [250, 350]:
        if time.time() > deadline - 120:
            break
        x_init = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_init, h_best), n_try)
        alloc = min(100, (deadline - time.time()) * 0.25)
        if alloc < 40:
            break
        h2, c2 = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c2 < c_best:
            h_best, c_best, n = h2, c2, n_try
    # Perturbation restarts
    if time.time() < deadline - 80:
        for trial in range(5):
            if time.time() > deadline - 40:
                break
            scale = 0.005 + 0.01 * trial
            hp = project_feasible(h_best + rng.normal(0, scale, n), n)
            alloc = min(50, (deadline - time.time() - 20) * 0.4)
            if alloc < 20:
                break
            hr, cr = full_optimize(hp, n, time.time() + alloc, rng)
            if cr < c_best:
                h_best, c_best = hr, cr
    # Final polish
    if time.time() < deadline - 25:
        hf = slsqp_opt(h_best, n, 1500000, 4000)
        cf = compute_c5(hf, n)
        if cf < c_best:
            h_best, c_best = hf, cf
    if time.time() < deadline - 5:
        hf = cd_opt(h_best, n, 5000, 600, deadline - 2, 1000.0)
        hf = project_feasible(hf, n)
        cf = compute_c5(hf, n)
        if cf < c_best:
            h_best, c_best = hf, cf
    h_best = project_feasible(h_best, n)
    c_best = compute_c5(h_best, n)
    return h_best, c_best, n
