def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=4000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    def obj(x):
        return smooth_max(compute_all_lags(x, n), alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_local_search(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    scales = [0.003, 0.0015, 0.0007, 0.0003, 0.0001, 0.00005]
    for scale in scales:
        if time.time() > deadline:
            break
        progress = True
        while progress and time.time() < deadline:
            progress = False
            c = compute_all_lags(h_curr, n)
            top_idx = np.argsort(c)[-4:]
            for k in top_idx:
                lag = k - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                if lag > 0:
                    for i in range(n - m):
                        j = i + m
                        if h_curr[i] > h_curr[j] + 0.003:
                            delta = min(scale * 4, (h_curr[i] - h_curr[j]) * 0.4)
                            if delta < 1e-9:
                                continue
                            h_curr[i] -= delta
                            h_curr[j] += delta
                            c_new = compute_c5(h_curr, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                progress = True
                            else:
                                h_curr[i] += delta
                                h_curr[j] -= delta
                else:
                    for i in range(m, n):
                        j = i - m
                        if h_curr[i] > h_curr[j] + 0.003:
                            delta = min(scale * 4, (h_curr[i] - h_curr[j]) * 0.4)
                            if delta < 1e-9:
                                continue
                            h_curr[i] -= delta
                            h_curr[j] += delta
                            c_new = compute_c5(h_curr, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                progress = True
                            else:
                                h_curr[i] += delta
                                h_curr[j] -= delta
    return h_curr, best_c

def optimize_one(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [5000, 15000, 40000, 100000, 200000, 400000]
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_refine(best_h, n, alpha, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 10:
        h_new, c_new = fine_local_search(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Optimize from initial
    alloc1 = min(250, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_one(h_init, n, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try mirror
    if time.time() < deadline - 200:
        h_mirror = 1.0 - best_h
        h_mirror = project_feasible(h_mirror, n)
        alloc2 = min(120, (deadline - time.time()) * 0.2)
        h_opt, c_opt = optimize_one(h_mirror, n, time.time() + alloc2, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Try perturbations
    if time.time() < deadline - 100:
        for attempt in range(8):
            if time.time() > deadline - 70:
                break
            scale = 0.008 + 0.006 * attempt
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc3 = min(60, (deadline - time.time() - 50) * 0.35)
            if alloc3 < 25:
                break
            h_opt, c_opt = optimize_one(hp, n, time.time() + alloc3, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try different n (coarser)
    for n_alt in [200, 250]:
        if time.time() > deadline - 100:
            break
        if n_alt == n:
            continue
        x_best = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_alt)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_alt)
        alloc4 = min(80, (deadline - time.time()) * 0.2)
        if alloc4 < 30:
            break
        h_opt, c_opt = optimize_one(h_start, n_alt, time.time() + alloc4, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_alt

    # Phase 5: Final aggressive polish on best solution
    n = len(best_h)
    if time.time() < deadline - 30:
        h_new = slsqp_refine(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 3:
        h_new, c_new = fine_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
