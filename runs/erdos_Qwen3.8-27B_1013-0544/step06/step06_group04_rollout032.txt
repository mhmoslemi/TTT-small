```python
    W2 = w[k2]  # (n, n)
    
    # grad[m] = sum_p W1[m,p]*(1-h[p]) - sum_p W2[m,p]*h[p]
    grad = W1 @ (1.0 - h) - W2 @ h
    grad *= dx
    
    return grad, max_c
```

This is beautiful! The gradient computation is just two matrix-vector products, O(n^2) with good cache locality. For n=250, this is essentially instant.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def init_f(h, n):
    nt1 = 2 * n - 1
    off = np.arange(nt1) - (n - 1)
    f = np.zeros((n, nt1))
    for p in range(n):
        m1 = (off >= -p) & (off <= n - 1 - p)
        f[p, m1] += h[p + off[m1]]
        m2 = (off >= p - (n - 1)) & (off <= p)
        f[p, m2] -= (1.0 - h[p - off[m2]])
    return f

def apply_swap(h, f, c, i, j, n):
    p = np.arange(n)
    f[:, i - p + n - 1] += 1.0
    f[:, p - i + n - 1] += 1.0
    f[:, j - p + n - 1] -= 1.0
    f[:, p - j + n - 1] -= 1.0
    c = c + f[j] - f[i]
    h[i] = 1.0
    h[j] = 0.0
    return h, c

def best_improvement_ls(h, n, deadline, max_iters=5000, seed=42):
    h = binarize(h, n).copy()
    c = get_corr(h, n).astype(float).copy()
    f = init_f(h, n)
    dx = 2.0 / n
    cur_c = float(c.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        base = c[None, :] - f[zeros][:, :]
        scores = np.max(base[:, None, :] + f[ones][None, :, :], axis=2)
        flat_idx = int(np.argmin(scores))
        i_idx = flat_idx // len(ones)
        j_idx = flat_idx % len(ones)
        i = zeros[i_idx]
        j = ones[j_idx]
        new_c_max = scores[i_idx, j_idx]
        if new_c_max >= cur_c - 1e-14:
            stuck += 1
            if stuck > 30:
                for _ in range(rng.randint(3, 8)):
                    ri = int(rng.choice(zeros))
                    rj = int(rng.choice(ones))
                    h, c = apply_swap(h, f, c, ri, rj, n)
                    f = init_f(h, n)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_c = float(c.max()) * dx
                stuck = 0
            continue
        h, c = apply_swap(h, f, c, int(i), int(j), n)
        f = init_f(h, n)
        cur_c = float(c.max()) * dx
        stuck = 0
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    ex = np.exp(alpha * (c - max_c))
    w = ex / ex.sum()
    m_idx = np.arange(n)[:, None]
    p_idx = np.arange(n)[None, :]
    W1 = w[m_idx + n - 1 - p_idx]
    W2 = w[p_idx + n - 1 - m_idx]
    grad = (W1 @ (1.0 - h) - W2 @ h) * dx
    return grad, max_c

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    mc = c.max()
    return mc + np.log(np.exp(alpha * (c - mc)).sum()) / alpha

def proj_grad_descent(h, n, alpha, deadline, lr_init=1.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    n_target = n / 2.0
    for it in range(20000):
        if time.time() > deadline:
            break
        grad, _ = smooth_gradient(h, n, alpha)
        f0 = smooth_obj(h, n, alpha)
        ls_ok = False
        lr_try = lr
        for _ in range(25):
            h_new = project_box_sum(h - lr_try * grad, n_target)
            f1 = smooth_obj(h_new, n, alpha)
            if f1 <= f0 - 1e-4 * lr_try * np.dot(grad, h_new - h) + 1e-12:
                ls_ok = True
                break
            lr_try *= 0.5
        h = h_new if ls_ok else project_box_sum(h - 0.01 * grad, n_target)
        actual_c = compute_c5(h, n)
        if actual_c < best_c:
            best_c = actual_c
            best_h = h.copy()
        if ls_ok:
            lr = min(lr * 1.2, 5.0)
        else:
            lr *= 0.7
    return best_h, best_c

def subgrad_descent(h, n, deadline, lr_init=0.5, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_target = n / 2.0
    dx = 2.0 / n
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = get_corr(h, n) * dx
        k_star = int(np.argmax(c))
        off = k_star - (n - 1)
        grad = np.zeros(n)
        for m in range(n):
            p1 = m - off
            if 0 <= p1 < n:
                grad[m] += (1.0 - h[p1])
            p2 = m + off
            if 0 <= p2 < n:
                grad[m] -= h[p2]
        h_new = project_box_sum(h - lr * grad, n_target)
        actual_c = compute_c5(h_new, n)
        if actual_c < best_c:
            best_c = actual_c
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.999, 0.01)
    return best_h, best_c
```
