from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    act = cc[idx]
    m = act.max()
    s = alpha * (act - m)
    if s.max() > 400:
        return float(m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]
            g[m:] += 1.0 - h[:n-m]
        else:
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g

def outer_step(h, n, trust, n_top=15):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_k = np.argsort(c)[-n_top:]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1,
            cp.abs(hv - h) <= trust]
    for k in top_k:
        lag = k - (n-1)
        g = lag_grad(h, n, lag)
        cons.append(c[k] + dx*(g @ (hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    hn = hv.value
    if hn is None:
        return h.copy()
    return project_feasible(hn, n)

def outer_loop(h, n, deadline, max_it=40):
    hc = h.copy()
    bh = hc.copy()
    bc = compute_c5(hc, n)
    tr = 0.2
    for _ in range(max_it):
        if time.time() > deadline - 2:
            break
        hn = outer_step(hc, n, tr)
        cn = compute_c5(hn, n)
        if cn < bc - 1e-13:
            bc = cn
            bh = hn.copy()
        step = np.max(np.abs(hn - hc))
        if step < 1e-9:
            break
        hc = hn
        tr = max(0.002, tr * 0.82)
    return bh, bc

def active_lbfgs(h, n, top_m=5, alpha=50000, maxiter=4000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max, h, args=(n, top_idx, alpha),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    # Fallback to SLSQP
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    bc = compute_c5(h, n)
    dx = 2.0 / n
    # Multi-scale local search
    scales = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in scales:
        if time.time() > deadline:
            break
        for _ in range(500):  # Max iterations per scale
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            # Gradient step in direction that reduces the active lag
            g = lag_grad(h, n, lag)
            # Project gradient onto feasible directions
            # Subtract mean to maintain sum constraint
            g_centered = g - g.mean()
            h_trial = project_feasible(h + d * g_centered, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < bc - 1e-14:
                bc = c_trial
                h = h_trial
                continue
            # Pairwise swaps
            found = False
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                di = h[i] - h[j]
                if abs(di) < 1e-10:
                    continue
                delta = d * 0.3
                if di > 0:
                    delta = min(delta, di * 0.5)
                    h[i] -= delta; h[j] += delta
                else:
                    delta = min(delta, -di * 0.5)

                c_new = compute_c5(h, n)

                    found = True
                    break
                else:
                    if di > 0:
                        h[i] += delta; h[j] -= delta
                    else:
                        h[i] -= delta; h[j] += delta
            if not found:
                break
    return h, bc

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def proj(h, n):
    ts = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def smax(x, n, idx, alpha):
    cc = all_lags(x, n)
    a = cc[idx]
    m = a.max()
    s = alpha * (a - m)
    if s.max() > 400:
        return float(m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def sc(x, n):
    return float(x.sum() - n / 2.0)

def sj(x, n):
    return np.ones(n)

def lg(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]
            g[m:] += 1.0 - h[:n-m]
        else:
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g

def ostep(h, n, tr, nt=12):
    dx = 2.0 / n
    c = all_lags(h, n)
    tk = np.argsort(c)[-nt:]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1, cp.abs(hv - h) <= tr]
    for k in tk:
        lag = k - (n-1)
        g = lg(h, n, lag)
        cons.append(c[k] + dx*(g @ (hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    hn = hv.value
    if hn is None:
        return h.copy()
    return proj(hn, n)

def oloop(h, n, dl, mi=35):
    hc = h.copy()
    bh = hc.copy()
    bc = compute_c5(hc, n)
    tr = 0.15
    for _ in range(mi):

        hn = ostep(hc, n, tr)
        cn = compute_c5(hn, n)

            bh = hn.copy()
        st = np.max(np.abs(hn - hc))
        if st < 1e-9: break
        hc = hn
        tr = max(0.002, tr * 0.8)
    return bh, bc

def abl(h, n, tm=5, alpha=50000, mi=4000):
    c = all_lags(h, n)
    ti = np.argsort(c)[-tm:]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smax, h, args=(n, ti, alpha), method='L-BFGS-B',
                    bounds=bnds, options={'maxiter': mi, 'ftol': 1e-17})
        if r.x is not None:

    except Exception:

    cons = [{'type': 'eq', 'fun': sc, 'jac': sj, 'args': (n,)}]

    except Exception:
        pass
    return h.copy()

def lsearch(h, n, dl):
    h = h.copy()

    dx = 2.0 / n

def fft_filter(h, keep):
