import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

def _obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 400:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sc(x, n):
    return float(x.sum() - n / 2.0)

def _sjac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, mit=3000, ti=None):
    b = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _sc, 'jac': _sjac, 'args': (n,)}]
    try:
        r = minimize(_obj, h, args=(n, alpha, ti), method='SLSQP',
                     bounds=b, constraints=cs,
                     options={'maxiter': mit, 'ftol': 1e-14})
        return project_feasible(r.x, n)
    except:
        return h.copy()

def lp_solve(h, n, trust, top_k=60):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    idx = np.argsort(c)[-top_k:]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cs = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1, cp.abs(hv - h) <= trust]
    for ki in idx:
        lag = int(ki) - (n - 1)
        g = _lag_grad(h, n, lag)
        cs.append(c[ki] + dx * (g @ (hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cs)
    for sv in [cp.PDLP, cp.GLOP, cp.CBC]:
        try:
            prob.solve(solver=sv, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                v = hv.value
                if v is not None:
                    return project_feasible(v, n)
        except:
            pass
    return h.copy()

def outer_loop(h, n, deadline, top_k=60):
    best_h, best_c = h.copy(), compute_c5(h, n)
    trust = 0.15
    for _ in range(60):
        if time.time() > deadline - 2:
            break
        prev = h.copy()
        h_new = lp_solve(h, n, trust, top_k)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        step = np.max(np.abs(h - prev))
        if step < 1e-9:
            break
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def grad_desc(h, n, deadline, top_m=5, mu=0.8):
    best_h, best_c = h.copy(), compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.01
    for _ in range(200000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        ti = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for ki in ti:
            g += _lag_grad(h, n, int(ki) - (n - 1))
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        v = mu * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
        else:
            v *= 0.5
            lr *= 0.9
            if lr < 1e-10:
                break
    return best_h, best_c

def local_search(h, n, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        imp = True
        while imp:
            if time.time() > deadline:
                break
            imp = False
            c = compute_all_lags(h, n)
            k = int(np.argmax(c))
            lag = k - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(0, n - m, 2):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 0.01:
                    amt = min(d, (h[i] - h[j]) * 0.4)
                    if amt < 1e-8:
                        continue
                    h[i] -= amt
                    h[j] += amt
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        best_h = h.copy()
                        imp = True
                    else:
                        h[i] += amt
                        h[j] -= amt
                elif h[j] > h[i] + 0.01:
                    amt = min(d, (h[j] - h[i]) * 0.4)
                    if amt < 1e-8:
                        continue
                    h[i] += amt
                    h[j] -= amt
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        best_h = h.copy()
                        imp = True
                    else:
                        h[i] -= amt
                        h[j] += amt
    return best_h, best_c

def optimize_n(h0, n, deadline):
    h = project_feasible(h0.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    
    # Phase 1: Fast gradient descent
    if time.time() < deadline - 30:
        t1 = min(time.time() + 40, deadline - 30)
        h, c = grad_desc(h, n, t1)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 2: LP outer approximation
    if time.time() < deadline - 30:
        t2 = min(time.time() + 60, deadline - 30)
        h, c = outer_loop(h, n, t2)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 3: SLSQP with increasing alpha
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline - 20:
            break
        h_new = slsqp(best_h, n, alpha, mit=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    # Phase 4: Active set SLSQP
    if time.time() < deadline - 15:
        c_all = compute_all_lags(best_h, n)
        top = np.argsort(c_all)[-8:]
        for alpha in [100000, 500000]:
            if time.time() > deadline - 10:
                break
