    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search_best(h, n, deadline, max_sweeps=100, seed=42):
    """Best-improvement local search."""
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_d = None
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max())
                if nc < best_nc:
                    best_nc = nc
                    best_j = int(j)
                    best_d = d
            if best_j >= 0 and best_nc < cur_c:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + best_d
                cur_c = best_nc
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c * dx
