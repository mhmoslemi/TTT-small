        p1[v1] = 1 - h[j[v1] + d]
        p2 = np.zeros(n)
        v2 = (j - d >= 0) & (j - d < n)
        p2[v2] = h[j[v2] - d]
        G[i] = p1 - p2
    
    # LP: minimize t
    # Variables: [t, Δh_0, ..., Δh_{n-1}] (n+1 variables)
    # Constraints:
    #   G @ Δh - t ≤ -c[active]  (i.e., G @ Δh + c[active] ≤ t)
    #   -h ≤ Δh ≤ 1-h (bound constraints)
    #   sum(Δh) = 0 (equality constraint)
    
    # In scipy linprog form: A_ub @ x ≤ b_ub, A_eq @ x = b_eq
    # x = [t, Δh]
    
    # G @ Δh - t ≤ -c[active]
    # [−1, G] @ [t, Δh] ≤ −c[active]
    A_ub = np.zeros((n_active, n+1))
    A_ub[:, 0] = -1
    A_ub[:, 1:] = G
    b_ub = -c[active]
    
    # sum(Δh) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1
    b_eq = [0]
    
    # Bounds: t is free, Δh_j ∈ [-h_j, 1-h_j]
    bounds = [(None, None)] + [(-h[i], 1-h[i]) for i in range(n)]
    
    res = linprog(c=np.concatenate([[1], np.zeros(n)]),
                  A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        dh = res.x[1:]
        h_new = np.clip(h + dh, 0, 1)
        # Re-project to ensure sum constraint (numerical issues)
        h_new = project_feasible(h_new, n)
        return h_new
    return h

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_init, n, deadline, n_restarts=20, perturb_size=3, seed=42):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        
        if restart == 0:
            h = h_init.copy()
        else:
            h = best_h.copy()
            # Perturb: flip k random bits
            k = rng.randint(1, perturb_size + 1)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(k):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                oi = rng.choice(ones)
                zi = rng.choice(zeros)
                h[oi] = 0
                h[zi] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        
        c = np.correlate(h, 1-h, mode='full')
        M = c.max()
        local_best = M
        local_best_h = h.copy()
        no_improve = 0
        max_no_improve = 50000
        
        while time.time() < deadline and no_improve < max_no_improve:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = h[b], h[a]
            new_M = np.correlate(h, 1-h, mode='full').max()
            if new_M < M:
                M = new_M
                no_improve = 0
                if M < local_best:
                    local_best = M
                    local_best_h = h.copy()
                    if M * dx < best_c:
                        best_c = M * dx
                        best_h = h.copy()
            else:
                h[a], h[b] = h[b], h[a]
                no_improve += 1
        
        # After local search, if we found something better, continue
        if local_best > best_c / dx:
            # Reset to best
            pass
    
    return best_h, best_c

  - c_k(h) ≤ C for k not in A

  - For j with h_j = 0: ∑_{k∈A} λ_k ∂c_k/∂h_j ≥ μ
  - For j with h_j = 1: ∑_{k∈A} λ_k ∂c_k/∂h_j ≤ μ
  - ∑λ_k = 1, ∑h_j = n/2

The equations are: |A| (active constraints) + |F| (KKT for free vars) + 1 (sum λ) + 1 (sum h) = |A| + |F| + 2.

For the system to be determined: n + |A| + 1 = |A| + |F| + 2, so n - |F| = 1, meaning |F| = n-1. That's not right - most variables are at bounds, so they don't contribute equations.

The correct count:
- Unknowns: h_j for j in F (|F| unknowns), λ_k for k in A (|A| unknowns), μ (1), C (1). Total: |F| + |A| + 2.
- Equations: 
  - c_k(h) = C for k in A: |A| equations
  - ∑_{k∈A} λ_k ∂c_k/∂h_j = μ for j in F: |F| equations
  - ∑λ_k = 1: 1 equation

So the system is determined! |F| + |A| + 2 unknowns = |A| + |F| + 2 equations.
