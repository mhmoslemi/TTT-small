```python
        # Equivalently: for all k, effects[j,k] > c[k] + effects[i,k] - cur_max

        critical = np.where(c >= cur_max - 1e-10)[0]

            if time.time() > deadline:
                break
            # For this i, the threshold at each lag is:
            # threshold[k] = c[k] + effects[i,k] - cur_max
            # We need effects[j, k] > threshold[k] for all k in critical
            # (checking only critical lags is sufficient if the margin is large enough)
            
            # Quick filter: at the most critical lag
            k_star = int(c.argmax())
            need_j = effects[zeros, k_star] > effects[i, k_star]
            if not need_j.any():
                continue
            candidates = zeros[need_j]
            
            # For each candidate, check all critical lags (and a few nearby)
            best_j = -1
            best_margin = 0

                margin = np.min(effects[j, critical] - (c[critical] + effects[i, critical] - cur_max))
                if margin > best_margin:

                    best_j = j

                    h[best_j] = 1.0
                    c = new_c
                    cur_max = float(c.max())
                    improved = True

            break
    
    return h, float(cur_max * dx)

def two_swap_search(h, n, deadline, n_trials=50000, seed=42):
    """Random 2-swap search to escape local optima."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(c.max())
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Random 2-swap
        i1, i2 = rng.choice(ones, size=2, replace=False)
        j1, j2 = rng.choice(zeros, size=2, replace=False)
        
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        
        new_c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        new_max = float(new_c.max())
        
        if new_max < cur_max - 1e-12:
            c = new_c
            cur_max = new_max
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            # Revert
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
    
    return h, float(cur_max * dx)

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

        # For ones: eff_i[ones] should be small (we want to remove them)
        # For zeros: eff_i[zeros] should be large (we want to add them)
        # Actually, we want effect[i,k_star] < effect[j,k_star] for the swap to help
        
        ones_eff = eff_i[ones]
        zeros_eff = eff_i[zeros]
        
        # Sort ones by eff (ascending - we want to remove low-eff ones)
        # Sort zeros by eff (descending - we want to add high-eff zeros)
        ones_sorted = ones[np.argsort(ones_eff)]
        zeros_sorted = zeros[np.argsort(-zeros_eff)]
        
        # Try the most promising pairs
        # For each i in ones_sorted, find best j in zeros_sorted
        for i_idx in range(min(len(ones_sorted), 50)):
            if time.time() > deadline:
                break
            i = ones_sorted[i_idx]
            # Need effect[j, k_star] > effect[i, k_star]
            mask = zeros_eff > eff_i[i]
            if not mask.any():
                continue
            candidates = zeros_sorted[mask]
            if len(candidates) == 0:
                continue
            j = candidates[0]
            
            # Full verification
            # Compute Δc for all lags
            delta_c = np.zeros(n_lags)
            for k_off in range(n_lags):
                kk = k_off - (n - 1)
                if kk == 0:
                    continue
                di = 0.0
                dj = 0.0
                if 0 <= i + kk < n:
                    di -= (1.0 - h[i + kk])
                if 0 <= i - kk < n:
                    di += h[i - kk]
                if 0 <= j + kk < n:
                    dj -= (1.0 - h[j + kk])
                if 0 <= j - kk < n:
                    dj += h[j - kk]
                delta_c[k_off] = di - dj
            
            new_max = (c + delta_c).max()
            if new_max < cur_max - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                c += delta_c
                cur_max = float(c.max())
                improved = True
                break
        
        if not improved:
            break
    
    return h, float(cur_max * dx)
```

Hmm, this is still O(n * n_lags) per candidate verification, which is O(n²) per candidate. With 50 candidates per sweep, that's 50 * n² = 50 * 62500 = 3M per sweep. For 1000 sweeps, that's 3B. Too slow.

Let me optimize the verification. Instead of computing delta_c for all lags, I can use a precomputed effects matrix.

For n=250, the effects matrix is 250 x 499 = 124750 entries. Computing it takes O(n²) = 62500 operations. Then each candidate verification is O(n_lags) = O(500). With 50 candidates, that's 25000 per sweep. Very fast!

Let me rewrite:

```python
def fast_1swap_search(h, n, deadline, max_sweeps=200000):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(c.max())
    n_lags = 2 * n - 1
    
    # Precompute effects matrix: effects[i, k_off] = change in c[k] if h[i] flips 1→0
    def compute_effects(h):
        eff = np.zeros((n, n_lags))
        for k_off in range(n_lags):
            k = k_off - (n - 1)
            if k == 0:
                continue
            if k > 0:
                # eff[i] = -(1-h[i+k]) for i=0..n-1-k, +h[i-k] for i=k..n-1
                eff[0:n-k, k_off] = -(1.0 - h[k:n])
                eff[k:n, k_off] += h[0:n-k]
            else:
                m = -k
                eff[m:n, k_off] = -(1.0 - h[0:n-m])
                eff[0:n-m, k_off] += h[m:n]
        return eff
    
    effects = compute_effects(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        improved = False
        k_star = int(c.argmax())
        
        # For 1-swap (i:1→0, j:0→1): Δc = effects[i] - effects[j]
        # Need: max(c + effects[i] - effects[j]) < cur_max
        
        # Filter: at k_star, need effects[j, k_star] > effects[i, k_star] + (cur_max - c[k_star])
        # Since c[k_star] = cur_max, need effects[j, k_star] > effects[i, k_star]
        
        ones_eff = effects[ones, k_star]
        zeros_eff = effects[zeros, k_star]
        
        # For each one (sorted by eff ascending), find best zero
        order_i = np.argsort(ones_eff)
        for idx in order_i[:100]:
            if time.time() > deadline:
                break
            i = ones[idx]
            # Find zeros with eff > eff[i]
            mask = zeros_eff > effects[i, k_star]
            if not mask.any():
                continue
            cand_j = zeros[mask]
            # Check full condition for each candidate
            for j in cand_j[:10]:
                new_c = c + effects[i] - effects[j]
                if new_c.max() < cur_max - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = new_c
                    cur_max = float(c.max())
                    # Update effects
                    effects = compute_effects(h)
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h, float(cur_max * dx)
```

This is much faster. The effects computation is O(n * n_lags) = O(n²) = 62500. The inner loop is O(n_lags) = O(n) per candidate. With 10 candidates and 100 iterations, that's 100 * 10 * 500 = 500000 per sweep. Very fast!

But wait, after each swap, I recompute the entire effects matrix, which is O(n²). If I do many swaps, this adds up. Let me think about incremental updates.

When h[i] changes from 1 to 0 (or 0 to 1), the effects for other positions change because they depend on h values at shifted positions.

If h[p] changes by δ (±1):
- effects[q, k] changes if q+k = p or q-k = p
  - q+k = p → q = p-k: effects[p-k, k] changes by... 
    The term -(1-h[p]) appears in effects[p-k, k] (as the "i+k=p" term for i=p-k).
    If h[p] changes by δ, (1-h[p]) changes by -δ, so -(1-h[p]) changes by +δ.
    So effects[p-k, k] += δ.
  - q-k = p → q = p+k: effects[p+k, k] changes by...
    The term +h[p] appears in effects[p+k, k] (as the "i-k=p" term for i=p+k).
    If h[p] changes by δ, h[p] changes by δ, so effects[p+k, k] += δ.

So after flipping h[p] by δ:
- For all k where 0 ≤ p-k < n: effects[p-k, k_off(k)] += δ
- For all k where 0 ≤ p+k < n: effects[p+k, k_off(k)] += δ

This is O(n) per flip. Much better than O(n²)!

Now let me put it all together. I'll also add a 2-swap search and continuous optimization.

For the 2-swap, I'll use a structured approach:
- For each zero j, compute the "best" one i to pair with (based on the critical lag)
- Try the top few 2-swap candidates

Actually, for 2-swaps, the effect is:
Δc[k] = effects[i1,k] + effects[i2,k] - effects[j1,k] - effects[j2,k]

This is the sum of individual effects. To find good 2-swaps, I can:
1. For each pair of zeros (j1, j2), compute the combined "benefit" vector: benefit = effects[j1] + effects[j2]
2. For each pair of ones (i1, i2), compute the combined "cost" vector: cost = effects[i1] + effects[i2]
3. A 2-swap is good if max(c + cost - benefit) < cur_max

But trying all pairs is O(n⁴). Instead, I'll use a heuristic:
- Sort ones by effect at k_star (ascending)
- Sort zeros by effect at k_star (descending)
- Try pairs from the top of each list
