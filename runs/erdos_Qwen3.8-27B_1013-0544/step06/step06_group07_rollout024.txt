            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = swap_delta(h, n, i, j, m_range)
            nc = float((corr + delta).max() * dx)
            dc = nc - cur_c
            if dc < best_dc:
                best_dc = dc
                best_i = i
                best_j = j
                best_delta = delta
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = corr + best_delta
            cur_c += best_dc
        else:
            improved = False
            for i in zeros:
                for j in ones:
                    delta = swap_delta(h, n, i, j, m_range)
                    nc = float((corr + delta).max() * dx)
                    if nc < cur_c - 1e-15:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = corr + delta
                        cur_c = nc
                        improved = True
                        break
                if improved:
                    break
                if time.time() > deadline:
                    return h, cur_c
            if not improved:
                break
    return h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    inits = []
    inits.append(round_to_binary(best_h, n))
    h_rand = np.zeros(n)
    h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    inits.append(h_rand)
    h_pert = best_h + 0.3 * rng.randn(n)
    inits.append(round_to_binary(h_pert, n))
    h_block = np.zeros(n)
    bs = max(4, n // 10)
    for bq in range(0, n, bs * 2):
        h_block[bq:bq + bs] = 1.0
    for _ in range(n // 8):
        zb = np.where(h_block == 0)[0]
        ob = np.where(h_block == 1)[0]
        h_block[zb[rng.randint(len(zb))]] = 1.0
        h_block[ob[rng.randint(len(ob))]] = 0.0
    inits.append(h_block)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h_gold = np.zeros(n)
    idx_gold = np.sort(((np.arange(n) * phi * n) % n).astype(int))
    h_gold[idx_gold[:n // 2]] = 1.0
    if h_gold.sum() != n // 2:
        idx2 = np.argsort(h_gold)[::-1]
        h_gold = np.zeros(n)
        h_gold[idx2[:n // 2]] = 1.0
    inits.append(h_gold)
    init_cs = [compute_c5(h, n) for h in inits]
    init_order = np.argsort(init_cs)
    n_sa_total = 4
    sa_budget = (budget_s - 25) * 0.55 / n_sa_total
    for sa_idx in range(n_sa_total):
        if time.time() > dl - 50:
            break
        if sa_idx < len(inits):
            init_i = init_order[sa_idx]
        else:
            init_i = init_order[0]
        sa_dl = min(dl - 25, time.time() + sa_budget)
        t_h = 0.04 if sa_idx < 2 else 0.02
        t_l = 0.00001 if sa_idx < 2 else 0.000005
        nr = 5 if sa_idx < 2 else 6
        hb, cb = replica_sa(inits[init_i], n, sa_dl, seed=seed + sa_idx * 137,
                            n_repl=nr, t_high=t_h, t_low=t_l, swap_interval=15)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    if time.time() < dl - 30:
        sa_dl = dl - 15
        hb, cb = replica_sa(best_h, n, sa_dl, seed=seed + 999,
                            n_repl=5, t_high=0.015, t_low=0.000003, swap_interval=12)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    if time.time() < dl - 8:
        hg, cg = greedy_local(best_h, n, dl - 4, max_iters=5000, seed=seed + 500)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    if time.time() < dl - 12:
        hb, cb = replica_sa(best_h, n, dl - 5, seed=seed + 2000,
                            n_repl=4, t_high=0.008, t_low=0.00001, swap_interval=20)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        if time.time() < dl - 3:
            hg, cg = greedy_local(best_h, n, dl - 2, max_iters=2000, seed=seed + 3000)
            if cg < best_c:
                best_c = cg
                best_h = hg.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
