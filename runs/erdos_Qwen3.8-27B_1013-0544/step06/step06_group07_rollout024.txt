def compute_c5(h, n):

def project_feasible(h, n):

            s = h.sum()
            if s < 1e-14:

            hb4b, cb4 = binary_sa(hb4, n, min(p4 + 5, dl - 20), seed=seed + 500)
            if cb4 < best_c:
                best_c = cb4
                best_h = hb4b.copy()
    # Phase 5: Try n=500 (15%)
    if n < 400 and time.time() < dl - 40:
        n2 = 500
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, n2)
        hbin2 = round_to_binary(hh, n2)
        hb5, cb5 = binary_sa(hbin2, n2, dl - 5, seed=seed + 700, t_start=0.025, t_end=0.00005)
        if cb5 < best_c:
            best_c = cb5
            best_h = hb5.copy()
            n = n2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
