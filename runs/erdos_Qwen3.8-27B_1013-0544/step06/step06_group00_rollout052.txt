            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_linearize(h, n):
    """Solve the LP linearization of the minimax problem at point h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    
    # For each lag δ, compute the linearization coefficients
    # d_δ[j] = (1-h[j+δ]) - h[j-δ] (with boundary handling)
    # Constraint: ∑_j d_δ[j] * h[j] ≤ t/dx - corr[k] + ∑_j d_δ[j]*h[j]

    hv = cp.Variable(n)
    tv = cp.Variable()
    
    constraints = [hv.sum() == n / 2.0, hv >= 0, hv <= 1]
    
    for k in range(2 * n - 1):
        delta = k - (n - 1)

        d = np.zeros(n)
        for j in range(n):
            jp = j + delta
            if 0 <= jp < n:
                d[j] += (1.0 - h[jp])
            jm = j - delta
            if 0 <= jm < n:
                d[j] -= h[jm]
        
        rhs = -corr[k] + d @ h
        constraints.append(d @ hv - tv / dx <= rhs)
    
    prob = cp.Problem(cp.Minimize(tv), constraints)
    prob.solve(solver=cp.GLPK)
    
    if hv.value is not None:

from scipy.optimize import linprog

def lp_step(h, n):
    """Solve the LP linearization at point h, return new h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    n_lags = 2 * n - 1
    
    # Build constraint matrix
    A_rows = []
    b_ub = []
    
    for k in range(n_lags):
        delta = k - (n - 1)
        d = np.zeros(n)
        if delta >= 0:
            if delta < n:
                d[:n-delta] += (1.0 - h[delta:])
                d[delta:] -= h[:n-delta]
        else:
            m = -delta
            if m < n:
                d[m:] += (1.0 - h[:n-m])
                d[:n-m] -= h[m:]
        
        # Constraint: d @ h_new - t/dx ≤ -corr[k] + d @ h
        # i.e., d @ h_new + (-1/dx) * t ≤ -corr[k] + d @ h
        A_rows.append(np.concatenate([d, [-1.0/dx]]))
        b_ub.append(-corr[k] + d @ h)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = project_feasible(res.x[:n], n)
        return h_new, res.x[n]
    return None, None

def project_feasible(h, n):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1 - h, mode='full')
    n_lags = 2 * n - 1
    A_rows = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)

                d[:n-m] = d[:n-m] - h[m:]
        A_rows[k, :n] = d
        A_rows[k, n] = -1.0 / dx

    c_obj[n] = 1.0
