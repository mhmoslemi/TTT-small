def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_smooth(h, n, alpha, lam):
    """Compute gradient of smooth max + sum penalty."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    
    # grad[j] = dx * sum_k w_k * [(1-h[j+k])*1[valid] - h[j-k]*1[valid]]
    # + 2*lam*(sum(h)-n/2)
    n_full = 2 * n - 1
    offset = n - 1  # shift k is at index offset+k in corr
    
    grad = np.zeros(n)
    # For each shift k, w_k contributes to grad
    # k ranges from -(n-1) to (n-1)

        # (1-h[j+k]) for valid j: need 0<=j+k<n, i.e., -k<=j<n-k if k>0, or 0<=j<n if k<=0... 
        # More precisely: 0<=j<n (always) and 0<=j+k<n
        # If k >= 0: j ranges 0 to n-1-k
        # If k < 0: j ranges -k to n-1
        lo = max(0, -k)
        hi = min(n, n - k)  # exclusive
        if lo < hi:

        # -h[j-k] for valid j: need 0<=j-k<n, i.e., k<=j<n+k if k>0, or 0<=j<n-k if k<0

    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

    h = h.copy()

    best_h = h.copy()
    prev_f = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        f_curr = smooth_obj(h, n, alpha, lam)
        if prev_f is not None and abs(prev_f - f_curr) < ftol:
            break
        prev_f = f_curr
        
        g = compute_grad_smooth(h, n, alpha, lam)
        
        # Backtracking line search

        step = eta

            h_new = project_feasible(h_new, n)

        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def smooth_obj(h, n, alpha, lam):

    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()

    sm = shifted.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return lse + lam * (h.sum() - n / 2.0) ** 2

def compute_grad_smooth(h, n, alpha, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    n_full = 2 * n - 1
    offset = n - 1
    grad = np.zeros(n)
    for k_idx in range(n_full):
        k = k_idx - offset
        wk = w[k_idx] * dx
        if wk < 1e-15:
            continue
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo + k:hi + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - k:hi2 - k]
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def proj_grad_descent(h, n, alpha, lam, eta0, max_iter, deadline, ftol=1e-15):
    h = h.copy()
    f_prev = smooth_obj(h, n, alpha, lam)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eta = eta0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_grad_smooth(h, n, alpha, lam)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        step = eta
        improved = False
        for _ in range(25):
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            f_new = smooth_obj(h_new, n, alpha, lam)
            if f_new <= f_prev - 1e-4 * step * np.dot(g, h_new - h):
                improved = True
                break
            elif f_new < f_prev:
                improved = True
                break
            step *= 0.5
        if not improved:
            h_new = project_feasible(h - step * g, n)
            if smooth_obj(h_new, n, alpha, lam) < f_prev:
                h = h_new
                f_prev = smooth_obj(h, n, alpha, lam)
            else:
                break
        h = h_new
        f_curr = smooth_obj(h, n, alpha, lam)
        if abs(f_prev - f_curr) < ftol:
            f_prev = f_curr
            break
        f_prev = f_curr
        eta = step * 1.5 if step < eta0 else min(eta0, step)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n

                delta[msk] -= h[imn[msk]]

                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0

                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - i)

                    delta[k1] += 1.0

                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_obj(x, n, alpha, lam):
    return smooth_obj(x, n, alpha, lam)

    return x.sum() - len(x) / 2.0
