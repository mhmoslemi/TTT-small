import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_TARGET = 0.0
_G_ALPHA = 1.0
_G_PEN = 0.0
_G_BOUNDS = ()
PAIR_FRACTIONS = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 0.995, 1.0)


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = 0.5 * float(n)
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        y = np.clip(x - tau, 0.0, 1.0)
        if y.sum() > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        if diff > 0.0:
            room = 1.0 - y
            s = room.sum()
            if s > 1e-14:
                y = np.clip(y + (diff / s) * room, 0.0, 1.0)
        else:
            s = y.sum()
            if s > 1e-14:
                y = np.clip(y + (diff / s) * y, 0.0, 1.0)
    return y


def enforce_sum(h, n):
    target = 0.5 * float(n)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        if diff > 0.0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + diff)
    return np.clip(h, 0.0, 1.0)


def round_to_binary(h, n):
    hb = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1]
    hb[idx[:n // 2]] = 1.0
    return hb


def h_to_bits(h):
    bits = 0
    for k in range(len(h)):
        if h[k] > 0.5:
            bits |= (1 << k)
    return bits


def bits_to_h(bits, n):
    h = np.zeros(n, dtype=float)
    for k in range(n):
        if (bits >> k) & 1:
            h[k] = 1.0
    return h


def bits_c5(bits, n, mask):
    z = mask ^ (bits & mask)
    best = 0
    for m in range(1, n):
        c = (bits & (z >> m)).bit_count()
        if c > best:
            best = c
        c = (z & (bits >> m)).bit_count()
        if c > best:
            best = c
    return float(best) * (2.0 / float(n))


def binary_sa(h, n, deadline, seed):

    h0 = round_to_binary(h, n)
    bits = h_to_bits(h0)

    cur = bits_c5(bits, n, mask)

    best = cur
    ones = np.where(h0 == 1.0)[0].astype(int)

        j = int(ones[bi])

                ones[bi] = i
            cur = bits_c5(bits, n, mask)

def weighted_grad(x, w, n, dx):
    cum = np.empty(2 * n, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(w)
    ar = np.arange(n, dtype=int)
    S = cum[2 * n - 1 - ar] - cum[n - 1 - ar]
    conv = np.convolve(x, w, mode='full')
    T = conv[n - 1:2 * n - 1]
    w0 = w[n - 1]
    return (S * (1.0 - 2.0 * x) - (T - w0 * x)) * dx


def soft_parts(x):
    n = _G_N
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode='full')
    f = c * dx
    m = float(np.max(f))
    alpha = float(_G_ALPHA)
    sh = alpha * (f - m)
    sh = np.clip(sh, -50.0, 0.0)
    e = np.exp(sh)
    z = float(e.sum())
    if z <= 0.0:
        idx = int(np.argmax(f))
        w = np.zeros_like(f)
        w[idx] = 1.0
        logz = 0.0
    else:
        w = e / z
        logz = float(np.log(z))
    return m, w, alpha, dx, logz


def soft_fg(x):
    m, w, alpha, dx, logz = soft_parts(x)
    val = m + logz / alpha
    s = float(np.sum(x) - _G_TARGET)
    if _G_PEN != 0.0:
        val += 0.5 * _G_PEN * s * s
    g = weighted_grad(x, w, _G_N, dx)
    if _G_PEN != 0.0:
        g = g + _G_PEN * s
    return val, g


def soft_val(x):
    m, w, alpha, dx, logz = soft_parts(x)
    val = m + logz / alpha
    s = float(np.sum(x) - _G_TARGET)
    if _G_PEN != 0.0:
        val += 0.5 * _G_PEN * s * s
    return float(val)


def soft_grad(x):
    m, w, alpha, dx, logz = soft_parts(x)
    g = weighted_grad(x, w, _G_N, dx)
    s = float(np.sum(x) - _G_TARGET)
    if _G_PEN != 0.0:
        g = g + _G_PEN * s
    return g


def cons_fun(x):
    return np.array([float(np.sum(x) - _G_TARGET)])


def cons_jac(x):
    return np.ones((1, _G_N))


def lbfgs_opt(h, n, alpha, maxiter, maxfun):
    global _G_N, _G_TARGET, _G_ALPHA, _G_PEN, _G_BOUNDS
    n = int(n)
    _G_N = n
    _G_TARGET = 0.5 * float(n)
    _G_ALPHA = float(alpha)
    _G_PEN = 500.0
    _G_BOUNDS = [(0.0, 1.0) for _ in range(n)]
    x = np.asarray(h, dtype=float)
    opts = {'maxiter': int(maxiter), 'maxfun': int(maxfun), 'ftol': 1e-14, 'gtol': 1e-12}
    try:
        res = minimize(soft_fg, x, method='L-BFGS-B', jac=True, bounds=_G_BOUNDS, options=opts)
        x = res.x
    except Exception:
        try:
            res = minimize(soft_val, x, method='L-BFGS-B', jac=soft_grad, bounds=_G_BOUNDS, options=opts)
            x = res.x
        except Exception:
            x = h
    return project_feasible(x, n)


def slsqp_opt(h, n, alpha, maxiter):
    global _G_N, _G_TARGET, _G_ALPHA, _G_PEN, _G_BOUNDS
    n = int(n)
    _G_N = n
    _G_TARGET = 0.5 * float(n)
    _G_ALPHA = float(alpha)
    _G_PEN = 0.0
    _G_BOUNDS = [(0.0, 1.0) for _ in range(n)]
    x = np.asarray(h, dtype=float)
    cons = ({'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},)
    try:
        res = minimize(soft_val, x, method='SLSQP', jac=soft_grad, bounds=_G_BOUNDS,
                       constraints=cons, options={'maxiter': int(maxiter), 'ftol': 1e-16})
        x = res.x
    except Exception:
        x = h
    return project_feasible(x, n)


def continuous_from(h, n, deadline, schedule):
    n = int(n)
    h = project_feasible(h, n)
    best = h.copy()
    bc = compute_c5(h, n)
    cur = h
    for alpha, mi, mf in schedule:
        rem = deadline - time.time()
        if rem < 10.0:
            break
        mi_use = int(min(int(mi), max(30, int(0.6 * rem))))
        mf_use = int(min(int(mf), max(800, int(30.0 * rem))))
        htry = lbfgs_opt(cur, n, alpha, mi_use, mf_use)
        ct = compute_c5(htry, n)
        if ct < bc:
            bc = ct
            best = htry.copy()
        if ct < bc + 0.0008:
            cur = htry
    return best, bc


def upsample(h_old, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    xo = np.linspace(0.0, 2.0, max(1, n_old))
    xn = np.linspace(0.0, 2.0, n_new)
    y = np.interp(xn, xo, h_old)
    return project_feasible(y, n_new)

    best_h = enforce_sum(project_feasible(best_h, n), n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, int(n)
