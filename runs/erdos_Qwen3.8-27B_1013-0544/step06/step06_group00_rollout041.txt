def c5v(c, n):

def project_feasible(y, n):

    h = np.clip(h, 0.0, 1.0)

        return h
    lo, hi = -1.0, 1.0

    a = float(alpha)
    m = float(np.max(c))
    z = a * (c - m)

def smooth_weights(c, alpha):

    m = float(np.max(c))
    w = np.exp(a * (c - m))

def grad_weights(h, w, n):

    B = np.convolve(h, w[::-1], mode='full')[mid:mid + n]
    return A - B

                rho *= 0.55
                if rho < 1e-4:
                    break
        else:
            g = G.mean(axis=0)
            st = 0.08
            for _ in range(6):
                y = project_feasible(h - st * g, n)
                cy = cvec(y)
                Cy = c5v(cy, n)
                if Cy < C - 1e-12:
                    h, c, C = y, cy, Cy
                    moved = True
                    if Cy < best_c:
                        best_c = Cy
                        best_h = h.copy()
                    break
                st *= 0.5
            if not moved:
                break
    return best_h, best_c

def _gs_sum(x):
    return x.sum() - 0.5 * float(_GS_N[0])

def _gs_obj(x):
    x = np.asarray(x, dtype=float)
    c = cvec(x)
    return smooth_value(c, _GS_ALPHA[0]) * (2.0 / float(_GS_N[0]))

def _gs_jac(x):
    x = np.asarray(x, dtype=float)
    c = cvec(x)
    w = smooth_weights(c, _GS_ALPHA[0])
    return grad_weights(x, w, _GS_N[0]) * (2.0 / float(_GS_N[0]))

def slsqp_polish(h0, n, deadline, alphas=None, maxfun=1400):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = c5v(cvec(h), n)
    if alphas is None:
        alphas = (30.0, 120.0, 500.0, 2000.0)
    for alpha in alphas:
        if time.time() > deadline:
            break
        _GS_N[0] = int(n)
        _GS_ALPHA[0] = float(alpha)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_gs_obj, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _gs_sum}],
                           jac=_gs_jac,
                           options={'maxiter': 250, 'ftol': 1e-14,
                                    'maxfun': maxfun})
        except Exception:
            res = None
        if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = cvec(h2)
            c2v = c5v(c2, n)
            if c2v < best_c:
                best_c = c2v
                best_h = h2.copy()
                h = h2
    return best_h, best_c

def to_binary(h, n):
    hb = np.zeros(n, dtype=float)
    exact = 0.5 * float(n)
    k = int(round(exact))
    if n and k > 0:
        hd = np.argsort(h)[::-1][:k]
        hb[hd] = 1.0
        rem = exact - k
        if abs(rem) > 1e-12 and n - k > 0:
            hb[int(np.argsort(h)[::-1][k])] = max(0.0, min(1.0, 2.0 * rem))
    return hb

def build_vw(h, n):
    M = 2 * n - 1
    mid = n - 1
    V = np.zeros((n, M), dtype=float)
    W = np.zeros((n, M), dtype=float)
    V[:, mid] = h
    W[:, mid] = h
    for s in range(1, n):
        ip = mid + s
        W[:n - s, ip] = h[s:]
        V[s:, ip] = h[:n - s]
        inl = mid - s
        V[:n - s, inl] = h[s:]
        W[s:, inl] = h[:n - s]
    return V, W

def binary_greedy(h0, n, deadline, max_passes=120):
    h = to_binary(h0, n).copy()
    c = cvec(h).astype(float)
    best_h = h.copy()
    best_c = c5v(c, n)
    V, W = build_vw(h, n)
    zeros_list = [int(i) for i in np.where(h < 0.5)[0]]
    ones_list = [int(i) for i in np.where(h > 0.5)[0]]
    mid = n - 1
    moves = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ii = 0
        while ii < len(zeros_list) and time.time() < deadline:
            i = zeros_list[ii]
            if h[i] >= 0.5:
                zeros_list.pop(ii)
                continue
            if not ones_list:
                break
            js = np.array(ones_list, dtype=int)
            diff = (V[js] - V[i]) + (W[js] - W[i])
            diff[:, mid] -= 2.0
            base = c[None, :] + diff
            k1 = mid + (i - js)
            k2 = mid + (js - i)
            rows = np.arange(js.size)
            cand = base.max(axis=1)
            cand = np.maximum(cand, base[rows, k1] + 1.0)
            cand = np.maximum(cand, base[rows, k2] + 1.0)
            bp = int(np.argmin(cand))
            if cand[bp] < c.max() - 1e-12:
                j = int(js[bp])
                delta = diff[bp].copy()
                delta[k1[bp]] += 1.0
                delta[k2[bp]] += 1.0
                c += delta
                h[i] = 1.0
                h[j] = 0.0
                moves += 1
                if moves % 25 == 0:
                    c = cvec(h).astype(float)
                    V, W = build_vw(h, n)
                else:
                    V, W = build_vw(h, n)
                zeros_list.pop(ii)
                try:
                    ones_list.remove(j)
                except ValueError:
                    pass
                ones_list.append(i)
                Cv = c5v(c, n)
                if Cv < best_c:
                    best_c = Cv
                    best_h = h.copy()
                improved = True
            else:
                ii += 1
        if not improved:
            break
    c = cvec(h).astype(float)
    Cv = c5v(c, n)
    if Cv < best_c:
        best_c = Cv
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1100.0
    total = max(20.0, budget_s - 12.0)
    deadline = t0 + total

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    if n < 2:
        return h0, c5v(cvec(h0), n), n

    h0 = project_feasible(h0, n)
    best_n = n
    best_h = h0.copy()
    best_c = c5v(cvec(best_h), best_n)

    hr = best_h[::-1].copy()
    cc = cvec(hr)
    if c5v(cc, best_n) < best_c:
        best_h = hr
        best_c = c5v(cc, best_n)
    ha = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
    cc = cvec(ha)
    if c5v(cc, best_n) < best_c:
        best_h = ha
        best_c = c5v(cc, best_n)

    dl1 = t0 + 0.34 * total
    if time.time() < dl1:
        h2, c2 = optimize_pgrad(best_h, best_n, dl1)
        if c2 < best_c:
            best_h, best_c = h2, c2

    dl2 = t0 + 0.52 * total
    if time.time() < dl2:
        h2, c2 = optimize_active_lp(best_h, best_n, dl2, 260)
        if c2 < best_c:
            best_h, best_c = h2, c2

    dl3 = t0 + 0.60 * total
    if best_n <= 600 and time.time() < dl3:
        h2, c2 = slsqp_polish(best_h, best_n, dl3, maxfun=1600)
        if c2 < best_c:
            best_h, best_c = h2, c2

    dl4 = t0 + 0.80 * total
    if best_n <= 450 and time.time() < dl4 - 45:
        hb, cb = binary_greedy(best_h, best_n, dl4 - 45, 120)
        if cb < best_c:
            best_h, best_c = hb, cb
        dl4b = dl4 - 8
        if time.time() < dl4b:
            h2, c2 = optimize_pgrad(hb, best_n, dl4b,
                                    alphas=(0.1, 0.4, 1.5, 6.0, 25.0, 100.0, 400.0),
                                    max_steps=1100)
            if c2 < best_c:
                best_h, best_c = h2, c2
            if time.time() < dl4:
                h2, c2 = optimize_active_lp(h2, best_n, dl4, 70)
                if c2 < best_c:
                    best_h, best_c = h2, c2

    dl5 = t0 + 0.92 * total
    for nh in (320, 480, 640):
        if time.time() > dl5 - 25 or nh == best_n or nh > 800:
            continue
        hstart = interp_to(best_h, best_n, nh)
        h2, c2 = optimize_pgrad(hstart, nh, min(deadline, time.time() + 28),
                                alphas=(0.1, 0.5, 2.0, 10.0, 40.0, 150.0),
                                max_steps=650)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, nh

    dlf = deadline - 3.0
    if time.time() < dlf:
        h2, c2 = optimize_active_lp(best_h, best_n, dlf, 110)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if best_n <= 600 and time.time() < deadline - 1.0:
        h2, c2 = slsqp_polish(best_h, best_n, deadline - 1.0,
                              alphas=(100.0, 500.0), maxfun=900)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = c5v(cvec(best_h), best_n)
    return best_h, best_c, best_n
