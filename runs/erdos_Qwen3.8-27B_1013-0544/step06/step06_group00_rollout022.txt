import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            h = np.clip(h + diff / max(s, 1e-15), 0, 1)
        else:
            s = h.sum()
            h = np.clip(h + diff / max(s, 1e-15), 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_obj_grad(x, n, alpha, pen):
    """Compute log-sum-exp objective and its gradient."""
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_v = np.exp(shifted - sm)
    p = exp_v / exp_v.sum()
    logsum = sm + np.log(exp_v.sum())
    obj = m + logsum / alpha + pen * (x.sum() - target) ** 2
    
    # Gradient
    dx_ = dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - x[lag:])
            grad[lag:] -= w * x[:n - lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - x[:n - l])
            grad[:n - l] -= w * x[l:]
    grad *= dx_
    grad += 2.0 * pen * (x.sum() - target)
    return obj, grad

def lse_smooth(h, n, alpha, pen, maxiter=5000, gtol=1e-14):
    x0 = h.copy()
    res = minimize(lambda x: lse_obj_grad(x, n, alpha, pen)[0],
                   x0, jac=lambda x: lse_obj_grad(x, n, alpha, pen)[1],
                   method='L-BFGS-B', bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-16})
    return project_feasible(res.x, n)

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].astype(int)
    zeros_arr = np.where(h == 0)[0].astype(int)
    T = 0.02
    cool = 0.99998
    scale = 5000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0

            zeros_arr[ai] = j

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            h = np.clip(h + diff / max(s, 1e-15), 0, 1)
        else:
            s = h.sum()
            h = np.clip(h + diff / max(s, 1e-15), 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_obj_grad(x, n, alpha, pen):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_v = np.exp(shifted - sm)
    p = exp_v / exp_v.sum()
    logsum = sm + np.log(exp_v.sum())
    obj = m + logsum / alpha + pen * (x.sum() - target) ** 2
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - x[lag:])
            grad[lag:] -= w * x[:n - lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - x[:n - l])
            grad[:n - l] -= w * x[l:]
    grad *= dx
    grad += 2.0 * pen * (x.sum() - target)
    return obj, grad

    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
