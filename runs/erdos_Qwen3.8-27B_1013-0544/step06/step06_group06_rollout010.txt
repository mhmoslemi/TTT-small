def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def gradient_at(h, n, alpha):
    """Gradient of smooth max of correlation using log-sum-exp."""
    corr = get_corr(h, n)
    m = corr.max()
    shifted = corr - m
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = 2 * n - 1
    
    for mi in range(nn):
        w = weights[mi] * dx
        if w < 1e-12:
            continue
        shift = mi - (n - 1)
        if shift >= 0:
            # corr[mi] = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i]) * dx
            # dcorr/dh[j] = dx * (1-h[j-shift]) for j in [shift, n-1]
            #              + dx * (-h[j+shift]) for j in [0, n-1-shift]
            if shift == 0:
                grad += w * 0  # (1-h[j]) - h[j] = 1-2h[j], summed = n - 2*sum(h) = 0
                # Actually for shift=0: d/dh[j] of h[j]*(1-h[j]) = 1-2h[j]
                # Sum over j: n - 2*sum(h) = n - 2*(n/2) = 0
                # But we still need the per-element gradient:
                for j in range(n):
                    grad[j] += w * (1 - 2*h[j])
            else:
                # j in [shift, n-1]: contributes (1-h[j-shift])*w
                pos = np.arange(shift, n)
                grad[pos] += w * (1 - h[pos - shift])
                # j in [0, n-1-shift]: contributes -h[j+shift]*w
                neg = np.arange(0, n - shift)
                grad[neg] += w * (-h[neg + shift])
        else:
            s = -shift
            # corr[mi] = sum_{i=s}^{n-1} h[i-s]*(1-h[i]) * dx
            # dcorr/dh[j] = dx * (1-h[j+s]) for j in [0, n-1-s]
            #              + dx * (-h[j-s]) for j in [s, n-1]
            pos = np.arange(0, n - s)
            grad[pos] += w * (1 - h[pos + s])
            neg = np.arange(s, n)
            grad[neg] += w * (-h[neg - s])
    
    return grad

def build_delta(h, n):
    """Build the delta matrix: delta[p, m] = change in corr[m] (before *dx) when h[p] goes 0->1."""
    delta = np.zeros((n, 2 * n - 1))
    for p in range(n):
        # n1 = p + n - 1 - m must be in [0, n-1]
        # n2 = p + m - n + 1 must be in [0, n-1]
        # m range: max(0, p - (n-1)) to min(2n-2, p + (n-1))
        m_lo = max(0, p - (n - 1))
        m_hi = min(2 * n - 2, p + (n - 1))
        ms = np.arange(m_lo, m_hi + 1)
        n1 = p + n - 1 - ms  # indices into h
        n2 = p + ms - (n - 1)  # indices into h
        # delta[p, m] = (1 - h[n1]) - h[n2]

        vals = (1.0 - h[n1]) - h[n2]

        center = (n - 1) - m_lo
        if 0 <= center < len(ms):
            vals[center] = 1.0 - 2.0 * h[p]
        delta[p, ms] = vals
    return delta
