def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    temp = 0.003
    stall = 0
    no_improve = 0

        top_idx = np.argsort(corr)[-top_k:]

def smooth_max_c5(h, n, alpha):

    sh = alpha * (c - m)

    best_h = h.copy()

        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        # Subgradient: index of max
        am = c.argmax()
        s = am - (n - 1)
        
        # Gradient of corr[am] w.r.t. h
        grad = np.zeros(n)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] = (1 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - s:hi2 - s]
        
        # Add smoothing gradient for other near-max shifts
        top5 = np.argsort(c)[-5:]
        for t in top5:
            st = t - (n - 1)
            w = np.exp(alpha * (c[t] - c[am]))
            if w > 0.01:
                l1 = max(0, -st)
                h1 = min(n, n - st)
                if h1 > l1:
                    grad[l1:h1] += w * (1 - h[l1 + st:h1 + st])
                l2 = max(0, st)
                h2 = min(n, n + st)
                if h2 > l2:
                    grad[l2:h2] -= w * h[l2 - st:h2 - st]
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        h = h_new
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        
        step = max(0.001, step * 0.9999)
        alpha = min(100000, alpha * 1.001)
    
    return best_h, best_c

                grad[lo1:hi1] = (1 - h_batch[lo1 + s:hi1 + s])
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if hi2 > lo2:
                grad[lo2:hi2] -= h_batch[lo2 - s:hi2 - s]
            top_t = np.argsort(c)[-4:]
            for t in top_t[1:]:
                st = t - (n - 1)
                w = np.exp(alpha * (c[t] - m))
                if w > 0.05:
                    l1 = max(0, -st)
                    h1i = min(n, n - st)
                    if h1i > l1:
                        grad[l1:h1i] += w * (1 - h_batch[l1 + st:h1i + st])
                    l2 = max(0, st)
                    h2i = min(n, n + st)
                    if h2i > l2:
                        grad[l2:h2i] -= w * h_batch[l2 - st:h2i + st]
            h_batch = project_feasible(h_batch - step * grad, n)
            n_its += 1
        c_final = compute_c5(h_batch, n)
        if c_final < best_c - 1e-8:
            best_c = c_final
            best_h = h_batch.copy()
            stagnant = 0
            step = min(0.008, step * 1.1)
        else:
            stagnant += 1
            step = max(0.0005, step * 0.85)
            alpha = min(50000, alpha * 1.05)
            if stagnant > 15:
                h_batch = project_feasible(best_h + 0.02 * rng.randn(n), n)
                stagnant = 0
        h = h_batch
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=500000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    best_n = n
    # Phase 1: Binary local search main (30% time)
    p1 = t0 + 0.30 * (budget_s - 5)
    h1, c1 = binary_local_search(h0, n, p1, seed=seed, top_k=20, max_kick=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
        best_n = n
    # Phase 2: Second binary pass (10%)
    p2 = t0 + 0.40 * (budget_s - 5)
    if time.time() < p2:
        h2, c2 = binary_local_search(best_h, n, p2, seed=seed + 777, top_k=30, max_kick=7)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n
    # Phase 3: Continuous gradient optimization (25%)
    p3 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p3:
        h_c = project_feasible(best_h + 0.05 * rng.randn(n), n)
        hc, cc = continuous_optimize(h_c, n, p3, seed=seed + 1000)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
            best_n = n
    # Phase 4: SLSQP polish (15%)
    p4 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p4:
        h_s = project_feasible(best_h + 0.01 * rng.randn(best_n), best_n)
        h_s = slsqp_polish(h_s, best_n, p4, alpha=200000)
        cs = compute_c5(h_s, best_n)
        if cs < best_c:
            best_c = cs
            best_h = h_s.copy()
            best_n = best_n
    # Phase 5: Final continuous from perturbed best (10%)
    p5 = dl - 5
    if time.time() < p5:
        h_f = project_feasible(best_h + 0.04 * rng.randn(best_n), best_n)
        hf, cf = continuous_optimize(h_f, best_n, p5, seed=seed + 5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
            best_n = best_n
    # Round if close to binary
    if (best_h > 0.95).sum() + (best_h < 0.05).sum() > best_n * 0.9:
        hb = np.zeros(best_n)
        hb[np.argsort(best_h)[::-1][:best_n // 2]] = 1.0
        cb = compute_c5(hb, best_n)
        if cb <= best_c + 1e-9:
            best_h = hb
            best_c = cb
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
