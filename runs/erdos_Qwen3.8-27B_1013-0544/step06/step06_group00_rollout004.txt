import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h * (target / s) if s > 0 else h
    return np.clip(h, 0.0, 1.0)

def corr_and_grad(h, n, alpha):
    """Compute correlation and gradient of smoothed max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    shifted = alpha * corr * dx
    shifted -= shifted.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()  # weights, length 2n-1
    
    # grad[j] = dx * sum_k w[k] * dc_k/dh[j]
    # k ranges from -(n-1) to (n-1), index in w is k+(n-1)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1] * dx
        if wk < 1e-15:
            continue
        # dc_k/dh[j] = (j in S_k)*(1-h[j-k]) - (j+k in S_k)*h[j+k]
        # S_k = {m: max(0,k) <= m <= min(n-1, n-1+k)}
        lo = max(0, k)
        hi = min(n-1, n-1+k)
        if lo <= hi:

            js = np.arange(lo, hi+1)
            grad[js] += wk * (1.0 - h[js - k])

        # j in [max(0,k)-k, min(n-1, n-1+k)-k] = [max(-k,0), min(n-1-k, n-1)]
        lo2 = max(0, -k)

        if lo2 <= hi2:
            js = np.arange(lo2, hi2+1)
            grad[js] -= wk * h[js + k]

def projected_grad_descent(h, n, alpha, max_iter, lr_init, deadline):

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()

        f = compute_c5(h, n)
        if f < best_f:
            best_f = f
            best_h = h.copy()
    
    return best_h, best_f

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    s_max = s.max()
    exps = np.exp(s - s_max)
    logsum = s_max + np.log(exps.sum())
    obj = logsum / alpha
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1] * dx
        if wk < 1e-16:
            continue
        lo = max(0, k)
        hi = min(n-1, n-1+k)
        if lo <= hi:
            js = np.arange(lo, hi+1)
            grad[js] += wk * (1.0 - h[js - k])
        lo2 = max(0, -k)
        hi2 = min(n-1, n-1-k)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2+1)
            grad[js] -= wk * h[js + k]
    
    return obj, grad

def pgd_optimize(h, n, alphas, deadline, lr_init=50.0):
    """Projected gradient descent with increasing alpha (annealing)."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = lr_init
        for it in range(20000):
            if time.time() > deadline:
                break
            obj, grad = smooth_obj_and_grad(h, n, alpha)
            
            # Backtracking line search
            def f_eval(h_new):
                return compute_c5(h_new, n)
            
            f_current = f_eval(h)
            accepted = False
            for _ in range(40):
                h_new = project_feasible(h - lr * grad, n)
                f_new = f_eval(h_new)
                if f_new < f_current - 1e-14 or f_new < best_c:
                    h = h_new
                    accepted = True
                    if f_new < best_c:
                        best_c = f_new
                        best_h = h.copy()
                    break
                lr *= 0.5
                if lr < 1e-10:
                    break
            if not accepted:
                break
        
        # Update best
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
