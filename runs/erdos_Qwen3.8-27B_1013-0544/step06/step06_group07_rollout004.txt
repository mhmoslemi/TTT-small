import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h)) * (2.0 / n))


def fix_sum(out, target):
    diff = target - out.sum()
    if abs(diff) < 1e-15:
        return out
    if diff > 0.0:
        idx = np.flatnonzero(out < 1.0)
        for ii in idx:
            a = min(diff, 1.0 - out[ii])
            out[ii] += a
            diff -= a
            if abs(diff) < 1e-15:
                break
    else:
        idx = np.flatnonzero(out > 0.0)
        for ii in idx:
            a = min(-diff, out[ii])
            out[ii] -= a
            diff += a
            if abs(diff) < 1e-15:
                break
    return out


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    v = np.sort(h)[::-1]
    css = np.cumsum(v)
    rho = np.arange(n, 0, -1, dtype=float)
    tau = (css - target) / rho
    m = v - tau > 0.0
    if np.any(m):
        tau = tau[int(np.argmax(m))]
    else:
        tau = 0.0
    out = np.clip(h - tau, 0.0, 1.0)
    return fix_sum(out, target)


def sum_eq(x, target):
    return x.sum() - target


def sum_jac(x, target):
    return np.ones(x.size, dtype=float)


def combined_grad(h, n, w):
    grad = np.zeros(n, dtype=float)
    w0 = w[n - 1]
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[n - 1 + p]
        wn = w[n - 1 - p]
        s = wp + wn
        if s <= 1e-15:
            continue
        grad[:n - p] += wn - s * h[p:]
        grad[p:] += wp - s * h[:n - p]
    return (2.0 / n) * grad


def smooth_obj(x, alpha, n):
    h = np.asarray(x, dtype=float)
    g = np.correlate(h, 1.0 - h)
    gm = g.max()
    z = alpha * (g - gm)
    s = np.exp(z).sum()
    return float((gm + np.log(s)) / alpha * (2.0 / n))


def smooth_grad(x, alpha, n):
    h = np.asarray(x, dtype=float)
    g = np.correlate(h, 1.0 - h)
    gm = g.max()
    z = alpha * (g - gm)
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(2 * n - 1, dtype=float)
        w[int(np.argmax(g))] = 1.0
    else:
        w = e / s
    return combined_grad(h, n, w)


def slsqp_polish(h, n, deadline, alphas, maxiter=200, ftol=1e-11):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (0.5 * n,)}]
    for alpha in alphas:
        if time.time() > deadline - 20.0:
            break
        try:
            res = minimize(
                smooth_obj, h, method='SLSQP', args=(alpha, n),
                jac=smooth_grad, bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
            )
            if not hasattr(res, 'x') or res.x is None:
                break
            x2 = res.x
            if not np.all(np.isfinite(x2)):
                break
            h2 = project_feasible(x2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= best_c + 2e-3:
                h = h2
            else:
                break
        except Exception:
            break
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)


def round_to_binary_int(h, n):
    target = n // 2
    arr = np.asarray(h, dtype=float)
    b = (arr >= 0.5).astype(np.int32)
    if b.sum() == target:
        return b
    out = np.zeros(n, dtype=np.int32)
    out[np.argsort(arr, kind='stable')[::-1][:target]] = 1
    return out


def make_binary_starts(h, n, seed):
    rng = np.random.RandomState(seed)
    starts = [
        round_to_binary_int(h, n),
        round_to_binary_int(h + 0.08 * rng.randn(n), n),
    ]
    for _ in range(2):
        idx = rng.permutation(n)
        b = np.zeros(n, dtype=np.int32)
        b[idx[:n // 2]] = 1
        starts.append(b)
    tm = np.zeros(n, dtype=float)
    for i in range(n):
        x = i
        p = 0
        while x:
            p ^= x & 1
            x >>= 1
        tm[i] = p
    starts.append(round_to_binary_int(tm + 0.01 * rng.rand(n), n))
    return starts


def binary_sa(h_in, n, deadline, seed=42, t_start=0.8, t_end=0.008, patience=25000):
    h = round_to_binary_int(h_in, n)
    zeros = np.flatnonzero(h == 0).tolist()
    ones = np.flatnonzero(h == 1).tolist()
    one_minus = (1 - h).astype(np.int32)
    corr = np.correlate(h, one_minus).astype(np.int32)
    rev = one_minus[::-1].astype(np.int32).copy()

    cur_mc = int(corr.max())
    best_mc = cur_mc
    best_h = h.copy()

    L = 2 * n - 1
    tmp = np.empty(L, dtype=np.int32)
    delta = np.empty(L, dtype=np.int32)

    rng = np.random.RandomState(seed)
    cool_steps = 200000
    rate = (t_end / t_start) ** (1.0 / cool_steps)
    T = t_start
    no = 0
    step = 0

    while time.time() < deadline:
        if not zeros or not ones:
            break
        iz = rng.randint(len(zeros))
        oj = rng.randint(len(ones))
        i = zeros[iz]
        j = ones[oj]

        delta.fill(0)
        delta[i:i + n] += rev
        si = n - 1 - i
        delta[si:si + n] -= h
        delta[j:j + n] -= rev
        sj = n - 1 - j
        delta[sj:sj + n] += h
        delta[n - 1] -= 2
        delta[n - 1 + i - j] += 1
        delta[n - 1 + j - i] += 1

        np.add(corr, delta, out=tmp)
        nm = int(tmp.max())
        dc = nm - cur_mc

        if dc <= 0 or (T > 1e-6 and rng.random() < np.exp(-float(dc) / T)):
            np.copyto(corr, tmp)
            h[i] = 1
            h[j] = 0
            rev[n - 1 - i] = 0
            rev[n - 1 - j] = 1
            zeros[iz] = j
            ones[oj] = i
            cur_mc = nm
            if nm < best_mc:
                best_mc = nm
                best_h = h.copy()
                no = 0
            else:
                no += 1
        else:
            no += 1

        step += 1
        T *= rate

        if no > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                if zeros and ones:
                    a = rng.randint(len(zeros))
                    b = rng.randint(len(ones))
                    ii = zeros[a]
                    jj = ones[b]
                    h[ii] = 1
                    h[jj] = 0
                    zeros[a] = jj
                    ones[b] = ii
            one_minus = (1 - h).astype(np.int32)
            corr = np.correlate(h, one_minus).astype(np.int32)
            rev = one_minus[::-1].astype(np.int32).copy()
            cur_mc = int(corr.max())
            zeros = np.flatnonzero(h == 0).tolist()
            ones = np.flatnonzero(h == 1).tolist()
            no = 0
            T = min(t_start, T * 5.0)

    return best_h, float(best_mc * (2.0 / n))


def upsample_h(h, n, n2):
    h = np.asarray(h, dtype=float)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n2) + 0.5) * (2.0 / n2)
    return project_feasible(np.interp(xn, xo, h, left=h[0], right=h[-1]), n2)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    dl = t0 + B - 10.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
        h0 = h0 + 0.25 * np.random.RandomState(seed).randn(n0)

    h0 = project_feasible(h0, n0)
    best_n = n0
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    h0_best = best_h.copy()
    rng = np.random.RandomState(seed)

    alphas = [40, 150, 600, 2000]

    if n0 % 2 == 0:
        d1 = min(dl - 350.0, t0 + max(60.0, 0.30 * (B - 10.0)))
    else:
        d1 = min(dl - 300.0, t0 + max(60.0, 0.25 * (B - 10.0)))

    if time.time() < d1:
        mi = 250 if n0 <= 300 else 100
        hb, cb = slsqp_polish(h0, n0, d1, alphas, maxiter=mi)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n0
            h0_best = best_h.copy()

    if best_c > 0.3808 and time.time() < d1 - 20.0:
        hp = project_feasible(h0_best + 0.02 * rng.randn(n0), n0)
        hb, cb = slsqp_polish(hp, n0, d1, alphas[:3], maxiter=150, ftol=1e-12)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n0
            h0_best = best_h.copy()

    n2 = 500 if n0 < 500 else int(max(500, n0))
    if n2 != n0 and n2 <= 750 and time.time() < dl - 260.0:
        d2 = min(dl - 250.0, t0 + max(120.0, 0.60 * (B - 10.0)))
        hu = upsample_h(best_h, best_n, n2)
        mi2 = 100 if n2 <= 500 else 60
        hb, cb = slsqp_polish(hu, n2, d2, alphas, maxiter=mi2, ftol=1e-12)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n2

        if best_c > 0.3808 and time.time() < d2 - 60.0:
            hu2 = project_feasible(hu + 0.03 * rng.randn(n2), n2)
            hb, cb = slsqp_polish(hu2, n2, d2, alphas[:3], maxiter=70, ftol=1e-12)
            if cb < best_c:
                best_c, best_h, best_n = cb, hb.copy(), n2

    d_bin = dl - 35.0
    if n0 % 2 == 0 and time.time() < d_bin - 40.0:
        starts = make_binary_starts(h0_best, n0, seed)
        for k, st in enumerate(starts[:3]):
            end = min(d_bin - 30.0, time.time() + 35.0)
            if time.time() >= end:
                break
            hb, cb = binary_sa(st, n0, end, seed=seed + 11 + 17 * k, patience=15000)
            if cb < best_c:
                best_c, best_h, best_n = cb, np.asarray(hb, dtype=float).copy(), n0
                h0_best = best_h.copy()

    if best_n > n0 and best_n % 2 == 0 and time.time() < d_bin - 30.0:
        hb, cb = binary_sa(
            round_to_binary_int(best_h, best_n), best_n,
            min(d_bin - 10.0, time.time() + 45.0), seed=seed + 101, patience=25000
        )
        if cb < best_c:
            best_c, best_h = cb, np.asarray(hb, dtype=float).copy()

    if time.time() < dl - 12.0:
        hfin = np.asarray(best_h, dtype=float)
        hb, cb = slsqp_polish(hfin, best_n, dl - 3.0, [150, 800, 3000], maxiter=80, ftol=1e-13)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    best_h = project_feasible(np.asarray(best_h, dtype=float), best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
