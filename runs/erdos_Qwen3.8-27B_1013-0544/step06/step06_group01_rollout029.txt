def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

                                best_d = d_opt
                elif y_m < 1e17 and y_m < y_c - 1e-14:
                    best_d = d_m
                elif y_p < 1e17 and y_p < y_c - 1e-14:
                    best_d = d_p
                c[ks] += best_d * s
                if has_zero:
                    c[nm1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                cur_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, cur_max * dx

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    return float(m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha)

def _cf(x, n):
    return x.sum() - n / 2.0

def _cj(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha=100000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(200, 0.5)
        n0 = 200
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Phase 1: aggressive CD
    t_alloc = min(280, (deadline - t0) * 0.30)
    h_cd, c_cd = fast_cd(best_h, n0, 600, 250, t0 + t_alloc, rng, 3.0)
    h_cd = project_feasible(h_cd, n0)
    c_cd = compute_c5(h_cd, n0)
    if c_cd < best_c:
        best_c, best_h = c_cd, h_cd
    # Phase 2: SLSQP
    if time.time() < deadline - 120:
        for a in [30000, 100000, 300000]:
            if time.time() > deadline - 80:
                break
            hs = slsqp_polish(best_h, n0, a, 1200)
            cs = compute_c5(hs, n0)
            if cs < best_c:
                best_c, best_h = cs, hs
    # Phase 3: fine CD
    if time.time() < deadline - 80:
        hd, cd = fast_cd(best_h, n0, 400, 350, time.time() + 70, rng, 0.5)
        hd = project_feasible(hd, n0)
        cd = compute_c5(hd, n0)
        if cd < best_c:
            best_c, best_h = cd, hd
    # Phase 4: try n=200
    if time.time() < deadline - 150 and n0 != 200:
        x0 = np.linspace(0, 2, n0)
        x2 = np.linspace(0, 2, 200)
        h2 = project_feasible(np.interp(x2, x0, best_h), 200)
        alloc = min(100, (deadline - time.time()) * 0.30)
        if alloc > 50:
            h2, c2 = fast_cd(h2, 200, 500, 250, time.time() + alloc, rng, 3.0)
            h2 = project_feasible(h2, 200)
            c2 = compute_c5(h2, 200)
            if c2 < best_c:
                best_c, best_h = c2, h2
            if time.time() < deadline - 50:
                h2 = slsqp_polish(h2, 200, 100000, 1000)
                c2 = compute_c5(h2, 200)
                if c2 < best_c:
                    best_c, best_h = c2, h2
                h2, c2 = fast_cd(h2, 200, 200, 350, deadline - 30, rng, 0.3)
                h2 = project_feasible(h2, 200)
                c2 = compute_c5(h2, 200)
                if c2 < best_c:
                    best_c, best_h = c2, h2
    # Phase 5: restarts
    if time.time() < deadline - 80:
        nr = len(best_h)
        for r in range(6):
            if time.time() > deadline - 40:
                break
            sc = 0.02 + 0.015 * r
            hp = project_feasible(best_h + rng.normal(0, sc, nr), nr)
            alloc = min(55, (deadline - time.time() - 20) * 0.35)
            if alloc < 25:
                break
            hr, cr = fast_cd(hp, nr, 300, 250, time.time() + alloc, rng, 3.0)
            hr = project_feasible(hr, nr)
            cr = compute_c5(hr, nr)
            if cr < best_c:
                best_c, best_h = cr, hr
    # Phase 6: final SLSQP + CD
    if time.time() < deadline - 20:
        nf = len(best_h)
        hf = slsqp_polish(best_h, nf, 500000, 2000)
        cf = compute_c5(hf, nf)
        if cf < best_c:
            best_c, best_h = cf, hf
    if time.time() < deadline - 5:
        nf = len(best_h)
        hf, cf = fast_cd(best_h, nf, 200, 400, deadline - 2, rng, 0.1)
        hf = project_feasible(hf, nf)
        cf = compute_c5(hf, nf)
        if cf < best_c:
            best_c, best_h = cf, hf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
