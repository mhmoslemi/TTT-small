                break
            h += diff / s * room

            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_gk(h, shift, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n-shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    return dx * grad

def pgd(h, n, deadline, alpha_init=0.05, max_iter=200000):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline: break
        cvals = np.correlate(h, 1-h, mode='full') * (2.0/n)
        cur_c = cvals.max()
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        k_star = int(np.argmax(cvals))
        shift = k_star - (n - 1)
        g = grad_gk(h, shift, n)
        a = alpha
        accepted = False
        for _ in range(25):
            h_new = project_feasible(h - a * g, n)
            c_new = float(np.max(np.correlate(h_new, 1-h_new, mode='full')) * (2.0/n))
            if c_new < cur_c - 1e-15:
                h = h_new
                alpha = min(a * 1.3, 0.2)
                accepted = True
                break
            a *= 0.5
        if not accepted:
            alpha *= 0.7
            if alpha < 1e-12:
                break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + sm/alpha + np.log(np.exp(shifted - sm).sum())/alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):

    dx = 2.0 / n
    h = to_binary(h_init, n).copy()

    cur_c = corr.max() * dx

    ones_l = list(np.where(h == 1)[0])
    zeros_l = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        for i in zeros_l:
            if time.time() > deadline: break
            for j in ones_l:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_l.remove(j); zeros_l.remove(i)
                    ones_l.append(i); zeros_l.append(j)
                    improved = True
                    break
            if improved: break
        if not improved: break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c
