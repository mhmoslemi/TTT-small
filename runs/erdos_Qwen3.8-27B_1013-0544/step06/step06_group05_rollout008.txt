def greedy_binary_v2(h, n, deadline, max_sweeps=5000):
    dx = 2.0 / n
    m = n // 2
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()

    # Precompute A and B: shape (n, 2n-1)
    # A[p, d_idx] = (1-h[p + lags[d_idx]]) if valid, else 0
    # B[p, d_idx] = h[p - lags[d_idx]] if valid, else 0
    def compute_AB(h):
        A = np.zeros((n, 2*n-1))
        B = np.zeros((n, 2*n-1))
        for p in range(n):
            pp = p + lags
            pm = p - lags
            vp = (pp >= 0) & (pp < n)
            vm = (pm >= 0) & (pm < n)
            A[p, vp] = 1.0 - h[pp[vp]]
            B[p, vm] = h[pm[vm]]
        return A, B
    
    A, B = compute_AB(h)

        if time.time() > deadline:
            break

                break

Wait, I made an error. The objective is max_k c[k] * dx where c = np.correlate(h, 1-h, mode='full'). This is NOT a linear function of h! Because c[k] = sum_i h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]. The second term is quadratic in h!

So the objective is max of quadratic functions, and the optimum is NOT necessarily at a vertex. It could be in the interior of an edge or face.

This changes everything. The continuous solution can indeed be better than any binary solution.

So my approach should be:
1. Find a good binary solution (to get close to the optimum)
2. Use continuous optimization to find the exact optimum in the neighborhood

For the continuous optimization, since the objective is max of smooth functions (each c[k] * dx is smooth in h), we can use:
- Smooth max approximation: f(h) ≈ (1/α) * log(sum(exp(α * c[k] * dx)))
- Then use gradient-based optimization

The gradient of c[k] with respect to h[i]:
c[k] = sum_j h[j] * (1-h[j+k-lag]) where lag = k-(n-1)
∂c[k]/∂h[i] = (1-h[i+lag]) [if i+lag valid] - h[i-lag] [if i-lag valid]

This is exactly the delta_i[d] I computed earlier!

So the gradient of the smooth max is:
∇f = (1/α) * sum_k (exp(α*c[k]*dx) / sum(exp(α*c[j]*dx))) * ∂c[k]/∂h

This gives us a way to do gradient descent on the continuous problem.

1. Binarize initial_h_values
2. Greedy binary local search (to get a good binary starting point)
3. Continuous optimization:
   a. Project gradient descent with smooth max
   b. Or SLSQP
4. Re-binarize and do another round of binary search
5. Final continuous polish

For the continuous optimization, I'll use a projected gradient method:
- Compute gradient of smooth max
- Take a step in the negative gradient direction
- Project back to the feasible set {0≤h≤1, sum(h)=n/2}
- Use a decreasing step size

The projection onto {0≤h≤1, sum(h)=n/2} can be done efficiently:
- Start with h' = h - step_size * grad
- Need to find t such that sum(clip(h' - t, 0, 1)) = n/2
- This is a 1D root-finding problem (bisection)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def binarize(h, n):
    m = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:m]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def compute_overlap_vector(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def greedy_binary(h, n, deadline, max_sweeps=5000):
    dx = 2.0 / n
    c = compute_overlap_vector(h, n)
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    lags = np.arange(2*n-1) - (n-1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        for i in ones:
            if improved or time.time() > deadline:
                break
            # Compute delta_i (change when h[i] goes 1→0)
            ip = i + lags
            im = i - lags
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            delta_i = np.zeros(2*n-1)
            delta_i[vp] -= (1.0 - h[ip[vp]])
            delta_i[vm] += h[im[vm]]
            base = c + delta_i

            # delta_j[d] = (1-h[j+d]) - h[j-d]
            # new_c = base + delta_j
            # We want min_j max_d new_c[j,d]

            best_j = -1
            best_val = cur_max
            for j in zeros:
                jp = j + lags
                jm = j - lags
                vjp = (jp >= 0) & (jp < n)
                vjm = (jm >= 0) & (jm < n)
                delta_j = np.zeros(2*n-1)
                delta_j[vjp] += (1.0 - h[jp[vjp]])
                delta_j[vjm] -= h[jm[vjm]]
                nm = (base + delta_j).max()
                if nm < best_val:
                    best_val = nm
                    best_j = j
            
            if best_j >= 0 and best_val < cur_max - 1e-10:
                h[i] = 0.0
                h[best_j] = 1.0
                c = compute_overlap_vector(h, n)
