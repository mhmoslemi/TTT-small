        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode='full')
        new_max = float(new_corr.max())
        delta = new_max - cur_max
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            cur_max = new_max
            cur_corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, float(best_max * dx)

def greedy_search(h_init, n, deadline, seed=42, max_sweeps=500000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    cur = make_binary(h_init.copy(), n)
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = float(cur_corr.max())
    best = cur.copy()
    best_max = cur_max
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur == 1)[0])
        zeros = list(np.where(cur == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if time.time() > deadline:
                return best, float(best_max * dx)
            for j in zeros[:80]:
                cur[i] = 0.0
                cur[j] = 1.0
                new_corr = np.correlate(cur, 1.0 - cur, mode='full')
                new_max = float(new_corr.max())
                if new_max < cur_max - 1e-14:
                    cur_max = new_max
                    cur_corr = new_corr
                    if cur_max < best_max:
                        best_max = cur_max
                        best = cur.copy()
                    improved = True
                    break
                else:
                    cur[i] = 1.0
                    cur[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return best, float(best_max * dx)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_top_lags(h, n, alpha, n_top, maxiter, deadline):
    dx = 2.0 / n
    g0 = np.correlate(h, 1.0 - h, mode='full') * dx
    top_lags = np.argsort(g0)[-n_top:]
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g[top_lags] - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 6
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Generate diverse starting points
    starts = [h0.copy()]
    starts.append(legendre_init(n, seed=seed))
    starts.append(random_binary(n, seed=seed+1))
    starts.append(random_binary(n, seed=seed+2))
    starts.append(random_binary(n, seed=seed+99))
    
    # Phase 1: SA on multiple starts (25% budget)
    p1 = t0 + 0.25 * (budget_s - 6)
    time_per_start = max(1.0, (p1 - t0) / len(starts))
    for si, s in enumerate(starts):
        if time.time() > p1:
            break
        sa_dl = min(p1, t0 + (si+1) * time_per_start + 2)
        hb, cb = simulated_annealing(s, n, sa_dl, seed=seed+si*10)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 2: Greedy search on best binary (25% budget)
    p2 = t0 + 0.50 * (budget_s - 6)
    if time.time() < p2:
        hb, cb = greedy_search(best_h.copy(), n, p2, seed=seed+200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 3: SLSQP continuous polish (25% budget)
    p3 = t0 + 0.75 * (budget_s - 6)
    if time.time() < p3:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p3: break
            h = slsqp_optimize(best_h.copy(), n, alpha, 5000, p3)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < p3:
            h = slsqp_top_lags(best_h.copy(), n, 500000, 5, 5000, p3)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 4: Re-binarize + greedy + SA + SLSQP (25% budget)
    p4 = dl
    if time.time() < p4:
        h_bin = make_binary(best_h, n)
        hb, cb = greedy_search(h_bin, n, dl - 40, seed=seed+500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        if time.time() < dl - 40:
            hb, cb = simulated_annealing(best_h.copy(), n, dl - 40, seed=seed+600,
                                         t_init=1.0, t_min=0.002, iter_per_sec=3000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
        if time.time() < dl - 5:
            for alpha in [200000, 2000000]:
                if time.time() > dl - 3: break
                h = slsqp_optimize(best_h.copy(), n, alpha, 3000, dl - 2)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
