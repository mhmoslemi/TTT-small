def smooth_grad(h, n, alpha):
    """Compute gradient of log-sum-exp smooth max of correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # Softmax weights
    shifted = alpha * corr - alpha * corr.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        L = k - n_minus_1
        if L >= 0:
            # d(corr[k])/d(h[j]) = (1-h[j+L]) * [0≤j≤n-1-L] - h[j-L] * [L≤j≤n-1]
            js = np.arange(n - L)
            grad[js] += w[k] * (1 - h[js + L])
            js2 = np.arange(L, n)
            grad[js2] -= w[k] * h[js2 - L]
        else:
            M = -L
            # d(corr[k])/d(h[j]) = (1-h[j-M]) * [M≤j≤n-1] - h[j+M] * [0≤j≤n-1-M]
            js = np.arange(M, n)
            grad[js] += w[k] * (1 - h[js - M])
            js2 = np.arange(n - M)
            grad[js2] -= w[k] * h[js2 + M]
    
    return grad * dx

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 30 or (no_improve > 200 and kick_count < 30):
                kick_count += 1
                nk = min(3, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                h_new = h.copy()
                for a in range(nk):
                    h_new[zeros[ri[a]]] = 1.0
                    h_new[ones[rj[a]]] = 0.0
                h_bin = np.zeros(n)
                h_bin[np.argsort(h_new)[::-1][:n // 2]] = 1.0
                new_corr = np.correlate(h_bin, 1 - h_bin, mode='full').astype(float)
                new_c_r = new_corr.max() * dx
                if np.random.rand() < np.exp(-(new_c_r - cur_c) / max(temp, 0.001)):
                    h = h_bin
                    corr = new_corr
                    g = build_g(h, n)
                    ones = np.where(h == 1)[0].copy()
                    zeros = np.where(h == 0)[0].copy()
                    cur_c = corr.max() * dx
                temp = max(0.001, temp * 0.9)
                stall = 0
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        gv = np.correlate(x, 1 - x, mode='full') * dx
        m = gv.max()
        sh = alpha * (gv - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_diff}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def proj_gradient_optimize(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = n / (8.0 * alpha)
        mmt = 0.9
        v = np.zeros(n)
        h_cur = best_h.copy()
        patience = 0
        for it in range(20000):
            if time.time() > deadline:
                break
            corr = np.correlate(h_cur, 1 - h_cur, mode='full') * 2.0 / n
            shifted = alpha * corr - alpha * corr.max()
            w = np.exp(shifted)
            w /= w.sum()
            grad = np.zeros(n)
            nm1 = n - 1
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                L = k - nm1
                if L >= 0:
                    js = np.arange(n - L)
                    grad[js] += w[k] * (1.0 - h_cur[js + L])
                    js2 = np.arange(L, n)
                    grad[js2] -= w[k] * h_cur[js2 - L]
                else:
                    M = -L
                    js = np.arange(M, n)
                    grad[js] += w[k] * (1.0 - h_cur[js - M])
                    js2 = np.arange(n - M)
                    grad[js2] -= w[k] * h_cur[js2 + M]
            v = mmt * v + (1.0 - mmt) * grad
            h_new = h_cur - lr * v
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
            if patience > 300:
                lr *= 0.5
                v = np.zeros(n)
                patience = 0
                if lr < 1e-8:
                    break
            h_cur = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Binary search (25% time)
    p1 = t0 + 0.25 * (budget_s - 5)
    hb, cb = fast_binary_search(h0, n, p1, top_k=10, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 2: Projected gradient (40% time)
    p2 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p2 - 10:
        h_pg = project_feasible(best_h + 0.05 * rng.randn(n), n)
        hp, cp = proj_gradient_optimize(h_pg, n, p2, alphas=[50, 200, 1000, 5000, 20000])
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    # Phase 3: SLSQP multi-restart (20% time)
    p3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p3 - 20:
        alphas_s = [5000, 20000, 50000, 100000]
        for restart in range(6):
            if time.time() > p3 - 10:
                break
            pert = 0.02 / (restart + 1) * rng.randn(n)
            h_s = project_feasible(best_h + pert, n)
            for alpha in alphas_s:
                if time.time() > p3 - 5:
                    break
                h_s = slsqp_smooth(h_s, n, alpha, 400)
                c = compute_c5(h_s, n)
                if c < best_c:
                    best_c = c
                    best_h = h_s.copy()
    # Phase 4: Final polish
    if time.time() < dl - 3:
        hf = project_feasible(best_h + 0.01 * rng.randn(n), n)
        hf = slsqp_smooth(hf, n, 50000, 300)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf.copy()
    # Binarize + re-polish
    if time.time() < dl - 2:
        h_bin = (best_h > 0.5).astype(float)
        ones = np.where(h_bin == 1)[0]
        if len(ones) > n // 2:
            order = np.argsort(best_h[ones])
            h_bin[ones[order[:len(ones) - n // 2]]] = 0.0
        elif len(ones) < n // 2:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros])
            h_bin[zeros[order[:n // 2 - len(ones)]]] = 1.0
        best_h = project_feasible(h_bin + 0.03 * rng.randn(n), n)
        best_h = slsqp_smooth(best_h, n, 20000, 300)
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
