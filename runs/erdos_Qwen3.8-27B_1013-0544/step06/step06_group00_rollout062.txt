        i_hi = min(n, n - m)
        if i_lo < i_hi:
            grad_g[i_lo:i_hi] += w * (1.0 - x[i_lo + m:i_hi + m])
        # Negative part: -x[i-m] for 0 <= i-m < n => m <= i < n+m

        i_hi2 = min(n, n + m)
        if i_lo2 < i_hi2:
            grad_g[i_lo2:i_hi2] -= w * x[i_lo2 - m:i_hi2 - m]
    
    return dx * grad_g + 2.0 * penalty * (x.sum() - target)

    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Precompute delta for each zero position
        # delta_i[k] = (1-h[i+m_k])*valid - h[i-m_k]*valid
        delta_zeros = np.zeros((len(zeros), 2*n-1))
        for idx, i in enumerate(zeros):
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta_zeros[idx, msk] += (1.0 - h[im[msk]])
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            delta_zeros[idx, msk2] -= h[imn[msk2]]
        
        # Precompute delta for each one position (going from 1 to 0)
        delta_ones = np.zeros((len(ones), 2*n-1))
        for idx, j in enumerate(ones):
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta_ones[idx, msk] -= (1.0 - h[jm[msk]])
            jmn = j - m_range
            msk2 = (jmn >= 0) & (jmn < n)
            delta_ones[idx, msk2] += h[jmn[msk2]]
        
        # For each zero i, find the best one j
        # new_corr = corr + delta_i + delta_j
        # We want to minimize max(new_corr) * dx
        best_improve = 0
        best_i = -1
        best_j = -1
        best_delta = None
        
        for ii in range(len(zeros)):
            base = corr + delta_zeros[ii]
            # new_corr for all j: base + delta_ones[jj]
            # max over k: np.max(base[None,:] + delta_ones, axis=1)
            new_maxes = np.max(base[None, :] + delta_ones, axis=1) * dx
            jj = np.argmin(new_maxes)
            if new_maxes[jj] < cur_c - 1e-15:
                if cur_c - new_maxes[jj] > best_improve:
                    best_improve = cur_c - new_maxes[jj]
                    best_i = ii

        j = ones[best_j]
        h[i], h[j] = 1.0, 0.0
        corr += delta_zeros[best_i] + delta_ones[best_j]
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lbfexp_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha + penalty * (x.sum() - target) ** 2

def lbfexp_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    weights = exps / exps.sum()
    grad_g = np.zeros(n)
    m_range = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        mv = k - (n - 1)
        i_lo = max(0, -mv)
        i_hi = min(n, n - mv)
        if i_lo < i_hi:
            grad_g[i_lo:i_hi] += w * (1.0 - x[i_lo + mv:i_hi + mv])
        i_lo2 = max(0, mv)
        i_hi2 = min(n, n + mv)
        if i_lo2 < i_hi2:
            grad_g[i_lo2:i_hi2] -= w * x[i_lo2 - mv:i_hi2 - mv]
    return dx * grad_g + 2.0 * penalty * (x.sum() - target)

def lbfexp_optimize(h, n, alpha, penalty, maxiter=3000):
    try:
        res = minimize(lbfexp_objective, h.copy(), jac=lambda x: lbfexp_gradient(x, n, alpha, penalty),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        delta_z = np.zeros((len(zeros), 2 * n - 1))
        for idx, i in enumerate(zeros):
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta_z[idx, msk] += (1.0 - h[im[msk]])
            imn = i - m_range
            msk2 = (imn >= 0) & (imn < n)
            delta_z[idx, msk2] -= h[imn[msk2]]
        delta_o = np.zeros((len(ones), 2 * n - 1))
        for idx, j in enumerate(ones):
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta_o[idx, msk] -= (1.0 - h[jm[msk]])
            jmn = j - m_range
            msk2 = (jmn >= 0) & (jmn < n)
            delta_o[idx, msk2] += h[jmn[msk2]]
        best_improve = 1e-16
        best_ii, best_jj = -1, -1
        for ii in range(len(zeros)):
            base = corr + delta_z[ii]
            new_maxes = np.max(base[None, :] + delta_o, axis=1) * dx
            jj = np.argmin(new_maxes)
            imp = cur_c - new_maxes[jj]
            if imp > best_improve:
                best_improve = imp
                best_ii, best_jj = ii, jj
        if best_ii < 0:
            break
        i = zeros[best_ii]
        j = ones[best_jj]
        h[i], h[j] = 1.0, 0.0
        corr += delta_z[best_ii] + delta_o[best_jj]
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def subgradient_polish(h, n, steps=500, alpha0=0.01):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    for s in range(steps):
        a = alpha0 / (1 + s * 0.01)
        g = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(g)
        m_k = k_star - (n - 1)
        subgrad = np.zeros(n)
        i_lo = max(0, -m_k)
        i_hi = min(n, n - m_k)
        subgrad[i_lo:i_hi] = (1.0 - h[i_lo + m_k:i_hi + m_k])
        i_lo2 = max(0, m_k)
        i_hi2 = min(n, n + m_k)
        subgrad[i_lo2:i_hi2] -= h[i_lo2 - m_k:i_hi2 - m_k]
        subgrad = dx * subgrad
        h = h - a * subgrad
        h = project_feasible(h, n)
    return h

        best_c, best_h = c_sp, h_sheet.copy()
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
