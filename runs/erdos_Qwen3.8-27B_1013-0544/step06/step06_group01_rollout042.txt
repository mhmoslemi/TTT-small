def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_feasible(arr, n):
    h = np.clip(np.asarray(arr, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(120):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-15:
                break
            h = np.clip(h + (diff / s) * room, 0.0, 1.0)
        else:
            s = float(h.sum())
            if s < 1e-15:
                break
            h = np.clip(h * (1.0 + diff / s), 0.0, 1.0)
    for _ in range(300):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            k = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[k])
            if add <= 1e-16:
                break
            h[k] += add
        else:
            k = int(np.argmax(h))
            sub = min(-diff, h[k])
            if sub <= 1e-16:
                break
            h[k] -= sub
    return h

    rows = np.arange(n, dtype=np.int64)
    center = n - 1
    S = np.zeros((n, 2 * n - 1), dtype=float)
    for i in range(n):
        S[i, center + i - rows] += 1.0 - h
        S[i, center - i + rows] += -h
    return h, c, S, rows


def update_S_coord(S, rows, q, delta):
    if delta == 0.0:
        return
    center = len(rows) - 1
    S[:, center + rows - q] -= delta
    S[:, center - rows + q] -= delta


def eval_into(c, m, d, center, idx1, idx2, tmp):
    tmp[:] = c
    if d != 0.0:
        tmp += d * m
        d2 = d * d
        tmp[idx1] += d2
        tmp[idx2] += d2
        tmp[center] -= 2.0 * d2
    return float(np.max(tmp))


def line_search_workers(c, m, lo, hi, M, center, idx1, idx2, fracs, small, tmp):
    lo = float(lo)
    hi = float(hi)
    M = float(M)
    W = hi - lo
    if W <= 1e-13:
        return 0.0, M
    best_d = 0.0
    best_val = M
    for f in fracs:
        d = lo + float(f) * W
        if d < lo:
            d = lo
        elif d > hi:
            d = hi
        if abs(d) < 1e-14:
            continue
        val = eval_into(c, m, d, center, idx1, idx2, tmp)
        if val < best_val - 1e-11:
            best_val = val
            best_d = d
    for s in small:
        d = float(s)
        if lo < d < hi:
            val = eval_into(c, m, d, center, idx1, idx2, tmp)
            if val < best_val - 1e-11:
                best_val = val
                best_d = d
        d = -d
        if lo < d < hi:
            val = eval_into(c, m, d, center, idx1, idx2, tmp)
            if val < best_val - 1e-11:
                best_val = val
                best_d = d
    return best_d, best_val


def golden_pair(c, m, lo, hi, center, idx1, idx2, tmp, iters=28):
    lo = float(lo)
    hi = float(hi)
    if hi - lo <= 1e-13:
        d = 0.5 * (lo + hi)
        return d, eval_into(c, m, d, center, idx1, idx2, tmp)
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    x1 = b - phi * (b - a)
    x2 = a + phi * (b - a)
    f1 = eval_into(c, m, x1, center, idx1, idx2, tmp)
    f2 = eval_into(c, m, x2, center, idx1, idx2, tmp)
    for _ in range(iters):
        if f1 <= f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - phi * (b - a)
            f1 = eval_into(c, m, x1, center, idx1, idx2, tmp)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + phi * (b - a)
            f2 = eval_into(c, m, x2, center, idx1, idx2, tmp)
    d = 0.5 * (a + b)
    return d, eval_into(c, m, d, center, idx1, idx2, tmp)


def refine_pair(h, c, S, center, i, j, M, init_d, fracs, small, tmp):
    hi = 1.0 - float(h[i])
    lo = -float(h[j])
    W = hi - lo
    if W <= 1e-13:
        return 0.0, float(M)
    m = S[i] - S[j]
    idx1 = center + i - j
    idx2 = center + j - i
    d, val = line_search_workers(c, m, lo, hi, float(M), center, idx1, idx2, fracs, small, tmp)
    if init_d is not None:
        idv = float(init_d)
        if abs(idv) > 1e-8:
            win = max(0.02, 0.10 * W)
            a = max(lo, idv - win)
            b = min(hi, idv + win)
            if b - a > 1e-8:
                dg, vg = golden_pair(c, m, a, b, center, idx1, idx2, tmp, 26)
                if vg < val - 1e-12:
                    d, val = dg, vg
    return d, float(val)


def apply_pair(h, c, S, rows, i, j, d):
    n = len(rows)
    center = n - 1
    hi = 1.0 - float(h[i])
    lo = -float(h[j])
    d = float(d)
    if d > hi:
        d = hi
    if d < lo:
        d = lo
    if abs(d) < 1e-14:
        return 0.0
    m = S[i] - S[j]
    c += d * m
    d2 = d * d
    idx1 = center + i - j
    idx2 = center + j - i
    c[idx1] += d2
    c[idx2] += d2
    c[center] -= 2.0 * d2
    h[i] += d
    h[j] -= d
    update_S_coord(S, rows, i, d)
    update_S_coord(S, rows, j, -d)
    return d


def polish_gradient(h, c, S, rows, deadline, rng, sets):
    n = len(h)
    L = 2 * n - 1
    center = n - 1
    dx = 2.0 / n
    frq, smq, frf, smf = sets
    tmp = np.empty(L, dtype=float)

    best_h = h.copy()
    best_M = float(np.max(c))
    best_c5 = best_M * dx
    stall = 0
    acc = 0

    while time.time() < deadline - 2.0:
        M = float(np.max(c))
        if M > best_M + 1e-7:
            h, c, S, rows = build_state(best_h, n)
            M = float(np.max(c))
        if M < best_M - 1e-12:
            best_M = M
            best_h = h.copy()

        tol_act = max(0.2, 0.003 * M)
        act = np.where(c >= M - tol_act)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(c))], dtype=int)
        if act.size > 50:
            act = act[np.argsort(c[act])[-50:]]

        G = S[:, act].mean(axis=1)
        order = np.argsort(G)
        P = min(8, max(2, n // 6))
        lows = order[:P]
        highs = order[-P:][::-1]

        pairs = []
        for a in lows:
            ia = int(a)
            for b in highs:
                jb = int(b)
                if ia != jb:
                    pairs.append((ia, jb))

        top0 = int(np.argmax(c))
        G1 = S[:, top0]
        ord1 = np.argsort(G1)
        P1 = min(4, max(1, n // 10))
        for a in ord1[:P1]:
            ia = int(a)
            for b in ord1[-P1:][::-1]:
                jb = int(b)
                if ia != jb:
                    pairs.append((ia, jb))

        for _ in range(10):
            ia = int(rng.randint(n))
            jb = int(rng.randint(n))
            if ia != jb:
                pairs.append((ia, jb))

        max_pairs = 80
        if len(pairs) > max_pairs:
            rng.shuffle(pairs)
            pairs = pairs[:max_pairs]

        best_d = 0.0
        best_val = M
        best_i = -1
        best_j = -1
        inner_deadline = deadline - 3.0

        for ia, jb in pairs:
            if time.time() > inner_deadline:
                break
            hi = 1.0 - float(h[ia])
            lo = -float(h[jb])
            if hi - lo <= 1e-13:
                continue
            m = S[ia] - S[jb]
            idx1 = center + ia - jb
            idx2 = center + jb - ia
            d, val = line_search_workers(c, m, lo, hi, M, center, idx1, idx2, frq, smq, tmp)
            if val < best_val - 1e-11:
                best_val = val
                best_d = d
                best_i = ia
                best_j = jb

        applied = False
        if best_i >= 0 and best_val < M - 1e-10:
            d, val = refine_pair(h, c, S, center, best_i, best_j, M, best_d, frf, smf, tmp)
            if val < best_val - 1e-12:
                best_val = val
                best_d = d
            if best_val < M - 1e-10:
                dd = apply_pair(h, c, S, rows, best_i, best_j, best_d)
                if dd != 0.0:
                    applied = True
                    stall = 0
                    acc += 1
                    if acc % 80 == 0:
                        h, c, S, rows = build_state(h, n)
                        M2 = float(np.max(c))
                        if M2 < best_M - 1e-12:
                            best_M = M2
                            best_h = h.copy()
                        if M2 > best_M + 1e-7:
                            h, c, S, rows = build_state(best_h, n)

        if not applied:
            stall += 1
            if stall >= 10:
                break

    h, c, S, rows = build_state(h, n)
    M = float(np.max(c))
    if M < best_M - 1e-12:
        best_M = M
        best_h = h.copy()
    return best_h, best_M * dx


def polish_state(h, n, deadline, rng, sets):
    h, c, S, rows = build_state(h, n)
    best_h, c5 = polish_gradient(h, c, S, rows, deadline, rng, sets)
    return best_h, c5


def try_candidate(h_start, n, deadline, rng, sets):
    h = project_feasible(h_start, n)
    best_h, c5 = polish_state(h, n, deadline, rng, sets)
    return best_h, c5, n


def resample_h(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    if n == 1:
        return np.full(n_new, float(h[0]), dtype=float)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_new)
    return np.interp(y, x, h)


def smooth_random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2.0 * math.pi * x) + 0.2 * np.cos(4.0 * math.pi * x + 0.9)
    h = h + rng.normal(0.0, 0.05, n)
    return project_feasible(h, n)


def random_binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, k, replace=False)
    else:
        idx = np.empty(0, dtype=int)
    h[idx] = 1.0
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        if rem.size > 0:
            h[int(rem[int(rng.randint(rem.size))])] = 0.5
    return project_feasible(h, n)


def quantize_half(h, n, rng):
    h2 = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = np.argpartition(h, n - k)[-k:]
        h2[idx] = 1.0
    else:
        idx = np.empty(0, dtype=int)
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        if rem.size > 0:
            h2[int(rem[int(rng.randint(rem.size))])] = 0.5
    return project_feasible(h2, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    rng = np.random.RandomState(seed)

    frq = np.linspace(0.0, 1.0, 13)
    smq = np.array([0.001, 0.004, 0.012, 0.035, 0.1], dtype=float)
    frf = np.linspace(0.0, 1.0, 41)
    smf = np.array([0.0003, 0.001, 0.003, 0.009, 0.025, 0.08], dtype=float)
    sets = (frq, smq, frf, smf)

    try:
        h_arr = np.array(initial_h_values, dtype=float).copy()
        if h_arr.size == 0:
            raise ValueError
        n0 = int(h_arr.size)
        if n0 < 4:
            n0 = 300
            h_arr = np.full(n0, 0.5, dtype=float)
        h0 = project_feasible(h_arr, n0)
    except Exception:
        n0 = 300
        h0 = smooth_random_start(n0, rng)

    best_h = h0
    best_n = int(n0)
    best_c = compute_c5(best_h)

    if time.time() < deadline - 120.0:
        end = min(deadline - 120.0, t0 + 300.0)
        if end - time.time() > 30.0:
            h_try, c_try, n_try = try_candidate(best_h, best_n, end, rng, sets)
            if c_try < best_c:
                best_h, best_c, best_n = h_try, c_try, int(n_try)

    n_trials = 0
    for nn in (250, 350, 200, 300, 400):
        if nn == best_n:
            continue
        if n_trials >= 3:
            break
        if time.time() > deadline - 260.0:
            break
        end = min(deadline - 140.0, time.time() + 150.0)
        if end - time.time() < 60.0:
            break
        h_start = resample_h(best_h, nn)
        h_try, c_try, n_try = try_candidate(h_start, nn, end, rng, sets)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, int(n_try)
        n_trials += 1

    scales = (0.03, 0.08, 0.18, 0.40, 0.85)
    for s in scales:
        if time.time() > deadline - 160.0:
            break
        end = min(deadline - 100.0, time.time() + 70.0)
        if end - time.time() < 30.0:
            break
        h_start = best_h + rng.normal(0.0, float(s), best_n)
        h_try, c_try, n_try = try_candidate(h_start, best_n, end, rng, sets)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, int(n_try)

    for sp in (0, 1):
        if time.time() > deadline - 130.0:
            break
        end = min(deadline - 90.0, time.time() + 80.0)
        if end - time.time() < 40.0:
            break
        if sp == 0:
            h_start = quantize_half(best_h, best_n, rng)
        else:
            h_start = random_binary_start(best_n, rng)
        h_try, c_try, n_try = try_candidate(h_start, best_n, end, rng, sets)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, int(n_try)

    if time.time() < deadline - 60.0:
        h_try, c_try, n_try = try_candidate(best_h, best_n, deadline - 2.0, rng, sets)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, int(n_try)

    best_h = np.asarray(best_h, dtype=float).copy()
    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5(best_h)
    return best_h, float(final_c), int(best_n)
