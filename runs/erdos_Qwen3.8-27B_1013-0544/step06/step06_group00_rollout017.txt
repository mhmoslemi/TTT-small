    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_value_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = g.max()
    eg = np.exp(alpha * (g - gm))
    se = eg.sum()
    val = gm + np.log(se) / alpha
    w = eg / se
    b = 1.0 - h
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        lo = max(0, n - 1 - m)
        hi = min(n, 2 * n - 1 - m)
        if hi > lo:
            T1[lo:hi] += wm * b[lo + m - (n-1):hi + m - (n-1)]
        lo = max(0, m - n + 1)
        hi = min(n, m)
        if hi > lo:
            T2[lo:hi] += wm * h[lo - m + (n-1):hi - m + (n-1)]
    grad = dx * (T1 - T2)
    return val, grad

def proj_grad(h, n, alpha, max_iter, deadline, lr=None):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    if lr is None:
        lr = 1.0 / (alpha * n * (2.0/n)**2)  # approximate Lipschitz
        lr = min(lr, 0.5)
    cur_val, _ = lse_value_and_grad(h, n, alpha)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = lse_value_and_grad(h, n, alpha)
        # Backtracking
        step = lr
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        for _ in range(50):
            h_new = project_feasible(h - step * grad, n)
            v_new, _ = lse_value_and_grad(h_new, n, alpha)
            if v_new <= cur_val - 1e-4 * step * gnorm**2:
                break
            step *= 0.5
        else:
            break
        h = h_new
        cur_val = v_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = step * 1.05
    return best_h, best_c5

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            best_delta = None
            best_j = -1
            best_nc = cur_c
            for j in ones[:]:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1+(j-i)
                delta[k1] += 1.0
                k2 = n-1+(i-j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_delta = delta
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def lse_value_and_grad(h, N, alpha):

    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = g.max()

    se = eg.sum()
    val = gm + np.log(se) / alpha
    w = eg / se
    b = 1.0 - h
    
    T1 = np.zeros(N)
    T2 = np.zeros(N)
    for m in range(2 * N - 1):

        if wm < 1e-13:
            continue
        # T1[j] gets b[j-m+(N-1)] when max(0,m-N+1) <= j <= min(N-1, m)
        lo = max(0, m - N + 1)
        hi = min(N, m + 1)  # exclusive, since j <= m means j < m+1
        if hi > lo:

            # b index: j-m+(N-1) ranges from lo-m+(N-1) to hi-1-m+(N-1)
            T1[lo:hi] += wm * b[lo - m + (N-1) : hi - m + (N-1)]
        
        # T2[j] gets h[j+m-(N-1)] when max(0,N-1-m) <= j <= min(N-1, 2N-2-m)
        lo = max(0, N - 1 - m)
        hi = min(N, 2 * N - 1 - m)  # exclusive, j <= 2N-2-m means j < 2N-1-m
        if hi > lo:
            T2[lo:hi] += wm * h[lo + m - (N-1) : hi + m - (N-1)]
    
    grad = dx * (T1 - T2)
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_val_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = g.max()
    eg = np.exp(alpha * (g - gm))
    se = eg.sum()
    val = gm + np.log(se) / alpha
    w = eg / se
    b = 1.0 - h
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        if hi > lo:
            T1[lo:hi] += wm * b[lo - m + (n-1):hi - m + (n-1)]
        lo = max(0, n - 1 - m)
        hi = min(n, 2 * n - 1 - m)
        if hi > lo:
            T2[lo:hi] += wm * h[lo + m - (n-1):hi + m - (n-1)]
    grad = dx * (T1 - T2)
    return val, grad

def proj_grad(h, n, alpha, max_iter, deadline, lr=None):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    if lr is None:
        lr = n / (4.0 * alpha)
        lr = min(lr, 0.5)
    cur_val, _ = lse_val_grad(h, n, alpha)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = lse_val_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr
        succ = False
        for _ in range(40):
            h_new = project_feasible(h - step * grad, n)
            v_new, _ = lse_val_grad(h_new, n, alpha)
            if v_new <= cur_val - 1e-5 * step * gnorm * gnorm:
                succ = True
                break
            step *= 0.5
        if not succ:
            break
        h = h_new
        cur_val = v_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = step * 1.02
    return best_h, best_c5

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            best_nc = cur_c
            best_j = -1
            best_delta = None
            for j in ones[:]:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_delta = delta
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Binary local search (fast)
    dl1 = t0 + 0.05 * budget_s
    hb, cb = greedy_binary(h0, n, dl1, max_sweeps=200)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb.copy()

    # Phase 2: Projected gradient with increasing alpha (main workhorse)
    alphas = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 15000]
    h = project_feasible(h0, n)
    for i, alpha in enumerate(alphas):
        frac = 0.05 + 0.04 * (i + 1)
        dl = min(t0 + frac * budget_s, deadline)
        if time.time() > dl:
            break
        h = proj_grad(h, n, alpha, max_iter=3000, deadline=dl)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: SLSQP refinement at high alpha
    dl3 = min(t0 + 0.60 * budget_s, deadline)
    h3 = project_feasible(best_h, n)
    for alpha in [5000, 20000, 50000]:
        if time.time() > dl3:
            break
        h3 = slsqp_refine(h3, n, alpha, 500, 1e-18, dl3)
        c5 = compute_c5(h3, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h3.copy()

    # Phase 4: Binary search from current best
    dl4 = min(t0 + 0.72 * budget_s, deadline)
    hb2, cb2 = greedy_binary(best_h, n, dl4, max_sweeps=200)
    if cb2 < best_c5:
        best_c5 = cb2
        best_h = hb2.copy()

    # Phase 5: More projected gradient from binary result
    hl = project_feasible(best_h, n)
    for alpha in [3000, 10000, 30000]:
        dl = min(t0 + 0.85 * budget_s, deadline)
        if time.time() > dl:
            break
        hl = proj_grad(hl, n, alpha, max_iter=2000, deadline=dl)
        c5 = compute_c5(hl, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hl.copy()

    # Phase 6: SLSQP final polish
    dl6 = deadline - 10
    h6 = project_feasible(best_h, n)
    for alpha in [50000, 100000]:
        if time.time() > dl6:
            break
        h6 = slsqp_refine(h6, n, alpha, 300, 1e-18, dl6)
        c5 = compute_c5(h6, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h6.copy()

    # Phase 7: Perturbation restarts
    for trial in range(4):
        if time.time() > deadline - 3:
            break
        for scale in [0.01, 0.03, 0.05]:
            if time.time() > deadline - 3:
                break
            hp = project_feasible(best_h + scale * rng.randn(n), n)
            hp = proj_grad(hp, n, 500, max_iter=500, deadline=deadline - 3)
            c5 = compute_c5(hp, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
            hp = proj_grad(hp, n, 5000, max_iter=500, deadline=deadline - 3)
            c5 = compute_c5(hp, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

    # Final SLSQP polish
    if time.time() < deadline - 2:
        hf = project_feasible(best_h, n)
        hf = slsqp_refine(hf, n, 100000, 200, 1e-18, deadline - 1)
        c5 = compute_c5(hf, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hf.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
