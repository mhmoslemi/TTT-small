def alpha_obj_val(x,n,a):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    u = a*(c-m)
    um = float(u.max())
    e = np.exp(u-um)
    return m + math.log(float(e.sum()))/a

def alpha_obj_grad(x,n,a):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    u = a*(c-m)
    um = float(u.max())
    e = np.exp(u-um)
    s = float(e.sum())
    w = e/s
    f = m + math.log(s)/a
    grad = np.zeros(n, dtype=float)
    nz = np.nonzero(w > 1e-5)[0]
    for p in nz:
        wp = float(w[p])
        l = int(p) - (n-1)
        if l >= 0:
            cnt = n - l
            if cnt > 0:
                grad[l:cnt+l] += wp*(1.0 - x[:cnt])
                grad[:cnt] -= wp*x[l:cnt+l]
        else:
            d = -l
            cnt = n - d
            if cnt > 0:
                grad[:cnt] += wp*(1.0 - x[d:d+cnt])
                grad[d:d+cnt] -= wp*x[:cnt]
    return f, grad

import numpy as np
import time
import math
from scipy.optimize import minimize
from itertools import combinations

_sm_n = 250
_sm_alpha = 100.0


def c5_value(h, n):
    if n < 2:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, 'full'))) * (2.0 / n)


def project_boxsum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    np.clip(x, 0.0, 1.0, out=x)
    s = float(x.sum())
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)


def fix_sum_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    diff = target - float(h.sum())
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)[::-1]
        add = diff
        for k in idx:
            if add <= 1e-15:
                break
            d = min(1.0 - float(h[k]), add)
            h[k] += d
            add -= d
    else:
        idx = np.argsort(h)[::-1]
        add = -diff
        for k in idx:
            if add <= 1e-15:
                break
            d = min(float(h[k]), add)
            h[k] -= d
            add -= d
    return h


def round_to_binary(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    target = n // 2
    out = np.zeros(n, dtype=np.int8)
    if target > 0:
        idx = np.argsort(h)[::-1][:target]
        out[idx] = 1
    return out


def make_default_h(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) < 0.5).astype(float)
    return project_boxsum(h, 0.5 * n)


def resample_to_n(h, n1, n2):
    if n1 == n2:
        return np.asarray(h, dtype=float).copy()
    xo = (np.arange(n1) + 0.5) * (2.0 / n1)
    xn = (np.arange(n2) + 0.5) * (2.0 / n2)
    return np.interp(xn, xo, np.asarray(h, dtype=float))


def apply_swap_int(corr, h, b, i, j, n):
    L = 2 * n - 1
    new = corr.copy()
    br = b[::-1]
    new[i:i + n] += br
    new[n - 1 - i:L - i] -= h
    new[j:j + n] -= br
    new[n - 1 - j:L - j] += h
    new[n - 1] -= 2
    new[n - 1 + (j - i)] += 1
    new[n - 1 + (i - j)] += 1
    return new


def binary_sa(h0, n, deadline, seed, t_start=2.5, t_end=0.02, directed_prob=0.78, patience=25000):
    n = int(n)
    h = round_to_binary(h0, n).astype(np.int8)
    b = (1 - h).astype(np.int8)
    corr = np.correlate(h, b, 'full').astype(np.int16)
    dx = 2.0 / n
    cur = int(corr.max())
    best = cur
    best_h = h.copy()
    zeros = np.nonzero(h == 0)[0]
    ones = np.nonzero(h == 1)[0]
    rng = np.random.RandomState(seed)
    T = t_start
    cool = (t_end / t_start) ** (1.0 / 200000.0)
    noimp = 0
    step = 0
    top_lags = None
    top_len = min(32, 2 * n - 1)

    while time.time() < deadline and zeros.size > 0 and ones.size > 0:
        step += 1
        if step % 80 == 0 or top_lags is None:
            top_lags = np.argsort(corr)[-top_len:]
        if top_lags is None:
            top_lags = np.arange(2 * n - 1)

        i = -1
        j = -1
        if rng.random() < directed_prob:
            p = int(top_lags[rng.randint(len(top_lags))])
            lag = p - (n - 1)
            ok = False
            if lag >= 0:
                cnt = n - lag
                if cnt > 0:
                    mask = (h[lag:lag + cnt] == 1) & (h[:cnt] == 0)
                    idxs = np.nonzero(mask)[0]
                    if idxs.size:
                        r = int(idxs[rng.randint(idxs.size)])
                        one_pos = r + lag
                        zero_pos = r
                        ok = True
            else:
                d = -lag
                cnt = n - d
                if cnt > 0:
                    mask = (h[:cnt] == 1) & (h[d:cnt + d] == 0)
                    idxs = np.nonzero(mask)[0]
                    if idxs.size:
                        r = int(idxs[rng.randint(idxs.size)])
                        one_pos = r
                        zero_pos = r + d
                        ok = True
            if ok:
                if rng.random() < 0.5:
                    j = one_pos
                    i = int(zeros[rng.randint(len(zeros))])
                else:
                    i = zero_pos
                    j = int(ones[rng.randint(len(ones))])

        if i < 0:
            i = int(zeros[rng.randint(len(zeros))])
            j = int(ones[rng.randint(len(ones))])
        if i == j:
            continue

        new = apply_swap_int(corr, h, b, i, j, n)
        nc = int(new.max())
        delta = nc - cur
        if delta <= 0 or (T > 1e-8 and rng.random() < math.exp(-delta / T)):
            h[i] = 1
            h[j] = 0
            b[i] = 0
            b[j] = 1
            corr = new
            cur = nc
            zeros = np.nonzero(h == 0)[0]
            ones = np.nonzero(h == 1)[0]
            if nc < best:
                best = nc
                best_h = h.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
        T = max(t_end, T * cool)

        if noimp > patience:
            kk = 3 + rng.randint(8)
            h = best_h.copy()

                z = np.nonzero(h == 0)[0]
                o = np.nonzero(h == 1)[0]
                if z.size == 0 or o.size == 0:
                    break
                ii = int(z[rng.randint(len(z))])
                jj = int(o[rng.randint(len(o))])
                h[ii] = 1
                h[jj] = 0
            corr = np.correlate(h, (1 - h).astype(np.int8), 'full').astype(np.int16)
            cur = int(corr.max())
            zeros = np.nonzero(h == 0)[0]
            ones = np.nonzero(h == 1)[0]
            b = (1 - h).astype(np.int8)
            noimp = 0
            top_lags = None
            T = min(t_start, T * 3.0)

    return best_h.astype(float), best * dx


def exhaustive_binary_small(n, deadline):
    n = int(n)
    target = n // 2
    best_count = 10 ** 9
    best_arr = None
    h = np.zeros(n, dtype=np.int8)
    lower = (n * n // 4 + (2 * n - 2) - 1) // (2 * n - 2)
    for comb in combinations(range(n), target):
        if time.time() > deadline:
            break
        h.fill(0)
        h[list(comb)] = 1
        c = int(np.correlate(h, 1 - h, 'full').max())
        if c < best_count:
            best_count = c
            best_arr = h.copy()
            if best_count <= lower:
                break
    if best_arr is None:
        return None, 1e9
    return best_arr.astype(float), float(best_count) * (2.0 / n)


def alpha_obj_val(x, n, a):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    u = a * (c - m)
    um = float(u.max())
    e = np.exp(u - um)
    return m + math.log(float(e.sum())) / a


def alpha_obj_grad(x, n, a):
    c = np.correlate(x, 1.0 - x, 'full')
    m = float(c.max())
    u = a * (c - m)
    um = float(u.max())
    e = np.exp(u - um)
    s = float(e.sum())
    w = e / s
    f = m + math.log(s) / a
    grad = np.zeros(int(n), dtype=float)
    nz = np.nonzero(w > 1e-5)[0]
    for p in nz:
        wp = float(w[p])
        lag = int(p) - (n - 1)
        if lag >= 0:
            cnt = n - lag
            if cnt > 0:
                grad[lag:cnt + lag] += wp * (1.0 - x[:cnt])
                grad[:cnt] -= wp * x[lag:cnt + lag]
        else:
            d = -lag
            cnt = n - d
            if cnt > 0:
                grad[:cnt] += wp * (1.0 - x[d:cnt + d])
                grad[d:cnt + d] -= wp * x[:cnt]
    return f, grad


def smooth_fun(x):
    return alpha_obj_grad(x, _sm_n, _sm_alpha)[0]


def smooth_jac(x):
    return alpha_obj_grad(x, _sm_n, _sm_alpha)[1]


def smooth_con(x):
    return float(np.sum(x)) - 0.5 * float(_sm_n)


def smooth_slqp(h, n, deadline, alphas=(50, 200, 1000, 5000, 20000), iters=80):
    global _sm_n, _sm_alpha
    n = int(n)
    best = project_boxsum(h, 0.5 * n)
    best_c = c5_value(best, n)
    for a in alphas:
        if time.time() > deadline - 5:
            break
        _sm_n = n
        _sm_alpha = float(a)
        start = best.copy()
        if a > 200:
            rng = np.random.RandomState(1000 + int(min(8900, a % 8900)))
            start = project_boxsum(best + 0.02 * rng.randn(n), 0.5 * n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': smooth_con}]
        try:
            res = minimize(smooth_fun, start, method='SLSQP', jac=smooth_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': int(iters), 'ftol': 1e-9})
            cand = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(cand)):
                cand = start
            cand = project_boxsum(cand, 0.5 * n)
        except Exception:
            cand = start
        cc = c5_value(cand, n)
        if cc < best_c:
            best_c = cc
            best = cand
    return best, best_c


def projected_smooth(h, n, deadline, alphas=(100, 1000, 5000)):
    n = int(n)
    target = 0.5 * n
    h = project_boxsum(h, target)
    best_h = h.copy()
    best_c = c5_value(h, n)
    rng = np.random.RandomState(12345)
    for a in alphas:
        if time.time() > deadline - 5:
            break
        h = project_boxsum(h, target)
        f = alpha_obj_val(h, n, float(a))
        step = 0.5
        bad = 0
        while time.time() < deadline - 2:
            f, grad = alpha_obj_grad(h, n, float(a))
            f = float(f)
            gnorm = math.sqrt(float(grad.dot(grad)))
            if gnorm < 1e-12:
                break
            s = step / (0.1 + gnorm)
            accepted = False
            for _ in range(6):
                h2 = project_boxsum(h - s * grad, target)
                f2 = alpha_obj_val(h2, n, float(a))
                if f2 < f - 1e-14:
                    h = h2
                    f = f2
                    step = min(s * 1.25, 2.0)
                    cc = c5_value(h, n)
                    if cc < best_c:
                        best_c = cc
                        best_h = h.copy()
                    accepted = True
                    bad = 0
                    break
                s *= 0.5
            if not accepted:
                bad += 1
                step *= 0.7
                if bad > 40:
                    h = project_boxsum(best_h + 0.015 * rng.randn(n), target)
                    f = alpha_obj_val(h, n, float(a))
                    step = 0.5
                    bad = 0
    return best_h, best_c
