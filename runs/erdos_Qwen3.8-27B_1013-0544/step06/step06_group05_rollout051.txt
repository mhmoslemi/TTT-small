```python
def simulated_annealing(h, n, deadline, n_restarts=5):
    dx = 2.0 / n
    best_h = h.copy()
    best_max = np.inf
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        
        if restart == 0:
            cur = h.copy()
        else:
            # Perturb the best solution
            cur = best_h.copy()
            n_swaps = np.random.randint(5, 20)
            for _ in range(n_swaps):
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                cur[i], cur[j] = 0.0, 1.0
        
        cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
        cur_max = cur_corr.max()
        
        T = 5.0  # initial temperature
        T_min = 0.01
        cooling = 0.9995
        
        while time.time() < deadline:
            # Pick a random swap
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Make the swap
            cur[i], cur[j] = 1.0 - cur[i], 1.0 - cur[j]
            
            new_corr = np.correlate(cur, 1.0 - cur, mode='full')
            new_max = new_corr.max()
            
            delta = new_max - cur_max
            
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                # Accept
                cur_max = new_max
                cur_corr = new_corr
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = cur.copy()
            else:
                # Reject
                cur[i], cur[j] = 1.0 - cur[i], 1.0 - cur[j]
            
            T *= cooling
            if T < T_min:
                break
    
    return best_h, best_max * dx
```

```python
    b[idx[:n//2]] = 1.0
```

```python
def thue_morse_bin(n, seed=0):
    """Generate a binary sequence with n/2 ones based on Thue-Morse."""
    rng = np.random.RandomState(seed)
    # Generate Thue-Morse values
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Adjust to have exactly n/2 ones
    n_ones = int(tm.sum())
    n_zeros = n - n_ones
    if n_ones > n // 2:
        # Need to flip some 1s to 0s
        ones_idx = np.where(tm == 1)[0]
        rng.shuffle(ones_idx)
        for k in range(n_ones - n // 2):
            tm[ones_idx[k]] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(tm == 0)[0]
        rng.shuffle(zeros_idx)
        for k in range(n // 2 - n_ones):
            tm[zeros_idx[k]] = 1.0
    return tm

def sa_guided(h0, n, deadline, seed, T_init=2.0, cooling=0.99998, guide_prob=0.6):
    """Guided simulated annealing."""
    rng = np.random.RandomState(seed)
    cur = h0.copy()
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_mc = float(cur_corr.max())
    best = cur.copy()
    best_mc = cur_mc
    T = T_init
    stall = 0
    
    while time.time() < deadline and T > 0.01:
        # Find bottleneck lag
        lag_max = int(np.argmax(cur_corr)) - (n - 1)
        
        oi = np.where(cur == 1)[0]
        zi = np.where(cur == 0)[0]
        
        # Guided choice of i
        if rng.rand() < guide_prob and len(oi) > 0:
            # Find i where cur[i]=1 and cur[i-lag_max]=0
            if lag_max >= 0:
                lo, hi = lag_max, n
            else:
                lo, hi = 0, n + lag_max
            candidates = []
            for i in range(lo, hi):
                if cur[i] == 1 and cur[i - lag_max] == 0:
                    candidates.append(i)
            if candidates:
                i = candidates[rng.randint(len(candidates))]
            else:
                i = oi[rng.randint(len(oi))]
        else:
            i = oi[rng.randint(len(oi))]
        
        j = zi[rng.randint(len(zi))]
        
        cur[i] = 0.0
        cur[j] = 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
        new_mc = float(new_corr.max())
        d = new_mc - cur_mc
        if d < 0 or rng.rand() < np.exp(-d / T):
            cur_mc = new_mc
            cur_corr = new_corr
            if new_mc < best_mc:
                best_mc = new_mc
                best = cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
        
        T *= cooling
    
    # Restart mechanism is handled externally
    return best, best_mc
```
