def force_sum(h, target):
    h=np.array(h,dtype=float,copy=True)
    np.clip(h,0.0,1.0,out=h)
    diff=target - float(h.sum())
    if abs(diff) > 1e-15:
        idx=int(np.argmin(np.abs(h-0.5)))
        max_add=1.0-h[idx]
        min_add=h[idx]
        if diff > 0:
            add=min(diff, max_add)
        else:
            add=max(diff, -min_add)
        h[idx] += add
        rem = diff - add
        if abs(rem) > 1e-15:
           if rem > 0:
              order = np.where(h < 1.0)[0]
              for ii in order:
                 if rem <=0: break
                 a = min(rem, 1.0 - h[ii])

           else:
              order = np.where(h > 0.0)[0]

                 if rem >=0: break
                 a = min(-rem, h[ii])
                 h[ii] -= a; rem += a
    return h

import numpy as np
import time


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def force_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = float(target) - float(h.sum())
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        if diff > 0.0:
            add = min(diff, 1.0 - h[idx])
        else:
            add = max(diff, -h[idx])
        h[idx] += add
        rem = diff - add
        if abs(rem) > 1e-15:
            if rem > 0.0:
                order = np.where(h < 1.0)[0]
                for ii in order:
                    if rem <= 0.0:
                        break
                    a = min(rem, 1.0 - h[ii])
                    h[ii] += a
                    rem -= a
            else:
                order = np.where(h > 0.0)[0]
                for ii in order:
                    if rem >= 0.0:
                        break
                    a = min(-rem, h[ii])
                    h[ii] -= a
                    rem += a
    return h


def scale_scan(h, n, deadline):
    target = float(n) / 2.0
    h = force_sum(np.array(h, dtype=float, copy=True), target)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_lam = 1.0

    base = h - 0.5
    dmax = float(np.max(np.abs(base))) if h.size else 0.0
    if dmax < 1e-12:
        return best_h, best_c

    lam_max = min(2.0, 1.0 / dmax)
    if lam_max <= 0.0:
        return best_h, best_c

    lams = np.linspace(0.0, lam_max, 121)
    for lam in lams:
        if time.time() >= deadline:
            break
        lamf = float(lam)
        h2 = 0.5 + lamf * base
        np.clip(h2, 0.0, 1.0, out=h2)
        h2 = force_sum(h2, target)
        c = compute_c5(h2, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h2.copy()
            best_lam = lamf

    lo = max(0.0, best_lam - 0.05)
    hi = min(lam_max, best_lam + 0.05)
    if hi > lo:
        lams2 = np.linspace(lo, hi, 41)
        for lam in lams2:
            if time.time() >= deadline:
                break
            lamf = float(lam)
            h2 = 0.5 + lamf * base
            np.clip(h2, 0.0, 1.0, out=h2)
            h2 = force_sum(h2, target)
            c = compute_c5(h2, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h2.copy()

    best_h = force_sum(best_h, target)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def cosine_candidate_best(n, deadline):
    n = int(n)
    if n % 2 != 0:
        n -= 1
    if n < 16:
        return None, float("inf")

    target = float(n) / 2.0
    idx = np.arange(n)
    two_pi = 2.0 * np.pi
    phases = two_pi * np.arange(16) / 16.0

    best_h = None
    best_c = float("inf")

    for ph in phases:
        if time.time() >= deadline:
            break
        basis = np.cos(two_pi * idx / float(n) + float(ph))
        h = 0.5 + 0.5 * basis
        h = force_sum(h, target)
        hh, cc = scale_scan(h, n, deadline)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    return best_h, best_c

    except Exception:

            best_h = bh.copy()

    best_h = force_sum(best_h, float(best_n) / 2.0)

    return best_h, float(best_c), int(best_n)
