import numpy as np
from scipy.optimize import minimize
import time

_n_cur = 400
_alpha_cur = 1000
_dl_cur = 999999.0

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    mx = corr.max()
    sh = alpha * (corr - mx)
    es = np.exp(sh)
    se = es.sum()
    p = es / se
    obj = mx + np.log(se) / alpha
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        pm = p[m] * dx
        if pm < 1e-15:
            continue
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += pm * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= pm * h[lo2 + s:hi2 + s]
    return obj, grad

def _obj_f(x):
    if time.time() > _dl_cur:
        raise ValueError("timeout")
    o, _ = smooth_obj_grad(x, _n_cur, _alpha_cur)
    return o

def _jac_f(x):
    if time.time() > _dl_cur:
        raise ValueError("timeout")
    _, g = smooth_obj_grad(x, _n_cur, _alpha_cur)
    return g

def _cons_f(x):

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline):

    _n_cur = n
    _alpha_cur = alpha
    _dl_cur = deadline
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
    try:
        res = minimize(_obj_f, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_jac_f,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_sweep(h, n, deadline, alphas, maxiter, ftol):
    best = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_opt(h, n, alpha, maxiter, ftol, deadline - 3)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best = h_new.copy()
            h = h_new.copy()
        else:
            h = best.copy()
    return best, best_c

    xo = np.linspace(0, 2, n_from)
    xn = np.linspace(0, 2, n_to)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, n_to)

    # Phase 2: Main SLSQP with analytic gradient
    p2 = t0 + 0.55 * total
