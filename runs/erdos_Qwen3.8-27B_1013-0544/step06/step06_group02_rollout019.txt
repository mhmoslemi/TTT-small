import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _cons(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, alphas=None, max_iters=300):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=200000, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def build_F(h, n):
    """Build F matrix: F[j,m] = (1-h[j-k])_v - h[j+k]_v, k=m-(n-1)."""
    F = np.zeros((n, 2*n-1))
    k = np.arange(-(n-1), n)
    for j in range(n):
        idx_neg = j - k
        mask_neg = (idx_neg >= 0) & (idx_neg < n)
        F[j, mask_neg] = (1.0 - h[idx_neg[mask_neg]])
        idx_pos = j + k
        mask_pos = (idx_pos >= 0) & (idx_pos < n)
        F[j, mask_pos] -= h[idx_pos[mask_pos]]
    return F

def batch_pair_search(h, n, deadline, deltas=None, verify_every=50):
    """Fast batch pair search using vectorized evaluation."""
    if deltas is None:
        deltas = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    best_c = float(c.max())
    
    for d in deltas:
        improved_any = True
        count = 0
        while improved_any and time.time() < deadline:
            improved_any = False
            F = build_F(h, n)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c * dx
                if h[i] + d > 1.0 + 1e-12:
                    continue
                F_i = F[i]
                base = c + d * F_i
                # For all j: check max(c + d*F_i - d*F[j])
                cand = (base[None, :] - d * F).max(axis=1)
                feasible = (h >= d - 1e-12).copy()
                feasible[i] = False
                idx = np.where((cand < best_c - 1e-13) & feasible)[0]
                if len(idx) > 0:
                    j_best = idx[np.argmin(cand[idx])]
                    h[i] += d
                    h[j_best] -= d
                    c = base - d * F[j_best]
                    best_c = float(c.max())
                    improved_any = True
                    count += 1
                    if count % verify_every == 0:
                        c = compute_c5_vec(h, n)
                        best_c = float(c.max())
                        # Re-verify feasibility
                        h = project_feasible(h, n)
                        c = compute_c5_vec(h, n)
                        best_c = float(c.max())
    return h, best_c * dx

def gradient_descent_opt(h, n, deadline, lr_init=0.05, alpha_init=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.5
        if lr < 1e-14:
            break
    return best_h, best_c

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)
