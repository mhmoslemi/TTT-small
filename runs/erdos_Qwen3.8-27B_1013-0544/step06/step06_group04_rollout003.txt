def project_feasible(h,n,max_iter=200):
    target = 0.5*n
    h = np.clip(np.asarray(h,dtype=float),0.0,1.0).copy()
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
        h = np.clip(h,0.0,1.0)
    return h

```python
import numpy as np
import time
from scipy.optimize import minimize

_PARAM_N = 0
_PARAM_X = None


def sort_key(t):
    return t[0]


def project_feasible(h, n, max_iter=200):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def eval_counts(h, n):
    N = 2 * n
    F = np.fft.rfft(h, N)
    A = np.fft.irfft(F * np.conj(F), N)[:n]
    A = np.asarray(A, dtype=float)
    A = np.maximum(A, 0.0)
    prefix = np.empty(n + 1)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(h)
    S = float(prefix[n])
    c0 = S - A[0]
    if n <= 1:
        return float(c0), float(c0), np.zeros(0), np.zeros(0)
    S0 = prefix[1:n][::-1]
    S1 = S - prefix[1:n]
    cpos = S1 - A[1:]
    cneg = S0 - A[1:]
    cmax = float(max(c0, np.max(cpos), np.max(cneg)))
    return cmax, float(c0), cpos, cneg


def gradient_weights(h, n, w0, wpos, wneg):
    g = np.empty(n)
    g[:] = w0 * (1.0 - 2.0 * h)
    for idx in range(n - 1):
        wp = float(wpos[idx])
        wq = float(wneg[idx])
        ws = wp + wq
        d = idx + 1
        if ws > 1e-14:
            g[:-d] -= ws * h[d:]
            g[d:] -= ws * h[:-d]
        if wp > 1e-14:
            g[d:] += wp
        if wq > 1e-14:
            g[:n - d] += wq
    return g


def soft_weights(h, n, beta):
    cmax, c0, cpos, cneg = eval_counts(h, n)
    vals = np.empty(2 * n - 1)
    vals[0] = c0
    vals[1:n] = cpos
    vals[n:2 * n - 1] = cneg
    m = float(vals.max())
    sh = beta * (vals - m)
    e = np.exp(sh)
    e_sum = float(e.sum())
    if e_sum <= 0.0:
        w0 = 1.0
        wpos = np.zeros(max(0, n - 1))
        wneg = np.zeros(max(0, n - 1))
    else:
        w = e / e_sum
        w0 = w[0]
        wpos = w[1:n]
        wneg = w[n:2 * n - 1]
    obj = m + (float(np.log(e_sum)) / beta if e_sum > 0.0 else 0.0)
    return float(obj), cmax, w0, wpos, wneg


def smooth_obj(x, n, beta, mu):
    x = np.asarray(x, dtype=float)
    obj, _, _, _, _ = soft_weights(x, n, beta)
    s = x.sum() - 0.5 * n
    val = obj + mu * s * s
    if not np.isfinite(val):
        return 1e6
    return float(val)


def smooth_jac(x, n, beta, mu):
    x = np.asarray(x, dtype=float)
    _, _, w0, wpos, wneg = soft_weights(x, n, beta)
    g = gradient_weights(x, n, w0, wpos, wneg)
    s = x.sum() - 0.5 * n
    if mu != 0.0:
        g += 2.0 * mu * s
    return g


def lbfgs_once(h, n, beta, mu, maxiter):
    h = project_feasible(h, n, 100)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_obj,
            h,
            jac=smooth_jac,
            args=(n, beta, mu),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'gtol': 1e-8,
                     'maxls': 10, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            h2 = project_feasible(res.x, n, 300)
            c = compute_c5(h2, n)
            if np.isfinite(c):
                return h2, c
    except Exception:
        pass
    return h, compute_c5(h, n)


def lbfgs_phase(h, n, deadline, betas):
    best_h = np.asarray(h, dtype=float).copy()
    best_c = compute_c5(best_h, n)
    cur_h = best_h
    for beta in betas:
        if time.time() > deadline:
            break
        maxit = 35 if n <= 300 else 18
        hh, cc = lbfgs_once(cur_h, n, float(beta), 100.0, maxit)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
            cur_h = hh
        elif cc < best_c + 1e-5:
            cur_h = hh
        else:
            cur_h = best_h
    return best_h, best_c


def make_h_params(p, n):
    x = _PARAM_X
    h = 0.5 * np.ones(n)
    for k in range(3):
        a = p[3 * k]
        f = p[3 * k + 1]
        ph = p[3 * k + 2]
        if abs(a) > 1e-14:
            h += a * np.cos(np.pi * f * x + ph)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n, 60)


def param_objective(p):

    if n <= 0:
        return 1.0
    h = make_h_params(np.asarray(p, dtype=float), n)
    c = compute_c5(h, n)

            p[5] = rng.uniform(0.0, p2)
            p[6] = rng.uniform(0.0, 0.30)
            p[7] = rng.uniform(2.0, 6.0)
            p[8] = rng.uniform(0.0, p2)
        p[0::3] = np.clip(p[0::3], -0.75, 0.75)
        p[1::3] = np.clip(p[1::3], 0.1, 6.0)
        p[2::3] = np.clip(p[2::3], 0.0, p2)
        candidates.append(p)
    scored = []
    for p in candidates:
        if time.time() > deadline:
            break
        scored.append((param_objective(p), p))
    scored.sort(key=sort_key)
    best_p = scored[0][1] if scored else known
    best_c = scored[0][0] if scored else compute_c5(make_h_params(best_p, n), n)
    top = scored[:min(8, len(scored))]
    for _, p in top:
        if time.time() > deadline - 25:
            break
        try:
            res = minimize(
                param_objective_args,
                np.asarray(p, dtype=float),
                args=(n,),
                method='Powell',
                bounds=bounds,
                options={'maxiter': 22, 'ftol': 1e-10, 'disp': False}
            )
            if hasattr(res, 'fun') and np.isfinite(res.fun) and res.fun < best_c:
                best_c = float(res.fun)
                best_p = np.asarray(res.x, dtype=float)
        except Exception:
            pass
    h = make_h_params(best_p, n)
    h = project_feasible(h, n, 300)
    c = compute_c5(h, n)
    return h, c


def interpolate_to(h, n_old, n_new):
    if n_new == int(n_old):
        return project_feasible(h, n_new, 200)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    hh = np.interp(x_new, x_old, np.asarray(h, dtype=float))
    return project_feasible(hh, n_new, 300)


def binarize(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.int8)
    b[idx[:n // 2]] = 1
    return b


def build_binary_db(h, n):
    h = np.asarray(h, dtype=np.int8)
    h32 = h.astype(np.int32)
    L = n - 1
    cP = np.empty(L, dtype=np.int32)
    cQ = np.empty(L, dtype=np.int32)
    for d in range(1, n):
        m = n - d
        cP[d - 1] = int(np.dot(h32[d:], 1 - h32[:m]))
        cQ[d - 1] = int(np.dot(h32[:m], 1 - h32[d:]))
    Gp = np.zeros((n, L), dtype=np.int8)
    Gq = np.zeros((n, L), dtype=np.int8)
    for d in range(1, n):
        idx = d - 1
        m = n - d
        Gp[d:, idx] = 1 - h[:m]
        Gp[:m, idx] += -h[d:]
        Gq[:m, idx] = 1 - h[d:]
        Gq[d:, idx] += -h[:m]
    state = {'cP': cP, 'cQ': cQ, 'Gp': Gp, 'Gq': Gq}
    cur = int(max(cP.max(), cQ.max()))
    return state, cur


def flip_update(state, a, s, n):
    pos = n - 1 - a
    if pos > 0:
        d = np.arange(1, pos + 1)
        state['Gp'][a + d, d - 1] += s
        state['Gq'][a + d, d - 1] += s
    neg = a
    if neg > 0:
        d = np.arange(1, neg + 1)
        state['Gp'][a - d, d - 1] += s
        state['Gq'][a - d, d - 1] += s


def swap_effect(state, i, j, n):
    d = abs(i - j)
    p = state['cP'] + state['Gp'][i] - state['Gp'][j]
    q = state['cQ'] + state['Gq'][i] - state['Gq'][j]
    if d > 0:
        idx = d - 1
        p[idx] += 1
        q[idx] += 1
    newc = int(max(p.max(), q.max()))
    return p, q, newc


def accept_swap(state, h, i, j, n, p, q):
    state['cP'] = p.astype(np.int32, copy=True)
    state['cQ'] = q.astype(np.int32, copy=True)
    h[i] = 1
    h[j] = 0
    flip_update(state, int(i), -1, n)
    flip_update(state, int(j), 1, n)


def score_binary(state, h, n):
    z = np.flatnonzero(h == 0)
    o = np.flatnonzero(h == 1)
    if z.size == 0 or o.size == 0:
        return None
    L = n - 1
    score = np.full((z.size, o.size), -1000000, dtype=np.int32)
    cP = state['cP']
    cQ = state['cQ']
    Gp = state['Gp']
    Gq = state['Gq']
    for idx in range(L):
        tmp = (cP[idx] + Gp[z, idx])[:, None] - Gp[o, idx][None, :]
        np.maximum(score, tmp, out=score)
        tmp = (cQ[idx] + Gq[z, idx])[:, None] - Gq[o, idx][None, :]
        np.maximum(score, tmp, out=score)
    return score, z, o


def binary_anneal(state, h, n, deadline, seed, cur):
    best_cur = cur
    best_h = h.copy()
    t_start = time.time()
    total = max(1.0, float(deadline - t_start))
    rng = np.random.RandomState(seed)
    T0 = 5.0
    T1 = 0.20
    for _ in range(1000000):
        if time.time() > deadline:
            break
        prog = min(1.0, (time.time() - t_start) / total)
        T = T0 * (T1 / T0) ** prog
        i = rng.randint(n)
        while h[i] != 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] != 1:
            j = rng.randint(n)
        p, q, newc = swap_effect(state, i, j, n)
        accept = newc < cur
        if not accept and newc > cur and T > 1e-9:
            gap = float(newc - cur)
            if gap < 12.0 * T:
                accept = rng.random() < np.exp(-gap / max(T, 1e-12))
        if accept:
            accept_swap(state, h, i, j, n, p, q)
            cur = newc
            if cur < best_cur:
                best_cur = cur
                best_h = h.copy()
    return best_h.astype(float), (float(best_cur) * (2.0 / n))


def binary_search(h, n, deadline, seed):
    if n <= 1 or n % 2 != 0:
        return project_feasible(h, n, 100), compute_c5(h, n)
    h = binarize(h, n)
    state, cur = build_binary_db(h, n)
    best_c = float(cur) * (2.0 / n)
    best_h = h.astype(float).copy()
    swap_deadline = deadline - 60.0
    while time.time() < swap_deadline:
        got = score_binary(state, h, n)
        if got is None:
            break
        score, z, o = got
        m1 = o.size
        fl = score.ravel()
        if fl[0] > cur - 1:
            break
        order = np.argsort(fl)
        improved = False
        for pos in order:
            s = int(fl[pos])
            if s > cur - 1:
                break
            ii = z[pos // m1]
            jj = o[pos % m1]
            p, q, newc = swap_effect(state, int(ii), int(jj), n)
            if newc < cur:
                accept_swap(state, h, int(ii), int(jj), n, p, q)
                cur = newc
                improved = True
                if cur < best_c * n / 2.0 - 1e-12:
                    best_c = float(cur) * (2.0 / n)
                    best_h = h.astype(float).copy()
                break
        if not improved:
            break
    if time.time() < deadline - 3.0:
        hb, cb = binary_anneal(state, h, n, deadline - 1.0, seed, cur)
        if cb < best_c:
            best_h, best_c = hb, cb
    best_h = project_feasible(best_h, n, 200)
    return best_h, compute_c5(best_h, n)


def run(seed=42, budget_s=1100, **kwargs):
    global _PARAM_N, _PARAM_X
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 250
        _PARAM_N = n0
        _PARAM_X = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = make_h_params(np.array([0.5, 1.0, 0.0, 0.12, 2.0, 0.0, 0.06, 3.0, 0.0]), n0)

    h0 = project_feasible(h0, n0, 300)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    if budget_s > 80 and time.time() < deadline - 250:
        pd = t0 + min(170.0, 0.16 * float(budget_s))
        h1, c1 = lbfgs_phase(h0, n0, pd, [8.0, 25.0, 80.0, 250.0])
        if c1 < best_c:
            best_c, best_h = c1, h1.copy()

    if n0 < 500 and time.time() < deadline - 260:
        n_up = 500
        h_up = interpolate_to(best_h, best_n, n_up)
        ud = t0 + min(430.0, 0.40 * float(budget_s))
        h2, c2 = lbfgs_phase(h_up, n_up, ud, [8.0, 25.0, 80.0])
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2.copy(), n_up

    if time.time() < deadline - 220:
        n_param = 500 if n0 < 500 else max(n0, 250)
        pd = t0 + min(640.0, 0.60 * float(budget_s))
        h3, c3 = param_phase(n_param, min(pd, deadline - 170), seed + 17)
        if c3 < best_c:
            best_c, best_h, best_n = c3, h3.copy(), n_param
        if time.time() < deadline - 140:
            hd, cd = lbfgs_phase(best_h, best_n, deadline - 140, [25.0, 80.0, 250.0])
            if cd < best_c:
                best_c, best_h = cd, hd.copy()

    if n0 % 2 == 0 and n0 <= 300 and time.time() < deadline - 90:
        if best_n == n0:
            binit = best_h
        else:
            binit = interpolate_to(best_h, best_n, n0)
        hb, cb = binary_search(binit, n0, deadline - 35.0, seed + 7)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), n0

    if time.time() < deadline - 12:
        hh, hh_c = lbfgs_phase(best_h, best_n, deadline - 3.0, [80.0, 250.0, 800.0])
        if hh_c < best_c:
            best_c, best_h = hh_c, hh.copy()

    best_h = project_feasible(best_h, best_n, 300)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
