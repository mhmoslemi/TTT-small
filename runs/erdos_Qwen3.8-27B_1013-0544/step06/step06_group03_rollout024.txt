    - After: s[p-m]=1, s[p]=0, so NOW counted. Δ = +1
  - Similarly for q as second index:
    - Before: s[q-m]=1, s[q]=0, so counted. 
    - After: s[q-m]=1, s[q]=1, so NOT counted. Δ = -1

So Δf_m = dx * [I(p valid, s[p+m]=0) * (-1) + I(q valid, s[q+m]=0) * (+1) + I(p-m valid, s[p-m]=1) * (+1) + I(q-m valid, s[q-m]=1) * (-1)]

Where "p valid" means 0 ≤ p ≤ n-m-1 (for m > 0).

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        obj = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
    else:
        obj = m + np.log(np.sum(np.exp(shifted))) / alpha
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        m_k = k - (n - 1)
        if m_k == 0:
            grad += w * dx * (1.0 - 2.0 * x)
        elif m_k > 0:
            grad[:n - m_k] += w * dx * (1.0 - x[m_k:])
            grad[m_k:] -= w * dx * x[:n - m_k]
        else:
            m2 = -m_k
            grad[m2:] += w * dx * (1.0 - x[:n - m2])
            grad[:n - m2] -= w * dx * x[m2:]
    return obj, grad

def slsqp_with_grad(h, n, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_grad, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_push(h, n, mu=0.3):
    h2 = (1.0 + mu) * h - 0.5 * mu
    return project_feasible(h2, n)

def interpolate_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)

def binary_local_search(s, n, deadline, rng, max_rounds=500):
    s = s.copy()
    dx = 2.0 / n
    c = np.correlate(s.astype(float), 1.0 - s.astype(float), mode='full') * dx
    best_c = float(c.max())
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(s == 1)[0]
        zeros = np.where(s == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_ones = min(len(ones), 40)
        n_zeros = min(len(zeros), 40)
        idx_ones = rng.choice(len(ones), size=min(n_ones, len(ones)), replace=False)
        idx_zeros = rng.choice(len(zeros), size=min(n_zeros, len(zeros)), replace=False)
        for ii in idx_ones:
            if improved and c.max() < best_c - 0.005:
                break
            i = ones[ii]
            for jj in idx_zeros:
                j = zeros[jj]
                if i == j:
                    continue
                s[i] = 0
                s[j] = 1
                c_new = np.correlate(s.astype(float), 1.0 - s.astype(float), mode='full') * dx
                val = float(c_new.max())
                if val < best_c - 1e-12:
                    best_c = val
                    c = c_new
                    improved = True
                    ones = np.where(s == 1)[0]
                    zeros = np.where(s == 0)[0]
                    if val < best_c - 0.001:
                        break
                else:
                    s[i] = 1
                    s[j] = 0
        if not improved:
            no_improve += 1
            if no_improve >= 15:
                break
        else:
            no_improve = 0
    return s, best_c

def try_binary_refinement(h, n, deadline, rng):
    s = (h > 0.5).astype(float)
    if s.sum() != n // 2:
        sorted_idx = np.argsort(-h)
        s = np.zeros(n)
        s[sorted_idx[:n//2]] = 1
    c_bin = compute_c5(s, n)
    if c_bin < float('inf'):
        s, c_bin = binary_local_search(s, n, deadline, rng)
        h_from_bin = s + 0.02 * rng.normal(size=n)
        h_from_bin = project_feasible(h_from_bin, n)
        for alpha in [20000, 50000, 100000]:
            if time.time() > deadline:
                break
            h_new = slsqp_with_grad(h_from_bin, n, alpha, maxiter=1500)
            h_new = binary_push(h_new, n, 0.2)
            c_new = compute_c5(h_new, n)
            if c_new < c_bin:
                c_bin = c_new
                h_from_bin = h_new
        return h_from_bin, c_bin
    return h, compute_c5(h, n)

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in [30000, 100000, 300000, 1000000]:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_with_grad(best_h, n, alpha, maxiter=2000)
        h_new = binary_push(h_new, n, 0.15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        if time.time() > deadline - 10:
            break
    return best_h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    if lag == 0:
        return h, best_c
    m = abs(lag)
    for d in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.8, (1.0 - h[j]) * 0.8)
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-6:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.8, (1.0 - h[i]) * 0.8)
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    if time.time() < deadline - 150:
        n = len(best_h)
        h_bin, c_bin = try_binary_refinement(best_h, n, min(deadline - 100, time.time() + 200), rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    n_candidates = [150, 200, 250, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        h_start = interpolate_to_n(best_h, len(best_h), n_try)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 35) * 0.5)
            if alloc < 20:
                break
            hp = binary_push(hp, n, 0.1)
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_new = slsqp_with_grad(best_h, n, 500000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = fine_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
