def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0

def generate_mseq_start(n, seed=42):
    """Generate a starting binary sequence inspired by m-sequences."""
    rng = np.random.RandomState(seed)
    # Start with alternating pattern, then add some structure
    h = np.zeros(n)
    h[::2] = 1.0  # positions 0, 2, 4, ... are 1
    # Adjust to get exactly n/2 ones
    n_ones = h.sum()
    target = n // 2
    if n_ones > target:
        # Remove some 1s from even positions
        even_pos = list(range(0, n, 2))
        rng.shuffle(even_pos)
        for p in even_pos[:int(n_ones - target)]:

        odd_pos = list(range(1, n, 2))
        rng.shuffle(odd_pos)
        for p in odd_pos[:int(target - n_ones)]:
            h[p] = 1.0
    return h

def sa_fast(h_bin, n, deadline, seed=42, t_start=0.05, t_end=0.00005, patience=30000):
    """Fast binary simulated annealing."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    zeros = [int(i) for i in np.where(h == 0)[0]]
    ones = [int(i) for i in np.where(h == 1)[0]]
    nz, no = len(zeros), len(ones)
    
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 500000.0)
    T = t_start
    
    while time.time() < deadline:
        rz = rng.randint(nz)
        ro = rng.randint(no)
        i = zeros[rz]
        j = ones[ro]
        
        # Compute delta for swap i(0->1), j(1->0)
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + j - i
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + i - j
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        nc = float(new_corr.max()) * dx
        dc = nc - cur_c
        
        if dc < 0 or (T > 1e-15 and rng.random() < np.exp(-dc / T)):
            zeros[rz] = j
            ones[ro] = i
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool_rate
        if T < t_end:
            T = t_end
        
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                rz2 = rng.randint(nz)
                ro2 = rng.randint(no)
                ii, jj = zeros[rz2], ones[ro2]
                zeros[rz2] = jj
                ones[ro2] = ii
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            cur_c = float(corr.max()) * dx
            no_improve = 0
            T = min(t_start * 0.6, T * 4.0)
    
    return best_h, best_c

def steepest_descent(h_bin, n, max_iters=200):
    """Steepest descent: try all single swaps, accept the best improving one."""
    dx = 2.0 / n
    h = h_bin.copy()
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64).copy()
    cur_c = float(corr.max()) * dx
    
    for _ in range(max_iters):
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_swap = None
        best_nc = cur_c
        
        for i in zeros:
            for j in ones:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + j - i
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + i - j
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                nc = float((corr + delta).max()) * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_swap = (int(i), int(j))
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 1.0
        h[j] = 0.0
        # Recompute correlation (safer than incremental for correctness)
        corr = np.correlate(h, 1 - h, mode='full').astype(np.float64).copy()
        cur_c = float(corr.max()) * dx
    
    return h, cur_c

def smooth_slsqp(h, n, alpha, maxiter=2000):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + np.log(np.exp(sh - sm).sum() + 1e-300) / alpha
    bounds = [(0.0, 1.0)] * n
