For coordinate descent, the key insight is:
- Fix all h[j] for j ≠ i
- c_k = c_k_base + b_k * h[i] (linear in h[i])
- We want to minimize smooth_max(c_k_base + b_k * t) over t ∈ [0,1]

The 1D function f(t) = smooth_max(c_base + b*t) is convex (smooth max of affine functions is convex).

To minimize a convex 1D function, I can:
1. Compute the derivative f'(t) = sum(w_k * b_k) where w_k are the soft weights
2. Use bisection on the derivative (since f' is monotonically increasing)

This is very efficient! Each evaluation of f' is O(n), and bisection converges in O(log(1/eps)) iterations.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h_slice_k(x, k, nv))
            grad[k:] += wm * (-h_slice_0(x, nv))
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h_slice_0(x, nv))
            grad[:nv] += wm * (-h_slice_k(x, kk, nv))
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def h_slice_k(x, k, nv):
    return x[k:k+nv]

def h_slice_0(x, nv):
    return x[0:nv]

def lbfgs_phase(h, n, deadline, alpha=500000, mu=1e-3, maxit=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, mu),
                      jac=smooth_max_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-14,
                               'maxfun': maxit * 12, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def coord_descent(h, n, deadline, alpha=1000):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full') * dx
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            b = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                bk = 0.0
                if 0 <= i + k < n:
                    bk += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    bk -= h[i - k]
                b[k + n - 1] = bk * dx
            old_t = h[i]
            c_base = c_full - b * old_t
            lo = 0.0
            hi = 1.0
            for _ in range(35):
                mid = (lo + hi) * 0.5
                c_t = c_base + b * mid
                m = c_t.max()
                sh = alpha * (c_t - m)
                mx = sh.max()
                wp = np.exp(sh - mx)
                wp /= wp.sum()
                g = float(np.dot(wp, b))
                if g > 0:
                    hi = mid
                else:
                    lo = mid
                if hi - lo < 1e-13:
                    break
            new_t = (lo + hi) * 0.5
            if abs(new_t - old_t) > 1e-11:
                delta = new_t - old_t
                c_full += b * delta
                h[i] = new_t
                c_max = c_full.max()
                if c_max < best_c - 1e-14:
                    best_c = c_max
                    improved = True
                else:
                    c_full -= b * delta
                    h[i] = old_t
        if not improved:
            break
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c
