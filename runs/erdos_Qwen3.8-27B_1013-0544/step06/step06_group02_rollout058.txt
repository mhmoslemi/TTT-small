import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_at_lag(h, n, m):
    dx = 2.0 / n
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])
        g[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv])
        g[:nv] += (-h[kk:])
    return g * dx

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += w[m] * (1.0 - h[k:])
            grad[k:] += w[m] * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w[m] * (1.0 - h[:nv])
            grad[:nv] += w[m] * (-h[kk:])
    return grad * dx

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def golden_search(h, n, i, j, lo, hi, max_iter=15):
    def f(a):
        h[i] += a
        h[j] -= a
        val = compute_c5(h, n)
        h[i] -= a
        h[j] += a
        return val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    best_a = c if fc < fd else d
    best_f = min(fc, fd)
    for _ in range(max_iter):
        if hi - lo < 1e-11:
            break
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = f(d)
        if fc < best_f:
            best_f, best_a = fc, c
        if fd < best_f:
            best_f, best_a = fd, d
    return best_a, best_f

def lbfgs_phase(h, n, deadline):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha, mu in [(500, 10.0), (5000, 50.0), (50000, 200.0), (500000, 1000.0), (5000000, 5000.0)]:
        if time.time() > deadline:

        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

        except Exception:

    return h, best_c

def subgrad_descent(h, n, deadline, top_k=3):
    best_h, best_c = h.copy(), compute_c5(h, n)
    lr = 0.005
    for it in range(50000):

        c = compute_c5_vec(h, n)
        thr = c.max() - 1e-8
        active = np.where(c >= thr)[0]

            active = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for m in active:
            g += grad_at_lag(h, n, m)
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        g /= gn
        step = lr
        improved = False

            c_new = compute_c5(h_new, n)

                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            if lr < 1e-13:
                break
        elif lr < 0.015:
            lr *= 1.1
    return best_h, best_c

                    a_b, c_b = golden_search(h, n, i, j, alo, ahi)
                    if c_b < best_c - 1e-14:
                        h[i] += a_b; h[j] -= a_b

                elif dd.min() > 1e-10:

                        best_c = c_b; improved = True
