    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_objective(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter=500, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(softmax_objective, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.03, patience=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    T = t_start
    
    while time.time() < deadline and T > 1e-10:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < -1e-15 or rng.random() < np.exp(-max(0, delta_c) / T):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= 0.99995
        if no_improve > patience:
            nk = 3 + rng.randint(10)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(0.02, T * 3.0)
    return best_h, best_c

def coordinate_polish(h, n, deadline, seed=42):
    """Try small perturbations at each coordinate."""
    rng = np.random.RandomState(seed)
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    steps = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    
    while time.time() < deadline:
        improved = False
        for i in rng.choice(n, size=n, replace=False):
            if time.time() > deadline:
                break
            orig = h[i]
            for s in steps:
                h_test = h.copy()
                h_test[i] = orig + s
                h_test = project_feasible(h_test, n)
                c = compute_c5(h_test, n)
                if c < best_c - 1e-14:
                    h = h_test
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c
