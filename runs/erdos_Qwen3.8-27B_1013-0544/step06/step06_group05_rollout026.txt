    """Compute change in correlation when flipping h[i]: 0->1 and h[j]: 1->0"""

    #   h[i] appears as first factor when i is in range: contributes (1-h[i+L]) if i+L valid
    #   h[i] appears as second factor when i-L is in range: contributes -h[i-L] if i-L valid (because 1-h goes from 1 to 0)

def compute_all_j_effects(h, n):
    """For each position j and each lag, compute the effect of h[j]: 1->0"""
    L = np.arange(-(n-1), n)  # shape (2n-1,)
    j = np.arange(n)  # shape (n,)

    jl = j[:, None] + L[None, :]
    valid_jl = (jl >= 0) & (jl < n)
    
    # j-L: shape (n, 2n-1)  
    jml = j[:, None] - L[None, :]
    valid_jml = (jml >= 0) & (jml < n)
    
    h_jl = np.zeros((n, 2*n-1))
    h_jl[valid_jl] = h[jl[valid_jl]]
    
    h_jml = np.zeros((n, 2*n-1))
    h_jml[valid_jml] = h[jml[valid_jml]]

def fast_binary_search(h, n, deadline, max_sweeps=50000, seed=42):
    dx = 2.0 / n
    cur = h.copy()
    corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_corr = cur_max
    best_h = cur.copy()
    
    L = np.arange(-(n-1), n, dtype=np.int64)
    nl = 2 * n - 1
    j_idx = np.arange(n, dtype=np.int64)
    jl = j_idx[:, None] + L[None, :]
    jml = j_idx[:, None] - L[None, :]
    vjl = (jl >= 0) & (jl < n)
    vjml = (jml >= 0) & (jml < n)
    
    def get_j_eff():
        hjl = np.zeros((n, nl))
        hjl[vjl] = cur[jl[vjl]]
        hjml = np.zeros((n, nl))
        hjml[vjml] = cur[jml[vjml]]
        return -(1.0 - hjl) * vjl + hjml * vjml
    
    j_eff = get_j_eff()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        order = rng.permutation(len(zeros))
        for oi in order:
            if time.time() > deadline:
                break
            i = zeros[oi]
            if cur[i] != 0:
                continue
            # Compute i_effect
            il = i + L
            vil = (il >= 0) & (il < n)
            iml = i - L
            viml = (iml >= 0) & (iml < n)
            hil = np.zeros(nl)
            hil[vil] = cur[il[vil]]
            himl = np.zeros(nl)
            himl[viml] = cur[iml[viml]]
            i_eff = (1.0 - hil) * vil - himl * viml
            
            # Approximate new corr for all j
            approx_nc = corr[None, :] + i_eff[None, :] + j_eff[ones, :]
            approx_nm = approx_nc.max(axis=1)
            
            # Find candidates that look good
            cands = np.where(approx_nm < cur_max - 0.5)[0]
            if len(cands) == 0:
                continue
            
            # Try top candidates (up to 5)
            cand_perms = cands[np.argsort(approx_nm[cands])][:5]
            found = False
            for ci in cand_perms:
                if time.time() > deadline:
                    break
                j = ones[ci]
                cur[i] = 1.0
                cur[j] = 0.0
                new_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
                new_max = float(new_corr.max())
                if new_max < cur_max:
                    corr = new_corr
                    cur_max = new_max
                    j_eff = get_j_eff()
                    improved = True
                    if cur_max < best_corr:
                        best_corr = cur_max
                        best_h = cur.copy()
                    found = True
                    break
                else:
                    cur[i] = 0.0
                    cur[j] = 1.0
            if found:
                break  # Restart sweep
        if not improved:
            break
    return best_h, float(best_corr * dx)

```python
def slsqp_obj_grad(h, n, alpha):
    """Return (objective, gradient) for smooth max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Scale to avoid overflow in exp
    mc = corr.max()
    shifted = alpha * (corr - mc)
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    obj = (mc + (log_sum - max_shifted) / alpha)  # = mc + (1/alpha)*log_sum_exp(alpha*(corr-mc))
    
    # Weights
    w = np.exp(shifted - max_shifted) / np.exp(shifted - max_shifted).sum()
    
    # Gradient
    L = np.arange(-(n-1), n)
    p = np.arange(n)
    pl = p[:, None] + L[None, :]
    pml = p[:, None] - L[None, :]
    vpl = (pl >= 0) & (pl < n)
    vpml = (pml >= 0) & (pml < n)
    
    # A[p] = sum_m w[m] * (1-h[p+L_m]) * vpl[p,m]
    # B[p] = sum_m w[m] * h[p-L_m] * vpml[p,m]
    
    h_pl = np.zeros((n, 2*n-1))
    h_pl[vpl] = h[pl[vpl]]
    h_pml = np.zeros((n, 2*n-1))
    h_pml[vpml] = h[pml[vpml]]
    
    grad = (w[None, :] * (1 - h_pl) * vpl).sum(axis=1) - (w[None, :] * h_pml * vpml).sum(axis=1)
    
    return obj, grad
```
