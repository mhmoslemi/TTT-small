        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Compute delta vectors (for all positions)
        # add_delta[i, m] = (1-h[i-m+n-1]) - h[i+m-n+1] for i in zeros
        # rem_delta[j, m] = -(1-h[j-m+n-1]) + h[j+m-n+1] for j in ones
        
        # Only need these at top_idx lags
        i_arr = zeros[:, None]  # (nz, 1)
        m_arr = top_idx[None, :]  # (1, top_k)
        n_1 = n - 1
        
        p1 = i_arr - m_arr + n_1
        p2 = i_arr + m_arr - n_1
        v1 = (p1 >= 0) & (p1 < n)
        v2 = (p2 >= 0) & (p2 < n)
        hp1 = np.where(v1, h[np.clip(p1, 0, n-1)], 0.0)
        hp2 = np.where(v2, h[np.clip(p2, 0, n-1)], 0.0)
        add_d = (1.0 - hp1) - hp2  # (nz, top_k)
        
        j_arr = ones[:, None]
        p1j = j_arr - m_arr + n_1
        p2j = j_arr + m_arr - n_1
        v1j = (p1j >= 0) & (p1j < n)
        v2j = (p2j >= 0) & (p2j < n)
        hp1j = np.where(v1j, h[np.clip(p1j, 0, n-1)], 0.0)
        hp2j = np.where(v2j, h[np.clip(p2j, 0, n-1)], 0.0)
        rem_d = -(1.0 - hp1j) + hp2j  # (no, top_k)
        
        # For each (i_idx, j_idx): new values at top_k lags = corr[top_idx] + add_d[i_idx] + rem_d[j_idx]
        # Lower bound on new max = max over top_k lags
        base = corr[top_idx]  # (top_k,)
        # new_vals[i_idx, j_idx, m_idx] = base[m_idx] + add_d[i_idx, m_idx] + rem_d[j_idx, m_idx]
        lb = (base[None, None, :] + add_d[:, None, :] + rem_d[None, :, :]).max(axis=2)  # (nz, no)
        
        flat = lb.argmin()
        ii, jj = np.unravel_index(flat, lb.shape)
        
        # Verify with full correlation
        i = zeros[ii]
        j = ones[jj]
        h_test = h.copy()
        h_test[i] = 1.0
        h_test[j] = 0.0
        full_corr = np.correlate(h_test, 1 - h_test, mode='full')
        full_c = full_corr.max() * dx
        
        if full_c < best_c - 1e-12:
            h = h_test
            best_h = h.copy()
            best_c = full_c
            # Update

            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            stall = 0
