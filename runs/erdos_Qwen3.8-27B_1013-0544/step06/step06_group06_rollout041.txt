    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0, 1)

def thue_morse_seq(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    ones_count = int(t.sum())
    if ones_count > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        t[idx[:ones_count - target]] = 0
    elif ones_count < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        t[idx[:target - ones_count]] = 1
    return t.astype(float)

def sa_search(h, n, deadline, seed=42, temp0=0.3, cool=0.998):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_val = float(corr.max())
    best_val = cur_val
    best_h = h.copy()
    temp = temp0
    kick = 0
    
    while time.time() < deadline:
        # Random swap
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        
        h[i] = 1.0
        h[j] = 0.0
        c = np.correlate(h, 1 - h, mode='full')
        new_val = float(c.max())
        
        if new_val <= cur_val or rng.rand() < np.exp(-(new_val - cur_val) / max(temp, 0.005)):
            # Accept
            cur_val = new_val
            corr = c
            # Update tracking
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
        
        temp = max(0.005, temp * cool)
        
        # Periodic reheat
        if int(best_val) == int(cur_val) and cur_val - best_val > 2:
            if rng.rand() < 0.1:
                temp = 0.5
                kick += 1
    
    return best_h, best_val * dx

def slsqp_refine(h, n, alpha=500000, maxiter=300, deadline=None):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        s = (c - m) * alpha
        return m + np.log(np.exp(s - s.max()).sum() + 1e-300) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except:
        pass
    return h
