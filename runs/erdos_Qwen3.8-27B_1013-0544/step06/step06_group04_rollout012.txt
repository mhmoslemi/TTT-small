import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_obj(x, n, alpha, mu):
    corr = compute_overlap(x, n)
    m = corr.max()
    shifted = alpha * (corr - m)
    val = m + np.log(np.exp(shifted).sum()) / alpha
    penalty = mu * (x.sum() - n / 2.0) ** 2
    return val + penalty

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    a = alpha * corr
    a_max = a.max()
    p = np.exp(a - a_max)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - n + 1
        if lag >= 0:
            end = n - lag
            if end > 0:
                grad[:end] += pk * dx * (1.0 - x[lag:lag + end])
            if n - 1 >= lag:
                grad[lag:] -= pk * dx * x[:n - lag]
        else:
            start = -lag
            if n - start > 0:
                grad[start:] += pk * dx * (1.0 - x[:n - start])
            end = n + lag
            if end > 0:
                grad[:end] -= pk * dx * x[-lag:]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

    idx = np.argsort(h)[::-1]
    b = np.zeros(n)

    return b

                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
