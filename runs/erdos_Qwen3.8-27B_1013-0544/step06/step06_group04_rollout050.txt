def local_search_fast(h, n, deadline, seed=42, max_sweeps=10000):

        excess = n_ones - target
        choose = np.random.RandomState(seed).permutation(ones)[:excess]
        h[choose] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        deficit = target - n_ones
        choose = np.random.RandomState(seed).permutation(zeros)[:deficit]
        h[choose] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    no_improve_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in zeros:
            i = int(i)
            for j in ones:
                j = int(j)
                d = swap_delta_vec(h, i, j, n)
                new_corr = corr + d
                nc = float(new_corr.max()) * dx

                    cur_c = nc
                    if cur_c < best_c:

                        best_h = h.copy()
                    improved = True
                    no_improve_count = 0
                    break
            if improved:
                break
        if not improved:
            no_improve_count += 1
            if no_improve_count > 5:

    return best_h, best_c

def projected_gradient(h, n, deadline, lr=None, seed=42):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)

            break

            mask = h < 1.0

            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()

    best_c = compute_c5(h, n)[0]

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(corr.max()) * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Find active set (lags near max)
        threshold = cur_c / dx - 0.5  # within 0.5*dx of max
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute subgradient: average gradient over active lags
        # Gradient of c[mlag] w.r.t. h[k]:
        # c[mlag] = sum_l h[l+mlag] * (1-h[l])
        # dc[mlag]/dh[k] = (1-h[k-mlag]) - h[k+mlag] (with validity checks)
        
        # For the max, the subgradient is the average of gradients of active functions
        grad = np.zeros(n)
        for mlag_idx in active:
            mlag = mlag_idx - (n-1)
            # dc/dh[k] = (1-h[k-mlag])*[valid] - h[k+mlag]*[valid]
            k_minus_mlag = np.arange(n) - mlag
            k_plus_mlag = np.arange(n) + mlag
            valid_1 = (k_minus_mlag >= 0) & (k_minus_mlag < n)
            valid_2 = (k_plus_mlag >= 0) & (k_plus_mlag < n)
            grad += (1.0 - np.where(valid_1, h[k_minus_mlag], 0.0)) * valid_1
            grad -= np.where(valid_2, h[k_plus_mlag], 0.0) * valid_2
        grad /= len(active)
        
        # Take step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        # Project to sum constraint
        for _ in range(50):
            s = h_new.sum()
            diff = target - s
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                mask = h_new < 1.0
                if mask.sum() == 0:
                    h_new = h  # revert
                    break
                h_new[mask] += diff / mask.sum()
            else:
                mask = h_new > 0.0
                if mask.sum() == 0:
                    h_new = h  # revert
                    break
                h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        # Adaptive learning rate
        new_c = compute_c5(h, n)[0]
        if new_c > cur_c + 0.01 * dx:
            lr *= 0.5
            h = best_h.copy()
        elif it % 100 == 0:
            lr = min(lr * 1.05, 0.1)
    
    return best_h, best_c
