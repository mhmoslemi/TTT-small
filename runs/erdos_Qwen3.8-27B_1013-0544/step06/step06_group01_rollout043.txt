```
def precompute_coords(n):
   info=[]
   L=2*n-1
   for p in range(n):
      lo=min(p,n-1-p); hi=L-1-lo
      ks=np.arange(lo,hi+1)
      fwd=(ks >= n-1-p) & (ks <= p+n-1)
      bwd=(ks >= p) & (ks <= p+n-1)
      fwd_pos=np.nonzero(fwd)[0]
      bwd_pos=np.nonzero(bwd)[0]

      bwd_idx=p+n-1-ks[bwd]
      idx0=n-1-lo
      info.append((ks,fwd_pos,bwd_pos,fwd_idx,bwd_idx,idx0,lo,hi))
   return info
```

```
def smooth_eval(x, n, beta, mu, dx, target):
    h = x  # assume in bounds
    a = 1.0 - h
    c = np.correlate(h, a, 'full')[::-1]
    m = float(c.max())
    # To avoid overflow, use beta*(c-m) <=0
    z = beta*(c-m)
    e = np.exp(z)
    se = e.sum()
    if se <=0: ...
    p = e/se
    f_soft = m + np.log(se)/beta
    f = f_soft*dx + mu*(h.sum()-target)**2
    grad = np.zeros(n)
    # active indices to reduce loops: active = np.where(p > 1e-16)[0]
    active = np.where(p > 1e-16)[0]
    for r in active:
       l = r - (n-1)
       if l == 0:
          grad += p[r]*(1-2*h)
       elif l > 0:
          d = l
          # c_l = sum_{i=0}^{n-d-1} h_i(1-h_{i+d})
          grad[:-d] += p[r]*(1-h[d:])
          grad[d:] -= p[r]*h[:-d]
       else:
          d = -l
          # c_{-d}=sum_{i=d}^{n-1} h_i(1-h_{i-d})
          grad[d:] += p[r]*(1-h[:-d])
          grad[:-d] -= p[r]*h[d:]
    grad *= dx
    grad += 2.0*mu*(h.sum()-target)
    return f, grad
```

def smooth_eval(x, n, beta, mu, dx, target):
    h = np.asarray(x, dtype=float)
    a = 1.0 - h
    c = np.correlate(h, a, 'full')[::-1]
    m = float(np.max(c))
    z = beta * (c - m)
    # To avoid log(0) if underflow
    exp = np.exp(z)
    s = float(np.sum(exp))
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        soft = m
    else:
        p = exp / s
        soft = m + np.log(s) / beta
    f = soft * dx + mu * (float(h.sum()) - target) ** 2
    grad = np.zeros(n)
    active = np.where(p > 1e-17)[0]
    for r in active:
        w = float(p[r])
        l = int(r) - (n - 1)
        if l == 0:
            grad += w * (1.0 - 2.0 * h)
        elif l > 0:
            d = l
            grad[:-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:-d]
        else:
            d = -l
            grad[d:] += w * (1.0 - h[:-d])
            grad[:-d] -= w * h[d:]
    grad *= dx
    grad += 2.0 * mu * (float(h.sum()) - target)
    return f, grad

def project_feasible(h,n):
    h=np.clip(np.asarray(h,float).copy(),0.0,1.0)
    tgt=n/2.0
    for _ in range(300):
        diff=tgt-h.sum()
        if abs(diff)<1e-14: break
        if diff>0:
            room=1.0-h; tot=room.sum()
            if tot<=0: break
            h += diff*room/tot
        else:
            tot=h.sum()
            if tot<=0: break
            h += diff*h/tot
    h=np.clip(h,0.0,1.0)
    diff=tgt-h.sum()
    if abs(diff)>1e-14:
        if diff>0: idx=int(np.argmin(h))
        else: idx=int(np.argmax(h))
        h[idx]+=diff
        h[idx]=min(max(h[idx],0.0),1.0)
    return h

def pair_cd(h,n,deadline,grid=16,max_passes=10000,rng=None,first=None,second=None,block=15):
    h=project_feasible(h,n).copy()
    if first is None:
        first,second=precompute_pair_matrices(n)
    if rng is None: rng=np.random.RandomState(12345)
    L=2*n-1
    c=compute_c_asc(h,n)
    best_coeff=float(c.max()); best_h=h.copy(); cur_coeff=best_coeff
    buf=np.empty((grid+2,L),float)
    no_stall=0; force_r=True; pass_i=0; order=np.arange(n)
    tol=1e-12
    while pass_i<max_passes and time.time()<deadline:
        if force_r or pass_i%block==0:
            jitter=np.abs(h-0.5)+1e-9*rng.random(n)
            r=int(np.argmin(jitter)); force_r=False
        oc=order.copy(); rng.shuffle(oc)
        improved=False
        for p in oc:

def smooth_eval(x,n,beta,mu,dx,target):
    h=np.asarray(x,float)
    c=compute_c_asc(h,n)
    m=float(c.max())
    e=np.exp(beta*(c-m))
    s=float(e.sum())

       p=np.zeros_like(c); p[int(np.argmax(c))]=1.0; soft=m
    else:
       p=e/s; soft=m+np.log(s)/beta

    grad *= dx
    grad += 2.0*mu*(float(h.sum())-target)
    return f,grad

def perturb_h(h,n,rng,scale,mode=0):
   hp=h.copy()
   if mode==0:
      hp += rng.normal(0.0, scale, n)
   else:
      x=np.linspace(0,1,n); jit=np.zeros(n)
      for k in range(1,8):
         amp=scale/np.sqrt(float(k))
         if rng.rand()<0.5: jit += amp*np.cos(2*np.pi*k*x)
         else: jit += amp*np.sin(2*np.pi*k*x)
      hp += jit
   return project_feasible(hp,n)

```python
import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        diff = tgt - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 0:
                break
            h += diff * room / tot
        else:
            tot = h.sum()
            if tot <= 0:
                break
            h += diff * h / tot
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
    return h


def compute_c_asc(h, n):
    a = 1.0 - h
    return np.convolve(h, a[::-1], mode='full')[::-1]


def exact_c5(h, n):
    return float(np.max(compute_c_asc(h, n)) * (2.0 / n))


def precompute_pair_matrices(n):
    idx = np.arange(n)
    first = (idx[None, :] - idx[:, None]) + (n - 1)
    second = (idx[:, None] - idx[None, :]) + (n - 1)
    return first.astype(np.intp, copy=False), second.astype(np.intp, copy=False)


def pair_cd(h, n, deadline, grid=16, max_passes=10000, rng=None,
            first=None, second=None, block=15):
    h = project_feasible(h, n).copy()
    if first is None or second is None:
        first, second = precompute_pair_matrices(n)
    if rng is None:
        rng = np.random.RandomState(12345)

    L = 2 * n - 1
    c = compute_c_asc(h, n)
    best_coeff = float(c.max())
    best_h = h.copy()
    cur_coeff = best_coeff

    buf = np.empty((grid + 2, L), dtype=float)
    no_stall = 0
    force_r = True
    pass_i = 0
    order = np.arange(n)
    tol = 1e-12

    while pass_i < max_passes and time.time() < deadline:
        if force_r or pass_i % block == 0:
            jitter = np.abs(h - 0.5) + 1e-9 * rng.random(n)
            r = int(np.argmin(jitter))
            force_r = False

        oc = order.copy()
        rng.shuffle(oc)
        improved = False

        for p in oc:
            if time.time() > deadline:
                break
            if p == r:
                continue

            hp = float(h[p])
            hr = float(h[r])
            low = max(-hp, hr - 1.0)
            high = min(1.0 - hp, hr)
            if high - low < 1e-13:
                continue

            deltas = np.linspace(low, high, grid)
            if low < 0.0 < high:
                deltas = np.append(deltas, 0.0)
            G = len(deltas)

            diff = np.zeros(L, dtype=float)
            fi = first[p]
            si = second[p]
            diff[fi] = 1.0 - h
            diff[si] -= h
            rfi = first[r]
            rsi = second[r]
            diff[rfi] -= 1.0 - h
            diff[rsi] += h

            np.multiply(deltas[:, None], diff[None, :], out=buf[:G, :])
            buf[:G, :] += c
            d2 = deltas * deltas
            buf[:G, n - 1] += -2.0 * d2
            dd = abs(r - p)
            if dd > 0:
                buf[:G, n - 1 + dd] += d2
                buf[:G, n - 1 - dd] += d2

            maxes = buf[:G, :].max(axis=1)
            bi = int(np.argmin(maxes))
            new_coeff = float(maxes[bi])

            if new_coeff < cur_coeff - tol:
                dval = float(deltas[bi])
                c += diff * dval
                c[n - 1] += -2.0 * dval * dval
                if dd > 0:
                    c[n - 1 + dd] += dval * dval
                    c[n - 1 - dd] += dval * dval
                h[p] = hp + dval
                h[r] = hr - dval
                cur_coeff = new_coeff
                improved = True

        pass_i += 1
        c = compute_c_asc(h, n)
        b2 = float(c.max())
        if b2 < best_coeff - 1e-14:
            best_coeff = b2
            best_h = h.copy()
        if b2 > best_coeff + 1e-9:
            h = best_h.copy()
            c = compute_c_asc(h, n)
            b2 = best_coeff
        cur_coeff = b2

        if not improved:
            no_stall += 1
            if no_stall >= 6:
                break
            force_r = True
        else:
            no_stall = 0

    h = project_feasible(best_h, n)
    return h, exact_c5(h, n)


def smooth_eval(x, n, beta, mu, dx, target):
    h = np.asarray(x, dtype=float)
    a = 1.0 - h
    c = np.convolve(h, a[::-1], mode='full')[::-1]

    m = float(np.max(c))
    e = np.exp(beta * (c - m))
    s = float(np.sum(e))

    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        soft = m
    else:
        p = e / s
        soft = m + np.log(s) / beta

    S = np.convolve(p[::-1], a, mode='full')[n - 1:2 * n - 1]
    U = np.convolve(p, h, mode='full')[n - 1:2 * n - 1]
    grad = S - U

    ss = float(h.sum())
    grad *= dx
    grad += 2.0 * mu * (ss - target)
    f = soft * dx + mu * (ss - target) ** 2
    return f, grad


def lbfgs_fun_tuple(x, n, beta, mu, dx, target):
    return smooth_eval(x, n, beta, mu, dx, target)


def lbfgs_polish(h, n, deadline, betas, mus, maxiters=100):
    best_h = project_feasible(h, n)
    best_c = exact_c5(best_h, n)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    if not isinstance(maxiters, (list, tuple)):
        maxiters = [maxiters] * len(betas)
    if not isinstance(mus, (list, tuple)):
        mus = [mus] * len(betas)

    for i, beta in enumerate(betas):
        if time.time() > deadline - 3:
            break
        mu = float(mus[min(i, len(mus) - 1)])
        mit = int(maxiters[min(i, len(maxiters) - 1)])
        rem = deadline - time.time()
        if rem < 10:
            break
        mit = min(mit, max(10, int(rem * 3)))

        cur = best_h.copy()
        try:
            res = minimize(
                lbfgs_fun_tuple,
                cur,
                args=(n, beta, mu, dx, target),
                method='L-BFGS-B',
                bounds=bounds,
                jac=True,
                options={'maxiter': mit, 'ftol': 1e-10, 'maxls': 20}
            )
            if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                h2 = project_feasible(np.asarray(res.x, dtype=float), n)
                c2 = exact_c5(h2, n)
                if c2 < best_c - 1e-13:
                    best_h = h2
                    best_c = c2
        except Exception:
            pass

    return best_h, best_c


def resample_h(h, n):
    n0 = len(h)
    if n0 == n:
        return project_feasible(h, n)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n)
    return project_feasible(np.interp(x1, x0, h), n)


def perturb_h(h, n, rng, scale, mode=0):
    hp = h.copy()
    if mode == 0:
        hp += rng.normal(0.0, scale, n)
    else:
        x = np.linspace(0.0, 1.0, n)
        jit = np.zeros(n)
        for k in range(1, 8):
            amp = scale / np.sqrt(float(k))
            if rng.rand() < 0.5:
                jit += amp * np.cos(2.0 * np.pi * k * x)
            else:
                jit += amp * np.sin(2.0 * np.pi * k * x)
        hp += jit
    return project_feasible(hp, n)


def quick_refine(h, n, deadline, rng, first, second):

    c = exact_c5(h, n)

    if time.time() < deadline - 12:

            h, n, min(deadline - 5, time.time() + 40),

    return h, c

    scale = 0.03
    count = 0
    while time.time() < restart_deadline and count < 40:
        now = time.time()
        alloc = min(25, restart_deadline - now)
        if alloc < 18:
            break
        mode = 1 if count % 4 == 3 else 0
        hn = perturb_h(best_h, n_best, rng, scale, mode)
        d = min(restart_deadline, now + alloc)
        hq, cq = quick_refine(hn, n_best, d, rng, first, second)
        if cq < best_c - 1e-13:
            best_h, best_c = hq, cq
            scale = max(0.005, scale * 0.5)
        else:
            scale = min(0.15, scale * 1.4)
        count += 1

    now = time.time()
    if now < deadline - 25:
        d = min(deadline - 15, now + 45)
        hf, cf = lbfgs_polish(
            best_h, n_best, d,
            [20000, 100000],
            [1000, 5000],
            [150, 200]
        )
        if cf < best_c:
            best_h, best_c = hf, cf

    now = time.time()
    if now < deadline - 5:
        d = min(deadline - 2, now + 60)
        hf, cf = pair_cd(
            best_h, n_best, d, grid=48, max_passes=20000,
            rng=rng, first=first, second=second, block=10
        )
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n_best)
    best_c = exact_c5(best_h, n_best)
    return best_h, best_c, int(n_best)
```
