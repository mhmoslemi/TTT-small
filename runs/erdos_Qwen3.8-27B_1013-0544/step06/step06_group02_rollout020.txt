def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            if tot < 1e-15:

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_at(h, n, k_idx):
    """Gradient of correlation at lag k_idx w.r.t. h, times dx."""
    s = k_idx - (n - 1)
    dx = 2.0 / n
    a = np.zeros(n)
    if s >= 0:
        a[:n-s] += (1.0 - h[s:])
        a[s:] += (-h[:n-s])
    else:
        t = -s
        a[t:] += (1.0 - h[:n-t])
        a[:n-t] += (-h[t:])
    return a * dx

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10:
            continue
        grad += w[m_idx] * grad_c_at(h, n, m_idx)
    return grad

    mx = sh.max()

def lbfgs_run(h, n, deadline, obj_fn, grad_fn, max_iter=500, alpha=1000):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-16,
                               'gtol': 1e-12, 'maxfun': max_iter * 20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    return best_h, best_c

def grad_pair_search(h, n, deadline, top_m=200, step_init=0.01):
    """Pair search guided by gradient: try top-m pairs with largest |g[i]-g[j]|."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = step_init
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        stagnant = 0
        while time.time() < deadline:
            g = smooth_max_grad(h, n, alpha)
            # Compute all |g[i]-g[j]| and pick top-m
            # For efficiency, just use the gradient to define a "direction"
            # Pairs where g[i] > g[j] suggest moving from i to j
            # Try top-m pairs based on |g[i]-g[j]|
            # Instead of O(n^2), use a heuristic:
            # Sort by |g[i]| and try pairs of high and low gradient
            order_high = np.argsort(g)[-top_m:]
            order_low = np.argsort(g)[:top_m]
            improved = False
            for i in order_high:
                if time.time() > deadline:
                    return best_h, best_c
                for j in order_low:
                    if i == j:
                        continue
                    for direction in [1, -1]:
                        d = direction * step
                        if h[i] + d >= 0 and h[i] + d <= 1 and h[j] - d >= 0 and h[j] - d <= 1:
                            h[i] += d
                            h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
                stagnant += 1
                if stagnant > 3:
                    break
            else:
                stagnant = 0
                # Occasionally try larger steps
                if np.random.random() < 0.1:
                    step *= 1.2
        # Re-identify active lags for next alpha
    return best_h, best_c

    if deltas is None:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_one_k(h, n, k_idx):
    s = k_idx - (n - 1)
    a = np.zeros(n)
    if s >= 0:
        a[:n-s] += (1.0 - h[s:])
        a[s:] += (-h[:n-s])
    else:
        a[-s:] += (1.0 - h[:n+s])
        a[:n+s] += (-h[-s:])
    return a

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10: continue
        grad += w[m_idx] * grad_one_k(x, n, m_idx) * dx
    return grad

        if 0 <= l + s < n: gl += 1.0 - h[l + s]
        if 0 <= l - s < n: gl -= h[l - s]

def grad_pair_search(h, n, deadline, step_init=0.005, top_m=100):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_one_k(h, n, k_idx):
    s = k_idx - (n - 1)
    a = np.zeros(n)
    if s >= 0:
        a[:n-s] += (1.0 - h[s:])
        a[s:] += (-h[:n-s])
    else:
        a[-s:] += (1.0 - h[:n+s])
        a[:n+s] += (-h[-s:])
    return a

def _obj_smooth(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def _grad_smooth(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10:
            continue
        grad += w[m_idx] * grad_one_k(x, n, m_idx) * dx
    return grad

def _obj_topk(x, n, k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-k:]
    top = c[idx]
    alpha = 2000
    m = top.max()
    sh = alpha * (top - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def _grad_topk(x, n, k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-k:]
    top = c[idx]
    alpha = 2000
    m = top.max()
    sh = alpha * (top - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(k):
        grad += w[ii] * grad_one_k(x, n, int(idx[ii])) * dx
    return grad

def coord_dir(h, n, j, l):
    dx = 2.0 / n
    nd = 2 * n - 1
    d = np.zeros(nd)
    for k in range(nd):
        s = k - (n - 1)
        gj = 0.0
        if 0 <= j + s < n:
            gj += 1.0 - h[j + s]
        if 0 <= j - s < n:
            gj -= h[j - s]
        gl = 0.0
        if 0 <= l + s < n:
            gl += 1.0 - h[l + s]
        if 0 <= l - s < n:
            gl -= h[l - s]
        d[k] = (gj - gl) * dx
    return d

def coord_descent(h, n, deadline, offsets=None, max_passes=1000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if offsets is None:
        offsets = [1, 2, 5, 10]
    stagnant = 0
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        for j in order:
            j = int(j)
            for off in offsets:
                l = (j + off) % n
                c = compute_c5_vec(h, n)
                d = coord_dir(h, n, j, l)
                d_lo = max(-h[j], -(1.0 - h[l]))
                d_hi = min(1.0 - h[j], h[l])
                if d_hi - d_lo < 1e-12:
                    continue
                lo, hi = d_lo, d_hi
                for _ in range(40):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = np.max(c + d * m1)
                    f2 = np.max(c + d * m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                delta = (lo + hi) / 2.0
                if abs(delta) < 1e-13:
                    continue
                h[j] += delta
                h[l] -= delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] -= delta
                    h[l] += delta
        if not improved:
            stagnant += 1
            if stagnant > 8:
                break
        else:
            stagnant = 0
    return best_h, best_c

def grad_pair_search(h, n, deadline, step_init=0.003, top_m=60):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = step_init
    for alpha in [200, 1000, 5000]:
        stag = 0
        while time.time() < deadline:
            g = _grad_smooth(h, n, alpha)
            hi_idx = np.argsort(g)[-top_m:]
            lo_idx = np.argsort(g)[:top_m]
            improved = False
            for i in hi_idx:
                if time.time() > deadline:
                    return best_h, best_c
                for j in lo_idx:
                    if i == j:
                        continue
                    for sgn in [1, -1]:
                        dd = sgn * step
                        if h[i] + dd < 0 or h[i] + dd > 1:
                            continue
                        if h[j] - dd < 0 or h[j] - dd > 1:
                            continue
                        h[i] += dd
                        h[j] -= dd
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= dd
                            h[j] += dd
            if not improved:
                step *= 0.5
                if step < 1e-9 or stag > 5:
                    break
                stag += 1
            else:
                stag = 0
    return best_h, best_c
