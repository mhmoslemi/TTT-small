import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_optimize(h, n, deadline, alpha_start=100, alpha_end=5000, n_iter=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if n_iter is None:
        n_iter = int((deadline - time.time()) * 50)
    n_iter = max(1, n_iter)
    lr = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, n_iter - 1))
        grad = lse_gradient(h, n, alpha)
        # Adaptive step size
        step = lr / (1.0 + it * 0.01)
        h_new = project_feasible(h - step * grad, n)
        c = compute_c5(h_new, n)
        if c < best_c:

            h = h_new

                best_h = h_half.copy()
                h = h_half
            else:
                lr *= 0.9
        # Occasionally boost learning rate if stuck
        if it % 50 == 0:
            lr = max(lr, 0.005)
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_exact(h, n):
    target = n / 2.0
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    g = g - g.max()
    exp_g = np.exp(alpha * g)
    p = exp_g / exp_g.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.empty(n)
    for j in range(n):
        mask1 = (j - lags >= 0) & (j - lags < n)
        mask2 = (j + lags >= 0) & (j + lags < n)
        grad[j] = 1.0
        if mask1.any():
            grad[j] -= np.dot(p[mask1], h[j - lags[mask1]])
        if mask2.any():
            grad[j] -= np.dot(p[mask2], h[j + lags[mask2]])
    return grad

def smooth_step(h, n, alpha, lr, max_backtrack=10):
    grad = lse_gradient(h, n, alpha)
    step = lr
    for _ in range(max_backtrack):
        h_new = project_exact(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new, step
        step *= 0.5
    return h, compute_c5(h, n), step * 0.5

def smooth_optimize(h, n, deadline, n_iters=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 20.0
    lr = 0.05
    for it in range(n_iters):
        if time.time() > deadline:
            break
        h, c, lr = smooth_step(h, n, alpha, lr)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Schedule
        alpha *= 1.05
        if it % 10 == 0:
            lr = max(lr, 0.02)
    return best_h, best_c

            delta[k2] += 1.0
    return delta
