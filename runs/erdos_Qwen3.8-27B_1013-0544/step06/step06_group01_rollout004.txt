        h[j] -= d

                best_h = h.copy()

            # Reject
            h[i] -= d
            h[j] += d
        
        T *= cooling

c[n-1] = sum_i h[i]*(1-h[i])

When h[p] changes by d:
New c[n-1] = c[n-1] + d*(1-2*h[p]) - d²

In my formula, c[n-1] is at index position `p` in the array `c[idx]` (since idx = n-1-p+j, and when j=p, idx = n-1).

The linear effect gives: c[n-1] += d*(1-2*h[p]) (from the effect vector at position p)
The quadratic correction: c[n-1] -= d²

nc[p] -= d * d           # quadratic correction (position p in the affected array corresponds to c[n-1])

            hp = project_feasible(h_best + rng.normal(0, noise_scale, n), n)
            alloc = min(per_start - 20, 80)
            if alloc < 15:
                break
            hr, cr = cd_fast(hp, n, max_passes=1000, grid=40,
                             deadline=time.time() + alloc, rng=rng)
            if cr < c_best:
                h_best, c_best = hr, cr

    # Phase 5: Final polish
    if time.time() < deadline - 20:
        hf = slsqp_polish(h_best, n, alpha=500000, maxiter=3000)
        cf = compute_c5(hf, n)
        if cf < c_best:
            h_best, c_best = hf, cf
    if time.time() < deadline - 5:
        h4, c4 = cd_fast(h_best, n, max_passes=2000, grid=64,
                         deadline=time.time() + 10, rng=rng)
        if c4 < c_best:
            h_best, c_best = h4, c4

    h_best = project_feasible(h_best, len(h_best))
    c_best = compute_c5(h_best, len(h_best))
    return h_best, c_best, len(h_best)
