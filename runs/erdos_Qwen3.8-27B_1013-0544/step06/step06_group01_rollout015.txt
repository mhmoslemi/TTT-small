import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coordinate_descent_exact(h, n, max_passes=200, deadline=None):
    """Coordinate descent with exact 1D optimization.
    For each coordinate p, the objective max_k c(k) is piecewise linear-quadratic in h[p].
    We find the best value by checking critical points.
    """
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(np.max(corr)) * dx
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if h[p] < 1e-15 and h[p] > 1 - 1e-15:
                continue  # at boundary
            # Try a few candidate values: 0, 1, current ± small steps, and the value that minimizes the max correlation
            # The correlation c(k) as a function of h[p] = t is:
            # c(k) = c_k_base + t * (coeff_forward[k] - coeff_backward[k]) - t^2 * delta_aligned[k]
            # where the specific terms depend on which indices are involved
            # For simplicity, let's just try a few strategic values
            candidates = [0.0, 1.0, h[p]]
            # Add values that might be optimal
            if h[p] > 0.5:
                candidates.extend([0.3, 0.4, 0.5])
            else:
                candidates.extend([0.6, 0.7, 0.8])
            
            best_val = current_c5
            best_t = h[p]
            for t in candidates:
                if t == h[p]:
                    continue
                h[p] = t
                # Quick check: only update affected correlations
                # Rather than recomputing full correlation, just check if max improved
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(new_corr)) * dx
                if new_c5 < best_val:
                    best_val = new_c5
                    best_t = t
                    corr = new_corr
                else:
                    h[p] = best_t
            if best_t != h[p]:
                h[p] = best_t
                current_c5 = best_val
                improved = True
            h[p] = np.clip(h[p], 0.0, 1.0)
        if not improved:
            break
    h = project_feasible(h, n)
    current_c5 = compute_c5(h, n)
    return h, current_c5

def upper_envelope_critical_points(c_ks, s_ks):
    """Find the d values where the upper envelope of lines c_ks[i] + d*s_ks[i] has vertices.
    Returns sorted array of critical d values."""
    m = len(s_ks)
    if m <= 1:
        return np.array([])
    
    # Sort by slope
    order = np.argsort(s_ks)
    s_sorted = s_ks[order]
    c_sorted = c_ks[order]
    
    # Build upper envelope using stack
    # Each element in stack is (slope, intercept, d_start)
    # The envelope is the max of all lines
    from collections import deque
    stack = []  # list of (s, c, d_left)
    
    for i in range(m):
        s_i, c_i = s_sorted[i], c_sorted[i]
        # Remove lines from stack that are dominated
        while stack:
            s_top, c_top, _ = stack[-1]
            if s_i == s_top:
                # Same slope, keep the one with higher intercept
                if c_i > c_top:
                    stack.pop()
                else:
                    break
            else:
                # Find intersection d = (c_top - c_i) / (s_i - s_top)
                d_cross = (c_top - c_i) / (s_i - s_top)
                if d_cross <= stack[-1][2]:
                    # The top line is never active
                    stack.pop()
                else:
                    stack[-1] = (s_top, c_top, d_cross)  # update d_left of top
                    break
        if len(stack) < 2:
            d_left = -np.inf
        else:
            s_prev, c_prev, _ = stack[-1]
            d_left = (c_prev - c_i) / (s_i - s_prev)
        stack.append((s_i, c_i, d_left))
    
    # Critical points are the d_left values (interior ones)
    critical = [s[2] for s in stack[1:] if np.isfinite(s[2])]
    return np.array(critical)

def obj_lse(x, n, alpha):
    """Log-sum-exp smoothed objective."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    # Numerically stable log-sum-exp
    max_s = shifted.max()
    lse = m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    # Add penalty for constraint violation
    penalty = 100.0 * (x.sum() - n/2.0)**2
    return float(lse * dx + penalty)

def obj_grad_lse(x, n, alpha):
    """Gradient of the LSE objective."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    weights = np.exp(shifted - max_s)
    weights /= weights.sum()
    # Gradient of max wrt x: sum_k weights[k] * dc[k]/dx
    # dc[k]/dx is... complex. Let's use numerical gradient or just return weights
    # Actually, for SLSQP we can just use the objective without gradient
    grad = np.zeros(n)
    # dc[k]/dx[p] = (1-x[p+k-(n-1)]) - x[p+(n-1-k)] - delta_{k,n-1}*2*(x[p]-0.5)
    # This is complex. Let's just use SLSQP without gradient (finite differences)
    return float((m + (max_s + np.log(weights.sum())) / alpha) * dx)

    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    lse = m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
    return float(lse * dx)

def slsqp_opt(h0, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(lse_objective, h0, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def cd_step(h, corr, n, p, grid):
    """Single coordinate descent step. Returns (new_h_val, new_corr, improved, current_max)."""
    d_lo = -h[p]
    d_hi = 1.0 - h[p]
    if d_hi - d_lo < 1e-14:
        return h[p], corr, False, np.max(corr)
    
    n1 = n - 1
    k_lo = max(0, n1 - p)
    k_hi = min(2*n-2, max(2*n-2-p, n1+p))
    
    # Unaffected max
    um = 0.0
    if k_lo > 0:
        um = float(np.max(corr[:k_lo]))
    if k_hi < 2*n-2:
        um = max(um, float(np.max(corr[k_hi+1:])))
    
    ks = np.arange(k_lo, k_hi + 1)
    s = np.zeros(len(ks))
    m1 = p + ks - n1
    va = (m1 >= 0) & (m1 <= n-1)
    s[va] += (1.0 - h[m1[va]])
    m2 = p + n1 - ks
    vv = (m2 >= 0) & (m2 <= n-1)
    s[vv] -= h[m2[vv]]
    
    current_max = float(np.max(corr[ks]))
    overall_max = max(um, current_max)
    
    best_val = overall_max
    best_d = 0.0
    # Use adaptive grid: focus on values near 0 and near boundaries
    deltas = np.linspace(d_lo, d_hi, grid)
    for d in deltas:
        if abs(d) < 1e-14:
            continue
        new_vals = corr[ks] + d * s
        if n1 >= k_lo and n1 <= k_hi:
            ci = n1 - k_lo
            new_vals[ci] -= d * d
        val = max(um, float(np.max(new_vals)))
        if val < best_val - 1e-15:
            best_val = val
            best_d = d
    
    if abs(best_d) > 1e-13:
        corr[ks] += best_d * s
        if n1 >= k_lo and n1 <= k_hi:
            ci = n1 - k_lo
            corr[ci] -= best_d * best_d
        new_hp = np.clip(h[p] + best_d, 0.0, 1.0)
        return new_hp, corr, True, max(um, float(np.max(corr[ks])))
    
    return h[p], corr, False, overall_max

def cd_optimize(h0, n, deadline, rng, grid=60, max_passes=3000):

                improved = True
        if not improved:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
