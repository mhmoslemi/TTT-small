import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_grad(h, n, lags, alpha):
    shifted = alpha * (lags - lags.max())
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    nlags = len(lags)
    grad = np.zeros(n)
    for idx in range(nlags):
        if w[idx] < 1e-12:
            continue
        lag = idx - (n - 1)
        grad += w[idx] * lag_gradient(h, n, lag)
    return grad

def fista_optimize(h0, n, deadline, alpha=50000, max_iter=20000):
    h = h0.copy()
    h_prev = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 1.0 / (4.0 / n + 1e-8)
    lags = compute_all_lags(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + ((t - 1) / (t + 1)) * (h - h_prev)
        y = np.clip(y, 0, 1)
        lags_y = compute_all_lags(y, n)
        g = smooth_max_grad(y, n, lags_y, alpha)
        lr = step
        for _ in range(30):
            cand = y - lr * g
            c_val = compute_c5(project_feasible(cand, n), n)
            pred = F_val = (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (lags_y - lags_y.max()))) + 1e-30) + lags_y.max()
            if c_val <= pred + 0.5 * lr * np.sum(g * (cand - y)):
                break
            lr *= 0.5
        h_new = project_feasible(y - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        lags = compute_all_lags(h, n)
        if it % 500 == 0 and it > 0:
            step = max(1e-10, step * 0.999)
    return best_h, best_c

def slsqp_active(h0, n, top_m, alpha, maxiter=3000):
    lags = compute_all_lags(h0, n)
    top_idx = np.argsort(lags)[-top_m:]
    def obj(x):
        l = compute_all_lags(x, n)
        a = l[top_idx]
        m = a.max()
        s = alpha * (a - m)
        mx = s.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
        return m + np.log(np.sum(np.exp(s))) / alpha
    def obj_grad(x):
        l = compute_all_lags(x, n)
        a = l[top_idx]
        m = a.max()
        s = alpha * (a - m)
        mx = s.max()
        if mx > 500:
            w = np.exp(s - mx)
        else:
            w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            if w[i] < 1e-12:
                continue
            lag = idx - (n - 1)
            g += w[i] * lag_gradient(x, n, lag)
        return g
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',

        return h0.copy()
