import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_onto_constraint(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-12:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            if k > 0:
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            if kk > 0:
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def _smo_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _smo_max_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _eq_con(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_refine(h, n, deadline, max_iters=500):
    h = project_onto_constraint(h, n / 2.0)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    alphas = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_smo_max_obj, h, jac=_smo_max_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 20})
            h_new = project_onto_constraint(res.x, n / 2.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def projected_grad_desc(h, n, deadline, alpha=5000, lr_init=0.005):
    h = project_onto_constraint(h, n / 2.0)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stall = 0

        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        gn = np.linalg.norm(g)

        g = g / gn

            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:

                best_c = c_new
                best_h = h.copy()

            lr *= 0.7

                break

                lr = lr_init * 0.5

def bit_flip_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        while time.time() < deadline:

                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):

                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = max(0.0, h[i] - d)
                            h[j] = min(1.0, h[j] + d)
                    elif h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        c = compute_c5(h, n)

                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = min(1.0, h[i] + d)
                            h[j] = max(0.0, h[j] - d)
            if not improved:
                break
    return best_h, best_c

def greedy_pair_swap(h, n, deadline, max_rounds=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, 5000)
        order = np.argsort(g)
        improved = False
        for _pi in range(50):
            if time.time() > deadline:
                return best_h, best_c
            i = order[_pi]
            j = order[n - 1 - _pi]
            if g[i] - g[j] < 1e-10:
                continue
            ratio = (1.0 - h[i]) / (h[j] + 1e-15) if h[j] > 0 else 10.0
            move = min(0.1, 0.5 * (1.0 - h[i]), 0.5 * h[j]) * min(1.0, ratio)
            if move < 1e-6:
                continue
            h[i] += move
            h[j] -= move
            h = project_onto_constraint(h, n / 2.0)
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()
        if not improved:
            break
    return best_h, best_c

def round_to_01(h, n):
    target = n / 2
    idx = np.argsort(-h)
    h01 = np.zeros(n)
    h01[idx[:int(target)]] = 1.0
    return h01

def combinatorial_local_search(h01, n, deadline):
    best_c = compute_c5(h01, n)
    best_h = h01.copy()
    while time.time() < deadline:
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for a in ones[:200]:
            if time.time() > deadline:
                break
            for b in zeros[:200]:
                best_h[a] = 0.0
                best_h[b] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    ones = np.where(best_h > 0.5)[0]
                    zeros = np.where(best_h < 0.5)[0]
                else:
                    best_h[a] = 1.0
                    best_h[b] = 0.0
        if not improved:
            break
    return best_h, best_c
