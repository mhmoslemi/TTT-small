from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)

def smax_grad(h, n, alpha):
    """Analytical gradient of the soft-max smoothed objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    x = alpha * (c - m)
    x = x - x.max()
    e = np.exp(x)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        s1 = 0.0
        k_lo = p
        k_hi = min(2*n-2, p+n-1)
        if k_lo <= k_hi:
            js = p - np.arange(k_lo, k_hi+1) + n - 1
            s1 = np.sum(w[k_lo:k_hi+1] * (1.0 - h[js]))
        s2 = 0.0
        k_lo2 = max(0, n-1-p)
        k_hi2 = min(2*n-2, n-1+p)
        if k_lo2 <= k_hi2:
            js2 = p + np.arange(k_lo2, k_hi2+1) - n + 1
            s2 = np.sum(w[k_lo2:k_hi2+1] * h[js2])
        grad[p] = (s1 - s2) * dx / alpha
    return grad

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def lbfgsb_opt(h, n, alpha, maxiter, penalty_mult=1e6):
    def obj(x):
        base = softmax_obj(x, n, alpha)
        s = x.sum() - n/2.0
        return base + penalty_mult * s * s
    
    def grad(x):
        g = smax_grad(x, n, alpha)
        s = x.sum() - n/2.0
        return g + 2.0 * penalty_mult * s * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def fast_cd(h, n, max_passes, deadline, rng, penalty=5.0, grid=80):
    h = h.copy()

    current_max = float(np.max(c))
    sumh = float(h.sum())
    dx = 2.0 / n
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]

            if d_hi - d_lo < 1e-14:

            k_lo = max(0, min(p, n - 1 - p))
