import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return (m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha) * dx
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def smooth_obj_grad(x, n, alpha):

def lbbb_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        exp_s = np.exp(s - mx)
        obj = (m + (mx + np.log(exp_s.sum())) / alpha) * dx
    else:
        exp_s = np.exp(s)
        obj = (m + np.log(exp_s.sum()) / alpha) * dx
    w = exp_s / exp_s.sum()
    g = np.zeros(n)
    for p in range(n):
        t1_lo = max(0, p)
        t1_hi = min(2*n-2, p+n-1)
        if t1_lo <= t1_hi:
            g[p] += np.sum(w[t1_lo:t1_hi+1])
        t2_lo = max(0, n-1-p)
        t2_hi = min(2*n-2, p+n-1)
        if t2_lo <= t2_hi:
            i_v = p - (np.arange(t2_lo, t2_hi+1) - n + 1)
            g[p] -= np.dot(w[t2_lo:t2_hi+1], x[i_v])
        t3_lo = max(0, n-1-p)
        t3_hi = min(2*n-2, 2*n-2-p)
        if t3_lo <= t3_hi:
            j_v = p + (np.arange(t3_lo, t3_hi+1) - n + 1)
            g[p] -= np.dot(w[t3_lo:t3_hi+1], x[j_v])
    g *= dx
    return float(obj), g

def lbbb_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj_grad, h, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fast_cd(h, n, max_passes=500, grid=80, deadline=None, rng=None, penalty=5.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            a_lo = max(0, min(p, n-1-p))
            a_hi = min(2*n-2, max(p+n-1, 2*n-2-p))
            um = 0.0
            if a_lo > 0:
                um = max(um, float(np.max(c[:a_lo])))
            if a_hi < 2*n-2:
                um = max(um, float(np.max(c[a_hi+1:])))
            ks = np.arange(a_lo, a_hi+1)
            s = np.zeros(len(ks))
            t1m = (ks >= max(0, p)) & (ks <= min(2*n-2, p+n-1))
            s[t1m] += 1.0
            t2m = (ks >= max(0, n-1-p)) & (ks <= min(2*n-2, p+n-1))
            if np.any(t2m):
                s[t2m] -= h[p - (ks[t2m] - n + 1)]
            t3m = (ks >= max(0, n-1-p)) & (ks <= min(2*n-2, 2*n-2-p))
            if np.any(t3m):
                s[t3m] -= h[p + (ks[t3m] - n + 1)]
            deltas = np.linspace(d_lo, d_hi, grid+1)[1:-1]
            best_d = 0.0
            base_pen = penalty * (sumh - n/2.0)**2
            best_val = max(um, current_max) + base_pen
            for d in deltas:
                new_c = c[ks] + d * s
                ci = (ks == (n-1))
                if np.any(ci):
                    new_c[ci] = c[n-1] + d * s[ci] - d*d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - n/2.0)**2
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                d = best_d
                c[ks] += d * s
                c[n-1] -= d*d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0/n)

def subgrad_polish(h, n, steps=200, eta0=0.001):
    h = h.copy()
    for i in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        active = np.where(c >= cmax - 1e-10)[0]
        g = np.zeros(n)
        for k in active:
            for p in range(n):
                val = 0.0
                if max(0, p) <= k <= min(2*n-2, p+n-1):
                    val += 1.0
                if max(0, n-1-p) <= k <= min(2*n-2, p+n-1):
                    val -= h[p - k + n - 1]
                if max(0, n-1-p) <= k <= min(2*n-2, 2*n-2-p):
                    val -= h[p + k - n + 1]
                g[p] += val / len(active)
        g *= (2.0/n)
        eta = eta0 / (1.0 + 0.01 * i)
        h = project_feasible(h - eta * g, n)
    return h

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i-radius)
                hi = min(n, i+radius+1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i]+d <= 1.0 and h[j]-d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i]-d >= 0.0 and h[j]+d <= 1.0:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
