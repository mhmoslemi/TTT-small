import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(x, n, alpha, penalty=1000.0):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    exp_shifted = np.exp(shifted - shifted.max())
    Z = exp_shifted.sum()
    w = exp_shifted / Z
    obj_smooth = m + (shifted.max() + np.log(Z)) / alpha
    sum_x = x.sum()
    target = n / 2.0
    pen_val = penalty * (sum_x - target) ** 2
    pen_grad = 2.0 * penalty * (sum_x - target)
    obj = obj_smooth + pen_val
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wk * (1.0 - x[s:])
            grad[s:] += wk * (-x[:n - s])
        else:
            t = -s
            grad[t:] += wk * (1.0 - x[:n - t])
            grad[:n - t] += wk * (-x[t:])
    grad += pen_grad
    return float(obj), grad

def lbfgsb_opt(h, n, alpha, penalty, maxiter):
    def fun(x):
        return smooth_max_grad(x, n, alpha, penalty)[0]
    def jac(x):
        return smooth_max_grad(x, n, alpha, penalty)[1]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'maxls': 20})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_desc(h, n, deadline, grid=60):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    sumh = float(h.sum())
    target = n / 2.0
    current_max = float(np.max(c)) * dx
    for _pass in range(300):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                return h, current_max
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            s_lo = min(-p, p - (n - 1))
            s_hi = max(n - 1 - p, p)
            s_lo = max(s_lo, -(n - 1))
            s_hi = min(s_hi, n - 1)
            k_lo = s_lo + (n - 1)
            k_hi = s_hi + (n - 1)
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo])) * dx
            if k_hi < n_corr - 1:
                um = max(um, float(np.max(c[k_hi + 1:])) * dx)
            ks = np.arange(k_lo, k_hi + 1)
            s_arr = k_lo  # placeholder
            s_vec = np.zeros(len(ks))
            for idx in range(len(ks)):
                k = ks[idx]
                s = k - (n - 1)
                if 0 <= p + s < n:
                    s_vec[idx] += (1.0 - h[p + s])
                if 0 <= p - s < n:
                    s_vec[idx] -= h[p - s]
            ci = (ks == (n - 1))
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0

def projected_grad_descent(h, n, alpha, deadline, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = smooth_max_grad(h, n, alpha, penalty=0.0)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        h_trial = h - eta * grad / grad_norm
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-12:
            h = h_trial
            best_c = c_trial
            eta = min(eta * 1.05, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-5:
                break
    return h, best_c
