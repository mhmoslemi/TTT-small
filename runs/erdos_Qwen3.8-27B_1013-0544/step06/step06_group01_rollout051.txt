import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n, c, max_c, grid, rng):
    n_changed = 0
    indices = np.arange(n)
    if rng is not None:
        rng.shuffle(indices)
    for p in indices:
        lo = -h[p]
        hi = 1.0 - h[p]
        if hi - lo < 1e-14:
            continue
        k_lo = 0
        k_hi = min(2 * n - 2, p + n - 1)
        um = 0.0
        if k_hi < 2 * n - 1:
            tail = c[k_hi + 1:]
            if len(tail) > 0:
                um = float(np.max(tail))
        ks = np.arange(k_lo, k_hi + 1)
        s = np.zeros(len(ks))
        a_idx = p - ks + n - 1
        va = (a_idx >= 0) & (a_idx < n)
        s[va] += (1.0 - h[a_idx[va]])
        b_idx = p + ks - (n - 1)
        vb = (b_idx >= 0) & (b_idx < n)
        s[vb] -= h[b_idx[vb]]
        deltas = np.linspace(lo, hi, grid)
        best_d = 0.0
        best_val = max(um, max_c)
        for d in deltas:
            if abs(d) < 1e-14:
                continue
            nv = c[ks] + d * s
            ci = (ks == (n - 1))
            if np.any(ci):
                nv[ci] -= d * d
            val = max(um, float(np.max(nv)))
            if val < best_val - 1e-15:
                best_val = val
                best_d = d
        if abs(best_d) > 1e-12:
            actual_d = min(max(best_d, lo), hi)
            c[ks] += actual_d * s
            ci = (ks == (n - 1))
            if np.any(ci):
                c[n - 1] -= actual_d * actual_d
            h[p] = np.clip(h[p] + actual_d, 0.0, 1.0)
            max_c = max(um, float(np.max(c[ks])))
            n_changed += 1
    return h, c, max_c, n_changed

def cd_optimize(h, n, max_passes, grid, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(c))
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        h, c, max_c, nc = cd_pass(h, n, c, max_c, grid, rng)
        if nc == 0:
            break
    return h, max_c * (2.0 / n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    if zm > 500:
        val = m + (zm + np.log(np.sum(np.exp(z - zm)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(z))) / alpha
    return float(val * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_candidate(h_start, n, deadline, rng):

    best_c = compute_c5(h, n)
    # Phase A: CD with moderate grid
    t_a = min(100, (deadline - time.time()) * 0.35)
    if t_a > 20:
        h_cd, c_cd = cd_optimize(h, n, 600, 60, time.time() + t_a, rng)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)

    # Phase B: SLSQP with multiple alphas
    for a in [5000, 50000, 200000]:

    # Phase C: Fine CD

    # Phase D: Final SLSQP
