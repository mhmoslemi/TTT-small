def fourier_optimize(h, n, m, deadline):

    H = np.fft.fft(h)
    # Keep DC and first m-1 modes (and their conjugates)
    mask = np.zeros(n, dtype=bool)
    for k in range(m):
        mask[k] = True
        mask[n-k] = True  # conjugate
    # Optimize in the subspace of modified coefficients
    def obj(coefs):
        H_new = H.copy()
        H_new[mask] = coefs
        h_new = np.real(np.fft.ifft(H_new))
        h_new = project_feasible(h_new, n)

    # Use Nelder-Mead or Powell on the reduced space

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=500, deadline=None, rng=None, penalty=5.0, grid=80):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2*n-2, max(p+n-1, 2*n-2-p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2*n-2:
                um = max(um, float(np.max(c[k_hi+1:])))
            ks = np.arange(k_lo, k_hi+1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p+n-1)
            bwd = (ks >= n-1-p) & (ks <= 2*n-2-p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n-1)]
            best_d = 0.0
            best_val = max(um, current_max) + penalty * (sumh - n/2.0)**2

                current_max = max(um, float(c[ks].max()))

        if not improved:

    return h, current_max*(2.0/n)

    H = np.fft.fft(h)
    n_keep = max(2, int(n * keep_frac / 2))
    mask = np.zeros(n, dtype=bool)

        mask[k] = True

    return project_feasible(np.real(np.fft.ifft(H)), n)
