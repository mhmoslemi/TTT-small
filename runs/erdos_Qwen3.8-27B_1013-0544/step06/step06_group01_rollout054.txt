    if m * alpha > 500:

        mx = sh.max()

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    sc = alpha * (c - c.max())
    return float((c.max() + np.log(np.sum(np.exp(sc)))) / alpha * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    sc = alpha * (c - cm)
    return float((cm + np.log(np.sum(np.exp(sc)))) / alpha * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h, n)

def polish_schedule(h, n, deadline, alphas, maxiter_each=1500):
    h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_opt(h, n, a, maxiter_each)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
    return h, best_c

def fast_cd(h, n, deadline, grid=50, penalty=3.0, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    target_sum = n / 2.0
    idx = np.arange(n)
    while time.time() < deadline:
        if rng is not None:
            rng.shuffle(idx)
        improved = False
        for p in idx:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = min(p, n - 1 - p)
            k_hi = max(p + n - 1, 2 * n - 2 - p)
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - target_sum) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n-1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - target_sum) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                hf = np.where(fwd)[0]
                hb = np.where(bwd)[0]
                if len(hf) > 0:
                    c[ks[hf]] += d * (1.0 - h[p - ks[hf] + n - 1])
                if len(hb) > 0:
                    c[ks[hb]] -= d * h[p + ks[hb] - (n - 1)]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def expand_optimize(h_src, n_src, n_dst, deadline, rng):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = project_feasible(h_dst, n_dst)
    c_dst = compute_c5(h_dst, n_dst)
    t_mid = min(time.time() + (deadline - time.time()) * 0.4, deadline - 30)
    h_p, c_p = polish_schedule(h_dst, n_dst, t_mid, alphas=[30000, 100000, 300000], maxiter_each=1200)
    if c_p < c_dst:
        h_dst, c_dst = h_p, c_p
    if time.time() < deadline - 10:
        h_cd, c_cd = fast_cd(h_dst, n_dst, deadline - 5, grid=40, penalty=2.0, rng=rng)
        if c_cd < c_dst:
            h_dst, c_dst = h_cd, c_cd
    return h_dst, c_dst
