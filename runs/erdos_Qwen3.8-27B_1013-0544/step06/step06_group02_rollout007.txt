import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sml_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def sml_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        if time.time() > deadline - 30 and alpha > 5000:
            break
        t_start = time.time()
        try:
            res = minimize(sml_obj, h, jac=lambda x: sml_grad(x, n, alpha) + 2e-6*(x.sum()-n/2),
                          args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 20, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def nesterov_pd(h, n, deadline, alpha=5000, max_iters=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        x = project_feasible(h - v, n)
        g = sml_grad(x, n, alpha)
        g += 1e-5 * (x.sum() - n / 2)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        h_new = project_feasible(x - lr * g, n)
        v = lr * (h - x) / (1 + lr)  # Nesterov momentum
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def stochastic_block_search(h, n, deadline, rng, block_sizes=None, ntrials_per_block=20):

        block_sizes = [2, 3, 5, 8, 12, 20]

from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sml_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def sml_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue

        if k >= 0:

            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(sml_obj, h, jac=lambda x, a=alpha: sml_grad(x, n, a) + 1e-6*(x.sum()-n/2),
                          args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 20, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def proj_grad_descent(h, n, deadline, alpha=10000, max_iters=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = sml_grad(h, n, alpha)
        g += 1e-5 * (h.sum() - n / 2)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        v = momentum * v + (1 - momentum) * g
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - lr * v, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr = max(lr * 2.0, 0.005)
            v *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c

def stochastic_blocks(h, n, deadline, rng, ntrials=30):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    block_sizes = [2, 3, 5, 8, 15, 25]
    no_improve = 0
    while time.time() < deadline:
        improved = False
        for bs in block_sizes:
            if time.time() > deadline:
                return h, best_c
            for _ in range(ntrials):
                if time.time() > deadline:
                    return h, best_c
                start = rng.randint(0, max(1, n - bs))
                idx = slice(start, start + bs)
                block_sum = h[idx].sum()
                best_trial_c = best_c
                best_trial_h = None
                for t in range(8):
                    noise = rng.standard_normal(bs) * 0.08
                    nb = h[idx] + noise
                    nb = np.clip(nb, 0.0, 1.0)
                    diff = block_sum - nb.sum()
                    if diff > 0:
                        room = 1.0 - nb
                        if room.sum() > 1e-15:
                            nb += diff * room / room.sum()
                    elif nb.sum() > 1e-15:
                        nb += diff * nb / nb.sum()
                    nb = np.clip(nb, 0.0, 1.0)
                    h_new = h.copy()
                    h_new[idx] = nb
                    h_new = project_feasible(h_new, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_trial_c:
                        best_trial_c = c_new
                        best_trial_h = h_new
                if best_trial_h is not None:
                    h = best_trial_h
                    best_c = best_trial_c
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve > 3:
                for _ in range(5):
                    if time.time() > deadline:
                        return h, best_c
                    k = rng.randint(3, min(30, n // 4) + 1)
                    idx = rng.choice(n, size=k, replace=False)
                    h[idx] = rng.uniform(0, 1, size=k)
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c:
                        best_c = c_new
                        no_improve = 0
                        break
                else:
                    break
        else:
            no_improve = 0
    return h, best_c

def binary_push(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for _ in range(200000):
        if time.time() > deadline:
            return h, best_c
        i = rng.randint(n)
        target = 1.0 if h[i] > 0.5 else 0.0
        diff = target - h[i]
        if abs(diff) < 1e-6:
            continue
        if rng.rand() < 0.7:
            j = rng.randint(n)
            if j == i:
                continue
            if h[j] + diff >= 0 and h[j] + diff <= 1:
                h[i] = target
                h[j] += diff
            else:
                continue
        else:
            room = np.where((1.0 - h >= diff) & (np.arange(n) != i), 1.0, 0.0)
            if room.sum() == 0:
                continue
            j = rng.choice(np.where(room > 0)[0])
            h[i] = target
            h[j] += diff
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
        else:
            h[i] -= diff if h[i] != target else 0
            if h[i] < 0:
                h[i] = 0
            if h[j] > 1:
                h[j] = 1
            h = project_feasible(h, n)
    return h, best_c

def pair_search_fine(h, n, deadline, deltas=None, radius=40):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
