import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_lse_grad(h, alpha, n):
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()
    lse_val = m + shifted.max() / alpha + np.log(exps.sum()) / alpha
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-13:
            continue
        shift = k_idx - (n - 1)
        if shift >= 0:
            if shift == 0:
                grad[:] += wk * (1.0 - 2.0 * h)
            else:
                grad[:n - shift] += wk * omh[shift:]
                grad[shift:] -= wk * h[:n - shift]
        else:
            s = -shift
            grad[s:] += wk * omh[:n - s]
            grad[:n - s] -= wk * h[s:]
    return lse_val, grad

def nesterov_opt(h_init, n, alpha, deadline, n_max_iter=5000):
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    t = 1.0
    eta = 1.0 / (alpha * 2.0 / n + 1e-10)
    eta = min(eta, 0.5 / n)
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, n)
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_curr + (t - 1.0) / t_new * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        f_val, grad = compute_lse_grad(y, alpha, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = project_feasible(h_curr - eta * grad)
        f_new, _ = compute_lse_grad(h_new, alpha, n)
        slope = np.dot(grad, h_new - h_curr)
        if f_new > f_val + 0.5 * eta * slope:
            eta *= 0.5
            h_new = project_feasible(h_curr - eta * grad)
        h_prev = h_curr
        h_curr = h_new
        t = t_new
        if it % 50 == 0:
            c5 = compute_c5(h_curr, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
    c5 = compute_c5(h_curr, n)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_curr.copy()
    return best_h, best_c5

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float) * dx
    cur_c = corr.max()
    best_h = h.copy()
    best_c = cur_c
    ones_set = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list[:]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_set[:]:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max()
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_set.remove(best_j)
                zeros_list.remove(i)
                ones_set.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def resample(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(xn, xo, h), n_new)
