    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    total_steps = max(int((deadline - time.time()) * 2000), 50000)
    cool_rate = (t_end / t_start) ** (1.0 / total_steps)
    T = t_start

    while time.time() < deadline:
        r = rng.random()
        if r < 0.35 and len(corr) > 0:
            top_idx = np.argsort(corr)[::-1][:8]
            s_shift = top_idx[rng.randint(8)]
            actual_shift = s_shift - (n - 1)
            found = False
            if actual_shift >= 0:
                valid_i = np.arange(n - actual_shift)
                cands = valid_i[(h[valid_i] == 1) & (h[valid_i + actual_shift] == 0)]
                if len(cands) > 0:
                    i = cands[rng.randint(len(cands))]
                    j = ones_idx[rng.randint(len(ones_idx))]
                    if j != i and h[j] == 1:
                        found = True
            else:
                valid_i = np.arange(n + actual_shift)
                cands = valid_i[(h[valid_i] == 0) & (h[valid_i - actual_shift] == 1)]
                if len(cands) > 0:
                    j = cands[rng.randint(len(cands))]
                    i = zeros_idx[rng.randint(len(zeros_idx))]
                    if i != j and h[i] == 0:
                        found = True
            if not found:
                i = zeros_idx[rng.randint(len(zeros_idx))]
                j = ones_idx[rng.randint(len(ones_idx))]
        else:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]

        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 4.0)
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=400, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(4):
        if time.time() > deadline:
            break
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            for i in zeros:
                for j in ones:
                    d = swap_delta(h, n, i, j, m_range)
                    nc = (corr + d).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + d
                        cur_c = nc
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) > 0 and len(zi) > 0:
            for _ in range(6):
                a = rng.choice(oi)
                b = rng.choice(zi)
                h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP smooth (10% time)
    p1 = t0 + 0.10 * (budget_s - 5)
    h = project_feasible(best_h, n)
    for alpha in [2000, 20000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 600, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Multiple directed SA runs (50% time)
    p2_end = t0 + 0.60 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    starts = [h_bin,
              round_to_binary(periodic_init(n, 7, 0, 3.0/7.0), n),
              round_to_binary(periodic_init(n, 11, 2, 5.5/11.0), n),
              round_to_binary(periodic_init(n, 9, 4, 4.0/9.0) + 0.3 * rng.randn(n), n),
              round_to_binary(best_h + 0.2 * rng.randn(n), n)]
    seg_lengths = [0.22, 0.12, 0.10, 0.08, 0.08]
    seg_start = t0 + 0.10 * (budget_s - 5)
    for si, start_h in enumerate(starts):
        seg_end = seg_start + seg_lengths[si] * (budget_s - 5)
        seg_end = min(seg_end, p2_end)
        if time.time() > seg_end - 1:
            break
        hb, cb = directed_sa(start_h, n, seg_end, seed=seed + si * 111,
                             t_start=0.03, t_end=0.00002, patience=40000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        seg_start = seg_end

    # Phase 3: Greedy polish (8% time)
    p3 = t0 + 0.68 * (budget_s - 5)
    hg, cg = greedy_binary(best_h, n, p3, max_sweeps=300, seed=seed + 5)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 4: SLSQP from best binary (10% time)
    p4 = t0 + 0.78 * (budget_s - 5)
    if time.time() < p4 - 3:
        hp = project_feasible(best_h + 0.04 * rng.randn(n), n)
        for alpha in [50000, 300000, 1500000]:
            if time.time() > p4:
                break
            hp = slsqp_smooth(hp, n, alpha, 1200, 1e-15)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 5: Final SA round (7% time)
    p5 = dl - 12
    if time.time() < p5 - 5:
        h_bin2 = round_to_binary(best_h, n)
        hb2, cb2 = directed_sa(h_bin2, n, p5, seed=seed + 500,
                               t_start=0.015, t_end=0.00005, patience=30000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 6: Final greedy
    if time.time() < dl - 3:
        hg2, cg2 = greedy_binary(best_h, n, dl - 2, max_sweeps=200, seed=seed + 600)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()

    # Phase 7: Higher resolution attempt
    if time.time() < dl - 10 and n <= 250:
        n2 = n * 2
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, n2)
        hbin2 = round_to_binary(hh, n2)
        hb3, cb3 = directed_sa(hbin2, n2, dl - 3, seed=seed + 700,
                               t_start=0.02, t_end=0.0001, patience=20000)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
            n = n2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
