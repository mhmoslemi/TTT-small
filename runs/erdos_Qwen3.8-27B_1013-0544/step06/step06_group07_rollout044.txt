    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Brief SLSQP (8% of budget)
    p1 = t0 + 0.08 * (budget_s - 3)
    if time.time() < p1:
        hs = project_feasible(h0 + 0.05 * rng.randn(n), n)
        for alpha in [200000, 1000000]:
            if time.time() > p1:
                break
            hs = slsqp_quick(hs, n, alpha, 600)
            c = compute_c5(hs, n)
            if c < best_c:
                best_c = c
                best_h = hs.copy()

    # Phase 2: Main SA (62% of budget) with multiple generations
    p2 = t0 + 0.70 * (budget_s - 3)
    h_bin = round_to_binary(best_h, n)
    sa_seeds = [seed, seed + 7, seed + 13, seed + 29]
    sa_temps = [0.05, 0.035, 0.06, 0.025]
    sa_pat = [25000, 20000, 30000, 18000]
    sa_durs = [0.20, 0.18, 0.16, 0.08]

    for idx in range(len(sa_seeds)):
        seg_end = t0 + (0.08 + sum(sa_durs[:idx + 1])) * (budget_s - 3)
        seg_end = min(seg_end, p2)
        if time.time() > seg_end:
            break
        start = round_to_binary(best_h, n)
        hb, cb = sa_adaptive(start, n, seg_end, sa_seeds[idx],
                            t_start=sa_temps[idx], patience=sa_pat[idx])
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy best-improvement (18% of budget)
    p3 = t0 + 0.88 * (budget_s - 3)
    if time.time() < p3 - 5:
        hg, cg = greedy_bi(round_to_binary(best_h, n), n, p3, max_sweeps=300, n_trials=25000)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Phase 4: Final SA polish (10% of budget)
    if time.time() < dl - 5:
        hb2, cb2 = sa_adaptive(round_to_binary(best_h, n), n, dl - 3, seed + 99,
                              t_start=0.02, patience=12000, base_temp=0.008)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Final greedy polish
    if time.time() < dl - 1:
        hg2, cg2 = greedy_bi(round_to_binary(best_h, n), n, dl - 0.5, max_sweeps=80, n_trials=15000)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()

    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
