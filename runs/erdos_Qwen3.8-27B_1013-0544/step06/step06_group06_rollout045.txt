def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    s = np.arange(2 * n - 1) - (n - 1)
    for p in range(n):
        mask1 = (s >= p - n + 1) & (s <= p)
        idx1 = p - s[mask1]
        g[p, mask1] += 1.0 - h[idx1]
        mask2 = (s >= -p) & (s <= n - 1 - p)
        idx2 = p + s[mask2]
        g[p, mask2] -= h[idx2]
    return g

def thue_morse_seq(n):
