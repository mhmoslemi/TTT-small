    target = n / 2.0

    h = np.zeros(n)
    step = n // (n // 2)
    for i in range(n // 2):
        pos = min(int(i * step + rng.randint(-2, 3)), n - 1)
        if h[pos] == 0:
            h[pos] = 1.0
    while h.sum() < n // 2:
        pos = rng.randint(n)
        if h[pos] == 0:
            h[pos] = 1.0
    return h
