    m = 2 * n - 1
    G = np.zeros((n, m), dtype=float)
    off = n - 1
    G[:, off] = 1.0 - 2.0 * h
    for d in range(1, n):

        im = off - d
        G[d:, ip] += 1.0
        G[d:, ip] -= h[:-d]
        G[:-d, ip] -= h[d:]
        G[:-d, im] += 1.0
        G[:-d, im] -= h[d:]
        G[d:, im] -= h[:-d]
    return G

        if np.isfinite(tv) and tv < mc - 1e-10:
            d = res.x[:n].astype(float)
            if np.all(np.isfinite(d)):
                return d, mc, tv
    return None, mc, float('inf')


def try_step(h, delta, best_c, n):
    for lam in (1.0, 0.8, 0.64, 0.5, 0.4, 0.32, 0.25, 0.2, 0.16, 0.125, 0.1, 0.08, 0.06, 0.04, 0.02):
        hn = h + lam * delta
        hn = adjust_sum_fast(hn, n)
        c = exact_c5(hn, n)
        if c < best_c - 1e-12:
            return hn, c
    return None, float('inf')


def z_from_h(h, eps):
    e = max(1e-12, float(eps))
    hc = np.clip(h, e, 1.0 - e)
    z = np.log(hc / (1.0 - hc))
    return z - z.mean()


def sigmoid(x):
    x = np.clip(np.asarray(x, dtype=float), -50.0, 50.0)
    return 1.0 / (1.0 + np.exp(-x))


def h_from_z(z, n):
    z = np.asarray(z, dtype=float)
    z = z - z.mean()
    target = 0.5 * n
    c = float(np.mean(z))
    err = 0.0
    for _ in range(8):
        h = sigmoid(z - c)
        s = float(h.sum())
        err = s - target
        if abs(err) < 1e-11:
            break
        W = float(np.dot(h, 1.0 - h))
        if W < 1e-12:
            break
        c += err / W
    if abs(err) > 1e-8:
        lo = float(np.min(z)) - 40.0
        hi = float(np.max(z)) + 40.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            s = float(sigmoid(z - mid).sum())
            if s > target:
                lo = mid
            else:
                hi = mid
        c = 0.5 * (lo + hi)
    return sigmoid(z - c)


def z_loss_value(z, tau, n):
    h = h_from_z(z, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = (c - m) / tau
    se = np.exp(s)
    return float(m + tau * math.log(float(se.sum())))


def z_loss_grad(z, tau, n):
    h = h_from_z(z, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = (c - m) / tau
    se = np.exp(s)
    p = se / float(se.sum())
    G = h_gradient_matrix(h, n)
    g = p @ G.T
    W = h * (1.0 - h)
    Wsum = float(W.sum())
    if Wsum < 1e-14:
        grad = g - g.mean()
    else:
        gw = float(np.dot(g, W))
        grad = W * (g - gw / Wsum)
    grad -= grad.mean()
    return grad


def lbfgs_tau(z, n, tau, maxiter):
    try:
        res = minimize(
            z_loss_value,
            z,
            args=(tau, n),
            jac=z_loss_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'maxcor': 50}
        )
    except Exception:
        return None, None
    if res.x is not None:
        zc = np.asarray(res.x, dtype=float)
        zc = zc - zc.mean()
        return zc, h_from_z(zc, n)
    return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + max(10.0, float(budget_s)) - 5.0
    try:
        h0 = np.array(initial_h_values, dtype=float).ravel()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = exact_c5(best_h, n)

    rem0 = max(1.0, deadline - time.time())
    p_lp = time.time() + 0.35 * rem0
    step = 0.03
    stall = 0
    while time.time() < min(p_lp, deadline - 60.0):
        d, mc, tv = lp_step(best_h, n, step)
        if d is None:
            stall += 1
            step *= 0.6
        else:
            hn, nc = try_step(best_h, d, best_c, n)
            if hn is not None:
                best_h = hn
                best_c = nc
                stall = 0
                if step < 0.05:
                    step *= 1.1
            else:
                stall += 1
                step *= 0.6
        if stall > 30 or step < 1e-4:
            break

    p_lb = deadline - 45.0
    if time.time() < p_lb:
        z = z_from_h(best_h, 1e-6)
        no_imp = 0
        taus = (10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                0.015625, 0.0078125, 0.003906, 0.001953, 0.000976)
        for tau in taus:
            if time.time() > p_lb:
                break
            maxiter = 220 if (deadline - time.time()) > 90.0 else 80
            zout, hout = lbfgs_tau(z, n, tau, maxiter)
            if hout is not None:
                hp = adjust_sum_fast(hout, n)
                if abs(hp.sum() - 0.5 * n) > 1e-10:
                    hp = project_feasible(hp, n)
                c = exact_c5(hp, n)
                if c < best_c - 1e-12:
                    best_h = hp
                    best_c = c
                    if zout is not None:
                        z = zout
                    no_imp = 0
                else:
                    if zout is not None:
                        z = zout
                    no_imp += 1
                    if no_imp > 6:
                        break
                    if no_imp > 3:
                        z = z_from_h(best_h, 1e-6)
                        no_imp = 2
            else:
                no_imp += 1
                if no_imp > 6:
                    break

    step = 0.01
    stall = 0
    while time.time() < deadline - 8.0 and stall < 30:
        d, mc, tv = lp_step(best_h, n, step)
        if d is not None:
            hn, nc = try_step(best_h, d, best_c, n)
            if hn is not None:
                best_h = hn
                best_c = nc
                stall = 0
                if step < 0.02:
                    step *= 1.1
            else:
                stall += 1
                step *= 0.5
        else:
            stall += 1
            step *= 0.5

    backup_h = best_h.copy()
    backup_c = best_c
    hfin = project_feasible(best_h, n)
    hfin = np.clip(hfin, 0.0, 1.0)
    cfin = exact_c5(hfin, n)
    if cfin <= backup_c + 1e-12:
        best_h = hfin
        best_c = cfin
    else:
        best_h = adjust_sum_fast(backup_h, n)
        best_c = exact_c5(best_h, n)

    return best_h, best_c, n
