import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=500, deadline=None, n_grid=48):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    grid = np.linspace(0, 1, n_grid)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            a = h[i]
            g = np.zeros(2 * n - 1)
            m_lo = max(-(n - 1), i - (n - 1))
            m_hi = min(n - 1, i)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                g[n - 1 + ms] += (1.0 - h[i - ms])
            m_lo2 = max(-(n - 1), -i)
            m_hi2 = min(n - 1, n - 1 - i)
            if m_lo2 <= m_hi2:
                ms2 = np.arange(m_lo2, m_hi2 + 1)
                g[n - 1 + ms2] += (-h[i + ms2])
            C0 = corr[n - 1] - a * (1.0 - a)
            A = corr - a * g
            M = A[None, :] + g[None, :] * grid[:, None]
            M[:, n - 1] = C0 + grid - grid * grid
            vals = M.max(axis=1)
            idx = vals.argmin()
            best_t = grid[idx]
            if 0 < idx < n_grid - 1:
                lo = grid[idx - 1]
                hi = grid[idx + 1]
                for _ in range(12):
                    mid = 0.5 * (lo + hi)
                    v_mid = max((A + g * mid).max(), C0 + mid - mid * mid)
                    v_lo = max((A + g * lo).max(), C0 + lo - lo * lo)
                    if v_lo > v_mid:
                        lo = mid
                    else:
                        hi = mid
                best_t = 0.5 * (lo + hi)
            if abs(best_t - a) > 1e-12:
                corr += g * (best_t - a)
                corr[n - 1] = C0 + best_t - best_t * best_t
                h[i] = best_t
        h = project_feasible(h, n)
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        c = float(corr.max() * dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -50, 50)
        return m + np.log(np.exp(shifted).sum()) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
