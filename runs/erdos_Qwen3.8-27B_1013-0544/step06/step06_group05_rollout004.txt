def compute_c5(h, n):
    n = int(n)
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def max_count(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def project_h(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    np.clip(h, 0.0, 1.0, out=h)
    target = float(target)
    res = target - float(h.sum())
    if abs(res) <= 1e-10:

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mu = 0.0
    for _ in range(70):
        mu = 0.5 * (lo + hi)
        p = np.clip(h - mu, 0.0, 1.0)
        s = float(p.sum())
        if s > target:

        else:
            hi = mu

    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    return h


def compute_gvals(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    g0 = float(np.dot(h, 1.0 - h))
    pref = np.concatenate(([0.0], np.cumsum(h)))
    gpos = np.empty(n - 1, dtype=float)
    gneg = np.empty(n - 1, dtype=float)
    for d in range(1, n):
        a = h[:-d]
        b = h[d:]
        ab = float(np.dot(a, b))
        gpos[d - 1] = float(pref[n - d] - ab)
        gneg[d - 1] = float(pref[n] - pref[d] - ab)
    return g0, gpos, gneg


def grad_from_weights(h, n, w0, wpos, wneg):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    grad = np.zeros(n, dtype=float)
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = float(wpos[d - 1])
        wn = float(wneg[d - 1])
        if wp == 0.0 and wn == 0.0:
            continue
        a = h[:-d]
        b = h[d:]
        if wp != 0.0:
            grad[:-d] += wp * (1.0 - b)
            grad[d:] -= wp * a
        if wn != 0.0:
            grad[:-d] -= wn * b
            grad[d:] += wn * (1.0 - a)
    return grad


def exact_subgrad(h, n, eps):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    g0, gpos, gneg = compute_gvals(h, n)
    mp = float(np.max(gpos)) if len(gpos) else 0.0
    mn = float(np.max(gneg)) if len(gneg) else 0.0
    m = max(g0, mp, mn)

    cp = gpos >= (m - eps)
    cn = gneg >= (m - eps)
    active = (1.0 if g0 >= (m - eps) else 0.0)
    active += float(np.count_nonzero(cp)) + float(np.count_nonzero(cn))
    if active <= 0.0:
        active = 1.0

    w0 = (1.0 if g0 >= (m - eps) else 0.0) / active
    wpos = np.zeros(n - 1, dtype=float)
    wneg = np.zeros(n - 1, dtype=float)
    wpos[cp] = 1.0 / active
    wneg[cn] = 1.0 / active

    return m, grad_from_weights(h, n, w0, wpos, wneg)

        if improved:
            h = project_h(best_h, 0.5 * n)
            f = best_f
            stagn = 0
        else:
            step = 0.05
            for _ in range(6):
                h2 = project_h(h - step * grad, 0.5 * n)
                f2 = float(max_count(h2))
                if f2 < f - 1e-10:
                    h = h2
                    f = f2
                    improved = True
                    break
                step *= 0.5
            if not improved:
                stagn += 1
                if stagn >= 4:
                    break

    return project_h(h, 0.5 * n)


def soft_value(h, n, alpha):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    alpha = max(1.0, float(alpha))
    g0, gpos, gneg = compute_gvals(h, n)
    vals = np.concatenate((np.array([g0]), gpos, gneg))
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    sh = np.clip(sh, -40.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / alpha


def soft_value_grad(h, n, alpha):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(n)
    alpha = max(1.0, float(alpha))
    g0, gpos, gneg = compute_gvals(h, n)
    vals = np.concatenate((np.array([g0]), gpos, gneg))
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    sh = np.clip(sh, -40.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(vals)
        active = np.where(vals >= m - 1e-9)[0]
        if len(active) == 0:
            active = [0]
        w[active] = 1.0 / float(len(active))
    else:
        w = e / s
    grad = grad_from_weights(h, n, float(w[0]), w[1:n], w[n:2 * n - 1])
    f = m + float(np.log(s)) / alpha
    return f, grad


def obj_state(x):
    n = int(_STATE['n'])
    alpha = float(_STATE['alpha'])
    return soft_value(x, n, alpha)


def jac_state(x):
    n = int(_STATE['n'])
    alpha = float(_STATE['alpha'])
    _, g = soft_value_grad(x, n, alpha)
    return g


def cons_state(x):
    return float(np.sum(x)) - 0.5 * int(_STATE['n'])


def cons_jac_state(x):
    return np.ones(int(_STATE['n']))


def callback_state(x):
    n = int(_STATE.get('n', 0))
    dl = float(_STATE.get('deadline', 1e100))
    if time.time() > dl:
        raise RuntimeError("budget")
    if n <= 0:
        return
    h = project_h(x, 0.5 * n)
    c = compute_c5(h, n)
    if c < _BEST['c']:
        _BEST['h'] = h.copy()
        _BEST['c'] = float(c)


def slsqp_refine(h, n, alpha, maxiter, deadline):
    n = int(n)
    alpha = float(alpha)
    h = project_h(h, 0.5 * n)
    c0 = compute_c5(h, n)
    _BEST['h'] = h.copy()
    _BEST['c'] = float(c0)

    _STATE['n'] = n
    _STATE['alpha'] = alpha
    _STATE['deadline'] = float(deadline)

    x0 = h.copy()
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {'type': 'eq', 'fun': cons_state, 'jac': cons_jac_state}

    res = None
    try:
        res = minimize(obj_state, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac_state, callback=callback_state,
                       options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': False})
    except Exception:
        res = None

    if res is not None and hasattr(res, 'x') and res.x is not None:
        h2 = project_h(res.x, 0.5 * n)
        c2 = compute_c5(h2, n)
        if c2 < _BEST['c']:
            _BEST['h'] = h2
            _BEST['c'] = float(c2)

    return _BEST['h'].copy(), float(_BEST['c'])


def binarize_balanced(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    k = n // 2
    order = np.argsort(h)[::-1]
    x = np.zeros(n, dtype=float)
    x[order[:k]] = 1.0
    return x


def interpolate_parent(p, n):
    p = np.asarray(p, dtype=float).ravel()
    p = np.nan_to_num(p, nan=0.0, posinf=1.0, neginf=0.0)
    np.clip(p, 0.0, 1.0, out=p)
    n0 = p.size
    n = int(n)
    if n0 == 0:
        return project_h(np.full(n, 0.5), 0.5 * n)
    if n0 == n:
        return project_h(p, 0.5 * n)

    b = np.floor(np.linspace(0, n, n + 1) * (float(n0) / float(n))).astype(int)
    b = np.clip(b, 0, n0)
    if b[-1] < n0:
        b[-1] = n0

    pref = np.concatenate(([0.0], np.cumsum(p)))
    counts = np.diff(b)
    vals = np.empty(n, dtype=float)
    for i in range(n):
        if counts[i] > 0:
            vals[i] = (pref[b[i + 1]] - pref[b[i]]) / float(counts[i])
        else:
            vals[i] = p[min(int(b[i]), n0 - 1)]
    return project_h(vals, 0.5 * n)


def make_random_binary(n, seed):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    perm = rng.permutation(n)
    x = np.zeros(n, dtype=float)
    x[perm[:n // 2]] = 1.0
    return x


def make_structured_binary(n, seed):
    n = int(n)
    m = n // 2
    x = np.zeros(n, dtype=float)
    phi = 0.6180339887498949
    k = 0
    cnt = 0
    while cnt < m and k < 20 * m + 10:
        i = int((k * phi * n) % n)
        if x[i] == 0.0:
            x[i] = 1.0
            cnt += 1
        k += 1
    if cnt < m:
        rng = np.random.RandomState(int(seed))
        zeros = np.flatnonzero(x == 0.0)
        if zeros.size >= m - cnt:
            pick = rng.choice(zeros, size=m - cnt, replace=False)
            x[pick] = 1.0
    return x


def optimize_candidate(h, n, deadline, seed):
    n = int(n)
    rng = np.random.RandomState(seed)
    h = project_h(h, 0.5 * n)
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = float(c)

    for eps in (0.1, 0.5, 1.5):
        if time.time() >= deadline:
            break
        sub = min(deadline, time.time() + 20.0)
        h2 = fw_exact_refine(best_h, n, 120, sub, eps, rng)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)

    if n <= 320:
        mm = 60 if n <= 240 else 35
        for alpha in (1000.0, 10000.0):
            if time.time() >= deadline - 8.0:
                break
            h2, c2 = slsqp_refine(best_h, n, alpha, mm, deadline)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)

    if time.time() < deadline - 5.0:
        for eps in (0.25, 1.0):
            if time.time() >= deadline:
                break
            h2 = fw_exact_refine(best_h, n, 60, min(deadline, time.time() + 10.0), eps, rng)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)

    if time.time() < deadline - 12.0:
        h3 = project_h(best_h + 0.03 * rng.randn(n), 0.5 * n)
        h3 = fw_exact_refine(h3, n, 80, min(deadline, time.time() + 15.0), 0.5, rng)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_h = h3.copy()
            best_c = float(c3)

    return best_h, float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    horizon = max(30.0, float(budget_s) - 10.0)
    deadline = t0 + horizon
    rng = np.random.RandomState(seed)

    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=float).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=1.0, neginf=0.0)
        np.clip(p, 0.0, 1.0, out=p)
        if p.size > 8:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        n_par = parent.size
        if n_par % 2 == 1:
            n_par -= 1
        if n_par > 350:
            n_par = 250
        if n_par < 10:
            parent = None
            n_par = 250
        if parent is not None:
            if n_par == parent.size:
                h_par = project_h(parent, 0.5 * n_par)
            else:
                h_par = interpolate_parent(parent, n_par)
        else:
            h_par = project_h(np.full(250, 0.5), 125.0)
            n_par = 250
    else:
        n_par = 250
        h_par = project_h(np.full(n_par, 0.5), 0.5 * n_par)

    best_h = None
    best_c = 1e100
    best_n = 0

    specs = []
    if parent is not None:
        specs.append((h_par, n_par, 0.32))
        specs.append((binarize_balanced(h_par, n_par), n_par, 0.12))
        specs.append((make_random_binary(n_par, seed + 1), n_par, 0.12))
        specs.append((make_structured_binary(n_par, seed + 2), n_par, 0.06))
        h_alt = interpolate_parent(parent, 300)
        specs.append((h_alt, 300, 0.20))
        specs.append((binarize_balanced(h_alt, 300), 300, 0.08))
    else:
        for idx, nn in enumerate((128, 192, 256)):
            specs.append((make_random_binary(nn, seed + 10 * idx), nn, 0.20))
            specs.append((make_structured_binary(nn, seed + 20 + 10 * idx), nn, 0.10))

    for idx, (h_start, n_start, frac) in enumerate(specs):
        end = t0 + float(frac) * horizon
        if time.time() >= deadline - 2.0:
            break
        sub_dead = min(deadline - 2.0, end)
        if sub_dead <= time.time():
            continue
        h_opt, c_opt = optimize_candidate(h_start, n_start, sub_dead, seed + 100 + idx)
        if c_opt < best_c:
            best_c = float(c_opt)
            best_h = h_opt.copy()
            best_n = int(n_start)
        if time.time() >= deadline - 8.0:
            break

    if best_h is None:
        best_n = 250
        best_h = project_h(np.full(best_n, 0.5), 0.5 * best_n)
        best_c = compute_c5(best_h, best_n)

    if time.time() < deadline - 12.0:
        for eps in (0.1, 0.5, 2.0):
            if time.time() >= deadline - 3.0:
                break
            h2 = fw_exact_refine(best_h, best_n, 100, min(deadline - 3.0, time.time() + 10.0), eps, rng)
            c2 = compute_c5(h2, best_n)
            if c2 < best_c:
                best_c = float(c2)
                best_h = h2.copy()

    if best_n <= 320 and time.time() < deadline - 25.0:
        for alpha in (10000.0, 100000.0):
            if time.time() >= deadline - 3.0:
                break
            h2, c2 = slsqp_refine(best_h, best_n, alpha, 80, deadline - 3.0)
            if c2 < best_c:
                best_c = float(c2)
                best_h = h2.copy()

    if time.time() < deadline - 1.0:
        h2 = fw_exact_refine(best_h, best_n, 80, deadline - 1.0, 0.5, rng)
        c2 = compute_c5(h2, best_n)
        if c2 < best_c:
            best_c = float(c2)
            best_h = h2.copy()

    best_h = project_h(best_h, 0.5 * int(best_n))
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
