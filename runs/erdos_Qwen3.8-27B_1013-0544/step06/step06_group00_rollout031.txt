```python
def compute_gradient(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    corr_hp = np.correlate(h, p, 'full')  # length 3n-2
    q = p[::-1]
    corr_hq = np.correlate(h, q, 'full')  # length 3n-2
    
    term1 = corr_hp[2*n-2 : 3*n-2][::-1]  # length n
    term2 = corr_hq[2*n-2 : 3*n-2][::-1]  # length n
    
    grad = (1.0 - term1 - term2) * dx / alpha
    return grad
```

```python
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Bisection on threshold
    def proj(tau):
        return np.clip(h - tau, 0, 1).sum()
    
    lo, hi = -1.0, 1.0

        if proj(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)
```

```python
                delta[n-1 + (j-i)] += 1.0
                delta[n-1 + (i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_set.discard(j)
                    zeros_set.discard(i)
                    ones_set.add(i)
                    zeros_set.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def projected_grad(h, n, alpha, n_steps, lr, mom, deadline):
    h = project_feasible(h.copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        g = fast_gradient(h, alpha, n)
        v = mom * v - lr * g
        h = project_feasible(h + v, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=2000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, 'full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T = 0.01
    cool = 0.99997
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        delta[n-1 + (j-i)] += 1.0
        delta[n-1 + (i-j)] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp((cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_time = budget_s - 5
    deadline = t0 + total_time
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    # Phase 1: Binary greedy (8%)
    p1 = t0 + 0.08 * total_time
    hb, cb = greed_binary(h0, n, p1, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: Projected gradient with increasing alpha (42%)
    p2 = t0 + 0.50 * total_time
    h = best_h.copy()
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > p2 - 10:
            break
        lr = min(2.0 / (alpha * 0.005 + 1), 0.5)
        steps = int(8e7 / (n * np.log2(n) * max(1, alpha / 100)))
        steps = max(steps, 2000)
        for trial in range(3):
            if time.time() > p2 - 5:
                break
            h = projected_grad(h, n, alpha, steps, lr, 0.9, p2 - 5)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # Phase 3: Anneal binary from continuous (20%)
    p3 = t0 + 0.70 * total_time
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=2000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Phase 4: Greedy from annealed result (10%)
    p4 = t0 + 0.80 * total_time
    hb2, cb2 = greed_binary(best_h, n, p4, max_sweeps=300)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    # Phase 5: High-alpha gradient polish (12%)
    p5 = deadline - 2
    for alpha in [100000, 500000, 2000000, 10000000]:
        if time.time() > p5 - 2:
            break
        lr = min(2.0 / (alpha * 0.005 + 1), 0.1)
        h = projected_grad(best_h, n, alpha, 30000, lr, 0.95, p5 - 2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Noise restarts (final 5%)
    if time.time() < deadline - 3:
        for trial in range(5):
            if time.time() > deadline - 3:
                break
            h = best_h + 0.01 * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [500000, 5000000]:
                if time.time() > deadline - 3:
                    break
                lr = min(2.0 / (alpha * 0.005 + 1), 0.1)
                h = projected_grad(h, n, alpha, 20000, lr, 0.95, deadline - 3)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
