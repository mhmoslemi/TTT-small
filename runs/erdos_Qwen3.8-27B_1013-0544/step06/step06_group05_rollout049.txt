from scipy.optimize import minimize, Lfvs
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    """Compute gradient of smoothed objective (1/alpha)*log(sum(exp(alpha*c_k)))"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    p /= p.sum()
    
    # Gradient of c[lag] with respect to h[j]
    # c[lag_idx] = sum_i h[i] * (1 - h[i - lag]) where lag = lag_idx - (n-1)
    # d/dh[j] c[lag_idx] = (1 - h[j-lag]) - h[j+lag] (with bounds)
    grad = np.zeros(n)
    for lag_idx in range(2*n - 1):
        lag = lag_idx - (n - 1)
        # Contribution from i = j: (1 - h[j-lag])
        # Contribution from i-lag = j, i = j+lag: -h[j+lag]
        if lag >= 0:
            # h[j-lag] is valid when j >= lag
            idx = np.arange(lag, n)
            grad[idx] += p[lag_idx] * (1.0 - h[idx - lag])
            # h[j+lag] is valid when j+lag < n, i.e., j < n-lag

            grad[idx2] += p[lag_idx] * (-h[idx2 + lag])
        else:
            # lag < 0, let l = -lag

            idx2 = np.arange(l, n)
            grad[idx2] += p[lag_idx] * (-h[idx2 - l])

import time

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    p /= p.sum()

        lag = lag_idx - (n - 1)
        w = p[lag_idx]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += w * (-h[lag:])
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += w * (-h[:n - l])
    return grad * dx

def gradient_descent(h_init, n, deadline, alpha_start=10.0, alpha_end=100000.0,
                     n_iter=8000, lr_start=0.05, lr_end=0.001, momentum=0.85):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = float(it) / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = compute_gradient(h, n, alpha)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    p /= p.sum()
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        lag = li - (n - 1)
        w = p[li]
        if w < 1e-15:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    return grad * dx

def proj_grad_opt(h0, n, deadline, alpha0=20.0, alpha1=200000.0,
                  iters=5000, lr0=0.02, lr1=0.0003, mu=0.85):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(iters):
        if time.time() > deadline:
            break
        t = float(it) / max(1, iters - 1)
        alpha = alpha0 * (alpha1 / alpha0) ** t
        lr = lr0 * (lr1 / lr0) ** t
        g = smooth_grad(h, n, alpha)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def binarize(h, n):
    s = np.sort(h)[::-1]
    thresh = s[n // 2]
    hb = (h >= thresh).astype(float)
    cnt = int(hb.sum())
    if cnt > n // 2:
        idx = np.where(hb == 1)[0]
        order = np.argsort(-h[idx])
        hb[idx[order[cnt - n // 2:]]] = 0.0
    elif cnt < n // 2:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])
        hb[idx[order[:n // 2 - cnt]]] = 1.0
    return hb

def slsqp_final(h0, n, maxiter=5000):

    def obj(x):

        m = g.max()
        a = 10000.0
        sh = a * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / a
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
