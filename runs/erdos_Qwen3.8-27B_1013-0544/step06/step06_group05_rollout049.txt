import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def make_signature(h, n):
    """Compute sig[p][lag] = h[p+lag] + h[p-lag] for all p, lag"""
    # sig has shape (n, 2n-1) where lag = col - (n-1)
    sig = np.zeros((n, 2*n-1))
    for p in range(n):
        for col in range(2*n-1):
            lag = col - (n-1)
            s = 0.0
            if 0 <= p + lag < n:
                s += h[p + lag]
            if 0 <= p - lag < n:
                s += h[p - lag]
            sig[p, col] = s
    return sig

            break

        critical_mask = corr >= cur_max - 2.0
        critical_lags = np.where(critical_mask)[0]
        
        best_improvement = 0  # We want to minimize new_max, so maximize (cur_max - new_max)
        best_i = -1
        best_j = -1
        best_new_max = cur_max

def simulated_annealing(h, n, deadline, rng, T_start=0.5, T_end=0.01, max_iters=50000):

            if 0 <= p - lag < n:
                s += h[p - lag]
            sig[p, col] = s
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)

        new_max = new_corr.max()

            # Accept
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            
            # Update sig
            for col in range(2*n-1):

                if 0 <= i - lag < n:
                    sig[i-lag, col] += 1
                if 0 <= i + lag < n:
                    sig[i+lag, col] += 1
                if 0 <= j - lag < n:
                    sig[j-lag, col] -= 1
                if 0 <= j + lag < n:
                    sig[j+lag, col] -= 1
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # Reject: don't change anything

def obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2n-1,)
    a = alpha
    ac = a * corr
    m = ac.max()
    exp_ac = np.exp(ac - m)
    w = exp_ac / exp_ac.sum()  # softmax weights, shape (2n-1,)
    val = (m + np.log(exp_ac.sum())) / a
    
    # Gradient
    grad = np.zeros(n)
    for col in range(2*n-1):
        lag = col - (n-1)
        wc = w[col] * dx
        # d(corr[col])/dh[j] = (1-h[j+lag]) * 1[j+lag valid] - h[j-lag] * 1[j-lag valid]
        # For j where j+lag is valid (0 <= j+lag < n):
        #   contribution: wc * (1-h[j+lag])
        # For j where j-lag is valid (0 <= j-lag < n):
        #   contribution: -wc * h[j-lag]
        if lag >= 0:
            # j+lag valid: j in [0, n-1-lag]
            js = np.arange(0, n - lag)
            grad[js] += wc * (1 - h[js + lag])
            # j-lag valid: j in [lag, n-1]
            js2 = np.arange(lag, n)
            grad[js2] -= wc * h[js2 - lag]
        else:
            # j+lag valid: j in [-lag, n-1] (since lag < 0, j >= -lag)
            js = np.arange(-lag, n)
            grad[js] += wc * (1 - h[js + lag])
            # j-lag valid: j in [0, n+lag-1] (since lag < 0, n+lag = n-|lag|)
            js2 = np.arange(0, n + lag)
            grad[js2] -= wc * h[js2 - lag]
