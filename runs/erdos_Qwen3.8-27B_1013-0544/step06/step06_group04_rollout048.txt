import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_objective(h, n, alpha):
    """Log-sum-exp smoothed max of correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    smax = shifted.max()
    return m + (smax + np.log(np.exp(shifted - smax).sum())) / alpha

def smooth_grad(h, n, alpha):
    """Gradient of the smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    smax = shifted.max()
    exp_shifted = np.exp(shifted - smax)
    w = exp_shifted / exp_shifted.sum()
    # d(corr[m])/dh[i] ... this is complex, use numerical gradient
    # Actually let's just use SLSQP's numerical gradient
    return None  # Let SLSQP do it

def smooth_optimize(h_init, n, alphas, maxiter=200, time_limit=30):
    """Run SLSQP with increasing alpha."""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_start = time.time()
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        
        def obj(x):
            return smooth_objective(x, n, alpha)
        
        try:

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

def smooth_grad(corr, h, n, alpha, dx):
    """Compute gradient of smoothed objective w.r.t. h."""
    g = corr * dx
    # softmax weights
    shifted = alpha * g
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    c = n - 1
    grad = np.zeros(n)
    
    # grad[i] = dx * sum_k p_k * (1 - h[i+k-c] - h[i-k+c]) (with boundary handling)
    # = dx * (sum_k p_k) - dx * sum_k p_k * h[i+k-c] - dx * sum_k p_k * h[i-k+c]
    # = dx - dx * sum_k p_k * h[i+k-c] - dx * sum_k p_k * h[i-k+c]
    
    # The second term: sum_k p_k * h[i+k-c] for valid i+k-c
    # Let l = i+k-c, so k = l+i-c. As k ranges 0..2n-2, l ranges i-c..i-c+2n-2 = i-(n-1)..i+n-1
    # Valid l: max(0, i-c) <= l <= min(n-1, i+c)
    # p indexed by k = l + i - c
    
    for i in range(n):
        # Term 2: sum over valid l of p[l+i-c] * h[l]
        lo = max(0, i - c)
        hi = min(n-1, i + c)
        if lo <= hi:
            k_idx = np.arange(lo, hi+1) + i - c
            grad[i] -= np.dot(p[k_idx], h[lo:hi+1])
        
        # Term 3: sum over valid l of p[l-i+c] * h[l]  [where i-k+c=l means k=i-l+c, and p is indexed by k]
        # k = i-l+c, l ranges such that 0<=k<2n-1 and 0<=l<n
        # k = i-l+c: need 0<=i-l+c<2n-1, i.e., i+c-(2n-2)<=l<=i+c, and 0<=l<n
        lo = max(0, i+c-(2*n-2))
        hi = min(n-1, i + c)
        if lo <= hi:
            k_idx = i - np.arange(lo, hi+1) + c
            grad[i] -= np.dot(p[k_idx], h[lo:hi+1])
        
        grad[i] += dx  # the "sum_k p_k * 1" part = dx * sum p_k = dx
    
    return grad

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_grad(h, n, alpha):
    """Gradient of log-sum-exp smoothed max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    shifted = alpha * g
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    c = n - 1
    grad = dx * np.ones(n)  # the "1" part of d(corr[k])/dh[i] = 1 - ...
    
    # Subtract: dx * sum_k p_k * h[i+k-c] (for valid indices)
    # This is like a correlation/convolution
    for i in range(n):
        lo = max(0, i - c)
        hi = min(n - 1, i + c)
        if lo <= hi:
            k_idx = np.arange(lo, hi + 1) + i - c
            grad[i] -= dx * np.dot(p[k_idx], h[lo:hi+1])
        lo = max(0, i + c - (2*n-2))
        hi = min(n-1, i + c)
        if lo <= hi:
            k_idx = i - np.arange(lo, hi+1) + c
            grad[i] -= dx * np.dot(p[k_idx], h[lo:hi+1])
    
    return grad

def projected_gradient_descent(h_init, n, alphas, n_iter=200, time_limit=60):
    """PGD with projection onto {0≤h≤1, sum(h)=n/2}."""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_start = time.time()
    
    for alpha in alphas:
        # Adaptive step size
        lr = 0.01 / alpha * alpha  # hmm, need to think about this
        lr = 2.0 / alpha  # Lipschitz constant estimate
        for it in range(n_iter):
            if time.time() - t_start > time_limit:
                break
            grad = smooth_grad(h, n, alpha)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            # Simple acceptance: always accept (or use backtracking)
            h = h_new
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

        corr = np.correlate(x, 1.0 - x, mode='full') * dx

        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
