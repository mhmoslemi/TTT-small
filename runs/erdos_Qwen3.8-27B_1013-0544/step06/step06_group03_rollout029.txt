```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_BC(h, n):
    """Build B and C matrices for binary h.
    B[p, k] = (1-h[p+lag]) if 0<=p+lag<n, else 0, where lag = k-(n-1)
    C[p, k] = h[p-lag] if 0<=p-lag<n, else 0, where lag = k-(n-1)
    """

    C = np.zeros((n, 2*n-1))

        # B[p, n-1-p : 2*n-1-p] = 1 - h[0:n-p]

        # C[p, p : n+p] = h[0:p+1]
        C[p, p : n+p] = h[0:p+1]
    return B, C

    """Local search on binary string with exactly n/2 ones."""
    h = h_bin.copy().astype(float)
    c = compute_all_lags(h, n)

    for sweep in range(max_sweeps):

            break
        B, C = build_BC(h, n)
        D = B - C
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        improved = False
        # For each one i, find best zero j to swap with
        for i in ones:
            if time.time() > deadline:
                return h, c_max

            # Need: D[j,:] < threshold for all k (and j in zeros)
            threshold = D[i] + (c_max - c) / dx

            # diffs = D[zeros] - threshold has shape (n_zeros, 2n-1)
            # valid = np.all(diffs < 0, axis=1)
            # But this is O(n_zeros * (2n-1)) which is fine
            diffs = D[zeros] - threshold[np.newaxis, :]
            valid = np.all(diffs < 0, axis=1)
            if valid.any():

                # new_max[j] = max(c + dx*(D[zeros[j]] - D[i]))
                # For valid ones:
                candidate_idxs = zeros[valid]
                # Compute new max for each
                new_d = D[candidate_idxs][:, np.newaxis, :] - D[i][np.newaxis, :]
                new_c = c[np.newaxis, :] + dx * new_d
                new_c[:, n-1] = 0  # Diagonal fix (lag=0 stays 0 for binary)

                new_maxs = new_c.max(axis=1)
                best_local = np.argmin(new_maxs)

                h[i] = 0
                h[j] = 1
                c = compute_all_lags(h, n)
                c_max = float(c.max())
                improved = True
                break  # Restart sweep after accepting a swap
        
        if not improved:
            break
    
    return h, c_max
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def build_D(h, n):
    """Build D matrix where D[p, k] = (1-h[p+lag])_{valid} - h[p-lag]_{valid}
    with lag = k - (n-1). Set diagonal (k=n-1) to 0 for correctness with binary h."""
    D = np.zeros((n, 2*n-1))
    for p in range(n):
        D[p, n-1-p : 2*n-1-p] += 1.0 - h[0:n-p]
        D[p, p : n+p] -= h[0:p+1]
    D[:, n-1] = 0.0  # Fix diagonal for binary case
    return D

def binary_greedy(h_bin, n, deadline, max_sweeps=300):

                return h, c_max
            threshold = D[i] + (c_max - c) / dx
            diffs = D[zeros] - threshold[np.newaxis, :]

                j = candidates[best_local]
                # Apply swap
                h[i] = 0
                h[j] = 1
```

```python
def update_D_after_swap(D, n, i, j):
    """Update D after swapping h[i]: 1→0 and h[j]: 0→1."""
    # h[i]: 1→0: add +1 to D[p, n-1+i-p] for p=0..i (skip p=i since diagonal)
    for p in range(i):
        D[p, n-1+i-p] += 1.0
    for p in range(i+1, n):
        D[p, n-1+p-i] += 1.0
    # h[j]: 0→1: add -1 to D[p, n-1+j-p] for p=0..j (skip p=j)
    for p in range(j):
        D[p, n-1+j-p] -= 1.0
    for p in range(j+1, n):
        D[p, n-1+p-j] -= 1.0
```

This is O(n) per swap.

```python
def binary_greedy(h_bin, n, deadline, max_sweeps=500):
    h = h_bin.copy().astype(float)
    c = compute_all_lags(h, n)
    c_max = float(c.max())
    dx = 2.0 / n
    D = build_D(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        
        for i in ones:
            if time.time() > deadline:
                return h, c_max
            threshold = D[i] + (c_max - c) / dx
            # For each zero j, check if D[j] < threshold element-wise
            # Vectorized: (D[zeros] < threshold).all(axis=1)
            valid = (D[zeros] < threshold[np.newaxis, :]).all(axis=1)
            if valid.any():
                candidates = zeros[valid]
                # Find best (largest decrease in max)
                new_c_array = c[np.newaxis, :] + dx * (D[candidates] - D[i][np.newaxis, :])
                new_maxs = new_c_array.max(axis=1)
                best_local = np.argmin(new_maxs)
                j = candidates[best_local]
                # Apply
                h[i] = 0
                h[j] = 1
                update_D_after_swap(D, n, i, j)
                c = compute_all_lags(h, n)
                c_max = float(c.max())
                improved = True
                break
        
        if not improved:
            break
    
    return h, c_max
```

```python
def round_to_binary_h(h, n, seed):
    """Round h to binary with exactly n/2 ones, using soft rounding."""
    rng = np.random.RandomState(seed)
    # Add small noise to break ties
    h_noisy = h + rng.normal(0, 0.01, n)
    # Sort and pick top n/2
    n_ones = n // 2
    idx = np.argsort(h_noisy)[::-1][:n_ones]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_D(h, n):
    D = np.zeros((n, 2 * n - 1))
    for p in range(n):
        D[p, n - 1 - p: 2 * n - 1 - p] = 1.0 - h[0: n - p]
        D[p, p: n + p] -= h[0: p + 1]
    D[:, n - 1] = 0.0
    return D

def update_D_swap(D, n, i, j):
    for p in range(i):
        D[p, n - 1 + i - p] += 1.0
    for p in range(i + 1, n):
        D[p, n - 1 + p - i] += 1.0
    for p in range(j):
        D[p, n - 1 + j - p] -= 1.0
    for p in range(j + 1, n):
        D[p, n - 1 + p - j] -= 1.0

def round_binary(h, n, seed):
    rng = np.random.RandomState(seed)

    n_ones = n // 2
