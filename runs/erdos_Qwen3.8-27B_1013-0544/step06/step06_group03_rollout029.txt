def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_obj(x, n, alpha, top_m=0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_m > 0:
        idx = np.argsort(c)[-top_m:]
        c = c[idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, top_m=0, maxiter=5000, ftol=1e-16):

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_top(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_full(h, n, alpha, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        return smooth_max_full(x, n, alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_top(h, n, alpha, top_idx, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        return smooth_max_top(x, n, alpha, top_idx)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def aggressive_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(2000):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            improved = False
            # Try all pairs for the worst lag
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-7:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-7:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1.0 - h[i]) * 0.5)
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
            if not improved:
                break
    return h, best_c

def multi_lag_local_search(h, n, deadline, n_top=3):

def binarize_and_refine(h, n, deadline):
    """Push values toward 0/1 then refine."""
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    
    # Also try soft thresholding
    h_soft = np.where(h > 0.7, 1.0, np.where(h < 0.3, 0.0, h))
    h_soft = project_feasible(h_soft, n)
    
    best_h, best_c = h.copy(), compute_c5(h, n)
    
    for candidate in [h_bin, h_soft]:
        c_cand = compute_c5(candidate, n)
        if c_cand < best_c:
            best_h, best_c = candidate, c_cand
        # Refine from this candidate
        if time.time() < deadline:
            h_ref, c_ref = aggressive_local_search(candidate, n, min(deadline, time.time() + 15))
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref
    
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP sweep
    for alpha in [3000, 10000, 30000, 100000, 300000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_full(best_h, n, alpha, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Active set (top lags)
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        for top_m in [5, 10, 20]:
            if time.time() > deadline - 10:
                break
            top_idx = np.argsort(c)[-top_m:]
            for alpha in [30000, 100000, 300000]:
                if time.time() > deadline - 10:
                    break
                h_new = slsqp_top(best_h, n, alpha, top_idx, maxiter=3000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    
    # Phase 3: Binarize and refine
    if time.time() < deadline - 10:
        h_bin, c_bin = binarize_and_refine(best_h, n, min(deadline - 5, time.time() + 20))
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    
    # Phase 4: Aggressive local search
    if time.time() < deadline - 5:
        h_ls, c_ls = aggressive_local_search(best_h, n, min(deadline - 2, time.time() + 25))
        if c_ls < best_c:

    # Phase 5: Multi-lag local search

        h_ml, c_ml = multi_lag_local_search(best_h, n, min(deadline - 1, time.time() + 15), n_top=5)

            best_c = c_ml
            best_h = h_ml
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
