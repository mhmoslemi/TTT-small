    bounds = [(0.0, 1.0)] * n

        h_clip = np.clip(x, 0, 1)
        c = np.correlate(h_clip, 1 - h_clip, mode='full') * dx
        m = c.max()
        if alpha < 5000:
            return m
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def constraint(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                      constraints={'type': 'eq', 'fun': constraint},
                      options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-6})
        return project_feasible(res.x, n)
    except Exception:
        # Try SLSQP as fallback
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
            return project_feasible(res.x, n)
        except Exception:
            return h
