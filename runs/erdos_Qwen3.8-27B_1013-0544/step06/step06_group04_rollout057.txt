import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -2.0, 3.0
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    if f(lo) < 0:
        return np.clip(h, 0.0, 1.0)
    if f(hi) > 0:
        return np.full(n, h.sum() / n)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, n):
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def swap_corr_delta(h, i, j, n):
    """O(n) computation of correlation change when h[i]:0->1, h[j]:1->0."""
    delta = np.zeros(2 * n - 1)
    # For each lag k = -(n-1), ..., (n-1):
    # corr[n-1+k] for k>=0: sum_{t=0}^{n-1-k} h[t]*(1-h[t+k])
    # corr[n-1-k] for k>0:  sum_{t=0}^{n-1-k} h[t+k]*(1-h[t])
    
    # Forward: k = 0, 1, ..., n-1
    ks = np.arange(n)
    # i as first factor: valid if i <= n-1-k, i.e., k <= n-1-i
    mask_i1 = ks <= (n - 1 - i)
    # i as second factor: valid if i >= k and i-k <= n-1-k, i.e., k <= min(i, n-1)
    mask_i2 = ks <= i
    # j as first factor: valid if j <= n-1-k, i.e., k <= n-1-j
    mask_j1 = ks <= (n - 1 - j)
    # j as second factor: valid if j >= k
    mask_j2 = ks <= j
    
    # Change in forward corr[n-1+k]:
    # (1_{i<=n-1-k} - 1_{j<=n-1-k}) - ΔA[k]
    # where ΔA[k] = change in autocorrelation at lag k
    
    # ΔA[k] from i: 
    #   +h'[i+k] if i+k < n and i+k != j, else +0 (if i+k==j, h'[j]=0)
    #   +h'[i-k] if i-k >= 0 and i-k != j, else +0 (if i-k==j, but wait i-k==j means k=i-j)
    # ΔA[k] from j:
    #   -h'[j+k] if j+k < n and j+k != i, else -0 (if j+k==i, h'[i]=1 so change is 1-1=0... 
    #     wait: old = h[j]*h[j+k] = 1*h'[j+k] (if j+k != i) or 1*0=0 (if j+k==i, h[i]=0)
    #     new = h'[j]*h'[j+k] = 0*h'[j+k] = 0
    #     change = 0 - old = -h[j+k] (if j+k != i and valid) or 0 - 0 = 0 (if j+k==i)
    #   Similarly for j-k:
    #     old = h[j-k]*h[j] = h'[j-k]*1 (if j-k != i) or 0*1=0 (if j-k==i, h[i]=0)
    #     new = h'[j-k]*h'[j] = h'[j-k]*0 = 0
    #     change = 0 - old = -h[j-k] (if j-k != i and valid) or 0 (if j-k==i)
    
    # Let me just compute it element by element for simplicity
    # For forward k:
    for k in range(n):
        # i part of ΔA[k]
        di = 0.0
        if i + k < n:
            if i + k == j:
                di += 0.0  # h'[j]=0, so new term = 1*0=0, old = 0*h[j]=0, change=0
            else:

        if i - k >= 0:
            if i - k == j:
                di += 0.0  # h'[j]=0, new=h'[j]*1=0, old=h[j]*0=0, change=0
            else:
                di += h[i - k]  # new=h[i-k]*1, old=h[i-k]*0, change=h[i-k]
        
        # j part of ΔA[k]
        dj = 0.0
        if j + k < n:
            if j + k == i:
                dj += 0.0  # old=1*0=0, new=0*1=0, change=0
            else:
                dj += -h[j + k]  # old=1*h[j+k], new=0, change=-h[j+k]
        if j - k >= 0:
            if j - k == i:
                dj += 0.0  # old=0*1=0, new=1*0=0, change=0
            else:
                dj += -h[j - k]  # old=h[j-k]*1, new=0, change=-h[j-k]
        
        da = di + dj
        # Sum change (linear part)
        si = 1.0 if i <= n - 1 - k else 0.0
        sj = 1.0 if j <= n - 1 - k else 0.0
        delta[n - 1 + k] += (si - sj) - da
    
    # Backward: k = 1, 2, ..., n-1, corr[n-1-k]
    for k in range(1, n):
        # corr[n-1-k] = sum_{t=0}^{n-1-k} h[t+k]*(1-h[t])

        # i appears as t (second factor) if 0 <= i <= n-1-k: h[i+k]*(1-new) - h[i+k]*(1-old)
        # j appears as t+k: new=0, old=1*(1-h[j-k]), change = -(1-h[j-k])
        # j appears as t: h[j+k]*(1-new) - h[j+k]*(1-old) = h[j+k]*1 - 0 = h[j+k]

        # where ΔA[k] is the same as above (autocorrelation at lag k is symmetric)
        
        di = 0.0
        if i + k < n:
            if i + k == j:
                di += 0.0
            else:
                di += h[i + k]
        if i - k >= 0:
            if i - k == j:
                di += 0.0
            else:
                di += h[i - k]
        
        dj = 0.0
        if j + k < n:
            if j + k == i:
                dj += 0.0
            else:
                dj += -h[j + k]
        if j - k >= 0:
            if j - k == i:
                dj += 0.0
            else:
                dj += -h[j - k]
        
        da = di + dj
        si = 1.0 if i >= k else 0.0
        sj = 1.0 if j >= k else 0.0
        delta[n - 1 - k] += (si - sj) - da
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -2.0, 3.0
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    if f(lo) < 0:
        return np.clip(h, 0.0, 1.0)
    if f(hi) > 0:
        return np.full(n, h.sum() / n)
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, n):
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def swap_corr_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    for k in range(1, n):
        di = 0.0
        if i + k < n:
            di += 0.0 if (i + k == j) else h[i + k]
        if i - k >= 0:
            di += 0.0 if (i - k == j) else h[i - k]
        dj = 0.0
        if j + k < n:
            dj += 0.0 if (j + k == i) else (-h[j + k])
        if j - k >= 0:
            dj += 0.0 if (j - k == i) else (-h[j - k])
        da = di + dj
        si = 1.0 if i <= n - 1 - k else 0.0
        sj = 1.0 if j <= n - 1 - k else 0.0
        delta[n - 1 + k] += (si - sj) - da
        bi = 1.0 if i >= k else 0.0
        bj = 1.0 if j >= k else 0.0
        delta[n - 1 - k] += (bi - bj) - da
    return delta

def binary_local_search(h, n, deadline, max_sweeps=5000, seed=42, try_count=30):

    dx = 2.0 / n

    best_c = cur_c

            best_j = -1

def slsqp_smooth(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
