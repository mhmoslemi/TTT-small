import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_v2(x, n, alpha, top_idx=None, var_weight=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        active = c[top_idx]
    else:
        active = c
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = (m + np.log(np.sum(np.exp(shifted - m)))) / alpha
    if var_weight > 0:
        lse += var_weight * np.var(active)
    return float(lse)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, alpha, maxiter, top_idx=None, var_weight=0.0):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_v2, h, args=(n, alpha, top_idx, var_weight),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def adaptive_slsqp(h, n, deadline, top_m=8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    alphas = [200, 1000, 5000, 20000, 80000, 300000, 1000000]

        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        h_new = slsqp_pass(h_curr, n, alpha, maxiter=4000, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new

def cvxpy_outer(h, n, deadline, max_iters=40, top_m=12):
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
                cp.abs(h_var - h_curr) <= trust]
        for k in top_idx:
            lag = k - (n - 1)
            m = abs(lag)
            if m >= n:
                continue
            if lag == 0:
                lin = c[k] + dx * (np.sum((1 - 2*h_curr) * (h_var - h_curr)))
            elif lag > 0:
                grad = np.zeros(n)
                grad[:n-m] -= h_curr[m:]
                grad[m:] += (1.0 - h_curr[:n-m])
                lin = c[k] + dx * (grad @ (h_var - h_curr))
            else:
                grad = np.zeros(n)
                grad[:n-m] += (1.0 - h_curr[m:])
                grad[m:] -= h_curr[:n-m]
                lin = c[k] + dx * (grad @ (h_var - h_curr))
            cons.append(lin <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return best_h, best_c
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

        while improved:

            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))

                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.9, 1.0 - h[j] - 1e-9)

                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.9, 1.0 - h[i] - 1e-9)

                if h[i] > h[j]:
