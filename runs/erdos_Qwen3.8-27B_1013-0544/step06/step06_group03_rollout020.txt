- Move that excess to regions where h(x+k*) = 1 (but h(x) = 0)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif 0 < m < n:
        if lag > 0:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_top_obj(x, n, alpha, top_idx):
    c = compute_all_lags(x, n)
    a = c[top_idx]
    m = a.max()
    s = alpha * (a - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp_run(h, n, alpha, maxiter=2000, top_idx=None):
    def obj(x):
        if top_idx is not None:
            return smooth_top_obj(x, n, alpha, top_idx)
        return smooth_max_obj(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_phase(h, n, deadline, max_iters=500):

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, 2) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.CBC, time_limit=10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=2000)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate'] or h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

        n_cur = len(best_h)
        h_new = slsqp_run(best_h, n_cur, alpha=200000, maxiter=1000)
        c_new = compute_c5(h_new, n_cur)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
