import numpy as np
import time
from scipy.optimize import minimize

_cache = {}


def project_feasible(h, target=None):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    d = target - float(h.sum())
    if abs(d) > 1e-13:
        if d > 0.0:
            r = float((1.0 - h).sum())
            if r > 1e-14:
                h = h + (d / r) * (1.0 - h)
        else:
            s = float(h.sum())
            if s > 1e-14:
                h = h + (d / s) * h
    d = target - float(h.sum())
    if abs(d) > 1e-11:
        if d > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size:
            h[idx] += d / idx.size
    return np.clip(h, 0.0, 1.0)


def compute_f(h):
    n = len(h)
    om = 1.0 - h
    fp = np.empty(n, dtype=float)
    fm = np.empty(max(0, n - 1), dtype=float)
    fp[0] = float(np.dot(h, om))
    for d in range(1, n):
        L = n - d
        fp[d] = float(np.dot(h[:L], om[d:]))
        fm[d - 1] = float(np.dot(h[d:], om[:L]))
    return fp, fm


def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    if n <= 1:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def max_f(fp, fm):
    return float(fp.max() if fm.size == 0 else max(fp.max(), fm.max()))


def grad_lag(h, d, positive):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if d == 0:
        return 1.0 - 2.0 * h
    L = n - d
    if positive:
        g[:L] += 1.0 - h[d:]
        g[d:] -= h[:L]
    else:
        g[d:] += 1.0 - h[:L]
        g[:L] -= h[d:]
    return g


def grad_weighted(h, wp, wm):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if wp.size and abs(wp[0]) > 1e-12:
        g += wp[0] * (1.0 - 2.0 * h)
    for d in range(1, n):
        w = wp[d]
        if abs(w) > 1e-10:
            L = n - d
            g[:L] += w * (1.0 - h[d:])
            g[d:] -= w * h[:L]
        w = wm[d - 1]
        if abs(w) > 1e-10:
            L = n - d
            g[d:] += w * (1.0 - h[:L])
            g[:L] -= w * h[d:]
    return g


def _prepare(x, beta):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    old_beta = _cache.get("beta")
    old_h = _cache.get("h")
    if old_beta == beta and old_h is not None and np.array_equal(old_h, h):
        return h, _cache["fp"], _cache["fm"], _cache["wp"], _cache["wm"], _cache["m"], _cache["ls"]
    fp, fm = compute_f(h)
    vals = np.concatenate((fp, fm))
    m = float(vals.max())
    a = beta * (vals - m)
    a -= a.max()
    z = np.exp(a)
    zsum = float(z.sum())
    w = z / max(zsum, 1e-300)
    ls = float(np.log(max(zsum, 1e-300)))
    _cache["h"] = h.copy()
    _cache["beta"] = beta
    _cache["fp"] = fp
    _cache["fm"] = fm
    _cache["wp"] = w[:fp.size]
    _cache["wm"] = w[fp.size:]
    _cache["m"] = m
    _cache["ls"] = ls
    return h, fp, fm, w[:fp.size], w[fp.size:], m, ls


def pen_objective(x, beta, mu):
    h, fp, fm, wp, wm, m, ls = _prepare(x, beta)
    s = float(h.sum()) - 0.5 * h.size
    return float(m + ls / beta + mu * s * s)


def pen_jacobian(x, beta, mu):
    h, fp, fm, wp, wm, m, ls = _prepare(x, beta)
    g = grad_weighted(h, wp, wm)
    s = float(h.sum()) - 0.5 * h.size
    return g + (2.0 * mu * s)


def run_lbfgsb(h0, beta, mu, maxiter):
    h0 = project_feasible(h0)
    bounds = [(0.0, 1.0) for _ in range(len(h0))]
    res = minimize(
        pen_objective,
        h0,
        method="L-BFGS-B",
        jac=pen_jacobian,
        args=(beta, mu),
        bounds=bounds,
        options={
            "maxiter": int(maxiter),
            "maxfun": int(max(100, 3 * maxiter)),
            "ftol": 1e-12,
            "gtol": 1e-10,
            "disp": 0
        }
    )
    return project_feasible(np.asarray(res.x, dtype=float))


def binary_from(h):
    n = len(h)
    target = 0.5 * n
    whole = int(np.floor(target))
    rem = target - whole
    b = np.zeros(n, dtype=float)
    if whole > 0:
        b[np.argsort(h)[::-1][:whole]] = 1.0
    if rem > 1e-12 and whole < n:
        b[np.argsort(h)[::-1][whole]] = rem
    return project_feasible(b)


def thue_morse_init(n, rng):
    b = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        b[i] = b[i >> 1] ^ (i & 1)
    t = b.astype(float)
    target = 0.5 * n
    s = float(t.sum())
    if s > target + 1e-12:
        ones = np.where(t == 1.0)[0]
        excess = int(np.rint(s - target))
        if ones.size and excess > 0:
            idx = rng.choice(ones.size, min(excess, ones.size), replace=False)
            t[ones[idx]] = 0.0
    elif s < target - 1e-12:
        zeros = np.where(t == 0.0)[0]
        need = int(np.rint(target - s))
        if zeros.size and need > 0:
            idx = rng.choice(zeros.size, min(need, zeros.size), replace=False)
            t[zeros[idx]] = 1.0
    d = target - float(t.sum())
    if abs(d) > 1e-8:
        if d > 0:
            idx = np.where(t < 1.0)[0]
            if idx.size:
                t[idx[0]] = min(1.0, t[idx[0]] + d)
        else:
            idx = np.where(t > 0.0)[0]
            if idx.size:
                t[idx[0]] = max(0.0, t[idx[0]] + d)
    return project_feasible(t)


def random_binary(n, rng):
    b = np.zeros(n, dtype=float)
    target = int(n // 2)
    if target > 0:
        b[rng.choice(n, target, replace=False)] = 1.0
    return project_feasible(b)


def pair_coeff(h, i, j):
    n = len(h)
    Aplus = np.empty(n, dtype=float)
    Aminus = np.empty(max(0, n - 1), dtype=float)
    Aplus[0] = 2.0 * (h[j] - h[i])
    for d in range(1, n):
        L = n - d
        gi = 0.0
        if i + d < n:
            gi += 1.0 - h[i + d]
        if i - d >= 0:
            gi -= h[i - d]
        gj = 0.0
        if j + d < n:
            gj += 1.0 - h[j + d]
        if j - d >= 0:
            gj -= h[j - d]
        Aplus[d] = gi - gj

        gim = 0.0
        if i >= d:
            gim += 1.0 - h[i - d]
        if i < L:
            gim -= h[i + d]
        gjm = 0.0
        if j >= d:
            gjm += 1.0 - h[j - d]
        if j < L:
            gjm -= h[j + d]
        Aminus[d - 1] = gim - gjm
    return Aplus, Aminus


def line_search_pair(h, fp, fm, i, j, cap):
    if cap <= 1e-12:
        return 0.0, max_f(fp, fm)
    Aplus, Aminus = pair_coeff(h, i, j)
    d0 = abs(i - j)
    small = min(cap, 0.25)
    d1 = np.linspace(0.0, cap, 44)
    d2 = np.linspace(0.0, small, 88)
    deltas = d1 if small >= cap * 0.95 else np.unique(np.concatenate((d1, d2)))
    D = deltas[:, None]
    DD = D * D
    p = fp[None, :] + Aplus[None, :] * D
    p[:, 0] -= 2.0 * DD
    if d0 > 0:
        p[:, d0] += DD
    if fm.size:
        m = fm[None, :] + Aminus[None, :] * D
        if d0 > 0:
            m[:, d0 - 1] += DD
        vals = np.maximum(p.max(axis=1), m.max(axis=1))
    else:
        vals = p.max(axis=1)
    b = int(np.argmin(vals))
    return float(deltas[b]), float(vals[b])


def local_exchange_search(h, deadline, seed=42, top_m=6, max_kick=80):
    n = len(h)
    hcur = project_feasible(h)
    fp, fm = compute_f(hcur)
    cur_f = max_f(fp, fm)
    best_h = hcur.copy()
    best_f = cur_f
    rng = np.random.RandomState(seed)
    no = 0
    kick = 0
    temp = 0.05
    k = max(1, min(n // 2, max(20, n // 4)))

    while time.time() < deadline:
        vals = np.concatenate((fp, fm))
        am = min(top_m, vals.size)
        acts = []
        for aid in np.argsort(vals)[-am:]:
            if aid < n:
                d = int(aid)
                fval = float(fp[d])
                g = grad_lag(hcur, d, True)
            else:
                d = int(aid - n + 1)
                fval = float(fm[d - 1])
                g = grad_lag(hcur, d, False)
            acts.append((d, fval, g))

        order = np.argsort(hcur)
        lo = order[:k]
        hi = order[-k:]
        ext = np.unique(np.concatenate((lo, hi)))
        R = ext
        D = ext
        caps = np.minimum(hcur[D][None, :], 1.0 - hcur[R][:, None])
        same = R[:, None] == D[None, :]
        caps[same] = 0.0
        if caps.size == 0 or float(caps.max()) < 1e-8:
            break
        pc = caps[caps > 1e-8]
        if pc.size == 0:
            break
        med = float(np.median(pc))
        if no > 50:
            delta_sel = min(1.0, med)
        else:
            delta_sel = min(0.05, 0.5 * med)
        if delta_sel < 1e-6:
            delta_sel = min(0.05, float(pc.max()) * 0.5)

        eff = np.minimum(delta_sel, caps)
        dmat = np.abs(R[:, None] - D[None, :])
        eff2 = eff * eff

        d0, f0, g0 = acts[0]
        approx = f0 + eff * (g0[R][:, None] - g0[D][None, :])
        if d0 == 0:
            approx -= 2.0 * eff2
        else:
            approx += eff2 * (dmat == d0)

        for dd, ff, gg in acts[1:]:
            tv = ff + eff * (gg[R][:, None] - gg[D][None, :])
            if dd == 0:
                tv -= 2.0 * eff2
            else:
                tv += eff2 * (dmat == dd)
            np.maximum(approx, tv, out=approx)

        if no > 30:
            approx += rng.normal(0.0, 0.1, caps.shape)
        approx = np.where(caps > 1e-8, approx, 1e100)

        flat = int(np.argmin(approx))
        ii, jj = divmod(flat, len(D))
        i = int(R[ii])
        j = int(D[jj])
        cap = float(caps[ii, jj])
        if cap < 1e-8:
            break

        bd, bf = line_search_pair(hcur, fp, fm, i, j, cap)
        if bf < cur_f - 1e-10:
            hcur[i] = min(1.0, hcur[i] + bd)
            hcur[j] = max(0.0, hcur[j] - bd)
            if abs(float(hcur.sum()) - 0.5 * n) > 1e-8:
                hcur = project_feasible(hcur)
            fp, fm = compute_f(hcur)
            cur_f = max_f(fp, fm)
            if cur_f < best_f:
                best_f = cur_f
                best_h = hcur.copy()
            no = 0
        else:
            no += 1
            if no > 80 and kick < max_kick:
                ri = int(rng.randint(len(R)))
                rj = int(rng.randint(len(D)))
                ci = float(caps[ri, rj])
                if ci > 1e-6:
                    kd, kf = line_search_pair(hcur, fp, fm, int(R[ri]), int(D[rj]), ci)
                    if kf <= cur_f + 0.25:
                        prob = np.exp(-(kf - cur_f) / max(temp, 1e-3))
                        if kf < cur_f - 1e-10 or rng.rand() < prob:
                            hcur[R[ri]] = min(1.0, hcur[R[ri]] + kd)
                            hcur[D[rj]] = max(0.0, hcur[D[rj]] - kd)
                            if abs(float(hcur.sum()) - 0.5 * n) > 1e-8:
                                hcur = project_feasible(hcur)
                            fp, fm = compute_f(hcur)
                            cur_f = max_f(fp, fm)
                            if cur_f < best_f:
                                best_f = cur_f
                                best_h = hcur.copy()
                            temp = max(0.005, temp * 0.85)
                            kick += 1
                            no = 20
        if no > 4000:
            break

    return best_h, compute_c5(best_h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    eff = max(60.0, total - 20.0)
    dl = t0 + eff
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = random_binary(n, rng)

    best_h = project_feasible(h0)
    best_c = compute_c5(best_h, n)

    if eff > 100.0:
        p1 = t0 + 0.08 * eff
        try:
            hs, cs = local_exchange_search(best_h, p1, seed, 6, 40)
        except Exception:
            hs, cs = best_h, best_c
        if cs < best_c:
            best_h, best_c = hs, cs

    h_bin = binary_from(best_h)
    h_tm = thue_morse_init(n, rng)
    h_rand = random_binary(n, rng)
    alt = (h_bin, h_tm, h_rand)

    pmain = t0 + 0.72 * eff
    if eff > 120.0:
        for beta in (4.0, 2.0, 6.0, 8.0, 1.5, 0.8, 3.0):
            for sig in (0.0, 0.02, 0.05):
                if time.time() > pmain - 25.0:
                    break
                iters = 220 if beta <= 4.0 else 140
                if sig > 0.0:
                    hstart = project_feasible(best_h + sig * rng.normal(0.0, 1.0, n))
                else:
                    hstart = project_feasible(best_h)
                try:
                    hr = run_lbfgsb(hstart, beta, 50.0, iters)
                except Exception:
                    hr = hstart
                c = compute_c5(hr, n)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = hr.copy()

        if time.time() < pmain - 40.0:
            for base in alt:
                for beta in (4.0, 2.0):
                    if time.time() > pmain - 25.0:
                        break
                    try:
                        hr = run_lbfgsb(project_feasible(base), beta, 50.0, 200)
                    except Exception:
                        hr = base
                    c = compute_c5(hr, n)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_h = hr.copy()

                    if time.time() > pmain - 25.0:
                        break
                    hstart = project_feasible(base + 0.03 * rng.normal(0.0, 1.0, n))
                    try:
                        hr = run_lbfgsb(hstart, beta, 50.0, 150)
                    except Exception:
                        hr = hstart
                    c = compute_c5(hr, n)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_h = hr.copy()

    if time.time() < dl - 20.0:
        try:
            hs, cs = local_exchange_search(best_h, dl - 10.0, seed + 7, 6, 80)
        except Exception:
            hs, cs = best_h, best_c
        if cs < best_c:
            best_h, best_c = hs, cs

    if time.time() < dl - 15.0:
        try:
            hr = run_lbfgsb(best_h, 4.0, 100.0, 80)
        except Exception:
            hr = best_h
        hr = project_feasible(hr)
        c = compute_c5(hr, n)
        if c < best_c:
            best_h, best_c = hr, c

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
