def smooth_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    g = corr * dx
    a = alpha * g
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    sum_exp = exp_a.sum()
    obj = (a_max + np.log(sum_exp)) / alpha
    sum_dev = x.sum() - n / 2.0
    obj += penalty * sum_dev * sum_dev
    return obj

def smooth_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    g = corr * dx
    a = alpha * g
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    sum_exp = exp_a.sum()
    w = exp_a / sum_exp  # soft-max weights
    
    # Gradient of smooth max:
    # dC/dx[i] = dx * sum_m w[m] * [(1-x[i-k])*[0<=i-k<n] - x[i+k]*[0<=i+k<n]]
    # where k = m - (n-1)
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    k = m_idx - (n - 1)
    
    idx1 = i_idx - k
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - x[np.clip(idx1, 0, n-1)], 0.0)
    
    idx2 = i_idx + k
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, x[np.clip(idx2, 0, n-1)], 0.0)
    
    grad = dx * ((term1 @ w) - (term2 @ w))
    
    # Penalty gradient
    sum_dev = x.sum() - n / 2.0
    grad += 2.0 * penalty * sum_dev
    
    return grad
