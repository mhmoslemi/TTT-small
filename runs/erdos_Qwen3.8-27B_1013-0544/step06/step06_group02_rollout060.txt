            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(h, n, alpha):
    c = c5_full(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def sm_grad(h, n, alpha):
    c = c5_full(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-12: continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

            c2 = compute_c5(h2, n)
            if c2 < best:

        except Exception:
