```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def adam_optimize(h, n, deadline, alpha_init=50.0, alpha_end=50000.0, 
                  lr_init=0.01, n_anneal=500):
    """Adam optimizer with smooth max annealing."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    total_iters = max(100, int((deadline - time.time()) / 0.02))
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        t += 1
        # Anneal alpha
        frac = min(1.0, it / n_anneal)
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        
        g = grad_smooth_max(h, n, alpha)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr_init * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def binary_push(h, n, iters=100, temp=0.1):
    """Push values toward 0 or 1 while maintaining feasibility."""
    h = h.copy()
    for _ in range(iters):
        # Compute "energy" of being in the middle
        mid_penalty = h * (1 - h)  # High when h is near 0.5
        # Push toward nearest boundary
        # Values > 0.5 push up, values < 0.5 push down
        delta = temp * (2 * h - 1)  # Push larger values up, smaller down
        # But we need to maintain sum constraint
```

```python
def proj_box_sum(y, n):
    """Project y onto {0 <= h <= 1, sum(h) = n/2} exactly."""
    target = n / 2.0
    # Binary search for lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute the remaining difference
        # Find indices where 0 < h < 1
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h
```

```python
    return proj_box_sum(h, n / 2.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def grad_top_k(h, n, topk=5):
    """Subgradient using top-k maximizing shifts."""
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    # Get top-k shifts
    idx = np.argsort(c)[-topk:]
    weights = c[idx] - c.min() + 1e-6
    weights /= weights.sum()
    grad = np.zeros(n)
    for wi, m in zip(weights, idx):
        k = int(m) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def adam_opt(h, n, deadline, alpha=1000.0, lr=0.01, topk=None):
    """Adam optimizer with exact projection."""
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        if topk is not None:
            g = grad_top_k(h, n, topk)
        else:
            g = grad_smooth_max(h, n, alpha)
        
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        
        lr_t = lr * np.sqrt(1 - b2 ** t) / (1 - b1 ** t)
        h_new = proj_box_sum(h - lr_t * m_hat / (np.sqrt(v_hat) + eps), target)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        # Anneal lr
        lr *= 0.9999
        if lr < 1e-6:
            break
    
    return best_h, best_c

def grad_descent_project(h, n, deadline, alpha=1000.0, lr=0.005):
    """Simple projected gradient with exact projection."""
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm  # Normalize
        
        # Backtracking
        step = lr
        for _ in range(50):
            h_new = proj_box_sum(h - step * g * g_norm, target)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                break
            step *= 0.5
        
        if step < 1e-10:
            break
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    target = n / 2.0
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius if radius else n))
                hi = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi):

                        h[i] += d; h[j] -= d

                        else:
```

Let me think about a different approach: **Coordinate descent with smart ordering**.

For each coordinate i, the optimal value of h[i] (given all other h[j], j≠i) can be found by:
- The objective is piecewise linear in h[i] (since each term h[i]*(1-h[i+k]) is quadratic in h[i])

- If i is the "second" index (i-k, i), then h[i] appears as h[i-k]*(1-h[i])

For shift k ≥ 0:
c[k] = sum_{j=0}^{n-k-1} h[j] * (1-h[j+k])

If j = i: term is h[i] * (1-h[i+k])
If j+k = i (i.e., j = i-k): term is h[i-k] * (1-h[i])

So ∂c[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with bounds checking)

This is linear in h[i]! So c[k] as a function of h[i] (with other h fixed) is:
c[k] = h[i] * (1-h[i+k]) + h[i-k] * (1-h[i]) + (terms not involving h[i])
     = h[i] - h[i]*h[i+k] + h[i-k] - h[i]*h[i-k] + const
     = h[i] * (1 - h[i+k] - h[i-k]) + h[i-k] + const

So c[k] is linear in h[i] when other variables are fixed! This means the max over k is a piecewise linear function of h[i], and we can find the optimal h[i] by checking where the max changes.

This gives us an exact coordinate descent step! For each coordinate i:
1. Compute the current c[k] for all k
2. For each k, compute the slope and intercept of c[k] as a function of h[i]
3. The max is a piecewise linear function of h[i]
4. Find the h[i] that minimizes the max (this is a 1D convex minimization)
5. Update h[i]

This is exact and guaranteed to not increase the objective! It's like a "best response" in a game.

```python
def coord_descent_step(h, n, i):
    """Find optimal h[i] given all other coordinates fixed."""
    # For each shift k, compute c[k] as a function of h[i]
    # c[k] = A_k * h[i] + B_k where A_k and B_k don't depend on h[i]
    # Actually, c[k] is linear in h[i] as shown above
    
    # Let's compute the slope and intercept for each k
    dx = 2.0 / n
    h_i = h[i]
    h_with_zero = h.copy()
    h_with_zero[i] = 0.0
    h_with_one = h.copy()
    h_with_one[i] = 1.0
    
    c_zero = np.correlate(h_with_zero, 1 - h_with_zero, mode='full') * dx
    c_one = np.correlate(h_with_one, 1 - h_with_one, mode='full') * dx
    
    # c[k] = c_zero[k] + t * (c_one[k] - c_zero[k]) where t = h[i]
    slope = c_one - c_zero  # This is the coefficient of h[i]
    intercept = c_zero  # This is c[k] when h[i] = 0
    
    # We want to find t ∈ [0,1] that minimizes max_k (intercept[k] + t * slope[k])
    # This is a 1D problem: minimize the upper envelope of lines
    
    # The maximum of lines is a convex piecewise linear function
    # The minimum is where the subgradient contains 0
    # i.e., where the positive slopes sum equals the negative slopes sum in some sense
    
    # Simple approach: ternary search or bisection on the subgradient
    # Subgradient of max_k(f_k(t)) at t is the convex hull of {slope[k] : f_k(t) = max}
    
    # Let's use a simple approach: evaluate at many points and find the min
    # Or better: use the fact that the optimal t is where the "active set" changes
    
    # Quick approach: evaluate at a grid and refine
    ts = np.linspace(0, 1, 200)
    vals = np.max(intercept[:, None] + ts[None, :] * slope[:, None], axis=0)
    t_best = ts[np.argmin(vals)]
    
    # Refine with finer grid around t_best
    lo = max(0, t_best - 0.01)
    hi = min(1, t_best + 0.01)
    ts_fine = np.linspace(lo, hi, 200)
    vals_fine = np.max(intercept[:, None] + ts_fine[None, :] * slope[:, None], axis=0)
    t_best = ts_fine[np.argmin(vals_fine)]
    
    return t_best
```

For shift k ≥ 0:
c[k] = sum_{j=0}^{n-k-1} h[j] * (1-h[j+k])

The terms involving h[i]:

- From second term: -h[i-k] if i ≥ k, else 0

c_{-k'} = sum_{j=k'}^{n-1} h[j] * (1-h[j-k'])

Terms involving h[i]:

1. Finding all intersection points of the lines
2. Evaluating the max at each intersection point
3. Taking the minimum

There are at most n-1 intersection points (for 2n-1 lines), so this is O(n²) in the worst case.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def proj_box_sum(y, target):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def coord_step(h, n, i, target):
    hi_old = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    h[i] = 1.0
    c1 = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    h[i] = hi_old
    slope = c1 - c0
    intercept = c0 - hi_old * slope
    ts = np.linspace(0, 1, 80)
    vals = np.max(intercept[:, None] + ts[None, :] * slope[:, None], axis=0)
    t_best = ts[np.argmin(vals)]
    lo = max(0.0, t_best - 0.0125)
    hi_b = min(1.0, t_best + 0.0125)
    ts2 = np.linspace(lo, hi_b, 80)
    vals2 = np.max(intercept[:, None] + ts2[None, :] * slope[:, None], axis=0)
    t_best = ts2[np.argmin(vals2)]
    return t_best

def coord_descent(h, n, deadline, max_passes=50):
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            t_opt = coord_step(h, n, i, target)
            if abs(t_opt - h[i]) > 1e-7:
                h[i] = t_opt
                h = proj_box_sum(h, target)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    improved_any = True
        if not improved_any:
            break
    return best_h, best_c

def adam_opt(h, n, deadline, alpha=1000.0, lr=0.01):
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    curr_lr = lr
    while time.time() < deadline:
        t += 1
        g = grad_smooth_max(h, n, alpha)
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        lr_t = curr_lr * np.sqrt(1 - b2 ** t) / (1 - b1 ** t)
        h_new = proj_box_sum(h - lr_t * m_hat / (np.sqrt(v_hat) + eps), target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        curr_lr *= 0.99995
        if t > 500000 or curr_lr < 1e-7:
            break
    return best_h, best_c

def fista(h, n, deadline, alpha=1000.0):
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    z = h.copy()
    t_k = 1.0
    step_size = n / (32.0 * alpha)
    step_size = min(step_size, 0.05)
    while time.time() < deadline:
        g = grad_smooth_max(z, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-13:
            break
        x_new = proj_box_sum(z - step_size * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = proj_box_sum(z, target)
        x = x_new
        t_k = t_new
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - (radius if radius else n))
                hi_b = min(n, i + (radius if radius else n) + 1)
                for j in range(lo, hi_b):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
```
