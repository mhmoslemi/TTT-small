def project_feasible(h, target_sum):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_c5(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (c - c.max())
    smax = shifted.max()
    return (c.max() + (np.exp(shifted - smax).sum()) * dx) / (1.0 + 0) * 0 + (c.max() + (np.log(np.exp(shifted - smax).sum()) + smax) / alpha)

def compute_gradient_smoothed(h, n, alpha):
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (result - result.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    idx = np.arange(n)[:, None]

    mask2 = (imm >= 0) & (imm < n)
    imm_clipped = np.clip(imm, 0, n - 1)
    term2 = np.where(mask2, h[imm_clipped], 0.0)
    J = (term1 - term2) * dx
    grad = J @ weights
    return grad

        s = s - s.max()
        return m + np.log(np.exp(s).sum()) / alpha

        s2 = s - s.max()
        w = np.exp(s2)

        return compute_gradient_smoothed(x, n, alpha)
