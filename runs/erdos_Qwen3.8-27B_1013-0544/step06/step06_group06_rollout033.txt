```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    target = n // 2
    # Sort by value, keep top target as 1
    idx = np.argsort(h)[::-1]
    h_new = np.zeros(n)
    h_new[idx[:target]] = 1.0
    return h_new

def compute_all_deltas(h, n):
    """Compute delta vector for flipping each position"""
    corr_base = np.correlate(h, 1 - h, mode='full').astype(float)
    deltas = np.zeros((n, 2 * n - 1))
    for p in range(n):
        h_flip = h.copy()
        h_flip[p] = 1.0 - h_flip[p]
        corr_flip = np.correlate(h_flip, 1 - h_flip, mode='full').astype(float)
        deltas[p] = corr_flip - corr_base
    return corr_base, deltas

def best_single_swap(corr, deltas, zeros, ones):
    """Find the best single swap (z, o) that minimizes max(corr + delta[z] + delta[o])"""
    d_z = deltas[zeros]  # (num_zeros, 2n-1)
    d_o = deltas[ones]   # (num_ones, 2n-1)
    # combined[z, o, k] = corr[k] + d_z[z, k] + d_o[o, k]
    # To save memory, do it in chunks
    best_val = np.inf
    best_zi = 0
    best_oi = 0
    chunk_size = max(1, 2000000 // (len(ones) * (2*n-1)))  # limit to ~16MB
    for z_start in range(0, len(zeros), chunk_size):
        z_end = min(z_start + chunk_size, len(zeros))
        combined = corr[None, None, :] + d_z[z_start:z_end, None, :] + d_o[None, :, :]
        maxes = combined.max(axis=2)
        flat = maxes.argmin()
        zi = flat // len(ones)
        oi = flat % len(ones)
        val = maxes[zi, oi]
        if val < best_val:
            best_val = val
            best_zi = z_start + zi
            best_oi = oi
    return best_val, zeros[best_zi], ones[best_oi]

def binary_local_search(h, n, deadline, seed=42):
    """Main binary swap local search with annealing kicks"""
    rng = np.random.RandomState(seed)
    h = project_binary(h, n)
    corr, deltas = compute_all_deltas(h, n)
    best_c = float(np.max(corr) * (2.0/n))
    best_h = h.copy()
    temp = 0.01
    kick_count = 0
    no_improve = 0
    stall = 0
    dx = 2.0 / n
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_val, z_best, o_best = best_single_swap(corr, deltas, zeros, ones)
        new_c = best_val * dx
        
        if new_c < best_c - 1e-15:
            # Make the swap
            h[z_best] = 1.0
            h[o_best] = 0.0
            # Recompute corr and deltas
            corr, deltas = compute_all_deltas(h, n)
            cur_c = float(np.max(corr) * dx)
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
            stall = 0
            temp = max(0.002, temp * 0.999)
        else:
            # No improving swap found
            stall += 1
            no_improve += 1
            
            if stall > 50 or (no_improve > 500 and kick_count < 50):
                kick_count += 1
                # Random kick: flip k random pairs
                k = rng.randint(1, 4)
                for _ in range(k):
                    zi = rng.choice(len(zeros))
                    oi = rng.choice(len(ones))
                    h[zeros[zi]] = 1.0
                    h[ones[oi]] = 0.0
                    # Update zeros and ones
                    zeros = np.where(h == 0)[0]
                    ones = np.where(h == 1)[0]
                
                corr, deltas = compute_all_deltas(h, n)
                cur_c = float(np.max(corr) * dx)
                
                # Accept with probability
                if cur_c < best_c or rng.rand() < np.exp(-(cur_c - best_c) / temp):
                    pass  # keep the kicked state
                else:
                    h = best_h.copy()
                    corr, deltas = compute_all_deltas(h, n)
                    cur_c = best_c
                    zeros = np.where(h == 0)[0]
                    ones = np.where(h == 1)[0]
                
                temp = max(0.001, temp * 0.8)
                stall = 0
    
    return best_h, best_c
```

```python
def lse_objective_and_grad(h, n, alpha):
    """Compute LSE objective and its gradient"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx  # shape (2n-1,)
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()  # should be 0
    exps = np.exp(shifted - sm)
    weights = exps / exps.sum()  # shape (2n-1,)
    obj = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if weights[r] < 1e-12:
            continue
        lag = r - (n - 1)
        g_k = np.zeros(n)
        j = np.arange(n)
        vf = (j + lag >= 0) & (j + lag < n)
        vb = (j - lag >= 0) & (j - lag < n)
        g_k[vf] = dx * (1 - h[j[vf] + lag])
        g_k[vb] -= dx * h[j[vb] - lag]
        grad += weights[r] * g_k
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_new = np.zeros(n)
    h_new[idx[:target]] = 1.0
    return h_new

def compute_deltas(h, n):
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    deltas = np.empty((n, 2 * n - 1), dtype=np.float64)
    for p in range(n):
        h2 = h.copy()
        h2[p] = 1.0 - h2[p]
        deltas[p] = np.correlate(h2, 1 - h2, mode='full')
    deltas -= corr  # broadcast
    return corr, deltas

def find_best_swap(corr, deltas, zeros, ones, n):
    d_z = deltas[zeros]
    d_o = deltas[ones]
    nz, no, L = len(zeros), len(ones), 2 * n - 1
    best_val = np.inf
    best_zi, best_oi = 0, 0
    chunk = max(1, 5000000 // (no * L))
    for cs in range(0, nz, chunk):
        ce = min(cs + chunk, nz)
        combined = corr[None, None, :] + d_z[cs:ce, None, :] + d_o[None, :, :]
        maxes = combined.max(axis=2)
        flat = int(maxes.argmin())
        zi, oi = flat // no, flat % no
        if maxes[zi, oi] < best_val:
            best_val = maxes[zi, oi]
            best_zi, best_oi = cs + zi, oi
    return best_val, zeros[best_zi], ones[best_oi]

def thue_morse(n, seed=0):
    rng = np.random.RandomState(seed)
    t = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    target = n // 2
    cur = int(t.sum())
    if cur > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        t[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        t[idx[:target - cur]] = 1
    return t

def binary_search(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, n)
    corr, deltas = compute_deltas(h, n)
    dx = 2.0 / n
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    temp = 0.02
    stall = 0
    no_improve = 0
    kick_count = 0
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_val, z_b, o_b = find_best_swap(corr, deltas, zeros, ones, n)
        
        if best_val < corr.max() - 1e-15:
            h[z_b] = 1.0
            h[o_b] = 0.0
            corr, deltas = compute_deltas(h, n)
            cur_c = float(corr.max() * dx)
            stall = 0
            temp = max(0.003, temp * 0.9995)
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 80 or (no_improve > 500 and kick_count < 200):
                kick_count += 1
                k = int(rng.randint(1, 6))
                for _ in range(k):
                    zeros = np.where(h == 0)[0]
                    ones = np.where(h == 1)[0]
                    if len(zeros) == 0 or len(ones) == 0:
                        break
                    h[zeros[int(rng.randint(len(zeros)))]] = 1.0
                    h[ones[int(rng.randint(len(ones)))]] = 0.0
                corr, deltas = compute_deltas(h, n)
                kick_c = float(corr.max() * dx)

                    cur_c = kick_c

                    corr, deltas = compute_deltas(h, n)
                    cur_c = best_c
                temp = max(0.002, temp * 0.85)
                stall = 0

def slsqp_refine(h_init, n, deadline):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in [5000, 20000, 100000, 500000]:
        if time.time() > deadline - 15:
            break
        
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full') * dx
            m = corr.max()
            sh = alpha * (corr - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def obj_grad(x):
            corr = np.correlate(x, 1 - x, mode='full') * dx
            m = corr.max()
            sh = alpha * (corr - m)
            sm = sh.max()
            exps = np.exp(sh - sm)
            w = exps / exps.sum()
            grad = np.zeros(n)
            for r in range(2 * n - 1):
                if w[r] < 1e-12:
                    continue
                lag = r - (n - 1)
                j = np.arange(n)
                vf = (j + lag >= 0) & (j + lag < n)
                vb = (j - lag >= 0) & (j - lag < n)
                gk = np.zeros(n)
                gk[vf] = dx * (1 - x[j[vf] + lag])
                gk[vb] -= dx * x[j[vb] - lag]
                grad += w[r] * gk
            return grad
        
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None:
                h_new = np.clip(res.x, 0, 1)
                # Fix sum
                diff = n / 2.0 - h_new.sum()
                if diff > 0:
                    room = 1.0 - h_new
                    s = room.sum()
                    if s > 1e-10: h_new += diff / s * room
                elif diff < 0:
                    s = h_new.sum()
                    if s > 1e-10: h_new += diff / s * h_new
                h_new = np.clip(h_new, 0, 1)
                c = compute_c5(h_new, n)
                if c < best_c:
                    best_c = c
                    best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
```
