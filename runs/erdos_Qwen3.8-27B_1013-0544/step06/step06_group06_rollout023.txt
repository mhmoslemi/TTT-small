So d(corr_k)/d(h_j) = (1-h_{j+k}) - h_{j-k} (where terms with out-of-range indices are 0).

For the soft-max: F(h) = (1/alpha) * log(sum_k exp(alpha * corr_k(h)))
dF/dh_j = (1/alpha) * sum_k exp(alpha*corr_k) * d(corr_k)/dh_j / sum_k exp(alpha*corr_k)
         = sum_k p_k * d(corr_k)/dh_j
where p_k = exp(alpha*corr_k) / sum_m exp(alpha*corr_m)

This is a weighted average of the gradients, with weights concentrated on the "active" (maximum) correlations.

Then I do a projected gradient step:
h_new = h - lr * dF/dh
h_new = project(h_new) [clip to [0,1] and adjust sum]

The projection onto {h: 0≤h≤1, sum h = n/2} can be done efficiently.

```python
    # h[p] appears as (1-h[i]) when i = p: contributes -h[p+shift]

    # grad[p] = sum_k w_k * [(1-h[p+shift_k])_valid - h[p-shift_k]_valid]
    # where shift_k = k - (n-1) and w_k = p_k (softmax weights)

    #   = sum over k where -p <= shift_k < n-p
    #   = sum over k where n-1-p <= k <= n-1+n-1-p = 2n-2-p
    #   of w[k] * (1 - h[p + k - (n-1)])

    shifts = np.arange(2*n-1) - (n-1)  # -[n-1], ..., n-1

    weights /= weights.sum()
    
    grad = np.zeros(n)
    for p in range(n):

        valid1 = (s1 >= 0) & (s1 < n)

        s2 = p - shifts
        valid2 = (s2 >= 0) & (s2 < n)
```

Now, for the optimization, I'll use:
- Adaptive learning rate
- Momentum
- Periodic restart with perturbation

Let me also think about a key insight: at the optimum, the solution will have most h values at 0 or 1 (complementarity), with only a few "fractional" values where the gradient is balanced. This is similar to the KKT conditions for an LP.

So my strategy will be:
1. Start from a good binary solution
2. Run projected gradient with soft-max to find the continuous optimum
3. Periodically "round" the solution back to binary and re-optimize

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_grad(h, n, alpha):
    """Compute gradient of (1/alpha)*log(sum exp(alpha*corr_k))"""
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    max_corr = corr.max()
    shifted = corr - max_corr
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for p in range(n):
        # grad[p] = sum_k w[k]*(1-h[p+k-(n-1)])_valid - sum_k w[k]*h[p-k+(n-1)]_valid
        lo1 = max(0, n - 1 - p)
        hi1 = min(2 * n - 1, 2 * n - 1 - p)
        if lo1 < hi1:
            h_idx = np.arange(lo1, hi1) + p - (n - 1)
            grad[p] += np.dot(w[lo1:hi1], 1.0 - h[h_idx])
        lo2 = max(0, n - 1 - p)
        hi2 = min(2 * n - 1, n - 1 + p + 1)
        if lo2 < hi2:
            h_idx = p - np.arange(lo2, hi2) + (n - 1)
            grad[p] -= np.dot(w[lo2:hi2], h[h_idx])
    
    return grad, max_corr

def continuous_optimize(h0, n, deadline, alpha_start=1000):
    """Projected gradient with soft-max and adaptive learning rate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    lr = 0.1
    momentum = np.zeros(n)
    mom_factor = 0.9
    stall = 0
    
    while time.time() < deadline:
        grad, max_corr = soft_max_grad(h, n, alpha)
        # Adaptive learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = lr / gnorm
        momentum = mom_factor * momentum - step * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.05, 1.0)
        else:
            stall += 1
            if stall > 20:
                # Perturb and restart
                h = best_h + 0.01 * np.random.randn(n)
                h = project_feasible(h, n)
                momentum = np.zeros(n)
                lr = 0.05
                alpha = max(100, alpha * 0.5)
                stall = 0
            else:
                h = project_feasible(h0 + 0.5 * (h_new - h0), n)  # backtrack
                lr *= 0.8
        
        # Increase alpha gradually (sharper soft-max)
        if stall % 50 == 0 and alpha < 1e6:
            alpha *= 1.5
    
    return best_h, best_c
```
