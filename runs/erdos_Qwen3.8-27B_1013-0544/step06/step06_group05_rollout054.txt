def sa_fast(h, n, deadline, seed=42):
    """Fast SA using O(n) delta computation per step."""
    dx = 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute current overlaps for all shifts
    # overlap[s] for s in [1, n-1] (positive shifts)
    # By symmetry, overlap[-s] relates to overlap[s]
    # Actually let's just compute all 2n-1 values
    
    # corr[k] for k in range(2n-1), lag = k - (n-1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.int64)
    best_val = corr.max()
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    T = float(best_val)
    T_min = 0.5
    steps = 0
    
    while time.time() < deadline and T > T_min:
        T *= 0.9999
        steps += 1
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for all lags
        # For shift s (lag s), delta = h[j+s] - h[i+s] + h[j-s] - h[i-s]
        # where h[x] = 0 outside [0,n)
        
        # Efficient computation:
        # h[j+s] is non-zero only for s in [-j, n-1-j]
        # h[i+s] is non-zero only for s in [-i, n-1-i]
        # h[j-s] is non-zero only for s in [j-n+1, j]
        # h[i-s] is non-zero only for s in [i-n+1, i]
        
        # For all s in [-(n-1), n-1]:
        # delta(s) = h[j+s] - h[i+s] + h[j-s] - h[i-s]

        s_range = np.arange(-(n-1), n)
        # h[j+s] for s in range: valid where 0 <= j+s < n
        vs = s_range + j
        hj = np.where((vs >= 0) & (vs < n), h[vs.clip(0, n-1)], 0.0)
        vs = s_range + i
        hi = np.where((vs >= 0) & (vs < n), h[vs.clip(0, n-1)], 0.0)
        vs = j - s_range
        hj_s = np.where((vs >= 0) & (vs < n), h[vs.clip(0, n-1)], 0.0)
        vs = i - s_range
        hi_s = np.where((vs >= 0) & (vs < n), h[vs.clip(0, n-1)], 0.0)
        
        delta = hj - hi + hj_s - hi_s
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < best_val or rng.rand() < np.exp(-(new_max - best_val * 0) / T):

                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]

                    best_val = new_max
                    best_h = h.copy()
    
    return best_h, float(best_val * dx)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def sa_fast(h, n, deadline, seed=42, T_init=None, cool_rate=0.9999, T_min=0.3):
    """Fast simulated annealing with O(n) delta per step."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    curr_max = corr.max()
    best_val = curr_max
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    if T_init is None:
        T = float(curr_max)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_ones = n // 2
    
    while time.time() < deadline and T > T_min:
        T *= cool_rate
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Delta computation: for lag s, delta(s) = h[j+s] - h[i+s] + h[j-s] - h[i-s]
        # where h[x]=0 outside [0,n)
        # Vectorized:
        s = np.arange(-(n-1), n)
        
        vs = j + s
        hj_s = h[vs] * ((vs >= 0) & (vs < n))
        vs = i + s
        hi_s = h[vs] * ((vs >= 0) & (vs < n))
        vs = j - s
        hj_m = h[vs] * ((vs >= 0) & (vs < n))
        vs = i - s
        hi_m = h[vs] * ((vs >= 0) & (vs < n))
        
        delta = hj_s - hi_s + hj_m - hi_m
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < curr_max or (new_max > curr_max and rng.rand() < np.exp(-(new_max - curr_max) / T)):
            corr = new_corr
            curr_max = new_max
            h[i] = 1.0
            h[j] = 0.0
            # Update ones and zeros
            idx = np.where(ones == i)[0]
            if len(idx) > 0:
                ones = np.delete(ones, idx[0])
            idx = np.where(zeros == j)[0]
            if len(idx) > 0:
                zeros = np.delete(zeros, idx[0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
    
    return best_h, float(best_val * dx)
```

    if T_init is None:
        T = float(curr_max)
    
    s_arr = np.arange(-(n-1), n)  # All possible lags
    
    while time.time() < deadline and T > 0.1:
        T *= cool_rate
        
        # Get a random 0 and 1 position
        i = np.where(h == 0)[0][rng.randint(n//2)]
        j = np.where(h == 1)[0][rng.randint(n//2)]
        
        # Compute delta vector
        hjp = _safe_gather(h, j + s_arr, n)
        hip = _safe_gather(h, i + s_arr, n)
        hjm = _safe_gather(h, j - s_arr, n)
        him = _safe_gather(h, i - s_arr, n)

        new_max = np.max(corr + delta)
        
        if new_max < curr_max or rng.rand() < np.exp(-(new_max - curr_max) / T):
            corr = corr + delta
            curr_max = new_max
            h[i] = 1.0
            h[j] = 0.0
            if curr_max < best_val:
                best_val = curr_max
                best_h = h.copy()
            # No need to recompute corr from scratch since we have the delta
    
    return best_h, float(best_val * dx)

def _safe_gather(h, indices, n):
    """Gather h at indices, returning 0 for out-of-bounds."""
    result = np.zeros(len(indices))
    valid = (indices >= 0) & (indices < n)
    result[valid] = h[indices[valid]]
    return result
```

def directed_search(h, n, deadline, seed=42, max_iterations=500000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_max = corr.max()
    best_val = curr_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    iter_count = 0
    while time.time() < deadline and iter_count < max_iterations:
        iter_count += 1
        max_k = int(np.argmax(corr))
        d = max_k - (n - 1)  # The lag with maximum correlation
        
        # Find positions p (where h[p]=0) such that h[p-d]=1 (reduces corr at lag d)
        # And positions q (where h[q]=1) such that h[q+d]=0
        # Then check if the swap (p→1, q→0) reduces the overall max
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        # Filter zeros: p-d must be in [0,n) and h[p-d]=1
        valid_p = zeros[(zeros - d >= 0) & (zeros - d < n)]
        valid_p = valid_p[h[valid_p - d] == 1]
        
        # Filter ones: q+d must be in [0,n) and h[q+d]=0  
        valid_q = ones[(ones + d >= 0) & (ones + d < n)]
        valid_q = valid_q[h[valid_q + d] == 0]
        
        found = False
        if len(valid_p) > 0 and len(valid_q) > 0:
            # Try some pairs
            n_try = min(10, len(valid_p))
            m_try = min(10, len(valid_q))
            for _ in range(30):
                p = valid_p[rng.randint(len(valid_p))]
                q = valid_q[rng.randint(len(valid_q))]
                if p == q:
                    continue
                
                s_arr = np.arange(-(n-1), n)
                hjp = _safe_gather(h, q + s_arr, n)
                hip = _safe_gather(h, p + s_arr, n)
                hjm = _safe_gather(h, q - s_arr, n)
                him = _safe_gather(h, p - s_arr, n)
                delta = hjp - hip + hjm - him
                new_max = np.max(corr + delta)
                
                if new_max < curr_max:
                    corr = corr + delta
                    curr_max = new_max
                    h[p] = 1.0
                    h[q] = 0.0
                    best_val = curr_max
                    best_h = h.copy()
                    found = True
                    break
        
        if not found:
            # Escape: random swap with SA acceptance
            T = 1.0
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            s_arr = np.arange(-(n-1), n)
            hjp = _safe_gather(h, j + s_arr, n)
            hip = _safe_gather(h, i + s_arr, n)
            hjm = _safe_gather(h, j - s_arr, n)
            him = _safe_gather(h, i - s_arr, n)
            delta = hjp - hip + hjm - him
            new_max = np.max(corr + delta)
            
            if new_max < curr_max or rng.rand() < np.exp(-(new_max - curr_max) / T):
                corr = corr + delta
                curr_max = new_max
                h[i] = 1.0
                h[j] = 0.0
                if curr_max < best_val:
                    best_val = curr_max
                    best_h = h.copy()
    
    return best_h, float(best_val * dx)

def _safe_gather(h, indices, n):
    """Safe gather: h[indices] with 0 for out-of-bounds."""
    result = np.zeros(len(indices), dtype=np.float64)
    valid = (indices >= 0) & (indices < n)
    result[valid] = h[indices[valid]]
    return result

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones (top n/2 values)."""
    h = h.copy()
    threshold = np.partition(h, n - n // 2)[n - n // 2]
    h_bin = (h >= threshold).astype(float)
    # Fix count if needed
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        # Set some to 0 (the ones closest to threshold)
        ones_idx = np.where(h_bin == 1)[0]
        sorted_ones = ones_idx[np.argsort(h[ones_idx])[::-1]]
        for i in range(n_ones - n // 2):
            h_bin[sorted_ones[i]] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        sorted_zeros = zeros_idx[np.argsort(h[zeros_idx])[::-1]]
        for i in range(n // 2 - n_ones):
            h_bin[sorted_zeros[i]] = 1.0
    return h_bin

    best_val = curr_max
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    if T_init is None:

   - In our case a=i (the 0→1 position): change is +(1-h[a+L]) if valid

2. h appears as (1-h[.]) at position a: the term is h[a-L]*(1-h[a]), valid if a-L is in range
   - Valid if 0 ≤ a-L < n AND 0 ≤ a < n (a is always in [0,n) since it's a valid position)
   - For L≥0: need a-L ≥ 0, i.e., a ≥ L
   - For L<0: need a-L < n, i.e., a < n+L = n-|L|
   - Combined: valid if 0 ≤ a-L < n
   - Change: -h[a-L] (since h[a] went from 0 to 1, so (1-h[a]) went from 1 to 0)

3. h[b] as first factor: term = h[b]*(1-h[b+L]), valid if 0 ≤ b+L < n
   - Change: -(1-h[b+L]) (h[b] went from 1 to 0)

4. h appears as (1-h[.]) at position b: term = h[b-L]*(1-h[b]), valid if 0 ≤ b-L < n
   - Change: +h[b-L] (h[b] went from 1 to 0, so (1-h[b]) went from 0 to 1)

δ_L = [1-h[a+L]]·[0≤a+L<n] - h[a-L]·[0≤a-L<n] - [1-h[b+L]]·[0≤b+L<n] + h[b-L]·[0≤b-L<n]
