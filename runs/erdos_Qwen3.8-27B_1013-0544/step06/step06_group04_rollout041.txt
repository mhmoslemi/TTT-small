    m = x.max()
    if alpha < 1e8:
        s = alpha * (x - m)
        s2 = s.max()
        return m + (s2 + np.log(np.exp(s - s2).sum())) / alpha
    return m

    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    if alpha < 1e8:
        s = alpha * (c - m)
        s2 = s.max()
        w = np.exp(s - s2)

If h is "random" (h(x) = 0.5 + noise), then by concentration, most shifts will have overlap close to the mean (≈ 0.25), and the maximum will be slightly above. But we're getting 0.38, which is higher. This suggests the shifts are NOT independent (they're correlated due to the structure of h).

For the optimal h, we want to break these correlations. This is where the optimization comes in.

**Key realization**: The problem is equivalent to designing a binary (or near-binary) sequence of length n with n/2 ones, such that the "cross-correlation" with its complement is as flat as possible. This is related to "low autocorrelation sequences" in signal processing!

In signal processing, we want sequences with low peak sidelobe level (PSL). Our problem is similar but with the cross-correlation with the complement.

    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    lo2 = max(0, i)
    hi2 = min(2 * n - 2, n - 1 + i)
    m2 = np.arange(lo2, hi2 + 1)
    delta[m2] -= h[n - 1 + i - m2]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    lo2 = max(0, j)
    hi2 = min(2 * n - 2, n - 1 + j)
    m2 = np.arange(lo2, hi2 + 1)
    delta[m2] += h[n - 1 + j - m2]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search_b(h, n, deadline, max_sweeps=3000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = (corr + d).copy()
                    cur_c = nc
                    if nc < best_c:
                        best_c, best_h = nc, h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Progressive SLSQP smooth optimization
    alphas = [500, 5000, 50000, 500000, 5000000, 50000000]
    for alpha in alphas:
        if time.time() > deadline - 100:
            break
        hr = slsqp_smooth(best_h.copy(), n, alpha, maxiter=400)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()

    # Phase 2: Projected gradient refinement
    if time.time() < deadline - 50:
        for alpha in [50000, 500000, 5000000, 50000000]:
            if time.time() > deadline - 20:
                break
            hg, cg = projected_gradient(best_h.copy(), n, alpha, n_iter=3000, eta0=0.008)
            if cg < best_c:
                best_c, best_h = cg, hg.copy()

    # Phase 3: SLSQP again with very high alpha after gradient refinement
    if time.time() < deadline - 30:
        hr = slsqp_smooth(best_h.copy(), n, 500000000, maxiter=600)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()

    # Phase 4: Binary local search
    if time.time() < deadline - 20:
        hb, cb = local_search_b(best_h, n, deadline - 10, max_sweeps=5000, seed=seed)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 5: Restart with perturbations (smooth path)
    n_restarts = 15
    for r in range(n_restarts):
        if time.time() > deadline - 15:
            break
        h_p = best_h.copy()
        n_sw = rng.randint(3, 8)
        for _ in range(n_sw):
            a, b = int(rng.randint(n)), int(rng.randint(n))
            h_p[a], h_p[b] = h_p[b], h_p[a]
        h_p = project_feasible(h_p, n)
        if time.time() < deadline - 10:
            hr = slsqp_smooth(h_p, n, 5000000, maxiter=300)
            c = compute_c5(hr, n)
            if c < best_c:
                best_c, best_h = c, hr.copy()
        if time.time() < deadline - 8:
            hg, cg = projected_gradient(best_h.copy(), n, 50000000, n_iter=1500, eta0=0.005)
            if cg < best_c:
                best_c, best_h = cg, hg.copy()
        if time.time() < deadline - 5:
            hb, cb = local_search_b(best_h, n, min(deadline - 3, time.time() + 15),
                                     max_sweeps=2000, seed=seed + r * 7919)
            if cb < best_c:
                best_c, best_h = cb, hb.copy()

    # Phase 6: Try larger n
    if time.time() < deadline - 80:
        n2 = 500
        x1 = np.linspace(0, 2, n, endpoint=False)
        x2 = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x2, x1, best_h)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)
        if c2 < best_c:
            best_c, best_h, n = c2, h2.copy(), n2
        if time.time() < deadline - 40:
            hr = slsqp_smooth(best_h.copy(), n, 1000000, maxiter=400)
            c = compute_c5(hr, n)
            if c < best_c:
                best_c, best_h = c, hr
        if time.time() < deadline - 20:
            hg, cg = projected_gradient(best_h.copy(), n, 10000000, n_iter=2000, eta0=0.005)
            if cg < best_c:
                best_c, best_h = cg, hg

    # Final SLSQP polish
    if time.time() < deadline - 5:
        hr = slsqp_smooth(best_h.copy(), len(best_h), 1000000000, maxiter=400)
        c = compute_c5(hr, len(best_h))
        if c < best_c:
            best_c, best_h = c, hr

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
