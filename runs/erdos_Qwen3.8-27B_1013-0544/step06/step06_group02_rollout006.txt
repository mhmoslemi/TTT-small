import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max approximation."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def solve_lp_subproblem(h, n, active_indices, epsilon=0.01):

    h = np.array(h, dtype=float)
    c_vals = compute_c5_vec(h, n)
    C = c_vals.max()
    
    # Active set: indices where c_val >= C - epsilon
    if active_indices is None:

        return h.copy()

    # In the full correlation, index m corresponds to lag k = m - (n-1)
    grads = []

    for m_idx in active_indices:
        k = m_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[0:nv] = 1.0 - h[k:n]
            g[k:n] = -h[0:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:n] = 1.0 - h[0:nv]
            g[0:nv] = -h[kk:n]
        dx = 2.0 / n
        g *= dx
        grads.append(g)
        consts.append(c_vals[m_idx] - np.dot(g, h))
    
    A = np.array(grads)  # shape (len(active), n)
    b = np.array(consts)
    
    # LP: min t s.t. A @ h + b <= t, 0 <= h <= 1, sum(h) = n/2

        h_var >= 0,

def nesterov_gradient(h, n, deadline, alpha=500, lr_init=0.05):
    """Nesterov accelerated projected gradient."""
    h = project_feasible(h, n).copy()
    m_vec = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    t = 1.0
    
    for it in range(5000):
        if time.time() > deadline:
            break
        
        # Nesterov lookahead
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t_new - 1) / t
        
        y = h + beta * m_vec
        y = np.clip(y, 0, 1)
        
        g = grad_smooth_max(y, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Backtracking line search
        h_new = None
        for _ in range(30):
            cand = h - lr * g
            cand = project_feasible(cand, n)
            c_cand = compute_c5(cand, n)
            if c_cand < best_c:
                h_new = cand
                best_c = c_cand
                best_h = cand.copy()
                break
            lr *= 0.5

            lr *= 0.5
            alpha *= 1.3
            if lr < 1e-12:
                break
        
        m_vec = h_new - h
        h = h_new
        t = t_new
    
    return best_h, best_c

def pair_search_fine(h, n, deadline, deltas=None):

    radius = min(30, n // 5)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def discrete_grad_lag(h, n, m_idx):
    dx = 2.0 / n
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] = 1.0 - h[k:n]
        g[k:n] = -h[0:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:n] = 1.0 - h[0:nv]
        g[0:nv] = -h[kk:n]
    return g * dx

def solve_lp_subproblem(h, n, active_indices):
    h = np.array(h, dtype=float)
    c_vals = compute_c5_vec(h, n)
    A_rows = []
    b_vec = []
    for m_idx in active_indices:
        g = discrete_grad_lag(h, n, m_idx)
        A_rows.append(g)
        b_vec.append(np.dot(g, h) - c_vals[m_idx])
    A = np.array(A_rows)
    b = np.array(b_vec)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [A @ h_var - t_var <= b, h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    solvers = [cp.GLOP, cp.CBC, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return np.clip(h_var.value, 0, 1)
        except Exception:
            continue
    return h.copy()

def active_set_lp(h, n, deadline, frac=0.2, eps_frac=0.01, max_iter=300):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = compute_c5_vec(h, n)
        C = c_vals.max()
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
        rng_thresh = C - c_vals.min() + 1e-10
        thresh = C - eps_frac * rng_thresh
        active = np.where(c_vals >= thresh)[0]
        n_max = max(5, int(len(c_vals) * frac))
        if len(active) > n_max:
            order = np.argsort(c_vals)[::-1]

        h_new = solve_lp_subproblem(h, n, active)

                c_new2 = compute_c5(h_new2, n)
                if c_new2 < C - 1e-13:
                    h = h_new2
                    continue
            break
    return best_h, best_c

def nesterov_grad(h, n, deadline, alpha=500, lr_init=0.03):

    best_c = compute_c5(h, n)

    lr = lr_init

    for _ in range(5000):

            break
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t_new - 1) / t
        y = np.clip(h + beta * m_vec, 0, 1)
        g = grad_smooth_max(y, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = None

            cand = project_feasible(h - lr * g, n)
            cc = compute_c5(cand, n)
            if cc < best_c - 1e-15:
                h_new = cand
                best_c = cc
                best_h = cand.copy()
                break
            lr *= 0.5
        if h_new is None:
            lr *= 0.5
            alpha *= 1.2
            if lr < 1e-12:
                break
        m_vec = h_new - h
        h = h_new
        t = t_new
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=40, stride=1):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, stride):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
