    """Gradient of (1/alpha)*log(sum(exp(alpha*c))) w.r.t. c"""
    m = c.max()
    shifted = alpha * (c - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    return alpha * weights  # This is the gradient of LSE w.r.t. the c values

    best_c = compute_c5(best, n)

            m = c.max()
