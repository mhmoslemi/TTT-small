def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0

    best_c = compute_c5(best_h, n)
    mf, sf, mb, sb = make_grad_mats(n)
    rng = np.random.RandomState(seed)
    # Phase 1: initial smooth
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 400, 1e-14, mf, sf, mb, sb)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: binary SA (main workhorse)
    p2 = t0 + 0.55 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    for att in range(5):
        if time.time() > p2:
            break
        if att == 0:
            start = h_bin
            s = seed
        elif att == 1:
            start = round_to_binary(best_h + 0.05 * rng.randn(n), n)
            s = seed + 100
        elif att == 2:
            start = round_to_binary(best_h + 0.15 * rng.randn(n), n)
            s = seed + 200
        elif att == 3:
            start = h_bin.copy()
            zi = np.where(start == 0)[0]
            oi = np.where(start == 1)[0]
            for _ in range(15):
                start[zi[rng.randint(len(zi))]] = 1.0
                start[oi[rng.randint(len(oi))]] = 0.0
            s = seed + 300
        else:
            start = h_bin.copy()
            zi = np.where(start == 0)[0]
            oi = np.where(start == 1)[0]
            for _ in range(30):
                start[zi[rng.randint(len(zi))]] = 1.0
                start[oi[rng.randint(len(oi))]] = 0.0
            s = seed + 400
        hb, cb = binary_sa(start, n, p2, seed=s, t_start=0.035, t_end=0.00005, patience=50000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: smooth from best binary
    p3 = t0 + 0.82 * (budget_s - 5)
    h = best_h + 0.01 * rng.randn(n)
    h = project_feasible(h, n)
    for alpha in [5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p3:
            break
        h = slsqp_smooth(h, n, alpha, 600, 1e-15, mf, sf, mb, sb)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = h + 0.003 * rng.randn(n)
        h = project_feasible(h, n)
    # Phase 4: projected gradient polish
    p4 = dl - 15
    if time.time() < p4 - 10:
        h = best_h + 0.002 * rng.randn(n)
        h = project_feasible(h, n)
        hp, cp = projected_grad(h, n, 500000, 5000, 0.005, mf, sf, mb, sb)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    # Phase 5: final SLSQP polish
    if time.time() < dl - 5:
        h = best_h.copy()
        h = slsqp_smooth(h, n, 5000000, 500, 1e-15, mf, sf, mb, sb)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
