```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    """Compute all correlation values c_m * dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def compute_c_unique(h, n):
    """Compute the n unique correlation values (using symmetry)."""
    dx = 2.0 / n
    # c[m] for m = 0, 1, ..., n-1
    c = np.zeros(n)
    for p in range(n):
        c[p] = np.dot(h[p:], 1 - h[:n-p]) * dx
    return c

    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            if tot < 1e-14:

            h += (diff / tot) * h
```

```python
def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Gradient of smooth max of c_p values."""
    c = compute_c_unique(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    g = np.zeros(n)
    
    # Only compute for lags with significant weight
    for p in range(n):
        if w[p] < 1e-10:
            continue
        wp = w[p] * dx
        if p == 0:
            g += wp * (1 - 2 * h)
        else:
            g[p:] += wp * (1 - h[:n-p])
            g[:n-p] += wp * (-h[p:])
    return g

def smooth_max_obj(h, n, alpha):
    c = compute_c_unique(h, n)
    s = alpha * (c - c.max())
    return float(c.max() + np.log(np.sum(np.exp(s))) / alpha)

def smooth_max_obj_pen(h, n, alpha, mu):
    c = compute_c_unique(h, n)
    s = alpha * (c - c.max())
    val = c.max() + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad_pen(h, n, alpha, mu):
    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def optimize_lbfgs(h, n, deadline, max_alpha=200000):
    """L-BFGS-B with increasing alpha."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [10, 50, 200, 1000, 5000, 20000, 100000, max_alpha]
    mus = [1.0, 5.0, 10.0, 50.0, 100.0, 500.0, 2000.0, 5000.0]
    
    for i, (a, m) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj_pen, h, jac=smooth_max_grad_pen,
                          args=(n, a, m), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 5000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgrad_refine(h, n, deadline, max_iters=100000):
    """Subgradient method using only the active constraint."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_c_unique(h, n)
        p_star = int(np.argmax(c))
        
        # Gradient of c_{p_star}
        dx = 2.0 / n
        g = np.zeros(n)
        if p_star == 0:
            g = dx * (1 - 2 * h)
        else:
            g[p_star:] += dx * (1 - h[:n-p_star])
            g[:n-p_star] += dx * (-h[p_star:])
        
        # Try step
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_new = project_feasible(h - lr * factor * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                break
        else:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
    
    return best_h, best_c

def vectorized_pair_search(h, n, deadline, max_radius=50, step=0.01):
    """Vectorized pair search using precomputed deltas."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def compute_c_unique(h, n):
    dx = 2.0 / n
    c = np.empty(n)
    for p in range(n):
        c[p] = np.dot(h[p:n], 1.0 - h[0:n-p]) * dx
    return c

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-13: break
            h += (d/s) * r
        else:
            s = h.sum()
            if s < 1e-13: break
            h += (d/s) * h
    return np.clip(h, 0.0, 1.0)

def grad_c(h, n, p):
    """Gradient of c_p w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if p == 0:
        return dx * (1.0 - 2.0 * h)
    g[p:] += (1.0 - h[:n-p])
    g[:n-p] -= h[p:]
    return g * dx

def obj_smooth(h, n, alpha, mu):
    c = compute_c_unique(h, n)
    s = alpha * (c - c.max())
    val = c.max() + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (h.sum() - n*0.5)**2
    return val

def grad_smooth(h, n, alpha, mu):
    c = compute_c_unique(h, n)
    cs = alpha * (c - c.max())
    w = np.exp(cs - cs.max())
    w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        if w[p] > 1e-10:
            g += w[p] * grad_c(h, n, p)
    g += 2.0 * mu * (h.sum() - n*0.5)
    return g

def phase_lbfgs(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0,1)] * n
    configs = [(50,1),(200,5),(1000,10),(5000,50),(20000,100),(100000,500),(500000,2000)]
    for alpha, mu in configs:
        if time.time() > deadline - 2: break
        try:
            res = minimize(obj_smooth, h, jac=grad_smooth,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter':300,'ftol':1e-18,'gtol':1e-14,'maxfun':3000})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c, h = cn, hn
        except: pass
    return h, best_c

def phase_subgrad(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    stuck = 0
    for _ in range(200000):
        if time.time() > deadline: break
        c = compute_c_unique(h, n)
        p = int(np.argmax(c))
        g = grad_c(h, n, p)
        ok = False
        for f in [1, 0.5, 0.25, 0.1, 0.05, 0.02]:
            hn = project_feasible(h - lr*f*g, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c, h, best_h = cn, hn, hn.copy()
                stuck = 0
                ok = True
                break
        if not ok:
            lr *= 0.7
            stuck += 1
            if stuck > 200: break
    return best_h, best_c

def phase_grad_pair(h, n, deadline):
    """Gradient-guided pair moves."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c_unique(h, n)
            p = int(np.argmax(c))
            g = grad_c(h, n, p)
            # Move from high gradient to low gradient
            idx = np.argsort(g)
            for k in range(min(20, n//2)):
                if time.time() > deadline: return h, best_c
                i = idx[n-1-k]
                j = idx[k]
                if g[i] - g[j] < 1e-8: continue
                if h[i]-d >= 0 and h[j]+d <= 1:
                    h[i]-=d; h[j]+=d
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn; improved = True
                    else:
                        h[i]+=d; h[j]-=d
    return h, best_c

def phase_pair(h, n, deadline, radius=60):
    """Exhaustive local pair search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                lo, hi = max(0,i-radius), min(n,i+radius+1)
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i]+d <= 1 and h[j]-d >= 0:
                        h[i]+=d; h[j]-=d
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14: best_c=cn; improved=True
                        else: h[i]-=d; h[j]+=d
                    if h[i]-d >= 0 and h[j]+d <= 1:
                        h[i]-=d; h[j]+=d
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14: best_c=cn; improved=True
                        else: h[i]+=d; h[j]-=d
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c_unique(h, n):
    dx = 2.0 / n
    c = np.empty(n)
    for p in range(n):
        c[p] = np.dot(h[p:n], 1.0 - h[0:n-p]) * dx
    return c

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-13:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_c(h, n, p):
    dx = 2.0 / n
    if p == 0:
        return dx * (1.0 - 2.0 * h)
    g = np.zeros(n)
    g[p:] += (1.0 - h[:n-p])
    g[:n-p] -= h[p:]
    return g * dx

def obj_smooth(h, n, alpha, mu):
    c = compute_c_unique(h, n)
    s = alpha * (c - c.max())
    val = c.max() + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (h.sum() - n * 0.5) ** 2
    return float(val)

def grad_smooth(h, n, alpha, mu):
    c = compute_c_unique(h, n)
    cs = alpha * (c - c.max())
    w = np.exp(cs - cs.max())
    w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        if w[p] > 1e-10:
            g += w[p] * grad_c(h, n, p)
    g += 2.0 * mu * (h.sum() - n * 0.5)
    return g

def phase_lbfgs(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

               (20000, 100.0), (100000, 500.0), (500000, 2000.0), (1000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_smooth, h, jac=grad_smooth,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 4000})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c, h = cn, hn
        except Exception:
            pass
    return h, best_c

def phase_subgrad(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    stuck = 0
    for _ in range(300000):
        if time.time() > deadline:
            break
        c = compute_c_unique(h, n)
        p = int(np.argmax(c))
        g = grad_c(h, n, p)
        ok = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            hn = project_feasible(h - lr * f * g, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c, h, best_h = cn, hn, hn.copy()
                stuck = 0
                ok = True
                break
        if not ok:
            lr *= 0.7
            stuck += 1
            if stuck > 300:
                break
    return best_h, best_c

def phase_grad_pair(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c_unique(h, n)
            p = int(np.argmax(c))
            g = grad_c(h, n, p)
            idx = np.argsort(g)
            for k in range(min(30, n // 2)):
                if time.time() > deadline:
                    return h, best_c
                i = idx[n - 1 - k]
                j = idx[k]
                if g[i] - g[j] < 1e-8:
                    continue
                if h[i] - d >= 0 and h[j] + d <= 1:
                    h[i] -= d
                    h[j] += d
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                    else:
                        h[i] += d
```
