            tot = room.sum()
            if tot < 1e-14:

            tot = h.sum()
            if tot < 1e-14:

        y = project_feasible(x_new + (t - 1.0) / t_new * (x_new - x), n)
        x = x_new
        t = t_new
        if it % 200 == 100 and lr < 0.03:
            lr *= 1.03
    return best_h, best_c

def phase_slsqp(h, n, deadline, alpha=1000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hn = project_feasible(res.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def phase_grad_pair(h, n, deadline, base_step=0.002):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [500, 5000, 50000, 200000]:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    break
                gi = g[i]
                if abs(gi) < 1e-12:
                    continue
                for j in range(n):
                    if i == j:
                        continue
                    mv = base_step * np.sign(gi) * max(1, abs(int(gi * 1000)))
                    mv = min(mv, base_step * 3)
                    if 0 <= h[i] - mv and h[i] - mv <= 1 and 0 <= h[j] + mv and h[j] + mv <= 1:
                        h[i] -= mv; h[j] += mv
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += mv; h[j] -= mv
                    elif 0 <= h[i] + mv and h[i] + mv <= 1 and 0 <= h[j] - mv and h[j] - mv >= 0:
                        h[i] += mv; h[j] -= mv
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= mv; h[j] += mv
            if not improved:
                break
    return best_h, best_c

def phase_pair_fine(h, n, deadline, radius=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def phase_restart(h, n, deadline, rng, noise_scale=0.02):
    """Try a randomized perturbation and re-optimize briefly."""
    h_pert = h + rng.randn(n) * noise_scale
    h_pert = project_feasible(h_pert, n)
    alloc = min(40, deadline - time.time() - 5)
    if alloc < 15:
        return h, compute_c5(h, n)
    hp, cp = phase_lbfags(h_pert, n, time.time() + alloc, max_iter=100)
    return hp, cp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B main (extended alpha range)
    t1 = min(t0 + 320, deadline - 400)
    best_h, best_c = phase_lbfags(best_h, n, t1, max_iter=250)

    # Phase 2: FISTA acceleration
    if time.time() < deadline - 300:
        h2, c2 = phase_fista(best_h, n, min(time.time() + 100, deadline - 300), alpha=5000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: SLSQP refinement
    if time.time() < deadline - 220:
        for a in [500000, 5000000, 50000000]:
            if time.time() > deadline - 200:
                break
            h3, c3 = phase_slsqp(best_h, n, min(time.time() + 35, deadline - 200), a, 1500)
            if c3 < best_c:
                best_c, best_h = c3, h3

    # Phase 4: Second L-BFGS-B pass (high alpha)
    if time.time() < deadline - 160:
        best_h, best_c = phase_lbfags(best_h, n, min(time.time() + 70, deadline - 160), max_iter=150)

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 100:
        best_h, best_c = phase_grad_pair(best_h, n, min(time.time() + 70, deadline - 100))

    # Phase 6: Fine pair search
    if time.time() < deadline - 60:
        best_h, best_c = phase_pair_fine(best_h, n, min(time.time() + 90, deadline - 60), radius=80)

    # Phase 7: Randomized restarts
    if time.time() < deadline - 50:
        for _ in range(3):
            if time.time() > deadline - 40:
                break
            hr, cr = phase_restart(best_h, len(best_h), time.time() + 30, rng,
                                   noise_scale=0.015)
            if cr < best_c:
                best_c, best_h = cr, hr

    # Phase 8: Try other n values
    if time.time() < deadline - 35:
        n_cur = len(best_h)
        for n_try in [200, 250, 400, 500]:
            if time.time() > deadline - 25:
                break
            alloc = min(30, deadline - time.time() - 15)
            if alloc < 12:
                break
            x_old = np.linspace(0, 2, n_cur)
            x_new = np.linspace(0, 2, n_try)
            h_i = np.interp(x_new, x_old, best_h)
            h_i = project_feasible(h_i, n_try)
            ht, ct = phase_lbfags(h_i, n_try, time.time() + alloc * 0.7, max_iter=100)
            if ct < best_c:
                best_c, best_h = ct, ht
                n_cur = n_try

    # Phase 9: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = phase_pair_fine(best_h, nf, deadline - 2, radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
