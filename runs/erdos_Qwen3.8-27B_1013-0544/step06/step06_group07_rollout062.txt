            break

            if s < 1e-14:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()

                cur_c = nc
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            T *= cool_rate
            if no_improve > patience:
                for _ in range(3 + rng.randint(6)):
                    ii = rng.randint(n)
                    jj = rng.randint(n)
                    if h[ii] == 0 and h[jj] == 1:
                        h[ii] = 1.0
                        h[jj] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = float(corr.max()) * dx
                no_improve = 0
                T = min(t_start, T * 2.5)
    return best_h, best_c
