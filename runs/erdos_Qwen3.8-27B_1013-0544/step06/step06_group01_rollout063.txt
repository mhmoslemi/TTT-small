def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m > 500:
        return float((m + (m - np.min(np.where(c > m - 50, c, m - 50)) )) * dx)  # fallback
    sh = alpha * (c - m)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

    m = float(np.max(v))
    if pen > 0:
        m += pen * (sumh + d - target) ** 2
    return m

def cd_optimize(h, n, deadline, rng=None, max_passes=5000, pen=10.0):
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    nm1 = n - 1
    n_full = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            sens = np.zeros(n_full)
            j_lo = max(0, p)
            j_hi = min(n_full - 1, p + n - 1)
            if j_lo <= j_hi:
                sens[j_lo:j_hi+1] += 1.0 - h[p + nm1 - np.arange(j_lo, j_hi + 1)]
            j_lo2 = max(0, nm1 - p)
            j_hi2 = min(n_full - 1, 2 * n - 2 - p)
            if j_lo2 <= j_hi2:
                sens[j_lo2:j_hi2+1] -= h[p + np.arange(j_lo2, j_hi2 + 1) - n + 1]

            lo = -h[p]
            hi = 1.0 - h[p]
            if hi - lo < 1e-13:
                continue
            sumh = float(h.sum())
            a = lo
            b = hi
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            f1 = _f1d(c1, c, sens, nm1, sumh, target, pen)
            f2 = _f1d(c2, c, sens, nm1, sumh, target, pen)
            for _ in range(38):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - gr * (b - a)
                    f1 = _f1d(c1, c, sens, nm1, sumh, target, pen)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + gr * (b - a)
                    f2 = _f1d(c2, c, sens, nm1, sumh, target, pen)
            best_d = (a + b) / 2.0
            old_hp = h[p]
            h[p] = np.clip(old_hp + best_d, 0.0, 1.0)
            actual_d = h[p] - old_hp
            if abs(actual_d) > 1e-12:
                c += actual_d * sens
                c[nm1] -= actual_d * actual_d
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = np.clip(alpha * (c - m), -500, 500)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(x.shape[0])

def slsqp_polish(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_softmax_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def two_block_start(n, frac=0.25):
    k = max(1, int(frac * n))
    h = np.zeros(n)
    h[:k] = 1.0
    h[-k:] = 1.0
    return project_feasible(h, n)

def periodic_start(n, period, duty):
    on = max(1, int(duty * period))
    pat = np.zeros(period)
    pat[:on] = 1.0
    h = np.tile(pat, n // period + 1)[:n]
    return project_feasible(h, n)

def topk_equalize_cd(h, n, deadline, rng, max_passes=3000, topk=5, pen=10.0):
    """CD that targets reducing the top-k lags simultaneously."""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    nm1 = n - 1
    n_full = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            sens = np.zeros(n_full)
            j_lo = max(0, p)
            j_hi = min(n_full - 1, p + n - 1)
            if j_lo <= j_hi:
                sens[j_lo:j_hi+1] += 1.0 - h[p + nm1 - np.arange(j_lo, j_hi + 1)]
            j_lo2 = max(0, nm1 - p)
            j_hi2 = min(n_full - 1, 2 * n - 2 - p)
            if j_lo2 <= j_hi2:
                sens[j_lo2:j_hi2+1] -= h[p + np.arange(j_lo2, j_hi2 + 1) - n + 1]

            lo = -h[p]
            hi = 1.0 - h[p]
            if hi - lo < 1e-13:
                continue
            sumh = float(h.sum())

            # Use soft-max of top-k as objective
            def _f_topk(d):
                v = c + d * sens
                v[nm1] -= d * d
                m = float(np.max(v))
                # Add pressure on top-k
                sorted_v = np.sort(v)[::-1]
                topk_mean = float(np.mean(sorted_v[:topk]))
                obj = 0.7 * m + 0.3 * topk_mean
                if pen > 0:
                    obj += pen * (sumh + d - target) ** 2
                return obj

            a = lo; b = hi
            c1 = b - gr * (b - a); c2 = a + gr * (b - a)
            f1 = _f_topk(c1); f2 = _f_topk(c2)
            for _ in range(35):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - gr * (b - a); f1 = _f_topk(c1)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + gr * (b - a); f2 = _f_topk(c2)
            best_d = (a + b) / 2.0
            old_hp = h[p]
            h[p] = np.clip(old_hp + best_d, 0.0, 1.0)
            actual_d = h[p] - old_hp
            if abs(actual_d) > 1e-12:
                c += actual_d * sens
                c[nm1] -= actual_d * actual_d
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: aggressive CD
    t_sub = min(320, deadline - time.time() - 100)
    if t_sub > 40:
        h, c = cd_optimize(best_h, n, time.time() + t_sub, rng, max_passes=8000, pen=10.0)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 2: topk equalize
    if time.time() < deadline - 180:
        t_sub = min(120, (deadline - time.time() - 80) * 0.5)
        if t_sub > 30:
            h, c = topk_equalize_cd(best_h, n, time.time() + t_sub, rng, max_passes=4000, topk=5, pen=10.0)
            if c < best_c:
                best_h, best_c = h, c

    # Phase 3: SLSQP polish
    if time.time() < deadline - 150:
        for a in [50000, 300000]:
            if time.time() > deadline - 80:
                break
            h_s = slsqp_polish(best_h, n, alpha=a, maxiter=2500)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Phase 4: CD with low penalty (precise sum)
    if time.time() < deadline - 120:
        t_sub = min(100, (deadline - time.time() - 40) * 0.6)
        if t_sub > 20:
            h, c = cd_optimize(best_h, n, time.time() + t_sub, rng, max_passes=8000, pen=0.3)
            if c < best_c:
                best_h, best_c = h, c

    # Phase 5: structured restarts
    if time.time() < deadline - 100:
        starts = [
            two_block_start(n, 0.25),
            two_block_start(n, 0.35),
            periodic_start(n, 4, 0.5),
            periodic_start(n, 3, 1.0/3.0),
            project_feasible(best_h + rng.normal(0, 0.06, n), n),
            project_feasible(best_h + rng.normal(0, 0.12, n), n),
            project_feasible(best_h + rng.normal(0, 0.2, n), n),
        ]
        for s in starts:
            if time.time() > deadline - 60:
                break
            alloc = min(50, (deadline - time.time() - 30) * 0.5)
            if alloc < 15:
                break
            h, c = cd_optimize(s, n, time.time() + alloc, rng, max_passes=3000, pen=10.0)
            if c < best_c:
                best_h, best_c = h, c

    # Phase 6: final CD + SLSQP
    if time.time() < deadline - 30:
        h, c = cd_optimize(best_h, n, deadline - 12, rng, max_passes=8000, pen=0.1)
        if c < best_c:
            best_h, best_c = h, c
    if time.time() < deadline - 15:
        h_s = slsqp_polish(best_h, n, alpha=500000, maxiter=3000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
