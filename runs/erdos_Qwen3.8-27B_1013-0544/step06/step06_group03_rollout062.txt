def slsqp_refine(h, n, alpha=50000, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def binary_local_search(b, n, deadline, max_rounds=500):
    b = b.copy()
    best_c = compute_c5(b, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c_mv = best_c
        best_pair = None
        for i in ones:
            for j in zeros:
                b[i] = 0
                b[j] = 1
                c_mv = compute_c5(b, n)
                b[i] = 1
                b[j] = 0
                if c_mv < best_c_mv - 1e-14:
                    best_c_mv = c_mv
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            b[i] = 0
            b[j] = 1
            best_c = best_c_mv
        else:
            break
    return b, best_c

def sim_anneal_binary(b, n, deadline, rng, T0=0.02, cooling=0.995, max_iters=30000):
    b = b.copy()
    c_curr = compute_c5(b, n)
    best_c = c_curr
    best_b = b.copy()
    T = T0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if T < 1e-7:

        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        c_new = compute_c5(b, n)
        delta = c_new - c_curr
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        T *= cooling
    return best_b, best_c

        if c_new < compute_c5(h, n) + 1e-10:
            h = h_new
        else:
            step_size *= 0.9
        if step_size < 1e-6:
            break
    return best_h, best_c

def slsqp_refine(h, n, alpha=50000, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search
    if time.time() < deadline - 300:
        b = to_binary(h_init, n)
        bl_end = min(deadline - 250, t0 + 150)
        b, bc = binary_local_search(b, n, bl_end)
        if bc < best_c:
            best_c = bc
            best_h = b.copy()

    # Phase 2: SLSQP on best (continuous)
    if time.time() < deadline - 200:
        h_ref = slsqp_refine(best_h, n, alpha=50000, maxiter=3000)
        c_ref = compute_c5(h_ref, n)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # Phase 3: Re-binarize + local search
    if time.time() < deadline - 150:
        b2 = to_binary(best_h, n)
        b2, bc2 = binary_local_search(b2, n, min(deadline - 100, time.time() + 80))
        if bc2 < best_c:
            best_c = bc2
            best_h = b2.copy()

    # Phase 4: Simulated annealing (multiple restarts)
    if time.time() < deadline - 100:
        for restart in range(3):
            if time.time() > deadline - 60:
                break
            b3 = to_binary(best_h, n)
            # Add random perturbation for restarts > 0
            if restart > 0:
                n_swap = 3 + restart * 2
                ones = np.where(b3 == 1)[0]
                zeros = np.where(b3 == 0)[0]
                for _ in range(n_swap):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    b3[i], b3[j] = b3[j], b3[i]
                    ones = np.where(b3 == 1)[0]
                    zeros = np.where(b3 == 0)[0]
            sa_end = min(deadline - 40, time.time() + 40)
            b3, bc3 = sim_anneal_binary(b3, n, sa_end, rng, T0=0.015, cooling=0.997)
            if bc3 < best_c:
                best_c = bc3
                best_h = b3.copy()
            # Local search after SA
            if time.time() < deadline - 20:
                b3, bc3 = binary_local_search(b3, n, min(deadline - 15, time.time() + 30))
                if bc3 < best_c:
                    best_c = bc3
                    best_h = b3.copy()

    # Phase 5: Subgradient method
    if time.time() < deadline - 80:
        h_sg, c_sg = subgradient_steps(best_h, n, min(deadline - 40, time.time() + 50), rng=rng)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 6: Final SLSQP + binary polish
    if time.time() < deadline - 30:
        h_ref = slsqp_refine(best_h, n, alpha=100000, maxiter=4000)
        c_ref = compute_c5(h_ref, n)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    if time.time() < deadline - 5:
        b4 = to_binary(best_h, n)
        b4, bc4 = binary_local_search(b4, n, deadline - 2)
        if bc4 < best_c:
            best_c = bc4
            best_h = b4.copy()

    # Final SLSQP
    if time.time() < deadline - 2:
        h_fin = slsqp_refine(best_h, n, alpha=200000, maxiter=3000)
        c_fin = compute_c5(h_fin, n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
