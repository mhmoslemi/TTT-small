Now, for the 1D minimax problem:
Given lines l_k(t) = a_k * t + b_k for k = 1,...,M, find t* ∈ [0,1] minimizing max_k l_k(t).

Algorithm:
1. Compute l_k(0) and l_k(1) for all k
2. The optimal t* is either:
   - At a boundary (0 or 1)
   - At an intersection of two lines l_i(t*) = l_j(t*) that is the maximum at t*

To find the optimal:
- Sort lines by their value at t=0
- The upper envelope changes at intersection points
- The minimum of the upper envelope is at a "valley" point

A simpler approach for small M:
- Generate all pairwise intersection points t_ij = (b_j - b_i)/(a_i - a_j) in [0,1]
- Plus t=0 and t=1
- Evaluate max_k l_k at each candidate point
- Return the best

For M = 5-10 active lags, this is at most ~50 candidate points, which is fast.

This coordinate descent approach is O(n * M^2) per full sweep, which is very fast for n=300 and M=5-10.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    # Fix sum exactly
    h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minimax(a_vec, b_vec, t_lo=0.0, t_hi=1.0):
    """Solve min_t max_k (a_k * t + b_k) for t in [t_lo, t_hi]"""
    M = len(a_vec)
    # Candidate points: boundaries and pairwise intersections
    candidates = [t_lo, t_hi]
    for i in range(M):
        for j in range(i+1, M):
            denom = a_vec[i] - a_vec[j]
            if abs(denom) > 1e-15:
                t = (b_vec[j] - b_vec[i]) / denom
                if t_lo <= t <= t_hi:
                    candidates.append(t)
    best_t = t_lo
    best_val = float('inf')
    for t in candidates:
        val = max(a_vec * t + b_vec)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Restart perturbations on best solution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 35) * 0.5)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive refinement
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        # Coordinate descent with large M
        if time.time() < deadline - 15:
            h_cd = coord_descent(best_h, n, M=20, max_sweeps=150)
            c_cd = compute_c5(h_cd, n)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
        # SLSQP
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 5000000, maxiter=5000, ftol=1e-17)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Fine local search
        if time.time() < deadline - 2:
            best_h, best_c = local_search(best_h, n, deadline - 1, top_k=4)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
