                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                stagnant += 1
    return h, best_c

def full_sweep_search(h, n, deadline, steps):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in steps:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - 80)
            hi = min(n, i + 81)
            for j in range(lo, hi):
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                    else:
                        h[i] -= d
                        h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                    else:
                        h[i] += d
                        h[j] -= d
    return h, best_c

def perturb_reopt(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for scale in [0.005, 0.002, 0.001]:
        if time.time() > deadline - 40:
            break
        for trial in range(4):
            if time.time() > deadline - 10:
                break
            h_p = h + scale * rng.randn(n)
            h_p = project_feasible(h_p, n)
            h_new, c_new = lbfgs_run(h_p, n, 200000, 0.01, 150)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS with increasing alpha
    for alpha, mu in [(50, 1.0), (200, 0.5), (1000, 0.1), (5000, 0.05),
                      (20000, 0.01), (100000, 0.01), (500000, 0.005)]:
        if time.time() > deadline - 450:
            break
        h_new, c_new = lbfgs_run(best_h, n, alpha, mu, 200)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: SLSQP fine-tuning
    if time.time() < deadline - 350:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > deadline - 150:
                break
            h_new, c_new = slsqp_run(best_h, n, alpha, 800)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 3: Guided local search (primarily gradient-guided)
    if time.time() < deadline - 150:
        best_h, best_c = guided_local_search(best_h, n, deadline - 100)

    # Phase 4: Perturbation and re-optimization
    if time.time() < deadline - 100:
        best_h, best_c = perturb_reopt(best_h, n, deadline - 50, rng)

    # Phase 5: Full sweep with tiny steps
    if time.time() < deadline - 30:
        best_h, best_c = full_sweep_search(best_h, n, deadline - 5,
                                           [0.0002, 0.0001, 0.00005, 0.00002])

    # Phase 6: Final guided polish
    if time.time() < deadline:
        best_h, best_c = guided_local_search(best_h, n, deadline)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
