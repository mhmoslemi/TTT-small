import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_smooth_obj(h, n, alpha, lam):
    """Smooth approximation of max(c)*dx + penalty"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    se = e.sum()
    obj = (m + np.log(se) / alpha) * dx + lam * (h.sum() - n / 2.0) ** 2
    return obj

def lse_smooth_grad(h, n, alpha, lam):
    """Gradient of smooth objective"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    se = e.sum()
    w = e / se  # softmax weights, sum to 1
    
    # grad of LSE part: dx * sum_k w[k] * dc[k]/dh[p]
    # dc[k]/dh[p] = 1{valid} - h[p+d]*1{valid} - h[p-d]*1{valid}
    # where d = k - (n-1)
    
    grad = np.zeros(n)
    for p in range(n):
        k_lo = max(0, n - 1 - p)
        k_hi = min(2 * n - 2, 2 * n - 2 - p)
        ks = np.arange(k_lo, k_hi + 1)
        ds = ks - (n - 1)
        pd = p + ds
        pmd = p - ds
        term1 = 1.0 - h[pd]
        valid2 = (pmd >= 0) & (pmd < n)
        term2 = np.where(valid2, h[pmd], 0.0)
        grad[p] = np.sum(w[ks] * (term1 - term2))
    
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    
    val = (m + np.log(se) / alpha) * dx + lam * (h.sum() - n / 2.0) ** 2
    return val, grad

def opt_lbfgsb(h, n, deadline, alpha_start=100, alpha_end=10000, lam=100.0):
    """Optimize using L-BFGS-B with adaptive alpha"""
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    
    # Try different alphas (smoothness)
    alphas = [alpha_start]
    a = alpha_start
    while a < alpha_end and time.time() < deadline - 10:
        a = min(a * 2, alpha_end)
        alphas.append(a)
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(lse_smooth_grad, h, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8},
                          args=(n, alpha, lam))
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h, best_c = h_new, c_new
        except Exception:
            pass
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    se = float(e.sum())
    w = e / se
    
    # Gradient: dx * sum_k w[k] * dc[k]/dh[p] + 2*lam*(sum(h)-n/2)
    grad = np.zeros(n)
    for p in range(n):
        d_lo = max(0, -(n - 1 - p))  # = max(0, p-n+1) ... 
        # Actually let me just use the k range directly
        k_lo = max(0, n - 1 - p)
        k_hi = min(2 * n - 2, 2 * n - 2 - p)
        ks = np.arange(k_lo, k_hi + 1)
        ds = ks - (n - 1)
        pd = p + ds  # always in [0, n-1]
        pmd = p - ds
        term1 = 1.0 - h[pd]
        valid2 = (pmd >= 0) & (pmd < n)
        term2 = np.where(valid2, h[pmd], 0.0)
        grad[p] = np.dot(w[ks], term1 - term2)
    
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    
    val = (m + np.log(se) / alpha) * dx + lam * (h.sum() - n / 2.0) ** 2
    return val, grad

def opt_lbfgsb(h, n, deadline, alphas=None, lam=50.0, maxiter=300):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(lse_obj_grad, h, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10},
                          args=(n, alpha, lam))
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h, best_c = h_new, c_new
        except Exception:
            pass
    return h, best_c

def coord_desc(h, n, max_passes=100, grid=50, deadline=None, penalty=1.0):
    """Simple coordinate descent"""
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    
    for _pass in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                return h, current_max * (2.0/n)
            if h[p] >= 1.0 - 1e-12 or h[p] <= 1e-12:
                continue
            # Find affected k range
            k_lo = max(0, n - 1 - p)
            k_hi = min(2 * n - 2, 2 * n - 2 - p)
            ks = np.arange(k_lo, k_hi + 1)
            ds = ks - (n - 1)
            # For changing h[p] by delta:
            # c[k] changes by: delta * (1-h[p+ds]) - delta * h[p-ds] (if valid) - delta^2 (for the h[p]*h[p] term when ds=0)
            # Actually: c[k] = sum_i h[i]*(1-h[i+ds])
            # If we change h[p] by delta:
            #   Term where i=p: delta*(1-h[p+ds])
            #   Term where i+ds=p, i.e., i=p-ds: h[p-ds]*(-delta) = -delta*h[p-ds]
            #   Special case ds=0 (k=n-1): the term h[p]*(1-h[p]) changes by delta*(1-2h[p])-delta^2
            
            pd = p + ds
            pmd = p - ds
            # delta_c[k] = delta*(1-h[pd]) - delta*h[pmd]*1{pmd valid and pmd != pd} 
            #             - delta^2 * 1{ds==0}
            # Wait, when ds=0, pd=pmd=p, and the term is h[p]*(1-h[p]), 
            # which changes to (h[p]+delta)*(1-h[p]-delta) = h[p]-h[p]^2 + delta*(1-2h[p]) - delta^2
            # So the change is delta*(1-2h[p]) - delta^2
            # For ds!=0: change is delta*(1-h[p+ds]) - delta*h[p-ds] (if p-ds is valid)
            
            # Let me just compute the total change for each candidate delta
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            deltas = np.linspace(d_lo, d_hi, grid)
            
            # Weight for the center (k=n-1, ds=0)
            # When ds=0: delta_c = delta*(1-2*h[p]) - delta^2
            # When ds!=0: delta_c = delta*(1-h[p+ds]) - delta*h[p-ds] (if valid)
            
            center_idx = (ks == (n-1))
            # Linear part: delta * (1-h[pd]) - delta*h[pmd]*valid
            valid2 = (pmd >= 0) & (pmd < n) & (~center_idx)
            linear = (1.0 - h[pd]) - np.where(valid2, h[pmd], 0.0)
            # For center: linear part is (1-2*h[p]), quadratic is -delta^2
            if np.any(center_idx):
                linear[center_idx] = 1.0 - 2.0 * h[p]
            
            # Quadratic part: only at center
            quad = np.zeros_like(ks, dtype=float)
            quad[center_idx] = -1.0
            
            # Change in c: delta * linear + delta^2 * quad
            # New c: c + delta*linear + delta^2*quad (for affected ks)
            
            # Unaffected max
            mask = np.ones(2*n-1, dtype=bool)
            mask[ks] = False
            unaffected_max = float(np.max(c[mask])) if np.any(mask) else 0.0
            
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n/2.0)**2
            
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c_affected = c[ks] + d * linear + d*d * quad
                val = max(unaffected_max, float(np.max(new_c_affected)))
                if penalty > 0:
                    val += penalty * (sumh + d - n/2.0)**2
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-13:
                c[ks] += best_d * linear + best_d*best_d * quad
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(unaffected_max, float(np.max(c[ks])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max * (2.0 / n)

def pair_desc(h, n, deadline, deltas=None, radius=30):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        any_improve = True
        while any_improve and time.time() < deadline:
            any_improve = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-13:
                            best_c = cv; any_improve = True
                        else:
                            h[i] -= d; h[j] += d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:

        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_c = compute_c5(best_h, n_init)

    t_sub = time.time() + min(200, (deadline - t0) * 0.25)
    h_opt, c_opt = opt_lbfgsb(best_h, n_init, t_sub, 
                               alphas=[50, 100, 200, 500, 1000, 2000, 5000], lam=50)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Coordinate descent
    t_sub = time.time() + min(200, (deadline - time.time()) * 0.3)
    h_cd, c_cd = coord_desc(best_h, n_init, max_passes=200, grid=60,
                            deadline=t_sub, penalty=1.0)
    h_cd = project_feasible(h_cd, n_init)
    c_cd = compute_c5(h_cd, n_init)
    if c_cd < best_c:
        best_h, best_c = h_cd, c_cd
    
    # Phase 3: Try n=200
    if n_init != 200 and time.time() < deadline - 150:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, 200)
        h_200 = project_feasible(np.interp(x_new, x_old, best_h), 200)
        t_sub = time.time() + min(150, (deadline - time.time()) * 0.3)
        h_o, c_o = opt_lbfgsb(h_200, 200, t_sub,
                               alphas=[50, 200, 500, 2000], lam=50)
        h_o, c_o = coord_desc(h_o, 200, max_passes=100, grid=50, deadline=deadline-60)
        h_o = project_feasible(h_o, 200)
        c_o = compute_c5(h_o, 200)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    
    # Phase 4: Multi-restart
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(2, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            scale = 0.01 + 0.03 * r / max(1, n_restarts)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            t_sub = time.time() + min(60, (deadline - time.time() - 30) * 0.4)
            if t_sub - time.time() < 20:
                break
            h_o, c_o = opt_lbfgsb(hp, n, t_sub,
                                   alphas=[200, 1000, 5000], lam=50, maxiter=200)
            h_o = project_feasible(h_o, n)
            c_o = compute_c5(h_o, n)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    
    # Phase 5: Final coordination descent + pair
    if time.time() < deadline - 30:
        n = len(best_h)
        t_sub = time.time() + min(40, deadline - time.time() - 15)
        h_cd, c_cd = coord_desc(best_h, n, max_passes=100, grid=80,
                                deadline=t_sub, penalty=0.5)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    
    if time.time() < deadline - 10:
        n = len(best_h)
        best_h, best_c = pair_desc(best_h, n, deadline - 3,
                                    deltas=[0.001, 0.0005, 0.0002], radius=30)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sc = alpha * (c - m)
    mx = sc.max()
    e = np.exp(sc - mx)
    se = float(e.sum())
    w = e / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        n_lo = max(0, -d)
        n_hi = min(n - 1, n - 1 - d)
        idx1 = n_lo + np.arange(n_hi - n_lo + 1)
        pd = idx1 + d
        grad[pd] += wk * (1.0 - h[idx1])
        pm_idx = n_lo + np.arange(n_hi - n_lo + 1)
        pm_d = pm_idx + d
        grad[pm_idx] -= wk * h[pm_d]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    val = (m + (mx + np.log(se)) / alpha) * dx + lam * (h.sum() - n / 2.0) ** 2
    return val, grad

def opt_lbfgsb(h, n, deadline, alphas=None, lam=50.0, maxiter=300):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(lse_obj_grad, h, method='L-BFGS-B',
                           bounds=[(0, 1)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                                    'maxls': 20},
                           args=(n, alpha, lam))
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h, best_c = h_new, c_new
        except Exception:
            pass
    return h, best_c

def coord_desc(h, n, max_passes=200, grid=60, deadline=None, penalty=1.0):
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            if h[p] <= 1e-12 or h[p] >= 1.0 - 1e-12:
                continue
            d_lo = max(0, p - n + 1)
            d_hi = n - 1 - p
            if d_lo > d_hi:
                continue
            ds = np.arange(d_lo, d_hi + 1)
            ks = ds + (n - 1)
            pd = p + ds
            pmd = p - ds
            valid2 = (pmd >= 0) & (pmd < n)
            linear = (1.0 - h[pd]) - np.where(valid2, h[pmd], 0.0)
            center = (ds == 0)
            if np.any(center):
                linear[center] = 1.0 - 2.0 * h[p]
            quad = np.zeros(len(ds))
            quad[center] = -1.0
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[ks] = False
            unaffected = float(np.max(c[mask])) if np.any(mask) else 0.0
            deltas = np.linspace(-h[p], 1.0 - h[p], grid)
            best_d, best_val = 0.0, current_max
            for dd in deltas:
                if abs(dd) < 1e-14:
                    continue
                nc = c[ks] + dd * linear + dd * dd * quad
                val = max(unaffected, float(np.max(nc)))
                if penalty > 0:
                    val += penalty * (sumh + dd - n / 2.0) ** 2
                if val < best_val - 1e-13:
                    best_val, best_d = val, dd
            if abs(best_d) > 1e-13:
                c[ks] += best_d * linear + best_d * best_d * quad
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(unaffected, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def pair_desc(h, n, deadline, deltas=None, radius=25):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005]
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-13:
                            best_c, any_imp = cv, True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B at initial resolution
    t1 = time.time() + min(200, (deadline - t0) * 0.25)
    h_o, c_o = opt_lbfgsb(best_h, n_init, t1,

    if c_o < best_c:

    # Phase 2: Coordinate descent
    t2 = time.time() + min(180, (deadline - time.time()) * 0.2)
    h_cd, c_cd = coord_desc(best_h, n_init, max_passes=300, grid=80, deadline=t2, penalty=2.0)
    h_cd = project_feasible(h_cd, n_init)
    c_cd = compute_c5(h_cd, n_init)
    if c_cd < best_c:
        best_h, best_c = h_cd, c_cd
    # Phase 3: Try n=200 if different
    if n_init != 200 and time.time() < deadline - 120:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, 200)
        h_200 = project_feasible(np.interp(x_new, x_old, best_h), 200)
        t3 = time.time() + min(120, (deadline - time.time()) * 0.25)
        h_o, c_o = opt_lbfgsb(h_200, 200, t3, alphas=[100, 500, 2000], lam=50, maxiter=300)
        ho, co = coord_desc(h_o, 200, max_passes=150, grid=60, deadline=deadline - 50, penalty=1.0)

    # Phase 4: Multi-restart
    if time.time() < deadline - 80:

            scale = 0.005 + 0.02 * r / max(1, n_r)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            t_sub = time.time() + min(50, (deadline - time.time() - 25) * 0.5)
            if t_sub - time.time() < 15:

            h_o, c_o = opt_lbfgsb(hp, n, t_sub, alphas=[200, 1000, 5000], lam=50, maxiter=200)
            h_o = project_feasible(h_o, n)
            c_o = compute_c5(h_o, n)
            if c_o < best_c:

    # Phase 5: Final CD + pair
    if time.time() < deadline - 25:
        n = len(best_h)
        t5 = time.time() + min(30, deadline - time.time() - 10)
