- Inner loop: minimize sum_k lambda_k * f_k(h) s.t. g(h)=0, 0<=h<=1

def mirror_descent(h, n, max_outer=200, inner_maxiter=500, eta=0.1):
    h = h.copy()
    c = np.correlate(h, 1-h, mode='full')
    # Initialize lambda (distribution over lags)
    lambda_vec = np.ones(2*n-1) / (2*n-1)
    for outer in range(max_outer):
        # Inner: minimize sum_k lambda_k * c[k] subject to constraints
        # This is a quadratic program
        def obj(x):
            c_x = np.correlate(x, 1-x, mode='full')
            return float(np.dot(lambda_vec, c_x))
        def jac(x):
            # Gradient of sum_k lambda_k * c[k] w.r.t. x
            # c[k] = sum_i x[i]*(1-x[i+k-(n-1)])
            # dc[k]/dx[p] = (1-x[p+k-(n-1)]) for valid k where p is first index
            #              - x[p-(n-1-k)] for valid k where p is second index
            grad = np.zeros(n)
            for k in range(2*n-1):
                if lambda_vec[k] == 0:
                    continue
                m = k - (n-1)  # shift
                # p as first index: i=p, need 0<=p<n and 0<=p+m<n
                lo1 = max(0, -m)
                hi1 = min(n, n-m)
                if lo1 < hi1:
                    grad[lo1:hi1] += lambda_vec[k] * (1 - x[lo1+m:hi1+m])
                # p as second index: i+m=p, i=p-m, need 0<=p-m<n and 0<=p<n
                lo2 = max(0, m)
                hi2 = min(n, n+m)
                if lo2 < hi2:
                    grad[lo2:hi2] -= lambda_vec[k] * x[lo2-m:hi2-m]
            return grad
        # Use SLSQP to solve the QP
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2}],
                      options={'maxiter': inner_maxiter})
        h = project_feasible(res.x, n)
        
        # Outer: update lambda
        c = np.correlate(h, 1-h, mode='full')
        # Gradient ascent: lambda <- softmax(eta * c)
        c_shifted = c - c.max()
        lambda_vec = np.exp(eta * c_shifted)
        lambda_vec /= lambda_vec.sum()
        
        eta *= 0.995  # Slowly decrease step size
    return h, compute_c5(h, n)

```python
def smooth_obj_grad(x, n, alpha):
    c = np.correlate(x, 1-x, mode='full')
    dx = 2.0/n
    m = float(c.max())
    k_max = int(np.argmax(c))
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    val = (m + log_sum/alpha) * dx
    w = exp_s / exp_s.sum()
    weights = alpha * w
    weights[k_max] += 1.0
    # Compute gradient: sum_j weights[j] * dc_j/dx[p] for each p
    # dc_j/dx[p] = (1-x[p-m_j])*[0<=p-m_j<n] - x[p+m_j]*[0<=p+m_j<n]
    # where m_j = j-(n-1)
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # m_j for each j
    # First term: sum_j weights[j] * (1-x[p-m_j]) for valid p-m_j
    # For each j, p-m_j is valid for p in [m_j, m_j+n-1] ∩ [0,n-1]
    # = [max(0,m_j), min(n-1, m_j+n-1)]
    # Second term: sum_j weights[j] * x[p+m_j] for valid p+m_j
    # For each j, p+m_j is valid for p in [-m_j, n-1-m_j] ∩ [0,n-1]
    # = [max(0,-m_j), min(n-1, n-1-m_j)]
    for p in range(n):
        # First term: indices j where 0 <= p-shifts[j] < n
        # i.e., shifts[j] in [p-n+1, p]
        mask1 = (shifts >= p-n+1) & (shifts <= p)
        idx1 = np.where(mask1)[0]
        if len(idx1) > 0:
            grad[p] += weights[idx1] @ (1-x[p-shifts[idx1]])
        # Second term: indices j where 0 <= p+shifts[j] < n
        mask2 = (shifts >= -p) & (shifts <= n-1-p)
        idx2 = np.where(mask2)[0]
        if len(idx2) > 0:
            grad[p] -= weights[idx2] @ x[p+shifts[idx2]]
    grad *= dx
    return val, grad
```

This is O(n * (2n-1)) per gradient = O(2n²) = 180K for n=300. Each gradient evaluation involves a few numpy operations on arrays of size ~n, so it's fast.
