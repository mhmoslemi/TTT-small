def continuous_optimize(h, n, deadline):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in [5000, 20000, 100000, 500000, 2000000]:
        if time.time() > deadline - 5:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full').astype(float) * dx
            m = g.max()
            sh = alpha * (g - m)
            # Clip for numerical stability
            sh = np.clip(sh, -500, 500)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def obj_grad(x):
            g = np.correlate(x, 1 - x, mode='full').astype(float) * dx
            m = g.max()
            sh = alpha * (g - m)
            sh = np.clip(sh, -500, 500)

                for p in range(n):

                    if 0 <= p - s < n:
                        gp += 1.0 - x[p - s]
                    if 0 <= p + s < n:
                        gp -= x[p + s]
                    grad[p] += wk * dx * gp
            return grad
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]

                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    
    return best_h, best_c

            if stall > 15:

                for a in range(nk):
                    ri = rng.randint(len(zeros))
                    rj = rng.randint(len(ones))
                    h[zeros[ri]] = 1.0
                    h[ones[rj]] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones = np.where(h == 1)[0].astype(np.int64).copy()
                zeros = np.where(h == 0)[0].astype(np.int64).copy()
                temp = max(0.001, temp * 0.93)
                stall = 0
    return best_h, best_c

def subgrad_polish(h, n, deadline):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.5
    no_imp = 0
    t_end = deadline
    while time.time() < t_end:
        corr = get_corr(h, n)
        c_val = corr.max()
        active = np.where(corr >= c_val - 1e-8)[0]
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            r1_lo, r1_hi = max(0, s), min(n, s + n)
            if r1_lo < r1_hi:
                idx = np.arange(r1_lo, r1_hi)
                grad[r1_lo:r1_hi] += (1.0 - h[idx - s])
            r2_lo, r2_hi = max(0, -s), min(n, n - s)
            if r2_lo < r2_hi:
                idx2 = np.arange(r2_lo, r2_hi)
                grad[r2_lo:r2_hi] -= h[idx2 + s]
        grad *= dx / len(active)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        h = h_new
        if no_imp > 30:
            step *= 0.85
            no_imp = 0
        if step < 1e-6:
            break
    return best_h, best_c

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary search from parent (25%)
    p1 = t0 + 0.25 * (budget_s - 5)
    hb, cb = binary_local_search(h0, n, p1, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Continuous optimization (35%)
    p2 = t0 + 0.60 * (budget_s - 5)
    if time.time() < p2:
        h_cont = project_feasible(best_h + 0.03 * rng.randn(n), n)
        hc, cc = continuous_polish(h_cont, n, p2)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

    # Phase 3: Subgradient polish (15%)
    p3 = t0 + 0.75 * (budget_s - 5)
    if time.time() < p3:
        hs, cs = subgrad_polish(best_h, n, p3)
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()

    # Phase 4: Binary from best (10%)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        hb4, cb4 = binary_local_search(best_h, n, p4, seed=seed + 700)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()

    # Phase 5: Final continuous polish (10%)
    if time.time() < dl - 3:
        h_fin = project_feasible(best_h + 0.01 * rng.randn(n), n)
        hf, cf = continuous_polish(h_fin, n, dl - 2)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Also try from Thue-Morse if time permits
    if time.time() < dl - 30 and best_c > 0.3808:
        h_tm = thue_morse(n)
        target = n // 2
        num_ones = int(h_tm.sum())
        if num_ones > target:
            oi = np.where(h_tm == 1)[0]
            rng.shuffle(oi)
            h_tm[oi[:num_ones - target]] = 0
        elif num_ones < target:
            zi = np.where(h_tm == 0)[0]
            rng.shuffle(zi)
            h_tm[zi[:target - num_ones]] = 1
        rem = dl - time.time() - 5
        if rem > 20:
            hb_t, cb_t = binary_local_search(h_tm, n, time.time() + rem * 0.5, seed=seed + 999)
            if cb_t < best_c:
                best_c = cb_t
                best_h = hb_t.copy()
            if time.time() < dl - 5:
                hc_t = project_feasible(best_h + 0.02 * rng.randn(n), n)
                hc_t, cc_t = continuous_polish(hc_t, n, dl - 2)
                if cc_t < best_c:
                    best_c = cc_t
                    best_h = hc_t.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
