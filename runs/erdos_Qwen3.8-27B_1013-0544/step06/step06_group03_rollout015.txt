def project_feasible(h,n):
    h=np.clip(np.asarray(h,float).copy(),0.0,1.0)
    tgt=0.5*n
    for _ in range(25):
        d=tgt-h.sum()
        if abs(d)<1e-13: break
        if d>0:
            r=1.0-h; t=r.sum()
            h += (d/t)*r if t>1e-14 else d/n
        else:
            t=h.sum()
            h += (d/t)*h if t>1e-14 else -d/n

            i=int(np.argmax(h)); a=min(-d,h[i]); h[i]-=a if a>0 else 0

def try_transfer(h,n,p,q,fracs,best):
    p=int(p); q=int(q)
    if p==q: return h,best,False

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(25):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (d / tot) * room
            else:
                h += d / n
        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (d / tot) * h
            else:
                h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    for _ in range(8):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            i = int(np.argmax(1.0 - h))
            a = min(d, 1.0 - h[i])
            if a > 1e-18:
                h[i] += a
        else:
            i = int(np.argmax(h))
            a = min(-d, h[i])
            if a > 1e-18:
                h[i] -= a
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_all_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))


def resample_h(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    if m == 1:
        return np.full(n, float(h[0]))
    xo = (np.arange(m) + 0.5) * (2.0 / m)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xn, xo, h)


def fallback_h(n, rng):
    h = np.empty(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    h = 0.6 * h + 0.4 * rng.rand(n)
    return project_feasible(h, n)


def soft_top(c, alpha, idx):
    a = c[idx]
    m = float(a.max())
    s = alpha * (a - m)
    mx = float(s.max())
    z = s - mx
    e = np.exp(z)
    ss = float(e.sum())
    return m + (mx + np.log(ss)) / alpha, e / ss


def weighted_gradient(h, n, w, idx):
    g = np.zeros(n)
    for wk, ix in zip(w, idx):
        f = int(ix) - (n - 1)
        wk = float(wk)
        if f == 0:
            g += wk * (1.0 - 2.0 * h)
        elif f > 0:
            nm = n - f
            g[:nm] -= wk * h[f:]
            g[f:] += wk * (1.0 - h[:nm])
        else:
            m = -f
            nm = n - m
            g[:nm] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:nm]
    return g


def active_obj(x, n, alpha, idx):
    return float(soft_top(compute_all_lags(x, n), alpha, idx)[0])


def active_jac(x, n, alpha, idx):
    return weighted_gradient(x, n, soft_top(compute_all_lags(x, n), alpha, idx)[1], idx)


def sum_fun(x, n):
    return float(np.sum(x) - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n)


def slsqp_active(h, n, alpha, tc, mi):
    idx = np.argsort(compute_all_lags(h, n))[-tc:]
    con = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        r = minimize(active_obj, h, args=(n, alpha, idx), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, jac=active_jac, constraints=con,
                     options={'maxiter': mi, 'ftol': 1e-12})
        return project_feasible(r.x, n)
    except Exception:
        return h


def slsqp_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = compute_c5(h, n)
    cfg = ((2000, 8), (15000, 10), (120000, 8))
    maxruns = 3 if n <= 350 else (2 if n <= 600 else 1)
    for al, tc in cfg[:maxruns]:
        if time.time() > deadline - 6:
            break
        mi = 450 if n <= 350 else (250 if n <= 600 else 120)
        hn = slsqp_active(h, n, al, tc, mi)
        cn = compute_c5(hn, n)
        if cn < best - 1e-13:
            h, best = hn, cn
    return h, best


def binary_counts(b, n):
    return np.correlate(b, 1.0 - b, mode='full')


def binary_score(cnt):
    mx = int(cnt.max())
    k = min(12, len(cnt))
    if k >= cnt.size:
        top = int(cnt.sum())
    else:
        top = int(np.sum(np.partition(cnt, -k)[-k:]))
    return mx, top


def binary_polish(h, n, deadline, target_c, rng):
    if n % 2:
        return h.copy(), compute_c5(h, n)
    m = n // 2
    order = np.argsort(h)
    b = np.zeros(n)
    b[order[n - m:]] = 1.0
    b = project_feasible(b, n)
    cnt = binary_counts(b, n)
    bm, bt = binary_score(cnt)
    if bm * 2.0 / n <= target_c + 1e-12:
        return b, compute_c5(b, n)
    stall = 0
    while time.time() < deadline:
        cnt = binary_counts(b, n)
        idxs = np.argsort(cnt)[-8:]
        cands = set()
        for ix in idxs:
            f = int(ix) - (n - 1)
            if f > 0:
                nm = n - f
                contrib = b[f:] * (1.0 - b[:nm])
                kk = min(5, nm)
                if kk:
                    pp = np.argsort(contrib)[-kk:]
                    for p in pp:
                        i = int(p)
                        j = i + f
                        if b[i] < 0.5 and b[j] > 0.5:
                            cands.add((i, j))
            elif f < 0:
                mm = -f
                nm = n - mm
                contrib = b[:nm] * (1.0 - b[mm:])
                kk = min(5, nm)
                if kk:
                    pp = np.argsort(contrib)[-kk:]
                    for p in pp:
                        i = int(p)
                        j = i + mm
                        if b[i] > 0.5 and b[j] < 0.5:
                            cands.add((i, j))
        ones = np.nonzero(b > 0.5)[0]
        zeros = np.nonzero(b < 0.5)[0]
        if ones.size and zeros.size:
            for _ in range(12):
                cands.add((int(rng.choice(ones)), int(rng.choice(zeros))))
        improved = False
        for i, j in list(cands)[:100]:
            if time.time() > deadline:
                break
            if b[i] == b[j]:
                continue
            oi, oj = b[i], b[j]
            b[i], b[j] = oj, oi
            cnt2 = binary_counts(b, n)
            mx2, tp2 = binary_score(cnt2)
            if mx2 < bm or (mx2 == bm and tp2 < bt):
                bm, bt = mx2, tp2
                improved = True
                break
            b[i], b[j] = oi, oj
        if improved:
            stall = 0
            if bm * 2.0 / n <= target_c + 1e-12:
                break
        else:
            stall += 1
            if stall > 25:
                break
    b = project_feasible(b, n)
    return b, compute_c5(b, n)


def try_transfer(h, n, p, q, fracs, best):
    p = int(p)
    q = int(q)
    if p == q:
        return h, best, False
    md = min(float(h[q]), 1.0 - float(h[p]))
    if md <= 1e-9:
        return h, best, False
    for frac in fracs:
        d = frac * md
        op, oq = float(h[p]), float(h[q])
        h[p] += d
        h[q] -= d
        c = compute_c5(h, n)
        if c < best - 1e-12:
            return h, c, True
        h[p], h[q] = op, oq
    return h, best, False
