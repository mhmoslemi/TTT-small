    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c * dx
            best_j = -1
            best_nc = cur_c + 1
            for j in ones:
                if i == j:
                    continue
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max()
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + swap_delta(h, n, i, best_j, m_range)
                # Recompute to avoid accumulation error
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max()
                improved = True
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            # Perturb to escape local optimum
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                a = oi[rng.randint(len(oi))]
                b = zi[rng.randint(len(zi))]
                h[a] = 0.0
                h[b] = 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max()
    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < deadline and res.success:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Quick SLSQP to get a good starting point
    p1_end = t0 + 0.08 * (budget_s - 8)
    h_s = project_feasible(h0 + 0.05 * rng.randn(n0), n0)
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1_end:
            break
        h_s = slsqp_smooth(h_s, n0, alpha, 800, 1e-14, p1_end)
        c = compute_c5(h_s, n0)
        if c < best_c:
            best_c = c
            best_h = h_s.copy()
            best_n = n0

    # Phase 2: Binary SA at multiple resolutions
    p2_end = t0 + 0.72 * (budget_s - 8)
    n_list = [n0]
    if n0 == 250:
        n_list = [250, 200, 300]
    elif n0 <= 200:
        n_list = [n0, 250, 300]
    else:
        n_list = [n0, 250]

    time_per_n = (p2_end - time.time()) / len(n_list)
    for ni in n_list:
        if time.time() > p2_end:
            break
        n_trial = ni
        h_init = project_feasible(np.interp(
            np.linspace(0, 2, n_trial), np.linspace(0, 2, best_n), best_h
        ), n_trial)
        h_bin = round_to_binary(h_init, n_trial)
        # Also try random and perturbed starts
        starts = [h_bin]
        h_rand = np.zeros(n_trial)
        idx_rand = rng.permutation(n_trial)
        h_rand[idx_rand[:n_trial // 2]] = 1.0
        starts.append(h_rand)
        h_pert = round_to_binary(h_init + 0.15 * rng.randn(n_trial), n_trial)
        starts.append(h_pert)

        t_per_start = time_per_n / len(starts)
        for si, start in enumerate(starts):
            if time.time() > p2_end:
                break
            end_t = min(time.time() + t_per_start, p2_end)
            hb, cb = binary_sa(start, n_trial, end_t,
                               seed=seed + si * 100 + n_trial,
                               t_start=0.03, patience=25000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                best_n = n_trial

    # Phase 3: Greedy polish on best
    p3_end = t0 + 0.85 * (budget_s - 8)
    hg, cg = greedy_binary(best_h, best_n, p3_end, max_passes=15, seed=seed + 50)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 4: Final SLSQP smoothing (non-binary might beat binary)
    p4_end = dl - 5
    if time.time() < p4_end - 10:
        h_final = project_feasible(best_h + 0.02 * rng.randn(best_n), best_n)
        for alpha in [10000, 100000, 1000000]:
            if time.time() > p4_end:
                break
            h_final = slsqp_smooth(h_final, best_n, alpha, 1500, 1e-14, p4_end)
            c = compute_c5(h_final, best_n)
            if c < best_c:
                best_c = c
                best_h = h_final.copy()

    # Phase 5: One more SA round if time
    if time.time() < dl - 20:
        h_bin2 = round_to_binary(best_h, best_n)
        hb2, cb2 = binary_sa(h_bin2, best_n, dl - 15, seed=seed + 999,
                             t_start=0.02, patience=20000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        if time.time() < dl - 5:
            hg2, cg2 = greedy_binary(best_h, best_n, dl - 3, max_passes=8, seed=seed + 800)
            if cg2 < best_c:
                best_c = cg2
                best_h = hg2.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
