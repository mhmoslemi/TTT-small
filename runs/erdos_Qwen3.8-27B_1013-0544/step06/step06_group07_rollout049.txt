import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def fast_local_search(h, n, max_sweeps=50, max_iter_per_sweep=200):
    """First-improvement local search for binary h."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        count = 0
        for _ in range(max_iter_per_sweep):
            found = False
            # Try random swaps until we find an improvement
            for attempt in range(10):
                i = zeros[np.random.randint(len(zeros))]
                j = ones[np.random.randint(len(ones))]
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    found = True
                    break
            if not found:

        if not improved:

    if cur_c < best_c:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.02, patience=20000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    deltas_cache = {}
    step = 0
    cool_rate = (0.00001 / t_start) ** (1.0 / 200000.0)
    T = t_start
    no_improve = 0
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        step += 1
        if step % 5000 == 0:
            T *= cool_rate
        if no_improve > patience:
            # Restart with perturbation
            nk = rng.randint(3, 10)
            z = np.where(h == 0)[0]
            o = np.where(h == 1)[0]
            for _ in range(nk):
                a = z[rng.randint(len(z))]
                b = o[rng.randint(len(o))]
                h[a], h[b] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            T = t_start
            no_improve = 0
    
    return best_h, best_c

def targeted_search(h, n, deadline, seed=42):
    """Targeted local search focusing on dominant lags."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    while time.time() < deadline:
        # Find top lags
        top_k = np.argsort(corr)[-3:]
        for m_idx in top_k:
            m_lag = int(m_idx - (n - 1))
            if m_lag == 0:
                continue
            # Find positions contributing to this lag
            if m_lag > 0:
                valid_i = np.arange(0, n - m_lag)
            else:
                valid_i = np.arange(-m_lag, n)
            # Positions where h[i]=1 and h[i+m_lag]=0
            ones_i = valid_i[h[valid_i] == 1]
            zeros_t = ones_i[h[ones_i + m_lag] == 0]
            if len(zeros_t) == 0:
                continue
            # Try to flip target positions
            target_pos = zeros_t[rng.randint(len(zeros_t))]
            ones = np.where(h == 1)[0]
            j = ones[rng.randint(len(ones))]
            if j == target_pos:
                continue
            delta = swap_delta(h, n, target_pos, j, m_range)
            nc = (corr + delta).max() * dx
            if nc < cur_c:
                h[target_pos] = 1.0
                h[j] = 0.0
                corr = corr + delta
                cur_c = nc
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
    
    return best_h, best_c

def greedy_binary(h_init, n, deadline, seed=42):
    """Greedy first-improvement local search."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    while time.time() < deadline:
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # Try all zero positions
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Perturb and continue
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            nk = rng.randint(3, 8)
            z = np.where(h == 0)[0]
            o = np.where(h == 1)[0]

                a = z[rng.randint(len(z))]
                b = o[rng.randint(len(o))]
                h[a], h[b] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
