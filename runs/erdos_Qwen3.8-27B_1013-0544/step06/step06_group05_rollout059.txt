def slsqp_smooth(h, n, alpha, maxiter, deadline):

    def obj(x):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

E[i, L_idx] represents the change in corr[L_idx] when we turn h[zeros_idx[i]] from 0 to 1.

When we set h[i] = 1 (was 0):
- The term k=i: changes from 0 * (1-h[i-L]) to 1 * (1-h[i-L]). Delta: +(1-h[i-L]) if 0 ≤ i-L < n
- The term k where k-L = i, i.e., k = i+L: changes from h[i+L] * (1-0) to h[i+L] * (1-1) = 0. Delta: -h[i+L] if 0 ≤ i+L < n

        diff = target - h.sum()

            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[sorted_idx[:n // 2]] = 1.0

def vectorized_local_search(h, n, deadline, max_sweeps=100000):
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    n2 = 2 * n - 1
    L_vals = np.arange(n2, dtype=np.float64) - (n - 1)
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros_idx = np.where(h < 0.5)[0]
        ones_idx = np.where(h > 0.5)[0]
        nz = len(zeros_idx)
        no = len(ones_idx)
        if nz == 0 or no == 0:
            break

        i_arr = zeros_idx.astype(np.float64)
        j_arr = ones_idx.astype(np.float64)

        i_mL = i_arr[:, None] - L_vals[None, :]
        m1 = (i_mL >= 0) & (i_mL < n)
        idx1 = i_mL.astype(np.int32).clip(0, n - 1)
        E1 = np.where(m1, 1.0 - h[idx1], 0.0)

        i_pL = i_arr[:, None] + L_vals[None, :]
        m2 = (i_pL >= 0) & (i_pL < n)
        idx2 = i_pL.astype(np.int32).clip(0, n - 1)
        E2 = np.where(m2, h[idx2], 0.0)
        E = E1 - E2

        j_mL = j_arr[:, None] - L_vals[None, :]
        m3 = (j_mL >= 0) & (j_mL < n)
        idx3 = j_mL.astype(np.int32).clip(0, n - 1)
        F1 = np.where(m3, 1.0 - h[idx3], 0.0)

        j_pL = j_arr[:, None] + L_vals[None, :]
        m4 = (j_pL >= 0) & (j_pL < n)
        idx4 = j_pL.astype(np.int32).clip(0, n - 1)
        F2 = np.where(m4, h[idx4], 0.0)
        F = -F1 + F2

        best_nm = cur_max
        best_i = -1
        best_j = -1
        for ii in range(nz):

        cur_max = corr.max()
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()

                       constraints=cons_c, options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h

    n2 = 2 * n - 1
    L_vals = np.arange(n2, dtype=np.float64) - (n - 1)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros_idx = np.where(h < 0.5)[0]
        ones_idx = np.where(h > 0.5)[0]
        nz = len(zeros_idx)
        no = len(ones_idx)
        if nz == 0 or no == 0:
            break
        i_arr = zeros_idx.astype(np.float64)
        j_arr = ones_idx.astype(np.float64)
        i_mL = i_arr[:, None] - L_vals[None, :]
        m1 = (i_mL >= 0) & (i_mL < n)
        idx1 = i_mL.astype(np.int32).clip(0, n - 1)
        E1 = np.where(m1, 1.0 - h[idx1], 0.0)
        i_pL = i_arr[:, None] + L_vals[None, :]
        m2 = (i_pL >= 0) & (i_pL < n)
        idx2 = i_pL.astype(np.int32).clip(0, n - 1)
        E2 = np.where(m2, h[idx2], 0.0)
        E = E1 - E2
        j_mL = j_arr[:, None] - L_vals[None, :]
        m3 = (j_mL >= 0) & (j_mL < n)
        idx3 = j_mL.astype(np.int32).clip(0, n - 1)
        F1 = np.where(m3, 1.0 - h[idx3], 0.0)
        j_pL = j_arr[:, None] + L_vals[None, :]
        m4 = (j_pL >= 0) & (j_pL < n)
        idx4 = j_pL.astype(np.int32).clip(0, n - 1)
        F2 = np.where(m4, h[idx4], 0.0)
        F = -F1 + F2
        best_nm = cur_max
        best_i = -1
        best_j = -1
        for ii in range(nz):
            G = corr + E[ii]
            nms = (G[None, :] + F).max(axis=1)
            mv = nms.min()
            if mv < best_nm - 1e-12:
                best_nm = mv
                best_i = ii
                best_j = nms.argmin()
        if best_i == -1:
            break
        h[zeros_idx[best_i]] = 1.0
        h[ones_idx[best_j]] = 0.0
        corr += E[best_i] + F[best_j]
        cur_max = corr.max()
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    return best_h, best_max * dx

def perturb(h, n, k, rng):
    h = h.copy()
    for _ in range(k):
        o = np.where(h > 0.5)[0]
        z = np.where(h < 0.5)[0]
        if len(o) == 0 or len(z) == 0:
            break
        h[rng.choice(z)] = 1.0
        h[rng.choice(o)] = 0.0
    return h

class SoftMaxObj:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
    def __call__(self, x):
        dx = 2.0 / self.n
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = self.alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / self.alpha

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return x.sum() - self.n / 2.0

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    obj_fn = SoftMaxObj(n, alpha)
    cons_fn = SumConstraint(n)
    bounds = [(0.0, 1.0)] * n
    cons_c = [{'type': 'eq', 'fun': cons_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons_c, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h_bin = binarize(h0, n)
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Main vectorized local search (45% budget)
    p1 = t0 + 0.45 * (budget_s - 5)
    hb, cb = vectorized_local_search(best_h, n, p1, max_sweeps=200000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 2: Iterated Local Search (35% budget)
    p2 = t0 + 0.80 * (budget_s - 5)
    n_rest = 35
    for r in range(n_rest):
        if time.time() > p2:
            break
        k = rng.randint(4, 13)
        h_p = perturb(best_h, n, k, rng)
        c_p = compute_c5(h_p, n)
        if c_p > best_c * 1.6:
            continue
        rem = p2 - time.time()
        tp = rem / max(1, (n_rest - r))
        hp, cp = vectorized_local_search(h_p, n, t0 + time.time() + tp, max_sweeps=40000)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    # Phase 3: SLSQP continuous polish + re-binarize (15% budget)
    p3 = dl
    if time.time() < p3:
        for alpha in [50000, 500000]:
            if time.time() > p3 - 5:
                break
            h_s = slsqp_smooth(best_h.copy(), n, alpha, 3000, p3)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()
            h_b = binarize(h_s, n)
            c_b = compute_c5(h_b, n)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b.copy()
            if time.time() < p3 - 3:
                hb2, cb2 = vectorized_local_search(best_h, n, p3, max_sweeps=30000)
                if cb2 < best_c:
                    best_c = cb2
                    best_h = hb2.copy()
    # Final verification
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
