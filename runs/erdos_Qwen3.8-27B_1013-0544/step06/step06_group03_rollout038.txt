                            best_c = compute_c5(h, n); improved = True
                        else:
                            h[j] += delta; h[i] -= delta
                    elif h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4)
                        if delta < 1e-10:
                            continue
                        h[j] += delta; h[i] -= delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n); improved = True
                        else:
                            h[j] -= delta; h[i] += delta
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 25:
        for alpha in [50, 500, 5000, 50000]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_step(best_h, n, alpha, maxiter=2500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new

    if time.time() < deadline - 15:
        for top_m in [3, 5, 10, 15]:
            if time.time() > deadline - 15:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_m:]
            for alpha in [10000, 100000, 500000]:
                if time.time() > deadline - 15:
                    break
                h_new = slsqp_step(best_h, n, alpha, top_idx=top_idx, maxiter=2000, ftol=1e-16)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new

    if time.time() < deadline - 8:
        h_new, c_new = grad_descent(best_h, n, min(deadline - 4, time.time() + 25), max_rounds=400)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 2:
        best_h, best_c = fine_search(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    n_candidates = [200, 250, 350, 400, 500]
    if h_init is not None and n_init not in n_candidates:
        n_candidates.insert(0, n_init)

    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 50:
            break
        alloc = min(70, remaining * 0.22)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        attempts = 0
        while time.time() < deadline - 35 and attempts < 8:
            attempts += 1
            scale = 0.008 + 0.004 * rng.random()
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 30) * 0.45)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_step(best_h, n, 1000000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = grad_descent(best_h, n, min(deadline - 2, time.time() + 15), max_rounds=200)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = fine_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    return best_h, best_c, len(best_h)
