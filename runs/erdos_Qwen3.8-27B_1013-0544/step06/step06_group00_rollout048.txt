import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def swap_delta(i, j, h, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def greedy_binary(h_init, n, deadline, max_sweeps=500, seed=42):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    rng = np.random.RandomState(seed)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = zeros_list[:]
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_delta = None
            for j in ones_list:
                delta = swap_delta(i, j, h, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def coord_descent(h_init, n, n_sweeps, deadline, seed=42):
    h = h_init.copy()
    target = n / 2.0
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    offset = n - 1
    rng = np.random.RandomState(seed)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = list(range(n))
        rng.shuffle(order)
        for j in order:
            if time.time() > deadline:
                break
            old_val = h[j]
            d = np.zeros(2 * n - 1)
            m_lo = max(j - (n - 1), -(n - 1))
            m_hi = min(j, n - 1)
            if m_lo <= m_hi:
                ar = np.arange(m_lo, m_hi + 1)
                d[ar + offset] += (1.0 - h[j - ar])
            m_lo2 = max(-j, -(n - 1))
            m_hi2 = min(n - 1 - j, n - 1)
            if m_lo2 <= m_hi2:
                ar = np.arange(m_lo2, m_hi2 + 1)
                d[ar + offset] -= h[j + ar]
            a = c - old_val * d
            pos_mask = d > 1e-14
            neg_mask = d < -1e-14
            lo_t = np.minimum(a, a + d).max()
            hi_t = np.maximum(a, a + d).max()
            if hi_t - lo_t < 1e-15:
                v_opt = old_val
            else:
                for _ in range(18):
                    mid_t = (lo_t + hi_t) / 2.0
                    v_lo = 0.0
                    v_hi = 1.0
                    if np.any(pos_mask):
                        tmp = (mid_t - a[pos_mask]) / d[pos_mask]
                        if np.any(tmp < v_hi):
                            v_hi = tmp.min()
                    if np.any(neg_mask):
                        tmp = (mid_t - a[neg_mask]) / d[neg_mask]
                        if np.any(tmp > v_lo):
                            v_lo = tmp.max()
                    if v_lo > v_hi:
                        lo_t = mid_t
                    else:
                        hi_t = mid_t
                t_opt = hi_t
                v_lo = 0.0
                v_hi = 1.0
                if np.any(pos_mask):
                    v_hi = min(v_hi, ((t_opt - a[pos_mask]) / d[pos_mask]).min())
                if np.any(neg_mask):
                    v_lo = max(v_lo, ((t_opt - a[neg_mask]) / d[neg_mask]).max())
                v_lo = max(0.0, min(v_lo, 1.0))
                v_hi = max(0.0, min(v_hi, 1.0))
                v_opt = (v_lo + v_hi) / 2.0 if v_lo <= v_hi else old_val
            if abs(v_opt - old_val) > 1e-12:
                c += (v_opt - old_val) * d
                h[j] = v_opt
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full').astype(float)
    return h

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    from scipy.optimize import minimize
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
