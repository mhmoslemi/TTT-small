def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0

def swap_delta(h, n, i, j, m_range):

    im = i + m_range
    msk = (im >= 0) & (im < n)

    imn = i - m_range
    msk = (imn >= 0) & (imn < n)

    jm = j + m_range

            T = max(T, 0.015)
    return best_h, best_c

def greedy_polish(h_init, n, deadline, m_range=None, seed=42):
    if m_range is None:
        m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)

    for _sweep in range(30):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_dc = 1.0
        best_i, best_j = -1, -1
        for i in zeros:
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_dc:
                    best_dc = dc
                    best_i, best_j = i, j
        if best_dc < -1e-14 and best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            delta = swap_delta(h, n, best_i, best_j, m_range)
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = corr - delta
            cur_c += best_dc
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        if not improved:
            break

    # Randomized 2-swap polish
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    for _ in range(200000):
        if time.time() > deadline:
            break
        i1 = zeros[rng.randint(len(zeros))]
        j1 = ones[rng.randint(len(ones))]
        mi = zeros != i1
        mj = ones != j1
        if mi.sum() < 1 or mj.sum() < 1:
            break
        i2 = zeros[mi][rng.randint(mi.sum())]
        j2 = ones[mj][rng.randint(mj.sum())]
        h[i1], h[j1] = 1.0, 0.0
        h[i2], h[j2] = 1.0, 0.0
        new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
        nc = new_corr.max() * dx
        if nc < cur_c - 1e-14:
            corr = new_corr
            cur_c = nc
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i1], h[j1] = 0.0, 1.0
            h[i2], h[j2] = 0.0, 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    m_range = np.arange(-(n - 1), n)

    # Phase 1: SLSQP (8%)
    t1 = t0 + 0.08 * (budget_s - 3)
    if time.time() < t1:
        h = best_h.copy()
        for alpha in [2000, 20000, 200000]:
            if time.time() > t1:
                break
            h = slsqp_smooth(h, n, alpha, 800, 1e-14)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 2: Enhanced SA (72%)
    t2 = t0 + 0.80 * (budget_s - 3)
    h_bin = round_to_binary(best_h, n)
    starts = [h_bin]
    starts.append(round_to_binary(best_h + 0.12 * rng.randn(n), n))
    starts.append(round_to_binary(best_h + 0.25 * rng.randn(n), n))
    sb = h_bin.copy()
    kick = rng.choice(n, size=n // 8, replace=False)
    sb[kick] = 1.0 - sb[kick]
    starts.append(sb)
    starts.append(round_to_binary(0.5 + 0.4 * np.cos(2 * np.pi * rng.uniform() * np.arange(n) / n + rng.uniform(2*np.pi)), n))

    n_attempts = len(starts)
    for att in range(n_attempts):
        frac_end = 0.08 + 0.72 * (att + 1) / n_attempts
        att_dl = t0 + frac_end * (budget_s - 3)
        if time.time() > att_dl:
            break
        s = seed + att * 77
        ts = 0.022 + att * 0.003
        hb, cb = enhanced_sa(starts[att], n, att_dl, seed=s, m_range=m_range, t0_temp=ts)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy polish (14%)
    t3 = t0 + 0.94 * (budget_s - 3)
    if time.time() < t3:
        hg, cg = greedy_polish(best_h, n, t3, m_range=m_range, seed=seed + 999)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Phase 4: Quick SA restart from best (6% remaining)
    if time.time() < dl - 2:
        hb4, cb4 = enhanced_sa(best_h, n, dl - 1, seed=seed + 2000, m_range=m_range, t0_temp=0.008)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
        hg4, cg4 = greedy_polish(best_h, n, dl - 1, m_range=m_range, seed=seed + 3000)
        if cg4 < best_c:
            best_c = cg4
            best_h = hg4.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
