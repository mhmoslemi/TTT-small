            # When h[i] goes 0->1 and h[j] goes 1->0:
            # g[p, k] changes because h changed at positions i and j
            # g[p, k] depends on h[p-kt] and h[p+kt]
            # If p-kt == i: h[i] was 0, now 1, so (1-h[i]) changes from 1 to 0: g[p,k] -= 1
            # If p+kt == i: h[i] was 0, now 1, so h[i] changes from 0 to 1: g[p,k] -= 1
            # If p-kt == j: h[j] was 1, now 0, so (1-h[j]) changes from 0 to 1: g[p,k] += 1

            #   If na == j: (1-h[j]) changes from 0 to 1, so g increases by 1 (if valid_a was true)
            #   If nb == j: h[j] changes from 1 to 0, so g increases by 1 (if valid_b was true)

            h[i] = 1.0
            h[j] = 0.0
            corr = corr + g_i - g_j
            g = build_g(h, n)
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            cur_c = corr.max()
            stall = 0

                best_h = h.copy()

                zeros = np.where(h == 0)[0].copy()

    kick_count = 0
    nm1 = n - 1
    
    while time.time() < deadline:
        gz = g[zeros]
        go = g[ones]
        D = corr[None, None, :] + gz[:, None, :] - go[None, :, :]
        new_maxes = D.max(axis=2)
        flat_min = new_maxes.argmin()
        best_ii = flat_min // len(ones)
        best_jj = flat_min % len(ones)
        best_val = new_maxes[best_ii, best_jj]
        
        i = zeros[best_ii]
        j = ones[best_jj]
        
        if best_val < cur_c - 1e-10:
            # Execute swap
            gi = g[i].copy()
            gj = g[j].copy()
            corr = corr + gi - gj
            update_g(g, i, j, n)
            h[i] = 1.0
            h[j] = 0.0
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            cur_c = corr.max()
            stall = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            stall += 1
            if stall > 10:
                kick_count += 1
                # Random kick
                nk = min(2, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    hi = zeros[ri[a]]
                    hj = ones[rj[a]]
                    g_hi = g[hi].copy()
                    g_hj = g[hj].copy()
                    corr = corr + g_hi - g_hj
                    update_g(g, hi, hj, n)
                    h[hi] = 1.0
                    h[hj] = 0.0
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                cur_c = corr.max()
                stall = 0
    
    return best_h, best_c * dx

def slsqp_optimize(h_init, n, deadline):
    dx = 2.0 / n
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(123)
    
    alphas = [1e5, 5e5, 2e6, 1e7]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        gmat = build_g(best_h, n)
        
        def slsqp_obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            sm = shifted.max()
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        
        def slsqp_grad(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            sm = shifted.max()
            exps = np.exp(shifted - sm)
            softmax = exps / exps.sum()
            # grad = dx * (gmat_old * softmax).sum(axis=1)
            # But gmat changes with x! We need to compute it fresh.
            # Actually, dc_k/dx_i = (2/n) * g[i,k] where g depends on x
            # g[i,k] = (1-x[na])*valid_a - x[nb]*valid_b
            # This is exactly what build_g computes.
            # For efficiency, compute it inline:
            p_idx = np.arange(n)[:, None]
            kt_idx = np.arange(-(n-1), n)[None, :]
            na = p_idx - kt_idx
            nb = p_idx + kt_idx
            valid_a = (na >= 0) & (na < n)
            valid_b = (nb >= 0) & (nb < n)
            ha = np.where(valid_a, x[np.clip(na, 0, n-1)], 0.0)
            hb = np.where(valid_b, x[np.clip(nb, 0, n-1)], 0.0)
            g_cur = (1.0 - ha) * valid_a - hb * valid_b
            return (g_cur * softmax[None, :]).sum(axis=1)
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        
        try:
            res = minimize(slsqp_obj, best_h, jac=slsqp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-16})
            x_opt = project_feasible(res.x, n)
            c_val = compute_c5(x_opt, n)
            if c_val < best_c:
                best_c = c_val
                best_h = x_opt.copy()
        except Exception:
            pass
    
    return best_h, best_c
