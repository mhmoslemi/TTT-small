def flip_deltas(h, n):
    L = 2*n-1
    out = np.zeros((n, L), dtype=np.int16)
    h = h.astype(np.int8)
    for a in range(n):
        d = out[a]
        if a > 0:
            left_d = a - np.arange(a)  # shape a, positive descending
            hleft = h[:a]
            if h[a] == 0:
                idx = n - 1 + left_d[hleft == 0]
                if idx.size: d[idx] += 1
                idx = n - 1 - left_d[hleft == 1]
                if idx.size: d[idx] -= 1
            else:
                idx = n - 1 + left_d[hleft == 0]
                if idx.size: d[idx] -= 1
                idx = n - 1 - left_d[hleft == 1]
                if idx.size: d[idx] += 1
        if a < n-1:
            right_d = np.arange(1, n-a)  # shape n-a-1
            hright = h[a+1:]
            if h[a] == 0:
                idx = n - 1 - right_d[hright == 0]
                if idx.size: d[idx] += 1

                idx = n - 1 + right_d[hright == 1]
                if idx.size: d[idx] += 1
    return out

def soft_obj(x, n, alpha):
    if not np.all(np.isfinite(x)):
        return 1e6
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx

    sh = alpha * (g - m)

    lse = msh + float(np.log(np.exp(sh - msh).sum()))

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize
try:
    from scipy.optimize import milp, LinearConstraint, Bounds
    HAS_MILP = True
except Exception:
    HAS_MILP = False

def c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project(h, n):
    h = np.nan_to_num(np.asarray(h, dtype=float), nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + d)
    return h


def binarize(h, n):
    N = n // 2
    h = np.nan_to_num(np.asarray(h, dtype=float), nan=0.0, posinf=1.0, neginf=0.0)
    idx = np.argsort(-h)[:N]
    b = np.zeros(n, dtype=np.int8)
    if N > 0:
        b[idx] = 1
    return b


def corr_count(h, n):
    a = np.asarray(h, dtype=np.int16)
    return np.correlate(a, 1 - a, mode='full').astype(np.int16, copy=False)


def adjust_sum(b, N):
    b = np.asarray(b, dtype=np.int8).copy()
    d = N - int(b.sum())
    if d > 0:
        idx = np.where(b == 0)[0]
        if idx.size >= d:
            b[idx[:d]] = 1
        else:
            for k in range(len(b)):
                if b[k] == 0 and d > 0:
                    b[k] = 1
                    d -= 1
    elif d < 0:
        idx = np.where(b == 1)[0]
        if idx.size >= -d:
            b[idx[:d]] = 0
        else:
            for k in range(len(b)):
                if b[k] == 1 and d < 0:
                    b[k] = 0
                    d += 1
    return b


def thue(n, shift=0):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return adjust_sum(np.roll(t, int(shift) % n), n // 2)


def random_bin(n, rng):
    b = np.zeros(n, dtype=np.int8)
    if n // 2 > 0:
        b[rng.choice(n, n // 2, replace=False)] = 1
    return b


def pattern_bin(n, mode):
    b = np.zeros(n, dtype=np.int8)
    if mode == 0:
        for i in range(n):
            b[i] = 1 if ((i // 3) % 2 == 0) else 0
    else:
        x = np.linspace(0.0, 4.0 * np.pi, n, endpoint=False)
        b = (np.sin(x) >= 0).astype(np.int8)
    return adjust_sum(b, n // 2)


def resample_parent(n, ph):
    if ph is None:
        return None
    n0 = len(ph)
    if n0 < 2:
        return None
    if n0 == n:
        return ph.astype(float)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x1, x0, ph, left=float(ph[0]), right=float(ph[-1]))


def make_starts(n, ph, seed):
    rng = np.random.RandomState(seed)
    s = [thue(n, 0), thue(n, 1), thue(n, 3), random_bin(n, rng), random_bin(n, rng),
         pattern_bin(n, 0), pattern_bin(n, 1)]
    rp = resample_parent(n, ph)
    if rp is not None:
        s.append(binarize(rp, n))
    return s


def flip_deltas(h, n):
    h = np.asarray(h, dtype=np.int8)
    L = 2 * n - 1
    out = np.zeros((n, L), dtype=np.int16)
    for a in range(n):
        d = out[a]
        if a > 0:
            left_d = a - np.arange(a, dtype=np.int64)
            left = h[:a]
            if h[a] == 0:
                idx = n - 1 + left_d[left == 0]
                if idx.size:
                    d[idx] += 1
                idx = n - 1 - left_d[left == 1]
                if idx.size:
                    d[idx] -= 1
            else:
                idx = n - 1 + left_d[left == 0]
                if idx.size:
                    d[idx] -= 1
                idx = n - 1 - left_d[left == 1]
                if idx.size:
                    d[idx] += 1
        if a < n - 1:
            right_d = np.arange(1, n - a, dtype=np.int64)
            right = h[a + 1:]
            if h[a] == 0:
                idx = n - 1 - right_d[right == 0]
                if idx.size:
                    d[idx] += 1
                idx = n - 1 + right_d[right == 1]
                if idx.size:
                    d[idx] -= 1
            else:
                idx = n - 1 - right_d[right == 0]
                if idx.size:
                    d[idx] -= 1
                idx = n - 1 + right_d[right == 1]
                if idx.size:
                    d[idx] += 1
    return out


def local_search_binary(h0, n, deadline, seed=0, max_sweeps=1000):
    h = binarize(h0, n).astype(np.int8)
    c = corr_count(h, n)
    cur = int(c.max())
    best_h, best_c = h.copy(), cur
    rng = np.random.RandomState(seed)
    sw = 0
    for _ in range(max_sweeps):
        if time.time() >= deadline:
            break
        dl = flip_deltas(h, n)
        z = np.where(h == 0)[0]
        o = np.where(h == 1)[0]
        if z.size == 0 or o.size == 0:
            break
        rng.shuffle(z)
        rng.shuffle(o)
        best_val, best_pair, best_delta = cur, None, None
        for i in z:
            ii = int(i)
            base = c + dl[ii]
            for j in o:
                jj = int(j)
                tmp = base + dl[jj]
                if int(tmp.max()) >= cur:
                    continue
                dd = abs(ii - jj)
                tmp[n - 1 + dd] += 1
                tmp[n - 1 - dd] += 1
                m = int(tmp.max())
                if m < best_val:
                    best_val, best_pair, best_delta = m, (ii, jj), tmp.copy()
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 1
        h[j] = 0
        c = (c + best_delta).astype(np.int16, copy=False)
        cur = int(c.max())
        if cur != best_val or sw > 0 and sw % 40 == 0:
            c = corr_count(h, n)
            cur = int(c.max())
        sw += 1
        if cur < best_c:
            best_c, best_h = cur, h.copy()
    return best_h, best_c


def sa_binary(h0, n, deadline, seed=0, t0=4.0, t1=0.1):
    h = binarize(h0, n).astype(np.int8)
    c = corr_count(h, n)
    cur = int(c.max())
    best_h, best_c = h.copy(), cur
    rng = np.random.RandomState(seed)
    dl = flip_deltas(h, n)
    z = np.where(h == 0)[0]
    o = np.where(h == 1)[0]
    max_it = max(50, int(max(0.0, deadline - time.time()) * 220))
    for it in range(max_it):
        if time.time() >= deadline or z.size == 0 or o.size == 0:
            break
        T = max(1e-3, t0 * (t1 / t0) ** (it / max(1, max_it - 1)))
        i = int(rng.choice(z))
        j = int(rng.choice(o))
        tmp = c + dl[i] + dl[j]
        dd = abs(i - j)
        tmp[n - 1 + dd] += 1
        tmp[n - 1 - dd] += 1
        new = int(tmp.max())
        diff = new - cur
        if diff < 0 or (diff <= 15 and rng.random() < np.exp(-diff / max(T, 1e-3))):
            h[i] = 1
            h[j] = 0
            c = tmp.astype(np.int16, copy=False)
            cur = new
            if cur < best_c:
                best_c, best_h = cur, h.copy()
            dl = flip_deltas(h, n)
            z = np.where(h == 0)[0]
            o = np.where(h == 1)[0]
    return best_h, best_c


def target_count(n):
    return int(np.floor(TARGET * (n / 2.0) + 1e-12))


def build_mip(n, T):
    P = n * (n - 1) // 2
    V = n + P
    pid = -np.ones((n, n), dtype=np.int32)
    y = n
    idx = 0
    for i in range(n - 1):
        pid[i, i + 1:] = np.arange(y + idx, y + idx + n - 1 - i, dtype=np.int32)
        idx += n - 1 - i
    rows, cols, vals, b = [], [], [], []
    for i in range(n - 1):
        for j in range(i + 1, n):
            w = int(pid[i, j])
            r = len(b)
            b.append(0.0)
            rows += [r, r]
            cols += [i, w]
            vals += [-1.0, 1.0]
            r = len(b)
            b.append(0.0)
            rows += [r, r]
            cols += [j, w]
            vals += [-1.0, 1.0]
            r = len(b)
            b.append(1.0)
            rows += [r, r, r]
            cols += [i, j, w]
            vals += [1.0, 1.0, -1.0]
    for d in range(1, n):
        r = len(b)
        b.append(float(T))
        for p in range(n - d):
            w = int(pid[p, p + d])
            rows += [r, r]
            cols += [p + d, w]
            vals += [1.0, -1.0]
        r = len(b)
        b.append(float(T))
        for p in range(n - d):
            w = int(pid[p, p + d])
            rows += [r, r]
            cols += [p, w]
            vals += [1.0, -1.0]
    A = sp.coo_matrix((vals, (rows, cols)), shape=(len(b), V)).tocsr()
    Ae = sp.csr_matrix(np.hstack([np.ones((1, n), dtype=float), np.zeros((1, P), dtype=float)]))
    return A, np.array(b, dtype=float), -np.inf * np.ones(len(b), dtype=float), Ae, float(n // 2)


def solve_mip(n, T, limit_s):
    if not HAS_MILP or limit_s < 5.0:
        return None
    try:
        A, b, lb, Ae, N = build_mip(n, T)
        V = A.shape[1]
        lbv = np.zeros(V, dtype=float)
        ubv = np.ones(V, dtype=float)
        integ = np.zeros(V, dtype=int)
        integ[:n] = 1
        cobj = np.zeros(V, dtype=float)
        res = milp(
            cobj,
            constraints=[LinearConstraint(A, lb, b), LinearConstraint(Ae, N, N)],
            integrality=integ,
            bounds=Bounds(lbv, ubv),
            options={'time_limit': float(max(1.0, limit_s))},
        )
        if res is None or getattr(res, 'x', None) is None:
            return None
        if getattr(res, 'status', 0) not in (0, 8, 1):
            return None
        xb = np.rint(np.asarray(res.x[:n], dtype=float)).astype(np.int8)
        if int(xb.sum()) != n // 2:
            return None
        if np.max(np.abs(np.asarray(res.x[:n], dtype=float) - xb.astype(float))) > 1e-5:
            return None
        cc = corr_count(xb, n)
        if int(cc.max()) <= T + 1e-6:
            return xb
    except Exception:
        return None
    return None


def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)


def soft_obj(x, n, alpha):
    if not np.all(np.isfinite(x)):
        return 1e6
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = float(np.max(g))
    if alpha <= 1.0:
        return m
    sh = alpha * (g - m)
    msh = float(np.max(sh))
    return m + (msh + float(np.log(np.exp(sh - msh).sum()))) / alpha


def smooth_candidate(h0, n, deadline, seed=0):
    h = project(h0, n)
    best_h, best_c = h.copy(), c5(h, n)
    rng = np.random.RandomState(seed)
    hp = project(h + 0.05 * rng.randn(n), n)
    starts = [h.copy()]
    if c5(hp, n) < best_c + 0.05:
        starts.append(hp)
    alphas = (80, 200, 600, 2000) if n <= 150 else (200, 800)
    for base in starts:
        for alpha in alphas:
            if time.time() >= deadline - 5:
                break
            rem = max(1.0, deadline - time.time())
            if n <= 120:
                mi = int(max(20, min(90, rem * 0.18)))
            elif n <= 200:
                mi = int(max(20, min(60, rem * 0.12)))
            else:
                mi = int(max(20, min(45, rem * 0.08)))
            try:
                bounds = [(0.0, 1.0) for _ in range(n)]
                cons = {'type': 'eq', 'fun': sum_constraint, 'args': (n,)}
                res = minimize(
                    soft_obj, base, args=(n, alpha), method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': mi, 'ftol': 1e-12, 'maxls': 20}
                )
                if res is not None and getattr(res, 'x', None) is not None:
                    x = project(np.asarray(res.x, dtype=float), n)
                    cc = c5(x, n)
                    if cc < best_c - 1e-12:
                        best_c, best_h = cc, x.copy()
                        base = x.copy()
            except Exception:
                pass
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(60.0, float(budget_s))
    deadline = t0 + B - 4.0

    best_h, best_c, best_n = None, 1e9, None
    best_for_n = {}
    parent_h, parent_n = None, None
    try:
        ph = np.asarray(initial_h_values, dtype=float)
        if ph.size > 4:
            parent_h = project(ph.copy(), len(ph))
            parent_n = len(parent_h)
            pc = c5(parent_h, parent_n)
            best_h, best_c, best_n = parent_h.copy(), pc, parent_n
            best_for_n[parent_n] = (parent_h.copy(), pc)
    except Exception:
        pass

    if best_h is None:
        n0 = 100
        h0 = thue(n0, 0).astype(float)
        best_h, best_c, best_n = h0.copy(), c5(h0, n0), n0
        best_for_n[n0] = (h0.copy(), c5(h0, n0))

    rng = np.random.RandomState(seed)
    cand_ns = (80, 100, 120, 150)

    binary_end = t0 + 0.25 * (B - 5.0)
    if binary_end > deadline - 180:
        binary_end = deadline - 180
    if binary_end < t0 + 25:
        binary_end = t0 + 25

    mip_end = t0 + 0.55 * (B - 5.0)
    if mip_end > deadline - 80:
        mip_end = deadline - 80
    if mip_end < binary_end:
        mip_end = binary_end

    for n in cand_ns:
        if time.time() >= binary_end - 1:
            break
        local_best_n, local_best_c = None, 10 ** 9
        starts = make_starts(n, parent_h, seed + n)
        for si, hh in enumerate(starts):
            if time.time() >= binary_end - 1:
                break
            tloc = time.time()
            hb, cb = local_search_binary(hh, n, min(binary_end - 0.5, tloc + 2.2),
                                         seed=seed + si * 17 + n, max_sweeps=80)
            if time.time() < binary_end - 5:
                hb2, cb2 = sa_binary(hb, n, min(binary_end - 0.5, time.time() + 3.5),
                                     seed=seed + si * 31 + n + 3)
                hb3, cb3 = local_search_binary(hb2, n, min(binary_end - 0.5, time.time() + 1.8),
                                               seed=seed + si * 91 + n + 5, max_sweeps=80)
                if cb3 < cb2:
                    hb2, cb2 = hb3, cb3
                if cb2 < cb:
                    hb, cb = hb2, cb2
            cc5 = cb * (2.0 / n)
            if cb < local_best_c:
                local_best_c, local_best_n = cb, hb.copy()
            if n not in best_for_n or cc5 < best_for_n[n][1] - 1e-12:
                best_for_n[n] = (hb.astype(float).copy(), cc5)
            if cc5 < best_c - 1e-12:
                best_c, best_h, best_n = cc5, best_for_n[n][0].copy(), n
        if local_best_n is not None and time.time() < binary_end - 2:
            hb, cb = local_search_binary(local_best_n, n, binary_end - 1.0,
                                         seed=seed + 5000 + n, max_sweeps=2000)
            cc5 = cb * (2.0 / n)
            if cc5 < best_for_n[n][1] - 1e-12:
                best_for_n[n] = (hb.astype(float).copy(), cc5)
            if cc5 < best_c - 1e-12:
                best_c, best_h, best_n = cc5, best_for_n[n][0].copy(), n

    for n in cand_ns:
        if time.time() >= mip_end - 5 or not HAS_MILP:
            break
        T = target_count(n)
        cur5 = best_for_n.get(n, (None, 1e9))[1]
        if cur5 <= T * (2.0 / n) + 1e-12:
            continue
        rem = mip_end - time.time()
        if rem < 10:
            break
        lim = min(160.0, rem * 0.55)
        if n >= 150:
            lim = min(90.0, lim)
        mh = solve_mip(n, T, lim)
        if mh is not None:
            mx = int(corr_count(mh, n).max())
            cc5 = mx * (2.0 / n)
            if n not in best_for_n or cc5 < best_for_n[n][1] - 1e-12:
                best_for_n[n] = (mh.astype(float).copy(), cc5)
            if cc5 < best_c - 1e-12:
                best_c, best_h, best_n = cc5, best_for_n[n][0].copy(), n
        if best_c <= TARGET and time.time() > mip_end - 80:
            break

    order = []
    for n in cand_ns:
        if n in best_for_n and n not in order:
            order.append(n)
    if parent_n is not None and parent_n not in order:
        order.append(parent_n)
    if best_n is not None and best_n not in order:
        order.append(best_n)
    for n in list(best_for_n.keys()):
        if n not in order:
            order.append(n)

    for n in order:
        if time.time() >= deadline - 10 or n not in best_for_n:
            break
        h0, c0 = best_for_n[n]
        if time.time() < deadline - 25:
            hh, cb = local_search_binary(h0, n, min(deadline - 20, time.time() + 3.0),
                                         seed=seed + 10000 + n, max_sweeps=120)
            cc5 = cb * (2.0 / n)
            if cc5 < c0 - 1e-12:
                best_for_n[n] = (hh.astype(float).copy(), cc5)
                if cc5 < best_c - 1e-12:
                    best_c, best_h, best_n = cc5, best_for_n[n][0].copy(), n
        if time.time() < deadline - 10:
            hh, cb = smooth_candidate(best_for_n[n][0], n, deadline, seed=seed + 20000 + n)
            cc5 = c5(hh, n)
            if cc5 < best_for_n[n][1] - 1e-12:
                best_for_n[n] = (hh.copy(), cc5)
                if cc5 < best_c - 1e-12:
                    best_c, best_h, best_n = cc5, hh.copy(), n
        if best_c > TARGET and n <= 150 and time.time() < deadline - 20:
            hp = project(best_for_n[n][0].copy() + 0.05 * rng.randn(n), n)
            hh, cb = smooth_candidate(hp, n, deadline, seed=seed + 30000 + n)
            cc5 = c5(hh, n)
            if cc5 < best_for_n[n][1] - 1e-12:
                best_for_n[n] = (hh.copy(), cc5)
                if cc5 < best_c - 1e-12:
                    best_c, best_h, best_n = cc5, hh.copy(), n

    for n, (hh, cc5) in best_for_n.items():
        if cc5 < best_c - 1e-12:
            best_c, best_h, best_n = cc5, hh.copy(), n

    best_h = project(best_h, int(best_n))
    best_c = c5(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
