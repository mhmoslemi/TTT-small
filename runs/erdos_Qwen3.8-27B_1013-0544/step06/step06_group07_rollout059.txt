```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / exp_c.sum()
    
    # p has length 2n-1, indexed by m = 0..2n-2
    # k = m - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        # A[i] = sum_m p[m] * (1-h[i+k]) * 1_{i+k in [0,n)}
        k_vals = np.arange(2*n-1) - (n-1)  # -(n-1) to n-1
        ik = i + k_vals
        mask = (ik >= 0) & (ik < n)
        # B[i] = sum_m p[m] * h[i-k] * 1_{i-k in [0,n)}
        ik_neg = i - k_vals
        mask_neg = (ik_neg >= 0) & (ik_neg < n)
        A_i = np.sum(p[mask] * (1 - h[ik[mask]]))
        B_i = np.sum(p[mask_neg] * h[ik_neg[mask_neg]])
        grad[i] = dx * (A_i - B_i)
    return grad
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0, 1)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta
```

```python
    mask_m = (ik_m >= 0) & (ik_m < n)
    ik_c = np.clip(ik, 0, n - 1)
    ik_m_c = np.clip(ik_m, 0, n - 1)

            omh = 1.0 - h[ik_c]
```
