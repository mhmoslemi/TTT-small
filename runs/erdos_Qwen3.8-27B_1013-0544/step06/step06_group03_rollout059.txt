            if tot < 1e-14:

    else:

        g[:n-m] += -h[m:]

                best_c = c_trial

        alpha *= 1.01

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] += (1.0 - h[m:])
            g[m:] += (-h[:n-m])
        else:
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] += (-h[m:])
    return g

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
        val = m
    else:
        w /= sw
        val = m + np.log(sw) / alpha
    
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        lag = k - (n-1)
        mg = abs(lag)

            else:
