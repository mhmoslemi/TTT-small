import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    if lag > 0:
        grad[m:] += (1.0 - h[:n - m]) * dx
        grad[:n - m] -= h[m:] * dx
    else:
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n - m] * dx
    return grad

def smooth_max_gradient(h, n, alpha, top_m=15):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    top_c = c[top_idx]
    max_tc = top_c.max()
    shifted = alpha * (top_c - max_tc)
    shifted = np.clip(shifted, -40.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-30:
        lag = top_idx[0] - (n - 1)
        return lag_gradient(h, n, lag)
    probs = weights / total
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        lag = top_idx[i] - (n - 1)
        grad += probs[i] * lag_gradient(h, n, lag)
    return grad

def projected_grad_with_momentum(h, n, deadline, alpha_start=30, alpha_end=800, top_m=12, momentum=0.5):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    t0 = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 3000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        step = 0.4 * (1.0 + 0.5) / (1.0 + 0.05 * frac)
        grad = smooth_max_gradient(h_curr, n, alpha, top_m)
        h_trial = h_curr - step * grad + momentum * (h_curr - h_prev)
        h_trial = project_feasible(h_trial, n)
        c_new = compute_c5(h_trial, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_trial.copy()
        h_prev = h_curr
        h_curr = h_trial
    return best_h, best_c

def lp_outer_approx(h, n, deadline, top_m=12):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for _ in range(30):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust
        ]
        for idx in top_idx:
            lag = idx - (n - 1)
            g = lag_gradient(h_curr, n, lag)
            cons.append(c[idx] + g @ (h_var - h_curr) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        if h_var.value is None:
            break
        h_new = project_feasible(h_var.value, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

def fine_local_search(h, n, deadline):

    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    rng = np.random.RandomState(123)
    for d in deltas:
        no_improve_count = 0
        while time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            top_k = int(np.argmax(c))
            lag = top_k - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            n_pairs = n - m
            if n_pairs <= 0:
                break
            n_tries = min(300, n_pairs)
            if n_pairs <= n_tries:
                indices = np.arange(n_pairs)
            else:
                indices = rng.choice(n_pairs, size=n_tries, replace=False)
            for i in indices:
                j = i + m
                if h[i] == h[j]:
                    continue
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.9, (1.0 - h[j]) * 0.9)
                else:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.9, (1.0 - h[i]) * 0.9)
                if delta < 1e-12:
                    continue
                if h[i] > h[j]:
                    h[i] -= delta
                    h[j] += delta
                else:
                    h[i] += delta
                    h[j] -= delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    if h[i] > h[j] - 2 * delta:
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
            if not improved:
                break
        if time.time() > deadline:
            break
    return h, best_c

def gradient_local_hybrid(h, n, deadline):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    for m_top in [3, 5, 7, 10]:
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        step = 0.3
        h_s = h_curr.copy()
        best_s_c = compute_c5(h_s, n)
        it = 0
        while time.time() < sub_deadline:
            it += 1
            step = 0.3 / (1.0 + 0.02 * it)
            c = compute_all_lags(h_s, n)
            top_idx = np.argsort(c)[-m_top:]
            grad = np.zeros(n)
            for idx in top_idx:
                grad += lag_gradient(h_s, n, idx - (n - 1))
            grad /= m_top
            h_trial = project_feasible(h_s - step * grad, n)
            c_new = compute_c5(h_trial, n)
            if c_new < best_s_c - 1e-15:
                best_s_c = c_new
            h_s = h_trial
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_s.copy()
        if best_s_c < compute_c5(h_curr, n):
            h_curr = h_s
    return best_h, best_c
