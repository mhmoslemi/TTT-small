from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

def subgradient_optimize(h, n, deadline, n_iter=5000):
    """Subgradient descent with Polyak step size and averaging."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    V = 0.0
    h_bar = h_curr.copy()
    momentum = np.zeros(n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        c_current = c.max()
        
        # Subgradient of the active lag
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        
        # Compute subgradient
        grad = np.zeros(n)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        elif lag > 0:
            grad[:n-m] -= h_curr[m:]
            grad[m:] += (1.0 - h_curr[:n-m])
        else:
            grad[:n-m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n-m]
        grad *= dx
        
        # Polyak step size
        V += 1.0
        gamma = (c_current - best_c) / (np.dot(grad, grad) + 1e-12)
        gamma = min(gamma, 0.5 / (it + 1))  # Cap step size
        
        # Momentum
        momentum = 0.9 * momentum - gamma * grad
        
        # Update and project
        h_new = project_feasible(h_curr + momentum, n)
        
        # Running average
        h_bar = (it * h_bar + h_new) / (it + 1)
        h_bar = project_feasible(h_bar, n)
        
        c_new = compute_c5(h_new, n)
        c_bar = compute_c5(h_bar, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_bar < best_c:
            best_c = c_bar
            best_h = h_bar.copy()
    
    return best_h, best_c
