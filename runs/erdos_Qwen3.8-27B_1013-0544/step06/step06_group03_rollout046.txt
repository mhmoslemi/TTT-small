def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_k is not None:
        c = c[top_k]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp(h, n, alpha, top_k=None, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        args = (n, alpha, top_k)
        res = minimize(smooth_obj, h, args=args, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def outer_approx_step(h, n, trust):

    h_var = cp.Variable(n)

            cp.abs(h_var - h) <= trust]

        cons.append(c[k] + dx * (g @ (h_var - h)) <= t_var)
