def project_feasible(h,n):
    h=np.array(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(100):
        diff=target-h.sum()
        if abs(diff)<1e-13:
            break
        if diff>0:
            room=1.0-h
            s=room.sum()
            if s<1e-15: break
            h=h + (diff/s)*room
        else:
            s=h.sum()
            if s<1e-15: break
            h=h + (diff/s)*h
    h=np.clip(h,0.0,1.0)

import numpy as np
import time

FRAC_SMALL = (0.0, 0.25, 0.50, 0.75, 1.0)
FRAC_MED = (0.0, 0.125, 0.25, 0.50, 0.75, 0.875, 1.0)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 0:
            h = h + (diff / s) * room
    elif diff < 0:
        s = h.sum()
        if s > 0:
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def overlap_counts(h, n):
    L = 2 * n - 1
    F = np.empty(L, dtype=float)
    F[n - 1] = float(np.dot(h, 1.0 - h))
    for s in range(1, n):
        F[n - 1 + s] = float(np.dot(h[:n - s], 1.0 - h[s:]))
        F[n - 1 - s] = float(np.dot(h[s:], 1.0 - h[:n - s]))
    return F


def make_lvec(h, i, j, n):
    L = 2 * n - 1
    off = n - 1
    v = np.zeros(L, dtype=float)
    omh = 1.0 - h
    v[off - i:off - i + n] += omh
    v[off - j:off - j + n] -= omh
    hr = h[::-1]
    v[i:i + n] -= hr
    v[j:j + n] += hr
    return v


def pair_max_delta(delta, F, Lv, c0, cp, cm):
    if delta == 0.0:
        return float(F.max())
    v = F + Lv * delta
    q = delta * delta
    v[c0] -= 2.0 * q
    v[cp] += q
    v[cm] += q
    return float(v.max())


def golden_min_pair(a, b, F, Lv, c0, cp, cm, iters=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1 = pair_max_delta(x1, F, Lv, c0, cp, cm)
    f2 = pair_max_delta(x2, F, Lv, c0, cp, cm)
    for _ in range(iters):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - gr * (b - a)
            f1 = pair_max_delta(x1, F, Lv, c0, cp, cm)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + gr * (b - a)
            f2 = pair_max_delta(x2, F, Lv, c0, cp, cm)
    mx = 0.5 * (a + b)
    fm = pair_max_delta(mx, F, Lv, c0, cp, cm)
    bestv, bestx = f1, x1
    if f2 < bestv:
        bestv, bestx = f2, x2
    if fm < bestv:
        bestv, bestx = fm, mx
    return bestv, bestx


def refine_pair(h, F, i, j, n):
    hi = min(1.0 - h[i], h[j])
    if hi <= 1e-8:
        return np.inf, 0.0
    Lv = make_lvec(h, i, j, n)
    off = n - 1
    d = abs(i - j)
    c0 = off
    cp = off + d
    cm = off - d
    npts = 33
    deltas = np.linspace(0.0, hi, npts)
    bestv = np.inf
    bestd = 0.0
    bestk = 0
    for k in range(npts):
        val = pair_max_delta(deltas[k], F, Lv, c0, cp, cm)
        if val < bestv:
            bestv = val
            bestd = deltas[k]
            bestk = k
    if bestk == 0:
        a = deltas[0]
        b = deltas[1]
    elif bestk == npts - 1:
        a = deltas[npts - 2]
        b = deltas[npts - 1]
    else:
        a = deltas[bestk - 1]
        b = deltas[bestk + 1]
    if b - a > 1e-13:
        gv, gd = golden_min_pair(a, b, F, Lv, c0, cp, cm, 22)
        if gv < bestv:
            bestv, bestd = gv, gd
    return bestv, bestd


def evaluate_list(h, F, n, pairs, K, fracs, deadline):
    vals = np.full(K, np.inf, dtype=float)
    ii = np.zeros(K, dtype=int)
    jj = np.zeros(K, dtype=int)
    dd = np.zeros(K, dtype=float)
    cnt = 0
    for pair in pairs:
        if (cnt & 1023) == 0 and time.time() > deadline:
            break
        cnt += 1
        i, j = pair
        hi = min(1.0 - h[i], h[j])
        if hi <= 1e-8:
            continue
        Lv = make_lvec(h, i, j, n)
        off = n - 1
        d = abs(i - j)
        c0 = off
        cp = off + d
        cm = off - d
        bv = np.inf
        bd = 0.0
        for f in fracs:
            dlt = f * hi
            val = pair_max_delta(dlt, F, Lv, c0, cp, cm)
            if val < bv:
                bv = val
                bd = dlt
        if bv < vals.max():
            pos = int(np.argmax(vals))
            vals[pos] = bv
            ii[pos] = int(i)
            jj[pos] = int(j)
            dd[pos] = bd
    order = np.argsort(vals)
    bestM = np.inf
    besti = -1
    bestj = -1
    bestd = 0.0
    ref_limit = min(K, 45)
    for q in range(ref_limit):
        if vals[order[q]] >= np.inf:
            break
        if time.time() > deadline:
            break
        i = int(ii[order[q]])
        j = int(jj[order[q]])
        rm, rd = refine_pair(h, F, i, j, n)
        if rm < bestM:
            bestM = rm
            besti = i
            bestj = j
            bestd = rd
    return bestM, besti, bestj, bestd


def generate_sample_pairs(h, F, n, rng, active_count=6, random_count=500):
    inc = np.where(h < 1.0 - 1e-7)[0]
    dec = np.where(h > 1e-7)[0]
    if len(inc) == 0 or len(dec) == 0:
        return []
    omh = 1.0 - h
    off = n - 1
    top_idx = np.argsort(F, kind="quicksort")[::-1][:active_count]
    st = set()
    for idx in top_idx:
        t = idx - off
        C = np.zeros(n, dtype=float)
        if t >= 0:
            C[:n - t] += omh[t:]
            C[t:] -= h[:-t]
        else:
            s = -t
            C[s:] += omh[:n - s]
            C[:n - s] -= h[s:]
        ci = C[inc]
        cd = C[dec]
        hi_idx = inc[np.argsort(ci, kind="mergesort")[-5:]]
        lo_idx = dec[np.argsort(cd, kind="mergesort")[:5]]
        for i in hi_idx:
            for j in lo_idx:
                if i != j:
                    st.add((int(i), int(j)))
        for i in lo_idx:
            for j in hi_idx:
                if i != j:
                    st.add((int(i), int(j)))
    made = 0
    for _ in range(random_count):
        i = int(rng.choice(inc))
        j = int(rng.choice(dec))
        if i != j:
            st.add((i, j))
            made += 1
            if len(st) > 1100:
                break
    return list(st)


def kick_h(h, n, rng, strength=0.035):
    h2 = np.array(h, dtype=float).copy()
    noise = rng.randn(n)
    noise -= noise.mean()
    h2 += strength * noise
    h2 = project_feasible(h2, n)
    for _ in range(3):
        inc = np.where(h2 < 1.0 - 1e-7)[0]
        dec = np.where(h2 > 1e-7)[0]
        if len(inc) == 0 or len(dec) == 0:
            break
        i = int(rng.choice(inc))
        j = int(rng.choice(dec))
        if i == j:
            continue
        amt = min(h2[j], 1.0 - h2[i]) * rng.random()
        h2[i] += amt
        h2[j] -= amt
    return project_feasible(h2, n)


def fix_count(t, n, rng):
    target = n // 2
    t = np.array(t, dtype=float)
    s = int(np.sum(t > 0.5))
    if s > target:
        idx = np.where(t > 0.5)[0]
        rng.shuffle(idx)
        t[idx[:s - target]] = 0.0
    elif s < target:
        idx = np.where(t <= 0.5)[0]
        rng.shuffle(idx)
        t[idx[:target - s]] = 1.0
    return t


def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return fix_count(t.astype(float), n, rng)


def sturm_init(n, alpha, rng):
    frac = (np.arange(n, dtype=float) * alpha) % 1.0
    t = np.where(frac < 0.5, 1.0, 0.0)
    return fix_count(t, n, rng)


def random_binary_init(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def resample_h(h0, n0, n1):
    if n1 == n0:
        return project_feasible(h0, n1)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n1) + 0.5) * (2.0 / n1)
    idx = np.searchsorted(x0, x1, side="right")
    idx = np.clip(idx, 0, n0 - 1)
    return project_feasible(h0[idx], n1)


def improve_pairs(start_h, n, budget, seed):
    t0 = time.time()
    dl = t0 + max(0.0, budget)
    h = project_feasible(start_h, n)
    if dl - t0 < 1.5:
        return h, compute_c5(h, n)

    F = overlap_counts(h, n)
    dx = 2.0 / n
    curM = float(F.max())
    bestc = compute_c5(h, n)
    besth = h.copy()
    rng = np.random.RandomState(seed)
    no_imp = 0
    temp = 0.0035

    while time.time() < dl - 1.0:
        full = no_imp >= 4 and (no_imp % 5 == 0)
        if full:
            inc = np.where(h < 1.0 - 1e-7)[0]
            dec = np.where(h > 1e-7)[0]
            prod = len(inc) * len(dec)
            if len(inc) > 0 and len(dec) > 0 and prod <= 30000:
                pairs = []
                for i in inc:
                    for j in dec:
                        if i != j:
                            pairs.append((int(i), int(j)))
                K = 55
                fracs = FRAC_SMALL
            else:
                pairs = generate_sample_pairs(h, F, n, rng, 10, 2200)
                K = 60
                fracs = FRAC_MED
        else:
            pairs = generate_sample_pairs(h, F, n, rng, 6, 500)
            K = 55
            fracs = FRAC_MED

        if not pairs:
            h = kick_h(h, n, rng)
            F = overlap_counts(h, n)
            curM = float(F.max())
            c = curM * dx
            if c < bestc:
                tc = compute_c5(h, n)
                if tc < bestc:
                    bestc = tc
                    besth = h.copy()
            no_imp = 0
            temp = 0.005
            continue

        bestM, bi, bj, bd = evaluate_list(h, F, n, pairs, K, fracs, dl - 0.5)
        moved = False
        if bi >= 0 and bd > 1e-8:
            if bestM < curM - 1e-8:
                h[bi] += bd
                h[bj] -= bd
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
                F = overlap_counts(h, n)
                curM = float(F.max())
                no_imp = 0
                temp = max(0.00012, temp * 0.90)
                moved = True
            elif bestM < curM * 1.000 + 3.0:
                delta_c = (bestM - curM) * dx
                if delta_c < 0.018 and rng.random() < np.exp(-delta_c / max(temp, 1e-7)):
                    h[bi] += bd
                    h[bj] -= bd
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
                    F = overlap_counts(h, n)
                    curM = float(F.max())
                    no_imp = 0
                    temp = max(0.00012, temp * 0.93)
                    moved = True

        if not moved:
            no_imp += 1
            temp = max(0.00012, temp * 0.97)
            if no_imp >= 12:
                h = kick_h(h, n, rng, 0.025 + 0.025 * rng.random())
                F = overlap_counts(h, n)
                curM = float(F.max())
                c = curM * dx
                if c < bestc:
                    tc = compute_c5(h, n)
                    if tc < bestc:
                        bestc = tc
                        besth = h.copy()
                no_imp = 0
                temp = 0.006

        c = curM * dx
        if c < bestc - 1e-10:
            tc = compute_c5(h, n)
            if tc < bestc:
                bestc = tc
                besth = h.copy()

    bestc = compute_c5(besth, len(besth))
    return besth, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = 8.0
    total = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 250

    if h0 is None or n0 % 2 != 0 or n0 < 60 or n0 > 900:

        h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    p1 = t0 + 0.60 * (total - reserve)
    if p1 - time.time() > 2.0:
        h1, c1 = improve_pairs(best_h, len(best_h), p1 - time.time(), seed)
        if c1 < best_c:
            best_c = c1

    p2 = t0 + 0.81 * (total - reserve)
    if time.time() < p2:
        ht = thue_morse_init(len(best_h), rng)
        h2, c2 = improve_pairs(ht, len(ht), max(1.0, 0.5 * (p2 - time.time()) - 1.0), seed + 11)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    if time.time() < p2 - 1.0:
        hs = sturm_init(len(best_h), (np.sqrt(5.0) - 1.0) / 2.0, rng)
        h3, c3 = improve_pairs(hs, len(hs), max(1.0, p2 - time.time() - 1.5), seed + 23)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    p3 = t0 + 0.93 * (total - reserve)
    if time.time() < p3:
        nb = len(best_h)
        n1 = nb + (50 if nb < 300 else 25)
        if n1 > 650:
            n1 = 650
        if n1 % 2 != 0:
            n1 += 1
        if n1 != nb:
            h1r = resample_h(best_h, nb, n1)
            h4, c4 = improve_pairs(h1r, n1, max(1.0, p3 - time.time() - 1.5), seed + 31)
            if c4 < best_c:
                best_c = c4
                best_h = h4.copy()

    t_end = t0 + total - reserve
    if time.time() < t_end:
        hf, cf = improve_pairs(best_h, len(best_h), max(1.0, t_end - time.time()), seed + 47)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, float(best_c), int(n)
