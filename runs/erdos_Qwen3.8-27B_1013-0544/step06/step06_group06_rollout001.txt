import numpy as np
import time
from scipy.optimize import minimize

_N = 0
_TARGET = 0.0
_ONES = None
_ALPHA = 1.0
_ACTIVE_TOL = 0.0
_SLSQP_DEADLINE = 1e30
_LAST_SLSQP_X = None
_RNG = np.random.RandomState(0)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def max_count(h):
    return float(np.max(corr_full(h)))


def c5_bound_from_count(cnt, n):
    return float(cnt) * (2.0 / float(n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    if abs(x.sum() - target) < 1e-12:
        return np.clip(x, 0.0, 1.0)

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - y
            mask = room > 1e-12
            if np.any(mask):
                y[mask] += diff * room[mask] / room[mask].sum()
        else:
            mask = y > 1e-12
            if np.any(mask):
                y[mask] += diff * y[mask] / y[mask].sum()
    return np.clip(y, 0.0, 1.0)


def grad_from_weights(h, w):
    w = np.asarray(w, dtype=float)
    sl = slice(_N - 1, 2 * _N - 1)
    t0 = np.convolve(_ONES, w, "full")[sl]
    fh = np.convolve(w, h, "full")[sl]
    sh = np.convolve(w[::-1], h, "full")[sl]
    return t0 - fh - sh


def smooth_count_from_c(c, alpha):
    cm = float(np.max(c))
    if alpha <= 0.0:
        return cm
    z = alpha * (c - cm)
    s = float(np.sum(np.exp(z)))
    if not np.isfinite(s) or s < 1.0:
        s = 1.0
    return cm + np.log(s) / alpha


def smooth_obj(h):
    c = corr_full(h)
    return smooth_count_from_c(c, _ALPHA)


def smooth_jac(h):
    c = corr_full(h)
    cm = float(np.max(c))
    z = _ALPHA * (c - cm)
    w = np.exp(z)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    return grad_from_weights(h, w)


def exact_obj(h):
    return max_count(h)


def active_jac(h):
    c = corr_full(h)
    cm = float(np.max(c))
    active = np.where(c >= cm - _ACTIVE_TOL)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))])
    if active.size > 200:
        active = active[np.argsort(c[active])[-200:]]
    w = np.zeros_like(c)
    w[active] = 1.0 / float(active.size)
    return grad_from_weights(h, w)

    _LAST_SLSQP_X = np.asarray(xk, dtype=float).copy()

def thue_morse_init(n, rng):
    x = np.arange(n, dtype=np.uint64)
    p = x
    for _ in range(6):
        p = p ^ (p >> 1)
    t = (p & 1).astype(float)
    target = n // 2
    ones = np.where(t == 1.0)[0]
    if ones.size > target:
        rng.shuffle(ones)
        t[ones[: ones.size - target]] = 0.0
    elif ones.size < target:
        zeros = np.where(t == 0.0)[0]
        rng.shuffle(zeros)
        t[zeros[: target - ones.size]] = 1.0
    return t


def sinusoid_init(n, amp, rng):

def projected_gradient(h0, alpha, deadline, max_iters=3000, base_start=0.04):

    active_mode = isinstance(alpha, str)

    c0 = corr_full(h)

    base = float(base_start)
    fails = 0
    it = 0
    active_tol = 0.05

    while time.time() < float(deadline) and it < int(max_iters):
        it += 1
        if active_mode:
            _ACTIVE_TOL = max(1e-4, active_tol)
            g = active_jac(h)
        else:
            g = smooth_jac(h)

        g = g - g.mean()
        nrm = float(np.max(np.abs(g)))
        if nrm < 1e-12:
            break

        u = -g / nrm
        step = np.inf
        pos = u > 1e-14
        if np.any(pos):
            step = min(step, float(np.min((1.0 - h[pos]) / u[pos])))
        neg = u < -1e-14
        if np.any(neg):
            step = min(step, float(np.min(h[neg] / (-u[neg]))))
        if not np.isfinite(step) or step <= 1e-10:
            step = 1e-4

        max_lam = min(base, step)
        if max_lam < 1e-9:
            base *= 0.5
            fails += 1
            if fails > 70:
                break
            continue

        trials = [max_lam, max_lam * 0.5, max_lam * 0.25, max_lam * 0.1, max_lam * 0.05]
        if step > base and it % 5 == 0:
            trials.append(min(step, base * 2.0))
        trials.sort(reverse=True)

        accepted = False
        for lam in trials:
            if lam <= 1e-11:
                continue
            xraw = h + lam * u
            mn = float(xraw.min())
            mx = float(xraw.max())
            if mn >= -1e-12 and mx <= 1.0 + 1e-12:
                x = np.clip(xraw, 0.0, 1.0)
                if abs(x.sum() - _TARGET) > 1e-9:
                    x = project_box_sum(xraw, _TARGET)
            else:
                x = project_box_sum(xraw, _TARGET)

            cnew = corr_full(x)
            cn = float(np.max(cnew))
            if cn < cur_c - 1e-12:
                h = x
                cur_c = cn
                cur_F = cur_c if active_mode else smooth_count_from_c(cnew, a_val)
                accepted = True
                break
            elif not active_mode and cn <= cur_c + 1e-9:
                fn = smooth_count_from_c(cnew, a_val)
                if fn < cur_F - 1e-10:
                    h = x
                    cur_c = cn
                    cur_F = fn
                    accepted = True
                    break

        if cur_c < best_c - 1e-13:
            best_c = cur_c
            best_h = h.copy()

        if accepted:
            fails = 0
            if not active_mode:
                active_tol = max(0.01, active_tol * 0.97)
            base = min(0.12, base * 1.08)
        else:
            fails += 1
            base *= 0.7
            if active_mode:
                active_tol = max(0.002, active_tol * 0.8)
            if fails == 25 or fails == 50:
                r = _RNG.randn(_N)
                r -= r.mean()
                rn = float(np.max(np.abs(r)))
                if rn > 1e-14:
                    h2 = project_box_sum(h + (0.005 / rn) * r, _TARGET)
                    c2 = corr_full(h2)
                    cn2 = float(np.max(c2))
                    h = h2
                    cur_c = cn2
                    cur_F = cn2 if active_mode else smooth_count_from_c(c2, a_val)
            if fails > 90:
                base = float(base_start)
                if fails > 180:
                    break

    return best_h
