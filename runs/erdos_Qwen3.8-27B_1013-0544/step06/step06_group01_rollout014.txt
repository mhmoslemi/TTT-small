    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def soft_max_c5(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if alpha * (c.max() - m) < -500:  # avoid overflow
        return float(m * dx)
    # log-sum-exp for numerical stability
    shifted = alpha * (c - m)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, step_size):
    """One subgradient step for minimizing C5."""
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(c)
    dx = 2.0 / n
    # Gradient of c[idx] wrt h
    grad = np.zeros(n)
    for j in range(n):
        k1 = j + idx - n + 1
        k2 = j - idx + n - 1
        if 0 <= k1 < n:
            grad[j] += 1.0 - h[k1]
        if 0 <= k2 < n:
            grad[j] -= h[k2]
    grad *= dx
    # Projected step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, c[idx] * dx

def softmax_grad(h, n, alpha):
    """Compute gradient of soft-max approximation of C5."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    
    for j in range(n):
        # First term
        m_lo = max(0, n - 1 - j)
        m_hi = min(2 * n - 2, 2 * n - 2 - j)
        if m_lo <= m_hi:
            idx = j + np.arange(m_lo, m_hi + 1) - n + 1
            grad[j] += np.dot(w[m_lo:m_hi + 1], one_minus_h[idx])
        # Second term
        m_lo2 = max(0, j)
        m_hi2 = min(2 * n - 2, j + n - 1)
        if m_lo2 <= m_hi2:
            idx = j - np.arange(m_lo2, m_hi2 + 1) + n - 1
            grad[j] -= np.dot(w[m_lo2:m_hi2 + 1], h[idx])
    
    return grad * dx, (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def gradient_descent(h, n, deadline, alphas):
    """Projected gradient descent with annealing alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        t_start = time.time()
        t_phase = min(60, deadline - time.time())
        step = 0.05 / alpha  # initial step size
        for _ in range(2000):
            if time.time() > deadline - 30:

            grad, obj_val = softmax_grad(h, n, alpha)
            # Backtracking line search
            step_new = step
            for _ls in range(20):
                h_try = project_feasible(h - step_new * grad, n)
                obj_try = compute_c5(h_try, n)
                if obj_try < compute_c5(h, n) or _ls == 19:
                    break
                step_new *= 0.5
            h = project_feasible(h - step_new * grad, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            step = step_new  # keep step size
    return best_h, best_c
