        fm = m + sm / alpha + np.log(e.sum()) / alpha

        grad = np.zeros(n)
        for ki in range(len(idx)):

            wk = w[ki]
            if k >= 0:
                # j+k valid for j in [0, n-1-k]
                grad[0:n-k] += wk * (1.0 - x[k:n])
                # j-k valid for j in [k, n-1]
                grad[k:n] -= wk * x[0:n-k]
            else:
                kk = -k
                # j+k valid for j in [kk, n-1]
                grad[kk:n] += wk * (1.0 - x[0:n-kk])
                # j-k valid for j in [0, n-1-kk]
                grad[0:n-kk] -= wk * x[kk:n]
        grad *= dx
        
        # Sum constraint penalty
        s_diff = x.sum() - n / 2.0
        pen = 50.0
        fm += 0.5 * pen * s_diff * s_diff
        grad += pen * s_diff * np.ones(n) / n
        
        return fm, grad
    
    try:
        res = minimize(obj_and_grad, h_init, method='L-BFGS-B',

                                'maxfun': maxiter * 2})

                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)

        if not improved:
            break

            h = (np.mod(np.arange(n) * alpha_s, 1.0) < 0.5).astype(float)
            h = project_feasible(h, n)
        elif trial == 5:
            # Low-frequency perturbation
            h = best_h.copy()
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.03 * np.sin(2 * np.pi * freq * np.arange(n) / n * 2 + phase)
            h = project_feasible(h + pert, n)
        else:
            h = project_feasible(best_h + rng.uniform(-0.05, 0.05, n), n)

        for alpha, topk, mi in [(50000, 40, 1500), (500000, 30, 1500), (5000000, 20, 1500)]:
            if time.time() > p5 - 2:
                break
            h = lbgfbsb_optimize(h, n, alpha, topk, mi, p5 - 2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 6: Final high-precision polish
    p6 = dl - 1
    if time.time() < p6 - 2:
        h = lbgfbsb_optimize(best_h, n, 50000000, 15, 5000, p6 - 1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h
        h = lbgfbsb_optimize(best_h, n, 200000000, 10, 3000, p6 - 1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
