                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
            else:
                bl = -lag
                grad[:n - bl] += (1.0 - h[bl:])
                grad[bl:] -= h[:n - bl]
        grad *= dx / top_m
        
        v = 0.9 * v - lr * grad
        h_new = project_feasible(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        if it % 1000 == 999:
            lr *= 0.95
    return best_h, best_c
```

This is much faster per iteration (only top_m gradients instead of all 2n-1). For top_m=10 and n=400, each iteration is:
- correlate: 50μs
- top-m gradient: 10 * (two array ops of size 400) ≈ 10 * 20μs = 200μs
- project: 10μs
- Total: ~260μs

Hmm, not much faster than the log-sum-exp version when only a few weights are active. Let me just use the log-sum-exp version.

Actually, wait. For large beta in the log-sum-exp, the number of active weights is similar to top_m. So the computational cost is similar. The advantage of log-sum-exp is that it provides a smooth gradient (weighted average), while the subgradient is discontinuous.

Let me just go with the log-sum-exp approach.

One more optimization: I can precompute the "active" indices and skip the loop over all 2n-1 weights. For large beta, only a few weights are non-negligible.

Let me finalize the code. I'll aim for clarity and correctness.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_obj_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    g = c * dx
    g_max = g.max()
    shifted = np.clip(beta * (g - g_max), -500, 0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    w = exp_s / s
    obj = g_max + np.log(s) / beta
    grad = np.zeros(n)
    n1 = n - 1
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - n1
        if lag >= 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            bl = -lag
            grad[:n - bl] += wi * (1.0 - h[bl:])
            grad[bl:] -= wi * h[:n - bl]
    grad *= dx
    return obj, grad

def adam_opt(h, n, beta, lr0, steps, deadline, decay=0.9):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    for it in range(steps):
        if time.time() > deadline:
            break
        obj, grad = soft_obj_grad(h, n, beta)
        t += 1
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        if it % 2000 == 1999:
            lr *= decay
    return best_h, best_c
