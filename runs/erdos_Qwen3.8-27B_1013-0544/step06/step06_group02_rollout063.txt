```python
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_A_from_g(g, n):
    """A @ h = correlate(h, g, 'full')"""
    A = np.zeros((2*n-1, n))
    ns1 = n - 1
    for k in range(n):
        # row ns1+k: A[ns1+k, i] = g[i+k] for i=0..n-1-k
        A[ns1+k, :n-k] = g[k:n]
    for k in range(1, n):
        # row ns1-k: A[ns1-k, i] = g[i-k] for i=k..n-1
        A[ns1-k, k:] = g[:n-k]
    return A

def build_A_from_h(h, n):
    """C @ g = correlate(h, g, 'full')"""
    C = np.zeros((2*n-1, n))
    ns1 = n - 1
    for k in range(n):
        # row ns1+k: C[ns1+k, i+k] = h[i] for i=0..n-1-k

        # row ns1-k: C[ns1-k, i-k] = h[i] for i=k..n-1
        C[ns1-k, :n-k] = h[k:]
    return C

def solve_lp_for_h(g, n, solver=None):

    if h_var.value is None:

    return np.clip(h_var.value, 0, 1), float(t_var.value)
```

```python
def corr_gradient(h, n, k):
    """Gradient of corr[k] w.r.t. h, where k is the shift (-(n-1) to n-1)."""
    g = np.zeros(n)
    if k >= 0:
        # (1-h[j+k]) for j = 0..n-1-k
        g[:n-k] += (1 - h[k:])
        # -h[j-k] for j = k..n-1
        g[k:] -= h[:n-k]
    else:
        kp = -k
        # (1-h[j-kp]) for j = kp..n-1
        g[kp:] += (1 - h[:-kp])
        # -h[j+kp] for j = 0..n-1-kp
        g[:n-kp] -= h[kp:]
    return g
```

Let me verify with the n=3 example. h = [a,b,c], k=1:
corr[1] = h[0]*(1-h[1]) + h[1]*(1-h[2]) = a-ah[1] + b-bh[2]... wait that's not right.

corr[1] = sum_{i=0}^{n-1-1} h[i]*(1-h[i+1]) = h[0]*(1-h[1]) + h[1]*(1-h[2])

∂corr[1]/∂h[0] = (1-h[1]) = 1-b ✓ (from first formula: g[0] += 1-h[1])
∂corr[1]/∂h[1] = (1-h[2]) - h[0] = (1-c) - a ✓ (g[1] += 1-h[2], g[1] -= h[0])
∂corr[1]/∂h[2] = -h[1] = -b ✓ (g[2] -= h[1])

Good, the formula works.

Now, for the guided pair search:
1. Find k* = argmax of correlate
2. Compute sg = corr_gradient(h, n, k*)
3. The directional derivative in the direction "move δ from j to i" is:
   δ * (sg[j] - sg[i])  (if sg[j] > sg[i], moving from j to i decreases C5)

Wait, if we do h[i] += δ, h[j] -= δ:
Change in corr[k*] ≈ δ * sg[i] - δ * sg[j] = δ * (sg[i] - sg[j])

We want this to be negative, so we need sg[i] < sg[j].
So: increase h at low-sg positions, decrease h at high-sg positions.

This gives a very efficient guided search:
- Find argmin and argmax of sg
- Try moving mass from argmax to argmin
- Check if it actually improves C5 (not just the linear approximation)
- If yes, accept and repeat

This is O(n) per step! Much faster than O(n * radius).

Let me implement this:

```python
def guided_pair_search(h, n, deadline, max_rounds=5000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full')
        k_star = np.argmax(c)
        k_shift = k_star - (n-1)
        sg = corr_gradient(h, n, k_shift)
        # Move from high sg to low sg
        i = np.argmin(sg)
        j = np.argmax(sg)
        if i == j:
            break
        # Try a move
        improved = False
        for delta in [0.01, 0.005, 0.002, 0.001]:
            if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                h[i] += delta; h[j] -= delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] -= delta; h[j] += delta
        if not improved:
            # Try the other direction
            for delta in [0.01, 0.005, 0.002, 0.001]:
                if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                    h[i] -= delta; h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] += delta; h[j] -= delta
        if not improved:
            break
    return best_h, best_c
```

This should be very fast: O(n) per step, and we only stop when no improvement is found.

The LP handles this correctly (it minimizes the max over all k). The guided pair search might get stuck because it only looks at one constraint.
