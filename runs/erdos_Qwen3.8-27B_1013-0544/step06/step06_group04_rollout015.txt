from scipy.optimize import minimize, linprog

def compute_c5_np(h,n):
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*2.0/float(n))
def project_feasible(h,n):
    h=np.clip(np.asarray(h,float),0.0,1.0)
    target=0.5*n
    if abs(h.sum()-target)<1e-12: return h
    lo,hi=-1.0,1.0
    for _ in range(60):
        nu=0.5*(lo+hi)
        s=np.clip(h-nu,0.0,1.0).sum()
        if s<target: hi=nu
        else: lo=nu
    return np.clip(h-0.5*(lo+hi),0.0,1.0)
def enforce_exact_sum(h,n):
    h=np.array(h,float); d=0.5*n-h.sum()
    if abs(d)>1e-14:
        idx=int(np.argmax(np.minimum(h,1.0-h))); h[idx]+=d; h=np.clip(h,0.0,1.0)
    return h
def resample_h(h,n):

import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

G_N = 0
G_M = 0
G_ALPHA = 1.0
G_TARGET = 0.0

def compute_c5_np(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / float(n))

def project_feasible(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        nu = 0.5 * (lo + hi)
        s = np.clip(h - nu, 0.0, 1.0).sum()
        if s < target:
            hi = nu
        else:
            lo = nu
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def enforce_exact_sum(h, n):
    h = np.array(h, float)
    d = 0.5 * n - h.sum()
    if abs(d) > 1e-14:
        idx = int(np.argmax(np.minimum(h, 1.0 - h)))
        h[idx] += d
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, n):
    m = len(h)
    if m == n:
        return project_feasible(h, n)
    if m == 1:
        return project_feasible(np.full(n, float(h[0])), n)
    xo = (np.arange(m) + 0.5) * 2.0 / m
    xn = (np.arange(n) + 0.5) * 2.0 / n
    return project_feasible(np.interp(xn, xo, h), n)

def term_counts(h, n):
    dx = 2.0 / float(n)
    total = float(h.sum())
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    aut = np.correlate(h, h, mode='full')
    ctr = n - 1
    cnt = np.empty(2 * n - 1)
    cnt[ctr] = total - aut[ctr]
    for m in range(1, n):
        A = 0.5 * (aut[ctr + m] + aut[ctr - m])
        u = total - pref[m] - A
        v = pref[n - m] - A
        if u < 0 and u > -1e-12:
            u = 0.0
        if v < 0 and v > -1e-12:
            v = 0.0
        cnt[ctr + m] = u * dx
        cnt[ctr - m] = v * dx
    return cnt

def smooth_max_vec(c, alpha):
    mc = float(np.max(c))
    L = c.size
    if mc <= 1e-15 or alpha <= 1e-3:
        return float(np.mean(c)), np.full(L, 1.0 / L)
    arg = alpha * (c - mc)
    w = np.exp(arg)
    s = float(w.sum())
    if s <= 0.0:
        return float(np.mean(c)), np.full(L, 1.0 / L)
    p = w / s
    return mc + math.log(s) / alpha, p

def smooth_value_h(h, n, alpha):
    return smooth_max_vec(term_counts(h, n), alpha)[0]

def smooth_grad_h(h, n, alpha):
    c = term_counts(h, n)
    _, p = smooth_max_vec(c, alpha)
    dx = 2.0 / float(n)
    g = np.zeros(n)
    ctr = n - 1
    pw = p[ctr]
    if pw:
        g += pw * (1.0 - 2.0 * h)
    for m in range(1, n):
        pu = p[ctr + m]
        pv = p[ctr - m]
        if pu == 0 and pv == 0:
            continue
        if pu:
            g[m:] += pu
        if pv:
            g[:n - m] += pv
        wm = pu + pv
        if wm:
            g[:n - m] -= wm * h[m:]
            g[m:] -= wm * h[:n - m]
    return dx * g

def lag_grad_count(h, n, idx):
    ctr = n - 1
    d = idx - ctr
    g = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        m = d
        g[m:] = 1.0
        g[:n - m] -= h[m:]
        g[m:] -= h[:n - m]
    else:
        m = -d
        g[:n - m] = 1.0
        g[:n - m] -= h[m:]
        g[m:] -= h[:n - m]
    return g

def make_anti_h(f):
    m = len(f)
    h = np.empty(2 * m)
    h[:m] = f
    h[m:] = 1.0 - f
    return h

def anti_objective(f):
    h = make_anti_h(f)
    return smooth_value_h(h, G_N, G_ALPHA)

def anti_gradient(f):
    h = make_anti_h(f)
    g = smooth_grad_h(h, G_N, G_ALPHA)
    return g[:G_M] - g[G_M:]

def direct_objective(x):
    return smooth_value_h(np.clip(x, 0.0, 1.0), G_N, G_ALPHA)

def direct_gradient(x):
    h = np.clip(x, 0.0, 1.0)
    g = smooth_grad_h(h, G_N, G_ALPHA)
    return g * ((x > 0.0) & (x < 1.0))

def direct_sum_fun(x):
    return x.sum() - G_TARGET

def direct_sum_jac(x):
    return np.ones_like(x)

def anti_from_h(h):
    n = len(h)
    if n < 2 or n % 2:
        return None
    m = n // 2
    return np.clip(0.5 * (h[:m] + (1.0 - h[m:2 * m])), 0.0, 1.0)

def generic_f_starts(m, count=4):
    out = []
    idx = np.arange(m)
    for r in (1, 2, 3, 4, 5):
        for amp in (0.40, 0.50):
            out.append(np.clip(0.5 + amp * np.sin(2.0 * math.pi * r * idx / m), 0.0, 1.0))
            out.append(np.clip(0.5 + amp * np.cos(2.0 * math.pi * r * idx / m), 0.0, 1.0))
            if len(out) >= count:
                return out
    for r in (1, 2, 3):
        out.append((np.sin(2.0 * math.pi * r * idx / m) > 0).astype(float))
        if len(out) >= count:
            return out
    return out

def pick_best_f_starts(f_list, n, limit):
    valid = []
    scores = []
    for f in f_list:
        if f is None:
            continue
        f = np.clip(np.asarray(f, float), 0.0, 1.0)
        c = compute_c5_np(make_anti_h(f), n)
        if np.isfinite(c):
            valid.append(f)
            scores.append(c)
    if not valid:
        return []
    sc = np.array(scores)
    return [valid[int(i)] for i in np.argsort(sc)[:limit]]

def improve_anti(f0, n, deadline, alphas, maxiter):
    global G_N, G_M, G_ALPHA
    G_N = n
    G_M = n // 2
    m = G_M
    f = np.clip(f0, 0.0, 1.0).copy()
    best_f = f.copy()

            f1 = np.clip(res.x, 0.0, 1.0)
            c1 = compute_c5_np(make_anti_h(f1), n)
        except Exception:
            break
        if c1 < best_c - 1e-15:
            best_c = c1
            best_f = f1.copy()
            f = f1.copy()
        else:
            f = best_f.copy()
    return best_f, best_c

def run_slsqp(h0, n, deadline, alphas, maxiter):

    G_N = n
    G_TARGET = 0.5 * n
    best = project_feasible(h0, n).copy()

def lp_solve(cobj, A, b, Aeq, beq, bounds):
    try:
        return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
    except Exception:
        try:
            return linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds)
        except Exception:
            return None

def lp_polish(h0, n, deadline, max_iters=100):
    h = project_feasible(h0, n).copy()
    cur = compute_c5_np(h, n)
    best = h.copy()
    best_c = cur
    rho = 0.05
    stall = 0
    dx = 2.0 / float(n)
    aw = min(100, 2 * n - 1)
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        c = term_counts(h, n)
        order = np.argsort(c)
        idxs = order[-aw:]
        A = []
        b = []
        for idx in idxs:
            g = lag_grad_count(h, n, int(idx))
            row = np.empty(n + 1)
            row[:n] = dx * g
            row[-1] = -1.0
            A.append(row)
            b.append(-float(c[idx]))
        A = np.asarray(A)
        b = np.asarray(b)
        lower = np.maximum(-rho, -h)
        upper = np.minimum(rho, 1.0 - h)
        bd = list(zip(lower, upper))
        bd.append((0.0, 2.0))
        Aeq = np.zeros((1, n + 1))
        Aeq[0, :n] = 1.0
        beq = np.zeros(1)
        co = np.zeros(n + 1)
        co[n] = 1.0
        res = lp_solve(co, A, b, Aeq, beq, bd)
        if res is None or not res.success or res.x is None or not np.all(np.isfinite(res.x)):
            rho *= 0.5
            stall += 1
            if stall > 30 or rho < 1e-9:
                break
            continue
        step = res.x[:n]
        acc = False
        if np.max(np.abs(step)) > 1e-14:
            scale = 1.0
            for _ in range(5):
                h1 = project_feasible(np.clip(h + scale * step, 0.0, 1.0), n)
                c1 = compute_c5_np(h1, n)
                if c1 < cur - 1e-13:
                    h = h1
                    cur = c1
                    if c1 < best_c:
                        best_c = c1
                        best = h1.copy()
                    acc = True
                    break
                scale *= 0.5
        if acc:
            stall = 0
            rho = max(0.7 * rho, 1e-7)
        else:
            stall += 1
            rho *= 0.5
            if stall > 30 and rho < 1e-6:
                break
    return best, best_c
