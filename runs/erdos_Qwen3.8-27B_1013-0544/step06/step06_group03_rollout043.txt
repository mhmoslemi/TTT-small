from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_eq_jac(x, n):
    return np.ones(n)

def _smooth_obj_top(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    a = c[top_idx]
    m = a.max()
    shifted = alpha * (a - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def _smooth_obj_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def _single_lag_obj(x, n, lag, alpha):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return 0.0
    if m == 0:
        vals = x * (1 - x) * dx
        s = vals.sum()
        return float(s)
    if lag > 0:
        vals = x[:n - m] * (1 - x[m:]) * dx
    else:
        vals = x[m:] * (1 - x[:n - m]) * dx
    return float(vals.sum())

def slsqp_top(h, n, top_idx, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_top, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_all(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_all, h, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def chase_worst_lag(h, n, deadline):
    """Iteratively optimize just the single worst lag with increasing alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()

    for alpha in alphas:
        if time.time() > deadline:
            break
