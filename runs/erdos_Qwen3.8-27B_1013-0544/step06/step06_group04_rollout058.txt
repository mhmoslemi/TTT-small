The key insight is that the overlap at shift k is related to the autocorrelation of h. Specifically:
∫h(x)(1-h(x+k))dx = 1 - ∫h(x)h(x+k)dx

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, 'full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]

    b[idx[:target]] = 1.0
    return b

def smooth_max_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, 'full') * dx
    a = alpha * (c - c.max())
    return c.max() + np.log(np.exp(a).sum()) / alpha

    k = np.arange(-(n - 1), n)
    idx1 = j[:, None] + k[None, :]
    idx2 = j[:, None] - k[None, :]
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    t1 = (w * np.where(v1, x[np.clip(idx1, 0, n - 1)], 0.0)).sum(axis=1)
    t2 = (w * np.where(v2, x[np.clip(idx2, 0, n - 1)], 0.0)).sum(axis=1)
