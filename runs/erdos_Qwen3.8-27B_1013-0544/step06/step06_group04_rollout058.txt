    """Compute gradient of softmax objective.
    h: array of length n
    w: array of length 2n-1 (softmax weights for each correlation)
    Returns gradient of shape (n,)
    """

    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        s = k - (n-1)  # shift
        wk = w[k] * dx
        if s >= 0:

            j = np.arange(n - s)
            grad[j] += wk * (1 - h[j+s])
            # B part: h[j-s] for j=s..n-1

            grad[j] -= wk * h[j-s]

            j = np.arange(n - s_abs)
            grad[j] -= wk * h[j+s_abs]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h = h + d / s * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + d / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h):
    n = len(h)
    t = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:t]] = 1.0
    return b

def grad_softmax(h, alpha, n):
    """Compute gradient of smooth max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    exp_c = np.exp((c - m) / alpha)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            grad[:n-s] += wk * (1 - h[s:])
            grad[s:] -= wk * h[:n-s]
        else:
            sa = -s
            grad[sa:] += wk * (1 - h[:n-sa])
            grad[:n-sa] -= wk * h[sa:]
    return grad * dx

def smooth_optimize(h, n, alpha, maxiter=1000, lr=0.01, deadline=None):
    """Gradient descent with projection for smooth max."""
    target = n / 2.0
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(maxiter):
        if deadline and time.time() > deadline:
            break
        g = grad_softmax(h, alpha, n)
        # Backtracking line search
        step = lr
        for _ in range(30):
            h_new = project_feasible(h - step * g, target)
            c_new = compute_c5(h_new, n)
            c_cur = compute_c5(h, n)
            if c_new < c_cur or step < 1e-10:
                break
            step *= 0.5
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if step < 1e-12:
            break
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=500, deadline=None):
    """SLSQP with softmax objective."""
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = (c - m) / alpha
        return m + np.log(np.exp(sh).sum()) / alpha
    def obj_grad(x):
        return grad_softmax(x, alpha, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() > deadline:  # won't reach this during minimize
            pass
        return project_feasible(res.x, target)
    except Exception:
        return h

def binary_ls(h, n, deadline, max_sweeps=3000, seed=42):
    """Binary local search."""
    h = binarize(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

                    improved = True
                    break

        if not improved:

    return best_h, best_c

    best_c = compute_c5(best_h, n)
    target = n / 2.0

    # Phase 1: Quick binary local search
    p1 = t0 + 70
    hb, cb = bls(best_h, n, p1, 4000, seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: SLSQP smooth optimization with increasing alpha
    p2 = t0 + 300
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > p2 - 15:
            break
        rem = max(5, p2 - 20 - time.time())
        mi = min(3000, max(200, int(rem * 80)))
        hr = slsqp_opt(best_h.copy(), n, a, mi, p2 - 5)
        c = compute_c5(hr, n)
        if c < best_c:
            best_c, best_h = c, hr.copy()

    # Phase 3: Gradient descent with momentum + restarts
    p3 = t0 + 650
    for restart in range(40):
        if time.time() > p3 - 15:
            break
        h_p = best_h.copy()
        nsw = rng.randint(2, 7)
        for _ in range(nsw):
            a, b = int(rng.randint(n)), int(rng.randint(n))
            h_p[a], h_p[b] = h_p[b], h_p[a]
        h_p = proj(h_p, target)
        a = 5000.0 * (1.0 + rng.random() * 50.0)
        t_rem = t0 + 650 - 10
        t_sa = min(t_rem, time.time() + 45)
        hb2, cb2 = grad_desc(h_p, n, a, 300, 0.003, t_sa)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
        if time.time() < p3 - 40:
            hr = slsqp_opt(hb2.copy(), n, 300000, 800, p3 - 10)
            c = compute_c5(hr, n)
            if c < best_c:
                best_c, best_h = c, hr.copy()

    # Phase 4: Try different n via interpolation
    p4 = t0 + 900
    if time.time() < p4 - 60:
        for n_new in [n + 20, n - 20]:
            if n_new < 50 or n_new > 500 or time.time() > p4 - 60:
                continue
            h_interp = interpolate_h(best_h, n, n_new)
            h_interp = proj(h_interp, n_new / 2.0)
            c_int = compute_c5(h_interp, n_new)
            if c_int < best_c:
                best_c, best_h = c_int, h_interp.copy()
                n = n_new
                target = n / 2.0
            if time.time() < p4 - 30:
                hr = slsqp_opt(best_h.copy(), n, 200000, 1500, p4 - 10)
                c = compute_c5(hr, n)
                if c < best_c:
                    best_c, best_h = c, hr.copy()
                    n = n
                    target = n / 2.0

    # Phase 5: Final polish
    if time.time() < deadline - 30:
        for a in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 30:
                break
            hr = slsqp_opt(best_h.copy(), n, a, 2000, deadline - 20)
            c = compute_c5(hr, n)
            if c < best_c:
                best_c, best_h = c, hr.copy()

    if time.time() < deadline - 10:
        hb, cb = bls(best_h, n, deadline - 5, 2000, seed + 7777)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    best_h = proj(best_h, target)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
