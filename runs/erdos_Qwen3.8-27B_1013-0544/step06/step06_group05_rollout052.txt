            break
        h = np.clip(H.value, 0, 1)
        # Project to sum = n/2
        h = project_feasible(h, n)

            idx = np.arange(n)

            m1 = (n1 >= 0) & (n1 < n)
            grad[m1] += (1.0 - h[n1[m1]])
            n2 = idx + lag
