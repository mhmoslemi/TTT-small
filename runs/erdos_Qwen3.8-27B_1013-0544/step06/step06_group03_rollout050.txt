    # Try a different n (interpolate best solution)
    if best_h is not None and time.time() < deadline - 100:
        for n_try in [n * 2, n * 3 // 2]:
            if n_try >= 1000 or n_try < 50:
                continue
            if time.time() > deadline - 80:
                break
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_best, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(80, deadline - time.time() - 50)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_n(h_interp, n_try, t0 + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish on best solution
    if best_h is not None and time.time() < deadline - 5:
        nf = len(best_h)
        h_new = slsqp_polish(best_h, nf, 5000000, 3, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, nf)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = fast_local_search(best_h, nf, deadline - 1)
        best_h = project_feasible(best_h, nf)
        best_c = compute_c5(best_h, nf)

    if best_h is None:
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = compute_c5(best_h, n0)

    return best_h, best_c, len(best_h)
