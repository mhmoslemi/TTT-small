```python
import numpy as np, time
from scipy.optimize import minimize
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

_CORR_FLAG = 0

def _ensure_corr_flag():
    global _CORR_FLAG
    if _CORR_FLAG:
        return
    a = np.array([1.0, 2.0, 0.0])
    v = np.array([0.0, 1.0, 1.0])
    c = np.correlate(a, v, mode='full')
    _CORR_FLAG = 1 if np.allclose(c, np.array([0.0, 2.0, 3.0])) else -1

def c5_vec(h, n):
    _ensure_corr_flag()
    c = np.correlate(h, 1.0 - h, mode='full')
    if _CORR_FLAG < 0:
        c = c[::-1]
    return c * (2.0 / float(n))

def c5_max(h, n):
    return float(np.max(c5_vec(h, n)))

def _clip_sum(v, lam):
    return float(np.sum(np.clip(v - lam, 0.0, 1.0)))

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    target = float(target)
    n = v.size
    if n == 0:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(52):
        mid = 0.5 * (lo + hi)
        if _clip_sum(v, mid) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-13:
        idx = int(np.argmax(1.0 - x) if r > 0 else np.argmax(x))
        x[idx] = float(np.clip(x[idx] + r, 0.0, 1.0))
    return x

def grad_from_weights(h, n, w):
    dx = 2.0 / float(n)
    g = np.zeros(n)
    for m in np.nonzero(w > 1e-12)[0]:
        k = int(m) - (n - 1)
        wm = float(w[m]) * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += -wm * h[:nv]
        else:
            q = -k
            nv = n - q
            g[q:] += wm * (1.0 - h[:nv])
            g[:nv] += -wm * h[q:]
    return g

def smooth_obj(x, n, alpha):
    c = c5_vec(x, n)
    m = float(c.max())
    if alpha <= 0.0:
        return m
    z = alpha * (c - m)
    z -= z.max()
    return m + float(np.log(np.sum(np.exp(z)))) / alpha

def smooth_grad(x, n, alpha):
    c = c5_vec(x, n)
    m = float(c.max())
    if alpha <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        z = alpha * (c - m)
        z -= z.max()
        w = np.exp(z)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= s
    return grad_from_weights(x, n, w)

def pen_obj(x, n, alpha, mu):
    s = x.sum() - 0.5 * float(n)
    return smooth_obj(x, n, alpha) + mu * s * s

def pen_grad(x, n, alpha, mu):
    s = x.sum() - 0.5 * float(n)
    g = smooth_grad(x, n, alpha)
    g += 2.0 * mu * s
    return g

    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best = c5_max(h, n)
    bounds = (np.zeros(n), np.ones(n))
    if quick:
        cfg = [(2000, 0.1, 30), (20000, 10.0, 35), (200000, 100.0, 25)]
    elif n > 600:
        cfg = [(1000, 0.05, 30), (10000, 1.0, 40), (100000, 10.0, 30)]
    else:
        cfg = [(100, 0.001, 40), (500, 0.05, 50), (2000, 0.2, 55),
               (10000, 1.0, 55), (80000, 8.0, 45)]
    for alpha, mu, mx in cfg:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mx, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 25})
            if res is None or getattr(res, 'x', None) is None or not np.all(np.isfinite(res.x)):
                continue
            x = project_box_sum(res.x, target)
            c = c5_max(x, n)
            if c < best - 1e-15:
                h, best = x, c
        except Exception:
            pass
    return h, best

def lp_direction(h, n, top, trust):
    if linprog is None:
        return None
    c = c5_vec(h, n)
    top = int(min(top, c.size))
    active = np.argsort(c)[-top:]
    dx = 2.0 / float(n)
    rows, bns = [], []
    for m in active:
        g = np.zeros(n)
        k = int(m) - (n - 1)
        if k >= 0:
            nv = n - k
            g[:nv] += dx * (1.0 - h[k:])
            g[k:] += -dx * h[:nv]
        else:
            q = -k
            nv = n - q
            g[q:] += dx * (1.0 - h[:nv])
            g[:nv] += -dx * h[q:]
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        bns.append(float(c[m] - np.dot(g, h)))
    A = np.vstack(rows)
    b = np.array(bns)
    obj = np.zeros(n + 1)
    obj[n] = 1.0
    lbx = np.maximum(0.0, h - trust)
    ubx = np.minimum(1.0, h + trust)
    try:
        res = linprog(obj, A_ub=A, b_ub=b,
                      A_eq=np.ones((1, n)), b_eq=np.array([0.5 * float(n)]),
                      bounds=list(zip(lbx, ubx)) + [(-2.0, 2.0)],
                      options={'time_limit': 2.0, 'disp': False})
    except Exception:
        return None
    if res is None or getattr(res, 'x', None) is None:
        return None
    x = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(x)) or x.min() < -1e-7 or x.max() > 1.0 + 1e-7:
        return None
    if abs(x.sum() - 0.5 * float(n)) > 1e-4:
        return None
    x = project_box_sum(x, 0.5 * float(n))
    p = x - h
    if np.max(np.abs(p)) < 1e-9:
        return None
    return p

def active_lp_phase(h, n, deadline, top=64, trust=0.04):
    target = 0.5 * float(n)
    best = c5_max(h, n)
    tc, toc, failed = float(trust), int(top), 0
    fracs = np.linspace(0.0, 1.0, 25)
    while time.time() < deadline - 2.0:
        p = lp_direction(h, n, toc, tc)
        if p is None:
            tc *= 0.5
            failed += 1
        else:
            best_local, best_x = best, None
            for fr in fracs[1:]:
                if time.time() > deadline - 2.0:
                    break
                x = project_box_sum(h + fr * p, target)
                c = c5_max(x, n)
                if c < best_local:
                    best_local, best_x = c, x.copy()
            if best_x is not None:
                h, best, failed = best_x, best_local, 0
                tc = min(0.25, tc * 1.2)
                toc = min(128, toc + 4)
            else:
                tc *= 0.6
                failed += 1
        if failed >= 3:
            failed = 0
            toc = min(128, toc + 24)
            tc = 0.10
        if tc < 0.002:
            break
    return h, best

def pair_gradient_refine(h, n, deadline, alpha=80000):
    best = c5_max(h, n)
    fracs = np.linspace(0.0, 1.0, 15)
    pc = 4 if n > 500 else 5
    for _ in range(5):
        if time.time() > deadline - 2.0:
            break
        g = smooth_grad(h, n, alpha)
        dm = g - float(np.mean(g))
        if np.max(dm) - np.min(dm) < 1e-10:
            break
        pos = np.argsort(dm)[-pc:]
        neg = np.argsort(dm)[:pc]
        improved = False
        for i in pos:
            if improved:
                break
            for j in neg:
                if i == j:
                    continue
                maxth = min(float(h[i]), 1.0 - float(h[j]))
                if maxth < 1e-9:
                    continue
                best_local, best_x = best, None
                for fr in fracs[1:]:
                    th = fr * maxth
                    x = h.copy()
                    x[i] = h[i] - th
                    x[j] = h[j] + th
                    c = c5_max(x, n)
                    if c < best_local:
                        best_local, best_x = c, x.copy()
                if best_x is not None:
                    h, best, improved = best_x, best_local, True
                    break
        if not improved:
            break
    return h, best

def threshold_candidates(h, n, deadline):
    target = 0.5 * float(n)
    best = c5_max(h, n)
    out = h.copy()
    for t in (0.2, 0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8):
        if time.time() > deadline:
            break
        v = (h >= t).astype(float)
        v = project_box_sum(v, target)
        c = c5_max(v, n)
        if c < best - 1e-14:
            out, best = v.copy(), c
    v = np.where(h < 0.34, 0.0, np.where(h > 0.66, 1.0, 0.5))
    v = project_box_sum(v, target)
    c = c5_max(v, n)
    if c < best - 1e-14:
        out, best = v.copy(), c
    return out, best

def polish_candidate(h, n, deadline, quick=False):
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    best = c5_max(h, n)
    if quick or n > 600:
        mlb, mlp, mp1, mp2 = 40, 90, 25, 12
    elif n > 400:
        mlb, mlp, mp1, mp2 = 100, 150, 30, 15
    else:
        mlb, mlp, mp1, mp2 = 170, 220, 40, 20
    sub = min(deadline - (mlp + mp1 + mp2 + 5), time.time() + mlb)
    if sub > time.time() + 2.0:
        h, best = lbfgs_phase(h, n, sub, quick=quick)
    sub = min(deadline - (mp1 + mp2 + 5), time.time() + mlp)
    if sub > time.time() + 4.0:
        h, best = active_lp_phase(h, n, sub, top=64 if n <= 400 else 48, trust=0.04)
    if time.time() < deadline - (mp2 + 4.0):
        h, best = pair_gradient_refine(h, n, deadline - (mp2 + 3.0), alpha=30000 if n <= 400 else 80000)
    if time.time() < deadline - 6.0:
        h, best = pair_gradient_refine(h, n, deadline - 5.0, alpha=200000 if n <= 400 else 500000)
    if time.time() < deadline - 1.0:
        h, best = threshold_candidates(h, n, deadline)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s) - 6.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2:
            raise ValueError
    except Exception:
        n0 = 300
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = 0.5 + 0.45 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)
    best_h = project_box_sum(h0, 0.5 * float(n0))
    best_c = c5_max(best_h, n0)
    main_dead = max(t0 + 40.0, min(deadline - 350.0, t0 + 620.0))
    if main_dead > deadline - 40.0:
        main_dead = deadline - 40.0
    best_h, best_c = polish_candidate(best_h, n0, main_dead, quick=False)
    if time.time() < deadline - 120.0 and best_c > 0.38081:
        for sig in (0.02, 0.05):
            if time.time() > deadline - 80.0:
                break
            ns = int(len(best_h))
            v = best_h + rng.randn(ns) * sig
            v = project_box_sum(v, 0.5 * float(ns))
            sub = min(deadline - 60.0, time.time() + 55.0)
            if sub <= time.time():
                break
            v, vc = polish_candidate(v, ns, sub, quick=True)
            if vc < best_c:
                best_h, best_c = v, vc
    for n_alt in (400, 600, 750):
        if time.time() > deadline - 70.0:
            break
        n_src = int(len(best_h))
        if n_alt <= n_src:
            continue
        rem = deadline - time.time()
        alloc = min(140.0, rem - 60.0)
        if alloc < 35.0:
            break
        xo = np.linspace(0.0, 2.0, n_src, endpoint=False)
        xn = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_box_sum(hi, 0.5 * float(n_alt))
        ha, ca = polish_candidate(hi, n_alt, time.time() + alloc, quick=(n_alt >= 600))
        if ca < best_c:
            best_h, best_c = ha, ca
    if time.time() < deadline - 8.0:
        best_h, best_c = polish_candidate(best_h, int(len(best_h)), deadline - 2.0, quick=True)
    best_h = project_box_sum(best_h, 0.5 * float(len(best_h)))
    best_c = c5_max(best_h, int(len(best_h)))
    return best_h, float(best_c), int(len(best_h))
```
