import time
import cvxpy as cp

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def fast_greedy(h, n, max_iters=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iters):
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = lag_gradient(h, n, lag)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] - grad[j] < 1e-12:
            break
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-12:
            break
        # Try a few fractions
        applied = False
        for frac in [0.5, 0.25, 0.1, 0.75, 0.05]:
            d = frac * max_d
            h[i] -= d
            h[j] += d

                applied = True

            h[i] += d
            h[j] -= d

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def fast_greedy(h, n, max_iters=20000, try_k=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = lag_gradient(h, n, lag)
        sorted_idx = np.argsort(-grad)
        found = False
        for a in range(min(try_k, n)):
            if found:
                break
            for b in range(min(try_k, n)):
                i = int(sorted_idx[a])
                j = int(sorted_idx[n - 1 - b])
                if i == j:
                    continue
                if grad[i] - grad[j] < 1e-12:
                    break
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-12:

                    d = frac * max_d

                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        found = True
                        break
                    h[i] += d
                    h[j] -= d
                if found:
                    break
        if not found:
            break
    return h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_lag_slsqp(h, n, top_m=5, alpha=50000, maxiter=3000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def active_obj(x):
        dx = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def make_bangbang(n, pattern, rng):
    h = np.zeros(n)
    k = n // 2
    if pattern == 0:
        h[:k] = 1.0
    elif pattern == 1:
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        h = h[:k] if h.sum() > k else h
    elif pattern == 2:
        idx = rng.choice(n, k, replace=False)
        h[idx] = 1.0
    elif pattern == 3:
        blocks = 3
        for i in range(blocks):
            start = int(i * n / blocks)
            end = int((i + 1) * n / blocks)
            mid = (start + end) // 2
            h[mid - k//(2*blocks):mid + k//(2*blocks)] = 1.0
    h = project_feasible(h, n)
    return h
