def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n

                       constraints=cons,

        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-15 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T = max(min_T, T * cool)
        if no_improve > patience:
            nk = 4 + rng.randint(10)
            zi = zeros_idx.copy()
            oi = ones_idx.copy()
            for _ in range(nk):
                ii = zi[rng.randint(len(zi))]
                jj = oi[rng.randint(len(oi))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = max(t_start * 0.6, T * 4.0)
    return best_h, best_c

def greedy_first_improve(h_init, n, deadline, seed=42, max_rounds=50):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                nk = 3 + rng.randint(6)
                for _ in range(nk):
                    a = oi[rng.randint(len(oi))]
                    b = zi[rng.randint(len(zi))]
                    h[a] = 0.0
                    h[b] = 1.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warmup (8% time)
    p1 = t0 + 0.08 * (budget_s - 5)
    h = project_feasible(h0, n)
    for alpha in [200, 800, 3000]:
        if time.time() > p1:
            break
        h = slsqp_single(h, n, alpha, 400, 1e-13)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Extended binary SA (55% time) - 5 attempts
    p2 = t0 + 0.63 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    time_per = (p2 - t0) / 5.0

    starts = []
    starts.append((h_bin, seed, 0.025))
    starts.append((round_to_binary(best_h + 0.1 * rng.randn(n), n), seed+777, 0.03))
    starts.append((round_to_binary(best_h + 0.25 * rng.randn(n), n), seed+1555, 0.04))
    rh = np.zeros(n)
    rh[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append((rh, seed+2333, 0.05))
    starts.append((round_to_binary(best_h + 0.4 * rng.randn(n), n), seed+3111, 0.05))

    for idx_sa in range(5):
        if time.time() > p2 - 1:
            break
        sh, ss, ts = starts[idx_sa]
        end_t = t0 + (idx_sa + 1) * time_per
        hb, cb = binary_sa(sh, n, min(end_t, p2 - 1), seed=ss, t_start=ts, patience=12000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy polish (8% time)
    p3 = t0 + 0.72 * (budget_s - 5)
    if time.time() < p3 - 3:
        hg, cg = greedy_first_improve(best_h, n, p3, seed=seed+50, max_rounds=80)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Phase 4: SLSQP refinement from best (20% time)
    p4 = t0 + 0.93 * (budget_s - 5)
    if time.time() < p4 - 10:
        hp = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [2000, 8000, 30000, 100000, 400000, 1500000]:
            if time.time() > p4:
                break
            hp = slsqp_single(hp, n, alpha, 1200, 1e-14)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 5: Final SA + SLSQP (7% time)
    if time.time() < dl - 8:
        h_bin2 = round_to_binary(best_h, n)
        end_t = dl - 12
        hb2, cb2 = binary_sa(h_bin2, n, end_t, seed=seed+9999, t_start=0.012, patience=8000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        if time.time() < dl - 2:
            hp2 = project_feasible(best_h, n)
            hp2 = slsqp_single(hp2, n, 500000, 600, 1e-15)
            c = compute_c5(hp2, n)
            if c < best_c:
                best_c = c
                best_h = hp2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
