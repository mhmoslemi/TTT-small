def project_box_sum(h, target):

import numpy as np
import time
from scipy.optimize import minimize

def fft_len(n):
    m = 2 * n - 1
    r = 1
    while r < m:
        r <<= 1
    return r

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        s = float(np.clip(h - m, 0.0, 1.0).sum())
        if s > target:
            lo = m
        else:
            hi = m
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(14):
        res = target - float(h.sum())
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += res
        if h[idx] < 0.0:
            h[idx] = 0.0
        elif h[idx] > 1.0:
            h[idx] = 1.0
    return h

def compute_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def fallback_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, int(n), endpoint=False)
    h = np.full(int(n), 0.5)
    for m in range(1, 5):
        amp = rng.uniform(0.06, 0.24)
        ph = 2.0 * np.pi * rng.random()
        h += amp * np.sin(np.pi * m * x + ph)
    return project_box_sum(h, float(n) / 2.0)

def make_perturb(h, seed, scale):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return h.copy()
    if scale <= 1e-14:
        return project_box_sum(h, float(n) / 2.0)
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    noise = rng.normal(0.0, float(scale), n)

        noise += amp * rng.uniform(-1.0, 1.0) * np.cos(np.pi * m * x)

def shuffle_blocks(h, seed, block):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n < 2 * int(block) or int(block) <= 1:
        return project_box_sum(h.copy(), float(n) / 2.0)
    m = n // int(block)
    rng = np.random.RandomState(seed)
    order = np.arange(m)
    rng.shuffle(order)
    out = np.empty(n, dtype=float)
    for b in range(m):
        out[b * block:(b + 1) * block] = h[order[b] * block:(order[b] + 1) * block]
    out[m * block:] = h[m * block:]
    return project_box_sum(out, float(n) / 2.0)

def lse_val_grad(h, beta):
    n = len(h)
    N = fft_len(n)
    h = np.asarray(h, dtype=float)
    hf = np.fft.rfft(h, N)
    ac = np.fft.irfft(hf * np.conjugate(hf), N)[:n]

    total = float(np.sum(h))
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(h)

    O = np.empty(2 * n - 1, dtype=float)
    center = n - 1
    O[center] = total - float(ac[0])
    for s in range(1, n):
        q = float(ac[s])
        O[center + s] = prefix[n - s] - q
        O[center - s] = total - prefix[s] - q

    mo = float(np.max(O))
    z = float(beta) * (O - mo)
    ep = np.exp(z)
    sump = float(ep.sum())
    if sump <= 0.0:
        p = np.zeros_like(O)
        p[int(np.argmax(O))] = 1.0
    else:
        p = ep / sump
    val = mo + np.log(max(sump, 1e-300)) / float(beta)
    grad = weighted_grad_fft(h, p, hf, N)
    return val, grad

def weighted_grad_fft(h, p, hf, N):
    n = len(h)
    center = n - 1
    g = p[center] * (1.0 - 2.0 * h)

    u = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    if n > 1:
        u[1:] = p[center + 1:]
        v[1:] = p[:center][::-1]

    cu = np.zeros(n, dtype=float)
    cv = np.zeros(n, dtype=float)
    if n > 1:
        cu[1:] = np.cumsum(u[1:])
        cv[1:] = np.cumsum(v[1:])

    r = np.arange(n, dtype=int)
    g += cu[n - 1 - r] + cv[r]

    uf = np.fft.rfft(u, N)
    vf = np.fft.rfft(v, N)
    hr = h[::-1]
    hrf = np.fft.rfft(hr, N)

    back_u = np.fft.irfft(uf * hf, N)[:n]
    back_v = np.fft.irfft(vf * hf, N)[:n]
    fwd_u = np.fft.irfft(uf * hrf, N)[:n][::-1]
    fwd_v = np.fft.irfft(vf * hrf, N)[:n][::-1]

    g -= (back_u + fwd_u + back_v + fwd_v)
    return g

            h_try = make_perturb(best_h, int(seed + 9000 + 11 * gi + att), float(scale))
            abort = min(phase3_end - 3.0, time.time() + 55.0)
            if abort <= time.time() + 0.5:
                break
            h_new = run_lbfgs(h_try, float(beta), float(abort), 30000)
            h_t, c_t = trial_candidate(h_new)
            if h_t is not None and c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()

    if time.time() < deadline - 8.0:
        if len(best_h) <= 320:
            h_new = run_slsqp(best_h, 500.0, float(deadline - 3.0), 30000)
            h_t, c_t = trial_candidate(h_new)
            if h_t is not None and c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()

    if time.time() < deadline - 4.0:
        beta_f = 1000.0 if len(best_h) <= 320 else 600.0
        h_new = run_lbfgs(best_h, float(beta_f), float(deadline - 2.0), 30000)
        h_t, c_t = trial_candidate(h_new)
        if h_t is not None and c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()

    saved_h = best_h.copy()
    saved_c = best_c
    best_h = project_box_sum(best_h, float(best_h.size) / 2.0)
    best_c = compute_c5(best_h)
    if best_c > saved_c:
        best_h = saved_h
        best_c = saved_c

    return best_h, float(best_c), int(best_h.size)
