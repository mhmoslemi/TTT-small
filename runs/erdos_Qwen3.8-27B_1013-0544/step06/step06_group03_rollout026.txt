def project_feasible(h, n):

            if tot < 1e-14:

    shifted = alpha * (c - m)

A better approach: use the subgradient of the max function. The subgradient of max(c[k]) with respect to h is the uniform combination of the gradients of all active lags (those at the maximum). So:
- Find all lags within epsilon of the max
- Use the average gradient of those lags as the descent direction
- Move in the negative gradient direction (projected onto the constraint set)

This is essentially what SLSQP with a smooth max does, but I can do it faster in a loop.

Let me implement a "gradient projection" method:
1. Compute c for all lags
2. Find active set: lags within epsilon of max
3. Compute average gradient of active lags
4. Take a step in the negative gradient direction
5. Project onto feasible set (0 <= h <= 1, sum(h) = n/2)
6. Repeat

This is basically projected subgradient descent. The step size can be adapted.

Actually, this is very similar to what SLSQP does internally. The advantage of my approach is that I can do many more iterations (since each iteration is O(n²) for the correlation + O(n) for the gradient, vs O(n³) or more for SLSQP).

def subgradient_search(h, n, deadline, eps=0.001):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    step = 0.01
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h, n)
        current_max = c.max()
        if current_max < best_c:
            best_c = current_max
        
        # Active set: lags within eps of max
        active = np.where(c >= current_max - eps)[0]
        
        # Compute average gradient
        avg_grad = np.zeros(n)
        for L in active:
            avg_grad += lag_gradient(h, n, L)
        avg_grad /= len(active)
        
        # Take step in negative gradient direction
        h_new = h - step * avg_grad
        h_new = project_feasible(h_new, n)
        
        # Check if it improved
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
            step = min(0.05, step * 1.1)  # increase step
        else:
            step *= 0.9  # decrease step
            if step < 0.0001:
                # Try random restart
                h = h + 0.01 * np.random.randn(n)
                h = project_feasible(h, n)
                step = 0.01
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, L):
    lag = L - (n - 1)
    grad = np.zeros(n)
    if lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    elif lag < 0:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def fast_local_search(h, n, deadline, rng):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    no_improve = 0
    for rnd in range(200000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        current_max = c.max()
        if current_max < best_c:
            best_c = current_max
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500 and rnd % 100 == 0:
            idx = rng.randint(n)
            jdx = rng.randint(n)
            if idx != jdx and h[idx] > 0.03 and h[jdx] < 0.97:
                d = 0.02 * rng.random()
                h[idx] -= d
                h[jdx] += d
                h = project_feasible(h, n)
            no_improve = 0
        top_idx = np.argsort(c)[-3:]
        grads = np.zeros((3, n))
        for mi in range(3):
            grads[mi] = lag_gradient(h, n, top_idx[mi])
        g0 = grads[0]
        order = np.argsort(g0)
        n_try = min(60, n // 5)
        low_pos = order[:n_try]
        high_pos = order[-n_try:]
        improved = False
        for ii in range(len(high_pos)):
            i = high_pos[ii]
            if h[i] < 1e-7:
                continue
            for jj in range(len(low_pos)):
                j = low_pos[jj]
                if i == j or h[j] > 1.0 - 1e-7:
                    continue
                d = min(0.004, h[i] * 0.35, (1.0 - h[j]) * 0.35)
                if d < 1e-10:
                    continue
                changes = d * (grads[:, j] - grads[:, i])
                if changes[0] >= -1e-13:
                    continue
                if (c[top_idx] + changes).max() <= current_max + 1e-13:
                    h[i] -= d
                    h[j] += d
                    improved = True
                    break
            if improved:
                break
        if not improved:
            d_small = 0.0008
            for ii in range(min(40, len(high_pos))):
                i = high_pos[ii]
                if h[i] < 1e-7:
                    continue
                for jj in range(min(40, len(low_pos))):
                    j = low_pos[jj]
                    if i == j or h[j] > 1.0 - 1e-7:
                        continue
                    d2 = min(d_small, h[i] * 0.35, (1.0 - h[j]) * 0.35)
                    if d2 < 1e-11:
                        continue
                    changes = d2 * (grads[:, j] - grads[:, i])
                    if changes[0] >= -1e-14:
                        continue
                    if (c[top_idx] + changes).max() <= current_max + 1e-13:
                        h[i] -= d2
                        h[j] += d2
                        improved = True
                        break
                if improved:
                    break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def subgradient_search(h, n, deadline, eps=0.002):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.008
    for it in range(50000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        current_max = c.max()
        if current_max < best_c:
            best_c = current_max
            step = min(0.04, step * 1.05)
        else:
            step *= 0.95
            if step < 0.0003:
                h = h + 0.008 * np.random.randn(n)
                h = project_feasible(h, n)
                step = 0.008
        active = np.where(c >= current_max - eps)[0]
        avg_grad = np.zeros(n)
        for L in active:
            avg_grad += lag_gradient(h, n, L)
        avg_grad /= max(1, len(active))
        h_new = h - step * avg_grad
        h_new = project_feasible(h_new, n)
        if compute_c5(h_new, n) < best_c:
            best_c = compute_c5(h_new, n)
            h = h_new
        else:
            h = h_new
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    remaining = deadline - time.time()
    if remaining > 40:
        t_slsqp = time.time() + min(45, remaining * 0.25)
        h1 = slsqp_optimize(best_h, n, 50000, maxiter=3000, ftol=1e-15)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    if time.time() < deadline - 30:
        t_sg = time.time() + min(50, (deadline - time.time()) * 0.3)
        h2, c2 = subgradient_search(best_h, n, t_sg)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    if time.time() < deadline - 20:
        t_ls = time.time() + min(80, (deadline - time.time()) * 0.5)
        h3, c3 = fast_local_search(best_h, n, t_ls, rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    if time.time() < deadline - 15:
        h4 = slsqp_optimize(best_h, n, 200000, maxiter=5000, ftol=1e-16)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    if time.time() < deadline - 5:
        h5, c5 = fast_local_search(best_h, n, deadline - 1, rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

        h_init = np.array(initial_h_values, dtype=float).copy()

        best_h, best_c = h_opt, c_opt

    for n_try in n_list:

        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:

            best_h = h_opt
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 30) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_final, c_final = fast_local_search(best_h, n, deadline - 1, rng)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
