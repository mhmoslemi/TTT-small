def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def _smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_active(x, n, alpha, top_idx):

                       bounds=bounds, constraints=cons,

    except Exception:
        return h.copy()

def subgradient_optimize(h, n, deadline, n_iter_max=50000):
    """Subgradient method for the minimax problem with projected steps."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    eta = 0.1
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Polyak-like step with decay
        eta_t = eta / (1.0 + 0.001 * it)
        momentum = beta * momentum - eta_t * grad
        h_new = h_curr + momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

        momentum = beta * momentum - eta_t * grad
        h_new = h_curr + momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def compress_binary(h, n, power):
    h_new = np.where(h >= 0.5,
                     0.5 + 0.5 * (2.0 * (h - 0.5)) ** power,
                     0.5 - 0.5 * (2.0 * (0.5 - h)) ** power)
    return project_feasible(h_new, n)

def targeted_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    n_inner = 0
    while time.time() < deadline and n_inner < 500:
        n_inner += 1
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if lag > 0:
            products = h[m:] * (1.0 - h[:n - m])
        else:
            products = h[:n - m] * (1.0 - h[m:])
        n_worst = min(15, n - m)
        worst_idx = np.argsort(products)[-n_worst:]
        improved = False
        for idx in worst_idx:
            i = int(idx)
            j = int(idx + m)
            hi, hj = h[i], h[j]
            if hi > hj:
                deltas = [(hi - hj) * s for s in [0.3, 0.15, 0.05, 0.02]]
                for d in deltas:
                    if d < 1e-9:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    h[i] += d
                    h[j] -= d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
            else:
                deltas = [(hj - hi) * s for s in [0.3, 0.15, 0.05, 0.02]]
                for d in deltas:
                    if d < 1e-9:
                        continue
                    h[j] -= d
                    h[i] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    h[j] += d
                    h[i] -= d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
            if not improved and time.time() < deadline:
                d = 0.03
                h[i] += d
                h[j] -= d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
    return h, best_c

def optimize_one(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 25:
        sub_dl = min(time.time() + 25, deadline - 25)
        h_sub, c_sub = subgradient_optimize(h, n, sub_dl, n_iter_max=8000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub
    if time.time() < deadline - 20:
        for alpha in [5000, 20000, 50000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_full(best_h, n, alpha, maxiter=2000, ftol=1e-13)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 20:
        for _ in range(4):
            if time.time() > deadline - 15:
                break
            c_arr = compute_all_lags(best_h, n)
            top_idx = np.argsort(c_arr)[-8:]
            for alpha in [10000, 50000, 100000]:
                if time.time() > deadline - 15:
                    break
                h_new = slsqp_active(best_h, n, alpha, top_idx, maxiter=2000, ftol=1e-13)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    if time.time() < deadline - 25:
        for power in [2.0, 4.0, 8.0]:
            if time.time() > deadline - 25:
                break
            h_comp = compress_binary(best_h, n, power)
            c_comp = compute_c5(h_comp, n)
            if c_comp < best_c:
                best_c = c_comp
                best_h = h_comp
            if time.time() < deadline - 15:
                c_a = compute_all_lags(h_comp, n)
                ti = np.argsort(c_a)[-5:]
                h_new = slsqp_active(h_comp, n, 30000, ti, maxiter=1500, ftol=1e-13)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    if time.time() < deadline - 5:
        c_arr = compute_all_lags(best_h, n)
        ti = np.argsort(c_arr)[-5:]
        h_new = slsqp_active(best_h, n, 200000, ti, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 2:
        best_h, best_c = targeted_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(280, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_one(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [150, 200, 250, 400]
    if h_init is not None:
        n_candidates = [x for x in n_candidates if x != n_init]
    for n_try in n_candidates:
        if time.time() > deadline - 90:
            break
        remaining = deadline - time.time() - 40
        if remaining < 70:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_i = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_i, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_one(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 50) / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 45:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 45) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_one(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if time.time() < deadline - 8:
            c_arr = compute_all_lags(best_h, n)
            ti = np.argsort(c_arr)[-5:]
            h_new = slsqp_active(best_h, n, 500000, ti, maxiter=5000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        if time.time() < deadline - 2:
            best_h, best_c = targeted_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
