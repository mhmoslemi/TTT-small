```python
                    best_c_test = c_test
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 1.0; h[j] = 0.0
            update_val(val, n, i, 1.0)
            update_val(val, n, j, -1.0)
            c = best_c_test
            current_max = best_new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            found_improvement = True
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        
        if not found_improvement:
            stuck += 1
            if stuck >= 20:
                # Random perturbation
                for _ in range(rng.randint(2, 6)):
                    ii = zeros[rng.randint(len(zeros))]
                    jj = ones[rng.randint(len(ones))]
                    h[ii] = 1.0; h[jj] = 0.0
                    update_val(val, n, ii, 1.0)
                    update_val(val, n, jj, -1.0)
                c = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
                current_max = c.max()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stuck = 0
    
    return best_h, float(best_max * 2.0 / n)
```

The issue with this approach is that for each (i, j) candidate, we do a full O(n²) correlation. If there are, say, 20 candidates per i and 125 i's, that's 2500 correlations per sweep = 2500 * 62500 = 156M ops. That might be too slow in Python.

Let me optimize: instead of copying h and computing full correlation, I can compute the exact Δc for the (i,j) swap using the correction formula, which is O(n_corr) = O(n) per swap.

Actually, let me just compute Δc exactly in O(1) per k using the formula I derived:

Δc[k] = val[j,k] - val[i,k] + correction(k; i,j)

where correction is non-zero for at most 3 k values. So:
