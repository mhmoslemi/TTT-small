import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Compute gradient of overlap at a given lag w.r.t. h (without dx factor)."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def fw_vertex(c, n):
    """Find s in {0≤s≤1, sum(s)=n/2} minimizing c·s."""
    idx = np.argsort(c)
    s = np.zeros(n)
    k = n // 2
    s[idx[:k]] = 1.0
    return s

def frank_wolfe(h, n, max_iter=5000, tol=1e-10):
    """Frank-Wolfe algorithm for min max overlap."""
    h_curr = h.copy()
    dx = 2.0 / n
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for t in range(max_iter):
        # Compute all lags
        c_all = compute_all_lags(h_curr, n)
        c_max = c_all.max()
        
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        
        # Find active lags (within tolerance of max)
        active = np.where(c_all >= c_max - 1e-10)[0]
        
        # Compute subgradient (average of active lag gradients)
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            grad += lag_gradient(h_curr, n, lag)
        grad /= len(active)
        
        # Find FW vertex
        s = fw_vertex(grad * dx, n)  # gradient of objective includes dx
        
        # Gap computation (for convergence check)
        gap = (h_curr - s) @ (grad * dx)
        if gap < tol:
            break
            
        # Step size
        alpha = 2.0 / (t + 2)

        best_obj = c_max
        for _ in range(20):
            h_try = h_curr + alpha * (s - h_curr)
            h_try = project_feasible(h_try, n)
            obj = compute_c5(h_try, n)
            if obj < best_obj - 1e-12:
                best_obj = obj
                h_curr = h_try
                break
            alpha *= 0.7
        else:
            break
    
    return best_h, best_c

def smooth_objective(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    if top_idx is not None:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c = c[top_idx]
    else:
        c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    if top_idx is not None:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c_active = c[top_idx]
    else:
        c = np.correlate(x, 1 - x, mode='full') * dx
        c_active = c
    m = c_active.max()
    shifted = alpha * (c_active - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    w = weights / weights.sum()  # softmax weights
    
    grad = np.zeros(n)
    if top_idx is not None:

            grad += w[i] * lag_gradient(x, n, lag) * dx
    else:
        for i, k_idx in enumerate(range(2 * n - 1)):
            lag = k_idx - (n - 1)
            grad += w[i] * lag_gradient(x, n, lag) * dx
    return grad

def slbfgs_optimize(h, n, alpha, maxiter=3000):
    """L-BFGS-B optimization of smoothed objective."""
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

def fw_step(h, n, max_iter=10000, tol=1e-12, deadline=None):
    h_curr = h.copy()
    dx = 2.0 / n
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for t in range(max_iter):
        if deadline and time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        c_max = c_all.max()
        
        # Active lags
        tol_lag = max(1e-10, c_max * 1e-6)
        active = np.where(c_all >= c_max - tol_lag)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k_idx in active:
            lag = int(k_idx) - (n - 1)
            grad += lag_grad(h_curr, n, lag)
        grad *= dx / len(active)
        
        # FW vertex: minimize grad·s s.t. 0≤s≤1, sum(s)=n/2
        idx = np.argsort(grad)
        s = np.zeros(n)
        s[idx[:n//2]] = 1.0
        
        # Gap
        gap = (h_curr - s) @ grad
        if gap < tol:
            break
        
        # Step size with line search
        alpha = 2.0 / (t + 2)
        improved = False
        for _ in range(30):
            h_try = h_curr + alpha * (s - h_curr)
            h_try = np.clip(h_try, 0, 1)
            if abs(h_try.sum() - n/2) < 1e-10:
                pass
            else:
                h_try = project_feasible(h_try, n)
            obj = compute_c5(h_try, n)
            if obj < c_max - 1e-14:
                h_curr = h_try
                if obj < best_c:
                    best_c = obj
                    best_h = h_curr.copy()
                improved = True
                break
            alpha *= 0.6
        if not improved:
            # Try smaller step
            alpha = 0.01
            h_try = h_curr + alpha * (s - h_curr)
            h_try = project_feasible(h_try, n)
            obj = compute_c5(h_try, n)
            if obj < c_max - 1e-14:
                h_curr = h_try
                if obj < best_c:
                    best_c = obj
                    best_h = h_curr.copy()
            else:
                break
    
    return best_h, best_c

def proximal_gradient(h, n, deadline, rho=100.0, lr=None):
    h_curr = h.copy()
    t_var = compute_c5(h_curr, n)
    dx = 2.0 / n
    best_h = h_curr.copy()
    best_c = t_var
    
    if lr is None:
        lr = 0.5 / (n * rho * dx)  # rough Lipschitz estimate
    
    for it in range(10000):
        if time.time() > deadline:
            break
        
        # Compute all overlaps
        c_all = np.correlate(h_curr, 1 - h_curr, mode='full')  # without dx
        c_all_scaled = c_all * dx
        
        # Gradient of t + (rho/2) * sum [max(0, c_k*dx - t)]²
        # = (rho * dx) * sum_{k: c_k*dx > t} (c_k*dx - t) * ∂c_k/∂h
        active = c_all_scaled > t_var
        if not active.any():
            break
        
        grad = np.zeros(n)
        n_active = active.sum()
        for k_idx in np.where(active)[0]:
            lag = int(k_idx) - (n - 1)
            grad += (c_all_scaled[k_idx] - t_var) * lag_grad(h_curr, n, lag)
        grad *= rho * dx / n_active
        
        # Gradient wrt t: 1 - rho * sum_{active} (c_k*dx - t)
        grad_t = 1.0 - rho * np.sum(c_all_scaled[active] - t_var)
        
        # Update
        t_new = t_var - lr * grad_t
        h_new = h_curr - lr * grad
        
        # Project h to [0,1] and sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Project sum
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                room = 1 - h_new
                tot = room.sum()
                if tot > 1e-14:
                    h_new += (diff / tot) * room
            else:
                tot = h_new.sum()
                if tot > 1e-14:
                    h_new += (diff / tot) * h_new
            h_new = np.clip(h_new, 0, 1)
        
        t_var = max(0, t_new)
        h_curr = h_new
        
        c_actual = compute_c5(h_curr, n)
        if c_actual < best_c:
            best_c = c_actual
            best_h = h_curr.copy()
        
        # Adaptive learning rate
        if it % 100 == 99:
            c_new = compute_c5(h_curr, n)
            if c_new > best_c * 1.1:
                lr *= 0.5
    
    return best_h, best_c

def compute_all_lags(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:

        g[m:] += 1.0 - h[:n-m]
    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def frank_wolfe(h, n, deadline, max_iter=50000):
    h_c = h.copy()
    dx = 2.0 / n
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    for t in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_c, n)
        c_max = c.max()
        if c_max < best_c - 1e-13:
            best_c = c_max
            best_h = h_c.copy()
        tol_a = max(1e-10, c_max * 1e-6)
        active = np.where(c >= c_max - tol_a)[0]
        g = np.zeros(n)
        for ki in active:
            g += lag_grad(h_c, n, int(ki) - (n - 1))
        g *= dx / len(active)
        idx = np.argsort(g)
        s = np.zeros(n)
        s[idx[:n // 2]] = 1.0
        gap = (h_c - s) @ g
        if gap < 1e-12:
            break
        alpha = 2.0 / (t + 2)
        improved = False
        for _ in range(40):
            h_try = h_c + alpha * (s - h_c)
            h_try = project_feasible(h_try, n)
            obj = compute_c5(h_try, n)
            if obj < c_max - 1e-14:
                h_c = h_try
                if obj < best_c:
                    best_c = obj
                    best_h = h_c.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if t > 200:
                break
    return best_h, best_c

def smooth_slsqp(h, n, alphas, deadline, top_m=5, maxiter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        c = compute_all_lags(best_h, n)
        top = np.argsort(c)[-top_m:]

        def _obj(x, _n=n, _top=top, _a=alpha):
            cc = np.correlate(x, 1 - x, mode='full') * (2.0 / _n)
            a = cc[_top]
            m = a.max()
            sh = _a * (a - m)
            mx = sh.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / _a)
            return float(m + np.log(np.sum(np.exp(sh))) / _a)

        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coord_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                i = int(np.argmax(h * (1 - h)))
                if h[i] >= 0.5:
                    h[i] = min(1.0, h[i] + d)
                else:
                    h[i] = max(0.0, h[i] - d)
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                else:
                    break
            else:
                m = abs(lag)
                if m >= n:
                    break
                if lag > 0:
                    contrib = h[:n - m] * (1 - h[m:])
                else:
                    contrib = h[m:] * (1 - h[:n - m])
                i_best = int(np.argmax(contrib))
                if lag > 0:
                    a, b = i_best, i_best + m
                else:
                    a, b = i_best + m, i_best
                improved = False
                for src, dst in [(a, b), (b, a)]:
                    if h[src] < 1e-6:
                        continue
                    delta = min(d, h[src] * 0.5)
                    if delta < 1e-8:
                        continue
                    h[src] -= delta
                    h[dst] += delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    h[src] += delta
                    h[dst] -= delta
                    h = project_feasible(h, n)
                if not improved:
                    break
    return h, best_c

def interp_to_n(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_s = np.linspace(0, 2, n_src)
    x_d = np.linspace(0, 2, n_dst)
    return np.interp(x_d, x_s, h_src)

def optimize_single(h, n, deadline):

    best_c = compute_c5(h, n)
    if time.time() < deadline - 5:

    best_h = project_feasible(best_h, n)

    return best_h, best_c
