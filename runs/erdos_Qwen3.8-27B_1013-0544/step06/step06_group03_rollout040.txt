def smooth_max_grad(h, n, alpha, top_idx=None):
    """Compute value and gradient of smooth max of selected lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_idx is not None:
        c_active = c[top_idx]
    else:
        c_active = c
    m = c_active.max()

    # Numerical stability
    mx = shifted.max()
    if mx > 500:
        shifted = shifted - mx
        log_sum = mx + np.log(np.sum(np.exp(shifted)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    val = m + log_sum / alpha
    weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))

    c = compute_all_lags(h, n)

    top_idx = np.argsort(c)[-top_k:]

    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Subgradient exploration
    if time.time() < deadline - 20:
        for alpha in [500, 2000, 10000]:
            if time.time() > deadline - 15:
                break
            alloc = time.time() + min(15, (deadline - time.time()) * 0.25)
            h_sub, c_sub = subgrad_descent(best_h, n, alpha, 500, alloc, top_k=10)
            if c_sub < best_c:
                best_c = c_sub
                best_h = h_sub
    # Phase 2: SLSQP active set
    if time.time() < deadline - 10:
        alloc = min(60, (deadline - time.time()) * 0.5)
        h_as, c_as = active_set_slsq(best_h, n, time.time() + alloc,
                                      alphas=[5000, 20000, 100000, 500000], top_k=10, maxiter=2000)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as
    # Phase 3: Fine local search
    if time.time() < deadline - 3:
        h_new, c_new = multi_lag_local_search(best_h, n, deadline - 1, top_k=8)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(300, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    # Try other n values
    n_candidates = [200, 250, 350, 400, 150]
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(80, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Random restarts
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        max_restarts = int(remaining / 40)
        for r in range(max_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 35) * 0.5)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 1000000, 3000, top_k=15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = multi_lag_local_search(best_h, n, deadline - 1, top_k=10)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
