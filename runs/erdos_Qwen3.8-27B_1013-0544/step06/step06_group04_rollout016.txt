```python
def logsumexp_gradient(h, n, alpha):
    """Compute gradient of (1/alpha) * log(sum(exp(alpha * corr_k))) wrt h."""
    corr = np.correlate(h, 1-h, mode='full')
    # Numerical stability
    max_val = np.max(corr)
    W = np.exp(alpha * (corr - max_val))
    Z = W.sum()
    first = np.correlate(W, 1-h, mode='full')[:n]
    second = np.correlate(W, h, mode='full')[:n]
    grad = (first - second) / Z
    obj = (np.log(Z) + alpha * max_val) / alpha
    return obj, grad
```

def project_feasible_v2(h):
    """Project h onto {0 <= h <= 1, sum(h) = len(h)/2}."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    # Bisection on tau
    lo, hi = -2.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2
        projected = np.clip(h - tau, 0, 1)
        s = projected.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = (lo + hi) / 2
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def lse_objgrad(x, n, alpha):
    corr = np.correlate(x, 1 - x, mode='full')
    mv = np.max(corr)
    W = np.exp(alpha * (corr - mv))
    Z = W.sum()
    obj = np.log(Z) / alpha + mv
    grad = (np.correlate(W, 1 - x, mode='full')[:n] - np.correlate(W, x, mode='full')[:n]) / Z
    return obj, grad

def slsqp_opt(h, n, alpha, maxiter=1000):
    def obj(x):
        return lse_objgrad(x, n, alpha)[0]
    def jac(x):
        return lse_objgrad(x, n, alpha)[1]
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x)
    except Exception:
        return h

def swap_delta(h, i, j, n):
    m = np.arange(2 * n - 1) - (n - 1)
    d = np.zeros(2 * n - 1)
    mi_pos = (i + m) >= 0
    mi_pos &= (i + m) < n
    d[mi_pos] += 1.0 - h[i + m[mi_pos]]
    mi_neg = (i - m) >= 0
    mi_neg &= (i - m) < n
    d[mi_neg] -= h[i - m[mi_neg]]
    mj_pos = (j + m) >= 0
    mj_pos &= (j + m) < n
    d[mj_pos] -= 1.0 - h[j + m[mj_pos]]
    mj_neg = (j - m) >= 0
    mj_neg &= (j - m) < n
    d[mj_neg] += h[j - m[mj_neg]]
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        d[k1] += 2.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        d[k2] += 2.0
    return d

def local_search(h, n, deadline, max_sweeps=5000, seed=42):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[int(i)], h[int(j)] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def projected_gradient(h, n, alpha, max_iter=3000, deadline=None):
    h = h.copy()
    obj, grad = lse_objgrad(h, n, alpha)
    best_obj = obj
    best_h = h.copy()
    lr = 1.0 / (alpha * 3)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        step = lr
        found = False
        for _ in range(40):
            h_trial = project_feasible(h - step * grad)
            obj_trial, _ = lse_objgrad(h_trial, n, alpha)
            if obj_trial <= obj:
                found = True
                break
            step *= 0.5
        if found:
            h = h_trial
            obj = obj_trial
        _, grad = lse_objgrad(h, n, alpha)
        lr = step * 1.5
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    return best_h, best_obj
