        if abs(diff) < 1e-13:
            break
        if diff > 0:

            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:
                break

    vals = c_ks + d * s_ks
    if self_idx >= 0:
        vals = vals.copy()
        vals[self_idx] -= d * d
    m = float(np.max(vals))
    if um > m:

    return m * dx + penalty * (sumh + d - target) ** 2

    # Phase 1: Main exact CD on initial n
    a1 = min(450, (deadline - t0) * 0.4)
    if a1 > 30:
        h1, c1 = exact_cd(bh, n0, maxp=50000, deadline=t0 + a1, pen=5.0, ng=30, rng=rng)
        h1 = project_feasible(h1, n0)
        c1 = compute_c5(h1, n0)
        if c1 < bc:
            bc, bh = c1, h1

    # Phase 2: Try n=500 via interpolation
    if time.time() < deadline - 300:
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, 500)
        h5 = project_feasible(np.interp(x_new, x_old, bh), 500)
        a2 = min(220, (deadline - time.time()) * 0.25)
        if a2 > 60:
            h2, c2 = exact_cd(h5, 500, maxp=20000, deadline=time.time() + a2, pen=5.0, ng=28, rng=rng)
            h2 = project_feasible(h2, 500)
            c2 = compute_c5(h2, 500)
            if c2 < bc:
                bc, bh = c2, h2

    # Phase 3: Subgradient steps on best
    nb = len(bh)
    if time.time() < deadline - 80:
        hsg = bh.copy()
        lr = 0.02
        for i in range(200):
            if time.time() > deadline - 60:
                break
            hsg = subgrad_step(hsg, nb, lr, nb / 2.0)
            csg = compute_c5(hsg, nb)
            if csg < bc:
                bc, bh = csg, hsg
            lr *= 0.99

    # Phase 4: Multi-restart
    if time.time() < deadline - 80:
        rem = deadline - time.time() - 40
        nr = max(2, int(rem / 50))
        for r in range(nr):
            if time.time() > deadline - 50:
                break
            if r % 3 == 0:
                hp = project_feasible(bh + rng.normal(0, 0.025, nb), nb)
            elif r % 3 == 1:
                al = 1.3 + 0.4 * r
                tmp = np.clip(bh, 0.01, 1.0) ** al
                hp = project_feasible(tmp, nb)
            else:
                hp = project_feasible(bh + rng.normal(0, 0.06, nb), nb)
            ar = min(40, (deadline - time.time() - 25) * 0.5)
            if ar < 15:
                break
            hr, cr = exact_cd(hp, nb, maxp=15000, deadline=time.time() + ar, pen=5.0, ng=25, rng=rng)
            hr = project_feasible(hr, nb)
            cr = compute_c5(hr, nb)
            if cr < bc:
                bc, bh = cr, hr

    # Phase 5: Final polish CD with low penalty
    if time.time() < deadline - 15:
        nb = len(bh)
        hf, cf = exact_cd(bh, nb, maxp=20000, deadline=deadline - 3, pen=0.5, ng=30, rng=rng)
        hf = project_feasible(hf, nb)
        cf = compute_c5(hf, nb)
        if cf < bc:
            bc, bh = cf, hf

    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh, len(bh))
    return bh, bc, len(bh)
