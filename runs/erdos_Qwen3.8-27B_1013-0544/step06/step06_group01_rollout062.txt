def project_feasible(h, n):

def lse_val(x, n, alpha, mu):

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha * m > 500:
        sh = alpha * (c - m)
        val = (m + np.log(np.sum(np.exp(sh))) / alpha) * dx
    else:
        val = np.log(np.sum(np.exp(alpha * c))) / alpha * dx
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

                        dc += -(1.0 - h[i+l])
                    if 0 <= i - l < n:
                        dc += h[i-l]
                    # j: was 0, now 1
                    if 0 <= j + l < n:
                        dc += (1.0 - h[j+l])
                    if 0 <= j - l < n:
                        dc += -h[j-l]
                    new_c[l + n - 1] = c[l + n - 1] + dc
                new_max = float(np.max(new_c))
                if new_max * dx < c5 - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = new_c
                    c5 = new_max * dx
                    improved = True
        if not improved:
            # Try a few random non-improving moves (simple annealing)
            if time.time() < deadline - 10:
                for _ in range(500):
                    if time.time() > deadline:
                        break
                    i = rng_int(0, n - 1)
                    j = rng_int(0, n - 1)
                    if i == j or h[i] < 0.5 or h[j] > 0.5:
                        continue
                    l_min = max(-(n-1), -i, i-(n-1), -j, j-(n-1))
                    l_max = min(n-1, n-1-i, i, n-1-j, j)
                    new_c = c.copy()
                    for l in range(l_min, l_max + 1):
                        if l == 0:
                            continue
                        dc = 0.0
                        if 0 <= i + l < n:
                            dc += -(1.0 - h[i+l])
                        if 0 <= i - l < n:
                            dc += h[i-l]
                        if 0 <= j + l < n:
                            dc += (1.0 - h[j+l])
                        if 0 <= j - l < n:
                            dc += -h[j-l]
                        new_c[l + n - 1] = c[l + n - 1] + dc
                    new_max = float(np.max(new_c))
                    delta = new_max * dx - c5
                    if delta < 0 or (np.random.random() < np.exp(-50 * delta)):
                        h[i] = 0.0
                        h[j] = 1.0
                        c = new_c
                        c5 = new_max * dx
                        improved = True
    return h, c5

def rng_int(lo, hi):
    return int(np.random.randint(lo, hi + 1))

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase 1: L-BFGS-B with continuation
    for alpha in [50, 200, 1000, 5000, 20000]:
        if time.time() > deadline - 60:
            break
        t_sub = min(20, deadline - time.time())
        if t_sub < 5:
            break
        h_new = lbfgsb_opt(h, n, alpha, 50.0, maxiter=300, deadline=time.time() + t_sub)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h, best_c = h_new, c_new
    # Phase 2: Coordinate descent
    if time.time() < deadline - 30:
        h_cd, c_cd = coord_descent(h, n, 300, 80, time.time() + min(80, deadline - time.time() - 10), rng, 5.0)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase 3: Discrete search
    if time.time() < deadline - 20:
        h_d, c_d = discrete_search(h, n, time.time() + min(60, deadline - time.time() - 5), window=50)
        if c_d < best_c:
            h, best_c = h_d, c_d
    # Phase 4: Refine discrete solution continuously
    if time.time() < deadline - 15:
        h_cont = h.astype(float)
        h_cont[h_cont > 0.5] = 0.7
        h_cont[h_cont <= 0.5] = 0.3
        h_cont = project_feasible(h_cont, n)
        h_new = lbfgsb_opt(h_cont, n, 10000, 100.0, maxiter=500, deadline=time.time() + 30)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h, best_c = h_new, c_new
        if time.time() < deadline - 10:
            h_cd, c_cd = coord_descent(h, n, 200, 100, time.time() + 30, rng, 1.0)
            h_cd = project_feasible(h_cd, n)
            c_cd = compute_c5(h_cd, n)
            if c_cd < best_c:
                h, best_c = h_cd, c_cd
    # Phase 5: Final discrete
    if time.time() < deadline - 5:
        h_d, c_d = discrete_search(h, n, deadline - 2, window=40)
        if c_d < best_c:
            h, best_c = h_d, c_d
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Optimize initial construction
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try different n values
    x_orig = np.linspace(0, 2, n_init)
    for n_try in [200, 250, 400, 500]:
        if time.time() > deadline - 200:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        alloc = min(150, (deadline - time.time()) * 0.25)
        if alloc < 50:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
            x_orig = x_new
    # Multi-restart from best with perturbation
    if time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restart = max(3, int(remaining / 70))
        for r in range(n_restart):
            if time.time() > deadline - 60:
                break
            scale = 0.03 + 0.05 * (r / max(1, n_restart - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(65, (deadline - time.time() - 30) * 0.4)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Also try structured starts
    if time.time() < deadline - 100:
        n = len(best_h)
        structured = []
        # Alternating pattern with noise
        alt = np.zeros(n)
        alt[::2] = 1.0
        alt = project_feasible(alt + rng.normal(0, 0.1, n), n)
        structured.append(alt)
        # Block pattern
        blk = np.zeros(n)
        blk[:n//4] = 1.0
        blk[3*n//4:] = 1.0
        blk = project_feasible(blk + rng.normal(0, 0.1, n), n)
        structured.append(blk)
        for hs in structured:
            if time.time() > deadline - 60:
                break
            alloc = min(60, deadline - time.time() - 30)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_full(hs, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 15:
        n = len(best_h)
        h_new = lbfgsb_opt(best_h, n, 50000, 200.0, maxiter=1000, deadline=time.time() + 30)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    if time.time() < deadline - 5:
        n = len(best_h)
        h_cd, c_cd = coord_descent(best_h, n, 200, 120, deadline - 3, rng, 0.5)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
