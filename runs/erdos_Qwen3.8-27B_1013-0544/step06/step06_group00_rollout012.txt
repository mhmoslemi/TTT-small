```python
            grad[j] += np.dot(w_arr, 1.0 - x_arr)
    
    return grad
```

Wait, this has an inner loop over j (n iterations) and each iteration does O(n) work, so total is O(n²). For n=400, that's 160,000 operations, which is fine.

But actually, I can vectorize the first sum using convolution:
First term for all j: -sum_s w[s+(n-1)] * x[j+s]
This is the "anti-convolution" or cross-correlation of w (re-indexed) and x.

If I define W_full = w (already in the right order, indexed by k=0..2n-2), then:
The first term at j is: -sum_{s: 0<=j+s<n} w[s+(n-1)] * x[j+s]

Let me define W2 as a length-n array where W2[t] = sum of contributions... no, this isn't a simple convolution because the valid range of s depends on j.

Actually, it IS a convolution if I zero-pad appropriately:
Define W_pad of length 2n-1: W_pad[s+(n-1)] = w[s] for s in [-(n-1), n-1].
Define X_pad of length 2n-1: X_pad[j] = x[j] for j in [0, n-1], 0 elsewhere.

Then the first term at position j is:
-sum_s W_pad[s+(n-1)] * X_pad[j+s]
= -sum_s W_pad[s+(n-1)] * X_pad[j+s]

This is the cross-correlation of W_pad and X_pad, evaluated at... let me think.

cross_corr[A, B][k] = sum_i A[i] * B[i+k]

So cross_corr[W_pad, X_pad][k] = sum_i W_pad[i] * X_pad[i+k]

I want: sum_s W_pad[s+(n-1)] * X_pad[j+s]
Let i = s+(n-1), so s = i-(n-1), and X_pad[j+s] = X_pad[j+i-(n-1)] = X_pad[i + (j-(n-1))]
So the sum is: sum_i W_pad[i] * X_pad[i + (j-(n-1))] = cross_corr[W_pad, X_pad][j-(n-1)]

So the first term at position j is: -cross_corr[W_pad, X_pad][j-(n-1)]

In numpy: np.correlate(W_pad, X_pad, mode='full')[j + (n-1)] would give this... wait, np.correlate(a, v)[k] = sum_i a[i+k] * v[i] (or is it the other way?).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_ls(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        if sm > 700:
            return m + (g - m).max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def barrier_obj_factory(n, T, alpha):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        shifted = alpha * (g - T)
        sm = shifted.max()
        if sm > 700:
            return sm / alpha + np.log(2 * n - 1) / alpha
        return (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return obj

def barrier_optimize(h, n, T, alpha, maxiter=2000):
    dx = 2.0 / n
    obj = barrier_obj_factory(n, T, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_search_target(h, n, T_lo, T_hi, alpha, max_iter_flag, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_iter_flag):
        if time.time() > deadline:
            break
        T_mid = 0.5 * (T_lo + T_hi)
        h_new = barrier_optimize(h, n, T_mid, alpha, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new < T_mid + 1e-12:
            T_hi = T_mid
            h = best_h.copy()
        else:
            T_lo = T_mid
    return best_h, best_c

def focused_slsqp(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for top_k in [3, 5, 8, 12]:
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        active = np.argsort(corr)[::-1][:top_k]
        for pen in [1.0, 10.0, 100.0]:
            if time.time() > deadline:
                break
            def obj(x):
                g = np.correlate(x, 1.0 - x, mode='full') * dx
                m = g.max()
                sh = 5000.0 * (g - m)
                sm = sh.max()
                base = m + (sm + np.log(np.exp(sh - sm).sum())) / 5000.0
                excess = np.maximum(g[active] - best_c + 1e-6, 0.0)
                return base + pen * np.sum(excess * excess)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 1500, 'ftol': 1e-18})
                if res.fun < obj(best_h):
                    nh = project_feasible(res.x, n)
                    nc = compute_c5(nh, n)
                    if nc < best_c:
                        best_c = nc
                        best_h = nh.copy()
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP log-sum-exp with progressive alpha
    p1 = t0 + 0.30 * total
    alphas = [30, 80, 200, 500, 1500, 5000, 15000, 50000, 150000, 500000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 8:
            break
        h = slsqp_ls(h, n, alpha, maxiter=2500, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Very high alpha refinement
    p2 = t0 + 0.45 * total
    alphas2 = [1000000, 5000000, 20000000]
    h = best_h.copy()
    for alpha in alphas2:
        if time.time() > p2 - 5:
            break
        h = slsqp_ls(h, n, alpha, maxiter=4000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Binary search barrier on target
    p3 = t0 + 0.65 * total
    T_lo = best_c - 0.005
    T_hi = best_c + 0.002
    best_h, best_c = binary_search_target(best_h, n, T_lo, T_hi,
                                          alpha=50000, max_iter_flag=20, deadline=p3)

    # Phase 4: Focused SLSQP on active constraints
    p4 = t0 + 0.78 * total
    best_h, best_c = focused_slsqp(best_h, n, p4)

    # Phase 5: Noise restarts
    p5 = t0 + 0.90 * total
    for trial in range(8):
        if time.time() > p5 - 8:
            break
        scale = 0.002 * (1.0 + 0.3 * trial)
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)

            if time.time() > p5 - 4:

            h = slsqp_ls(h, n, alpha, maxiter=1500, ftol=1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 6: Try other resolutions
    if time.time() < dl - 25:
        for nh in [300, 500]:
            if time.time() > dl - 15:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = np.interp(xn, xo, best_h)
            hh = project_feasible(hh, nh)
            for alpha in [500, 5000, 50000, 500000]:
                if time.time() > dl - 8:
                    break
                hh = slsqp_ls(hh, nh, alpha, maxiter=2000, ftol=1e-16)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c = ch
                    best_h = hh.copy()
                    n = nh

    # Final polish
    if time.time() < dl - 3:
        h = best_h + 0.001 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [10000000, 50000000]:
            if time.time() > dl - 1:
                break
            h = slsqp_ls(h, n, alpha, maxiter=2000, ftol=1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
