import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    diff = alpha * (c - m)
    mx = diff.max()
    w = np.exp(diff - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, max_iters=200):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50,1.0),(200,5.0),(1000,10.0),(5000,50.0),
               (20000,100.0),(100000,500.0),(500000,2000.0),(2000000,5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters*15,
                                   'maxls': 40, 'maxcor': 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    diff = alpha * (c - m)
    mx = diff.max()
    w = np.exp(diff - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, max_iters=250):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [20, 60, 200, 700, 2500, 10000, 40000, 150000, 500000, 1500000]
    for alpha in alphas:
        if time.time() > deadline - 1:
            break
        mu = max(0.1, alpha * 1e-6)
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 20,
                                   'maxls': 50, 'maxcor': 25})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_slsqp(h, n, deadline, alpha=100000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair_search(h, n, deadline, top_k=35):
    best_c = compute_c5(h, n)
    for alpha in [100, 500, 2000, 10000, 50000]:
        while time.time() < deadline - 1:
            g = grad_smooth_max(h, n, alpha)
            pos = np.argsort(g)[-top_k:]
            neg = np.argsort(g)[:top_k]
            improved = False
            for i in pos:
                if time.time() > deadline - 1:
                    return h, best_c
                for j in neg:
                    if i == j:
                        continue
                    for d in [0.004, 0.002, 0.001, 0.0005, 0.0002]:
                        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                            h[i] += d; h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True; break
                            h[i] -= d; h[j] += d
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            h[i] -= d; h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True; break
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def local_pair_search(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def random_perturb_optimize(h, n, deadline, rng, n_perturbs=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(n_perturbs):
        if time.time() > deadline - 30:
            break
        h_pert = h.copy()
        mask = rng.random(n) < 0.3
        noise = rng.randn(n) * 0.05 * mask
        h_pert = project_feasible(h_pert + noise, n)
        h_opt, c_opt = run_lbfgs(h_pert, n, min(time.time() + 60, deadline - 20), max_iters=100)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c
