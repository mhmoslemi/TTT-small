    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_c = cur_max * dx

        h[i], h[j] = 1.0, 0.0

            # Undo swap
            h[i], h[j] = 0.0, 1.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0

def binary_local_search(h, n, deadline, seed=42, two_swap_prob=0.1):

    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() < deadline and no_improve < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        else:

def slsqp_optimize(h, n, deadline, alphas=None):
    """SLSQP with log-sum-exp smoothing."""
    if alphas is None:
        alphas = [1000, 5000, 20000, 100000, 500000]
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            # Smooth max approximation
            shifted = alpha * (g - m)
            logsum = m + np.log(np.exp(shifted).sum()) / alpha
            return logsum
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        
        try:

                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or res.fun < best_c:

                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
        except Exception:
            pass
    
    return best_h, best_c

        alphas = [1000, 5000, 50000, 200000]
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            # For numerical stability
            logsum = m + np.log(np.exp(shifted).sum()) / alpha
            # Store for gradient
            obj.stored_g = g
            obj.stored_max = m
            return logsum
        
        def grad(x):
            g = obj.stored_g
            m = obj.stored_max
            shifted = alpha * (g - m)
            exp_shifted = np.exp(shifted)
            weights = exp_shifted / exp_shifted.sum()
            # Compute gradient using correlate
            w_rev = weights[::-1]
            first = np.correlate(1-x, weights, 'full')[n-1:2*n-1]
            second = np.correlate(x, w_rev, 'full')[2*n-2:3*n-2]
            return dx * (first - second)
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        def cons_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, best_h, jac=grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
            h_try = project_feasible(res.x, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        except Exception:
            pass
    
    return best_h, best_c

def compute_gradient(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process significant weights
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n-1)  # the actual lag
        # Contribution: dx * (1-h[j+lag]) - dx * h[j-lag] for valid j
        if lag >= 0:
            # (1-h[j+lag]) is valid for j = 0 to n-1-lag
            grad[:n-lag] += weights[k] * dx * (1-h[lag:])
            # h[j-lag] is valid for j = lag to n-1
            grad[lag:] += weights[k] * (-dx) * h[:-lag]
        else:
            lag2 = -lag
            # (1-h[j+lag]) = (1-h[j-lag2]) is valid for j = lag2 to n-1
            grad[lag2:] += weights[k] * dx * (1-h[:-lag2])
            # h[j-lag] = h[j+lag2] is valid for j = 0 to n-1-lag2
            grad[:n-lag2] += weights[k] * (-dx) * h[lag2:]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def gradient_c5_smooth(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w[k] * dx * (1.0 - h[lag:])
            grad[lag:] += w[k] * (-dx) * h[:n - lag]
        else:
            lag2 = -lag
            grad[lag2:] += w[k] * dx * (1.0 - h[:n - lag2])
            grad[:n - lag2] += w[k] * (-dx) * h[lag2:]
    return grad, m + np.log(exp_s.sum()) / alpha

        alphas = [500, 2000, 10000, 50000, 200000]

        if time.time() > deadline - 3:
            break

            _, val = gradient_c5_smooth(x, n, alpha)
            return val
        def grad(x):
            grad_v, _ = gradient_c5_smooth(x, n, alpha)
            return grad_v
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:

            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_ls(h, n, deadline, seed=42, two_prob=0.3):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_c = cur_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    no_imp = 0
    max_no_imp = 10000
    while time.time() < deadline and no_imp < max_no_imp:

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < two_prob and len(ones) >= 2 and len(zeros) >= 2:

            o1, o2 = rng.choice(ones, 2, replace=False)

                corr = np.correlate(h, 1 - h, mode='full')

                    best_h = h.copy()
                no_imp = 0

def sa_search(h, n, deadline, seed=42, t0=0.5):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_c = cur_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    n_iter = max(100, int((deadline - time.time()) * 8000))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = t0 * max(0.01, 1.0 - it / n_iter)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        z = int(rng.choice(zeros))
        o = int(rng.choice(ones))
        h[z], h[o] = 1.0, 0.0
        nc = float(np.correlate(h, 1 - h, mode='full').max())
        delta = (nc - cur_max) * dx
        accept = delta < 0 or rng.random() < np.exp(-delta / max(T * dx, 1e-10))
        if accept:
            cur_max = nc
            corr = np.correlate(h, 1 - h, mode='full')
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        else:
            h[z], h[o] = 0.0, 1.0
    return best_h, best_c
