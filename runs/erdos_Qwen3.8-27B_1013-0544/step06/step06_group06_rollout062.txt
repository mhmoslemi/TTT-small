def project_feasible(h, n):

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    h = np.clip(h, 0, 1)
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        np.random.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        np.random.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

```python
def compute_G(h, n):
    s_all = np.arange(2 * n - 1) - (n - 1)  # (2n-1,)
    # For each p, v1 = (p - s_all >= 0) & (p - s_all < n)
    # p - s_all = p - (arange(2n-1) - (n-1)) = p + n - 1 - arange(2n-1)
    # This is valid when 0 <= p+n-1-arange < n, i.e., p-n+1 <= arange <= p-1... 
    # Actually: 0 <= p - s < n where s = k - n + 1
    # p - s = p - k + n - 1. Valid: 0 <= p - k + n - 1 < n => 1-n <= p-k < 0-n+1... 
    # Let me just do it with broadcasting.
    
    # p ranges over [0, n), s_all ranges over [-(n-1), n-1]
    # v1: 0 <= p - s < n for each (p, s) pair
    # v2: 0 <= p + s < n for each (p, s) pair
    
    p_col = np.arange(n)[:, None]  # (n, 1)
    s_row = s_all[None, :]  # (1, 2n-1)
    
    v1 = (p_col - s_row >= 0) & (p_col - s_row < n)  # (n, 2n-1)
    v2 = (p_col + s_row >= 0) & (p_col + s_row < n)  # (n, 2n-1)
    
    # raw[p, k] = (1-h[p-s_k]) * v1[p,k] + (-h[p+s_k]) * v2[p,k]
    # p - s_k = p - (k - n + 1) = p + n - 1 - k
    # p + s_k = p + k - n + 1
    
    idx1 = p_col - s_row  # (n, 2n-1), the index into h for first term
    idx2 = p_col + s_row  # (n, 2n-1), the index into h for second term
    
    # Clip for safety (will be masked by v1, v2)
    idx1_clipped = np.clip(idx1, 0, n - 1)
    idx2_clipped = np.clip(idx2, 0, n - 1)
    
    term1 = (1 - h[idx1_clipped]) * v1  # (n, 2n-1)
    term2 = (-h[idx2_clipped]) * v2  # (n, 2n-1)
    
    raw = term1 + term2
    raw[:, n - 1] = 0  # self-term
    
    signs = (1 - 2 * h)[:, None]  # (n, 1)
    G = signs * raw
    return G
```

Let me verify the indices:
- s_all = arange(2n-1) - (n-1) = [-n+1, -n+2, ..., 0, ..., n-2, n-1]
- For position p and lag k (output index), s = k - (n-1)
- First term: index into h is p - s = p - (k-n+1) = p + n - 1 - k
  - Valid when 0 <= p + n - 1 - k < n, i.e., 1-n <= p-k < 0... 
  - Wait: 0 <= p-s < n where s = k-(n-1)
  - 0 <= p - (k-n+1) < n
  - 0 <= p+n-1-k < n
  - 1-n <= p-k < 1
  - p-n+1 <= k <= p
  
  So k ranges from max(0, p-n+1) to p. Since p < n, p-n+1 <= 0, so k ranges from 0 to p.
  And the index into h is p - (k-n+1) = p+n-1-k.

- Second term: index into h is p + s = p + k - n + 1
  - Valid when 0 <= p + k - n + 1 < n
  - n-1-p <= k < 2n-1-p
  - k ranges from max(0, n-1-p) to min(2n-2, 2n-2-p) = 2n-2-p
  - The index into h is p + k - n + 1.
