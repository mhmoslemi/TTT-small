  - For each pair (i,j) of ones, compute g[i] + g[j]
  - We want to minimize max(corr - g[i] - g[j] + g[k] + g[l])
  
- This is still O(n^4) in the worst case, but we can use the fact that we only care about the top_k lags:
  - For each (i,j) pair of ones and (k,l) pair of zeros:
    - new_max ≈ max_{m in top_k} (corr[m] - g[i,m] - g[j,m] + g[k,m] + g[l,m])
  
- For efficiency, we can precompute:
  - For ones: S1[i,j,m] = g[i,m] + g[j,m] for i < j (O(n^2 * top_k))
  - For zeros: S0[k,l,m] = g[k,m] + g[l,m] for k < l (O(n^2 * top_k))
  
- Then for each (i,j) and (k,l): new_max(i,j,k,l) = max_m (corr[m] - S1[i,j,m] + S0[k,l,m])
  
- This is O(n^4 * top_k) which is too slow for n=250.

Alternative: Greedy pair swaps
- For each zero k, find the best one i to swap with (single swap)
- Then for the resulting state, find the best zero l and one j to do a second swap
- Accept if the combined improvement is positive

This is O(n^3 * top_k) per iteration which is still a lot.

- Pick two random ones and two random zeros
- Check if the swap improves the objective

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ks = max(0, p)
        ke = min(2 * n - 2, p + n - 1)
        for k in range(ks, ke + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def apply_swap_state(h, corr, g, i, j, n, zeros, ones, ii, jj):
    p_arr = np.arange(n)
    mi = (p_arr + i < 2 * n - 1)
    g[p_arr[mi], p_arr[mi] + i] -= 2.0
    if 2 * i < 2 * n - 1:
        g[i, 2 * i] = 0.0
    mj = (p_arr + j < 2 * n - 1)
    g[p_arr[mj], p_arr[mj] + j] += 2.0
    if 2 * j < 2 * n - 1:
        g[j, 2 * j] = 0.0
    h[i] = 1.0
    h[j] = 0.0
    zeros = np.delete(zeros, ii)
    ones = np.delete(ones, jj)
    zeros = np.append(zeros, j)
    ones = np.append(ones, i)
    corr = corr + g[i] - g[j]
    return h, corr, g, zeros, ones

def hill_climb(h_start, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_start)[::-1]

    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    temp = 0.005
    no_improve = 0
    top_k = 8
    
    while time.time() < deadline:
        if no_improve > 400:
            top_k = min(50, 8 + no_improve // 60)
        elif no_improve < 50:
            top_k = 8
        
        top_idx = np.argsort(corr)[-top_k:]
        g_act = g[:, top_idx]
        D = corr[top_idx][None, None, :] + g_act[zeros][:, None, :] - g_act[ones][None, :, :]
        new_maxes = D.max(axis=2)
        flat = new_maxes.argmin()
        ii = flat // len(ones)
        jj = flat % len(ones)
        i = zeros[ii]
        j = ones[jj]
        new_corr = corr + g[i] - g[j]
        actual_new_c = new_corr.max() * dx
        
        if actual_new_c < cur_c - 1e-14:
            h, corr, g, zeros, ones = apply_swap_state(h, corr, g, i, j, n, zeros, ones, ii, jj)
            cur_c = actual_new_c
            stall = 0
            temp = max(0.001, temp * 0.999)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

            no_improve += 1

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh = np.clip(sh, -500, 500)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
