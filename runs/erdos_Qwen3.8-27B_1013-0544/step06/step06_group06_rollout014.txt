new_max ≈ max_{k in L} (corr[k] + F[i,k] + F[j,k])

For fixed i, we want to find j that minimizes max_{k in L} (corr[k] + F[i,k] + F[j,k])
= min_j max_{k in L} ((corr[k] + F[i,k]) + F[j,k])

Let A_k = corr[k] + F[i,k] for k in L.
Then we want min_j max_{k in L} (A_k + F[j,k]).

This is still not trivially fast, but for small K it's much faster.

                        zeros = np.append(zeros, j)
                        ones = ones[ones != j]
                        ones = np.append(ones, i)
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur = corr.max()
                stall = 0

    return best_h, best * dx

def subgradient_refine(h_start, n, deadline, max_iter=20000):
    dx = 2.0 / n
    h = h_start.copy()
    best = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.005
    top_m = 5

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 200 == 0:
            lr = max(0.0005, 0.005 * (0.995 ** it))

        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = c.max()
        # Get active lags
        thresh = cur_max - 0.002
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(c)[-top_m:]
        active = active[:top_m]

        # Compute average gradient over active lags
        # grad c_k / dh[p] = (1-h[p+k]) - h[p-k] where k = q - (n-1)
        grad = np.zeros(n)
        for q in active:
            k = q - (n - 1)
            for p in range(n):
                pk = p + k
                pmk = p - k
                term1 = (1 - h[pk]) if 0 <= pk < n else 1.0
                term2 = h[pmk] if 0 <= pmk < n else 0.0
                grad[p] += (-term1 + term2)  # negative because we want to decrease c_k

        # Project gradient to maintain sum constraint
        grad_mean = grad.mean()
        grad_centered = grad - grad_mean

        # Update
        h_new = h + lr * grad_centered
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < cur_max:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best = h.copy()

    return best, best_c

def slsqp_smooth(h, n, deadline, alphas):
    dx = 2.0 / n
    best = h.copy()
    best_c = compute_c5(h, n)

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            s = alpha * (c - m)
            s = s - s.max()
            return m + np.log(np.exp(s).sum()) / alpha

        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
            h2 = np.clip(res.x, 0, 1)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best = h2.copy()
                h = h2.copy()
        except Exception:
            pass
    return best, best_c

def thue_morse_binary(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    cnt = int(t.sum())
    if cnt > target:
        idx = np.where(t == 1)[0]
        np.random.shuffle(idx)
        t[idx[:cnt - target]] = 0
    elif cnt < target:
        idx = np.where(t == 0)[0]
        np.random.shuffle(idx)
        t[idx[:target - cnt]] = 1
    return t.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary greedy from initial (55% time)
    p1 = t0 + 0.55 * (budget_s - 8)
    h_bin = to_binary(h0, n)
    c_b = compute_c5(h_bin, n)
    if c_b < best_c:
        best_c = c_b
        best_h = h_bin.copy()
    hb, cb = binary_greedy(h_bin, n, p1, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Subgradient continuous refinement (15% time)
    p2 = t0 + 0.70 * (budget_s - 8)
    if time.time() < p2 - 5:
        h_cont = best_h + 0.03 * rng.randn(n)
        h_cont = project_feasible(h_cont, n)
        hr, cr = subgradient_refine(h_cont, n, p2 - 3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()

    # Phase 3: SLSQP refinement (10% time)
    p3 = t0 + 0.80 * (budget_s - 8)
    if time.time() < p3 - 5:
        h_s = best_h + 0.02 * rng.randn(n)
        h_s = project_feasible(h_s, n)
        hs, cs = slsqp_smooth(h_s, n, p3 - 3, [50000, 200000, 800000])
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()

    # Phase 4: Binary search from refined (10% time)
    p4 = t0 + 0.90 * (budget_s - 8)
    if time.time() < p4 - 5:
        h_bin2 = to_binary(best_h, n)
        c_b2 = compute_c5(h_bin2, n)
        if c_b2 < best_c:
            best_c = c_b2
            best_h = h_bin2.copy()
        hb2, cb2 = binary_greedy(h_bin2, n, p4 - 3, seed=seed + 77)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 5: Final SLSQP polish (5% time)
    if time.time() < dl - 5:
        h_f = best_h + 0.015 * rng.randn(n)
        h_f = project_feasible(h_f, n)
        hf, cf = slsqp_smooth(h_f, n, dl - 3, [200000, 1000000])
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Final SLSQP
    if time.time() < dl - 2:
        hf2, cf2 = slsqp_smooth(best_h, n, dl - 1, [500000])
        if cf2 < best_c:
            best_c = cf2
            best_h = hf2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
