def compute_gradient(h, alpha, n):
    """Compute gradient of log-sum-exp approximation of C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    shifted = alpha * (corr - corr.max())
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    # Gradient
    # dL/dh[j] = dx * sum_k p[k] * (1-h[j-k])*I[0<=j-k<n] - dx * sum_k p[k] * h[j+k]*I[0<=j+k<n]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-15:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 1e-15:
                h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr_max = corr.max()
    shifted = alpha * (corr - corr_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    A = np.convolve(1.0 - h, p, mode='full')[:n]
    B = np.correlate(h, p, mode='full')[:n]
    return (A - B) * dx

def pgd_step(h, n, alpha, max_iters, step_scale, deadline, beta_mom=0.85):
    h = project_feasible(h, n)
    cur_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    eta = step_scale / max(alpha, 1.0)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = compute_gradient(h, alpha, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        v = beta_mom * v + (1.0 - beta_mom) * grad
        h_new = project_feasible(h - eta * v, n)
        if it % 30 == 0:
            c5 = compute_c5(h_new, n)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            if stall > 4:
                eta *= 0.6
                v *= 0.5
                stall = 0
                if eta < 1e-10:
                    break
        h = h_new
    return best_h, best_c5

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx_i in range(len(zeros)):
            i = zeros[idx_i]
            best_j = -1
            best_nc = cur_c
            best_delta = None
            for j in ones:
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
