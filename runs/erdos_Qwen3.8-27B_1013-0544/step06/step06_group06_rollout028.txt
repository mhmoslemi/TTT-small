    for _ in range(100):

        if abs(diff) < 1e-12:
            break
        if diff > 0:

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

def project_feasible(x, target_sum):
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-14:
            break
    return np.clip(x - (lo+hi)/2, 0, 1)
