        m_hat = b1 * m_hat + (1 - b1) * grad
        v_hat = b2 * v_hat + (1 - b2) * grad * grad
        m_hat_adj = m_hat / (1 - b1 ** t)
        v_hat_adj = v_hat / (1 - b2 ** t)
        step = lr * m_hat_adj / (np.sqrt(v_hat_adj) + eps)
        h_new = project_feasible(h - step, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() - 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
