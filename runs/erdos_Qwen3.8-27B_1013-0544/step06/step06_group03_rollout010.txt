import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, n, max_iter=300):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += diff / room.sum() * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        s = s - s.max()
        return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_max_grad_full(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        s = s - s.max()
    w = np.exp(s)
    wn = w.sum()
    w = w / wn
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - x[lag:])
            grad[lag:] -= wk * x[:n - lag]
        else:
            m2 = -lag
            grad[m2:] += wk * (1.0 - x[:n - m2])
            grad[:n - m2] -= wk * x[m2:]
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    f = _smooth_max_val(c, alpha)
    s = x.sum() - n / 2.0
    return f + 0.5 * mu * s * s

def _penalty_grad(x, n, alpha, mu):
    g = _smooth_max_grad_full(x, n, alpha)
    s = x.sum() - n / 2.0
    g += mu * s
    return g

def optimize_penalty(h, n, alpha, mu, deadline, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h.copy(), args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds, jac=_penalty_grad,
                       options={'maxiter': maxiter, 'gtol': 1e-14,
                                'ftol': 1e-16, 'maxls': 50})
        return project_box_sum(res.x, n)
    except Exception:
        return h.copy()

def proj_grad_momentum(h, n, alpha, deadline, momentum=0.9,
                       max_iter=5000, step_init=0.02):
    h_curr = h.copy().astype(float)
    v = np.zeros(n)
    step = step_init
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _smooth_max_grad_full(h_curr, n, alpha)
        s = h_curr.sum() - n / 2.0
        g += g * 0 + s  # add sum constraint gradient (simple)
        # Actually the gradient of the indicator of sum constraint is just ones * s
        v = momentum * v - step * g
        h_new = h_curr + v
        h_new = project_box_sum(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Backtracking
        if c_new > compute_c5(h_curr, n) * 1.01 + 1e-12:
            step *= 0.7
        elif (it + 1) % 200 == 0:
            step = min(step * 1.1, 0.05)
        h_curr = h_new
    return best_h, best_c

def local_search_active(h, n, deadline, top_m=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(50):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        # For each active lag, try to reduce it
        improved = False
        for ki in top_idx:
            if time.time() > deadline:
                break
            lag = ki - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            # Try small perturbations in the direction that reduces this lag
            # Direction: for lag>0, increase h[i] where h[i+m] is low, decrease where high
            if lag > 0:
                direction = np.zeros(n)
                direction[:n - m] = h[lag:] - 0.5
                direction[lag:] = 0.5 - h[:n - m]
            else:
                m2 = -lag
                direction = np.zeros(n)
                direction[m2:] = h[:n - m2] - 0.5
                direction[:n - m2] = 0.5 - h[m2:]
            # Normalize
            dn = np.linalg.norm(direction)
            if dn < 1e-12:
                continue
            direction /= dn
            for scale in [0.02, 0.01, 0.005, 0.002, 0.001]:
                h_try = project_box_sum(h + scale * direction, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_passes=10):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        for i in range(n):
            if time.time() > deadline:
                break
            # Try to optimize h[i]
            old = h[i]
            # Grid search in a small neighborhood
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                val = old + delta
                if val < 0 or val > 1:
                    continue
                h[i] = val
                c_try = compute_c5(h, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h.copy()
                else:
                    h[i] = val - delta  # revert
                    break
            h[i] = old
    return best_h, best_c
