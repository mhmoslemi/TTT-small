def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, 0, 1)
    for _ in range(50):
        s = x.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            # Need to reduce: subtract t from elements that are > 0
            # x_new = clip(x - t, 0, 1)
            # Want sum(x_new) = target
            # Binary search on t in [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(60):
                mid = (lo + hi) / 2
                new_s = np.clip(x - mid, 0, 1).sum()
                if new_s > target:
                    lo = mid
                else:
                    hi = mid
            x = np.clip(x - (lo + hi) / 2, 0, 1)
        else:
            # Need to increase: add t to elements that are < 1
            lo, hi = 0.0, 1.0
            for _ in range(60):
                mid = (lo + hi) / 2
                new_s = np.clip(x + mid, 0, 1).sum()
                if new_s < target:
                    lo = mid
                else:
                    hi = mid
            x = np.clip(x + (lo + hi) / 2, 0, 1)
    return x

    W = np.zeros(n)
    for j in range(n):

        m_hi = min(n-1, j)
        W[j] = cum_w[m_hi + n - 1] - cum_w[m_lo + n - 1]

    return S, grad, c

def pgd_solve(h, n, deadline, alpha_list, lr=0.1):

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

        cur_lr = lr

                break

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_set = set(int(x) for x in np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros_set):
            if time.time() > deadline:
                break
            for j in sorted(ones_set):
                # Compute delta for swap i->1, j->0
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_set.remove(j)
                    zeros_set.remove(i)
                    ones_set.add(i)
                    zeros_set.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
