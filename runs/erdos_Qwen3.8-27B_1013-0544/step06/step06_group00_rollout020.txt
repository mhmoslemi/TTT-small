    dx = 2.0 / n
    k = k_star  # offset from center (n-1)
    # c[k] = sum_i h[i+k] * (1-h[i]) for valid i
    # dc[k]/dh[j] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]

    lo = max(0, k)

    if lo < hi:

    # Second term: -h[j+k] for j where 0 <= j+k < n, i.e., -k <= j < n-k
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if lo2 < hi2:

    return dx * grad

    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def corr_and_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c, float(c.max())

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_polish(h, n, deadline, n_iters=20000, top_k=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    G = np.zeros(n)
    t = 0
    lr_base = 0.05
    prev_grad = None
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        t += 1
        c, cur_c = corr_and_max(h, n)
        
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute average subgradient
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            # dc/dh[j] = (1-h[j-k])*[valid1] - h[j+k]*[valid2]
            if k >= 0:
                # (1-h[j-k]) for j in [k, n)
                grad[k:] += (1.0 - h[:n-k])
                # -h[j+k] for j in [0, n-k)
                grad[:n-k] -= h[k:]
            else:
                kk = -k
                # (1-h[j+k]) for j in [0, n-kk)
                grad[:n-kk] += (1.0 - h[kk:])
                # -h[j-kk] for j in [kk, n)
                grad[kk:] -= h[:n-kk]
        grad = dx * grad / len(top_idx)
        
        if prev_grad is not None:
            G += prev_grad ** 2
            lr = lr_base / (np.sqrt(G) + 1e-8)
            h_new = project_feasible(h - lr * prev_grad, n)
            c_new, cur_c_new = corr_and_max(h_new, n)
            if cur_c_new < cur_c + 0.01:  # accept if not too much worse
                h = h_new
                if cur_c_new < best_c:
                    best_c = cur_c_new
                    best_h = h.copy()
        prev_grad = grad.copy()
    
    return best_h, best_c

def binary_swap(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                # Compute change in max
                # Quick estimate: check if swapping i,j reduces the current max
                # Full computation:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                i_idx = zeros_list.index(i)
                j_idx = ones_list.index(best_j)
                # Recompute delta for the actual swap
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(best_j-i)] += 1.0
                delta[n-1+(i-best_j)] += 1.0
                h_b[i], h_b[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list[j_idx] = i
                zeros_list[i_idx] = best_j
                improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_subgrad(c_idx, h, n):
    k = c_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    else:
        kk = -k
        if kk == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n - kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n - kk]
    return grad

def subgrad_optimize(h, n, deadline, max_iters=50000, top_k=3, lr_base=0.08):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    stall_limit = 5000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += compute_subgrad(idx, h, n)
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.005:
            h = h_new
    return best_h, best_c

                    best_j = j
            if best_j < 0:

            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)

            imn = i - m_range

            delta[msk] += h_b[jmn[msk]]
            delta[n - 1] -= 2.0
            delta[n - 1 + (best_j - i)] += 1.0
            delta[n - 1 + (i - best_j)] += 1.0
            h_b[i], h_b[best_j] = 1.0, 0.0
            corr += delta
            cur_c = best_nc
            ones_list.remove(best_j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(best_j)
            improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c
