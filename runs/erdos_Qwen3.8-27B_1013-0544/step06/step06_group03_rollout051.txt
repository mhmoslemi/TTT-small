        alloc = min(250, (deadline - time.time()) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try other resolutions by interpolating
    n_candidates = []
    for nc in [n_init + 100, n_init + 200, n_init + 50]:
        if nc > 0 and nc != n_init:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        alloc = min(remaining * 0.4, 120)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Diversification - random restarts from best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        n_restarts = min(5, max(1, int((deadline - time.time() - 30) / 60)))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final polish on best solution
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_refine(best_h, n, alpha=1000000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_refine(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
