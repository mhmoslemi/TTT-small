def steepest_greedy_fast(h, n, deadline, max_sweeps=50):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    L = 2*n - 1  # length of correlation
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no = len(ones)
        
        # Pre-compute add deltas: shape (nz, L)
        add_deltas = np.zeros((nz, L))
        for ii, i in enumerate(zeros):
            im = i + m_range
            msk = (im >= 0) & (im < n)
            add_deltas[ii, msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            add_deltas[ii, msk] -= h[imn[msk]]
        
        # Pre-compute remove deltas: shape (no, L)
        rem_deltas = np.zeros((no, L))
        for jj, j in enumerate(ones):
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            rem_deltas[jj, msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            rem_deltas[jj, msk] += h[jmn[msk]]
        
        # Find best swap
        best_delta_c = 0.0
        best_ii, best_jj = -1, -1
        
        for ii in range(nz):
            # add_deltas[ii] + rem_deltas[jj] + corrections for all jj
            # The corrections are: delta[n-1] -= 2, delta[n-1+(j-i)] += 1, delta[n-1+(i-j)] += 1
            # For fixed ii (i=zeros[ii]), the corrections depend on jj through j=ones[jj]
            base = corr + add_deltas[ii]  # shape (L,)
            for jj in range(no):
                i = zeros[ii]
                j = ones[jj]
                new_corr = base + rem_deltas[jj]
                new_corr[n-1] -= 2.0
                k1 = n-1 + (j-i)
                new_corr[k1] += 1.0
                k2 = n-1 + (i-j)
                new_corr[k2] += 1.0

                dc = nc - corr.max()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def steepest_greedy(h, n, deadline, max_sweeps=200):

    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    L = 2 * n - 1
    best_c = cur_max * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        
        # Pre-compute deltas
        add_deltas = np.zeros((nz, L))
        for ii in range(nz):
            i = zeros[ii]
            im = i + m_range
            msk = (im >= 0) & (im < n)
            add_deltas[ii, msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            add_deltas[ii, msk] -= h[imn[msk]]
        
        rem_deltas = np.zeros((no, L))
        for jj in range(no):
            j = ones[jj]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            rem_deltas[jj, msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            rem_deltas[jj, msk] += h[jmn[msk]]
        
        best_dc = 0.0
        best_ii, best_jj = -1, -1
        rows = np.arange(no)
        
        for ii in range(nz):
            base = corr + add_deltas[ii]
            candidate = base[None, :] + rem_deltas
            candidate[:, n-1] -= 2.0
            k1s = n - 1 + (ones - zeros[ii])
            k2s = n - 1 + (zeros[ii] - ones)
            # Handle bounds
            msk1 = (k1s >= 0) & (k1s < L)
            msk2 = (k2s >= 0) & (k2s < L)
            candidate[rows[msk1], k1s[msk1]] += 1.0
            candidate[rows[msk2], k2s[msk2]] += 1.0
            maxes = candidate.max(axis=1)
            jj_min = maxes.argmin()
            dc = maxes[jj_min] - cur_max
            if dc < best_dc - 1e-10:
                best_dc = dc
                best_ii, best_jj = ii, jj_min
        
        if best_ii == -1:
            break
        
        h[zeros[best_ii]] = 1.0
        h[ones[best_jj]] = 0.0
        corr = np.correlate(h, 1-h, mode='full').astype(float)
        cur_max = corr.max()
        c = cur_max * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
