import numpy as np
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * (2.0 / float(n)))


def project_feasible(v, n):
    v = np.asarray(v, dtype=float)
    if v.size != n:
        v = np.resize(v, n)
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if n == 0:
        return v

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v - lam, 0.0, 1.0)

    for _ in range(8):
        diff = target - float(x.sum())
        if abs(diff) < 1e-13:
            break
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            x[mask] += diff / cnt
        else:
            x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


def grad_weighted(h, n, w):
    w = np.asarray(w, dtype=float)
    h = np.asarray(h, dtype=float)
    if w.sum() == 0.0:
        return np.zeros(n, dtype=float)
    b = 1.0 - h
    left = np.convolve(w[::-1], b, mode="full")
    right = np.convolve(w, h, mode="full")
    start = n - 1
    return left[start:start + n] - right[start:start + n]


def grad_lag(h, n, m):
    g = np.zeros(n, dtype=float)
    s = int(m) - (n - 1)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
    else:
        t = -s
        if t == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[t:] += 1.0 - h[:n - t]
            g[:n - t] -= h[t:]
    return g


def active_cycle(h, n):
    h = np.asarray(h, dtype=float)
    c = compute_corr(h)
    cmax = float(np.max(c))
    if cmax <= 1e-12:
        return h.copy(), cmax

    K = min(60, c.size)
    idx = np.argsort(c)[-K:]
    K = int(idx.size)

    G = np.empty((K, n), dtype=float)
    for i, m in enumerate(idx):
        G[i] = grad_lag(h, n, m)

    cidx = c[idx]
    GGT = G @ G.T
    I = np.eye(K)

    fracs = (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05)
    R = len(fracs)
    B = np.empty((K, R), dtype=float)
    for j, f in enumerate(fracs):
        T = cmax - cmax * f
        b = T - cidx
        B[:, j] = np.where(b < 0.0, b, 0.0)

    scale_mu = max(1.0, n / 300.0)
    mus = (0.1 * scale_mu, 1.0 * scale_mu, 10.0 * scale_mu)
    factors = (1.0, 0.5, 0.25, 0.1, 0.05)

    best_h = h.copy()
    best_c = cmax

    for mu in mus:
        M = GGT + (mu + 1e-12) * I
        try:
            Y = np.linalg.solve(M, B)
        except Exception:
            Y = np.linalg.pinv(M) @ B

        for j in range(R):
            v = G.T @ Y[:, j]
            v -= v.mean()
            vmax = float(np.max(np.abs(v)))
            if vmax < 1e-12:
                continue
            for f in factors:
                vs = v * f
                if vmax * f > 0.25:
                    vs *= 0.25 / (vmax * f)
                h2 = project_feasible(h + vs, n)
                c2 = compute_c5(h2, n)
                if c2 < best_c - 1e-14:
                    best_c = c2
                    best_h = h2.copy()
    return best_h, best_c


def project_gradient(h, n, deadline, beta=1.0, max_steps=50):
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    stall = 0

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        cur = float(np.max(c))
        if cur < best_c - 1e-14:
            best_c = cur
            best_h = h.copy()

        p = np.exp((c - cur) / max(beta, 1e-8))
        p /= p.sum()
        g = grad_weighted(h, n, p)
        g -= g.mean()
        sn = float(np.linalg.norm(g))
        if sn < 1e-12:
            p = np.ones(c.size, dtype=float) / c.size
            g = grad_weighted(h, n, p)
            g -= g.mean()
            sn = float(np.linalg.norm(g))
            if sn < 1e-12:
                break

        local_h = h
        local_c = cur
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            v = -mult * step * g / sn
            h2 = project_feasible(h + v, n)
            c2 = compute_c5(h2, n)
            if c2 < local_c - 1e-14:
                local_c = c2
                local_h = h2

        if local_c < cur - 1e-14:
            h = local_h
            if local_c < best_c - 1e-14:
                best_c = local_c
                best_h = h.copy()
            stall = 0
            step = min(0.12, step * 1.08)
        else:
            stall += 1
            step *= 0.8
            if step < 1e-5 or stall > 40:
                break
    return best_h, best_c


def make_smooth_noise(n, rng, amp, passes):
    z = rng.normal(0.0, 1.0, n)
    ker = np.ones(3, dtype=float) / 3.0
    for _ in range(max(1, int(passes))):
        z = np.convolve(z, ker, mode="same")
    z = z - z.mean()
    m = float(np.max(np.abs(z)))
    if m > 1e-14:
        z = z / m * amp
    return z


def polish(h, n, deadline, rng=None):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    last_update = time.time()

    while time.time() < deadline - 1.0:
        c_now = compute_c5(h, n)
        if c_now < best_c - 1e-14:
            best_c = c_now
            best_h = h.copy()

        h2, c2 = active_cycle(h, n)
        if c2 < best_c - 1e-14:
            h = h2.copy()
            best_h = h.copy()
            best_c = c2
            stall = 0
            last_update = time.time()
            continue

        t1 = min(deadline, time.time() + 18.0)
        h2, c2 = project_gradient(h, n, t1, beta=1.0, max_steps=28)
        if c2 < best_c - 1e-14:
            h = h2.copy()
            best_h = h.copy()
            best_c = c2
            stall = 0
            last_update = time.time()
            continue

        t2 = min(deadline, time.time() + 10.0)
        h2, c2 = project_gradient(h, n, t2, beta=5.0, max_steps=18)
        if c2 < best_c - 1e-14:
            h = h2.copy()
            best_h = h.copy()
            best_c = c2
            stall = 0
            last_update = time.time()
            continue

        if rng is not None and time.time() - last_update > 25.0:
            kick = min(0.22, 0.004 * (stall + 1))
            noise = make_smooth_noise(n, rng, kick, 2 + (stall % 3))
            h = project_feasible(h + noise, n)
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-14:
                best_c = c_now
                best_h = h.copy()
            stall += 1
        else:
            stall += 1

        if stall > 35:
            break

    if compute_c5(best_h, n) < best_c:
        best_c = compute_c5(best_h, n)
    return best_h, best_c


def binary_improve(h, n, deadline, rng):

                    if hb[i] + hb[i + s] == 1.0:

                        cands.append((i - t, i))
                        break

        for _ in range(70):

            b = int(rng.randint(n))
            if a != b:

        used = 0
        improved = False
        for a, b in cands:
            used += 1
            if used > 120:

            if hb[a] == hb[b]:
                continue
            hb[a], hb[b] = hb[b], hb[a]
            c2 = compute_c5(hb, n)
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = hb.copy()
                improved = True
                break
            hb[a], hb[b] = hb[b], hb[a]

            stall = 0

    return best_h, best_c
