import time
import numpy as np
from scipy.optimize import minimize

def exact_c5(h, n=None):
    h = np.asarray(h, dtype=float)
    if n is None:
        n = int(h.size)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def feasible_project(v, n, iters=35):
    v = np.asarray(v, dtype=float).ravel()
    n = int(n)
    S = 0.5 * n
    b = np.clip(v, 0.0, 1.0)
    if b.size != n:
        b = b[:n] if b.size > n else np.pad(b, (0, n - b.size))
    if abs(float(b.sum()) - S) < 1e-12:
        return b
    lo = float(b.min()) - 1.0
    hi = float(b.max()) + 1.0
    for _ in range(int(iters)):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(b - mid, 0.0, 1.0).sum())
        if s < S:
            hi = mid
        else:
            lo = mid
    out = np.clip(b - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = S - float(out.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - out))
            add = min(1.0 - out[i], diff)
            if add <= 0:
                break
            out[i] += add
        else:
            i = int(np.argmax(out))
            sub = min(out[i], -diff)
            if sub <= 0:
                break
            out[i] -= sub
    return out

def smooth_c5(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    s = (c - m) / alpha
    sm = float(s.max())
    e = np.exp(s - sm)
    soft = m + (sm + float(np.log(float(e.sum())))) / alpha
    return float(soft * (2.0 / n))

def smooth_c5_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    s = (c - m) / alpha
    sm = float(s.max())
    e = np.exp(s - sm)
    e /= float(e.sum())
    soft = m + (sm + float(np.log(float(e.sum())))) / alpha
    f = float(soft * (2.0 / n))
    w = e
    z = n - 1
    w0 = float(w[z])
    wp = np.zeros(n)
    wm = np.zeros(n)
    wp[1:] = w[n:]
    wm[1:] = w[:n - 1][::-1]
    pref = np.cumsum(wp)
    cumwm = np.cumsum(wm)
    suff = cumwm[n - 1 - np.arange(n)]
    lin = pref + suff + w0
    p = wp + wm
    p[0] = 0.0
    prod = np.convolve(h, p, mode="full")[:n] + np.convolve(h[::-1], p, mode="full")[:n][::-1]
    g = (lin - 2.0 * w0 * h - prod) * (2.0 / n)
    return f, g

def proj_grad_opt(h, n, deadline, alphas=(20.0, 8.0, 3.0, 1.0)):
    h = feasible_project(h, n, 45)
    best_h = h.copy()
    best_c = exact_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        f, g = smooth_c5_grad(h, n, alpha)
        it = 0
        while time.time() < deadline - 4 and it < 2200:
            it += 1
            gn = float(np.linalg.norm(g))
            if gn < 1e-10:
                break
            s0 = min(0.22, 0.22 / (gn + 1e-8))
            acc = False
            for mult in (1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.005, 0.001):
                st = s0 * mult
                if st <= 1e-13:
                    continue
                nv = feasible_project(h - st * g, n, 28)
                nvf = smooth_c5(nv, n, alpha)
                if nvf < f - 1e-12:

                    f = nvf
                    acc = True
                    break
            if not acc:
                break
            if it % 40 == 0:
                ec = exact_c5(h, n)
                if ec < best_c:

            f, g = smooth_c5_grad(h, n, alpha)
        ec = exact_c5(h, n)
        if ec < best_c:
            best_c = ec
            best_h = h.copy()
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    return smooth_c5(h, n, alpha)

def slsqp_jac(x, n, alpha):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    return smooth_c5_grad(h, n, alpha)[1]

def slsqp_con(x, n):
    return float(np.asarray(x).sum()) - 0.5 * float(n)

def slsqp_con_jac(x, n):
    return np.ones(int(n), dtype=float)

def slsqp_opt(h0, n, alpha, maxiter):
    h0 = feasible_project(h0, n, 45)
    try:
        bounds = [(0.0, 1.0)] * int(n)
        cons = [{"type": "eq", "fun": slsqp_con, "jac": slsqp_con_jac, "args": (int(n),)}]
        res = minimize(slsqp_obj, h0, args=(int(n), float(alpha)), method="SLSQP",
                       bounds=bounds, constraints=cons, jac=slsqp_jac,
                       options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False})
        return feasible_project(res.x, n, 50)
    except Exception:
        return h0

def build_basis(n, K, use_sin=False):
    i = np.arange(int(n), dtype=float)
    out = []
    for r in range(1, int(K) + 1):
        fr = 2.0 * np.pi * r * i / float(n)
        out.append(np.cos(fr))
        if use_sin:
            out.append(np.sin(fr))
    if not out:
        return np.empty((0, n))
    return np.vstack(out)

def fit_fourier_params(h, n, K, use_sin, bound):
    i = np.arange(int(n), dtype=float)
    hc = np.asarray(h, dtype=float).ravel() - 0.5
    m = int(K) * (2 if use_sin else 1)
    p = np.zeros(m)
    idx = 0
    for r in range(1, int(K) + 1):
        fr = 2.0 * np.pi * r * i / float(n)
        a = float(np.dot(hc, np.cos(fr)) * (2.0 / n))
        p[idx] = np.arctanh(np.clip(a / bound, -0.99, 0.99))
        idx += 1
        if use_sin:
            b = float(np.dot(hc, np.sin(fr)) * (2.0 / n))
            p[idx] = np.arctanh(np.clip(b / bound, -0.99, 0.99))
            idx += 1
    return p

def fourier_h(params, basis, bound):
    a = bound * np.tanh(np.asarray(params, dtype=float).ravel())
    return 0.5 + basis.dot(a)

def fourier_obj(params, n, basis, bound, alpha):
    h = feasible_project(fourier_h(params, basis, bound), n, 25)
    if alpha <= 1e-12:
        return exact_c5(h, n)
    return smooth_c5(h, n, alpha)

def nelder_fourier(n, basis, p0, bound, alpha, maxiter):
    try:
        res = minimize(fourier_obj, p0, args=(int(n), basis, float(bound), float(alpha)),
                       method="Nelder-Mead",
                       options={"maxiter": int(maxiter), "xatol": 1e-8, "fatol": 1e-10})
        h = feasible_project(fourier_h(res.x, basis, bound), n, 40)
        return h, float(res.fun)
    except Exception:
        return None, None

def interpolate_h(h, n1):
    h = np.asarray(h, dtype=float).ravel()
    n0 = int(h.size)
    n1 = int(n1)
    if n0 == n1:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return np.interp(x1, x0, h)

def coord_derivative(p, h, n):
    L = 2 * int(n) - 1
    z = int(n) - 1
    s = np.zeros(L)
    s[z] = 1.0 - 2.0 * h[p]
    D = int(max(p, n - 1 - p))
    if D > 0:
        d = np.arange(1, D + 1)
        prod = np.zeros(D)
        cl = d <= p
        cr = d <= n - 1 - p
        if np.any(cl):
            prod[cl] += h[p - d[cl]]
        if np.any(cr):
            prod[cr] += h[p + d[cr]]
        s[z - d] += (d <= n - 1 - p).astype(float) - prod
        s[z + d] += (d <= p).astype(float) - prod
    return s

def try_pair(h, n, c, cur, i, j, grid):
    hi = float(h[i])
    hj = float(h[j])
    lo = -hi
    hi2 = min(hj, 1.0 - hi)
    if hi2 - lo < 1e-12:
        return False, cur
    si = coord_derivative(i, h, n)
    sj = coord_derivative(j, h, n)
    v = si - sj
    d = abs(i - j)
    z = n - 1
    deltas = np.linspace(lo, hi2, int(grid))
    if lo < 0.0 < hi2:
        deltas = np.append(deltas, 0.0)
    new = c + np.outer(deltas, v)
    d2 = deltas * deltas
    new[:, z] -= 2.0 * d2
    if d > 0:
        new[:, z - d] += d2
        new[:, z + d] += d2
    vals = np.max(new, axis=1)
    bi = int(np.argmin(vals))
    valc = float(vals[bi])
    if valc < cur - 1e-11:
        dv = float(deltas[bi])
        h[i] = float(np.clip(h[i] + dv, 0.0, 1.0))
        h[j] = float(np.clip(h[j] - dv, 0.0, 1.0))
        return True, valc
    return False, cur

def pair_search(h, n, deadline, radius=10, random_pairs=500, grid=26, rng=None):
    h = feasible_project(h, n, 45)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(c.max())
    if rng is None:
        rng = np.random.RandomState(17)
    for _ in range(2):
        improved = False
        for i in range(int(n) - 1):
            if time.time() > deadline:
                return h, cur * (2.0 / n)
            mj = min(int(n), i + int(radius) + 1)
            for j in range(i + 1, mj):
                up, curv = try_pair(h, n, c, cur, i, j, grid)
                if up:
                    c = np.correlate(h, 1.0 - h, mode="full")
                    cur = float(c.max())
                    improved = True
        for _ in range(int(random_pairs)):
            if time.time() > deadline:
                return h, cur * (2.0 / n)
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            up, curv = try_pair(h, n, c, cur, i, j if j > i else j, grid)
            if up:
                c = np.correlate(h, 1.0 - h, mode="full")
                cur = float(c.max())
                improved = True
        if not improved or time.time() > deadline:
            break
    h = feasible_project(h, n, 50)
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, float(c.max()) * (2.0 / n)

                best_h, best_c = h3.copy(), c3

    if time.time() < t_end - 6:
        best_h, best_c = pair_search(best_h, int(n), max(time.time() + 1, t_end - 4),
                                     radius=10, random_pairs=500, grid=26)

    best_h = feasible_project(best_h, int(n), 55)
    best_c = exact_c5(best_h, int(n))
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel().copy()
    except Exception:
        h0 = np.full(300, 0.5)
    best_n = int(h0.size)
    best_h = feasible_project(h0, best_n, 55)
    best_c = exact_c5(best_h, best_n)

    for hv in (best_h[::-1], 1.0 - best_h, 0.5 * (best_h + best_h[::-1]),
               np.roll(best_h, max(1, best_n // 8)), np.roll(best_h, -max(1, best_n // 8))):
        hv = feasible_project(hv, best_n, 45)
        cv = exact_c5(hv, best_n)
        if cv < best_c:
            best_h, best_c = hv.copy(), cv

    if time.time() < deadline - 260:
        four_end = min(deadline - 240, time.time() + min(260.0, 0.25 * (deadline - t0)))
        n_f = best_n if 120 <= best_n <= 520 else 300
        for nf, K, use_sin in ((n_f, 5, False), (n_f, 3, True), (n_f, 7, False)):
            if time.time() > four_end - 25:
                break
            base = feasible_project(interpolate_h(best_h, nf), nf, 45)
            basis = build_basis(int(nf), K, use_sin)
            if basis.size == 0:
                continue
            m = basis.shape[0]
            p_fit = fit_fourier_params(base, int(nf), K, use_sin, 0.55)
            starts = [p_fit + 0.02 * rng.normal(0.0, 1.0, m)]
            p1 = np.zeros(m)
            p1[0] = np.arctanh(0.5 / 0.55)
            p1[1:] = 0.12 * rng.normal(0.0, 1.0, m - 1)
            if use_sin and m > K:
                p1[K] = 0.05 * rng.normal()
            starts.append(p1)
            for p0 in starts:
                if time.time() > four_end - 20:
                    break
                h_cand, _ = nelder_fourier(int(nf), basis, p0, 0.55, 2.0, maxiter=520)
                if h_cand is not None:
                    h_cand = feasible_project(h_cand, int(nf), 45)
                    c_cand = exact_c5(h_cand, int(nf))
                    if c_cand < best_c + 0.002:
                        if time.time() < four_end - 8:
                            h_cand2, _ = nelder_fourier(int(nf), basis,
                                                        fit_fourier_params(h_cand, int(nf), K, use_sin, 0.55),
                                                        0.55, 0.0, maxiter=350)
                            if h_cand2 is not None:
                                h_cand2 = feasible_project(h_cand2, int(nf), 45)
                                c_cand2 = exact_c5(h_cand2, int(nf))
                                if c_cand2 < c_cand:
                                    h_cand, c_cand = h_cand2, c_cand2
                        if c_cand < best_c:
                            best_h, best_c, best_n = h_cand.copy(), c_cand, int(nf)

    if time.time() < deadline - 110 and best_n > 0:
        n_alt = int(min(600, max(240, round(1.5 * best_n))))
        if n_alt != best_n:
            ha = feasible_project(interpolate_h(best_h, n_alt), n_alt, 45)
            ca = exact_c5(ha, n_alt)
            alloc = min(110.0, max(0.0, deadline - time.time() - 130.0))
            if alloc > 45:
                ha, ca = polish_candidate(ha, n_alt, alloc)
                if ca < best_c:
                    best_h, best_c, best_n = ha.copy(), ca, n_alt

    if time.time() < deadline - 20:
        best_h, best_c = polish_candidate(best_h, best_n, max(0.0, deadline - time.time() - 6.0))

    for hv in (1.0 - best_h, 0.5 * (best_h + best_h[::-1]), np.roll(best_h, best_n // 4)):
        hv = feasible_project(hv, best_n, 50)
        cv = exact_c5(hv, best_n)
        if cv < best_c:
            best_h, best_c = hv.copy(), cv

    hp = feasible_project(best_h, best_n, 60)
    cp = exact_c5(hp, best_n)
    if cp < best_c:
        best_h, best_c = hp, cp
    else:
        best_h = feasible_project(best_h, best_n, 60)
        best_c = exact_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
