    # For k ≠ i,j and k+s = i: h[k]*(1-h'[i]) - h[k]*(1-h[i]) = h[k]*h'[i] - h[k]*h[i] = h[k]*(1-0) - h[k]*0 = h[k]
    #   Wait: h'[i]=1, h[i]=0, so 1-h'[i]=0, 1-h[i]=1. Change = h[k]*0 - h[k]*1 = -h[k]
    # For k ≠ i,j and k+s = j: h[k]*(1-h'[j]) - h[k]*(1-h[j]) = h[k]*1 - h[k]*0 = h[k]
    #   (h'[j]=0, h[j]=1)
    
    # This is getting complicated. The cross terms happen when:
    # - i+s = j (i.e., s = j-i): the term h'[i]*(1-h'[i+s]) = 1*(1-h'[j]) = 1*1 = 1, 
    #   while before it was h[i]*(1-h[i+s]) = 0*(1-h[j]) = 0. Change = +1.
    #   But in F[i, s=j-i]: (1-h[i+s]) - h[i-s] = (1-h[j]) - h[i-(j-i)] = 0 - h[2i-j]

    #   Total from F[i]-F[j] at s=j-i: (0 - h[2i-j]) + (1-h[2j-i] - 0) = 1 - h[2i-j] - h[2j-i]
    #   True change: +1 (from i term) + h[k] where k=j-s = j-(j-i) = i... wait
    #   Let me redo. At s = j-i:
    #   - h'[i]*(1-h'[i+s]) = 1*(1-h'[j]) = 1*(1-0) = 1; before: 0*(1-1) = 0. Delta = 1.
    #   - h'[j]*(1-h'[j+s]) = 0*(1-h'[2j-i]) = 0; before: 1*(1-h[j+s]) = 1*(1-h[2j-i]). Delta = -(1-h[2j-i]) = h[2j-i]-1.
    #   - For k where k+s = i, i.e., k = i-s = i-(j-i) = 2i-j:
    #     h'[k]*(1-h'[i]) - h[k]*(1-h[i]) = h[k]*0 - h[k]*1 = -h[k] = -h[2i-j]
    #   - For k where k+s = j, i.e., k = j-s = j-(j-i) = i:
    #     Already counted (k=i is the first term)
    #   - For k where k = j and j+s is valid:
    #     Already counted (second term)
    #   
    #   Total true change at s=j-i: 1 + (h[2j-i]-1) + (-h[2i-j]) = h[2j-i] - h[2i-j]
    #   (assuming 2i-j and 2j-i are both valid and ≠ i, j)
    
    #   From F[i]-F[j] at s=j-i: 
    #   F[i, j-i] = (1-h[i+(j-i)]) - h[i-(j-i)] = (1-h[j]) - h[2i-j] = 0 - h[2i-j] = -h[2i-j]
    #   F[j, j-i] = (1-h[j+(j-i)]) - h[j-(j-i)] = (1-h[2j-i]) - h[i] = (1-h[2j-i]) - 0 = 1-h[2j-i]
    #   F[i]-F[j] at s=j-i: -h[2i-j] - (1-h[2j-i]) = -h[2i-j] - 1 + h[2j-i] = h[2j-i] - h[2i-j] - 1
    
    #   True change: h[2j-i] - h[2i-j]
    #   F[i]-F[j] gives: h[2j-i] - h[2i-j] - 1
    #   Difference: 1
    #   
    # So there IS a cross-term! The simple F[i]-F[j] is wrong when i+s=j or j+s=i.
    
    # The cross-term correction:
    # When s = j-i: add +1 (from the h'[i]*(1-h'[j]) term being 1 instead of 0)
    # When s = i-j: by symmetry, also add +1
    
    # So the correct formula is:
    # corr_change[s] = F[i, s] - F[j, s] + correction
    # where correction = 1 if s = j-i, +1 if s = i-j, 0 otherwise
    # (but only if the relevant indices are valid)
    
    # Actually, let me just use the original swap_delta approach which handles boundaries correctly.
    # The precomputation of F is still useful as an approximation for filtering.

```python
def compute_F(h, n):
    """Compute F[i, s] = (1-h[i+s]) - h[i-s] with boundary handling"""
    m_range = np.arange(-(n-1), n)  # s values
    F = np.zeros((n, 2*n-1))
    for i in range(n):
        # (1-h[i+s]) for valid i+s
        idx_plus = i + m_range
        valid_plus = (idx_plus >= 0) & (idx_plus < n)
        F[i, valid_plus] += (1 - h[idx_plus[valid_plus]])
        # -h[i-s] for valid i-s
        idx_minus = i - m_range
        valid_minus = (idx_minus >= 0) & (idx_minus < n)
        F[i, valid_minus] -= h[idx_minus[valid_minus]]
    return F
```

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def compute_F(h, n):
    """F[i, s_index] = (1-h[i+s]) - h[i-s] where s = s_index - (n-1)"""
    m_range = np.arange(-(n-1), n)
    idx_plus = np.arange(n)[:, None] + m_range[None, :]
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    term_plus = np.where(valid_plus, 1.0 - h[np.clip(idx_plus, 0, n-1)], 0.0)
    idx_minus = np.arange(n)[:, None] - m_range[None, :]
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    term_minus = np.where(valid_minus, h[np.clip(idx_minus, 0, n-1)], 0.0)
    return term_plus - term_minus

def fast_local_search(h, n, deadline, max_iters=5000, verbose=False):
    """Best-improvement local search using vectorized computation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    
    F = compute_F(h, n)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    steps_since_recompute = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(zeros) < 1 or len(ones) < 1:
            break
        
        zF = F[zeros]  # (n_z, 2n-1)
        oF = F[ones]   # (n_o, 2n-1)

        new_maxes = combined.max(axis=2)

        h[best_j] = 0.0

def ilsb(h, n, deadline, seed, max_rounds=100):
    """Iterated Local Search with binary."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Local search
        cur_h, cur_c = fast_local_search(cur_h, n, deadline)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        
        # Perturb: flip k random bits
        # k increases with stagnation
        k = 2 + rnd // 10
        k = min(k, n // 4)
        zeros = np.where(cur_h == 0)[0]
        ones = np.where(cur_h == 1)[0]
        if len(zeros) < k or len(ones) < k:

        for _ in range(k):

            cur_h[ones[oi]] = 0.0
            zeros = np.where(cur_h == 0)[0]
            ones = np.where(cur_h == 1)[0]
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def compute_F(h, n):
    m_range = np.arange(-(n - 1), n)
    idx_p = np.arange(n)[:, None] + m_range[None, :]
    vp = (idx_p >= 0) & (idx_p < n)
    tp = np.where(vp, 1.0 - h[np.clip(idx_p, 0, n - 1)], 0.0)
    idx_m = np.arange(n)[:, None] - m_range[None, :]
    vm = (idx_m >= 0) & (idx_m < n)
    tm = np.where(vm, h[np.clip(idx_m, 0, n - 1)], 0.0)
    return tp - tm

    F = compute_F(h, n)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    steps = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break

        best_pair_max = cur_max

            temp = corr + F[zeros[zi]]
            nmj = (temp[None, :] - F[ones]).max(axis=1)
            mj = int(np.argmin(nmj))
            if float(nmj[mj]) < best_pair_max - 1e-10:
                best_pair_max = float(nmj[mj])
                best_ii = zi
                best_jj = mj
        if best_ii < 0:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        no_improve = 0
        bi = int(zeros[best_ii])
        bj = int(ones[best_jj])
        h[bi] = 1.0
        h[bj] = 0.0
        corr = corr + F[bi] - F[bj]
        steps += 1
        if steps % 8 == 0:
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_max = float(corr.max())
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
        F = compute_F(h, n)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
    best_c = float(np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx)

        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
