               (50000, 500.0), (200000, 2000.0), (1000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_pen, best_h, jac=jac_pen, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': 6000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_opt(h, n, deadline, alpha=500000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_slq, 'jac': jac_con, 'args': (n,)}]
    try:
        res = minimize(obj_slq, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def build_P(h, n):
    shifts = np.arange(-(n - 1), n)
    idx = np.arange(n)[:, None] + shifts[None, :]
    valid = (idx >= 0) & (idx < n)
    P = np.zeros((n, 2 * n - 1))
    P[valid] = h[idx[valid]]
    return P

def fast_pair_search(h, n, deadline, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5,
                      2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 1e-7]
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    threshold = best_c / dx
    for d in step_sizes:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            C = np.correlate(h, 1 - h, mode='full')
            P = build_P(h, n)
            P_rev = P[:, ::-1]
            best_move = None
            best_val = threshold
            for i in range(n):
                if h[i] + d <= 1.0 + 1e-15:
                    valid_j = (h >= d - 1e-15).copy()
                    valid_j[i] = False
                    if valid_j.any():
                        deltas = d * ((P - P[i, :]) + (P_rev - P_rev[i, :]))
                        new_maxes = np.max(C[None, :] + deltas, axis=1)
                        mask = valid_j & (new_maxes < best_val - 1e-14)
                        if mask.any():
                            cands = np.where(mask)[0]
                            bj = cands[np.argmin(new_maxes[cands])]
                            if new_maxes[bj] < best_val:
                                best_val = new_maxes[bj]
                                best_move = (i, bj, 1)
                if h[i] - d >= -1e-15:
                    valid_j2 = (h <= 1.0 - d + 1e-15).copy()
                    valid_j2[i] = False
                    if valid_j2.any():
                        deltas2 = -d * ((P - P[i, :]) + (P_rev - P_rev[i, :]))
                        new_maxes2 = np.max(C[None, :] + deltas2, axis=1)
                        mask2 = valid_j2 & (new_maxes2 < best_val - 1e-14)
                        if mask2.any():
                            cands2 = np.where(mask2)[0]
                            bj2 = cands2[np.argmin(new_maxes2[cands2])]
                            if new_maxes2[bj2] < best_val:
                                best_val = new_maxes2[bj2]
                                best_move = (i, bj2, -1)
            if best_move is None:
                break
            i, j, sgn = best_move
            h[i] += sgn * d
            h[j] -= sgn * d
            h = np.clip(h, 0.0, 1.0)
            best_c = best_val * dx
            threshold = best_c / dx
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(300, 0.5)
        n = 300
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS
    if time.time() < deadline - 600:
        t1 = min(time.time() + 80, deadline - 600)
        best_h, best_c = lbfgs_opt(best_h, n, t1)
    # Phase 2: SLSQP
    if time.time() < deadline - 500:
        t2 = min(time.time() + 60, deadline - 500)
        h2, c2 = slsqp_opt(best_h, n, t2, alpha=500000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    # Phase 3: Fast pair search (main refinement)
    remaining = deadline - time.time()
    if remaining > 10:
        best_h, best_c = fast_pair_search(best_h, n, deadline)
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
