def d_split(h,n):
    dx=2.0/n
    pref=np.empty(n+1); pref[0]=0.0; pref[1:]=np.cumsum(h)
    aa=np.correlate(h,h,mode='full')
    aut=(aa[n-1:]+aa[n-1::-1])*0.5
    pos=pref[np.arange(n,0,-1)]-aut
    neg=pref[n]-pref[np.arange(n)]-aut
    return pos,neg,dx

def smax_obj(x,n,alpha,mu=0.0):
    pos,neg,dx=d_split(x,n)
    vals=np.empty(2*n-1)
    vals[:n]=pos; vals[n:]=neg[1:]
    mv=vals.max()
    val=dx*(mv + np.log(np.sum(np.exp(alpha*(vals-mv))))/alpha)
    if mu:
        val += mu*(x.sum()-0.5*n)**2
    return float(val)

def smax_grad(x,n,alpha,mu=0.0):
    pos,neg,dx=d_split(x,n)
    vals=np.empty(2*n-1)
    vals[:n]=pos; vals[n:]=neg[1:]
    mv=vals.max()
    w=np.exp(alpha*(vals-mv)); w/=w.sum(); w*=dx
    g=np.zeros(n)
    if w[0] > 1e-16: g += w[0]*(1.0-2.0*x)
    for k in range(1,n):
        wp=w[k]
        if wp > 1e-16:
            nv=n-k
            g[:nv] += wp*(1.0-x[k:])
            g[k:] += wp*(-x[:nv])
    for k in range(1,n):
        wn=w[n-1+k]
        if wn > 1e-16:
            nv=n-k
            g[k:] += wn*(1.0-x[:nv])
            g[:nv] += wn*(-x[k:])
    if mu: g += 2.0*mu*(x.sum()-0.5*n)
    return g

def binary_anneal(bits,n,deadline,rng,temp=2.0):
    mask=(1<<n)-1
    cur=bin_max_bits(bits,n,mask)
    best=cur; best_bits=bits
    pows=[1<<i for i in range(n)]
    it=0
    while time.time()<deadline:
        if rng.random()<0.25:
            # double swap
            i=rng.randint(n)
            while not ((bits>>i)&1): i=rng.randint(n)
            k=rng.randint(n)
            while k==i or not ((bits>>k)&1): k=rng.randint(n)
            j=rng.randint(n)
            while (bits>>j)&1: j=rng.randint(n)
            l=rng.randint(n)
            while l==j or l==i or l==k or (bits>>l)&1: l=rng.randint(n)
            cand=bits ^ pows[i]^pows[k]^pows[j]^pows[l]
        else:

                best=cur; best_bits=bits

            temp=max(0.01,temp*0.98)

import numpy as np
from scipy.optimize import minimize
import time, math

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + diff)
    return h

def c5_val(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def d_split(h, n):
    dx = 2.0 / n
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    aa = np.correlate(h, h, mode='full')
    aut = 0.5 * (aa[n - 1:] + aa[n - 1::-1])
    pos = pref[np.arange(n, 0, -1)] - aut
    neg = pref[n] - pref[np.arange(n)] - aut
    return pos, neg, dx

def smax_obj(x, n, alpha, mu=0.0):
    pos, neg, dx = d_split(x, n)
    vals = np.empty(2 * n - 1)
    vals[:n] = pos
    vals[n:] = neg[1:]
    mv = vals.max()
    val = dx * (mv + np.log(np.sum(np.exp(alpha * (vals - mv)))) / alpha)
    if mu:
        val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)

def smax_grad(x, n, alpha, mu=0.0):
    pos, neg, dx = d_split(x, n)
    vals = np.empty(2 * n - 1)
    vals[:n] = pos
    vals[n:] = neg[1:]
    mv = vals.max()
    w = np.exp(alpha * (vals - mv))
    w /= w.sum()
    w *= dx
    g = np.zeros(n)
    if w[0] > 1e-16:
        g += w[0] * (1.0 - 2.0 * x)
    for k in range(1, n):
        wp = w[k]
        if wp > 1e-16:
            nv = n - k
            g[:nv] += wp * (1.0 - x[k:])
            g[k:] += wp * (-x[:nv])
    for k in range(1, n):
        wn = w[n - 1 + k]
        if wn > 1e-16:
            nv = n - k
            g[k:] += wn * (1.0 - x[:nv])
            g[:nv] += wn * (-x[k:])
    if mu:
        g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def smooth_phase(h, n, deadline, alphas, method='SLSQP', mulim=0.0):
    h = project_feasible(h, n)
    best = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
    if method == 'SLSQP':
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline - 2:
            break
        res = None
        if method == 'L-BFGS-B':
            mit = 60 if n > 250 else 120
            mu = max(1.0, min(5.0, 0.5 * mulim + a / 100000.0))
            opts = {'maxiter': mit, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': mit * 8, 'maxls': 30}
            try:
                res = minimize(smax_obj, h, args=(n, a, mu), jac=smax_grad,
                               method='L-BFGS-B', bounds=bounds, options=opts)
            except Exception:
                pass
        else:
            mit = 50 if n > 250 else 120
            opts = {'maxiter': mit, 'ftol': 1e-12}
            try:
                res = minimize(smax_obj, h, args=(n, a, 0.0), jac=smax_grad,
                               method='SLSQP', bounds=bounds, constraints=cons, options=opts)
            except Exception:
                pass
        if res is None or not np.all(np.isfinite(res.x)):
            continue
        hn = project_feasible(res.x, n)
        cn = c5_val(hn, n)
        if cn < best:
            best = cn
            h = hn
    return h, best

def pair_search(h, n, deadline):
    h = project_feasible(h, n)
    best = c5_val(h, n)
    for d in (0.002, 0.001, 0.0005):
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best
                lo = max(0, i - 18)
                hi = min(n, i + 19)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = c5_val(h, n)
                        if c < best - 1e-15:
                            best = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = c5_val(h, n)
                        if c < best - 1e-15:
                            best = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best

def bits_from_values(v, n, rng=None):
    t = n // 2
    if t <= 0 or t >= n:
        return 0
    v = np.asarray(v, dtype=float)
    if rng is not None:
        v = v + rng.rand(n) * 1e-3
    bits = 0
    for i in np.argsort(-v)[:t]:
        bits |= 1 << int(i)
    return bits

def random_bits(rng, n):
    bits = 0
    for i in rng.choice(n, n // 2, replace=False):
        bits |= 1 << int(i)
    return bits

def bits_to_array(bits, n):
    a = np.zeros(n, dtype=float)
    for i in range(n):
        if (bits >> i) & 1:
            a[i] = 1.0
    return a

def bin_max_bits(bits, n, mask=None):
    if mask is None:
        mask = (1 << n) - 1
    z = mask ^ bits
    best = 0
    for m in range(1, n):
        v = (bits & (z >> m)).bit_count()
        if v > best:
            best = v
        v = (bits & ((z << m) & mask)).bit_count()
        if v > best:
            best = v
    return best

def bin_active_sets(bits, n, best, mask, atol=0, cap=12):
    z = mask ^ bits
    pos = []
    neg = []
    for m in range(1, n):
        v = (bits & (z >> m)).bit_count()
        if v >= best - atol:
            pos.append((v, m))
        v = (bits & ((z << m) & mask)).bit_count()
        if v >= best - atol:
            neg.append((v, m))
    if len(pos) > cap:
        pos.sort(reverse=True)
        pos = [m for _, m in pos[:cap]]
    else:
        pos.sort(reverse=True)
        pos = [m for _, m in pos]
    if len(neg) > cap:
        neg.sort(reverse=True)
        neg = [m for _, m in neg[:cap]]
    else:
        neg.sort(reverse=True)
        neg = [m for _, m in neg]
    return pos, neg

def bin_score_active(bits, mask, ap, an):
    z = mask ^ bits
    s = 0
    for m in ap:
        v = (bits & (z >> m)).bit_count()
        if v > s:
            s = v
    for m in an:
        v = (bits & ((z << m) & mask)).bit_count()
        if v > s:
            s = v
    return s

def binary_hill(bits, n, deadline, rng, cap=12):
    mask = (1 << n) - 1
    cur = bin_max_bits(bits, n, mask)
    ap, an = bin_active_sets(bits, n, cur, mask, 0, cap)
    best = cur
    best_bits = bits
    pows = [1 << i for i in range(n)]
    stall = 0
    while time.time() < deadline:
        ones = []
        zeros = []
        for i in range(n):
            if (bits >> i) & 1:
                ones.append(i)
            else:
                zeros.append(i)
        ones = list(rng.permutation(ones))
        zeros = list(rng.permutation(zeros))
        improved = False
        fe = 0
        for i in ones:
            bi = pows[i]
            for j in zeros:
                cand = bits ^ bi ^ pows[j]
                sc = bin_score_active(cand, mask, ap, an)
                if sc > cur:
                    continue
                if sc < cur or fe < 25:
                    nm = bin_max_bits(cand, n, mask)
                    fe += 1
                    if nm < cur:
                        bits = cand
                        cur = nm
                        if cur < best:
                            best = cur
                            best_bits = bits
                        ap, an = bin_active_sets(bits, n, cur, mask, 0, cap)
                        improved = True
                        break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_bits, best

def binary_anneal(bits, n, deadline, rng, temp=2.0):
    mask = (1 << n) - 1
    cur = bin_max_bits(bits, n, mask)
    best = cur
    best_bits = bits
    pows = [1 << i for i in range(n)]
    it = 0
    while time.time() < deadline:
        if rng.random() < 0.25:
            i = rng.randint(n)
            while not ((bits >> i) & 1):
                i = rng.randint(n)
            k = rng.randint(n)
            while k == i or not ((bits >> k) & 1):
                k = rng.randint(n)
            j = rng.randint(n)
            while (bits >> j) & 1:
                j = rng.randint(n)
            l = rng.randint(n)
            while l == j or l == i or l == k or (bits >> l) & 1:
                l = rng.randint(n)
            cand = bits ^ pows[i] ^ pows[k] ^ pows[j] ^ pows[l]
        else:
            i = rng.randint(n)
            while not ((bits >> i) & 1):
                i = rng.randint(n)
            j = rng.randint(n)
            while (bits >> j) & 1:
                j = rng.randint(n)
            cand = bits ^ pows[i] ^ pows[j]
        nm = bin_max_bits(cand, n, mask)
        diff = nm - cur
        if diff <= 0:
            acc = True
        elif temp > 0.01 and diff <= int(temp) + 1:
            acc = rng.random() < math.exp(-diff / temp)
        else:
            acc = False
        if acc:
            bits = cand
            cur = nm
            if cur < best:
                best = cur
                best_bits = bits
        if cur > best + 8:
            bits = best_bits
            cur = best
        it += 1
        if (it & 1023) == 0:
            temp = max(0.01, temp * 0.98)
    return best_bits, best

def binary_optimize(bits, n, deadline, rng):
    mask = (1 << n) - 1
    cur = bin_max_bits(bits, n, mask)
    best = cur
    best_bits = bits
    while time.time() + 1.0 < deadline:
        seg = min(time.time() + 8.0, deadline - 3.0)
        b2, c2 = binary_hill(best_bits, n, seg, rng)
        if c2 < best:
            best = c2
            best_bits = b2
        if time.time() + 1.0 < deadline:
            endan = min(time.time() + 4.0, deadline - 2.0)
            b3, c3 = binary_anneal(best_bits, n, endan, rng)
            if c3 < best:
                best = c3
                best_bits = b3
            if time.time() + 1.0 < deadline:
                endh = min(time.time() + 3.0, deadline - 1.0)
                b4, c4 = binary_hill(best_bits, n, endh, rng)
                if c4 < best:
                    best = c4
                    best_bits = b4
    return best_bits, best

def interp_h(h, n):
    m = len(h)
    if m == n:
        return h.copy()
    x = (np.arange(m) + 0.5) * (2.0 / m)
    y = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(y, x, h), n)
