    return h_new

import time

def subgradient_descent(h, n, deadline, lr_init=0.01, n_active=5):
    """Subgradient descent on the max correlation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(10000):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        cur_max = float(np.max(c))
        # Find active constraints
        top_idx = np.argsort(c)[-n_active:]
        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in top_idx:
            shift = k_idx - (n - 1)
            lo = max(0, shift)
            hi = min(n, n + shift)
            if hi > lo:
                grad[lo:hi] += (1.0 - h[np.arange(lo, hi) - shift])
            lo = max(0, -shift)
            hi = min(n, n - shift)
            if hi > lo:
                grad[lo:hi] -= h[np.arange(lo, hi) + shift]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        # Step
        h_new = h - lr * grad / gnorm
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            # Backtrack
            lr *= 0.9
        # Decrease learning rate
        lr *= 0.999
    return h, best_c

def fast_cd(h, n, max_passes=500, grid=50, deadline=None, rng=None):
    """Faster coordinate descent with vectorized evaluation."""
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    target = n / 2.0

        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)

            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            # Determine which correlation entries are affected
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            # Correlation values NOT affected (outside [k_lo, k_hi])
            um = 0.0
            if k_lo > 0:

            s = np.zeros(len(ks))

def slsqp_smooth(h, n, alpha, maxiter=1000):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        max_sh = shifted.max()
        if max_sh > 500:
            logsum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
        else:
            logsum = np.log(np.sum(np.exp(shifted)))
        return float((m + logsum / alpha) * dx)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

        h_p += a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    return project_feasible(h_p, n)

- It appears in (1-h[i]) when i = p, contributing h[p+shift]*(1-h[p])

So c_full[k] is linear in h[p] for k ≠ n-1 (both terms are linear)
For k = n-1 (shift=0): c_full[n-1] = Σ h[i](1-h[i]), so h[p] contributes h[p] - h[p]², which is quadratic.

So for each coordinate p, the objective max_k c[k] is the max of:
- A set of linear functions (for k ≠ n-1)
- One quadratic function (for k = n-1)

The minimum of max of linear + one quadratic can be found exactly (or nearly so) by bisection or by checking critical points.

Actually, this is a 1D problem! For each p, we can find the optimal h[p] by:
1. Finding all k where c[k] changes as h[p] varies
2. The max is a piecewise function of h[p]
3. We can find the minimum by checking where the pieces cross

But this is complex. A simpler approach: for each p, the function f(d) = max_k (c[k] + d*s[k]) where s[k] is the sensitivity, plus the quadratic correction for k=n-1. This is a convex piecewise-linear (plus one quadratic) function of d. We can find its minimum by:
- Computing all breakpoints
- Checking the value at each breakpoint and at the vertex of the quadratic

This would give the EXACT optimal step for coordinate p, much better than grid search!

Let me implement this:

```python
def exact_cd_step(p, h, c, k_lo, k_hi, um, s):
    """Find exact optimal delta for coordinate p."""
    n = len(h)
    d_lo = -h[p]
    d_hi = 1.0 - h[p]
    
    # For k in [k_lo, k_hi], c[k] changes as:
    #   linear: c[k] + d * s[k]  (for k ≠ n-1)
    #   quadratic: c[n-1] + d*(1-2h[p]) + ... 
    
    # Wait, let me re-derive.
    # c_full[k] = Σ_i h[i+shift]*(1-h[i])
    # d c_full[k] / d h[p] = s[k] (computed as before)
    # But c_full[n-1] has the quadratic term: h[p] - h[p]²
    # So c_full[n-1] = c[n-1] + d*(1-2*h[p]) - d*d (approximately, since the d*d comes from the (1-h[p]) term)
    
    # Actually more precisely:
    # c_full[n-1] = Σ_i h[i](1-h[i])
    # If h[p] changes by d: the term becomes (h[p]+d)(1-h[p]-d) = h[p]-h[p]² + d(1-2h[p]) - d²
    # So c_full[n-1]_new = c[n-1] + d(1-2h[p]) - d²
    # For other k: c_full[k]_new = c[k] + d*s[k]
    
    # The objective is max(um, max_k_new c[k]_new)
    # = max(um, max over linear pieces, quadratic piece)
    
    # The max of linear functions is convex, and adding a convex function (the quadratic) keeps it convex.
    # So we can find the minimum by bisection on the derivative.
    
    def f(d):
        vals = c[k_lo:k_hi+1] + d * s
        ci = np.arange(k_lo, k_hi+1) == (n-1)
        if np.any(ci):
            idx = np.where(ci)[0]
            vals = vals.copy()
            vals[idx] = c[n-1] + d*(1-2*h[p]) - d*d
        return max(um, float(np.max(vals)))
    
    # Find minimum by golden section or ternary search
    # f is convex in d (max of convex functions)
    a, b = d_lo, d_hi
    # Check endpoints
    fa = f(a)
    fb = f(b)
    f0 = f(0.0)
    
    # If f(0) is already best, return 0
    # Otherwise use golden section
    if fa <= fb and fa <= f0:
        return a
    if fb <= fa and fb <= f0:
        return b
    
    # Golden section search
    phi = (np.sqrt(5) - 1) / 2
    for _ in range(60):
        d1 = b - phi * (b - a)
        d2 = a + phi * (b - a)
        f1 = f(d1)
        f2 = f(d2)
        if f1 < f2:
            b = d2
        else:
            a = d1
        if b - a < 1e-12:
            break
    
    return (a + b) / 2
```

This is much more accurate than grid search! And since f(d) involves just O(n) arithmetic (the max over the affected correlation values), each evaluation is fast.

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        max_sh = shifted.max()
        if max_sh > 500:
            logsum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
        else:
            logsum = np.log(np.sum(np.exp(shifted)))
        return float((m + logsum / alpha) * dx)
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

        b = rng.uniform(-scale, scale)
        h_p += a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    return project_feasible(h_p, n)

                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _eval_cd(d, c_sl, s, um, c_n1, qc, qi):
    vals = c_sl + d * s
    if qi >= 0:
        vals[qi] = c_n1 + d * qc - d * d
    m = float(np.max(vals))
    return max(um, m)

def exact_cd(h, n, max_passes=1000, deadline=None, rng=None):
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    n1 = n - 1
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n1 - p))
            k_hi = min(2 * n - 2, max(p + n1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n1)
            bwd = (ks >= n1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - n1]
            hp = h[p]
            c_n1v = c[n1]
            qc = 1.0 - 2.0 * hp
            qi = n1 - k_lo if (n1 >= k_lo and n1 <= k_hi) else -1
            c_sl = c[k_lo:k_hi + 1].copy()
            f0 = _eval_cd(0.0, c_sl, s, um, c_n1v, qc, qi)
            best_d = 0.0
            best_val = f0
            fd = _eval_cd(d_lo, c_sl, s, um, c_n1v, qc, qi)
            if fd < best_val:
                best_d, best_val = d_lo, fd
            fd = _eval_cd(d_hi, c_sl, s, um, c_n1v, qc, qi)
            if fd < best_val:
                best_d, best_val = d_hi, fd
            if best_val < f0 - 1e-11:
                a, b = d_lo, d_hi
                for _ in range(45):
                    d1 = b - phi * (b - a)
                    d2 = a + phi * (b - a)
                    f1 = _eval_cd(d1, c_sl, s, um, c_n1v, qc, qi)
                    f2 = _eval_cd(d2, c_sl, s, um, c_n1v, qc, qi)
                    if f1 < f2:
                        b = d2
                    else:
                        a = d1
                    if b - a < 1e-13:
                        break
                dm = (a + b) / 2.0
                fm = _eval_cd(dm, c_sl, s, um, c_n1v, qc, qi)
                if fm < best_val:
                    best_d, best_val = dm, fm
            if abs(best_d) > 1e-12 and best_val < f0 - 1e-11:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        logsum = np.log(np.sum(np.exp(sh)))
    return float((m + logsum / alpha) * dx)

def _slsqp_con(x, n):
    return x.sum() - n / 2.0

def _slsqp_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fourier_perturb(h, n, rng, scale=0.03, num_modes=3):
    x = np.linspace(0, 2, n)
    hp = h.copy()
    for m in range(1, num_modes + 1):
        hp += rng.uniform(-scale, scale) * np.cos(m * np.pi * x)
        hp += rng.uniform(-scale, scale) * np.sin(m * np.pi * x)
    return project_feasible(hp, n)

def bottleneck_refine(h, n, deadline, rng):
    """Target the specific k that gives the max and reduce it."""
    h = h.copy()
    best_c = compute_c5(h, n)
    n1 = n - 1
    for _ in range(50):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        k_max = int(np.argmax(c))
        shift = k_max - n1
        # Find coordinates that most affect c[k_max]
        # d c[k_max] / d h[p] = (1-h[p-shift]) - h[p+shift] (for valid indices)
        grad = np.zeros(n)
        lo = max(0, shift)
        hi = min(n, n + shift)
        if hi > lo:
            grad[lo:hi] += (1.0 - h[np.arange(lo, hi) - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[np.arange(lo2, hi2) + shift]
        # Move top in-magnitude coordinates
        top = np.argsort(np.abs(grad))[-10:]
        for p in top:
            d_max = 0.01 * np.sign(-grad[p])
            for step in [d_max, d_max*0.5, d_max*0.25]:
                h_new = h.copy()
                h_new[p] = np.clip(h_new[p] + step, 0, 1)
                h_new = project_feasible(h_new, n)
                cv = compute_c5(h_new, n)
                if cv < best_c - 1e-12:
                    h, best_c = h_new, cv
                    break
    return h, best_c
