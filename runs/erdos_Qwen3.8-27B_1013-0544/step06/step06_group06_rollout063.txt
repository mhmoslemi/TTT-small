    """Compute gradient of smooth max of corr values"""
    center = n - 1
    corr = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    shifted = corr - corr.max()
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()

    grad = np.zeros(n)

    grad += weights[center] * (1 - 2*h)

    # For each offset d (positive and negative):

def compute_smooth_obj_and_grad(h, n, alpha):
    center = n - 1
    corr = np.correlate(h, 1-h, mode='full')
    shifted = corr - corr.max()
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / exp_vals.sum()
    obj = corr.max() + np.log(exp_vals.sum()) / alpha
    
    # Gradient: grad[p] = sum_k weights[k] * d(corr_k)/d(h[p])
    # d(corr_k)/d(h[p]) for offset = k - center:
    #   = (1-h[p-offset])*I(0<=p-offset<n) - h[p+offset]*I(0<=p+offset<n)
    #   + (1-2h[p]) * I(offset==0)  [for the center term]
    
    # Let's compute this as:
    # grad = (1-2h) * weights[center]  [center contribution]
    #      + sum_{offset!=0} weights[center+offset] * [(1-h[p-offset])*I(valid) - h[p+offset]*I(valid)]
    
    # The sum can be decomposed:
    # sum_d w_{+d} * (1-h[p-d]) * I(p>=d) - sum_d w_{+d} * h[p+d] * I(p<n-d)
    # + sum_d w_{-d} * (1-h[p+d]) * I(p<n-d) - sum_d w_{-d} * h[p-d] * I(p>=d)
    
    # = sum_d [w_{+d} * (1-h[p-d]) - w_{-d} * h[p-d]] * I(p>=d)
    #   - sum_d [w_{+d} * h[p+d] - w_{-d} * (1-h[p+d])] * I(p<n-d)
    
    # Let A[d] = w_{+d}*(1) - w_{-d}*(0) ... this is getting complicated.
    
    # Let me just do it loop over offsets:
    grad = np.zeros(n)
    grad += (1-2*h) * weights[center]
    
    for d in range(1, n):
        if center + d < 2*n - 1:
            wd_pos = weights[center + d]
            # offset = +d
            # (1-h[p-d]) for p>=d: indices d..n-1, values (1-h[0..n-1-d])
            # -h[p+d] for p<n-d: indices 0..n-1-d, values -h[d..n-1]
            grad[d:] += wd_pos * (1 - h[:n-d])
            grad[:n-d] -= wd_pos * h[d:]
        
        if center - d >= 0:
            wd_neg = weights[center - d]
            # offset = -d
            # (1-h[p+d]) for p<n-d: indices 0..n-1-d, values (1-h[d..n-1])
            # -h[p-d] for p>=d: indices d..n-1, values -h[0..n-1-d]
            grad[:n-d] += wd_neg * (1 - h[d:])
            grad[d:] -= wd_neg * h[:n-d]
    
    return obj, grad, weights

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def build_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    center = n - 1
    for p in range(n):
        m_s1 = max(0, center + 1 - p)
        m_e1 = min(2 * n - 2, center + n - p)
        if m_s1 <= m_e1:
            vals = np.arange(m_s1, m_e1 + 1)
            idx = p + center - vals
            G[p, vals] += (1.0 - h[idx])
        m_s2 = max(0, center - p)
        m_e2 = min(2 * n - 2, center + n - 1 - p)
        if m_s2 <= m_e2:
            vals = np.arange(m_s2, m_e2 + 1)
            idx = p + vals - center
            G[p, vals] -= h[idx]
        G[p, center] = 0.0
    return G

def update_G(G, i, j, n):
    center = n - 1
    p_idx = np.arange(n)
    m1 = center + p_idx - i
    mask1 = (p_idx != i) & (m1 >= 0) & (m1 < 2 * n - 1)
    G[p_idx[mask1], m1[mask1]] -= 1.0
    m2 = center + i - p_idx
    mask2 = (p_idx != i) & (m2 >= 0) & (m2 < 2 * n - 1)
    G[p_idx[mask2], m2[mask2]] -= 1.0
    m3 = center + p_idx - j
    mask3 = (p_idx != j) & (m3 >= 0) & (m3 < 2 * n - 1)
    G[p_idx[mask3], m3[mask3]] += 1.0
    m4 = center + j - p_idx
    mask4 = (p_idx != j) & (m4 >= 0) & (m4 < 2 * n - 1)
    G[p_idx[mask4], m4[mask4]] += 1.0

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def binary_local_search(h, n, deadline, top_k=15, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    G = build_G(h, n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    temp = 0.01
    stall = 0
    no_improve = 0
    kick_count = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        top_k_actual = min(top_k, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k_actual:]
        best_swap_val = cur_c + 1e-10
        best_i = -1
        best_j = -1

        for m in top_idx:
            g_z = G[zeros, m]
            g_o = G[ones, m]
            bi = zeros[np.argmin(g_z)]
            bj = ones[np.argmax(g_o)]
            new_corr = corr + G[bi] - G[bj]
            nm = new_corr.max()
            if nm < best_swap_val:
                best_swap_val = nm
                best_i = bi
                best_j = bj

        if best_i >= 0 and best_swap_val * dx < cur_c - 1e-13:
            i, j = int(best_i), int(best_j)
            ii = np.where(zeros == i)[0][0]
            jj = np.where(ones == j)[0][0]
            corr = corr + G[i] - G[j]
            update_G(G, i, j, n)
            h[i] = 1.0
            h[j] = 0.0
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            cur_c = corr.max() * dx
            stall = 0
            temp = max(0.002, temp * 0.999)
            if cur_c < best_c - 1e-13:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 40 or (no_improve > 400 and kick_count < 50):
                kick_count += 1
                nk = min(2, len(zeros), len(ones))
                if nk > 0:
                    ri = rng.choice(len(zeros), nk, replace=False)
                    rj = rng.choice(len(ones), nk, replace=False)
                    for a in range(nk):
                        i2, j2 = int(zeros[ri[a]]), int(ones[rj[a]])
                        new_corr = corr + G[i2] - G[j2]
                        new_c = new_corr.max() * dx
                        if np.random.rand() < np.exp(-(new_c - cur_c) / max(temp, 0.002)):
                            ii2 = np.where(zeros == i2)[0][0]
                            jj2 = np.where(ones == j2)[0][0]
                            corr = new_corr
                            update_G(G, i2, j2, n)
                            h[i2] = 1.0
                            h[j2] = 0.0
                            zeros = np.delete(zeros, ii2)
                            ones = np.delete(ones, jj2)
                            zeros = np.append(zeros, j2)
                            ones = np.append(ones, i2)
                            cur_c = corr.max() * dx
                    if iter_count % 50 == 0:
                        corr = np.correlate(h, 1 - h, mode='full').astype(float)
                        G = build_G(h, n)
                        ones = np.where(h == 1)[0].copy()
                        zeros = np.where(h == 0)[0].copy()
                        cur_c = corr.max() * dx
                temp = max(0.002, temp * 0.9)
                stall = 0

    return best_h, best_c

    best_h = h.copy()
