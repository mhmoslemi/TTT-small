from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def lse_c5(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh_max = sh.max()
    return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha

def smooth_opt(h, n, alpha, maxiter=2000):
    def obj(x):
        return lse_c5(x, n, alpha)
    def cf(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cf}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

            break
        improved = False
        zeros = np.where(h == 0)[0].copy()
        ones = np.where(h == 1)[0].copy()

        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                h[i], h[j] = 1.0, 0.0

                h[i], h[j] = 0.0, 1.0
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break

                o = rng.choice(np.where(h==1)[0])
                h[z], h[o] = 1.0, 0.0
                cur_c = float(np.max(np.correlate(h, 1-h, mode='full'))) * dx
            else:
                break
    return best_h, best_c
