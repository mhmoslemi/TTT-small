```python
def compute_grad_matrix(h, n):
    """Returns G[k, m] = ∂corr[m]/∂h[k] for all k, m."""
    # corr[m] = sum_i h[i](1-h[i+s]) where s = m-(n-1)
    # G[k,m] = (1 - h[k+s]) - h[k-s]
    center = n - 1
    G = np.ones((n, 2*n-1))  # the "1" part
    
    for m in range(2*n-1):
        s = m - center
        # h[k+s] for k=0..n-1
        ks = np.arange(n) + s
        valid = (ks >= 0) & (ks < n)
        G[:, m] -= np.where(valid, h[ks[valid]], 0.0)
        # h[k-s]
        ks2 = np.arange(n) - s
        valid2 = (ks2 >= 0) & (ks2 < n)
        G[:, m] -= np.where(valid2, h[ks2[valid2]], 0.0)
    
    return G
```

```python
import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    """Project to {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_grad(h, n, weights):
    """Compute gradient of weighted sum of correlate values."""
    # G[k,m] = (1 - h[k+s_m]) - h[k-s_m]
    # grad[k] = sum_m weights[m] * G[k,m]
    center = n - 1
    k = np.arange(n)
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-15:
            continue
        s = m - center
        ks = k + s
        valid = (ks >= 0) & (ks < n)
        term1 = np.where(valid, 1.0 - h[ks[valid]], 1.0)
        ks2 = k - s
        valid2 = (ks2 >= 0) & (ks2 < n)
        term2 = np.where(valid2, h[ks2[valid2]], 0.0)
        grad += weights[m] * (term1 - term2)
    return grad

    """Optimize using smoothed max with increasing alpha."""
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    alphas = np.logspace(3, np.log10(alpha_max), 20)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        # Initialize with current h
        x = h.copy()
        iters = min(n_iter, max(100, int((deadline - time.time()) * 50)))
        for it in range(iters):
            if time.time() > deadline - 5:
                break
            corr = get_corr(x, n)
            obj_val = corr.max()
            # Softmax weights
            shifted = (corr - obj_val) * alpha
            weights = np.exp(shifted - shifted.max())
            weights /= weights.sum()
            # Only keep significant weights
            mask = weights > 1e-10
            if mask.sum() == 0:
                break
            grad = compute_grad(x, n, weights)
            # Projected gradient step with adaptive step size
            lr = 0.01 / alpha * 100  # tune this
            # Actually, let's use backtracking
            for attempt in range(20):
                x_new = x - lr * grad
                x_new = project_feasible(x_new, n)
                c_new = compute_c5(x_new, n)
                if c_new <= obj_val + 1e-10:
                    break
                lr *= 0.5
            x = x_new
            c_new = compute_c5(x, n)
            if c_new < best_c:
                best_c = c_new
                best_h = x.copy()
            h = x  # use current as starting point for next alpha
    
    return best_h, best_c
```

        x = h.copy()

        max_iter = min(2000, max(50, int((deadline - time.time()) * 100 / (len(alphas) - alphas.index(alpha)))))
        
        for it in range(max_iter):
            if time.time() > deadline - 3:
