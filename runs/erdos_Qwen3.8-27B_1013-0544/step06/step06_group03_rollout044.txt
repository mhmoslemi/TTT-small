def lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n - m] -= h[m:]
            grad[m:] += (1.0 - h[:n - m])
        else:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def lp_step(h_curr, n, trust, top_k=15):
    dx = 2.0 / n
    c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
    top_idx = np.argsort(c)[-top_k:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, np.inf) <= trust]
    for k in top_idx:
        lag = k - (n - 1)
        g = lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def lp_loop(h, n, deadline):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(200):
        if time.time() > deadline - 2:
            break
        h_new = lp_step(h_curr, n, trust, top_k=min(20, 5 + it // 3))
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

def targeted_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    ds = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in ds:
        if time.time() > deadline:
            break
        stagnation = 0
        while stagnation < 50:
            if time.time() > deadline:
                break
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            top_lags = np.argsort(c)[-3:]
            improved = False
            for k_star in top_lags:
                if time.time() > deadline:
                    break
                lag = k_star - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    continue
                if lag > 0:
                    contrib = h[:n - m] * (1 - h[m:])
                else:
                    contrib = h[m:] * (1 - h[:n - m])
                top_pairs = np.argsort(contrib)[-min(10, n - m):]
                for i in top_pairs:
                    j = i + m
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                    elif h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                    else:
                        continue
                    if delta < 1e-9:
                        continue
                    if h[i] > h[j]:
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        if h[i] < h[j] + delta + 1e-12:
                            h[i] += delta
                            h[j] -= delta
                        else:
                            h[i] -= delta
                            h[j] += delta
                if improved:
                    break
            if improved:
                stagnation = 0
            else:
                stagnation += 1
    return h, best_c

def binary_refine(h, n, deadline):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    improved = True
    while improved:
        if time.time() > deadline:
            break
        improved = False
        c = np.correlate(h_bin, 1 - h_bin, mode='full') * (2.0 / n)
        top_lags = np.argsort(c)[-3:]
        for k_star in top_lags:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            if lag > 0:
                critical = h_bin[:n - m] * (1 - h_bin[m:])
            else:
                critical = h_bin[m:] * (1 - h_bin[:n - m])
            for i in np.where(critical > 0)[0][:15]:
                j = i + m
                if h_bin[i] != h_bin[j]:
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                    c_new = compute_c5(h_bin, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_bin.copy()
                        ones = np.where(h_bin == 1.0)[0]
                        zeros = np.where(h_bin == 0.0)[0]
                        improved = True
                        break
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            if improved:
                break
    return best_h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t_lp_end = min(deadline - 30, time.time() + min(70, (deadline - time.time()) * 0.3))
    if time.time() < t_lp_end:
        h_lp, c_lp = lp_loop(best_h, n, t_lp_end)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp
    if time.time() < deadline - 20:
        h_sl = slsqp_solve(best_h, n, 50000, maxiter=4000)
        c_sl = compute_c5(h_sl, n)
        if c_sl < best_c:
            best_c = c_sl
            best_h = h_sl
    if time.time() < deadline - 15:
        h_lr, c_lr = targeted_local_search(best_h, n, deadline - 2)
        if c_lr < best_c:
            best_c = c_lr
            best_h = h_lr
    if time.time() < deadline - 5:
        h_sl2 = slsqp_solve(best_h, n, 500000, maxiter=3000)
        c_sl2 = compute_c5(h_sl2, n)
        if c_sl2 < best_c:
            best_c = c_sl2
            best_h = h_sl2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    if h_init is not None:
        alloc = min(200, max(80, (deadline - t0) * 0.30))
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 250, 400, 150]
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        alloc = min(100, remaining * 0.25)
        if alloc < 25:
            break
        if n_try == n_init:
            continue
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining // 70))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            alloc = min(60, (deadline - time.time() - 15) * 0.35)
            if alloc < 20:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_bin, c_bin = binary_refine(best_h, n, min(deadline - 25, time.time() + 40))
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_final, c_final = targeted_local_search(best_h, n, deadline - 3)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h_final = slsqp_solve(best_h, n, 1000000, maxiter=3000)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
