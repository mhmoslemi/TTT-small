    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    zero_set = list(np.where(h == 0)[0])
    one_set = list(np.where(h == 1)[0])
    
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 500000.0)
    T = t_start
    
    while time.time() < deadline:
        i = zero_set[rng.randint(len(zero_set))]
        j = one_set[rng.randint(len(one_set))]
        delta = swap_delta(h, n, i, j, m_range)

        nc = new_corr.max() * dx
