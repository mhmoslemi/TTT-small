def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap1_delta(h, n, i, j, m_range):
    """Delta for swapping h[i]: 0->1 and h[j]: 1->0"""
    delta = np.zeros(2 * n - 1)
    # i goes 0->1: contributes -h[i+k] - h[i-k]
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] -= h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    # j goes 1->0: contributes +h[j+k] + h[j-k]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] += h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    return delta

def swap2_delta(h, n, a, b, c, d, m_range):
    """Delta for 2-swap: a,c go 0->1; b,d go 1->0"""
    delta = np.zeros(2 * n - 1)
    for p in (a, c):
        im = p + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] -= h[im[msk]]
        imn = p - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
    for p in (b, d):
        im = p + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += h[im[msk]]
        imn = p - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] += h[imn[msk]]
    return delta

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 250000.0)
    T = t_start

    while time.time() < deadline:
        do2 = (two_swap_frac > 0 and len(zeros_idx) >= 2 and len(ones_idx) >= 2 and
               rng.random() < two_swap_frac)
        if do2:
            za = rng.randint(len(zeros_idx))
            zb = rng.randint(len(zeros_idx))
            while zb == za:
                zb = rng.randint(len(zeros_idx))
            oa = rng.randint(len(ones_idx))
            ob = rng.randint(len(ones_idx))
            while ob == oa:
                ob = rng.randint(len(ones_idx))
            a, c = zeros_idx[za], zeros_idx[zb]
            b, d = ones_idx[oa], ones_idx[ob]
            delta = swap2_delta(h, n, a, b, c, d, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            dc = nc - cur_c
            if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                h[a], h[c] = 1.0, 1.0
                h[b], h[d] = 0.0, 0.0
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
            delta = swap1_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            dc = nc - cur_c
            if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = max(3, int(kick_frac * len(ones_idx)))
            perm = rng.permutation(len(ones_idx))
            zcopy = zeros_idx.copy()
            for idx in perm[:nk]:
                h[ones_idx[idx]] = 0.0
                h[zcopy[rng.randint(len(zcopy))]] = 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = (0.0, 1.0)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[bounds] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_polish(h_bin, n, deadline, seed=42, max_rounds=200):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        ntries = min(len(zeros) * len(ones), 20000)
        for _ in range(ntries):
            if time.time() > deadline:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = swap1_delta(h, n, i, j, m_range)
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr = corr + delta
                cur_c = nc
                improved = True
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            nk = max(3, int(0.08 * len(ones)))
            perm = rng.permutation(len(ones))
            zc = zeros.copy()
            for idx in perm[:nk]:
                h[ones[idx]] = 0.0
                h[zc[rng.randint(len(zc))]] = 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP smooth (short)
    p1 = t0 + 0.06 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 400, 1e-14)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Main SA with 2-swap moves
    p2 = t0 + 0.55 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    # Attempt 1: direct
    if time.time() < p2:
        ea = p2 * 0.4 + t0 * 0.6
        hb, cb = improved_sa(h_bin, n, ea, seed=seed, t_start=0.06, t_end=0.0001,
                             patience=45000, two_swap_frac=0.35, kick_frac=0.10)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    # Attempt 2: perturbed
    if time.time() < p2:
        hp = round_to_binary(best_h + 0.12 * rng.randn(n), n)
        ea = p2 * 0.55 + t0 * 0.45
        hb, cb = improved_sa(hp, n, ea, seed=seed + 100, t_start=0.07, t_end=0.0002,
                             patience=40000, two_swap_frac=0.3, kick_frac=0.12)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    # Attempt 3: more perturbed
    if time.time() < p2:
        hp2 = round_to_binary(best_h + 0.2 * rng.randn(n), n)
        hb, cb = improved_sa(hp2, n, p2, seed=seed + 200, t_start=0.08, t_end=0.0003,
                             patience=35000, two_swap_frac=0.4, kick_frac=0.15)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 3: SLSQP refine
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3 - 5:
        hs = project_feasible(best_h + 0.015 * rng.randn(n), n)
        for alpha in [100000, 500000, 2000000]:
            if time.time() > p3:
                break
            hs = slsqp_smooth(hs, n, alpha, 800, 1e-14)
            c = compute_c5(hs, n)
            if c < best_c:
                best_c, best_h = c, hs.copy()

    # Phase 4: Final SA
    p4 = dl - 30
    if time.time() < p4:
        hb2 = round_to_binary(best_h, n)
        hb, cb = improved_sa(hb2, n, p4, seed=seed + 300, t_start=0.04, t_end=0.00005,
                             patience=30000, two_swap_frac=0.3, kick_frac=0.08)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 5: Greedy polish
    if time.time() < dl - 15:
        hg = round_to_binary(best_h, n)
        hg, cg = greedy_polish(hg, n, dl - 12, seed=seed + 400, max_rounds=150)
        if cg < best_c:
            best_c, best_h = cg, hg.copy()

    # Phase 6: Try higher resolution
    if time.time() < dl - 10 and n < 400:
        n2 = min(400, n + 50)
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hh = project_feasible(np.interp(xn, xo, best_h), n2)
        hb3 = round_to_binary(hh, n2)
        hb, cb = improved_sa(hb3, n2, dl - 5, seed=seed + 500, t_start=0.05, t_end=0.0001,
                             patience=25000, two_swap_frac=0.35, kick_frac=0.1)
        if cb < best_c:
            best_c, best_h, n = cb, hb.copy(), n2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
