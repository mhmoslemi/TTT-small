import time
import numpy as np
from scipy.optimize import minimize


def compute_c5_np(h, n):
    n = int(n)
    if n <= 1:
        return 0.0
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def project_feasible(h, n):
    n = int(n)
    S = 0.5 * n
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() < S:
            hi = mid
        else:
            lo = mid
    x = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            rem = diff
            for k in idx:
                if rem <= 0.0:
                    break
                add = min(rem, 1.0 - x[k])
                x[k] += add
                rem -= add
        else:
            idx = np.where(x > 0.0)[0]
            rem = diff
            for k in idx:
                if rem >= 0.0:
                    break
                add = max(rem, -x[k])
                x[k] += add
                rem -= add
    return x


def overlap_c(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    c = np.empty(2 * n - 1, dtype=float)
    off = n - 1
    omh = 1.0 - h
    c[off] = float(np.dot(h, omh))
    for s in range(1, n):
        L = n - s
        c[off + s] = float(np.dot(h[:L], omh[s:]))
        c[off - s] = float(np.dot(h[s:], omh[:L]))
    return c


def lse_counts(c, alpha):
    m = float(np.max(c))
    a = float(alpha)
    with np.errstate(under="ignore", over="ignore"):
        z = np.exp(a * (c - m))
    zs = float(z.sum())
    if not np.isfinite(zs) or zs <= 0.0:
        w = np.zeros_like(c, dtype=float)
        amax = np.where(c >= m - 1e-12)[0]
        if amax.size == 0:
            amax = np.array([int(np.argmax(c))])
        w[amax] = 1.0 / float(amax.size)
        return m, w
    f = m + float(np.log(zs)) / a
    w = z / zs
    return float(f), w


def overlap_grad_weights(h, w):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    off = n - 1
    active = np.where(w > 1e-14)[0]
    for idx in active:
        wv = float(w[idx])
        if wv == 0.0:
            continue
        s = int(idx - off)
        if s == 0:
            g += wv * (1.0 - 2.0 * h)
        elif s > 0:
            L = n - s
            g[:L] += wv * (1.0 - h[s:])
            g[s:] += -wv * h[:L]
        else:
            L = n + s
            s = -s
            g[s:] += wv * (1.0 - h[:L])
            g[:L] += -wv * h[s:]
    return g


def smooth_obj_alpha(x, alpha):
    f, _ = lse_counts(overlap_c(x), float(alpha))
    return f


def smooth_jac_alpha(x, alpha):
    _, w = lse_counts(overlap_c(x), float(alpha))
    return overlap_grad_weights(x, w)


def sum_constraint(x, S):
    return float(np.asarray(x, dtype=float).sum() - float(S))


def slsqp_smooth_once(h, n, alpha, maxiter, deadline):
    h = project_feasible(h, n)
    c0 = compute_c5_np(h, n)
    maxiter = int(maxiter)
    if maxiter <= 0 or time.time() > deadline:
        return h, c0
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_constraint, "args": (0.5 * n,)}]
    opts = {"maxiter": maxiter, "ftol": 1e-12, "maxls": 50, "disp": False}
    try:
        res = minimize(
            fun=smooth_obj_alpha,
            x0=h,
            jac=smooth_jac_alpha,
            args=(float(alpha),),
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options=opts,
        )
        x = project_feasible(res.x, n)
    except Exception:
        x = h
    return x, compute_c5_np(x, n)


def smooth_obj_pen(x, alpha, rho, S):
    f, _ = lse_counts(overlap_c(x), float(alpha))
    d = float(np.asarray(x, dtype=float).sum() - float(S))
    return f + float(rho) * d * d


def smooth_jac_pen(x, alpha, rho, S):
    _, w = lse_counts(overlap_c(x), float(alpha))
    g = overlap_grad_weights(x, w)
    d = float(np.asarray(x, dtype=float).sum() - float(S))
    g += 2.0 * float(rho) * d
    return g


def lbfgs_pen_once(h, n, alpha, rho, iters, deadline):
    h = project_feasible(h, n)
    c0 = compute_c5_np(h, n)
    iters = int(iters)
    if iters <= 0 or time.time() > deadline:
        return h, c0
    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": iters,
        "maxfun": max(50, 8 * iters),
        "ftol": 1e-12,
        "gtol": 1e-8,
        "maxls": 50,
    }
    try:
        res = minimize(
            fun=smooth_obj_pen,
            x0=h,
            jac=smooth_jac_pen,
            args=(float(alpha), float(rho), 0.5 * n),
            bounds=bounds,
            method="L-BFGS-B",
            options=opts,
        )
        x = project_feasible(res.x, n)
    except Exception:
        x = h
    return x, compute_c5_np(x, n)


def active_subgrad_polish(h, n, deadline, max_iter=100, active_tol=1e-3):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_M = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    steps = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005)
    no_improve = 0
    for _ in range(int(max_iter)):
        if time.time() > deadline:
            break
        c = overlap_c(h)
        M = float(np.max(c))
        if M < best_M - 1e-12:
            best_M = M
            best_h = h.copy()
        active = np.where(c >= M - float(active_tol))[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        w = np.zeros_like(c, dtype=float)
        w[active] = 1.0 / float(active.size)
        g = overlap_grad_weights(h, w)
        d = -g + g.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            continue
        d = d / norm
        improved = False
        for eta in steps:
            y = project_feasible(h + float(eta) * d, n)
            if float(np.max(np.abs(y - h))) < 1e-14:
                continue
            M2 = float(np.max(np.correlate(y, 1.0 - y, mode="full")))
            if M2 < best_M - 1e-11:
                best_M = M2
                best_h = y.copy()
                h = y.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 40:
                break
    return best_h, best_M * (2.0 / float(n))


def exact_pair_polish(h, n, deadline, max_sweeps=40, eval_limit=6000, top_k=24):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_M = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    evals = 0
    if n > 500:
        top_k = min(int(top_k), 16)
        eval_limit = min(int(eval_limit), 2500)
    for _ in range(int(max_sweeps)):
        if time.time() > deadline or evals >= int(eval_limit):
            break
        c = overlap_c(h)
        M = float(np.max(c))
        if M < best_M - 1e-12:
            best_M = M
            best_h = h.copy()
        active = np.where(c >= M - 1e-3)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        w = np.zeros_like(c, dtype=float)
        w[active] = 1.0 / float(active.size)
        g = overlap_grad_weights(h, w)
        inc = np.where(h < 1.0 - 1e-7)[0]
        dec = np.where(h > 1e-7)[0]
        if inc.size == 0 or dec.size == 0:
            break
        inc = inc[np.argsort(g[inc])[: int(top_k)]]
        dec = dec[np.argsort(-g[dec])[: int(top_k)]]
        improved = False
        for i in inc:
            if improved or evals >= int(eval_limit):
                break
            cap_i = 1.0 - h[i]
            if cap_i <= 1e-8:
                continue
            for j in dec:
                if i == j or evals >= int(eval_limit):
                    continue
                d = min(cap_i, h[j])
                if d <= 1e-9:
                    continue
                for frac in (1.0, 0.5, 0.2, 0.05):
                    nd = float(frac) * d
                    if nd < 1e-9:
                        continue
                    h[i] += nd
                    h[j] -= nd
                    evals += 1
                    M2 = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                    if M2 < best_M - 1e-10:
                        best_M = M2
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] -= nd
                    h[j] += nd
                    if evals >= int(eval_limit):
                        break
                if improved or evals >= int(eval_limit):
                    break
            if improved or evals >= int(eval_limit):
                break
        if not improved:
            break
        h = best_h.copy()
    return best_h, best_M * (2.0 / float(n))


def resample_h(base, m):
    base = np.asarray(base, dtype=float)
    m = int(m)
    nb = base.size
    if m == nb:
        return base.copy()
    old = (np.arange(nb) + 0.5) / float(nb)
    new = (np.arange(m) + 0.5) / float(m)
    return np.interp(new, old, base)


def binarize_balanced(h, n, seed=0):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n % 2 != 0:
        return h
    S = n // 2
    rng = np.random.RandomState(seed)
    vals = h + rng.uniform(0.0, 1e-12, n)
    idx = np.argsort(vals)[::-1][:S]
    x = np.zeros(n, dtype=float)
    if idx.size:
        x[idx] = 1.0
    return x


def default_initial(n):
    n = int(n)
    x = np.zeros(n, dtype=float)
    if n % 2 == 0:
        x[0:n:2] = 1.0
    else:
        x[0:n:2] = 1.0
    return project_feasible(x, n)

        rem = max(0.0, deadline - time.time())
        iters = int(min(80, max(20, int(rem * 0.30))))
        h5, c5 = slsqp_smooth_once(best_h, best_n, 30000.0, iters, deadline - 5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    if time.time() < deadline - 10:
        hf, cf = active_subgrad_polish(best_h, best_n, deadline - 6, max_iter=160)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    if time.time() < deadline - 4:
        hf, cf = exact_pair_polish(best_h, best_n, deadline - 2, max_sweeps=45, eval_limit=7000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_np(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
