     = δ * [(1-h[i+lag]) - h[i-lag] - (1-h[j+lag]) + h[j-lag]]

where lag = k-(n-1).

So for each lag, the change depends on h at 4 points. We can precompute these and update c_k efficiently.

But actually, the max might be at any k, so we need to know the new max. If we precompute the old max and the second max, and the change only affects a few k values, we can determine the new max quickly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = (c - m) * alpha
    ms = shifted.max()
    if ms > 500:
        val = m + (ms + np.log(np.sum(np.exp(shifted - ms)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

def lbfgs(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                               'gtol': 1e-10, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def subgrad_descent(h, n, n_iter=20000, deadline=None, top_k=3):
    best_c = compute_c5(h, n)
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(np.max(c)) * (2.0 / n)
        if cur < best_c:
            best_c = cur
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            m = abs(lag)
            g[:n - m] += (1.0 - h[m:])
            g[m:] += -h[:n - m]
        g /= top_k
        gn = np.abs(g).max()
        if gn < 1e-12:
            break
        g_dir = -g / gn
        found = False
        for s in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = project_feasible(h + s * g_dir, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                found = True
                break
        if not found:
            break
    return h, best_c

def pair_moves(h, n, deltas, radius, max_rounds=5):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        for _ in range(max_rounds):
            improved = False
            for i in range(n):
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
