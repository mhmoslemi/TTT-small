    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        # Compute weights based on how much each corr[k] contributes to the max
        # Use softmax for smoothness
        alpha = 500  # temperature parameter
        weights = np.exp(alpha * corr / dx)
        weights /= weights.sum()

    grad = np.zeros(n)

            continue
        # (1-h[p+s]) for valid p: p+s in [0,n-1] → p in [max(0,-s), min(n-1, n-1-s)]
