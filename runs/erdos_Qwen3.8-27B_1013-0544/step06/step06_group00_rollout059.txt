def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_D(h, i, j, n):
    """Compute the delta vector D[k] for changing h[i]+=δ, h[j]-=δ."""
    D = np.zeros(2 * n - 1)
    # For each lag k (index n-1+k in the correlation array):
    # Change = δ * [(1-h[i+k]) if valid - h[i-k] if valid - (1-h[j+k]) if valid + h[j-k] if valid]
    
    # Vectorized approach:
    # For term (1-h[i+k]): valid for k such that 0 <= i+k < n, i.e., -i <= k < n-i
    #   contribution to D[n-1+k] = (1-h[i+k]) for those k
    # For term -h[i-k]: valid for k such that 0 <= i-k < n, i.e., i-n+1 < k <= i
    #   contribution to D[n-1+k] = -h[i-k] for those k
    # For term -(1-h[j+k]): valid for k such that 0 <= j+k < n
    #   contribution to D[n-1+k] = -(1-h[j+k])
    # For term +h[j-k]: valid for k such that 0 <= j-k < n
    #   contribution to D[n-1+k] = h[j-k]
    
    k_range = np.arange(-(n-1), n)
    
    # i+k valid: 0 <= i+k < n → -i <= k < n-1-i+1 = n-i
    mask1 = (k_range >= -i) & (k_range < n - i)
    if mask1.any():
        idx = i + k_range[mask1]
        D[n-1+k_range[mask1]] += 1.0 - h[idx]
    
    # i-k valid: 0 <= i-k < n → i-(n-1) < k <= i → k > i-n, k <= i
    mask2 = (k_range > i - n) & (k_range <= i)
    if mask2.any():
        idx = i - k_range[mask2]
        D[n-1+k_range[mask2]] -= h[idx]
    
    # j+k valid: 0 <= j+k < n → -j <= k < n-j
    mask3 = (k_range >= -j) & (k_range < n - j)
    if mask3.any():
        idx = j + k_range[mask3]
        D[n-1+k_range[mask3]] -= 1.0 - h[idx]
    
    # j-k valid: 0 <= j-k < n → j-(n-1) < k <= j → k > j-n, k <= j
    mask4 = (k_range > j - n) & (k_range <= j)
    if mask4.any():
        idx = j - k_range[mask4]
        D[n-1+k_range[mask4]] += h[idx]
    
    return D

def optimal_delta_ternary(C, D, lo, hi, n_iter=40):
    """Find δ that minimizes max(C[k] + δ*D[k]) for δ in [lo, hi]."""
    if lo >= hi - 1e-15:
        return (lo + hi) / 2
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(C + m1 * D) <= np.max(C + m2 * D):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2

def coordinate_descent(h, n, deadline, seed=42, max_no_improve=5000):
    """Main coordinate descent loop."""
    rng = np.random.RandomState(seed)
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = C.max()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = cur_c
    no_improve = 0
    n_total = 2 * n - 1
    
    it = 0
    while time.time() < deadline:
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Determine direction: we'll try both directions
        # Direction 1: h[i] += δ, h[j] -= δ
        # Direction 2: h[i] -= δ, h[j] += δ (equivalent to negative δ in direction 1)
        
        # For direction 1: δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        
        if hi - lo < 1e-14:
            continue
        
        D = compute_D(h, i, j, n)
        delta = optimal_delta_ternary(C, D, lo, hi, n_iter=35)
        
        new_c = np.max(C + delta * D)
        if new_c < cur_c - 1e-14:
            h[i] += delta
            h[j] -= delta
            C += delta * D
            cur_c = new_c
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_improve += 1
        
        it += 1
        if it % 10000 == 0 and no_improve > max_no_improve:
            # Stuck, add perturbation
            break
    
    return best_h, best_c * dx, it

    kr = np.arange(-(n - 1), n)
    m1 = (i + kr >= 0) & (i + kr < n)
    D[n - 1 + kr[m1]] += 1.0 - h[i + kr[m1]]

    D[n - 1 + kr[m4]] += h[j - kr[m4]]

def coord_desc(h, n, deadline, seed=42, no_improve_limit=20000):

    C = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = C.max()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = cur_max
    no_improve = 0
    kr = np.arange(-(n - 1), n)
    
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])

            else:

                best_h = h.copy()
