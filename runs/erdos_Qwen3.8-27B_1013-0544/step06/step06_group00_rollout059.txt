```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

    h_b[idx[:n // 2]] = 1.0
    return h_b

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    return grad

    h = h_init.copy()
    best_h = h.copy()

        if deadline is not None and time.time() > deadline:
            break
        grad = gradient_softmax(h, n, alpha)
        
        # Backtracking line search
        for _ in range(20):
            h_new = project_feasible(h - lr * grad, n)
            obj_new = soft_obj(h_new, n, alpha)
            if obj_new < best_obj or obj_new < soft_obj(h, n, alpha):
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-8)
            h_new = project_feasible(h - lr * grad, n)
        
        h = h_new
        obj = soft_obj(h, n, alpha)
        if obj < best_obj:

            best_h = h.copy()
        
        # Reset learning rate if we made progress
        if obj < best_obj + 1e-10:
            lr = min(lr * 1.2, 2.0)
    
    return best_h, best_obj
```

```python
    h = to_binary(h_init, n).copy()
    c0 = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c0.max() * dx

    # R[i, p] = (1-h[i+d])*[0<=i+d<n] - h[i-d]*[0<=i-d<n]
```

```python
def compute_delta(i, j, h, n, lags):
    """Compute delta in correlation for swap i:0->1, j:1->0."""
    delta = np.zeros(2*n-1)
    # i: 0 -> 1
    v1 = (i + lags >= 0) & (i + lags < n)
    delta[v1] += 1.0 - h[i + lags[v1]]
    v2 = (i - lags >= 0) & (i - lags < n)
    delta[v2] -= h[i - lags[v2]]
    # j: 1 -> 0
    v3 = (j + lags >= 0) & (j + lags < n)
    delta[v3] -= 1.0 - h[j + lags[v3]]
    v4 = (j - lags >= 0) & (j - lags < n)
    delta[v4] += h[j - lags[v4]]
    return delta
```

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def soft_obj_and_grad(h, n, alpha):
    """Compute soft-max objective and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    dc = c * dx
    m = dc.max()
    shifted = alpha * (dc - m)
    sm = shifted.max()
    obj = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    w = np.exp(shifted - sm)
    w /= w.sum()
    
    # Gradient computation (O(n^2) but vectorized inner loop)
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for p in range(len(lags)):
        lag = lags[p]
        wl = w[p]
        if wl < 1e-18:
            continue
        if lag >= 0:
            lo1, hi1 = 0, n - lag
        else:
            lo1, hi1 = -lag, n
        grad[lo1:hi1] += dx * wl * (1.0 - h[lo1 + lag:hi1 + lag])
        if lag >= 0:
            lo2, hi2 = lag, n
        else:
            lo2, hi2 = 0, n + lag
        grad[lo2:hi2] -= dx * wl * h[lo2 - lag:hi2 - lag]
    return obj, grad

def pgd_step(h, n, alpha, lr, deadline):
    """One step of projected gradient descent with line search."""
    obj, grad = soft_obj_and_grad(h, n, alpha)
    for _ in range(30):
        h_new = project_feasible(h - lr * grad, n)
        obj_new, _ = soft_obj_and_grad(h_new, n, alpha)
        if obj_new < obj:
            return h_new, obj_new, lr
        lr *= 0.5
        if lr < 1e-10:
            break
    return h, obj, lr

def pgd_optimize(h_init, n, alpha, deadline, max_iter=5000):

    h = h_init.copy()

            stagnant = 0
        else:
            stagnant += 1
        h = h_new
        if stagnant > 50:

    return best_h, best_obj, lr
