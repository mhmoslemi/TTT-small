- Similarly for j.

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def make_obj(n, alpha, mu):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        g = grad_c5(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    return obj, grad

def lbfgs_phase(h, n, deadline, alphas=None, max_iter=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        obj, grad = make_obj(n, alpha, 0.0)
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iter * 20,
                                   'maxls': 60, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n),
             'args': ()}]
    obj, _ = make_obj(n, alpha, 0.0)
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': -1})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def exact_pair_refine(h, n, deadline, max_pairs_per_sweep=200):
    """Use exact pair optimization on a subset of promising pairs."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for sweep in range(50):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        # Active shifts
        active_threshold = c_max * 0.95
        active_m = np.where(c >= active_threshold)[0]
        
        # Find coordinates that appear in active shifts
        active_coords = set()
        for m in active_m:
            k = m - (n - 1)
            if k >= 0:
                for i in range(n - k):
                    active_coords.add(i)
                    active_coords.add(i + k)
            else:
                kk = -k
                for i in range(kk, n):
                    active_coords.add(i)
                    active_coords.add(i - kk)
        
        active_coords = sorted(active_coords[:100])  # limit to 100
        if len(active_coords) < 2:
            continue
        
        # Try pairs among active coords
        tried = 0
        improved = False
        for ii in range(len(active_coords)):
            if time.time() > deadline:
                return h, best_c
            if tried >= max_pairs_per_sweep:
                break
            i = active_coords[ii]
            for jj in range(ii + 1, min(ii + 20, len(active_coords))):
                if tried >= max_pairs_per_sweep:
                    break
                j = active_coords[jj]
                tried += 1
                
                # Exact pair optimization
                dx = 2.0 / n
                # Compute coefficients
                ci = np.zeros(2 * n - 1)
                cj = np.zeros(2 * n - 1)
                for m in range(2 * n - 1):
                    k = m - (n - 1)
                    # coef for h[i]
                    val_i = 0.0
                    if i + k >= 0 and i + k < n:
                        val_i += (1.0 - h[i + k])
                    if i - k >= 0 and i - k < n:
                        val_i += -h[i - k]
                    ci[m] = val_i
                    # coef for h[j]
                    val_j = 0.0
                    if j + k >= 0 and j + k < n:
                        val_j += (1.0 - h[j + k])
                    if j - k >= 0 and j - k < n:
                        val_j += -h[j - k]
                    cj[m] = val_j
                
                a = compute_c5_vec(h, n)
                b = (ci - cj) * dx
                
                L = max(-h[i], h[j] - 1.0)
                U = min(1.0 - h[i], h[j])
                if L >= U - 1e-12:
                    continue
                
                # Ternary search
                lo, hi = L, U
                for _ in range(45):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    if np.max(a + b * m1) < np.max(a + b * m2):
                        hi = m2
                    else:
                        lo = m1
                t = (lo + hi) / 2.0
                t = max(L, min(U, t))
                
                if abs(t) < 1e-12:
                    continue
                
                old_i, old_j = h[i], h[j]
                h[i] = old_i + t
                h[j] = old_j - t
                new_c = compute_c5(h, n)
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    improved = True
                else:
                    h[i] = old_i
                    h[j] = old_j
        
        if not improved:
            break
    
    return h, best_c
