    for _ in range(80):

       if s > target: lo=mid

    p=np.clip(v-(lo+hi)/2,0,1)

       else: p[:] += diff/n
       p=np.clip(p,0,1)
    return p

import time
import numpy as np
from scipy.optimize import minimize

_STOP = [0.0]


def c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def fix_sum(p, t):
    p = np.asarray(p, dtype=float).copy()
    for _ in range(30):
        d = t - float(p.sum())
        if abs(d) < 1e-15:
            break
        if d > 0.0:
            r = 1.0 - p
            s = float(r.sum())
            if s < 1e-15:
                break
            p = p + (d / s) * r
        else:
            s = float(p.sum())
            if s < 1e-15:
                break
            p = p + (d / s) * p
    return np.clip(p, 0.0, 1.0)


def project(v, t=None):
    n = len(v)
    if t is None:
        t = 0.5 * n
    t = float(t)
    v = np.asarray(v, dtype=float).copy()
    if v.min() > -1e-12 and v.max() < 1.0 + 1e-12 and abs(float(v.sum()) - t) < 1e-11:
        return fix_sum(v, t)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(64):
        m = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - m, 0.0, 1.0)))
        if s > t:
            lo = m
        else:
            hi = m
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum(p, t)


def reinterp(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x1, x0, h)


def binary_from(h):
    n = len(h)
    b = np.zeros(n, dtype=float)
    if n % 2 == 0 and n > 1:
        b[np.argsort(h)[-n // 2:]] = 1.0
    return b


def loss_grad(h, n, a):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = np.clip(a * (c - m), -700.0, 700.0)
    e = np.exp(z)
    s = float(e.sum())
    w = e / s
    val = m + float(np.log(s)) / a
    g1 = np.convolve(1.0 - h, w[::-1], mode='full')[n - 1:2 * n - 1]
    g2 = np.convolve(h, w, mode='full')[n - 1:2 * n - 1]
    return val, g1 - g2


def sum_eq(x, t):
    return float(x.sum() - t)


def sum_jac(x, t):
    return np.ones_like(x, dtype=float)


def slsqp_fun(x, n, a):
    return loss_grad(x, n, a)


def slsqp_cb(xk):
    return time.time() >= _STOP[0]


def slsqp(h, n, a, mit):
    t = 0.5 * n
    h0 = project(h, t)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (t,)}]
    opts = {'maxiter': int(mit), 'ftol': 1e-11}
    try:
        try:
            r = minimize(slsqp_fun, h0, args=(n, a), method='SLSQP',
                         bounds=bounds, constraints=cons, jac=True,
                         callback=slsqp_cb, options=opts)
        except TypeError:
            r = minimize(slsqp_fun, h0, args=(n, a), method='SLSQP',
                         bounds=bounds, constraints=cons, jac=True, options=opts)
        x = r.x
        if x is None:
            x = h0
        return project(x, t)
    except Exception:
        return h0


def pair_polish(h, n, deadline, rng, evals_limit=8000):
    h = np.asarray(h, dtype=float).copy()
    best = c5(h)
    bases = (0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0006, 0.0003, 0.00015)
    e = 0
    nb = len(bases)
    for bi, bbase in enumerate(bases):
        if time.time() > deadline or e >= evals_limit:
            break
        stage = e
        for _ in range(evals_limit // nb):
            if time.time() > deadline or e >= evals_limit:
                break
            if rng.rand() < 0.18:
                r = int(n)
            else:
                r = int(min(n, max(1, (n // 6 + 2) * rng.rand() * rng.rand() + 1)))
            i = int(rng.randint(0, n))
            lo = max(0, i - r)
            hi = min(n - 1, i + r)
            j = int(rng.randint(lo, hi + 1))
            if i == j:
                continue
            d = bbase * (0.5 + rng.rand())
            opts = []
            if h[i] + d <= 1.0 - 1e-15 and h[j] - d >= 1e-15:
                opts.append(1.0)
            if h[i] - d >= 1e-15 and h[j] + d <= 1.0 - 1e-15:
                opts.append(-1.0)
            if not opts:
                continue
            s = opts[int(rng.randint(0, len(opts)))]
            h[i] += s * d
            h[j] -= s * d
            val = c5(h)
            e += 1
            if val < best - 1e-15:
                best = val
            else:
                h[i] -= s * d
                h[j] += s * d
        if bi > 2 and e == stage:
            break
    return h, best


def binary_ls(h, n, deadline, rng, evals_limit=1500):
    if n % 2 != 0:
        return project(h), c5(h)
    b = binary_from(h)
    cur = c5(b)
    best = cur
    bh = b.copy()
    temp = 0.002
    for _ in range(int(evals_limit)):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(b > 0.5)
        zeros = np.flatnonzero(b <= 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[int(rng.randint(0, ones.size))]
        j = zeros[int(rng.randint(0, zeros.size))]
        b[i] = 0.0
        b[j] = 1.0
        val = c5(b)
        if val < best:
            best = val
            bh = b.copy()
        acc = val < cur - 1e-14
        if not acc and temp > 1e-12:
            diff = val - cur
            if diff < 50.0 * temp and rng.rand() < np.exp(-diff / temp):
                acc = True
        if acc:
            cur = val
        else:
            b[i] = 1.0
            b[j] = 0.0
        temp *= 0.9995
    bh = project(bh)
    cb = c5(bh)
    if cb < best:
        best = cb
    return bh, best


def optimize_candidate(h0, n, deadline, rng):
    n = int(n)
    _STOP[0] = max(0.0, deadline - 1.0)
    h_cur = project(h0)
    h_best = h_cur.copy()
    best = c5(h_best)

    if n <= 250:
        mit = 260
    elif n <= 400:
        mit = 170
    elif n <= 560:
        mit = 100
    else:
        mit = 55

    for a in (20.0, 80.0, 300.0, 1000.0, 3000.0):
        if time.time() > deadline - 18.0:
            break
        h2 = slsqp(h_cur, n, a, mit)
        c2 = c5(h2)
        if c2 < best:
            h_best = h2
            best = c2
        if c2 < best + 0.012:
            h_cur = h2

    ccur = c5(h_cur)
    if ccur < best:
        h_best, best = h_cur, ccur

    rt = max(50, 250000 // max(1, n))
    el = int(min(22000, max(800, max(0.0, deadline - time.time()) * rt))))
    if time.time() < deadline - 10.0 and el > 200:
        h_best, best = pair_polish(h_best, n, deadline - 4.0, rng, el)

    if time.time() < deadline - 8.0:
        h3 = slsqp(h_best, n, 8000.0, max(35, mit // 2))
        c3 = c5(h3)
        if c3 < best:
            h_best, best = h3, c3

    rt2 = max(30, 160000 // max(1, n))
    el2 = int(min(8000, max(300, max(0.0, deadline - time.time()) * rt2))))
    if time.time() < deadline - 3.0 and el2 > 100:
        h_best, best = pair_polish(h_best, n, deadline - 0.5, rng, el2)

    hbest = project(h_best)
    cbest = c5(hbest)
    return hbest, cbest


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    if budget_s <= 30.0:
        try:
            p = np.asarray(initial_h_values, dtype=float).copy()
        except Exception:
            p = np.full(240, 0.5, dtype=float)
        b = project(p)
        return b, c5(b), len(b)

    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(int(seed))

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = np.full(240, 0.5, dtype=float)
    if parent.size < 2:
        parent = np.full(240, 0.5, dtype=float)

    n0 = int(parent.size)
    best = project(parent)
    best_c = c5(best)

    alloc = min(380.0, max(80.0, 0.30 * budget_s))
    if t0 + alloc > deadline - 30.0:
        alloc = max(30.0, deadline - 30.0 - t0)
    if alloc > 20.0:
        h2, c2 = optimize_candidate(best, n0, t0 + alloc, rng)
        if c2 < best_c:
            best, best_c = h2, c2

    if time.time() < deadline - 110.0:
        hb, _ = binary_ls(best, n0, deadline - 40.0, rng, 1300)
        alloc = min(200.0, max(40.0, 0.16 * budget_s))
        now = time.time()
        if now + alloc > deadline - 35.0:
            alloc = max(35.0, deadline - 35.0 - now)
        if alloc > 25.0:
            h2, c2 = optimize_candidate(hb, n0, now + alloc, rng)
            if c2 < best_c:
                best, best_c = h2, c2

    tried = {n0}
    for fac in (0.85, 1.15, 0.70, 1.30, 1.60, 0.95):
        nn = int(round(n0 * fac / 2.0) * 2)
        if nn < 100 or nn > 700 or nn in tried:
            continue
        tried.add(nn)
        if time.time() > deadline - 85.0:
            break
        rem = deadline - time.time()
        alloc = min(135.0, max(35.0, 0.18 * rem))
        now = time.time()
        if now + alloc > deadline - 35.0:
            alloc = max(35.0, deadline - 35.0 - now)
        if alloc < 25.0:
            break
        hs = project(reinterp(best, nn))
        h2, c2 = optimize_candidate(hs, nn, now + alloc, rng)
        if c2 < best_c:
            best, best_c = h2, c2

    nbest = int(best.size)
    for s in (0.05, 0.12, 0.25):
        if time.time() > deadline - 85.0:
            break
        hp = project(best + rng.normal(0.0, s, nbest))
        rem = deadline - time.time()
        alloc = min(95.0, max(30.0, 0.18 * rem))
        now = time.time()
        if now + alloc > deadline - 35.0:
            alloc = max(30.0, deadline - 35.0 - now)
        if alloc < 25.0:
            break
        h2, c2 = optimize_candidate(hp, nbest, now + alloc, rng)
        if c2 < best_c:
            best, best_c = h2, c2

    if time.time() < deadline - 90.0 and nbest % 2 == 0:
        hb, _ = binary_ls(best, nbest, deadline - 35.0, rng, 900)
        alloc = min(110.0, max(30.0, deadline - 35.0 - time.time()))
        now = time.time()
        if now + alloc > deadline - 30.0:
            alloc = max(30.0, deadline - 30.0 - now)
        if alloc > 25.0:
            h2, c2 = optimize_candidate(hb, nbest, now + alloc, rng)
            if c2 < best_c:
                best, best_c = h2, c2

    if time.time() < deadline - 20.0:
        alloc = min(160.0, max(35.0, 0.45 * (deadline - time.time())))
        if time.time() + alloc > deadline - 5.0:
            alloc = max(30.0, deadline - 5.0 - time.time())
        if alloc > 25.0:
            h2, c2 = optimize_candidate(best, len(best), time.time() + alloc, rng)
            if c2 < best_c:
                best, best_c = h2, c2

    best = project(best)
    best_c = c5(best)
    return best, float(best_c), int(len(best))
