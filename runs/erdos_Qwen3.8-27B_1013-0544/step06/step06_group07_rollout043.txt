def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def weyl_init(n, alpha=None):
    """Create a binary sequence using Weyl equidistribution."""
    if alpha is None:
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
    target = n // 2
    positions = np.zeros(target, dtype=int)
    for i in range(target):
        positions[i] = int((n * (i * alpha) % 1.0))
    # Handle collisions
    h = np.zeros(n)
    used = set()
    for p in positions:
        if p not in used:
            h[p] = 1.0
            used.add(p)
    # Fill remaining if collisions
    count = int(h.sum())
    if count < target:
        remaining = np.where(h == 0)[0]
        extra = target - count
        rng = np.random.RandomState(0)
        for _ in range(extra):
            idx = rng.choice(remaining)
            if h[idx] == 0:
                h[idx] = 1.0
                remaining = np.where(h == 0)[0]
    return h

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

    T = t_start

    ones_idx = np.where(h == 1)[0]

                gmask = (h == 0).astype(int).copy()

                gpos = np.where(gmask > 0)[0]

                cpos = valid[cmask]

            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]

                no_improve = 0
