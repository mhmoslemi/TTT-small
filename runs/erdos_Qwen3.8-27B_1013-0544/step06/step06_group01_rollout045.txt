                for idx, d in enumerate(fine_deltas):
                    if abs(d) < 1e-15:
                        continue
                    new_c = affected_c + d * slopes
                    if klo <= m <= khi:
                        pos = m - klo
                        new_c[pos] -= d * d
                    val = max(um, float(np.max(new_c)))
                    if val < best_val:
                        best_val = val
                        best_d = d
            
            if abs(best_d) > 1e-12 and best_val < current_max - 1e-13:
                # Apply the update
                c[klo:khi+1] += best_d * slopes
                if klo <= m <= khi:
                    pos = m - klo
                    c[klo + pos] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[klo:khi+1])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max

def pair_search_fast(h, n, deadline, deltas_list, radius=50):
    """Pair search with incremental correlation updates."""
    h = h.copy()
    m = n - 1
    c = compute_corr(h, n)
    best_c = float(np.max(c))
    
    for delta in deltas_list:
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                # Try moving delta from j to i: h[i]+=delta, h[j]-=delta
                if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                    continue
                
                # Compute new max efficiently
                # h[i] changes by +delta, h[j] changes by -delta
                # Affected range for i: [max(0, m-i), min(2n-2, m-i+n-1)]
                # Affected range for j: [max(0, m-j), min(2n-2, m-j+n-1)]
                
                # For the max, we need to check:
                # 1. Unaffected correlations (unchanged)
                # 2. Affected correlations (changed)
                
                # The max of unaffected is at most the old max
                # We only need to check if any affected correlation exceeds the old max
                
                # Range for i
                iklo = max(0, m - i)
                ikhi = min(2*n-2, m - i + n - 1)
                # Range for j
                jklo = max(0, m - j)
                jkhi = min(2*n-2, m - j + n - 1)
                
                # Combined affected range
                a_klo = min(iklo, jklo)
                a_khi = max(ikhi, jkhi)
                
                # Unaffected max
                um = best_c  # Upper bound: the old max is an upper bound for unaffected
                # Actually, unaffected values are exactly c[k] for k outside affected range
                if a_klo > 0:
                    um = max(um, float(np.max(c[:a_klo]))) if a_klo < len(c) else um
                if a_khi < 2*n-2:
                    um = max(um, float(np.max(c[a_khi+1:])))
                
                # For affected range, compute new values
                # This is O(n) per pair, which is too slow for O(n²) pairs
                # Skip this approach - too slow
                
                # Instead, just do a quick check: if the current corr values 
                # at the affected positions are already well below best_c, skip
                # (optimization: early rejection)
                
                # For now, just do the full computation for promising pairs
                # This is still O(n) per pair...
                
                # Let's just try the move and evaluate
                h[i] += delta
                h[j] -= delta
                # Quick estimate: check only the affected positions
                # For i: corr[k] += delta*(1-2*h[i+k-m]) for k in [iklo, ikhi]
                # For j: corr[k] += (-delta)*(1-2*h[j+k-m]) for k in [jklo, jkhi]
                # Plus quadratic at m
                
                # This is getting complex. Let's just compute the full correlation
                # for now and rely on the radius to limit the number of pairs.
                c_new_max = float(np.max(compute_corr(h, n)))
                if c_new_max < best_c - 1e-14:
                    best_c = c_new_max
                    c = compute_corr(h, n)  # Update full correlation
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        
        if not improved:
            break
    
    return h, best_c

Hmm, the pair search is still too slow with full correlation computation. Let me think of a better way.

Actually, for the pair search, the key optimization is: after a move is accepted, we update the correlation array incrementally. For the evaluation of a candidate move, we only need to check if any correlation value exceeds the current max. We can do this by:
1. Computing the affected range
2. For each k in the affected range, computing the new value
3. Checking if the max of (new affected values, old unaffected max) < current best

The old unaffected max can be precomputed. The affected range size is at most 2n, so each evaluation is O(n). With radius R, we have O(nR) pairs per delta, and O(n) work per pair, so O(n²R) per delta. For n=300, R=50, that's 4.5M operations per delta - feasible.

But we can do better: precompute the slopes for each point, and use vectorized operations.

Actually, let me reconsider the whole approach. The previous code achieved 0.380864. The target is 0.38085. The difference is tiny. Maybe I should focus on making the CD more aggressive and adding a final polishing step.

Let me think about what's different between 0.380864 and 0.38085:
- 0.380864 = max_corr * dx where dx = 2/300 = 0.00667
- So max_corr ≈ 0.380864 / 0.00667 ≈ 57.13
- 0.38085 / 0.00667 ≈ 57.13 (almost the same)

The difference is about 0.000014 in C₅, which is 0.0021 in max_corr. Very small.

Let me focus on a cleaner, more efficient implementation:

1. **Fast CD with vectorized line search**: For each coordinate, compute the affected range once, then do a vectorized grid search over delta values.

2. **Block coordinate descent**: Instead of one coordinate at a time, optimize over blocks of coordinates simultaneously. This can escape shallow local minima.

3. **Simulated annealing / cooling**: Start with large steps and gradually decrease.

4. **Multiple n values**: Try n=100, 150, 200, 250, 300, 400.

Let me write a cleaner, more efficient version:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            h = np.where(h > 0.0, h, 0.0)
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot) if tot > 0 else 1.0
            h = np.clip(h, 0, 1)
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, max_passes=1000, grid=80, deadline=None, rng=None):
    """Fast coordinate descent with vectorized line search."""
    h = h.copy()
    m = n - 1
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    
    # Precompute slopes for each point (they don't change during CD)
    # slopes[p] = 1 - 2*h[p + k - m] for k in affected range
    # Actually these DO change as h changes. So we can't precompute.
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            span = d_hi - d_lo
            if span < 1e-12:
                continue
            
            klo = max(0, m - p)
            khi = min(2*n - 2, m - p + n - 1)
            num_affected = khi - klo + 1
            
            # Unaffected max
            if klo > 0 and khi < 2*n - 2:
                um = float(np.maximum(np.max(c[:klo]), np.max(c[khi+1:])))
            elif klo > 0:
                um = float(np.max(c[:klo]))
            elif khi < 2*n - 2:
                um = float(np.max(c[khi+1:]))
            else:
                um = -1e18
            
            # Affected values and their slopes
            start = p + klo - m
            end = p + khi - m
            aff_c = c[klo:khi+1]
            slopes = 1.0 - 2.0 * h[start:end+1]
            
            # Grid search for best delta
            # f(δ) = max(um, max(aff_c + δ*slopes - δ²*I[m in range]))
            deltas = np.linspace(d_lo, d_hi, grid)
            
            # Vectorized: new_c[grid_idx, k_idx] = aff_c[k_idx] + deltas[grid_idx]*slopes[k_idx]
            # new_c[grid_idx, pos_m] -= deltas[grid_idx]^2 if m in range
            new_c = aff_c[None, :] + deltas[:, None] * slopes[None, :]
            if klo <= m <= khi:
                pos = m - klo
                new_c[:, pos] -= deltas**2
            
            vals = np.maximum(um, new_c.max(axis=1))
            best_idx = int(np.argmin(vals))
            best_d = deltas[best_idx]
            best_val = vals[best_idx]
            
            # Fine refinement
            if best_val < current_max - 1e-12 and 0 < best_idx < grid - 1:
                lo = deltas[best_idx - 1]
                hi = deltas[best_idx + 1]
                fine = np.linspace(lo, hi, grid)
                new_cf = aff_c[None, :] + fine[:, None] * slopes[None, :]
                if klo <= m <= khi:
                    new_cf[:, pos] -= fine**2
                vals_f = np.maximum(um, new_cf.max(axis=1))
                best_fi = int(np.argmin(vals_f))
                best_d = fine[best_fi]
                best_val = vals_f[best_fi]
            
            if abs(best_d) > 1e-10 and best_val < current_max - 1e-13:
                c[klo:khi+1] += best_d * slopes
                if klo <= m <= khi:
                    c[m] -= best_d ** 2
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                current_max = max(um, float(np.max(c[klo:khi+1])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max

def slsqp_optimize(h, n, alpha=100000, maxiter=2000, deadline=None):
    """Smooth max approximation with SLSQP."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            return float((m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha) * dx)
        return float((m + np.log(np.sum(np.exp(s))) / alpha) * dx)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    if deadline is not None and time.time() > deadline:
        return h
    try:
        res = minimize(obj, h, args=(), method='SLSQP',
                      bounds=[(0.0, 1.0)]*n,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
