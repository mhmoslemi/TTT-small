import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_weighted_corr(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    nz = np.nonzero(weights)[0]
    for idx in nz:
        w = weights[idx]
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * dx * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * dx * h[lag:]
        else:
            k = -lag
            grad[:n - k] += w * dx * (1.0 - h[k:])
            grad[k:] -= w * dx * h[:n - k]
    return grad

def smooth_max_val(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = g - m
    thresh = -min(40.0, 3.0 / alpha)
    mask = shifted > thresh
    if mask.sum() == 0:
        return m
    sm = shifted[mask].max()
    w_sum = np.exp(shifted[mask] - sm).sum()
    return m + (sm + np.log(w_sum)) / alpha

def smooth_max_weights(g, alpha):
    m = g.max()
    shifted = g - m
    thresh = -min(40.0, 3.0 / alpha)
    mask = shifted > thresh
    sm = shifted[mask].max()
    w = np.zeros(len(g))
    w[mask] = np.exp(shifted[mask] - sm)
    w /= w.sum()
    return w

def obj_full(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def jac_full(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    w = smooth_max_weights(g, alpha)
    return grad_weighted_corr(x, n, w)

def obj_active(x, n, alpha, top_k):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(g)[::-1][:top_k]
    m = g[idx[0]]
    shifted = np.zeros(2 * n - 1)
    shifted[idx] = g[idx] - m
    sm = shifted.max()
    w_sum = np.exp(shifted[idx] - sm).sum()
    return m + (sm + np.log(w_sum)) / alpha

def jac_active(x, n, alpha, top_k):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(g)[::-1][:top_k]
    m = g[idx[0]]
    shifted = np.zeros(2 * n - 1)
    shifted[idx] = g[idx] - m
    sm = shifted.max()
    w = np.zeros(2 * n - 1)
    w[idx] = np.exp(shifted[idx] - sm)
    w /= w.sum()
    return grad_weighted_corr(x, n, w)

def con_fun(x, n):
    return x.sum() - n / 2.0

def con_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter, ftol, top_k=None):
    if top_k is not None:
        fun = partial(obj_active, n=n, alpha=alpha, top_k=top_k)
        jac = partial(jac_active, n=n, alpha=alpha, top_k=top_k)
    else:
        fun = partial(obj_full, n=n, alpha=alpha)
        jac = partial(jac_full, n=n, alpha=alpha)
    cf = partial(con_fun, n=n)
    cj = partial(con_jac, n=n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_greedy(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
