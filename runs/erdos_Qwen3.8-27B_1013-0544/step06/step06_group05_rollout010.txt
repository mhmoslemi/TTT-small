        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        j = rng.choice(ones)

        new_max = new_c.max()

            cur_max = new_max

                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max * dx

def legendre_init(n, p):
    """Create a binary sequence based on Legendre symbols mod p."""
    # Use the first n values of the Legendre sequence of length p
    h = np.zeros(n)
    for i in range(n):
        if i == 0:
            h[i] = 1.0  # arbitrary
        else:
            # Legendre symbol (i/p)
            ls = pow(i, (p-1)//2, p)
            if ls == p - 1:  # -1 mod p
                h[i] = 0.0
            else:  # +1
                h[i] = 1.0
    # Adjust to have exactly n//2 ones
    n_ones = int(h.sum())
    if n_ones > n // 2:
        excess = np.where(h == 1)[0]
        for idx in excess[n_ones - n//2:]:
            h[idx] = 0.0
    elif n_ones < n // 2:
        deficit = np.where(h == 0)[0]
        for idx in deficit[:n//2 - n_ones]:
            h[idx] = 1.0
    return h

def random_binary(n, rng):
    """Random binary sequence with exactly n//2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h
