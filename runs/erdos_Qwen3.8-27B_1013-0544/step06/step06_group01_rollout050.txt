def slsqp_smooth(h, n, alpha, maxiter=2000):
    """SLSQP with smooth log-sum-exp objective."""
    dx = 2.0/n

        m = c.max()
        return float((m + np.log(np.sum(np.exp(c-m)))) / alpha * dx)
    
    def con_fun(x):
        return x.sum() - n/2.0
    
    bounds = [(0.0, 1.0)]*n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n), 'args': ()}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fourier_to_h(params, n):
    K = len(params) // 2
    h = np.full(n, 0.5)
    j = np.arange(n)
    for k in range(K):
        h += params[2*k] * np.cos(2*np.pi*k*j/n) + params[2*k+1] * np.sin(2*np.pi*k*j/n)
    return h

def h_to_fourier(h, n, K):
    hc = h - 0.5
    params = np.zeros(2*K)
    j = np.arange(n)
    for k in range(K):
        params[2*k] = 2.0/n * np.sum(hc * np.cos(2*np.pi*K*j/n + 2*np.pi*(k-K)*j/n + 0))
        # Simpler:
        params[2*k] = 2.0/n * np.sum(hc * np.cos(2*np.pi*k*j/n))
        params[2*k+1] = 2.0/n * np.sum(hc * np.sin(2*np.pi*k*j/n))
    return params

def fast_cd(h, n, max_passes=300, grid=60, deadline=None, rng=None):
    h = h.copy()
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    target = n / 2.0
    pen = 5.0
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(n-1-p, p))
            k_hi = min(2*n-2, max(2*n-2-p, p+n-1))
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < 2*n-2:
                um = max(um, float(np.max(c[k_hi+1:])))
            ks = np.arange(k_lo, k_hi+1)
            s = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                offset = k - n + 1
                if offset == 0:
                    s[idx] = 1.0 - 2.0*h[p]
                else:
                    if 0 <= p+offset < n:
                        s[idx] += 1.0 - h[p+offset]
                    if 0 <= p-offset < n:
                        s[idx] -= h[p-offset]
            diag_idx = (ks == n-1)
            best_d = 0.0
            best_val = current_max + pen*(sumh-target)**2
            deltas = np.linspace(d_lo, d_hi, grid)
            for d in deltas:
                if abs(d) < 1e-15:
                    continue
                new_c = c[ks] + d*s
                if np.any(diag_idx):
                    new_c[diag_idx] -= d*d
                val = max(um, float(np.max(new_c)))
                val += pen*(sumh+d-target)**2
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                c[ks] += best_d * s
                if np.any(diag_idx):
                    c[ks[diag_idx]] -= best_d*best_d
                h[p] = np.clip(h[p]+best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max*(2.0/n)

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0/n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        return float((m + np.log(np.sum(np.exp(c-m))))/alpha*dx)
    def con_fun(x):
        return x.sum() - n/2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)]*n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def fourier_search(h_start, n, K, deadline, rng):
    """Global search in Fourier coefficient space using Nelder-Mead."""
    best_c = compute_c5(h_start, n)
    best_h = h_start.copy()
    n_tries = 3
    for attempt in range(n_tries):
        if time.time() > deadline:
            break
        if attempt == 0:
            params = h_to_fourier(h_start, n, K)
        else:
            # Random perturbation in Fourier space
            params = h_to_fourier(h_start, n, K) + rng.normal(0, 0.05/(attempt+1), 2*K)
        dx2 = 2.0/n
        def f(p):
            h_new = fourier_to_h(p, n)
            h_new = np.clip(h_new, 0, 1)
            viol = np.sum(np.maximum(0, h_new-1)**2) + np.sum(np.maximum(0, -h_new)**2)
            c = np.correlate(h_new, 1.0-h_new, mode='full')
            return c.max()*dx2 + 20.0*viol
        try:
            res = minimize(f, params, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10, 'adaptive': True})
            h_new = np.clip(fourier_to_h(res.x, n), 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except:
            pass
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=25):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i-radius)
                hi = min(n, i+radius+1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i]+d <= 1.0 and h[j]-d >= 0.0:
                        h[i]+=d; h[j]-=d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i]-=d; h[j]+=d
                    if h[i]-d >= 0.0 and h[j]+d <= 1.0:
                        h[i]-=d; h[j]+=d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i]+=d; h[j]-=d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # Phase A: CD
    t_cd = min((deadline-time.time())*0.25, 80)
    if t_cd > 20:
        h2, c2 = fast_cd(h, n, max_passes=400, grid=70, deadline=time.time()+t_cd, rng=rng)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c2
    # Phase B: SLSQP
    if time.time() < deadline - 60:
        for alpha in [30000, 150000, 500000]:
            if time.time() > deadline - 40:
                break
            h_s = slsqp_smooth(h, n, alpha=alpha, maxiter=1200)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                h, best_c = h_s, c_s
    # Phase C: Fourier global search
    if time.time() < deadline - 40:
        K = min(12, n//8)
        h_f, c_f = fourier_search(h, n, K, time.time()+80, rng)
        if c_f < best_c:
            h, best_c = h_f, c_f
    # Phase D: Final CD
    if time.time() < deadline - 25:
        h3, c3 = fast_cd(h, n, max_passes=200, grid=80, deadline=time.time()+50, rng=rng)
        h3 = project_feasible(h3, n)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            h, best_c = h3, c3
    # Phase E: Pair search
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline-3, [0.005, 0.002, 0.001, 0.0005], radius=30)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Main optimization on initial construction
    alloc1 = (deadline-t0)*0.45
    h_opt, c_opt = optimize_candidate(h_init, n_init, t0+alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try other n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in [200, 400]:
        if n_try == n_init:
            continue
        if time.time() > deadline - 180:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_init, h_init, period=2), n_try)
        alloc = min(130, (deadline-time.time())*0.25)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time()+alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    # Multi-restart
    if time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restart = max(2, int(remaining/55))
        for r in range(n_restart):
            if time.time() > deadline - 50:
                break
            scale = 0.01 + 0.04*(r/max(1,n_restart-1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(45, (deadline-time.time()-30)*0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time()+alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 20:
        n = len(best_h)
        h_f = slsqp_smooth(best_h, n, alpha=800000, maxiter=2000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 8:
        n = len(best_h)
        best_h, best_c = pair_search(best_h, n, deadline-3,
                                     [0.001, 0.0005, 0.0002], radius=30)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
