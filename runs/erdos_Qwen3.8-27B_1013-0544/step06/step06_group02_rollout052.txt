def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, n, alpha):
    """Compute smooth max of correlation values and its gradient."""
    c = compute_c5_all(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w_sum = w.sum()
    w /= w_sum
    dx = 2.0 / n
    obj = c.max() + (mx + np.log(w_sum)) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return obj, grad

    target = n / 2.0
    h = h.copy()
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Now adjust to meet sum constraint
    # We need sum(h + lambda * d) = target where d handles the adjustment
    # Simple approach: bisection on shift
    # If sum too high, reduce values (proportionally or via projection)
    # If sum too low, increase values
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Need to increase sum; add to elements with room
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            # Need to decrease sum; remove from elements
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    dx = 2.0 / n
    lr = lr_init
    while time.time() < deadline:
        c = compute_c5_all(h, n)
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)
            wm = dx / top_k
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        grad += (h.sum() - n / 2.0)
        improved = False
        step = lr
        for _ in range(40):
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_c

def coord_desc(h, n, deadline):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old = h[i]
            cands = [0.0, 1.0, max(0, old - 0.05), max(0, old - 0.01),
                     min(1, old + 0.01), min(1, old + 0.05)]
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c_val = compute_c5(h, n)
                if c_val < best_c - 1e-14:
                    best_c = c_val
                    improved = True
                else:
                    h = project_feasible(np.concatenate([h[:i], [old], h[i+1:]]), n)
        if not improved:
            break
    return h, best_c

def lbfgs_phase(h, n, deadline, max_iter=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_val(compute_c5_all(x, n), alpha) + 0.5 * (x.sum() - n/2.0)**2
        def g(x):
            _w, gg = compute_grad(x, n, alpha)
            gg += (x.sum() - n/2.0)
            return gg
        try:
            res = minimize(obj, h, jac=g, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': max_iter * 15})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def multi_restart(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_tries = max(5, int((deadline - time.time()) / 12))
    for i in range(n_tries):
        if time.time() > deadline - 5:
            break
        scale = 0.02 / (1.0 + 0.5 * i)
        hp = project_feasible(h + scale * rng.randn(n), n)
        hd, cd = adam_opt(hp, n, min(time.time() + 10, deadline - 3), alpha=10000, lr=0.003)
        if cd < best_c - 1e-15:
            best_c = cd
            best_h = hd
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B refinement
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iter=200)
    # Phase 2: Adam
    if time.time() < deadline - 350:
        t2 = min(time.time() + 150, deadline - 350)
        h2, c2 = adam_opt(best_h, n, t2, alpha=5000, lr=0.005)
        if c2 < best_c:
            best_c, best_h = c2, h2
    # Phase 3: Active set descent
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h3, c3 = active_desc(best_h, n, t3, top_k=5, lr_init=0.003)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Coordinate descent
    if time.time() < deadline - 180:
        t4 = min(time.time() + 70, deadline - 180)
        h4, c4 = coord_desc(best_h, n, t4)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: Multi-restart
    if time.time() < deadline - 100:
        t5 = min(time.time() + 130, deadline - 100)
        h5, c5 = multi_restart(best_h, n, t5, seed=seed)
        if c5 < best_c:
            best_c, best_h = c5, h5
    # Phase 6: Try other n
    if time.time() < deadline - 50:
        for n_try in [200, 400]:
            if time.time() > deadline - 35:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            hi = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(35, max(15, deadline - time.time() - 20))
            if alloc < 15:
                break
            ht, ct = adam_opt(hi, n_try, time.time() + alloc * 0.5, alpha=5000, lr=0.003)
            if time.time() < deadline - 10:
                ht, ct = active_desc(ht, n_try, time.time() + min(12, deadline - time.time() - 5), top_k=5, lr_init=0.002)
            if ct < best_c:
                best_c, best_h = ct, ht
    # Phase 7: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = adam_opt(best_h, nf, deadline - 1, alpha=50000, lr=0.001)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
