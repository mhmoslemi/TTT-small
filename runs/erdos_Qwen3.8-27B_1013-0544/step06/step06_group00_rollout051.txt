```python
    """Compute gradient of sum_{k in active_k} corr[k] w.r.t. h"""
    grad = np.zeros(n)
    for k in active_k:
        s = k - (n-1)  # shift
        # corr[k] = sum_i h[i] * (1-h[i+s]) for valid i
        # d corr[k] / d h[j] = (1-h[j+s])*I[0<=j+s<n] - h[j-s]*I[0<=j-s<n]
        # First term: valid j where 0 <= j+s < n => -s <= j < n-s
        lo1 = max(0, -s)
        hi1 = min(n, n-s)
        if hi1 > lo1:
            grad[lo1:hi1] += (1 - h[lo1+s:hi1+s])
        # Second term: valid j where 0 <= j-s < n => s <= j < n+s
        lo2 = max(0, s)
        hi2 = min(n, n+s)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2-s:hi2-s]
    return grad * dx
```

```python
    v = np.zeros(n)
```

```python
    best_c = compute_c5(h, n)
    while time.time() < deadline:

        grad, c = smooth_max_grad(h, n, alpha, top_k)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        c = compute_c5(h, n)
        if c < best_c:

    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def smooth_grad(h, n, alpha, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, 'full') * dx
    m_val = corr.max()
    shifted = alpha * (corr - m_val)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    # Get top active
    active = np.argsort(w)[::-1][:top_k]
    wa = w[active] / w[active].sum()
    grad = np.zeros(n)
    for idx, wi in zip(active, wa):
        s = idx - (n-1)
        lo1 = max(0, -s); hi1 = min(n, n-s)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1 - h[lo1+s:hi1+s])
        lo2 = max(0, s); hi2 = min(n, n+s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2-s:hi2-s]
    grad *= dx
    return grad, m_val

def adam_opt(h, n, alphas, deadline, top_k=15, lr=0.05, n_inner=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 2: break
        h_cur = best_h.copy()
        m_acc = np.zeros(n)
        v_acc = np.zeros(n)
        t = 0
        stage_deadline = min(deadline - 1, time.time() + 60)  # 60s per alpha or until overall deadline
        while time.time() < stage_deadline and t < n_inner:
            t += 1
            grad, c = smooth_grad(h_cur, n, alpha, top_k)
            m_acc = 0.9*m_acc + 0.1*grad
            v_acc = 0.999*v_acc + 0.001*grad*grad
```

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def smooth_grad(h, n, alpha, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, 'full') * dx
    m_val = corr.max()
    shifted = alpha * (corr - m_val)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    active = np.argsort(w)[::-1][:top_k]
    wa = w[active]
    wa /= wa.sum()
    grad = np.zeros(n)
    for idx, wi in zip(active, wa):
        s = int(idx) - (n - 1)
        lo1 = max(0, -s); hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1+s:hi1+s])
        lo2 = max(0, s); hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2-s:hi2-s]
    grad *= dx
    return grad, m_val
