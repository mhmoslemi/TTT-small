def projected_subgradient(h_init, n, max_iter, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    corr = np.correlate(h, 1 - h, mode='full')
    
    for t in range(max_iter):
        if time.time() > deadline:
            break
        # Find active shift
        k_star = int(np.argmax(corr))
        c_val = corr[k_star] * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Compute subgradient
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j - k_star < n:
                g[j] = (1.0 - h[j - k_star])
            if 0 <= j + k_star < n:
                g[j] -= h[j + k_star]
        g *= dx
        
        # Step size (polyak or diminishing)
        step = 1.0 / (2.0 * (t + 1))
        # Or use a fixed step with backtracking
        
        # Nesterov: step from y, not h
        y_new = project_feasible(y - step * g, n)
        
        # Update momentum
        momentum = t / (t + 2.0)
        y = y_new + momentum * (y_new - h)
        h = y_new
        
        # Update correlation
        corr = np.correlate(h, 1 - h, mode='full')
    
    return best_h, best_c

3. If sum < n/2, increase by adding to the smallest values (but not above 1)

A more efficient projection:
- After clipping, if sum > n/2: we need to reduce by (sum - n/2). We can subtract uniformly from all non-zero elements: h = h - (sum - n/2) / count_nonzero, then clip again. Repeat until converged.
- Similarly for sum < n/2.

In practice, this converges in 2-3 iterations. So the projection is O(n) with a small constant.

For small δ, the last term is negligible. So:

The first correction is a convolution of δ with (1-h), and the second is a convolution of h with δ. Both are O(n²) in general.

```python
def fast_subgradient(h_init, n, deadline, c0=0.5):
    """Projected subgradient with Nesterov acceleration."""
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    corr = np.correlate(h, 1 - h, mode='full')
    
    t = 0
    while time.time() < deadline:
        t += 1
        alpha = c0 / np.sqrt(t)
        
        # Find active shift
        k_star = int(np.argmax(corr))
        
        # Compute subgradient at active shift
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j - k_star < n:
                g[j] = (1.0 - h[j - k_star])
            if 0 <= j + k_star < n:
                g[j] -= h[j + k_star]
        g *= dx
        
        # Nesterov step
        y_new = project_fast(y - alpha * g, n)
        
        # Momentum update
        if t > 1:
            mom = (t - 1) / (t + 1.0)
            y = y_new + mom * (y_new - h)
        else:
            y = y_new.copy()
        h = y_new
        
        # Update correlation (O(n^2))
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Check improvement
        c_now = corr.max() * dx
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    
    return best_h, best_c
```

def project_fast(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x) = n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Need to reduce: find threshold t such that sum(clip(h-t, 0, 1)) = target
        # clip(h-t, 0, 1) = max(0, min(1, h-t))
        # For h-t > 1: contributes 1 (h > 1+t, but h<=1 so this only happens if t<0, which it won't be)
        # For 0 < h-t < 1: contributes h-t
        # For h-t < 0: contributes 0
        # So we need: sum_{h_i > t} (h_i - t) + count_{h_i >= 1+t} = target
        # Since h_i <= 1, the second term is 0 (for t>=0)
        # sum_{h_i > t} (h_i - t) = target
        # Let S = {i: h_i > t}, then sum_{i in S} h_i - t*|S| = target
        # t = (sum_{i in S} h_i - target) / |S|
        # This is a fixed-point equation. Use binary search on t.
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2
            val = np.clip(h - mid, 0.0, 1.0).sum()
            if val > target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
        h = np.clip(h - t, 0.0, 1.0)
    else:
        # Need to increase: find threshold t such that sum(clip(h+t, 0, 1)) = target
        lo, hi = 0.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2
            val = np.clip(h + mid, 0.0, 1.0).sum()
            if val < target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
        h = np.clip(h + t, 0.0, 1.0)
    return h
