def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _obj_top(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        res = minimize(_obj_top, h, args=(n, alpha, top_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def single_lag_gradient(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.002
    stall = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = _lag_grad(h, n, lag) * dx
        moved = False
        for _ in range(15):
            h_try = project_feasible(h - lr * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                h = h_try
                best_c = c_try
                lr = min(lr * 1.15, 0.02)
                moved = True
                stall = 0
                break
            lr *= 0.5
        if not moved:
            stall += 1
            lr = max(lr * 0.3, 1e-9)
            if lr < 1e-12 or stall > 30:
                break
    return h, best_c

def cvx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solved = False
    for solver in [cp.GLOP, cp.CBC, cp.SCIP]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def n_sweep(h_best, n_best, deadline):
    x_best = np.linspace(0, 2, n_best)
    best_c = compute_c5(h_best, n_best)
    best_h = h_best.copy()
    best_n = n_best
    n_range = list(range(max(100, n_best - 80), n_best + 120, 40))
    if n_best not in n_range:
        n_range.insert(0, n_best)
    for n_try in n_range:
        if time.time() > deadline:
            break
        x_try = np.linspace(0, 2, n_try)
        h_try = project_feasible(np.interp(x_try, x_best, h_best), n_try)
        h_try = slsqp(h_try, n_try, 50000, maxiter=2000, ftol=1e-14)
        c_try = compute_c5(h_try, n_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            best_n = n_try
    return best_h, best_c, best_n
