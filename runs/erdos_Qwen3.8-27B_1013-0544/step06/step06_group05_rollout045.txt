        g = np.correlate(x, 1.0 - x, mode='full') * dx
        vals = g[top_indices]
        m = vals.max()
        sh = alpha * (vals - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_structural_init(n, seed):
    """Create diverse structural initializations."""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    candidates = []
    # Sine-based
    for freq in [1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
        for ph in [0, np.pi / 2, np.pi]:
            h = (np.sin(freq * x + ph) > 0).astype(float)
            candidates.append(h)
    # Block pattern
    block = n // (2 * n % 7 + 7) if n % 7 != 0 else n // 14
    for offset in range(block):
        h = np.zeros(n)
        for i in range(n):
            if ((i - offset) // block) % 2 == 0:
                h[i] = 1.0
        candidates.append(h)
    # Random
    for _ in range(5):
        h = (rng.rand(n) > 0.5).astype(float)
        candidates.append(h)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search from binarized initial (45% budget)
    p1 = t0 + 0.45 * (budget_s - 10)
    h_bin = binarize_top_n_safe(best_h, n) if False else binarize_top(best_h, n)
    hb, cb = fast_binary_ls(h_bin, n, p1, max_rounds=500000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: SLSQP continuous polish (25% budget)
    p2 = t0 + 0.70 * (budget_s - 10)
    if time.time() < p2:
        for alpha in [1000, 5000, 20000, 100000]:
            if time.time() > p2:
                break
            h = slsqp_opt(best_h.copy(), n, p2, maxiter=4000, alpha=alpha)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < p2:
            h = slsqp_topk(best_h.copy(), n, p2, maxiter=3000, k=5, alpha=20000.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 3: Re-binarize + local search (20% budget)
    p3 = t0 + 0.90 * (budget_s - 10)
    if time.time() < p3:
        h_bin2 = binarize_top(best_h, n)
        hb2, cb2 = fast_binary_ls(h_bin2, n, p3, max_rounds=200000, seed=seed + 313)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 4: Try alternative n values if time remains
    remaining = dl - time.time()
    if remaining > 60:
        p4 = t0 + budget_s - 15
        for alt_n in [200, 300, 350]:
            if time.time() > p4:
                break
            alt_budget = min(remaining * 0.35, p4 - time.time())
            if alt_budget < 30:
                continue
            # Create initial for alt_n by sampling best_h
            interp_idx = (np.arange(alt_n) * n / alt_n).astype(int)
            interp_idx = np.clip(interp_idx, 0, n - 1)
            h_alt = best_h[interp_idx].copy()
            h_alt = project_feasible(h_alt, alt_n)
            h_alt_bin = binarize_top(h_alt, alt_n)
            hb_alt, cb_alt = fast_binary_ls(h_alt_bin, alt_n, time.time() + alt_budget * 0.6,
                                            max_rounds=80000, seed=seed + alt_n)
            if cb_alt < best_c:
                # Polish with SLSQP
                if time.time() < p4:
                    h_pol = slsqp_opt(hb_alt.copy(), alt_n, p4, maxiter=2000, alpha=50000)
                    c_pol = compute_c5(h_pol, alt_n)
                    if c_pol < best_c:
                        best_c = c_pol
                        best_h = h_pol.copy()
                else:
                    best_c = cb_alt
                    best_h = hb_alt.copy()

    # Phase 5: Final SLSQP polish
    if time.time() < dl - 3:
        h = slsqp_opt(best_h.copy(), len(best_h), dl - 2, maxiter=2000, alpha=500000)
        c = compute_c5(h, len(h))
        if c < best_c:
            best_c = c
            best_h = h.copy()

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
