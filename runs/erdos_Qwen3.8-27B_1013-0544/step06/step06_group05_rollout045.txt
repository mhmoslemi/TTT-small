def smooth_max_grad(c, dx, alpha):
    """Compute softmax weights and smooth max value."""
    x = alpha * c * dx
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    p = exp_x / s  # softmax probabilities
    f_val = (x_max + np.log(s)) / alpha
    return f_val, p

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project_continuous(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

def projected_gradient_descent(h, n, max_iter, deadline, alphas=(100, 500, 2000, 10000, 50000)):
    """PGD with log-sum-exp smooth max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = best_h.copy()
        for it in range(max_iter):
            if time.time() > deadline:
                break
            c = compute_corr(h, n)
            # Smooth max
            x = alpha * c * dx
            x_max = x.max()
            exp_x = np.exp(x - x_max)
            p = exp_x / exp_x.sum()
            
            # Gradient
            grad = np.zeros(n)
            for j in range(n):
                g = 0.0
                for m in range(2 * n - 1):
                    lag = m - (n - 1)
                    if 0 <= j - lag < n:
                        g += p[m] * (1.0 - h[j - lag])
                    if 0 <= j + lag < n:
                        g -= p[m] * h[j + lag]
                grad[j] = dx * g
            
            # Line search / fixed step
            lr = 0.5 / np.max(np.abs(grad)) + 1e-10
            h_new = h - lr * grad
            h_new = project_continuous(h_new, n)
            
            # Check improvement
            c_new = compute_c5(h_new, n)
            c_old = compute_c5(h, n)
            if c_new < c_old:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                lr *= 0.5
                h_new = h - lr * grad
                h_new = project_continuous(h_new, n)
                h = h_new
        
        # After each alpha, binarize and re-continue
        sorted_h = np.sort(h)[::-1]
        thresh = sorted_h[n // 2]
        h_bin = (h >= thresh).astype(float)
        # Ensure exact count
        n_ones = int(h_bin.sum())
        if n_ones != n // 2:
            # Fix
            if n_ones > n // 2:
                excess = np.where(h_bin == 1)[0]
                h_bin[excess[n_ones - n//2:]] = 0
            else:
                deficit = np.where(h_bin == 0)[0]
                h_bin[deficit[:n//2 - n_ones]] = 1
        best_h_bin_c = compute_c5(h_bin, n)
        if best_h_bin_c < best_c:
            best_c = best_h_bin_c
            best_h = h_bin.copy()
    
    return best_h, best_c
