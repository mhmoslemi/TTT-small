def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt2, c_opt2 = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2

    # Simulated annealing at best n
    if time.time() < deadline - 60:
        n = len(best_h)
        sa_end = min(deadline - 40, time.time() + 100)
        h_sa, c_sa = sa_search(best_h, n, sa_end, rng)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Post-SA fine optimization
    if time.time() < deadline - 20:
        n = len(best_h)
        if time.time() < deadline - 15:
            h_new = slsqp_step(best_h, n, 3000000, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h, best_c = fine_search(best_h, n, deadline - 2)

    # Final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = fine_search(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
