```
def lp_full_step(h_curr, n, deadline, K=200, trust=0.05):
  if time.time() > deadline - 30: return h_curr.copy()
  c=compute_lags(h_curr,n)
  if K >= len(c): idx=np.arange(len(c))
  else: idx=np.argpartition(c, -K)[-K:]
  hv=cp.Variable(n)
  tv=cp.Variable()
  cons=[cp.sum(hv)==n/2.0, hv>=0, hv<=1]
  # trust: cp.abs(hv-h_curr)<=trust maybe too many constraints; use hv >= h_curr - trust, hv <= h_curr + trust (2n). ok.
  cons.append(hv >= h_curr - trust)
  cons.append(hv <= h_curr + trust)
  dx=2.0/n
  for t in idx:
    lag=t-(n-1)
    g=lag_gradient(h_curr,n,lag)
    cons.append(c[t] + dx*(g @ (hv - h_curr)) <= tv)
  prob=cp.Problem(cp.Minimize(tv),cons)
  for solver in [cp.GLOP, cp.CBC, cp.SCIP]:
    try:
      prob.solve(solver=solver, verbose=False)
    except Exception:
      continue
    if prob.status in ('optimal','optimal_inaccurate') and hv.value is not None:
      h2=project_feasible(hv.value,n)
      return h2
  return h_curr.copy()
```

```
def lag_gradient(h,n,lag):
  g=np.zeros(n,dtype=float)
  m=abs(lag)
  if m==0:
    return 1.0-2.0*h
  if m>=n: return g
  if lag>0:
    g[:n-m] -= h[m:]
    g[m:] += 1.0 - h[:n-m]
  else:
    g[:n-m] += 1.0 - h[m:]
    g[m:] -= h[:n-m]
  return g
```

```python
import numpy as np
import time
from scipy.optimize import minimize
try:
    import cvxpy as cp
except Exception:
    cp = None

def project_feasible(p, n):
    p = np.asarray(p, dtype=float)
    target = 0.5 * float(n)
    if np.min(p) >= -1e-12 and np.max(p) <= 1.0 + 1e-12:
        x = np.clip(p, 0.0, 1.0)
        if abs(x.sum() - target) <= 1e-11:
            return x
    s0 = np.clip(p, 0.0, 1.0).sum()
    if s0 < target:
        lo, hi = 0.0, 1.0
        while np.clip(p + hi, 0.0, 1.0).sum() < target:
            hi *= 2.0
    else:
        hi, lo = 0.0, -1.0
        while np.clip(p + lo, 0.0, 1.0).sum() > target:
            lo *= 2.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(p + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(p + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - x
            tot = room.sum()
            if tot > 0:
                x += diff / tot * room
        else:
            tot = x.sum()
            if tot > 0:
                x += diff / tot * x
    return np.clip(x, 0.0, 1.0)

def compute_lags(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def compute_c5(h, n):
    c = compute_lags(h, n)
    return float(np.max(c)) if c.size else 0.0

def lag_gradient(h, n, lag):
    g = np.zeros(n, dtype=float)
    m = abs(int(lag))
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return g
    if lag > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def top_indices(c, tp):
    L = len(c)
    tp = int(tp)
    if tp <= 0:
        return np.array([], dtype=int)
    if tp >= L:
        return np.arange(L, dtype=int)
    return np.argpartition(c, -tp)[-tp:]

def smooth_of_values(vals, alpha):
    m = float(np.max(vals))
    s = float(alpha) * (vals - m)
    mx = float(np.max(s))
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / float(alpha))

def total_sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)

def total_sum_con_jac(x, n):
    return np.ones(len(x), dtype=float)

def full_smooth(x, n, ids, alpha):
    c = compute_lags(x, n)
    return smooth_of_values(c[ids], alpha)

def full_smooth_jac(x, n, ids, alpha):
    c = compute_lags(x, n)
    vals = c[ids]
    m = float(np.max(vals))
    s = float(alpha) * (vals - m)
    mx = float(np.max(s))
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / float(n)
    center = n - 1
    g = np.zeros(n, dtype=float)
    for i in range(len(ids)):
        wi = w[i]
        if wi > 1e-12:
            g += wi * lag_gradient(x, n, int(ids[i]) - center)
    return dx * g

def slsqp_full(h, n, ids, alpha, maxiter, deadline):
    h0 = project_feasible(h, n).copy()
    if not len(ids):
        return h0
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": total_sum_con, "jac": total_sum_con_jac, "args": (n,)}]
    try:
        res = minimize(full_smooth, h0, args=(n, ids, alpha), method="SLSQP",
                       bounds=bounds, constraints=cons, jac=full_smooth_jac,
                       options={"maxiter": int(maxiter), "ftol": 1e-13})
        x = res.x
    except Exception:
        return h0
    if x is None or time.time() > deadline:
        return h0
    return project_feasible(x, n)

def build_sym(y, n):
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    m = n // 2
    h = np.zeros(n, dtype=float)
    h[:m] = y
    if 2 * m < n:
        h[m] = 0.5
    h[n - m:] = 1.0 - y[::-1]
    return h

def core_from_h(h, n):
    h = np.asarray(h, dtype=float)
    m = n // 2
    y = 0.5 * (h[:m] + 1.0 - h[n - m:][::-1])
    return np.clip(y, 0.0, 1.0)

def anti_symmetrize(h, n):
    return build_sym(core_from_h(h, n), n)

def sym_smooth(y, n, ids, alpha):
    h = build_sym(y, n)
    c = compute_lags(h, n)
    return smooth_of_values(c[ids], alpha)

def sym_smooth_jac(y, n, ids, alpha):
    h = build_sym(y, n)
    c = compute_lags(h, n)
    vals = c[ids]
    m = float(np.max(vals))
    s = float(alpha) * (vals - m)
    mx = float(np.max(s))
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / float(n)
    center = n - 1
    gh = np.zeros(n, dtype=float)
    for i in range(len(ids)):
        wi = w[i]
        if wi > 1e-12:
            gh += wi * lag_gradient(h, n, int(ids[i]) - center)
    gh *= dx
    m = n // 2
    return gh[:m] - gh[n - m:][::-1]

def slsqp_sym(y, n, ids, alpha, maxiter, deadline):
    y0 = np.clip(np.asarray(y, dtype=float), 0.0, 1.0).copy()
    m = n // 2
    if not len(ids) or m == 0:
        return y0
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(sym_smooth, y0, args=(n, ids, alpha), method="SLSQP",
                       bounds=bounds, jac=sym_smooth_jac,
                       options={"maxiter": int(maxiter), "ftol": 1e-13})
        x = res.x
    except Exception:
        return y0
    if x is None or time.time() > deadline:
        return y0
    return np.clip(x, 0.0, 1.0)

def lp_full_step(h_curr, n, deadline, trust=0.08, k_max=170):
    if cp is None or time.time() > deadline - 45:
        return h_curr.copy()
    try:
        c = compute_lags(h_curr, n)
        L = len(c)
        k = min(L, int(k_max))
        idx = top_indices(c, k)
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0,
                hv >= h_curr - trust, hv <= h_curr + trust]
        dx = 2.0 / float(n)
        for t in idx:
            g = lag_gradient(h_curr, n, int(t) - (n - 1))
            cons.append(c[int(t)] + dx * cp.sum(g * (hv - h_curr)) <= tv)
        prob = cp.Problem(cp.Minimize(tv), cons)
        for name in ("GLOP", "CBC", "SCIP", "CVXOPT", "PDLP", "ECOS"):
            s = getattr(cp, name, None)
            if s is None:
                continue
            try:
                prob.solve(solver=s, verbose=False)
            except Exception:
                continue
            if prob.status in ("optimal", "optimal_inaccurate") and hv.value is not None:
                return project_feasible(hv.value, n)
    except Exception:
        pass
    return h_curr.copy()

def refine_full(h, n, deadline, rng):

    best_h = h.copy()

        (90, 25, 90), (55, 90, 140), (35, 350, 170), (24, 1400, 190),

    ]

        for top_m, alpha, maxiter in cfgs:

                break

            ids = top_indices(c, top_m)

            solver_count += 1

                best_c = c2
                best_h = h2.copy()
                work_h = h2.copy()
            else:
                if c2 < best_c + 0.004:
                    work_h = h2.copy()
                else:
                    work_h = best_h.copy()
            if solver_count % 4 == 0 and time.time() < deadline - 70:
                hlp = lp_full_step(work_h, n, deadline, trust=0.07 if solver_count % 8 == 0 else 0.035)
                clp = compute_c5(hlp, n)
                if clp < best_c - 1e-14:
                    best_c = clp
                    best_h = hlp.copy()
                    work_h = hlp.copy()
                elif clp < best_c + 0.012:
                    work_h = hlp.copy()
        if time.time() < deadline - 70 and cycle > 0 and rng is not None:
            hp = project_feasible(best_h + rng.normal(0.0, 0.018, n), n)
            chp = compute_c5(hp, n)
            if chp < best_c + 0.015:
                work_h = hp
        cycle += 1
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def refine_sym(y, n, deadline, rng):
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    best_h = build_sym(y, n)
    best_c = compute_c5(best_h, n)
    work_y = y.copy()
    configs = [
        (130, 25, 110), (70, 100, 170), (45, 400, 210), (32, 1800, 230),
        (24, 9000, 230), (18, 55000, 210), (13, 280000, 190), (9, 1400000, 160)
    ]
    cycle = 0
    while time.time() < deadline - 12 and cycle < 3:
        cfgs = configs if cycle == 0 else configs[-4:]
        for top_m, alpha, maxiter in cfgs:
            if time.time() > deadline - 12:
                break
            c = compute_lags(build_sym(work_y, n), n)
            ids = top_indices(c, top_m)
            y2 = slsqp_sym(work_y, n, ids, alpha, maxiter, deadline)
            h2 = build_sym(y2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h2
                work_y = y2.copy()
            else:
                if c2 < best_c + 0.004:
                    work_y = y2.copy()
                else:
                    work_y = core_from_h(best_h, n)
        if time.time() < deadline - 70 and cycle > 0 and rng is not None:
            m = n // 2
            yp = np.clip(work_y + rng.normal(0.0, 0.025, m), 0.0, 1.0)
            hp = build_sym(yp, n)
            chp = compute_c5(hp, n)
            if chp < best_c + 0.015:
                work_y = yp
        cycle += 1
    return build_sym(work_y, n) if best_c == compute_c5(build_sym(work_y, n), n) else build_sym(core_from_h(best_h, n), n), best_c

def make_anti_wave(n, mode):
    x = np.arange(n, dtype=float) * (2.0 / float(n)) - 1.0
    h = 0.5 * np.ones(n, dtype=float) + 0.5 * np.sin(0.5 * np.pi * float(mode) * x)
    return project_feasible(h, n)

def make_wave(n, freq, phase):
    x = np.arange(n, dtype=float) * (2.0 / float(n))
    h = 0.5 + 0.5 * np.cos(np.pi * float(freq) * x + float(phase))
    return project_feasible(h, n)

def make_anti_fourier(n, coeffs):
    x = np.arange(n, dtype=float) * (2.0 / float(n)) - 1.0
    h = 0.5 * np.ones(n, dtype=float)
    for j, cf in enumerate(coeffs):
        mode = 2 * j + 1
        h += float(cf) * np.sin(0.5 * np.pi * float(mode) * x)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(90.0, float(budget_s)) - 8.0
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        if n <= 0:
            raise ValueError
    except Exception:
        n = 300
        h0 = make_anti_wave(n, 3)

    if n > 620:
        old_n = n
        n = 300
        xo = np.linspace(0.0, 2.0, old_n)
        xn = np.arange(n, dtype=float) * (2.0 / float(n))
        h0 = np.interp(xn, xo, h0)

    h0 = project_feasible(h0, n)
    cands = [
        h0,
        anti_symmetrize(h0, n),
        make_anti_wave(n, 1),
        make_anti_wave(n, 3),
        make_anti_wave(n, 5),
        make_wave(n, 1, 0.0),
        make_wave(n, 1, 1.5707963267948966),
        make_wave(n, 2, 0.0)
    ]
    for _ in range(6):
        cands.append(make_anti_fourier(n, rng.normal(0.0, 0.16, 5)))

    best_h = cands[0].copy()
    best_c = compute_c5(best_h, n)
    for hc in cands[1:]:
        hc = project_feasible(hc, n)
        c = compute_c5(hc, n)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = hc.copy()

    if time.time() < deadline - 80:
        rem = deadline - time.time()
        t1 = time.time() + 0.50 * rem
        hf, cf = refine_full(best_h, n, t1, rng)
        if cf < best_c:
            best_c, best_h = cf, hf

    if time.time() < deadline - 70:
        ys = core_from_h(best_h, n)
        rem = deadline - time.time()
        t2 = deadline - max(35.0, 0.12 * rem)
        hs, cs = refine_sym(ys, n, t2, rng)
        if cs < best_c:
            best_c, best_h = cs, hs

    if time.time() < deadline - 90:
        rem = deadline - time.time()
        t3 = deadline - max(40.0, 0.15 * rem)
        hf, cf = refine_full(best_h, n, t3, rng)
        if cf < best_c:
            best_c, best_h = cf, hf

    if time.time() < deadline - 70:
        ys = core_from_h(best_h, n)
        rem = deadline - time.time()
        t4 = deadline - max(30.0, 0.12 * rem)
        hs, cs = refine_sym(ys, n, t4, rng)
        if cs < best_c:
            best_c, best_h = cs, hs

    if time.time() < deadline - 25:
        c = compute_lags(best_h, n)
        for top, alpha in ((14, 8000), (9, 50000), (7, 250000), (5, 1200000), (4, 4000000)):
            if time.time() > deadline - 10:
                break
            ids = top_indices(c, top)
            h2 = slsqp_full(best_h, n, ids, alpha, 130, deadline)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h2.copy()
                c = compute_lags(best_h, n)

    if time.time() < deadline - 22:
        ys = core_from_h(best_h, n)
        c = compute_lags(build_sym(ys, n), n)
        for top, alpha in ((14, 8000), (9, 50000), (7, 250000), (5, 1200000), (4, 4000000)):
            if time.time() > deadline - 10:
                break
            ids = top_indices(c, top)
            y2 = slsqp_sym(ys, n, ids, alpha, 150, deadline)
            h2 = build_sym(y2, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h2.copy()
                ys = core_from_h(best_h, n)
                c = compute_lags(best_h, n)

    if time.time() < deadline - 45:
        h2 = lp_full_step(best_h, n, deadline, trust=0.03, k_max=150)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    if not np.isfinite(best_c):
        best_h = make_anti_wave(n, 3)
        best_c = compute_c5(best_h, n)

    return best_h, float(best_c), int(n)
```
