def compute_c5(h, n):

def swap_delta(h, i, j, k, n):
    """Compute delta in corr[k] when swapping h[i](1->0) and h[j](0->1)"""
    # Delta = h[i+k] + h[i-k] - h[j+k] - h[j-k] (with bounds checking)
    d = 0.0
    ik = i + k
    if 0 <= ik < n:
        d += h[ik]
    ik = i - k
    if 0 <= ik < n:
        d += h[ik]
    jk = j + k
    if 0 <= jk < n:
        d -= h[jk]
    jk = j - k
    if 0 <= jk < n:
        d -= h[jk]
    return d

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_F(h, lags):
    """Compute F(pos, lag) = h[pos+lag]*I + (h[pos-lag]-1)*I for valid indices."""
    n = len(h)
    lc = len(lags)
    F = np.zeros((n, lc))
    for c in range(lc):
        lag = lags[c]
        if lag == 0:
            # F(pos, 0) = h[pos] + (h[pos]-1) = 2*h[pos] - 1
            F[:, c] = 2.0 * h - 1.0
        else:
            idx1 = np.arange(n) + lag
            idx2 = np.arange(n) - lag
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            F[:, c] = np.where(v1, h[idx1], 0.0) + np.where(v2, h[idx2] - 1.0, 0.0)
    return F

    max_val = corr.max()

        if time.time() > deadline:
            break

def apply_swap(h, corr, i, j, n):
    """Apply swap h[i]: 1->0, h[j]: 0->1. Returns updated h and corr."""
    # Compute delta for all lags using current (old) h
    for k in range(2*n - 1):
        lag = k - (n - 1)
        d = 0.0
        p1 = i + lag
        if 0 <= p1 < n:
            d += h[p1]
        p2 = i - lag
        if 0 <= p2 < n:
            d += (h[p2] - 1.0)

        p4 = j - lag
        if 0 <= p4 < n:
            d -= (h[p4] - 1.0)
        corr[k] += d
    h[i] = 0.0
    h[j] = 1.0
    return h, corr

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_F(h, lags):
    n = len(h)
    lc = len(lags)
    F = np.zeros((n, lc))
    for c in range(lc):
        lag = lags[c]
        if lag == 0:
            F[:, c] = 2.0 * h - 1.0
        else:
            idx1 = np.arange(n) + lag
            idx2 = np.arange(n) - lag
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            F[:, c] = np.where(v1, h[idx1], 0.0) + np.where(v2, h[idx2] - 1.0, 0.0)
    return F

def apply_swap(h, corr, i, j, n):
    lags = np.arange(-(n - 1), n)
    ip = i + lags
    im = i - lags
    jp = j + lags
    jm = j - lags
    vip = (ip >= 0) & (ip < n)
    vim = (im >= 0) & (im < n)
    vjp = (jp >= 0) & (jp < n)
    vjm = (jm >= 0) & (jm < n)
    delta = np.zeros(2 * n - 1)
    if vip.any():
        delta[vip] += h[ip[vip]]
    if vim.any():
        delta[vim] += h[im[vim]] - 1.0
    if vjp.any():
        delta[vjp] -= h[jp[vjp]]
    if vjm.any():
        delta[vjm] -= h[jm[vjm]] - 1.0
    corr += delta
    h[i] = 0.0
    h[j] = 1.0
    return h, corr

def local_search_fast(h, n, deadline, max_sweeps=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    corr = get_corr(h)
    max_val = corr.max()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c0 = corr >= max_val - 0.5
        c1 = (corr >= max_val - 1.5) & (corr < max_val - 0.5)
        c2 = (corr >= max_val - 2.5) & (corr < max_val - 1.5)
        chk = c0 | c1 | c2
        ci = np.where(chk)[0]
        lc = len(ci)
        if lc == 0:
            break
        lags = (ci - (n - 1)).astype(int)
        F = compute_F(h, lags)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        req = np.zeros(lc)
        for c in range(lc):
            k = ci[c]
            if c0[k]:
                req[c] = 1.0
            elif c1[k]:
                req[c] = 0.0
            else:
                req[c] = -1.0
        perm = rng.permutation(len(ones))
        improved = False
        for pi in perm:
            if time.time() > deadline:
                break
            i = ones[pi]
            need = F[i, :] + req
            Fz = F[zeros, :]
            fm = np.all(Fz >= need, axis=1)
            if fm.any():
                margins = (Fz - need)[fm].min(axis=1)
                j = zeros[fm][np.argmax(margins)]
                h, corr = apply_swap(h, corr, i, j, n)
                max_val = corr.max()
                improved = True
                break
        if not improved:
            break
    return h, max_val * (2.0 / n)

def perturb(h, n, n_swaps, rng):
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h[rng.choice(ones)] = 0.0
        h[rng.choice(zeros)] = 1.0
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        e = np.exp(sh - sm)
        w = e / e.sum()
        grad = np.zeros(2 * n - 1)
        for c in range(2 * n - 1):
            lag = c - (n - 1)
            idx1 = np.arange(n) + lag
            idx2 = np.arange(n) - lag
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            if v1.any():
                grad[idx1[v1]] -= dx * w[c]
            if v2.any():
                grad[idx2[v2]] -= dx * w[c]
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
