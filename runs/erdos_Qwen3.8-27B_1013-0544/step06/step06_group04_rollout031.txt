import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    for _ in range(50):
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
                lo = mid
            else:
                hi = mid
        lam = 0.5 * (lo + hi)
        result = np.clip(h - lam, 0.0, 1.0)
        if abs(result.sum() - target) < 1e-10:
            break
    return result

def lse_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    obj = m + (sm + np.log(w.sum())) / alpha  # wait, w.sum() = exp(-sm)*sum(exp(sh))... 
    # Actually: logsumexp(sh) = sm + log(sum(exp(sh-sm)))
    # So obj = m + logsumexp(sh)/alpha = m + (sm + log(sum(exp(sh-sm))))/alpha
    # And w = exp(sh-sm)/sum(exp(sh-sm)), so log(sum(exp(sh-sm))) = -log(w[k] for any k with sh[k]=sm)
    # Simpler: just compute it directly
    obj = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    # Gradient: dx * ifft(fft(w) * fft(v_flip_padded))[0:n]
    v = 1 - 2 * x
    N = 2 * n - 1
    Wf = np.fft.rfft(w, N)
    vp = np.zeros(N)
    vp[:n] = v[::-1]
    Vf = np.fft.rfft(vp, N)
    grad = dx * np.fft.irfft(Wf * Vf, N)[:n]
    return obj, grad

def slsqp_optimize(h, n, alphas, maxiter=300):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            o, _ = lse_obj_and_grad(x, n, alpha)
            return o
        def grad(x):
            _, g = lse_obj_and_grad(x, n, alpha)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def binary_local_search(h, n, deadline, max_sweeps=1000, seed=42):
    # Binarize
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    dx = 2.0 / n
    c = np.correlate(b, 1 - b, mode='full').copy()
    cur_max = float(c.max())
    best_c = cur_max
    best_b = b.copy()
    rng = np.random.RandomState(seed)
    s = 1 - 2 * b  # sign vector: +1 where b=0, -1 where b=1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(b == 0)[0].copy()
        ones = np.where(b == 1)[0].copy()
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            i = int(i)
            # Compute smear of s shifted by i
            # Δc[k] = s[k-i] - s[k-j] (for the best j)
            # First compute smear_i[k] = s[k-i] for valid k
            smear_i = np.zeros(2 * n - 1)
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            if lo <= hi:
                smear_i[lo:hi+1] = s[lo-i:hi-i+1]
            best_nc = cur_max
            best_j = -1
            for j in ones:
                j = int(j)
                smear_j = np.zeros(2 * n - 1)
                lo = max(0, j)
                hi = min(2 * n - 2, j + n - 1)
                if lo <= hi:
                    smear_j[lo:hi+1] = s[lo-j:hi-j+1]
                nc = float((c + smear_i - smear_j).max())
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_max - 1e-15:
                # Apply swap
                delta = smear_i - np.zeros(2*n-1)
                lo = max(0, best_j)
                hi = min(2*n-2, best_j+n-1)
                if lo <= hi:
                    delta[lo:hi+1] -= s[lo-best_j:hi-best_j+1]
                c = c + delta
                b[i] = 1.0
                b[best_j] = 0.0
                s[i, best_j] = -s[i], -s[best_j]  # wait, s = 1-2b, so when b flips, s flips
                s[i] = -s[i]
                s[best_j] = -s[best_j]
                cur_max = best_nc
                if cur_max < best_c:
                    best_c = cur_max
                    best_b = b.copy()
                improved = True
                break
        if not improved:
            break
    return best_b, best_c * dx

def lse_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    wss = w.sum()
    w /= wss
    obj = m + (sm + np.log(wss)) / alpha
    v = 1.0 - 2.0 * x
    N = 2 * n - 1
    Wf = np.fft.rfft(w, N)
    vp = np.zeros(N)
    vp[:n] = v[::-1]
    Vf = np.fft.rfft(vp, N)
    grad = dx * np.fft.irfft(Wf * Vf, N)[:n]
    return obj, grad
