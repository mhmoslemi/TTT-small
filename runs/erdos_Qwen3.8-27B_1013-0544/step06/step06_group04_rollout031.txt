def smooth_grad(h, n, alpha, top_k=20):
    """Compute gradient of smooth max using top-k active shifts."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    # Use top-k shifts for efficiency
    top_idx = np.argsort(corr)[::-1][:top_k]
    top_c = corr[top_idx] - max_c
    exp_top = np.exp(alpha * top_c)
    w = exp_top / exp_top.sum()
    
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        if w[idx] < 1e-8:
            continue
        k = int(top_idx[idx]) - (n - 1)
        wi = w[idx]
        if k >= 0:
            if k < n:
                g[:n-k] += wi * (1.0 - h[k:])
                g[k:] -= wi * h[:n-k]
        else:
            kp = -k
            if kp < n:
                g[kp:] += wi * (1.0 - h[:n-kp])
                g[:n-kp] -= wi * h[kp:]
    return g, float(max_c * (2.0/n))

    best_h = h.copy()

    stuck_count = 0
    current_lr = lr
    
    while time.time() < deadline:
        g, c = smooth_grad(h, n, alpha, top_k)

            stuck_count = 0

            stuck_count += 1

                current_lr *= 0.9
                stuck_count = 0
                v *= 0.5

                h_new = project_feasible(h + v, n)
                h = h_new
    
    return best_h, best_c

                break
            for j_pos in range(len(ones)):
                j = ones[j_pos]
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def multi_swap_search(h, n, deadline, seed=42, max_rounds=500):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        shuffled_ones = ones[:]
        shuffled_zeros = zeros[:]
        rng.shuffle(shuffled_ones)
        rng.shuffle(shuffled_zeros)
        for a in shuffled_zeros[:60]:
            if time.time() > deadline:
                break
            for b in shuffled_zeros[:60]:
                if a == b:
                    continue
                for c in shuffled_ones[:60]:
                    if time.time() > deadline:
                        break
                    for d_pos in range(min(60, len(shuffled_ones))):
                        d_idx = shuffled_ones[d_pos]
                        if d_idx == c:
                            continue
                        nh = h.copy()
                        nh[a] = 1.0; nh[b] = 1.0
                        nh[c] = 0.0; nh[d_idx] = 0.0
                        nc = float(np.max(np.correlate(nh, 1-nh, mode='full'))) * dx
                        if nc < cur_c - 1e-15:
                            h = nh
                            corr = np.correlate(h, 1-h, mode='full')
                            cur_c = nc
                            if nc < best_c:
                                best_c = nc
                                best_h = h.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def smooth_grad(h, n, alpha, top_k=100):
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    top_idx = np.argsort(corr)[::-1][:top_k]
    top_c = corr[top_idx] - max_c
    sh = alpha * top_c
    sh = sh - sh.max()
    exp_sh = np.exp(sh)
    w = exp_sh / exp_sh.sum()
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        if w[idx] < 1e-10:
            continue
        k = int(top_idx[idx]) - (n - 1)
        wi = w[idx]
        if k >= 0:
            if k < n:
                g[:n-k] += wi * (1.0 - h[k:])
                g[k:] -= wi * h[:n-k]
        else:
            kp = -k
            if kp < n:
                g[kp:] += wi * (1.0 - h[:n-kp])
                g[:n-kp] -= wi * h[kp:]
    smooth_val = max_c + np.log(exp_sh.sum()) / alpha
    return g, float(smooth_val * (2.0/n)), float(max_c * (2.0/n))

def smooth_grad_descent(h, n, deadline, alpha=10000, lr=0.02, momentum=0.85, top_k=80):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    stuck = 0
    cur_lr = lr
    while time.time() < deadline:
        g, _, c_exact = smooth_grad(h, n, alpha, top_k)
        if c_exact < best_c - 1e-13:
            best_c = c_exact
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck % 80 == 0:
                cur_lr *= 0.85
                v *= 0.3
        v = momentum * v - cur_lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_exact:
            h = h_new
        else:
            h_try = project_feasible(h + 0.4 * v, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_exact:
                h = h_try
                v *= 0.5
            else:
                v *= 0.3
                h = project_feasible(h + 0.1 * v, n)
    return best_h, best_c

def make_binary_start(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def downsample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return h_interp

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = make_binary_start(n0, seed)
    
    def update(h, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n
    
    # Phase 1: Binary local search on n0 (25% time)
    t1 = t_start + 0.25 * (budget_s - 5)
    hb, cb = local_search_binary(h0, n0, t1, max_sweeps=5000, seed=seed)
    update(hb, cb, n0)
    
    # Phase 2: Smooth gradient on n0 (15% time)
    t2 = t_start + 0.40 * (budget_s - 5)
    if time.time() < t2 - 5:
        h_smooth = best_h.copy() if best_n == n0 else project_feasible(h0, n0)
        for alpha in [5000, 20000, 50000, 100000]:
            if time.time() > t2 - 3:
                break
            t_sub = min(t2 - 2, time.time() + (t2 - time.time()) / max(1, 4 - [5000,20000,50000,100000].index(alpha)))
            hr, cr = smooth_grad_descent(h_smooth, n0, t_sub, alpha=alpha, lr=0.03, momentum=0.85, top_k=60)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                best_n = n0
                h_smooth = hr.copy()
    
    # Phase 3: Binarize and local search again (10% time)
    t3 = t_start + 0.50 * (budget_s - 5)
    if time.time() < t3 - 2:
        hb, cb = local_search_binary(best_h, n0, t3, max_sweeps=3000, seed=seed + 999)
        update(hb, cb, n0)
    
    # Phase 4: Try n=200 (25% time)
    n_alt = 200
    t4 = t_start + 0.75 * (budget_s - 5)
    h_alt = downsample(best_h, n0, n_alt)
    h_alt = project_feasible(h_alt, n_alt)
    if time.time() < t4 - 10:
        hb2, cb2 = local_search_binary(h_alt, n_alt, min(t4 - 20, time.time() + 0.12*(budget_s-5)), max_sweeps=5000, seed=seed + 42)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            best_n = n_alt
    if time.time() < t4 - 5:
        hr2, cr2 = smooth_grad_descent(best_h if best_n == n_alt else h_alt, n_alt, t4 - 3, alpha=30000, lr=0.02, momentum=0.85, top_k=60)
        if cr2 < best_c:
            best_c = cr2
            best_h = hr2.copy()
            best_n = n_alt
    
    # Phase 5: Multi-swap on best (10% time)
    t5 = t_start + 0.88 * (budget_s - 5)
    if time.time() < t5 - 3:
        hm, cm = multi_swap_search(best_h, best_n, t5, seed=seed + 777, max_rounds=200)
        update(hm, cm, best_n)
    
    # Phase 6: Final polish (remaining time)
    if time.time() < deadline - 10:
        hb3, cb3 = local_search_binary(best_h, best_n, deadline - 8, max_sweeps=5000, seed=seed + 1234)
        update(hb3, cb3, best_n)
    
    if time.time() < deadline - 4:
        hr3, cr3 = smooth_grad_descent(best_h, best_n, deadline - 2, alpha=200000, lr=0.005, momentum=0.9, top_k=40)
        if cr3 < best_c:
            best_c = cr3
            best_h = hr3.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
