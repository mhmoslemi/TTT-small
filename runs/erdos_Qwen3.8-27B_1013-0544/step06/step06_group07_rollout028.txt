from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

        best_h = h.copy()

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def compute_deltas(h, n, zeros, ones):
    """Compute delta arrays for all zeros and ones positions."""
    ncorr = 2 * n - 1
    nz = len(zeros)
    no = len(ones)
    delta_z = np.zeros((nz, ncorr))
    delta_o = np.zeros((no, ncorr))
    for idx, i in enumerate(zeros):
        ms = max(0, i - n + 1)
        me = min(ncorr - 1, i)
        if ms <= me:
            kv = i - np.arange(ms, me + 1)
            delta_z[idx, ms:me+1] += 1 - h[kv]
        me2 = min(ncorr - 1, n - 1 - i)
        if me2 >= 0:
            kv = i + np.arange(0, me2 + 1)
            delta_z[idx, 0:me2+1] -= h[kv]
    for idx, j in enumerate(ones):
        ms = max(0, j - n + 1)
        me = min(ncorr - 1, j)
        if ms <= me:
            kv = j - np.arange(ms, me + 1)
            delta_o[idx, ms:me+1] -= 1 - h[kv]
        me2 = min(ncorr - 1, n - 1 - j)
        if me2 >= 0:
            kv = j + np.arange(0, me2 + 1)
            delta_o[idx, 0:me2+1] += h[kv]
    return delta_z, delta_o

    """Run best-improvement local search until convergence or deadline."""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c = corr.max() * dx
    best_h = h.copy()
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no = len(ones)

def iterated_local_search(h0, n, deadline, seed=42):
    """Iterated local search with perturbations."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = corr.max() * dx
    best_h = h.copy()

    perturbation_sizes = [3, 5, 7, 10, 15, 20, 30, 40, 50, 70]
    pi = 0
    
    while time.time() < deadline - 2:
        # Run best-improvement to convergence
        sub_deadline = min(time.time() + sweep_time, deadline - 2)
        h, c = best_improvement(h, n, corr, sub_deadline)
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            pi = 0  # Reset perturbation size on improvement
        else:
            pi = min(pi + 1, len(perturbation_sizes) - 1)
        
        if time.time() > deadline - 2:
            break
        
        # Perturb
        k = perturbation_sizes[pi]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for _ in range(min(k, len(zeros), len(ones))):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i] = 1.0
            h[j] = 0.0
        
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
