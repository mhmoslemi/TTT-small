import numpy as np
import time
from scipy.optimize import minimize

_IDX_CACHE = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def resample_vec(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])

def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    if y.size != n:
        y = resample_vec(y, n)
    y[~np.isfinite(y)] = 0.5
    target = 0.5 * n

    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(72):
        tau = 0.5 * (lo + hi)
        z = np.clip(y - tau, 0.0, 1.0)
        s = float(z.sum())
        if s < target:
            hi = tau
        else:
            lo = tau

    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(300):
        diff = target - float(z.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - z
            tot = float(room.sum())
            if tot < 1e-15:
                break
            z = z + (diff / tot) * room
        else:
            tot = float(z.sum())
            if tot < 1e-15:
                break
            z = z + (diff / tot) * z
        z = np.clip(z, 0.0, 1.0)

    diff = target - float(z.sum())
    if abs(diff) > 1e-12:
        idxs = np.where((z > 1e-7) & (z < 1.0 - 1e-7))[0]
        if len(idxs) > 0:
            i = int(idxs[0])
            cand = z[i] + diff
            if 0.0 <= cand <= 1.0:
                z[i] = cand
    return np.clip(z, 0.0, 1.0)

def get_indices(n):
    if n not in _IDX_CACHE:
        p = np.arange(n)
        off = n - 1
        i1 = off + p[:, None] - p[None, :]
        i2 = off + p[None, :] - p[:, None]
        _IDX_CACHE[n] = (i1, i2)
    return _IDX_CACHE[n]

def _smooth_pieces(h, alpha, c=None):
    n = len(h)
    dx = 2.0 / n
    if c is None:
        c = compute_corr(h)
    m = float(np.max(c))
    if alpha <= 0.0:
        w = np.full(c.shape, 1.0 / len(c))
        val = float(dx * float(np.mean(c)))
        return val, w, c
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        e = np.zeros_like(c)
        e[np.argmax(c)] = 1.0
        s = 1.0
    val = float(dx * (m + np.log(s) / alpha))
    w = e / s
    return val, w, c

def _grad_from_w(h, w, n):
    active = np.flatnonzero(w > 1e-12)
    active_limit = min(80, max(24, n // 4))
    if active.size > 0 and active.size <= active_limit:
        g = np.zeros(n)
        y = 1.0 - h
        off = n - 1
        for li in active:
            wi = float(w[li])
            if wi <= 0.0:
                continue
            q = int(li - off)
            if q >= 0:
                L = n - q
                if L > 0:
                    g[q:] += wi * y[:L]
                    g[:L] -= wi * h[q:]
            else:
                r = -q
                L = n - r
                if L > 0:
                    g[:L] += wi * y[r:]
                    g[r:] -= wi * h[:L]
        return g

    i1, i2 = get_indices(n)
    y = 1.0 - h
    first = w[i1] @ y
    second = w[i2] @ h
    return first - second

def _smooth_val(x, n, alpha):
    val, _, _ = _smooth_pieces(x, alpha)
    return val

def _smooth_grad(x, n, alpha):
    val, w, c = _smooth_pieces(x, alpha)
    return (2.0 / len(x)) * _grad_from_w(x, w, len(x))

def _pen_funjac(x, n, alpha, mu):
    val, w, c = _smooth_pieces(x, alpha)
    err = float(x.sum()) - 0.5 * n
    val += 0.5 * mu * err * err
    g = (2.0 / n) * _grad_from_w(x, w, n)
    if mu != 0.0 and err != 0.0:
        g = g + (mu * err) * np.ones(n)
    return float(val), g

def _pen_val(x, n, alpha, mu):
    val, _, _ = _smooth_pieces(x, alpha)
    err = float(x.sum()) - 0.5 * n
    return float(val + 0.5 * mu * err * err)

def _pen_grad(x, n, alpha, mu):
    _, w, _ = _smooth_pieces(x, alpha)
    err = float(x.sum()) - 0.5 * n
    g = (2.0 / n) * _grad_from_w(x, w, n)
    if mu != 0.0 and err != 0.0:
        g = g + (mu * err) * np.ones(n)
    return g

def _eq_fun(x, n):
    return float(x.sum() - 0.5 * n)

def _eq_jac(x, n):
    return np.ones(n)

def _bounds(n):
    return [(0.0, 1.0)] * n

def lbfgs_penalty(x, n, alpha, mu, maxiter):
    x = project_feasible(x, n)
    bnds = _bounds(n)
    opts = {
        "maxiter": int(maxiter),
        "maxfun": int(max(1000, 6 * maxiter)),
        "maxcor": 20,
        "ftol": 1e-12,
        "gtol": 1e-6,
        "disp": False,
    }
    try:
        res = minimize(_pen_funjac, x.copy(), args=(n, alpha, mu),
                       method="L-BFGS-B", bounds=bnds, jac=True, options=opts)
        x2 = res.x
    except Exception:
        try:
            res = minimize(_pen_val, x.copy(), args=(n, alpha, mu),
                           method="L-BFGS-B", bounds=bnds, jac=_pen_grad, options=opts)
            x2 = res.x
        except Exception:
            return x
    if x2 is None or not np.all(np.isfinite(x2)):
        return x
    return project_feasible(x2, n)

def slsqp_smooth(x, n, alpha, maxiter):
    x = project_feasible(x, n)
    cons = ({
        "type": "eq",
        "fun": _eq_fun,
        "jac": _eq_jac,
        "args": (n,),
    },)
    opts = {"maxiter": int(maxiter), "ftol": 1e-12}
    try:
        res = minimize(_smooth_val, x.copy(), args=(n, alpha),
                       method="SLSQP", bounds=_bounds(n),
                       jac=_smooth_grad, constraints=cons, options=opts)
        x2 = res.x
    except Exception:
        return x
    if x2 is None or not np.all(np.isfinite(x2)):
        return x
    return project_feasible(x2, n)

def steepest_refine(x, n, limit, alpha=20000.0, maxit=60):
    x = project_feasible(x, n)
    dx = 2.0 / n
    val, w, c = _smooth_pieces(x, alpha)
    f = float(np.max(c) * dx)
    x_prev = x.copy()
    g_prev = None
    step = 0.004
    no_imp = 0

    for _ in range(int(maxit)):
        if time.time() > limit:
            break
        val, w, c = _smooth_pieces(x, alpha)
        f = float(np.max(c) * dx)
        gd = dx * _grad_from_w(x, w, n)
        direction = -(gd - float(np.mean(gd)))
        ng = float(np.linalg.norm(direction))
        if ng < 1e-11:
            break

        if g_prev is not None:
            dx_old = x - x_prev
            dg = gd - g_prev
            den = float(np.dot(dg, dg))
            num = float(np.dot(dx_old, dx_old))
            if den > 1e-18:
                step = 0.55 * step + 0.45 * (num / den)
                step = min(0.12, max(1e-6, step))

        x_prev = x.copy()
        g_prev = gd
        improved = False
        for mult in (1.0, 0.5, 0.25):
            y = project_feasible(x + step * mult * direction, n)
            cy = float(np.max(compute_corr(y)) * dx)
            if cy < f - 1e-12:
                x = y
                f = cy
                improved = True
                no_imp = 0
