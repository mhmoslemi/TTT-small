def thue_morse(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)
