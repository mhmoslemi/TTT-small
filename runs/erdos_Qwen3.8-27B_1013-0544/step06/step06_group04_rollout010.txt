    """Delta in correlation when swapping positions i and j (h[i]→h[j], h[j]→h[i]).
    For binary: h[i]=1, h[j]=0 → h[i]=0, h[j]=1."""
    d = np.zeros(2*n-1)
    # Position i changes from 1 to 0:
    # In c[k] = sum_i h[i]*(1-h[i+k]):
    #   Terms where i is first index: contribution 1*(1-h[i+k]) = 1-h[i+k] (removed)
    #   Terms where i is second index: contribution h[i-k]*(1-1) = 0 (no change)
    #   Wait, (1-h[i]) was the second factor. If h[i] goes from 1 to 0:
    #   Second factor changes from (1-1)=0 to (1-0)=1
    #   So terms where i is second index: h[i-k]*1 instead of h[i-k]*0 → adds h[i-k]
    
    # Let me be more careful. c_full[m] = sum_i h[i] * (1-h)[i + m - (n-1)]
    # Let b = 1-h. Then c_full[m] = sum_i h[i] * b[i + m - (n-1)]
    # For m = n-1+k: c_full[n-1+k] = sum_{valid i} h[i] * b[i+k]
    
    # Change in h at position i (from 1 to 0) and b at position i (from 0 to 1):
    #   c_full[n-1+k] changes by:
    #     - old term where i is first index: h[i]*b[i+k] = 1*b[i+k] = b[i+k] (removed, so -b[i+k])
    #     + new term where i is second index: h[i-k]*b[i] = h[i-k]*1 = h[i-k] (added, so +h[i-k])
    #   Valid when i+k < n (for first) and i-k >= 0 (for second)
    
    # Similarly for position j (from 0 to 1), b changes from 1 to 0:
    #   c_full[n-1+k] changes by:
    #     + new term where j is first index: h[j]*b[j+k] = 0*... wait h[j] was 0, now 1
    #     Actually: old h[j]=0, new h[j]=1; old b[j]=1, new b[j]=0
    #     Term where j is first index: old: 0*b[j+k]=0, new: 1*b[j+k]=b[j+k]. Change: +b[j+k]
    #     Term where j is second index: old: h[j-k]*1=h[j-k], new: h[j-k]*0=0. Change: -h[j-k]
    
    # Total change for swap(i,j) where h[i]=1,h[j]=0:
    # At lag k (c_full[n-1+k]):
    #   From i: -b[i+k] (if i+k<n) + h[i-k] (if i-k>=0)
    #   From j: +b[j+k] (if j+k<n) - h[j-k] (if j-k>=0)
    
    for k in range(-(n-1), n):
        delta = 0.0
        # From position i (1→0):
        if i + k >= 0 and i + k < n:
            delta -= (1 - h[i+k])  # -b[i+k]
        if i - k >= 0 and i - k < n:
            delta += h[i-k]  # +h[i-k]
        # From position j (0→1):
        if j + k >= 0 and j + k < n:
            delta += (1 - h[j+k])  # +b[j+k]
        if j - k >= 0 and j - k < n:
            delta -= h[j-k]  # -h[j-k]
        d[n-1+k] = delta
    return d

    # Log-sum-exp for numerical stability
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    log_sum = np.log(exp_c.sum()) + alpha * max_c
    f = log_sum / alpha
    
    # Gradient
    # w_k = exp(alpha * corr_k) / sum(exp(alpha * corr))
    w = exp_c / exp_c.sum()
    
    # grad[i] = sum_k w_k * (1-h[i+k])_valid - h[i-k]_valid
    grad = np.zeros(n)
    for k in range(-(n-1), n):

    def obj(x):
        corr = np.correlate(x, 1-x, mode='full') * dx
        max_c = corr.max()
        if max_c < -700/alpha:
            return alpha * max_c + np.log(corr.sum() * dx)
        exp_c = np.exp(alpha * (corr - max_c))
        return max_c + np.log(exp_c.sum()) / alpha

        max_c = corr.max()

        w = exp_c / exp_c.sum()

        for k in range(-(n-1), n):
            wk = w[n-1+k] * dx
            if wk < 1e-12:

            hi2 = min(n-1, n-1+k) + 1
            if lo2 < hi2:
                g[lo2:hi2] -= wk * x[lo2-k:hi2-k]
        return g
    
    # Use SLSQP for equality constraint
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    result = minimize(obj, h, jac=grad, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
