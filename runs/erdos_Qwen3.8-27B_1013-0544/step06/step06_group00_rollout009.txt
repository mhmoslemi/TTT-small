import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optim(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n, deadline, n_iter=5000, top_k=20):
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    lr = 0.001
    decay = 0.999
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        m = corr.max()
        
        # Find active set
        idx = np.argsort(corr)[::-1][:top_k]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in idx:
            k = k_idx - (n - 1)
            if k == 0:
                grad += dx - 2.0 * dx * h
            else:
                # dg_k/dh[j] = dx - dx*h[j+k] (if j+k<n) - dx*h[j-k] (if j-k>=0)
                grad += dx
                if k > 0:
                    grad[:-k] -= dx * h[k:]
                    grad[k:] -= dx * h[:-k]
                else:  # k < 0
                    kk = -k
                    grad[kk:] -= dx * h[:-kk]
                    grad[:-kk] -= dx * h[kk:]
        
        grad /= len(idx)
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still update h (for momentum-like effect)
            # or keep h if not improved (more conservative)
        
        lr *= decay

    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

class ObjFunc:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha
        self.dx = 2.0 / n
    def __call__(self, x):
        g = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = g.max()
        shifted = self.alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / self.alpha

class ConsFunc:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return x.sum() - self.n / 2.0

def slsqp_optim(h, n, alpha, maxiter=5000, ftol=1e-16):
    obj = ObjFunc(n, alpha)
    cons = [{'type': 'eq', 'fun': ConsFunc(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_gain = 1e-15
            best_j = -1
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max()
                if nc < cur_c - best_gain:  # Hmm, this logic is wrong
                    best_gain = cur_c - nc
                    best_j = j
            # Actually let me simplify: just find the first improving move
            # (or the best one)
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                # Recompute delta for the actual move
                delta = np.zeros(2 * n - 1)
                j = best_j
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]  # Wait, h is already updated!
                # This is getting complicated. Let me just recompute correlation.
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def subgradient_polish(h, n, deadline, top_k=30, n_max_iter=200000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    lr = 0.005
    decay = 0.9999
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        if no_improve > 5000:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-8:
                break
        
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        m = corr.max()
        
        # Active set: top top_k values
        idx = np.argsort(corr)[::-1][:top_k]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in idx:
            k = k_idx - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * h) * dx
            elif k > 0:
                # dg/dh[j] = (1-h[j-k])*[k<=j] - h[j+k]*[j <= n-1-k]
                grad[k:] += (1.0 - h[:-k]) * dx  # (1-h[j-k]) for j=k to n-1
                grad[:n-k] -= h[k:] * dx  # -h[j+k] for j=0 to n-1-k
            else:
                kk = -k
                # dg/dh[j] = (1-h[j+kk])*[0<=j<n-kk] - h[j-kk]*[kk<=j<n]
                grad[:n-kk] += (1.0 - h[kk:]) * dx
                grad[kk:] -= h[:-kk] * dx
        
        grad /= len(idx)
        
        # Momentum update
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
        
        lr *= decay
    
    return best_h, best_c
