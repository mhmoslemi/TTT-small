def compute_all_lags(h, n):

    hi = h.max() + 1.0
    for _ in range(80):

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

    corr = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    vals = corr[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    sh = np.clip(sh, -500, 500)
    ms = sh.max()
    return m + (ms + np.log(np.sum(np.exp(sh - ms)))) / alpha

def _obj_smooth_grad(x, n, alpha, idx):

    corr = np.correlate(x, 1 - x, mode='full') * dx
    vals = corr[idx]
    m = vals.max()

    ms = sh.max()
    e = np.exp(sh - ms)
    w = e / e.sum()

    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] < 1e-12:

    return np.sum(x) - n / 2.0

    return np.ones(n)

        return h.copy(), compute_c5(h, n)
