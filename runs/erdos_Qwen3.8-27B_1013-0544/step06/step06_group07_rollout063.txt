import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_t(d, c):
    tol = 1e-12
    f0 = d.max()
    active0 = np.where(d >= f0 - tol)[0]
    if len(active0) > 0 and c[active0].min() >= -tol:
        return 0.0
    f1 = (d + c).max()
    active1 = np.where(d + c >= f1 - tol)[0]
    if len(active1) > 0 and c[active1].max() <= tol:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        val = d + mid * c
        mx = val.max()
        active = np.where(val >= mx - tol)[0]
        if len(active) == 0:
            active = np.where(val == mx)[0]
        mc = c[active].min()
        Mc = c[active].max()
        if mc >= -tol:
            hi = mid
        elif Mc <= tol:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coordinate_descent(h, n, deadline, max_sweeps=200000):
    rng = np.random.RandomState(int(time.time()) % 100000)
    h = h.astype(float).copy()
    C = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = C.max()
    best_max = cur_max
    best_h = h.copy()
    target = n / 2.0
    n_sweeps_done = 0
    while time.time() < deadline:
        any_change = False
        order = np.arange(n)
        for i in range(n):
            n_sweeps_done += 1
            if n_sweeps_done > max_sweeps:
                break
            ii = order[i]
            delta = np.zeros(2 * n - 1)
            delta[n - 1 - ii:2 * n - 1 - ii] += 1.0 - h
            delta[ii:n + ii] -= h[::-1]
            d = C - h[ii] * delta
            t_new = find_optimal_t(d, delta)
            if abs(t_new - h[ii]) > 1e-12:
                C_new = C + (t_new - h[ii]) * delta
                C_new[n - 1] = C[n - 1] - h[ii] + h[ii] * h[ii] + t_new - t_new * t_new
                if C_new.max() <= cur_max + 1e-11:
                    C = C_new
                    h[ii] = t_new
                    cur_max = C.max()
                    any_change = True
        if n_sweeps_done > max_sweeps:
            break
        if not any_change:
            h = project_feasible(h, n)
            C = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = C.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            h = project_feasible(h + 0.03 * rng.randn(n), n)
            C = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = C.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            continue
        if n_sweeps_done % 500 < n:
            h = project_feasible(h, n)
            C = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = C.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    true_C = np.correlate(h, 1 - h, mode='full')
    if true_C.max() < best_max:
        best_max = true_C.max()
        best_h = h.copy()
    return best_h, float(best_max)

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00001):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 400000.0)
    T = t_start
    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > 50000:
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 4.0)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out
