```python
def simulated_annealing(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    h = h_init.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    corr = np.correlate(h, 1-h, 'full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    T = 5.0  # Initial temperature
    T_min = 0.01
    cooling = 0.999
    
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline and T > T_min:
        # Pick random 1 and random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1-h, 'full')
        new_max = new_corr.max()
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    return best_h, best_max * dx
```

```python
def iterative_reweight(h, n, deadline, n_iter=100):
    dx = 2.0 / n
    h_cur = h.copy()
    weights = np.ones(2*n-1) / (2*n-1)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute correlation
        corr = np.correlate(h_cur, 1-h_cur, 'full') * dx
        # Update weights (multiplicative)
        w_max = corr.max()
        eta = 0.1
        weights *= np.exp(eta * corr)
        weights /= weights.sum()
        
        # Optimize weighted sum
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            return float(np.dot(weights, g))
        
        def cons_fun(x):
            return x.sum() - n / 2.0

        except:
            pass
    
    return h_cur
```
