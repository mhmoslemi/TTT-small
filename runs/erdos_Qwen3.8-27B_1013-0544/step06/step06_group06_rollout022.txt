            if len(zeros) == 0 or len(ones) == 0:
                break
            z_pos = zeros[:, None] + offsets[None, :]
            o_pos = ones[:, None] + offsets[None, :]
            z_valid = (z_pos >= 0) & (z_pos < n)
            o_valid = (o_pos >= 0) & (o_pos < n)
            z_safe = np.clip(z_pos, 0, n - 1)
            o_safe = np.clip(o_pos, 0, n - 1)
            z_contrib = np.where(z_valid, 1.0 - h[z_safe], 0.0)
            o_contrib = np.where(o_valid, 1.0 - h[o_safe], 0.0)
            z_sums = z_contrib.sum(axis=1)
            o_sums = o_contrib.sum(axis=1)
            obj = z_sums[:, None] - o_sums[None, :]
            M = min(n_cand, obj.size)
            flat_idx = np.argsort(obj, axis=None)[:M]
            best_c_iter = cur_c + 1e-14
            best_swap = None
            for fi in range(M):
                ii = int(flat_idx[fi] // len(ones))
                jj = int(flat_idx[fi] % len(ones))
                i = zeros[ii]
                j = ones[jj]
                h[i] = 1.0
                h[j] = 0.0
                new_corr = get_corr(h)
                c_new = new_corr.max() * dx
                if c_new < best_c_iter:
                    best_c_iter = c_new
                    best_swap = (ii, jj, new_corr.copy())
                h[i] = 0.0
                h[j] = 1.0
            if best_swap is not None:
                ii, jj, new_corr = best_swap
                i = zeros[ii]
                j = ones[jj]
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_c = best_c_iter
                zeros = np.delete(zeros, ii)
                ones = np.delete(ones, jj)
                zeros = np.append(zeros, j)
                ones = np.append(ones, i)
                stall = 0
                if cur_c < best_c - 1e-14:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                stall += 1
                if no_improve > 100:
                    break
                no_improve += 1
        max_restarts -= 1
        if time.time() >= deadline:
            break
        nk = min(kick_size, len(zeros), len(ones))
        ri = rng.choice(len(zeros), nk, replace=False)
        rj = rng.choice(len(ones), nk, replace=False)
        h[zeros[ri]] = 1.0
        h[ones[rj]] = 0.0
        corr = get_corr(h)
        cur_c = corr.max() * dx
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def round_continuous(hc, n):
    h_bin = (hc >= 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        ones_idx = np.where(h_bin == 1)[0]
        flip = ones_idx[np.argsort(hc[ones_idx])[:cnt - target]]
        h_bin[flip] = 0.0
    elif cnt < target:
        zeros_idx = np.where(h_bin == 0)[0]
        flip = zeros_idx[np.argsort(hc[zeros_idx])[::-1][:target - cnt]]
        h_bin[flip] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: ILS from parent (40% time)
    p1 = t0 + 0.40 * (budget_s - 5)
    hb, cb = ils_local_search(h0, n, p1, seed=seed, top_k=5, n_cand=50, max_restarts=40, kick_size=3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Thue-Morse ILS (15% time)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2:
        h_tm = thue_morse_init(n, rng)
        hb, cb = ils_local_search(h_tm, n, p2, seed=seed + 200, top_k=5, n_cand=50, max_restarts=30, kick_size=3)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 3: SLSQP continuous polish (15% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3:
        h_cont = project_feasible(best_h + 0.01 * rng.randn(n), n)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p3 - 5:
                break
            h_cont = slsqp_smooth(h_cont, n, alpha, 800, p3 - 3)
            h_bin = round_continuous(h_cont, n)
            c_bin = compute_c5(h_bin, n)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()
            # Local perturb around rounding
            for _ in range(15):
                h_p = best_h.copy()
                fi = rng.choice(n)
                fj = rng.choice(n)
                if h_p[fi] == 0 and h_p[fj] == 1:
                    h_p[fi] = 1.0
                    h_p[fj] = 0.0
                    cp = compute_c5(h_p, n)
                    if cp < best_c:
                        best_c = cp
                        best_h = h_p.copy()
    
    # Phase 4: Extended ILS with different top_k (15% time)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        hb, cb = ils_local_search(best_h, n, p4, seed=seed + 500, top_k=3, n_cand=60, max_restarts=50, kick_size=2)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 5: Final polish with top_k=8
    if time.time() < dl - 10:
        hb, cb = ils_local_search(best_h, n, dl - 10, seed=seed + 800, top_k=8, n_cand=40, max_restarts=20, kick_size=2)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 6: Final SLSQP micro-polish
    if time.time() < dl - 3:
        hf = project_feasible(best_h + 0.005 * rng.randn(n), n)
        hf = slsqp_smooth(hf, n, 3000000, 400, dl - 2)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
