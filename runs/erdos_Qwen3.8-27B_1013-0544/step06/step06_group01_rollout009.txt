import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_fun(h, n, alpha, pen_w):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = (corr - m) * alpha
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if pen_w > 0:
        val += pen_w * (h.sum() - n / 2.0) ** 2
    return val

def smooth_grad_fun(h, n, alpha, pen_w):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = (corr - m) * alpha
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            grad[offset:] += w * (1.0 - h[:n - offset])
            grad[:n - offset] -= w * h[offset:]
        else:
            oa = -offset
            grad[:n - oa] += w * (1.0 - h[oa:])
            grad[oa:] -= w * h[:n - oa]
    
    if pen_w > 0:
        grad += 2.0 * pen_w * (h.sum() - n / 2.0)
    
    return grad

def run_lbfgs(h0, n, alpha_schedule, pen_w, deadline):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_schedule:
        if time.time() >= deadline:
            break
        res = minimize(smooth_obj_fun, h, jac=smooth_grad_fun, args=(n, alpha, pen_w),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 100000})
        h = project_feasible(res.x, n)
    return h
