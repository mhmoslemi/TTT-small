def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-12:

        if diff > 0:

            tot = room.sum()

            if tot < 1e-15:

            h += (diff / tot) * h

        # Update weights

def slsqp_step(h, n, alpha=100000):

    dx = 2.0 / n

        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
        return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

                    new_ca[idx_n1] -= d * d
                val = max(um, float(np.max(new_ca))) * dx
                val += 0.05 * (h.sum() + d - n / 2.0) ** 2
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                c[ks] += best_d * coeff
                idx_n1 = np.where(ks == n - 1)[0]
                if len(idx_n1) > 0:
                    c[n - 1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                cur = max(um, float(np.max(c))) * dx
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def slsqp_step(h, n, alpha=200000, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
        return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Phase 1: MW with continuation (main workhorse)
    sched1 = [(2.0, 30.0, 400), (5.0, 80.0, 400), (10.0, 150.0, 300)]
    for bs, be, ni in sched1:
        if time.time() > deadline - 60:
            break
        sub_dl = time.time() + min(350, (deadline - time.time()) * 0.35)
        h_mw, c_mw = mw_optimize(best_h, n, sub_dl, beta_start=bs, beta_end=be, n_iter=ni)
        if c_mw < best_c:
            best_c, best_h = c_mw, h_mw

    # Phase 2: CD polish
    if time.time() < deadline - 40:
        h_cd, c_cd = cd_pass(best_h, n, min(deadline - 35, time.time() + 200), max_passes=200, grid=100)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd

    # Phase 3: SLSQP
    if time.time() < deadline - 30:
        h_sl = slsqp_step(best_h, n, alpha=300000, maxiter=3000)
        c_sl = compute_c5(h_sl, n)
        if c_sl < best_c:
            best_c, best_h = c_sl, h_sl

    # Phase 4: MW with top-k focus
    if time.time() < deadline - 40:
        for tk in [20, 50, 100]:
            if time.time() > deadline - 30:
                break
            sub_dl = time.time() + min(120, (deadline - time.time()) * 0.2)
            h_mw2, c_mw2 = mw_optimize(best_h, n, sub_dl, beta_start=5.0, beta_end=100.0, n_iter=200, top_k=tk)
            if c_mw2 < best_c:
                best_c, best_h = c_mw2, h_mw2

    # Phase 5: Another CD round
    if time.time() < deadline - 20:
        h_cd2, c_cd2 = cd_pass(best_h, n, min(deadline - 15, time.time() + 120), max_passes=150, grid=120)
        if c_cd2 < best_c:
            best_c, best_h = c_cd2, h_cd2

    # Phase 6: Final SLSQP + MW + CD
    if time.time() < deadline - 15:
        h_sl2 = slsqp_step(best_h, n, alpha=800000, maxiter=2000)
        c_sl2 = compute_c5(h_sl2, n)
        if c_sl2 < best_c:
            best_c, best_h = c_sl2, h_sl2
    if time.time() < deadline - 8:
        h_mw3, c_mw3 = mw_optimize(best_h, n, deadline - 5, beta_start=20.0, beta_end=200.0, n_iter=200)
        if c_mw3 < best_c:
            best_c, best_h = c_mw3, h_mw3
    if time.time() < deadline - 3:
        h_cd3, c_cd3 = cd_pass(best_h, n, deadline - 1, max_passes=100, grid=150)
        if c_cd3 < best_c:
            best_c, best_h = c_cd3, h_cd3

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
