def project_box_sum(v, target):
   v=np.asarray(v,float).copy()
   if v.size==0: return v
   lo=float(v.min()) - 1.0
   hi=float(v.max()) + 1.0
   for _ in range(70):
      mid=0.5*(lo+hi)
      s=np.clip(v-mid,0.0,1.0).sum()
      if s > target: lo=mid
      else: hi=mid
   h=np.clip(v-0.5*(lo+hi),0.0,1.0)
   for _ in range(10):
      diff=target - h.sum()
      if abs(diff) < 1e-15: break
      if diff >0:
         idx=int(np.argmax(1.0-h))
         inc=min(diff, 1.0-h[idx])
         if inc <=0: break
         h[idx]+=inc
      else:
         idx=int(np.argmin(h))
         dec=min(-diff, h[idx])
         if dec <=0: break
         h[idx]-=dec

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def raw_max(h):
    return float(np.max(raw_corr(h)))

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).copy()
    if v.size == 0:
        return v
    bad = ~np.isfinite(v)
    if np.any(bad):
        v[bad] = 0.5
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        diff = target - float(h.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0.0:
                break
            h[idx] += inc
        else:
            idx = int(np.argmin(h))
            dec = min(-diff, h[idx])
            if dec <= 0.0:
                break
            h[idx] -= dec
    return h

import numpy as np
from scipy.optimize import minimize
import time, math

_STATE = {'alpha': 1.0, 'target': 0.0}

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def raw_max(h):
    return float(np.max(raw_corr(h)))

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).copy()
    if v.size == 0:
        return v
    bad = ~np.isfinite(v)
    if np.any(bad):
        v[bad] = 0.5
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        diff = target - float(h.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0.0:
                break
            h[idx] += inc
        else:
            idx = int(np.argmin(h))
            dec = min(-diff, h[idx])
            if dec <= 0.0:
                break
            h[idx] -= dec
    return h

def weighted_grad(h, p):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    off = n - 1
    for t in range(2 * n - 1):
        c = float(p[t])
        if c <= 1e-14:
            continue
        k = t - off
        if k == 0:
            grad += c * (1.0 - 2.0 * h)
        elif k > 0:
            b = n - k
            grad[:b] += c * (1.0 - h[k:])
            grad[k:] -= c * h[:b]
        else:
            r = -k
            b = n - r
            grad[r:] += c * (1.0 - h[:b])
            grad[:b] -= c * h[r:]
    return grad

def smooth_raw(h, alpha):
    raw = raw_corr(h)
    m = float(raw.max())
    z = alpha * (raw - m)
    e = np.exp(z)
    s = float(e.sum())
    val = m + math.log(s) / alpha
    p = e / s
    return val, weighted_grad(h, p)

def con_fun(x):
    return float(np.sum(x) - _STATE['target'])

def con_jac(x):
    return np.ones_like(x, dtype=float)

def obj_fun(x):
    return smooth_raw(x, float(_STATE['alpha']))[0]

def obj_jac(x):
    return smooth_raw(x, float(_STATE['alpha']))[1]

def slsqp_step(x0, n, alpha, maxiter):
    target = 0.5 * float(n)
    _STATE['alpha'] = float(alpha)
    _STATE['target'] = target
    x0 = project_box_sum(x0, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(
            obj_fun, x0, jac=obj_jac, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'iprint': 0}
        )
        return project_box_sum(res.x, target)
    except Exception:
        return x0

def continuous_polish(h_in, n, deadline, alist, seed):
    target = 0.5 * float(n)
    h0 = project_box_sum(h_in, target)
    best_h = h0.copy()
    best_r = raw_max(best_h)
    rng = np.random.RandomState(seed)
    starts = [h0]
    if deadline - time.time() > 5.0:
        starts.append(project_box_sum(h0 + 0.05 * rng.normal(size=n), target))
    for st in starts:
        if time.time() > deadline:
            break
        h = st.copy()
        for alpha, mit in alist:
            if time.time() > deadline:
                break
            x = slsqp_step(h, n, alpha, mit)
            r = raw_max(x)
            if r < best_r - 1e-12:
                best_r = r
                best_h = x.copy()
            h = best_h if r > best_r + 1e-6 else x
    return best_h, best_r

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n, dtype=float)
    out[idx[:target]] = 1.0
    return out

def random_continuous(n, seed):
    return project_box_sum(np.random.RandomState(seed).rand(n), 0.5 * n)

def thue_morse_binary(n, seed):
    L = 1
    while L < n:
        L *= 2
    s = [0]
    l = 1
    while l < L:
        for i in range(l):
            s.append(1 - s[i])
        l *= 2
    arr = np.array(s[:n], dtype=float)
    target = n // 2
    cnt = int(arr.sum())
    rng = np.random.RandomState(seed)
    if cnt < target:
        idx = np.where(arr == 0)[0]
        rng.shuffle(idx)
        arr[idx[:target - cnt]] = 1.0
    else:
        idx = np.where(arr == 1)[0]
        rng.shuffle(idx)
        arr[idx[:cnt - target]] = 0.0
    return arr

def kick_binary(h, n, k, seed):
    h = round_to_binary(h, n).copy()
    rng = np.random.RandomState(seed)
    for _ in range(k):
        z = np.where(h == 0)[0]
        o = np.where(h == 1)[0]
        if len(z) == 0 or len(o) == 0:
            break
        a = z[rng.randint(len(z))]
        b = o[rng.randint(len(o))]
        h[a] = 1.0
        h[b] = 0.0
    return h

def fill_swap_work(h, omh, i, j, work, n):
    work.fill(0.0)
    work[n - 1 - i:2 * n - 1 - i] += omh
    work[i:i + n] -= h[::-1]
    work[n - 1 - j:2 * n - 1 - j] -= omh
    work[j:j + n] += h[::-1]
    work[n - 1] -= 2.0
    work[n - 1 + j - i] += 1.0
    work[n - 1 + i - j] += 1.0

def init_sa(h, n):
    zeros = []
    ones = []
    pos0 = np.full(n, -1, dtype=int)
    pos1 = np.full(n, -1, dtype=int)
    for i in range(n):
        if h[i] < 0.5:
            pos0[i] = len(zeros)
            zeros.append(i)
        else:
            pos1[i] = len(ones)
            ones.append(i)
    return zeros, pos0, ones, pos1

def sa_apply(h, omh, i, j, zeros, pos0, ones, pos1):
    p0 = pos0[i]
    last = zeros[-1]
    zeros[p0] = last
    pos0[last] = p0
    zeros.pop()
    pos0[i] = -1

    p1 = pos1[j]
    last = ones[-1]
    ones[p1] = last
    pos1[last] = p1
    ones.pop()
    pos1[j] = -1

    ones.append(i)
    pos1[i] = len(ones) - 1
    zeros.append(j)
    pos0[j] = len(zeros) - 1

    h[i] = 1.0
    h[j] = 0.0
    omh[i] = 0.0
    omh[j] = 1.0

def binary_sa(start, n, deadline, seed, t_start=3.0, t_end=0.002, patience=20000):
    h = round_to_binary(start, n).copy()
    omh = 1.0 - h
    zeros, pos0, ones, pos1 = init_sa(h, n)
    work = np.zeros(2 * n - 1, dtype=float)
    corr = raw_corr(h).astype(float)
    cur = float(corr.max())
    best = cur
    best_h = h.copy()
    no_improve = 0
    step = 0
    cool = (t_end / max(1e-12, t_start)) ** (1.0 / 500000.0)
    T = float(t_start)
    rng = np.random.RandomState(seed)

    while time.time() < deadline:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        fill_swap_work(h, omh, i, j, work, n)
        arr = corr + work
        nc = float(arr.max())
        dc = nc - cur
        if dc < 0.0 or (T > 1e-12 and rng.random() < math.exp(-dc / T)):
            corr = arr
            cur = nc
            sa_apply(h, omh, i, j, zeros, pos0, ones, pos1)
            if nc < best - 1e-12:
                best = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        step += 1
        if step < 500000:
            T *= cool
        else:
            T = t_end

        if no_improve > patience:
            if rng.random() < 0.5:
                h = best_h.copy()
                omh = 1.0 - h
                corr = raw_corr(h).astype(float)
                cur = float(corr.max())
                zeros, pos0, ones, pos1 = init_sa(h, n)
            nk = 5 + rng.randint(10)
            for _ in range(nk):
                z = np.where(h == 0)[0]
                o = np.where(h == 1)[0]
                if len(z) == 0 or len(o) == 0:
                    break
                a = z[rng.randint(len(z))]
                b = o[rng.randint(len(o))]
                h[a] = 1.0
                h[b] = 0.0
            omh = 1.0 - h
            corr = raw_corr(h).astype(float)
            cur = float(corr.max())
            zeros, pos0, ones, pos1 = init_sa(h, n)
            no_improve = 0
            if cur < best:
                best = cur
                best_h = h.copy()
            T = min(t_start, T * 4.0)

    return best_h, raw_max(best_h)

def binary_bundle(h_in, n, deadline, seed):
    if n % 2 != 0 or n <= 0:
        return h_in, raw_max(h_in)
    best_h = round_to_binary(h_in, n)
    best_r = raw_max(best_h)
    rem = deadline - time.time()
    if rem < 5.0:
        return best_h, best_r

    rng = np.random.RandomState(seed)
