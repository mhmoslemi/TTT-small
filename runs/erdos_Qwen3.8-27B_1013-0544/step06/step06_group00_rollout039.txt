def project_sum(v,target,iters=200):
    v=np.clip(v.astype(float),0,1)
    for _ in range(iters):
       diff=target-v.sum()
       if abs(diff)<1e-13: break
       if diff>0:
          room=1-v; s=room.sum()
          if s<1e-14: break
          v += diff/s*room
       else:
          s=v.sum()
          if s<1e-14: break
          v += diff/s*v
    diff=target-v.sum()
    if abs(diff)>1e-12:
       idxs=np.flatnonzero((v>1e-12)&(v<1-1e-12))
       if idxs.size>0:
          i=idxs[0]
          v[i]=clip(v[i]+diff)
       else:
          # distribute over all
          if diff>0:
             i=int(np.argmax(1-v)); v[i]=min(1,v[i]+diff)
          else:
             i=int(np.argmax(v)); v[i]=max(0,v[i]+diff)
    return np.clip(v,0,1)

import numpy as np
import time
from scipy.optimize import minimize

STATE = {"alpha": 1.0, "n": 0, "sym": False}
_LAG_CACHE = {}
MAX_ACTIVE = 400

def project_sum(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - v.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - v
            s = room.sum()
            if s < 1e-14:
                break
            v += (diff / s) * room
        else:
            s = v.sum()
            if s < 1e-14:
                break
            v += (diff / s) * v
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        idxs = np.flatnonzero((v > 1e-12) & (v < 1.0 - 1e-12))
        i = int(idxs[0]) if idxs.size else int(np.argmax(1.0 - v if diff > 0 else v))
        v[i] = min(1.0, max(0.0, v[i] + diff))
    return np.clip(v, 0.0, 1.0)

def make_feasible(h, n):
    return project_sum(h, 0.5 * n)

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def expand_x(x):
    x = np.asarray(x, dtype=float)
    if STATE["sym"]:
        return np.concatenate((x, x[::-1]))
    return x

def lag_info(n):
    if n not in _LAG_CACHE:
        info = []
        for idx in range(2 * n - 1):
            l = idx - (n - 1)
            L = n - abs(l)
            s1 = max(0, l)
            s2 = max(0, -l)
            p1 = max(0, -l)
            p2 = max(0, l)
            info.append((
                slice(s1, s1 + L), slice(p1, p1 + L),
                slice(s2, s2 + L), slice(p2, p2 + L)
            ))
        _LAG_CACHE[n] = info
    return _LAG_CACHE[n]

def softmax_data(c, alpha):
    cm = float(np.max(c))
    sh = alpha * (c - cm)
    sh = np.clip(sh, -50.0, 0.0)
    e = np.exp(sh)
    tot = float(e.sum())

        e = np.zeros_like(e)
        e[int(np.argmax(c))] = 1.0
        tot = 1.0
    w = e / tot
    f = cm + np.log(tot) / alpha
    return w, float(f)

                xs, n0, True,
                [10.0, 40.0, 150.0, 600.0, 2500.0, 10000.0, 40000.0, 150000.0, 600000.0],
                [80, 100, 120, 150, 170, 200, 230, 260, 300],
                p1
            )
            if cs < best_c:
                best_c = cs
                best_h = hs
                best_n = n0

    base_full = h0 if best_h is h0 else best_h
    if len(base_full) != n0:
        base_full = resample(base_full, n0)
    base_full = make_feasible(base_full, n0)

    if time.time() < t0 + 0.68 * usable:
        _, hf, cf = run_smooth(
            base_full, n0, False,
            [30.0, 120.0, 500.0, 2000.0, 8000.0, 30000.0, 120000.0, 500000.0],
            [100, 130, 160, 190, 220, 250, 280, 320],
            t0 + 0.68 * usable
        )
        if cf < best_c:
            best_c = cf
            best_h = hf
            best_n = n0

    if time.time() < t0 + 0.69 * usable:
        second = h0 if base_full is not h0 else base_full.copy()
        _, hf, cf = run_smooth(
            second, n0, False,
            [1000.0, 5000.0, 20000.0, 100000.0, 400000.0],
            [120, 150, 180, 220, 260],
            t0 + 0.82 * usable
        )
        if cf < best_c:
            best_c = cf
            best_h = hf
            best_n = n0

    for nn in (500, 300):
        if time.time() > t_end - 130.0:
            break
        hh = resample(best_h, nn)
        hh = make_feasible(hh, nn)
        if nn % 2 == 0:
            xx = to_sym_x(hh)
            xx = project_sum(xx, 0.25 * nn)
            _, hh, ch = run_smooth(
                xx, nn, True,
                [50.0, 300.0, 1500.0, 8000.0, 40000.0, 150000.0],
                [110, 140, 170, 200, 230, 260],
                t_end - 90.0
            )
        else:
            hh, ch = hh, eval_c5(hh)
        if ch < best_c:
            best_c = ch
            best_h = hh
            best_n = nn

    if time.time() < t_end - 50.0:
        hfin = best_h if len(best_h) == best_n else resample(best_h, best_n)
        hfin = make_feasible(hfin, best_n)
        _, hf, cf = run_smooth(
            hfin, best_n, False,
            [100000.0, 500000.0, 2000000.0],
            [200, 260, 320],
            t_end - 8.0
        )
        if cf < best_c:
            best_c = cf
            best_h = hf
            best_n = len(best_h)

    best_h = make_feasible(best_h, best_n)
    best_c = eval_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
