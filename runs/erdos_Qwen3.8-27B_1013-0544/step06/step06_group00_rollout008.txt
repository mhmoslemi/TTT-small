```python
def compute_grad(h, n, alpha):
    """Compute gradient of log-sum-exp objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx  # length 2n-1
    # softmax
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / exp_c.sum()
    # gradient: for each i, grad[i] = sum_s p[s] * dc[s]/dh[i]
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)  # shifts
    for s_idx, s in enumerate(m_range):
        if p[s_idx] < 1e-15:
            continue
        # dc[s]/dh[i] for all i
        # h[i] as first factor: (1-h[i-s])*dx for valid i-s
        # h[i] as second factor: -h[i+s]*dx for valid i+s
        valid1 = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
        valid2 = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
        grad[valid1] += p[s_idx] * (1 - h[np.arange(n)[valid1] - s]) * dx
        grad[valid2] -= p[s_idx] * h[np.arange(n)[valid2] + s] * dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    logsum = m + np.log(np.exp(shifted).sum()) / alpha
    return logsum

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for s_idx in range(n_corr):
        pi = p[s_idx]
        if pi < 1e-15:
            continue
        s_val = s_idx - (n - 1)
        # First term: h[i] as first factor, contributes (1-h[i-s_val])*dx
        i_min = max(0, s_val)
        i_max = min(n, n + s_val)
        if i_max > i_min:
            idx = np.arange(i_min, i_max)
            grad[idx] += pi * (1.0 - x[idx - s_val]) * dx
        # Second term: h[i] as second factor (in 1-h), contributes -x[i+s_val]*dx
        i_min2 = max(0, -s_val)
        i_max2 = min(n, n - s_val)
        if i_max2 > i_min2:
            idx2 = np.arange(i_min2, i_max2)
            grad[idx2] -= pi * x[idx2 + s_val] * dx
    return grad

def pgd_optimize(h, n, alpha, deadline, max_iter=5000):
    """Projected gradient descent with backtracking."""
    h = h.copy()
    cur_obj = lse_objective(h, n, alpha)
    lr = 0.1 / alpha  # initial learning rate
    best_h = h.copy()
    best_obj = cur_obj
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = lse_gradient(h, n, alpha)
        # Backtracking line search
        for _ in range(20):
            h_new = project_feasible(h - lr * grad, n)
            obj_new = lse_objective(h_new, n, alpha)
            if obj_new < cur_obj:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        cur_obj = obj_new
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
    return best_h, best_obj

def slsqp_optimize(h, n, alpha, deadline, maxiter=500):
    def obj(x):
        return lse_objective(x, n, alpha)
    def grad(x):
        return lse_gradient(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_basis_start(n, rng, m=15):
    """Generate a starting point from triangular basis functions."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for _ in range(m):
        center = rng.uniform(0, 2)
        width = rng.uniform(0.1, 0.5)
        weight = rng.uniform(0.5, 2.0)
        tri = np.maximum(0, 1 - np.abs(x - center) / width)
        h += weight * tri
    return project_feasible(h, n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b
```

```python
def threshold_grad(h, n, theta):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    mask = corr > theta
    if not mask.any():
        return np.zeros(n)
    viol = np.where(mask, corr - theta, 0.0)
    # grad = 2 * sum_s viol_s * dc_s/dh
    grad = np.zeros(n)
    for s_idx in np.where(mask)[0]:
        vi = viol[s_idx]
        if vi < 1e-15:
            continue
        s_val = s_idx - (n-1)
        i_min = max(0, s_val)
        i_max = min(n, n + s_val)
        if i_max > i_min:
            idx = np.arange(i_min, i_max)
            grad[idx] += 2 * vi * (1.0 - h[idx - s_val]) * dx
        i_min2 = max(0, -s_val)
        i_max2 = min(n, n - s_val)
        if i_max2 > i_min2:
            idx2 = np.arange(i_min2, i_max2)
            grad[idx2] -= 2 * vi * h[idx2 + s_val] * dx
    return grad
```

This is much faster than the log-sum-exp gradient when only a few shifts violate!
