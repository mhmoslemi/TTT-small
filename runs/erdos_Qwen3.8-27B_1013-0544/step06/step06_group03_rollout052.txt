def lag_gradient(h, n, lag):

        return project_feasible(h, n), compute_c5(h, n)

        h_full[soft_mask] = x
        return compute_c5(h_full, n)

    soft_idx = np.where(soft_mask)[0]

        return float(np.sum(x) - soft_target)

        return np.ones(n_soft)

        h[soft_mask] = np.clip(res.x, 0, 1)

        pass

    return h, compute_c5(h, n)

def project_feasible(h, n):
    target_sum = n / 2.0

        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

        h_pgd, c_pgd = pgd_optimize(best_h, n, pgd_end, top_k=3, step_init=0.003)
        if c_pgd < best_c:
            best_c = c_pgd
            best_h = h_pgd
    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-3:]
        h_new = slsqp_solve(best_h, n, top_idx, 500000, maxiter=4000, use_exact=False)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 10:
        h_bin, c_bin = binary_refine(best_h, n, deadline - 5)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    if time.time() < deadline - 5:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-4:]
        h_new = slsqp_solve(best_h, n, top_idx, 1000000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = None
    best_c = float('inf')
    if n_init >= 50:
        alloc = min(250, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 250, 350, 450, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and abs(n_try - len(best_h)) < 30:
            continue
        remaining = deadline - time.time() - 60
        if remaining < 70:
            break
        alloc = min(70, remaining * 0.22)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        else:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_full(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.015, n)
            elif r % 4 == 1:
                hp = best_h.copy()
                flip_mask = rng.rand(n) < 0.08
                hp[flip_mask] = 1.0 - hp[flip_mask]
            elif r % 4 == 2:
                freq = rng.randint(2, 8)
                phase = rng.uniform(0, 2 * np.pi)
                hp = best_h + 0.04 * np.sin(freq * np.pi * np.arange(n) / n + phase)
            else:
                hp = best_h + rng.normal(0, 0.005, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 35) * 0.32)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if time.time() < deadline - 8:
            h_pgd, c_pgd = pgd_optimize(best_h, n, deadline - 6, top_k=2, step_init=0.001)
            if c_pgd < best_c:
                best_c = c_pgd
                best_h = h_pgd
        if time.time() < deadline - 4:
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-3:]
            h_new = slsqp_solve(best_h, n, top_idx, 2000000, maxiter=5000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
