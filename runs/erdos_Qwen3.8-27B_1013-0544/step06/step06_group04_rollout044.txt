import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    return float(m + np.log(np.exp(sh).sum()) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    w = np.exp(sh) / np.exp(sh).sum()
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        m_idx = s + (n - 1)
        wm = w[m_idx]
        if wm < 1e-13:
            continue
        wm *= dx
        j1_lo = max(0, -s)
        j1_hi = min(n - 1, n - 1 - s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += wm * (1.0 - h[j1_lo + s:j1_hi + s + 1])
        j2_lo = max(0, s)
        j2_hi = min(n - 1, n - 1 + s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] -= wm * h[j2_lo - s:j2_hi - s + 1]
    return grad

def slsqp_opt(h, n, alpha, maxiter=1000):
    def obj(x):
        return smooth_max_val(x, n, alpha)
    def jac(x):
        return smooth_max_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coordinate_descent(h, n, alpha, deadline, max_rounds=50):
    h = h.copy()
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > deadline:
                break
            def obj_1d(val):
                h2 = h.copy()
                h2[i] = val
                return smooth_max_val(h2, n, alpha)
            res = minimize(obj_1d, h[i], method='bounded', bounds=(0.0, 1.0),
                           options={'xatol': 1e-10, 'maxiter': 100})
            if res.success and res.fun < obj_1d(h[i]) - 1e-12:
                h[i] = res.x[0]
                improved = True
        if not improved:
            break
    return project_feasible(h, n)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta_fast(h, i, j, n):

    shift_i = i
    shift_j = j
    lo = max(0, n - 1 - shift_i)

    lo = max(0, n - 1 - shift_j)
