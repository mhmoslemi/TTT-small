```python
def continuous_coordinate_descent(h, n, max_iter=100, deadline=None):
    """Coordinate descent on the continuous relaxation"""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        improved = False
        for j in range(n):
            # Compute slopes for all k
            corr = np.correlate(h, 1 - h, mode='full')
            slopes = np.zeros(2*n-1)
            # ∂I_k/∂h[j] = (1-h[j+lag])·[valid] - h[j-lag]·[valid]
            # lag = k - (n-1)
            for k in range(2*n-1):
                lag = k - (n-1)
                s = 0.0
                if 0 <= j + lag < n:
                    s += (1 - h[j + lag])
                if 0 <= j - lag < n:
                    s += (-h[j - lag])
                slopes[k] = s
            
            # Current values
            # corr[k] = I_k at current h
            # New value as function of h[j]: corr[k] + slopes[k] * (h_new - h[j])
            # We want to minimize max_k (corr[k] + slopes[k] * (h_new - h[j]))
            # over h_new ∈ [0, 1]
            
            # This is a 1D problem: minimize max of linear functions
            # The optimal is where the "upper envelope" is minimized
            # Binary search on the value: find the minimum possible max
            
            # More efficient: the optimal h_new is where the maximum
            # of the active constraints is minimized
            # This is a linear programming problem in 1 variable
            
            # Simple approach: try h_new = 0, 1, and the crossing points
            # of pairs of the top correlations
            
            # Get top correlations
            top_idx = np.argsort(corr)[-20:]
            vals = corr[top_idx]
            sl = slopes[top_idx]
            
            # Try boundaries
            candidates = [0.0, 1.0]
            
            # Try crossing points of pairs
            for a in range(min(20, len(top_idx))):
                for b in range(a+1, min(20, len(top_idx))):
                    if abs(sl[a] - sl[b]) > 1e-10:
                        crossing = (vals[b] - vals[a]) / (sl[a] - sl[b]) + h[j]
                        if 0 <= crossing <= 1:
                            candidates.append(crossing)
            
            # Also try the current value
            candidates.append(h[j])
            
            # Evaluate each candidate
            best_val = np.inf
            best_candidate = h[j]
            for c_val in candidates:
                c_val = np.clip(c_val, 0, 1)
                new_corr = corr + sl * (c_val - h[j])  # Only top_k affected? No, all k.
                # Actually we need the full change:
                new_corr_full = corr + slopes * (c_val - h[j])
                m = new_corr_full.max()
                if m < best_val:
                    best_val = m
                    best_candidate = c_val
            
            if best_candidate != h[j]:
                h[j] = best_candidate
                improved = True
        
        # Project to feasible set
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
```

This is: minimize max_k (corr[k] + S[j,k] * δ)

The function f(δ) = max_k (corr[k] + S[j,k] * δ) is convex piecewise-linear in δ.
The minimum over δ ∈ [-(h[j]), 1-h[j]] can be found by:
1. If the minimum of f over all δ (unconstrained) is in the interval, that's the answer.
2. Otherwise, the answer is at a boundary.

The unconstrained minimum of f(δ) is where the "slope" of f changes from negative to positive. Since f is the max of linear functions, its slope at any point is the max of the slopes of the active functions.

A simpler approach: binary search on the value.
- Lower bound: min_k corr[k] (if all slopes are 0)
- Upper bound: max_k corr[k]
- For a given target T, check if there exists δ such that corr[k] + S[j,k]*δ ≤ T for all k.
  This is: δ ≤ (T - corr[k])/S[j,k] for S[j,k] > 0
           δ ≥ (T - corr[k])/S[j,k] for S[j,k] < 0
  So we need max_{S[j,k]<0} (T-corr[k])/S[j,k] ≤ min_{S[j,k]>0} (T-corr[k])/S[j,k]
  AND this δ must be in [-(h[j]), 1-h[j]].

This is a feasibility check that can be done in O(n) per check, and binary search takes O(log(1/ε)) iterations.

Total per coordinate: O(n * log(1/ε)) ≈ O(n * 20) = O(5000) for n=250.
Total per pass: O(n * 5000) = O(1.25M) for n=250.

```python
def continuous_cd(h, n, max_passes=20, deadline=None):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        
        for j in range(n):
            # Compute slopes S[j, k] for all k
            # S[j,k] = (1-h[j+lag])·[0≤j+lag<n] - h[j-lag]·[0≤j-lag<n], lag=k-(n-1)
            slopes = np.zeros(2*n-1)
            # First term: (1-h[j+lag]) when j+lag is in [0,n-1]
            # j+lag = j+k-n+1, valid when 0 ≤ j+k-n+1 ≤ n-1, i.e., n-1-j ≤ k ≤ 2n-2-j
            lo1 = max(0, n-1-j)
            hi1 = min(2*n-2, 2*n-2-j)
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1+1)
                slopes[idx] += (1 - h[j + idx - (n-1)])
            # Second term: -h[j-lag] when j-lag is in [0,n-1]
            # j-lag = j-k+n-1, valid when 0 ≤ j-k+n-1 ≤ n-1, i.e., j ≤ k ≤ j+n-1
            lo2 = max(0, j)
            hi2 = min(2*n-2, j+n-1)
            if lo2 <= hi2:
                idx = np.arange(lo2, hi2+1)
                slopes[idx] += (-h[j - idx + (n-1)])
            
            # Now find best δ ∈ [-(h[j]), 1-h[j]] to minimize max_k (corr[k] + slopes[k]*δ)
            # Binary search on T
            lo = -(h[j])
            hi = 1.0 - h[j]
            
            # Find the optimal δ by binary search on T
            # For a given T, check feasibility:
            # Need: corr[k] + slopes[k]*δ ≤ T for all k
            # If slopes[k] > 0: δ ≤ (T - corr[k]) / slopes[k]
            # If slopes[k] < 0: δ ≥ (T - corr[k]) / slopes[k]
            # If slopes[k] == 0: need corr[k] ≤ T
            
            # The feasible range for δ given T is:
            # [max_{s<0} (T-corr[k])/s[k], min_{s>0} (T-corr[k])/s[k]]
            # intersected with [lo, hi]
            
            # Binary search on T to find the minimum feasible T
            T_lo = corr.min()  # Lower bound on T
            T_hi = (corr + slopes * hi).max() if hi > 0 else corr.max()  # Upper bound
            T_lo = min(T_lo, (corr + slopes * lo).max())
            
            for _ in range(50):
                T_mid = (T_lo + T_hi) / 2
                # Check if there exists δ ∈ [lo, hi] such that corr[k] + slopes[k]*δ ≤ T_mid for all k
                pos_mask = slopes > 1e-15
                neg_mask = slopes < -1e-15
                zero_mask = ~pos_mask & ~neg_mask
                
                feasible = True
                if np.any(zero_mask) and corr[zero_mask].max() > T_mid:
                    feasible = False
                else:
                    if feasible:
                        delta_lo = lo
                        if np.any(neg_mask):
                            delta_lo = max(delta_lo, ((T_mid - corr[neg_mask]) / slopes[neg_mask]).max())
                        delta_hi = hi
                        if np.any(pos_mask):
                            delta_hi = min(delta_hi, ((T_mid - corr[pos_mask]) / slopes[pos_mask]).min())
                        if delta_lo > delta_hi + 1e-15:
                            feasible = False
                
                if feasible:
                    T_hi = T_mid
                else:
                    T_lo = T_mid
            
            # Now find the optimal δ for T = T_hi
            T = T_hi
            pos_mask = slopes > 1e-15
            neg_mask = slopes < -1e-15
            delta_lo = lo
            if np.any(neg_mask):
                delta_lo = max(delta_lo, ((T - corr[neg_mask]) / slopes[neg_mask]).max())
            delta_hi = hi
            if np.any(pos_mask):
                delta_hi = min(delta_hi, ((T - corr[pos_mask]) / slopes[pos_mask]).min())
            delta_lo = max(delta_lo, lo)
            delta_hi = min(delta_hi, hi)
            if delta_lo <= delta_hi:
                # Any δ in [delta_lo, delta_hi] works; pick the middle
                h[j] = h[j] + (delta_lo + delta_hi) / 2
            else:
                # Pick the boundary that's closest to feasible
                h[j] = h[j] + (lo + hi) / 2
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```
