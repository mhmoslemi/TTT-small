```python
import numpy as np
import time
from scipy.optimize import minimize

G_N = 0
G_DX = 0.0
G_ALPHA = 1.0
G_TARGET = 0.0
G_BASE = None
G_IDX = None
G_RTARGET = 0.0

def set_state(n):
    global G_N, G_DX, G_TARGET
    G_N = int(n)
    G_DX = 2.0 / float(G_N)
    G_TARGET = float(G_N) / 2.0

def proj(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            sr = float(r.sum())
            if sr < 1e-14:
                break
            h += (d / sr) * r
        else:
            sm = s
            if sm < 1e-14:
                break
            h += (d / sm) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - float(h.sum())
        if abs(d) < 1e-13:
            break
        if d > 0:
            k = int(np.argmax(1.0 - h))
        else:
            k = int(np.argmax(h))
        h[k] = float(np.clip(h[k] + d, 0.0, 1.0))
    return h

def compute_c5(h, n=None):
    if n is None:
        n = G_N
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def smooth_max(vals):
    m = float(np.max(vals))
    z = G_ALPHA * (vals - m)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / float(G_ALPHA)

def accum_grad(vals, hx):
    n = G_N
    L = 2 * n - 1
    center = n - 1
    m = float(np.max(vals))
    z = G_ALPHA * (vals - m)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(L)
        w[int(np.argmax(vals))] = 1.0
    else:
        w = e / s
    wd = G_DX * w
    g = np.zeros(n)
    for p in range(L):
        wp = wd[p]
        if wp <= 1e-15:
            continue
        d = center - p
        a = max(0, -d)
        b = n - 1 - max(0, d)
        ln = b - a + 1
        if ln <= 0:
            continue
        g[a:b + 1] += wp * (1.0 - hx[a + d:b + d + 1])
        g[a + d:b + d + 1] -= wp * hx[a:b + 1]
    return g

def red_soft_fun(x):
    x = np.asarray(x, dtype=float)
    hx = G_BASE.copy()
    hx[G_IDX] = x
    vals = G_DX * np.correlate(hx, 1.0 - hx, mode="full")
    return float(smooth_max(vals))

def red_soft_grad(x):
    x = np.asarray(x, dtype=float)
    hx = G_BASE.copy()
    hx[G_IDX] = x
    vals = G_DX * np.correlate(hx, 1.0 - hx, mode="full")
    g = accum_grad(vals, hx)
    return g[G_IDX]

def red_cons(x):
    return float(np.asarray(x, dtype=float).sum() - G_RTARGET)

def red_cons_jac(x):
    return np.ones_like(np.asarray(x, dtype=float))

def mk_bounds(n):
    return [(0.0, 1.0)] * int(n)

def red_state(h, eps):
    n = G_N
    target = G_TARGET
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo = h <= eps
    hi = h >= 1.0 - eps
    base = h.copy()
    base[lo] = 0.0
    base[hi] = 1.0
    free = ~(lo | hi)
    if not free.any():
        return None
    p = int(free.sum())
    need = target - float(base[~free].sum())
    if need < -1e-8 or need > p + 1e-8:
        return None
    if need < 0.0 and need > -1e-9:
        need = 0.0
    if need > p and need < p + 1e-9:
        need = float(p)
    return base, np.where(free)[0], float(need)

def slsqp_reduce(h, alpha, eps, miter):
    global G_ALPHA, G_BASE, G_IDX, G_RTARGET
    n = G_N
    target = G_TARGET
    h = proj(h, target)
    st = red_state(h, eps)
    if st is None:
        return h
    base, idx, need = st
    p = len(idx)
    if p == 0:
        return h
    x0 = h[idx].copy()
    x0 = proj(x0, need)
    G_ALPHA = float(alpha)
    G_BASE = base
    G_IDX = idx
    G_RTARGET = need
    miter = min(int(miter), max(40, 24000 // max(1, p)))
    cons = [{'type': 'eq', 'fun': red_cons, 'jac': red_cons_jac}]
    try:
        res = minimize(red_soft_fun, x0, method='SLSQP', bounds=mk_bounds(p),
                       constraints=cons, options={'maxiter': miter, 'ftol': 1e-12, 'disp': 0})
        x1 = np.asarray(res.x, dtype=float)
    except Exception:
        x1 = x0
    h1 = base.copy()
    h1[idx] = x1
    return proj(h1, target)

def polish(h, t_end, seed=0, fast=False):
    n = G_N
    target = G_TARGET
    h = proj(h, target)
    if np.all((h <= 1e-10) | (h >= 1.0 - 1e-10)):
        h = proj(h + 0.04 * np.random.RandomState(seed).randn(n), target)
    besth = h.copy()
    bestc = compute_c5(besth)
    if fast:
        sched = ((2000, 0.001, 170), (8000, 1e-9, 220), (20000, 0.0, 280), (50000, 0.0, 300))
    else:
        sched = ((30, 0.05, 100), (100, 0.02, 160), (500, 0.005, 230),
                 (2000, 0.001, 280), (8000, 1e-9, 340), (20000, 0.0, 420), (50000, 0.0, 320))
    for a, e, m in sched:
        if time.time() > t_end - 12:
            break
        src = besth
        if np.all((src <= 1e-12) | (src >= 1.0 - 1e-12)):
            src = proj(src + 0.02 * np.random.RandomState(seed + int(a)).randn(n), target)
        h2 = slsqp_reduce(src, a, e, m)
        c2 = compute_c5(h2)
        if c2 < bestc:
            bestc = c2
            besth = h2
    return besth, bestc

def make_binary(h):
    if abs(G_TARGET - round(G_TARGET)) > 1e-9:
        return proj(h, G_TARGET)
    target = int(round(G_TARGET))
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    b = (h >= 0.5).astype(float)
    diff = target - int(b.sum())
    if diff > 0:
        z = np.flatnonzero(b == 0)
        order = np.argsort(-h[z])[:diff]
        b[z[order]] = 1.0
    elif diff < 0:
        o = np.flatnonzero(b == 1)
        order = np.argsort(h[o])[:-diff]
        b[o[order]] = 0.0
    return b

def corr_counts(h):
    hi = np.rint(h).astype(np.int64)
    c = np.correlate(hi, 1.0 - hi, mode="full")
    return np.rint(c[::-1]).astype(np.int64)

def build_up(h):
    n = G_N
    L = 2 * n - 1
    hi = np.rint(h).astype(np.int64)
    up = np.zeros((n, L), dtype=np.int64)
    omh = 1 - hi
    rev = (-hi)[::-1]
    for p in range(n):
        up[p, n - 1 - p:n - 1 - p + n] += omh
        up[p, p:p + n] += rev
        up[p, n - 1] = 0
    return up

def binary_hill(h0, t_end, seed):
    if abs(G_TARGET - round(G_TARGET)) > 1e-9:
        return proj(h0, G_TARGET)
    n = G_N
    center = n - 1
    h = make_binary(h0).copy()
    counts = corr_counts(h)
    up = build_up(h)
    cur = int(counts.max())
    best = cur
    besth = h.copy()
    rng = np.random.RandomState(seed)
    block = 96
    while time.time() < t_end:
        zeros = np.flatnonzero(1.0 - h)
        ones = np.flatnonzero(h)
        if zeros.size == 0 or ones.size == 0:
            break
        bestc = cur
        bi = -1
        bj = -1
        for i in zeros:
            upi = up[i]
            for st in range(0, ones.size, block):
                o = ones[st:st + block]
                base = upi[None, :] - up[o] + counts[None, :]
                mb = base.max(axis=1)
                rows = np.where(mb < bestc)[0]
                if rows.size == 0:
                    continue
                sub = base[rows].copy()
                oo = o[rows]
                d = np.abs(i - oo)
                r = np.arange(sub.shape[0])
                np.add.at(sub, (r, center + d), 1)
                np.add.at(sub, (r, center - d), 1)
                mc = sub.max(axis=1)
                pos = np.where(mc < bestc)[0]
                if pos.size:
                    q = pos[int(np.argmin(mc[pos]))]
                    nc = int(mc[q])
                    if nc < bestc:
                        bestc = nc
                        bi = int(i)
                        bj = int(oo[q])
            if bestc <= 0:
                break
        if bi >= 0:
            h[bi] = 1.0
            h[bj] = 0.0
            counts = corr_counts(h)
            up = build_up(h)
            cur = int(counts.max())
            if cur < best:
                best = cur
                besth = h.copy()
        else:
            moved = False
            att = 0
            while att < 350 and not moved and time.time() < t_end:
                att += 1
                i = int(rng.choice(zeros))
                j = int(rng.choice(ones))
                base = counts + up[i] - up[j]
                d = abs(i - j)
                base = base.copy()
                base[center + d] += 1
                base[center - d] += 1
                nc = int(base.max())
                if nc < cur or (nc == cur and rng.rand() < 0.2):
                    h[i] = 1.0
                    h[j] = 0.0
                    counts = corr_counts(h)
                    up = build_up(h)
                    cur = int(counts.max())
                    if cur < best:
                        best = cur
                        besth = h.copy()
                    moved = True
            if not moved:
                break
    return besth

def thue_init(n, seed):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        x = i
        p = 0
        while x:
            p ^= (x & 1)
            x >>= 1
        t[i] = p
    if n % 2 == 0:
        target = n // 2
        s = int(t.sum())
        rng = np.random.RandomState(seed)
        if s > target:
            idx = np.flatnonzero(t == 1)
            rng.shuffle(idx)
            t[idx[:s - target]] = 0.0
        elif s < target:
            idx = np.flatnonzero(t == 0)
            rng.shuffle(idx)
            t[idx[:target - s]] = 1.0
    return proj(t, float(n) / 2.0)

def rand_balanced(n, seed):
    if n % 2:
        return np.full(n, 0.5, dtype=float)
    b = np.zeros(n, dtype=float)
    idx = np.random.RandomState(seed).choice(n, n // 2, replace=False)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(30.0, float(budget_s) - 20.0)
    use = max(30.0, float(budget_s) - 25.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = rand_balanced(n, seed)

    set_state(n)
    h0 = proj(h0, G_TARGET)
    besth = h0.copy()
    bestc = compute_c5(besth)

    end = min(t0 + 0.35 * use, deadline - 30.0)
    if time.time() < end - 8.0:
        hh, cc = polish(besth, end, seed, fast=False)
        if cc < bestc:
            bestc = cc
            besth = hh

    if bestc > 0.3808:
        end = min(t0 + 0.52 * use, deadline - 60.0)
        if n <= 550 and time.time() < end - 12.0:
            hb = make_binary(besth)
            hill_end = min(time.time() + 0.08 * use, end - 0.08 * use)
            if hill_end > time.time() + 10.0:
                hb = binary_hill(hb, hill_end, seed + 1)
            cb = compute_c5(hb)
            if cb < bestc:
                bestc = cb
                besth = hb
            if time.time() < end - 5.0:
                hh, cc = polish(hb, end, seed + 2, fast=False)
                if cc < bestc:
                    bestc = cc
                    besth = hh

    if bestc > 0.3808:
        end = min(t0 + 0.64 * use, deadline - 50.0)
        if time.time() < end - 8.0:
            hs = proj(besth + 0.005 * rng.randn(n), G_TARGET)
            hh, cc = polish(hs, end, seed + 10, fast=False)
            if cc < bestc:
                bestc = cc
                besth = hh

    if bestc > 0.3808:
        end = min(t0 + 0.75 * use, deadline - 50.0)
        if time.time() < end - 8.0:
            hs = proj(besth + 0.02 * rng.randn(n), G_TARGET)
            hh, cc = polish(hs, end, seed + 20, fast=False)
            if cc < bestc:
                bestc = cc
                besth = hh

    if bestc > 0.3808:
        end = min(t0 + 0.87 * use, deadline - 40.0)
        if time.time() < end - 8.0:
            ht = thue_init(n, seed + 30)
            if n % 2 == 0 and n <= 550:
                hill_end = min(time.time() + 0.035 * use, end - 0.05 * use)
                if hill_end > time.time() + 6.0:
                    ht = binary_hill(ht, hill_end, seed + 31)
            hh, cc = polish(ht, end, seed + 32, fast=False)
            if cc < bestc:
                bestc = cc
                besth = hh

    if bestc > 0.3808:
        end = min(t0 + 0.96 * use, deadline - 30.0)
        if time.time() < end - 8.0:
            hr = rand_balanced(n, seed + 40)
            hh, cc = polish(hr, end, seed + 41, fast=False)
            if cc < bestc:
                bestc = cc
                besth = hh

    if time.time() < deadline - 8.0:
        hh, cc = polish(besth, deadline - 3.0, seed + 50, fast=True)
        if cc < bestc:
            bestc = cc
            besth = hh

    if bestc > 0.3808:
        for sc, sd in ((0.002, 70), (0.008, 80), (0.02, 90)):
            rem = deadline - time.time()
            if rem < 25.0:
                break
            hs = proj(besth + sc * np.random.RandomState(sd).randn(n), G_TARGET)
            hh, cc = polish(hs, time.time() + min(16.0, rem - 6.0), sd, fast=True)
            if cc < bestc:
                bestc = cc
                besth = hh

    besth = proj(besth, G_TARGET)
    bestc = compute_c5(besth)
    return besth, float(bestc), int(n)
```
