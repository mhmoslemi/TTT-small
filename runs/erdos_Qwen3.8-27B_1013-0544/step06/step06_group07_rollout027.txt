def focused_local_search(h, n, deadline, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)

    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline:
        # Find bottleneck lags
        max_val = corr.max()
        bottleneck_mask = corr > max_val - dx * 0.1  # lags close to max
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_improvement = 0
        best_i, best_j = -1, -1
        best_delta = None
        
        # Try random swaps, focusing on reducing bottleneck
        n_tries = min(5000, len(zeros) * len(ones))
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j:

            delta = swap_delta(h, n, i, j, m_range)
            new_max = (corr + delta).max()
            improvement = max_val - new_max
            if improvement > best_improvement:
                best_improvement = improvement
                best_i, best_j = i, j
                best_delta = delta
        
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = corr + best_delta
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:

            for _ in range(n_kick):

def ibls(h, n, deadline, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    kick_size = 3
    patience = 50  # number of kicks without improvement before giving up
    
    for restart in range(200):
        if time.time() > deadline:
            break
        
        # Local search: best improvement
        local_iters = 0
        while local_iters < 500:
            if time.time() > deadline:
                return best_h, best_c
            
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            
            best_improvement = 1e-10
            best_ij = None
            best_delta = None
            
            for i in zeros:
                # Compute add_i
                add_i = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                add_i[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                add_i[msk] -= h[imn[msk]]
                
                for j in ones:
                    delta = add_i.copy()
                    # sub_j
                    jm = j + m_range
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    jmn = j - m_range
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    # corrections
                    delta[n-1] -= 2.0
                    delta[n-1+j-i] += 1.0
                    delta[n-1+i-j] += 1.0
                    
                    new_max = (corr + delta).max()
                    improvement = cur_c - new_max * dx
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_ij = (i, j)
                        best_delta = delta
            
            if best_ij is not None:
                i, j = best_ij
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + best_delta
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                local_iters += 1
            else:
                break
        
        # Kick
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for _ in range(kick_size):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i != j:
                h[i], h[j] = h[j], h[i]
        corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
        cur_c = corr.max() * dx
    
    return best_h, best_c

But wait, compute_c5 calls np.correlate which is O(n^2) (computing all lags). For the 1D optimization, we don't need to recompute the full correlation - we only need to update the parts that change.

When h[i] changes by delta, the change in correlation at lag k is:
delta_corr[k] = delta * (1 - h[i+k]) + delta * (-h[i-k]) = delta * (1 - h[i+k] - h[i-k])

(with appropriate boundary handling)

So we can precompute the "coefficient" for each k and evaluate the new correlation as:
new_corr[k] = old_corr[k] + (new_h_i - old_h_i) * coeff[k]

where coeff[k] = (1 - h[i+k]) - h[i-k] (with boundary terms handled).

This makes each evaluation O(n) instead of O(n^2)!

Let me implement this properly:

```python
def coord_optimize(h, n, i, corr):
    """Optimize h[i] given all other coordinates fixed.
    Returns (new_h_i, new_corr)."""
    dx = 2.0 / n
    old_i = h[i]
    
    # Compute coefficient for each lag k
    # corr_full[m] = sum_j h[j] * (1-h[j - (m-(n-1))])
    # = sum_j h[j] * (1-h[j-k]) where k = m-(n-1)
    # Change in h[i] affects:
    #   - When j = i: term h[i]*(1-h[i-k]) changes
    #   - When j-k = i, i.e., j = i+k: term h[i+k]*(1-h[i]) changes
    
    # For lag k (m = n-1+k):
    # Term 1: h[i]*(1-h[i-k]) → change by delta*(1-h[i-k]) if 0≤i-k<n
    # Term 2: h[i+k]*(1-h[i]) → change by -delta*h[i+k] if 0≤i+k<n
    # Total coefficient for lag k: (1-h[i-k])*[0≤i-k<n] - h[i+k]*[0≤i+k<n]
    
    coeff = np.zeros(2*n-1)
    # Term 1: for m = n-1+k where 0≤i-k<n → k from -(i) to n-1-i
    # m = n-1+k, so m from n-1-i to 2n-2-i
    lo1 = max(0, n-1-i)
    hi1 = min(2*n-2, n-1+n-1-i)
    for m in range(lo1, hi1+1):
        k = m - (n-1)
        coeff[m] += 1.0 - h[i-k]
    
    # Term 2: for m = n-1+k where 0≤i+k<n → k from -i to n-1-i
    # m = n-1+k, so m from n-1-i to 2n-2-i
    lo2 = max(0, n-1-i)
    hi2 = min(2*n-2, n-1+n-1-i)
    for m in range(lo2, hi2+1):
        k = m - (n-1)
        coeff[m] -= h[i+k]
    
    # Now minimize max(corr + delta * coeff) over delta ∈ [-old_i, 1-old_i]
    # This is: minimize max_k (corr[k] + delta * coeff[k])
    # = minimize max_k (coeff[k] * delta + corr[k])
    # which is the max of linear functions of delta → convex piecewise linear
    
    lo = -old_i
    hi = 1.0 - old_i
    
    # Find minimum of max of linear functions on [lo, hi]
    # Evaluate at endpoints and at "breakpoints"
    # Breakpoints are where two lines cross: corr[a]+delta*coeff[a] = corr[b]+delta*coeff[b]
    # delta = (corr[b]-corr[a])/(coeff[a]-coeff[b])
    
    # For efficiency, evaluate at a grid of points
    n_eval = 50
    deltas = np.linspace(lo, hi, n_eval)
    vals = np.max(corr[None,:] + deltas[:,None] * coeff[None,:], axis=1)
    best_idx = np.argmin(vals)
    best_delta = deltas[best_idx]
    
    # Refine with golden section
    # ... (optional)
    
    new_i = old_i + best_delta
    h[i] = new_i
    new_corr = corr + best_delta * coeff
    return h, new_corr
```

Wait, but this grid approach might miss the optimal point. The optimal point is at a "kink" of the piecewise linear function, which occurs at a crossing of two lines.

A better approach: since the function is convex piecewise linear, the minimum occurs either at an endpoint or at a point where the subgradient contains 0. At a kink point, the left derivative < 0 < right derivative.

We can find it by:
1. Evaluate at lo and hi
2. If f(lo) ≤ f(hi), the minimum is at lo or somewhere in between where the slope becomes positive
3. Use a binary search on the slope

Actually, the simplest correct approach: since the function is convex and piecewise linear, and we have finite number of "pieces" (at most 2n-1 lines), we can find all breakpoints and evaluate at each.

But that's O(n^2) breakpoints in the worst case (each pair of lines can cross once).

For n=250, that's 250^2/2 ≈ 31K breakpoints per coordinate, and 250 coordinates, so 7.8M evaluations total. Each evaluation is O(1) (just computing the max of 2n-1 values at that point). Total: 7.8M * 499 ≈ 3.9B operations. Too slow!

Let me use a simpler approach: ternary search or golden section search with ~50 evaluations. Since the function is convex, golden section will converge to the minimum in O(log(1/eps)) steps.

Actually, for a piecewise linear convex function, golden section search works but converges linearly (not quadratically like for smooth functions). 50 iterations should give good accuracy.

But each evaluation is O(n) (computing the max of corr + delta*coeff). So 50 * O(n) = 50 * 500 = 25K per coordinate. Total: 250 * 25K = 6.25M per sweep. Very fast!

Let me also add the "exact" solution: for a piecewise linear function, the minimum is at a kink or an endpoint. We can find the kink by binary search on the slope.

The slope of max_k (a_k*x + b_k) at point x is:
slope(x) = max_{k ∈ S(x)} a_k where S(x) = {k : a_k*x + b_k = max}

We want to find x where slope changes from negative to positive.

Algorithm:
1. x = (lo + hi) / 2
2. Compute slope(x)
3. If slope < 0: lo = x (minimum is to the right)
4. If slope > 0: hi = x (minimum is to the left)
5. If slope = 0 or 0 ∈ [min_active_slope, max_active_slope]: x is the minimum

This converges in O(log(1/eps)) steps.

Let me implement this:

```python
def find_1d_min(corr, coeff, lo, hi, max_iter=100):
    """Find min of max_k (corr[k] + coeff[k]*x) for x in [lo, hi]."""
    for _ in range(max_iter):
        x = 0.5 * (lo + hi)
        vals = corr + coeff * x
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        slopes_active = coeff[active]
        min_slope = slopes_active.min()
        max_slope = slopes_active.max()
        if min_slope <= 0 <= max_slope:
            return x  # optimum found
        elif max_slope < 0:
            lo = x  # function is decreasing, go right
        else:
            hi = x  # function is increasing, go left
    return 0.5 * (lo + hi)
```

This is clean and efficient! Each iteration is O(n), and we need about 30-50 iterations for good precision.

Now let me put together the full algorithm:

```python
def coordinate_descent(h, n, deadline, max_sweeps=10000):
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * (2.0/n)
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_i = h[i]
            # Compute coefficient
            coeff = np.zeros(2*n-1)
            # h[i] appears in corr at:
            # m = n-1+k: term h[i]*(1-h[i-k]) [when i-k valid] and h[i+k]*(1-h[i]) [when i+k valid]
            for m in range(2*n-1):
                k = m - (n-1)
                c = 0.0
                if 0 <= i-k < n:
                    c += 1.0 - h[i-k]
                if 0 <= i+k < n:
                    c -= h[i+k]
                coeff[m] = c
            
            # Minimize max(corr + delta*coeff) for delta ∈ [-old_i, 1-old_i]
            lo = -old_i
            hi = 1.0 - old_i
            best_delta = find_1d_min(corr, coeff, lo, hi)
            h[i] = old_i + best_delta
            corr = corr + best_delta * coeff
            new_c = corr.max() * (2.0/n)
            if new_c < cur_c - 1e-14:
                cur_c = new_c
                improved = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # No improvement, revert
                h[i] = old_i
                corr = corr - best_delta * coeff
        
        if not improved:
            break
    
    return best_h, best_c
```

Wait, there's a subtle issue. When I update h[i] and then move to h[i+1], the corr is already updated. But the coeff for h[i+1] depends on h[i] (which has changed). So I need to recompute coeff each time, which I'm doing (the inner loop over m).

The main question is: does coordinate descent converge to a good solution? It will converge to a coordinate-wise optimum, which might not be the global optimum. But combined with SLSQP and restarts, it should be effective.

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_coeff(h, n, i):
    coeff = np.zeros(2*n-1)
    lo1 = i
    hi1 = min(i+n-1, 2*n-2)
    if lo1 <= hi1:
