from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_smooth(h, n, alpha, pen=0.0):
    """Compute value and gradient of smoothed C5 + penalty."""
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -700, 700)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = m + np.log(exp_s.sum()) / alpha
    if pen > 0:
        d = h.sum() - target
        val += pen * d * d
    
    g = np.zeros(n)
    mv = np.arange(-(n-1), n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = mv[k]
        lo = max(0, m)
        hi = min(n, n + m)
        if hi > lo:
            g[lo:hi] += wk * dx * (1 - h[lo-m:hi-m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if hi2 > lo2:
            g[lo2:hi2] -= wk * dx * h[lo2+m:hi2+m]
    if pen > 0:
        g += 2 * pen * (h.sum() - target)
    return val, g

def optimize_continuous(h, n, deadline, alpha_start=100, alpha_end=5000000, n_steps=15):
    """Multi-stage L-BFGS-B optimization."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Geometric sequence of alphas
    log_a = np.logspace(np.log10(alpha_start), np.log10(alpha_end), n_steps)
    pen = 1e3
    
    for alpha in log_a:
        if time.time() > deadline - 5:
            break
        # L-BFGS-B
        def obj(x):
            return grad_smooth(x, n, alpha, pen)[0]
        def jac(x):
            return grad_smooth(x, n, alpha, pen)[1]
        
        bounds = [(0, 1)] * n
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-18,
                                                 'gtol': 1e-10, 'maxfun': 5000})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=500):
    """Greedy binary local search with first-improvement."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                # Compute delta for swap i->1, j->0
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    zeros_arr = np.where(h == 0)[0].copy()

        for it in range(n_iters):
            if time.time() > deadline:
                break
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = zeros_arr[ai]
            j = ones_arr[bi]

            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range

            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2*n-1:
                delta[k2] += 1.0

            accept = nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(min(0, scale * (cur_c - nc) / T)))

                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
        if T < 1e-12:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_and_grad(h, n, alpha, pen=0.0):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -700, 700)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = m + np.log(exp_s.sum()) / alpha
    if pen > 0:
        d = h.sum() - target
        val += pen * d * d
    
    g = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - nm1
        lo1 = max(0, m)
        hi1 = min(n, n + m)
        if hi1 > lo1:
            g[lo1:hi1] += wk * dx * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if hi2 > lo2:
            g[lo2:hi2] -= wk * dx * h[lo2 + m:hi2 + m]
    if pen > 0:
        g += 2.0 * pen * (h.sum() - target)
    return val, g

def optimize_continuous(h, n, deadline, alpha_list, pen=5000.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()

            break

            return smooth_obj_and_grad(x, n, alpha, pen)[0]

            return smooth_obj_and_grad(x, n, alpha, pen)[1]

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    nm1 = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range

                jmn = j - m_range

                delta[nm1] -= 2.0
                k1 = nm1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = nm1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx

                    cur_c = nc
