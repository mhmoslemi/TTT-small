def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha, g

```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    p = np.exp(shifted - sm) / np.exp(shifted - sm).sum()  # length 2n-1, p[k_idx] where k_idx = n-1+lag
    
    grad = np.zeros(n)
    # p is indexed by k_idx = n-1+lag, so lag = k_idx - (n-1)
    lags = np.arange(-(n-1), n)
    for idx, lag in enumerate(lags):
        w = p[idx] * dx
        if w < 1e-18:
            continue
        if lag >= 0:
            # (1-h[j-lag]) for j in [lag, n-1]
            grad[lag:] += w * (1 - h[:n-lag])
            # -h[j+lag] for j in [0, n-1-lag]
            grad[:n-lag] -= w * h[lag:]
        else:
            m_ = -lag
            # (1-h[j+m_]) for j in [0, n-1-m_]
            grad[:n-m_] += w * (1 - h[m_:])
            # -h[j-m_] for j in [m_, n-1]
            grad[m_:] -= w * h[:n-m_]
    return grad
```

        viol = x.sum() - n/2.0
        return val + penalty * viol * viol
    
    def jac(x):
        _, grad = smooth_max_and_grad(x, n, alpha)
        grad = grad + 2.0 * penalty * (x.sum() - n/2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    h_out = project_feasible(res.x, n)
    return h_out

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    val = m + logsum / alpha
    p = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
    grad = np.zeros(n)
    for idx in range(len(p)):
        w = p[idx] * dx
        if w < 1e-20:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= w * h[lag:]
        else:
            m_ = -lag
            grad[:n - m_] += w * (1.0 - h[m_:])
            grad[m_:] -= w * h[:n - m_]
    return val, grad

def optimize_step(h, n, alpha, maxiter=400):
    penalty = 5e3
    target = n / 2.0
    def obj(x):
        val, _ = smooth_max_val_grad(x, n, alpha)
        viol = x.sum() - target
        return val + penalty * viol * viol
    def jac(x):
        _, grad = smooth_max_val_grad(x, n, alpha)
        grad = grad + 2.0 * penalty * (x.sum() - target)
        return grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

                j = int(list(ones_set)[rng.integers(0, len(ones_set))])
        if i == j:

        if i in tabu and tabu.get(i, 0) > it - tabu_len:

        delta = np.zeros(2 * n - 1)
        im = i + m_range

        imn = i - m_range
