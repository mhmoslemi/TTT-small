def swap_delta(h, n, i, j, m_range):

    imn = i - m_range

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_fast(h_bin, n, deadline, seed=42, t_start=0.015, t_end=0.000005, patience=25000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    no_improve = 0
    total_steps = 0
    cool_rate = (t_end / t_start) ** (1.0 / 80000.0)
    T = t_start

    while time.time() < deadline:
        r1 = rng.randint(n)
        r2 = rng.randint(n)
        if h[r1] == h[r2]:
            continue
        i, j = (r1, r2) if h[r1] == 0 else (r2, r1)
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = float(new_corr.max()) * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-14 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c - 1e-12:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            total_steps += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 4 + rng.randint(6)
            for _ in range(nk):
                a = rng.randint(n)
                b = rng.randint(n)
                if h[a] != h[b]:
                    x, y = (a, b) if h[a] == 0 else (b, a)
                    h[x] = 1.0
                    h[y] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = float(corr.max()) * dx
            no_improve = 0
            T = min(t_start * 0.5, max(T * 2.0, t_end))

    return best_h, best_c

def greedy_local(h_init, n, deadline, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    max_rounds = 8

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        max_swaps = 200
        for _ in range(max_swaps):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            rng.shuffle(zeros)
            found = False
            for i in zeros:
                for j in ones:
                    delta = swap_delta(h, n, i, j, m_range)
                    nc = float((corr + delta).max()) * dx
                    if nc < cur_c - 1e-14:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + delta
                        cur_c = nc
                        found = True
                        break
                if found:
                    break
            if not found:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) > 0 and len(zi) > 0:
            nk = 3 + rng.randint(4)
            for _ in range(nk):
                h[rng.choice(oi)] = 0.0
                h[rng.choice(zi)] = 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = float(corr.max()) * dx

    return best_h, best_c

def slsqp_optimize(h, n, deadline):
    dx = 2.0 / n
    best = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [500, 5000, 50000]:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = float(g.max())
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 600, 'ftol': 1e-10})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best = h.copy()
        except Exception:
            pass
    return best, best_c

def make_structured_start(n, seed, ptype):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    if ptype == 0:
        h = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    elif ptype == 1:
        freq = 2 + (seed % 5)
        phase = rng.random() * 2 * np.pi
        h = (np.sin(2 * np.pi * freq * np.arange(n) / n + phase) > 0).astype(float)
    elif ptype == 2:
        block = 4 + rng.randint(6)
        for b in range(n // block):
            if rng.random() < 0.5:
                h[b*block:(b+1)*block//2] = 1.0
            else:
                h[b*block+block//2:(b+1)*block] = 1.0
    elif ptype == 3:
        step = 2 + rng.randint(3)
        h[::step] = 1.0
    else:
        idx = rng.choice(n, target, replace=False)
        h[idx] = 1.0
    cur = int(h.sum())
    if cur > target:
        oi = np.where(h == 1)[0]
        h[oi[target:]] = 0.0
    elif cur < target:
        zi = np.where(h == 0)[0]
        h[zi[:target - cur]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Brief SLSQP
    p1 = t0 + 0.07 * (budget_s - 5)
    hs = project_feasible(h0, n)
    hs, cs = slsqp_optimize(hs, n, p1)
    if cs < best_c:
        best_c = cs
        best_h = hs.copy()

    # Phase 2: Multi-start SA (main workhorse)
    p2 = t0 + 0.62 * (budget_s - 5)
    starts = [round_to_binary(best_h, n)]
    for pt in range(5):
        starts.append(make_structured_start(n, seed + pt * 997, pt))
    for _ in range(2):
        hp = best_h + 0.25 * rng.randn(n)
        starts.append(round_to_binary(hp, n))

    remaining = p2 - time.time()
    per_start = remaining / len(starts)
    sa_params = [
        (0.02, 0.00001, 30000),
        (0.015, 0.000005, 25000),
        (0.025, 0.00002, 35000),
        (0.012, 0.00001, 20000),
        (0.018, 0.000008, 28000),
        (0.022, 0.000015, 32000),
        (0.01, 0.000005, 22000),
        (0.016, 0.00001, 26000),
    ]
    for si in range(len(starts)):
        if time.time() > p2 - 3:
            break
        end_t = min(p2, time.time() + per_start * 1.4)
        ts, te, pat = sa_params[si % len(sa_params)]
        s = seed + si * 1337
        hb, cb = sa_fast(starts[si], n, end_t, seed=s, t_start=ts, t_end=te, patience=pat)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy local search
    p3 = t0 + 0.78 * (budget_s - 5)
    hg, cg = greedy_local(best_h, n, p3, seed=seed + 555)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 4: SA from perturbed best
    p4 = t0 + 0.88 * (budget_s - 5)
    if time.time() < p4 - 8:
        hb_init = round_to_binary(best_h, n)
        z = np.where(hb_init == 0)[0]
        o = np.where(hb_init == 1)[0]
        for _ in range(5 + rng.randint(4)):
            hb_init[rng.choice(z)] = 1.0
            hb_init[rng.choice(o)] = 0.0
        z = np.where(hb_init == 0)[0]
        o = np.where(hb_init == 1)[0]
        hb, cb = sa_fast(hb_init, n, p4, seed=seed + 4444,
                        t_start=0.012, t_end=0.00001, patience=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: Final greedy
    if time.time() < dl - 3:
        hg2, cg2 = greedy_local(best_h, n, dl - 2, seed=seed + 8888)
        if cg2 < best_c:
            best_c = cg2
            best_h = hg2.copy()

    # Phase 6: Try higher resolution
    if time.time() < dl - 45 and best_c > 0.3795:
        n2 = n * 2
        if n2 <= 600:
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, n2)
            hh = project_feasible(np.interp(xn, xo, best_h), n2)
            hbin = round_to_binary(hh, n2)
            hb3, cb3 = sa_fast(hbin, n2, dl - 5, seed=seed + 7777,
                              t_start=0.012, t_end=0.00001, patience=15000)
            if cb3 < best_c:
                best_c = cb3
                best_h = hb3.copy()
                n = n2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
