def project_feasible(h_vec, n):
    target_sum = n/2.0
    h = h_vec.copy()
    while True:
        h_sum = target_sum - sum(h)
        if abs(h_sum) < 1e-14: break
        if h_sum > 0:
            room = np.where(h < 1, 1-h, 0)
            if sum(room) > 1e-14*h_sum:
                h += h_sum * room / sum(room)
            else:
                h_clip = np.clip(h, 0, 1)
                h_clip[h_clip==0 if np.random.rand() > 0.5 else 1] = h  # Push to boundary
        else:
            tot = sum(h)
            h += h_sum * np.clip(h, 0, None) / tot
        h = np.clip(h, 0, 1)
    return h
