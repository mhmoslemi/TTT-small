import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    return float((m + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def con_fun(x, n):
    return x.sum() - n / 2.0

def con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def eval_cd_step(d, c, ks, s, um, h_p, n, penalty, sumh, target):
    if abs(d) < 1e-15:
        val = float(c[ks].max())
        if val < um:
            val = um
    else:
        new_c = c[ks] + d * s
        ci = (ks == (n - 1))
        if np.any(ci):
            new_c = new_c.copy()
            new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h_p) - d * d
        val = float(new_c.max())
        if val < um:
            val = um
    if penalty > 0:
        val += penalty * (sumh + d - target) ** 2
    return val

def fast_cd(h, n, max_passes=300, n_grid=40, deadline=None, rng=None, penalty=5.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    sumh = float(h.sum())
    target = n / 2.0
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = float(c[:k_lo].max())
            if k_hi < 2 * n - 2:
                r = float(c[k_hi + 1:].max())
                if r > um:
                    um = r
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s[bwd] -= h[p + ks[bwd] - (n - 1)]
            best_val = eval_cd_step(0.0, c, ks, s, um, h[p], n, penalty, sumh, target)
            best_d = 0.0
            deltas = np.linspace(d_lo, d_hi, n_grid)
            min_idx = 0
            min_val = best_val
            for gi, d in enumerate(deltas):
                v = eval_cd_step(d, c, ks, s, um, h[p], n, penalty, sumh, target)
                if v < min_val - 1e-14:
                    min_val = v
                    min_idx = gi
            if min_val < best_val - 1e-14:
                best_val = min_val
                best_d = deltas[min_idx]
            lo = deltas[max(0, min_idx - 1)]
            hi = deltas[min(n_grid - 1, min_idx + 1)]
            if hi > lo:
                fine = np.linspace(lo, hi, 16)
                for d in fine:
                    v = eval_cd_step(d, c, ks, s, um, h[p], n, penalty, sumh, target)
                    if v < best_val - 1e-14:
                        best_val = v
                        best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]
                if len(ids_f) > 0:
                    c[ks[ids_f]] += d * (1.0 - h[p - ks[ids_f] + n - 1])
                if len(ids_b) > 0:
                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n - 1)]
                c[n - 1] -= d * d
                h[p] = float(np.clip(h[p] + d, 0.0, 1.0))
                sumh = float(h.sum())
                cm = float(c[ks].max())
                if um > cm:
                    cm = um
                current_max = cm
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def fourier_perturb(h, n, rng, scale):
    h_new = h.copy()
    n_basis = min(n // 2, 15)
    for k in range(1, n_basis + 1):
        if rng.random() < 0.7:
            amp = scale * rng.uniform(0.3, 1.0) / (1 + k * 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            h_new += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    h_new += rng.normal(0, scale * 0.1, n)
    return project_feasible(h_new, n)

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [2e5, 8e4, 3e4, 1e4, 5e3, 2e3]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        h_s = slsqp_run(h, n, alpha=a, maxiter=1500)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    if time.time() < deadline - 40:
        t_cd = min(120, deadline - time.time() - 30)
        if t_cd > 20:
            h_cd, c_cd = fast_cd(h, n, max_passes=400, n_grid=36,
                                 deadline=time.time() + t_cd, rng=rng, penalty=3.0)
            h_cd = project_feasible(h_cd, n)
            c_cd = compute_c5(h_cd, n)
            if c_cd < best_c:
                h, best_c = h_cd, c_cd
    if time.time() < deadline - 15:
        h, best_c = pair_search(h, n, deadline - 3,
                                [0.005, 0.002, 0.001, 0.0005], radius=40)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Multi-restart with Fourier perturbations
    n = len(best_h)

                                     [0.001, 0.0005, 0.0002, 0.0001], radius=35)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
