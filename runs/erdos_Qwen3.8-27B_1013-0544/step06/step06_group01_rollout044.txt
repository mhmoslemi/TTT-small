import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val * dx)

def slsqp_step(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fast_cd(h, n, deadline, grid=40, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            if time.time() > deadline:
                break
            old = h[p]
            if old < 1e-10 or old > 1.0 - 1e-10:
                continue
            k_min = max(0, n - 1 - p)
            k_max = min(2 * n - 2, n + p - 1)
            ks = np.arange(k_min, k_max + 1)
            s = np.zeros(len(ks))
            idx1 = p + ks - n + 1
            v1 = (idx1 >= 0) & (idx1 < n)
            s[v1] += (1.0 - h[idx1[v1]])
            idx2 = p - ks + n - 1
            v2 = (idx2 >= 0) & (idx2 < n)
            s[v2] -= h[idx2[v2]]
            mask = np.zeros(2 * n - 1, dtype=bool)
            mask[k_min:k_max + 1] = True
            c_out = 0.0
            if np.any(~mask):
                c_out = float(np.max(c[~mask]))
            cur_max = float(np.max(c))
            # Coarse search
            d_range = np.linspace(-old, 1.0 - old, grid)
            best_d = 0.0
            best_obj = cur_max
            for d in d_range:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                diag = (ks == n - 1)
                if np.any(diag):
                    new_c[diag] -= d * d
                obj = max(c_out, float(np.max(new_c)))
                if obj < best_obj:
                    best_obj = obj
                    best_d = d
            # Fine search around best_d
            if abs(best_d) > 1e-12:
                lo = max(-old, best_d - (1.0 - 2.0 * old) / grid)
                hi = min(1.0 - old, best_d + (1.0 - 2.0 * old) / grid)
                d_fine = np.linspace(lo, hi, 20)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    new_c = c[ks] + d * s
                    diag = (ks == n - 1)
                    if np.any(diag):
                        new_c[diag] -= d * d
                    obj = max(c_out, float(np.max(new_c)))
                    if obj < best_obj:
                        best_obj = obj
                        best_d = d
            if abs(best_d) > 1e-12:
                h[p] = old + best_d
                c[ks] += best_d * s
                diag = (ks == n - 1)
                if np.any(diag):
                    c[ks[diag]] -= best_d * best_d
                improved_any = True
    return h, float(np.max(c)) * dx
