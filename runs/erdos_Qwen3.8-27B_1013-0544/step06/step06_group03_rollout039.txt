def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    ts = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-15: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-15: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0: g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]; g[m:] += 1.0 - h[:n-m]
        else:
            g[:n-m] += 1.0 - h[m:]; g[m:] -= h[:n-m]
    return g

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500: return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

    return float(x.sum() - n / 2.0)

def _ajian(x, n):
    return np.ones(n)

                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(r.x, n)

        return h.copy()

    c = compute_all_lags(h_curr, n)
    hv = cp.Variable(n)
    cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1]

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    ts = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-15: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-15: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
        else:
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
    return g

def _sobj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sobj_top(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full') * dx
    a = cc[idx]
    m = a.max()
    s = alpha * (a - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _acon(x, n):
    return float(x.sum() - n / 2.0)

def _ajian(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, idx=None, maxiter=5000, ftol=1e-15):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _acon, 'jac': _ajian, 'args': (n,)}]
    f = _sobj_top if idx is not None else _sobj
    a = (n, idx, alpha) if idx is not None else (n, alpha)
    try:
        r = minimize(f, h, args=a, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def outer_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1,
            cp.abs(hv - h_curr) <= trust]
    for k in range(2*n-1):
        lag = k - (n-1)
        g = lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (hv - h_curr)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.GLOP, cp.CBC]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or hv.value is None:
        return h_curr.copy()
    return project_feasible(hv.value, n)
