```python
            h[i], h[j] = 1.0, 0.0
            zeros = np.delete(zeros, ii)

            corr = corr + g[i] - g[j]

            # Actually, let's update g incrementally
            # When h[i] changes 0→1 and h[j] changes 1→0:

def l_bfgsb_optimize(h, n, deadline, seed=42):
    """Continuous optimization using L-BFGS-B with log-sum-exp smoothing"""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in [50000, 200000, 500000, 2000000]:
        if time.time() > deadline - 10:
            break
            
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = smooth_max(c, alpha)
            pen = 1e8 * (x.sum() - n / 2.0) ** 2
            return m + pen
        
        bounds = [(0.0, 1.0)] * n
        
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-8, 'maxfun': 5000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def targeted_flow(h, n, deadline, seed=42):
    """Targeted perturbation: move mass to reduce worst lags"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for iteration in range(50):
        if time.time() > deadline - 5:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lags = np.argsort(corr)[-5:]
        
        improved = False
        for k in worst_lags:
            lag = k - (n - 1)
            # Find positions where h[p] is high and h[p+lag] is low (contributing to c[k])
            if lag >= 0:
                valid_p = range(n - lag)
                contribs = []
                for p in valid_p:
                    val = h[p] * (1 - h[p + lag])
                    if val > 0.01:
                        contribs.append((val, p, p + lag))
            else:
                lag2 = -lag
                valid_p = range(lag2, n)
                contribs = []
                for p in valid_p:
                    # h[p-lag2] * (1-h[p]) contributes
                    val = h[p - lag2] * (1 - h[p])
                    if val > 0.01:
                        contribs.append((val, p - lag2, p))
            
            contribs.sort(reverse=True)
            for val, src, dst in contribs[:3]:
                # Move small amount from src to dst
                eps = min(0.05, h[src] * 0.3, (1 - h[dst]) * 0.5)
                if eps < 0.01:
                    continue
                h_new = h.copy()
                h_new[src] -= eps
                h_new[dst] += eps
                h_new = np.clip(h_new, 0, 1)
                # Project to feasible
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-10:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Add small noise and continue
            h = project_feasible(h + 0.01 * rng.randn(n), n)
    
    return best_h, best_c
```

Wait, I have a concern about the `build_g` function being called in every iteration of the greedy loop. For n=250, it's O(n²) = 62500 operations per call. At 5000 iterations per second (rough estimate for Python), that's 3 million operations per second, which might be too slow.

Let me optimize: instead of rebuilding g from scratch, I can update it incrementally.

When h[i] changes from 0 to 1:
- g[p, k] where p+lag = i (i.e., p = i - lag = i - k + (n-1)): the forward term (1-h[p+lag]) = (1-h[i]) changes from 1 to 0, so g[p,k] decreases by 1
- g[p, k] where p-lag = i (i.e., p = i + lag = i + k - (n-1)): the backward term h[p-lag] = h[i] changes from 0 to 1, so g[p,k] decreases by 1

When h[j] changes from 1 to 0:
- g[p, k] where p+lag = j: (1-h[j]) changes from 0 to 1, so g[p,k] increases by 1
- g[p, k] where p-lag = j: h[j] changes from 1 to 0, so g[p,k] increases by 1

So the net effect of swapping h[i]: 0→1, h[j]: 1→0:
- g[p,k] changes by -1 when p = i-k+(n-1) [forward for i] or p = i+k-(n-1) [backward for i]
- g[p,k] changes by +1 when p = j-k+(n-1) [forward for j] or p = j+k-(n-1) [backward for j]

For each k, there are at most 4 positions p affected. So the total update is O(2n * 4) = O(n) per swap!

Let me implement this:

```python
def update_g_after_swap(g, i, j, n, h_i_old=0, h_j_old=1):
    """Update g after h[i]: h_i_old→1 and h[j]: h_j_old→0"""
    # For h[i]: h_i_old → 1
    # Forward: p+lag=i → p=i-lag=i-k+(n-1). (1-h[i]) changes from (1-h_i_old) to 0, delta = -(1-h_i_old)
    # Backward: p-lag=i → p=i+lag=i+k-(n-1). h[i] changes from h_i_old to 1, delta = -(1-h_i_old) [g -= h[i], so delta_g = -(new - old) = -(1 - h_i_old)]
    # Wait, let me re-derive.
    # g[p,k] = (1-h[p+lag])*valid_fwd - h[p-lag]*valid_bwd
    # When h[i] changes from old to new:
    #   If p+lag=i: forward term changes from (1-old) to (1-new). g changes by (1-new)-(1-old) = old-new
    #   If p-lag=i: backward term changes from -old to -new. g changes by -new-(-old) = old-new
    # So in both cases, g changes by (old - new) = h_i_old - 1 = -1 (when h_i_old=0)
    
    # For h[j]: h_j_old → 0
    #   If p+lag=j: forward term changes from (1-h_j_old) to (1-0)=1. g changes by 1-(1-h_j_old) = h_j_old = 1
    #   If p-lag=j: backward term changes from -h_j_old to -0. g changes by 0-(-h_j_old) = h_j_old = 1
    # So in both cases, g changes by +1 (when h_j_old=1)
    
    delta_i = h_i_old - 1.0  # = -1 when h_i_old = 0
    delta_j = h_j_old  # = +1 when h_j_old = 1
    
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        # Positions affected by h[i] change
        p1 = i - lag  # = i - k + (n-1), forward: p+lag=i
        if 0 <= p1 < n:
            g[p1, k] += delta_i
        p2 = i + lag  # = i + k - (n-1), backward: p-lag=i
        if 0 <= p2 < n:
            g[p2, k] += delta_i
        # Positions affected by h[j] change
        p3 = j - lag  # forward: p+lag=j
        if 0 <= p3 < n:
            g[p3, k] += delta_j
        p4 = j + lag  # backward: p-lag=j
        if 0 <= p4 < n:
            g[p4, k] += delta_j
```

This is O(2n * 4) per swap, which is much faster than O(n²).

But wait, there's a subtlety: if p1 == p3 or p1 == p4 etc., we might double-count. But since i ≠ j and the conditions are different, this shouldn't happen in practice (the positions would have to coincide, which means i ± lag1 = j ± lag2 for the same k, which is possible but rare).

Actually, for a given k, p1 = i - (k-(n-1)) and p3 = j - (k-(n-1)). If i ≠ j, then p1 ≠ p3. Similarly for the other pairs. And p1 = i - lag, p2 = i + lag. If lag ≠ 0, p1 ≠ p2. If lag = 0, p1 = p2 = i, and we'd add delta_i twice, which is correct (both forward and backward terms are affected).

```python
def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    lags = np.arange(2 * n - 1) - (n - 1)  # shape (2n-1,)
    for k in range(2 * n - 1):
        lag = lags[k]
        # Forward: p + lag in [0, n) → p in [-lag, n-1-lag]
        p_start_f = max(0, -lag)
        p_end_f = min(n - 1, n - 1 - lag)
        if p_start_f <= p_end_f:
            g[p_start_f:p_end_f + 1, k] += (1.0 - h[p_start_f + lag:p_end_f + lag + 1])
        # Backward: p - lag in [0, n) → p in [lag, n-1+lag]
        p_start_b = max(0, lag)
        p_end_b = min(n - 1, n - 1 + lag)
        if p_start_b <= p_end_b:
            g[p_start_b:p_end_b + 1, k] -= h[p_start_b - lag:p_end_b - lag + 1]
    return g
```

```python
def build_g(h, n):
    # g[p, k] = (1-h[p+lag])*[valid_fwd] - h[p-lag]*[valid_bwd]
    # where lag = k - (n-1)
    p = np.arange(n)[:, None]  # (n, 1)
    k = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    lag = k - (n-1)  # (1, 2n-1)
    
    fwd_idx = p + lag  # (n, 2n-1)
    bwd_idx = p - lag  # (n, 2n-1)
    
    fwd_valid = (fwd_idx >= 0) & (fwd_idx < n)
    bwd_valid = (bwd_idx >= 0) & (bwd_idx < n)
    
    # Use clip to avoid out-of-bounds indexing
    fwd_clipped = np.clip(fwd_idx, 0, n-1)
    bwd_clipped = np.clip(bwd_idx, 0, n-1)
    
    g = (1.0 - h[fwd_clipped]) * fwd_valid - h[bwd_clipped] * bwd_valid
    return g
```

Actually, let me try a different approach for the continuous phase: **projected subgradient method**.

The objective f(h) = max_k c[k] * dx is the max of linear functions (well, bilinear, but locally approximately linear). The subgradient at h is the c[k*] / dx for the argmax k* (times the appropriate derivative).

Actually, c[k] = ∑_p h[p](1-h[p+k-(n-1)]), which is bilinear. The gradient with respect to h is:
∂c[k]/∂h[i] = (1-h[i+k-(n-1)])_valid - h[i-k+(n-1)]_valid = g[i,k] (from our earlier derivation!)

So the subgradient of f(h) = max_k c[k] * dx is:
∇f(h) = dx * g[:, k*] where k* = argmax_k c[k]

And we can do projected gradient descent:
h ← project_feasible(h - lr * ∇f(h))

This is simple and effective! Let me use this as the main continuous optimization method, with a decreasing learning rate schedule.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    m = g.max()
    s = alpha * (g - m)
    return m + (s.max() + np.log(np.exp(s - s.max()).sum())) / alpha

def build_g(h, n):
    p = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    lag = k - (n - 1)
    fwd_idx = p + lag
    bwd_idx = p - lag
    fwd_valid = (fwd_idx >= 0) & (fwd_idx < n)
    bwd_valid = (bwd_idx >= 0) & (bwd_idx < n)
    fwd_clipped = np.clip(fwd_idx, 0, n - 1)
    bwd_clipped = np.clip(bwd_idx, 0, n - 1)
    g = (1.0 - h[fwd_clipped]) * fwd_valid - h[bwd_clipped] * bwd_valid
    return g

def update_g(g, i, j, n):
    """Update g after h[i]: 0→1 and h[j]: 1→0"""
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        p1 = i - lag
        if 0 <= p1 < n:
            g[p1, k] -= 1.0
        p2 = i + lag
        if 0 <= p2 < n:
            g[p2, k] -= 1.0
        p3 = j - lag
        if 0 <= p3 < n:
            g[p3, k] += 1.0
        p4 = j + lag
        if 0 <= p4 < n:
            g[p4, k] += 1.0

def greedy_binary(h0, n, deadline, top_k=10, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    stall = 0
    no_improve = 0
    
    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        g_z = g[zeros][:, top_idx]
        g_o = g[ones][:, top_idx]
        base = corr[top_idx][None, None, :]
        D = base + g_z[:, None, :] - g_o[None, :, :]
        new_maxes = D.max(axis=2)
        flat = int(new_maxes.argmin())
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = new_maxes[ii, jj]
        i = int(zeros[ii])
        j = int(ones[jj])
        new_c = best_val * dx
        
        if new_c < cur_c - 1e-14:
            h[i], h[j] = 1.0, 0.0
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            corr = corr + g[i] - g[j]
            update_g(g, i, j, n)
            cur_c = corr.max() * dx
            stall = 0
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 30 or (no_improve > 150 and stall > 50):
                nk = min(3, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    h[zeros[ri[a]]] = 1.0
                    h[ones[rj[a]]] = 0.0
                zeros = np.where(h == 0.0)[0].copy()
                ones = np.where(h == 1.0)[0].copy()
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                g = build_g(h, n)
                cur_c = corr.max() * dx
                stall = 0
    
    return best_h, best_c

def l_bfgsb_smooth(h, n, alpha, deadline):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = smooth_max(c, alpha)
        pen = 1e8 * (x.sum() - n / 2.0) ** 2
        return m + pen
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-9, 'maxfun': 10000})
        return project_feasible(res.x, n)

def projected_subgrad(h, n, deadline, lr0=0.1, decay=0.999):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    no_improve = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        c_curr = corr.max() * dx
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star = int(np.argmax(corr))
        grad = dx * h * 0  # placeholder
        # grad = dx * g[:, k_star] where g = build_g(h, n)
        # But we can compute it directly:
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            fwd = i + lag
            bwd = i - lag
            if 0 <= fwd < n:
                grad[i] += (1.0 - h[fwd])
            if 0 <= bwd < n:
                grad[i] -= h[bwd]
        grad *= dx
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        h = h_new
        lr *= decay
        
        if no_improve > 500:
            break
    
    return best_h, best_c
```

```python
    lags = k_range - (n - 1)
    # For i: positions p where p+lag=i or p-lag=i
    p_fwd_i = i - lags  # p+lag=i → p=i-lag
    p_bwd_i = i + lags  # p-lag=i → p=i+lag
```
