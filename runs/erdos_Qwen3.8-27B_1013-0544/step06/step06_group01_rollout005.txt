        if c_cd2 < best_c:
            h, best_c = h_cd2, c_cd2
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline - 5,
                                [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
    if time.time() < deadline - 5:
        h_cd3, c_cd3 = fast_cd(h, n, max_passes=300, grid=400,
                               deadline=deadline - 2, rng=rng, penalty=0.1)
        h_cd3 = project_feasible(h_cd3, n)
        c_cd3 = compute_c5(h_cd3, n)
        if c_cd3 < best_c:
            h, best_c = h_cd3, c_cd3
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    alloc1 = min(350, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize(h_init, n, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    # Try symmetric perturbation
    if time.time() < deadline - 200:
        h_sym = best_h.copy()
        h_sym = 0.5 * (h_sym + h_sym[::-1])
        h_sym = project_feasible(h_sym, n)
        c_sym = compute_c5(h_sym, n)
        if c_sym < best_c:
            best_c, best_h = c_sym, h_sym
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc > 30:
            h_opt, c_opt = optimize(h_sym, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Multi-restart from best
    if time.time() < deadline - 150:
        remaining = deadline - time.time() - 60
        n_restart = max(2, int(remaining / 80))
        for r in range(n_restart):
            if time.time() > deadline - 70:
                break
            scale = 0.01 + 0.04 * (r / max(1, n_restart - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(70, (deadline - time.time() - 40) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    # Final polish
    if time.time() < deadline - 15:
        h_f = fista_phase(best_h, n, time.time() + 30, max_iter=1000)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        best_h, best_c = fast_cd(best_h, n, max_passes=500, grid=500,
                                 deadline=deadline - 2, rng=rng, penalty=0.01)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
