import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum using bisection."""
    lo = h.min() - target_sum - 1
    hi = h.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining diff
        if diff > 0:
            room = 1.0 - h_p
            nonzero = room > 1e-10
            if nonzero.sum() > 0:
                h_p[nonzero] += diff / nonzero.sum() * room[nonzero] / room[nonzero].max()
        else:
            nonzero = h_p > 1e-10
            if nonzero.sum() > 0:
                h_p[nonzero] += diff / nonzero.sum()
    return np.clip(h_p, 0.0, 1.0)

def gradient_smooth_max(h, n, alpha, top_k=20):
    """Compute gradient of smooth max (top-k version)."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Focus on top-k lags only
    k_idx = np.argsort(c)[-top_k:]
    c_top = c[k_idx]
    max_c = c_top.max()
    w = np.exp(alpha * (c_top - max_c))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, p in enumerate(k_idx):
        wp = w[idx]
        if wp < 1e-8:
            continue
        m = p  # lag
        j = np.arange(n)
        jp = j - m
        jm = j + m
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        if mask_p.any():
            g[mask_p] += (1.0 - h[jp[mask_p]])
        if mask_m.any():
            g[mask_m] -= h[jm[mask_m]]
        grad += wp * g
    return grad, float(c.max())

def pgd_optimize(h, n, deadline, alpha=100, top_k=10, max_iter=5000):
    """Projected gradient descent on smooth max."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    eta = 0.005
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, cur_max = gradient_smooth_max(h, n, alpha, top_k)
        
        # Try a few step sizes
        best_step_h = None
        best_step_c = cur_max
        for eta_try in [eta, eta * 0.5, eta * 2, eta * 0.1]:
            h_trial = h - eta_try * grad
            h_trial = project_feasible(h_trial, target_sum)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_step_c:
                best_step_c = c_trial
                best_step_h = h_trial
                eta = eta_try
        
        if best_step_h is not None:
            h = best_step_h
            if best_step_c < best_c:
                best_c = best_step_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Reduce step size
            eta *= 0.5
        
        if no_improve > 20:
            break
    
    return best_h, best_c

def swap_local_search(h, n, deadline, max_sweeps=5000):
    """Local search using best-improvement swaps."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cur_max = c.max()
        k_star = np.argmax(c)
        
        # Compute gradient at k_star
        m = k_star
        j = np.arange(n)
        jp = j - m
        jm = j + m
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        if mask_p.any():
            g[mask_p] = (1.0 - h[jp[mask_p]])
        if mask_m.any():
            g[mask_m] -= h[jm[mask_m]]
        
        # Find best swap: move from high g to low g
        # Source: where g is high and h > 0
        # Dest: where g is low and h < 1
        src_candidates = np.where(h > 0.001)[0]
        dst_candidates = np.where(h < 0.999)[0]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            break
        
        best_gain = 0
        best_i = -1
        best_j = -1
        
        # Check top sources against top destinations
        top_src = src_candidates[np.argsort(g[src_candidates])[-10:]]
        top_dst = dst_candidates[np.argsort(g[dst_candidates])[:10]]
        
        for si in top_src:
            for di in top_dst:
                if si == di:
                    continue
                # Gain = g[si] - g[di] (we want to reduce c[k_star])
                gain = g[si] - g[di]
                if gain > best_gain:
                    best_gain = gain
                    best_i = di
                    best_j = si
        
        if best_gain <= 0 or best_i < 0:
            break
        
        # Perform swap
        eps = min(h[best_j], 1.0 - h[best_i]) * 0.5
        h[best_i] += eps
        h[best_j] -= eps
        h = np.clip(h, 0.0, 1.0)
        
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
        else:
            h[best_i] -= eps
            h[best_j] += eps
            break
    
    return best_h, best_c

def random_perturb(h, n, k_swaps, rng):
    """Random perturbation: k random swaps."""

        h_p[j] -= eps
    h_p = np.clip(h_p, 0.0, 1.0)
    h_p = project_feasible(h_p, target_sum)
    return h_p

Let me reconsider the swap_local_search. The issue is that computing the full correlation at each step is O(n²). For n=250, that's 62500 operations per step. With 20000 steps, that's 1.25 billion operations, which might take 10-20 seconds in numpy. That's acceptable.

But actually, I can be smarter: after a swap, only the lags affected by the swap change. A swap of positions i and j only affects lags k where |k - i| < n and |k - j| < n... which is all of them. So I can't really save computation here.

However, I can use the gradient to avoid recomputing the full correlation. After a small swap, the new correlation is approximately:
c_new[p] ≈ c[p] + ε * (∂c[p]/∂h[i] - ∂c[p]/∂h[j])

I can use this to quickly evaluate candidate swaps, and only compute the exact correlation when I actually perform a swap.

def fast_swap_search(h, n, deadline, max_candidates=500):
    """Find best swap using gradient approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    k_star = int(np.argmax(c))
    
    # Compute gradient at k_star
    k = k_star - (n - 1)
    j = np.arange(n)
    jp = j - k
    jm = j + k
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    g = np.zeros(n)
    if mask_p.any():
        g[mask_p] = (1.0 - h[jp[mask_p]])
    if mask_m.any():
        g[mask_m] -= h[jm[mask_m]]
    
    # Also compute gradients for top few lags
    top_lags = np.argsort(c)[-5:]
    all_g = np.zeros((len(top_lags), n))
    for li, p in enumerate(top_lags):
        kl = p - (n - 1)
        jpl = j - kl
        jml = j + kl
        mpl = (jpl >= 0) & (jpl < n)
        mml = (jml >= 0) & (jml < n)
        gl = np.zeros(n)
        if mpl.any():
            gl[mpl] = (1.0 - h[jpl[mpl]])
        if mml.any():
            gl[mml] -= h[jml[mml]]
        all_g[li] = gl
    
    # Find best (i, j) swap
    # We want to reduce c[k_star] most, subject to not increasing max
    src_candidates = np.where(h > 0.01)[0]
    dst_candidates = np.where(h < 0.99)[0]
    
    if len(src_candidates) < 2 or len(dst_candidates) < 2:
        return h, compute_c5(h, n)
    
    # Rank sources by g (want high g to move from)
    # Rank dests by g (want low g to move to)
    src_rank = src_candidates[np.argsort(g[src_candidates])[::-1]][:20]
    dst_rank = dst_candidates[np.argsort(g[dst_candidates])[:20]]
    
    best_gain = 0
    best_i, best_j = -1, -1
    best_eps = 0
    
    for si in src_rank:
        for di in dst_rank:
            if si == di:
                continue
            # Gain for k_star
            gain_star = g[si] - g[di]
            if gain_star <= 0:
                continue
            # Check top lags don't worsen
            ok = True
            for li in range(len(top_lags)):
                p = top_lags[li]
                change = all_g[li, si] - all_g[li, di]
                if c[p] + gain_star * change / gain_star * 0.5 > cur_max + 1e-6:
                    # This is an approximation; skip
                    pass
            # Max eps we can do
            eps = min(h[si], 1.0 - h[di]) * 0.5
            if eps > best_eps and gain_star > best_gain:
                best_gain = gain_star
                best_i = di
                best_j = si
                best_eps = eps
    
    if best_i < 0:
        return h, compute_c5(h, n)
    
    # Perform swap and verify
    h[best_i] += best_eps
    h[best_j] -= best_eps
    h = np.clip(h, 0, 1)
    new_c = compute_c5(h, n)
    
    if new_c > cur_max * dx + 1e-10:
        # Revert
        h[best_i] -= best_eps
        h[best_j] += best_eps
    
    return h, compute_c5(h, n)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    """Project to [0,1]^n with sum = target_sum."""
    lo = h.min() - target_sum - 1
    hi = h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine correction
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h_p
            mask = room > 1e-12
            if mask.sum() > 0:
                h_p[mask] = np.clip(h_p[mask] + diff / room[mask].sum() * room[mask], 0, 1)
        else:
            mask = h_p > 1e-12
            if mask.sum() > 0:
                h_p[mask] = np.clip(h_p[mask] + diff / h_p[mask].sum() * h_p[mask], 0, 1)
    return h_p

def smooth_max_and_grad(h, n, alpha, top_k=15):
    """Return (smooth_max_value, gradient) of top-k smooth max."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_idx = np.argsort(c)[-top_k:]
    c_top = c[k_idx]
    max_c = c_top.max()
    w = np.exp(alpha * (c_top - max_c))
    w /= w.sum()
    
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(len(k_idx)):
        wp = w[idx]
        if wp < 1e-10:
            continue
        p = k_idx[idx]
        k = p - (n - 1)
        jp = j - k
        jm = j + k
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        if mp.any():
            grad[mp] += wp * (1.0 - h[jp[mp]])
        if mm.any():
            grad[mm] -= wp * h[jm[mm]]
    return float(c.max()), grad

def pgd_with_momentum(h, n, deadline, alpha=100, top_k=15, beta=0.9):
    """PGD with momentum."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    eta = 0.01
    no_improve = 0
    
    while time.time() < deadline:
        _, grad = smooth_max_and_grad(h, n, alpha, top_k)
        v = beta * v + grad
        h_trial = h - eta * v
        h_trial = project_feasible(h_trial, target_sum)
        c_trial = compute_c5(h_trial, n)
        
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
            no_improve = 0
            # Can increase eta slightly
            eta = min(eta * 1.1, 0.05)
        else:
            no_improve += 1
            eta *= 0.8
            h = h_trial  # Still move, even if worse (momentum helps)
            # Actually, let's keep h if it's not too bad
            if c_trial > best_c * 1.01:
                h = best_h.copy()
                v = np.zeros(n)
        
        if no_improve > 100:
            # Perturb and reset

            h = best_h + 0.01 * rng_local.randn(n)
            h = project_feasible(h, target_sum)
            v = np.zeros(n)
            eta = 0.01
            no_improve = 0
