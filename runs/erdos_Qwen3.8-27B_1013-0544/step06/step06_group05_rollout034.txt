```python
def binary_sa(h_init, n, deadline, n_iter=500000, rng=None):
    """Simulated annealing for binary swap problem with fast updates."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    
    best_h = h.copy()
    best_max = cur_max
    
    lags = np.arange(-(n-1), n)
    
    # Temperature schedule
    t_start = cur_max / 10.0
    t_end = 1e-10
    n_steps = n_iter
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Choose a random 0-position and 1-position
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta
        idx1 = i + lags
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - lags
        v2 = (idx2 >= 0) & (idx2 < n)
        idx3 = j + lags
        v3 = (idx3 >= 0) & (idx3 < n)
        idx4 = j - lags
        v4 = (idx4 >= 0) & (idx4 < n)
        
        delta = np.zeros(2*n-1)
        delta[v1] += (1.0 - h[idx1[v1]])
        delta[v2] -= h[idx2[v2]]
        delta[v3] -= (1.0 - h[idx3[v3]])
        delta[v4] += h[idx4[v4]]
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        # Temperature
        t = t_start * (t_end / t_start) ** (step / n_steps)
        
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(t, 1e-15)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, float(best_max * dx)
```

```python
def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def sa_binary(h_init, n, deadline, n_trials=2000000, seed=42):
    """Fast simulated annealing for binary swap problem."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    
    lags = np.arange(-(n-1), n)
    n_lags = 2 * n - 1
    half = n // 2
    
    # Precompute index arrays for efficiency
    # For speed, we'll use a simpler approach: just evaluate the max change
    
    t_start = 5.0  # in "count" units
    t_end = 0.01
    t = t_start
    
    cooling = (t_end / t_start) ** (1.0 / n_trials)
    
    for step in range(n_trials):
        if time.time() > deadline:
            break
        if step % 1000 == 0:
            t = t_start * cooling ** step
        
        # Pick random swap
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b or h[a] == h[b]:
            continue
        
        # Ensure a is the 0→1 position
        if h[a] == 1:
            a, b = b, a
        # Now h[a]=0, h[b]=1
        
        # Compute delta quickly - only need to check if new_max < cur_max + t
        # For speed, compute new_corr and its max
        idx1 = a + lags
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = a - lags
        v2 = (idx2 >= 0) & (idx2 < n)
        idx3 = b + lags
        v3 = (idx3 >= 0) & (idx3 < n)
        idx4 = b - lags
        v4 = (idx4 >= 0) & (idx4 < n)
        
        new_corr = corr.copy()
        new_corr[v1] += (1.0 - h[idx1[v1]])
        new_corr[v2] -= h[idx2[v2]]
        new_corr[v3] -= (1.0 - h[idx3[v3]])
        new_corr[v4] += h[idx4[v4]]
        
        new_max = float(new_corr.max())
        
        if new_max <= cur_max:
            h[a] = 1.0
            h[b] = 0.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # Accept with probability exp(-(new_max - cur_max) / t)
            if rng.random() < np.exp(-(new_max - cur_max) / max(t, 1e-10)):
                h[a] = 1.0
                h[b] = 0.0
                corr = new_corr
                cur_max = new_max
    
    return best_h, float(best_max * dx)

def greedy_best_swap(h_init, n, deadline, max_sweeps=500):
    """Greedy: for each 0-position, find the best 1-position to swap with."""
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            # Compute A_i(ℓ) = contribution of making h[i]=1
            idx1 = i + lags
            v1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - lags
            v2 = (idx2 >= 0) & (idx2 < n)
            
            A = np.zeros(2*n-1)
            A[v1] += (1.0 - h[idx1[v1]])
            A[v2] -= h[idx2[v2]]
            
            # Critical lags: where corr + A > cur_max
            critical = (corr + A > cur_max)
            if not critical.any():
                # Any j works, pick first one
                j = ones[0]
                # Just verify
                idx3 = j + lags
                v3 = (idx3 >= 0) & (idx3 < n)
                idx4 = j - lags
                v4 = (idx4 >= 0) & (idx4 < n)
                delta = A.copy()
                delta[v3] -= (1.0 - h[idx3[v3]])
                delta[v4] += h[idx4[v4]]
                new_max = float((corr + delta).max())
                h[i] = 1.0
                h[j] = 0.0
                corr += delta
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                ones = ones[np.where(ones != j)]
                improved = True
                continue
            
            # For critical lags, find best j
            best_j = -1
            best_new_max = cur_max + 1.0
            
            for j in ones:
                idx3 = j + lags
                v3 = (idx3 >= 0) & (idx3 < n)
                idx4 = j - lags
                v4 = (idx4 >= 0) & (idx4 < n)
                
                # Only check critical lags
                B = np.zeros(2*n-1)
                B[v3] = -(1.0 - h[idx3[v3]])
                B[v4] += h[idx4[v4]]
                
                # Check if feasible: corr + A + B <= cur_max for all lags
                # Fast check on critical lags first
                crit_vals = corr[critical] + A[critical] + B[critical]
                if crit_vals.max() > cur_max:
                    continue
                
                # Full check
                new_max = float((corr + A + B).max())
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_j = j
            
            if best_j >= 0:
                idx3 = best_j + lags
                v3 = (idx3 >= 0) & (idx3 < n)
                idx4 = best_j - lags
                v4 = (idx4 >= 0) & (idx4 < n)
                delta = A.copy()
                delta[v3] -= (1.0 - h[idx3[v3]])
                delta[v4] += h[idx4[v4]]
                h[i] = 1.0
                h[best_j] = 0.0
                corr += delta
                cur_max = best_new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                ones = ones[np.where(ones != best_j)]
                improved = True
        
        if not improved:
            break
    
    return best_h, float(best_max * dx)

def slsqp_polish(h, n, deadline, maxiter=5000):
    """SLSQP on continuous relaxation."""
    dx = 2.0 / n
    def obj_and_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        # Soft max
        alpha = 5000.0
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        soft = m + np.log(np.exp(shifted).sum()) / alpha
        return soft
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_block_pattern(n, block_ones, block_zeros):
    """Create a repeating block pattern."""
    period = block_ones + block_zeros
    pattern = np.zeros(n)
    for i in range(n):
        pos_in_period = i % period
        pattern[i] = 1.0 if pos_in_period < block_ones else 0.0
    # Adjust to have exactly n/2 ones
    n_ones = int(pattern.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (from the end of one-blocks)
        ones_idx = np.where(pattern == 1)[0]
        for idx in ones_idx[target:]:
            pattern[idx] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(pattern == 0)[0]
        for idx in zeros_idx[:target - n_ones]:
            pattern[idx] = 1.0
    return pattern
```

For the subgradient method:
- The objective is f(h) = max_k C(k) * dx
- The subgradient at h is: for the maximizing k*, ∂f/∂h[i] = dx * [(1-h[i+k*])δ(i valid for first) - h[i-k*]δ(i valid for second)]

Wait, C(k) = ∑_i h[i] * (1-h[i+k])
∂C(k)/∂h[i] = (1-h[i+k]) + h[i] * (-∂h[i+k]/∂h[i])
For i ≠ i+k: ∂C(k)/∂h[i] = (1-h[i+k])
For i = i+k (i.e., k=0): ∂C(0)/∂h[i] = (1-2h[i])

More precisely:
C(k) = ∑_{i: i+k valid} h[i] * (1-h[i+k])

∂C(k)/∂h[j] = (1-h[j+k]) [if j+k valid, j ≠ j+k i.e. k≠0]
             + ∑_{i: i+k=j} h[i] * (-1) [the derivative of (1-h[i+k]) w.r.t. h[j] when i+k=j]
             = (1-h[j+k])_{valid, k≠0} - h[j-k]_{valid}

For k=0:
C(0) = ∑_i h[i]*(1-h[i]) = ∑_i h[i] - h[i]²
∂C(0)/∂h[j] = 1 - 2h[j]

So the subgradient of f(h) = max_k C(k)*dx at h is:
For the set S = {k : C(k) = max_k C(k)}:
∂f/∂h[j] = dx * (1/|S|) * ∑_{k∈S} [(1-h[j+k])_{valid, k≠0} - h[j-k]_{valid}]

(For k=0: use 1-2h[j] instead)

This gives a direction to decrease the max. I can do projected gradient descent:
h_new = h - step * subgradient
h_new = clip(h_new, 0, 1)
h_new = project to sum constraint

This is O(n * |S|) per step, which is fast.

```python
def subgradient_polish(h_init, n, deadline, n_iter=100000):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    lags = np.arange(-(n-1), n)
    
    # Learning rate schedule
    lr0 = 0.1
    lr_end = 0.001
    decay = (lr_end / lr0) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        lr = lr0 * decay ** it
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = corr.max()
        
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        
        # Find maximizing lags
        S = np.where(corr >= c_max - 1e-10)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in S:
            k = k_idx - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * h)
            else:
                # (1-h[j+k]) for valid j+k
                jk = np.arange(n) + k
                valid = (jk >= 0) & (jk < n)
                g1 = np.zeros(n)
                g1[valid] = 1.0 - h[jk[valid]]
                grad += g1
                # -h[j-k] for valid j-k
                jk2 = np.arange(n) - k
                valid2 = (jk2 >= 0) & (jk2 < n)
                g2 = np.zeros(n)
                g2[valid2] = h[jk2[valid2]]
                grad -= g2
        
        grad /= len(S)
        grad *= dx
        
        # Update
        h = h - lr * grad
        
        # Project: clip to [0,1] and enforce sum constraint
        h = np.clip(h, 0.0, 1.0)
        # Project onto sum = n/2
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - h
                s = room.sum()
                if s > 1e-10:
                    h = np.clip(h + diff / s * room, 0.0, 1.0)
            else:
                s = h.sum()
                if s > 1e-10:
                    h = np.clip(h + diff / s * h, 0.0, 1.0)
    
    return best_h, best_c
```

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-10:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb

def directed_sa(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    lags = np.arange(-(n-1), n)
    n_lags = 2 * n - 1
    t = 4.0
    step = 0
    
    while time.time() < deadline:
        step += 1
        if step % 1000 == 0:
            t = max(4.0 * (0.999 ** step), 0.05)
        
        # Choose swap
        if rng.random() < 0.5 and cur_max > 0:
            # Smart: target max lags
            max_mask = corr >= cur_max - 0.5
            max_lags = np.where(max_mask)[0]
            if len(max_lags) == 0:
                max_lags = np.where(corr == cur_max)[0]
            k_star = max_lags[rng.randint(len(max_lags))]
            ell = k_star - (n - 1)
            
            # Find 1-positions contributing to C(ell)
            if ell >= 0:
                contrib_pos = np.where((h[:n-ell] == 1) & (h[ell:] == 0))[0]
            else:
                contrib_pos = np.where((h[-ell:] == 1) & (h[:n+ell] == 0))[0]
            
            # Find 0-positions that won't increase C(ell)
            zeros = np.where(h == 0)[0]
            if ell >= 0:
                good_zeros = zeros[(zeros + ell < n) & (h[zeros + ell] == 1)]
            else:
                good_zeros = zeros[(zeros - ell < n) & (h[zeros - ell] == 1)]
            
            if len(contrib_pos) > 0 and len(good_zeros) > 0:
                i = contrib_pos[rng.randint(len(contrib_pos))]  # 1→0
                j = good_zeros[rng.randint(len(good_zeros))]    # 0→1
                a, b = j, i
            elif len(contrib_pos) > 0:
                i = contrib_pos[rng.randint(len(contrib_pos))]
                j = zeros[rng.randint(len(zeros))]
                a, b = j, i
            else:
                a = rng.randint(n)
                b = rng.randint(n)
        else:
            a = rng.randint(n)
            b = rng.randint(n)
        
        if a == b or h[a] == h[b]:
            continue
        if h[a] == 1:
            a, b = b, a  # a is 0→1, b is 1→0
        
        # Compute delta
        idx1 = a + lags
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = a - lags
        v2 = (idx2 >= 0) & (idx2 < n)
        idx3 = b + lags
        v3 = (idx3 >= 0) & (idx3 < n)
        idx4 = b - lags
        v4 = (idx4 >= 0) & (idx4 < n)
        
        delta = np.zeros(n_lags)
        delta[v1] += (1.0 - h[idx1[v1]])
        delta[v2] -= h[idx2[v2]]
        delta[v3] -= (1.0 - h[idx3[v3]])
        delta[v4] += h[idx4[v4]]
        
        new_max = float((corr + delta).max())
        
        if new_max <= cur_max:
            h[a] = 1.0
            h[b] = 0.0
            corr += delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        elif rng.random() < np.exp(-(new_max - cur_max) / max(t, 1e-10)):
            h[a] = 1.0
            h[b] = 0.0
            corr += delta
            cur_max = new_max
    
    return best_h, float(best_max * dx)

def subgradient_opt(h_init, n, deadline):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c = float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0-h, mode='full') * dx
        c_max = float(corr.max())
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        
        S = np.where(corr >= c_max - 1e-12)[0]
        grad = np.zeros(n)
        for k_idx in S:
            k = k_idx - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * h)
            else:
                g1 = np.zeros(n)
                if k > 0:
                    g1[:n-k] = 1.0 - h[k:]
                else:
                    g1[-k:] = 1.0 - h[:n+k]
                grad += g1
                g2 = np.zeros(n)
                if k > 0:
                    g2[k:] = h[:n-k]
                else:
                    g2[:n+k] = h[-k:]
                grad -= g2
        grad /= len(S)
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step_size = min(0.5 / gnorm, 5.0 / max(c_max, 1e-10))
        h = h - step_size * grad * dx
        
        h = np.clip(h, 0.0, 1.0)
        # Project onto sum = n/2
        target = n / 2.0
        d = target - h.sum()
        if d > 1e-10:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-10:
                h = np.clip(h + d/s * room, 0.0, 1.0)
        elif d < -1e-10:
            s = h.sum()
            if s > 1e-10:
                h = np.clip(h + d/s * h, 0.0, 1.0)
    
    return best_h, best_c

def slsqp_polish(h, n, deadline, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full') * dx
        m = g.max()
        alpha = 2000.0
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':cons_fun}],
                       options={'maxiter':maxiter,'ftol':1e-16})
        return project_feasible(res.x, n)
    except:
        return h
