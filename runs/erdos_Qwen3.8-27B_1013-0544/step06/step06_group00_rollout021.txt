            break

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

        return h

```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    c = corr * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_shifted = np.exp(shifted - sm)
    weights = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    # For each shift k (index in the full correlation array)
    for k in range(2*n - 1):
        if weights[k] < 1e-12:
            continue
        s = k - (n - 1)  # the actual shift
        # ∂c_k/∂h[j] = dx * [(1-h[j-s])*1_{0<=j-s<n} - h[j+s]*1_{0<=j+s<n}]
        # Vectorized:
        # First term: for j where 0 <= j-s < n, i.e., s <= j < s+n
        j_start = max(0, s)
        j_end = min(n, s + n)
        if j_start < j_end:
            grad[j_start:j_end] += weights[k] * dx * (1 - h[j_start-s:j_end-s])
        # Second term: for j where 0 <= j+s < n, i.e., -s <= j < n-s
        j_start2 = max(0, -s)
        j_end2 = min(n, n - s)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= weights[k] * dx * h[j_start2+s:j_end2+s]
    
    return grad
```

This is O(n^2) in the worst case (when all weights are significant), but with the early termination (weights[k] < 1e-12), it'll be much faster in practice for large alpha.

def slsqp_smooth_grad(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def jac(x):
        return compute_gradient(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
