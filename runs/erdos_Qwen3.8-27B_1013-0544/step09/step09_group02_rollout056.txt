    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

def nesterov_smooth(h, n, alphas, deadline, max_iter_per_alpha=2000):
    """Nesterov-accelerated gradient descent on smooth approximation."""
    h = h.copy()
    y = h.copy()  # Nesterov lookahead
    beta = 0.9  # momentum
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        beta_t = 1.0  # initial momentum parameter
        y = h.copy()
        step = 0.05  # initial step size
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            # Compute gradient at y (the lookahead point)
            y_feas = project_feasible(y, n)
            grad = smooth_gradient(y_feas, n, alpha)
            
            # Nesterov update
            new_h = y - step * grad
            new_h = project_feasible(new_h, n)
            
            # Check improvement
            # (expensive to check full C5 every iteration, so check every 10)
            if it % 20 == 0:
                c_curr = compute_c5(y_feas, n)
                if c_curr < best_c:
                    best_h = y_feas.copy()
                    best_c = c_curr
            
            # Update momentum
            t = it + 1
            beta_t = (1 + np.sqrt(1 + 4 * beta_t**2)) / 2
            beta_new = (1 - 1/beta_t) / (1 + 1/beta_t)  # decaying momentum
            y = new_h + beta_new * (new_h - h)
            y = project_feasible(y, n)
            h = new_h
            
            # Adaptive step size
            if it % 50 == 0 and it > 0:
                # Try to increase step if we're making progress
                step = min(0.5, step * 1.1)
            
    final_c = compute_c5(best_h, n)
    return best_h, final_c

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=500, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        j1lo = max(0, s)
        j1hi = min(n, s + n)
        if j1lo < j1hi:
            idx = np.arange(j1lo, j1hi)
            grad[idx] += wk * (1.0 - h[idx - s])
        j2lo = max(0, -s)
        j2hi = min(n, n - s)
        if j2lo < j2hi:
            idx = np.arange(j2lo, j2hi)
            grad[idx] -= wk * h[idx + s]
    return grad

def smooth_obj_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def nesterov_opt(h, n, alphas, deadline, iters_per_alpha=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        h_prev = h_cur.copy()
        y = h_cur.copy()
        step = 0.03
        beta = 0.85
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            y_feas = project_feasible(y, n)
            g = smooth_gradient(y_feas, n, alpha)
            new_h = y_feas - step * g
            new_h = project_feasible(new_h, n)
            if it % 15 == 0:
                c_chk = compute_c5(new_h, n)
                if c_chk < best_c:
                    best_c = c_chk
                    best_h = new_h.copy()
            h_prev = h_cur.copy()
            h_cur = new_h
            y = h_cur + beta * (h_cur - h_prev)
            y = np.clip(y, 0.0, 1.0)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            if it > 0:
                step = min(0.3, step * 1.02)
    return best_h, best_c

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=500, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def gradient_perturb(h, n, n_pert, rng, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    top_k = np.where(c >= m - 0.01 * (2.0 / n))[0]
    directions = []
    for k in top_k:
        s = k - (n - 1)
        g = np.zeros(n)
        j1lo, j1hi = max(0, s), min(n, s + n)
        if j1lo < j1hi:
            g[j1lo:j1hi] += (1.0 - h[j1lo - s:j1hi - s])
        j2lo, j2hi = max(0, -s), min(n, n - s)
        if j2lo < j2hi:
            g[j2lo:j2hi] -= h[j2lo + s:j2hi + s]
        directions.append(-g)
    if not directions:
        return project_feasible(h + rng.normal(0, 0.005, n), n)
    d = np.mean(directions, axis=0)
    dnorm = np.linalg.norm(d)
    if dnorm < 1e-15:
        return project_feasible(h + rng.normal(0, 0.005, n), n)
    d = d / dnorm
    scale = 0.02
    hp = project_feasible(h + scale * d + rng.normal(0, 0.003, n), n)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)

    if c1 < best_c:

    # Phase 2: Nesterov smooth optimization (key new component)
    if time.time() < deadline - 300:

        h2, c2 = nesterov_opt(best_h, n, alphas, deadline, iters_per_alpha=2500)
        if c2 < best_c:

        h2b, c2b = coord_cd(best_h, n, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if c2b < best_c:
            best_h, best_c = h2b, c2b

    # Phase 3: Multi-resolution with Nesterov at each level
    x_cur = np.linspace(0, 2, len(best_h))
    for n_try in [350, 500, 800]:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=100, deadline=deadline, shuffle=True, rng=rng)
        if time.time() < deadline - 100:
            alphas_small = [500, 3000]
            h_opt2, c_opt2 = nesterov_opt(h_opt, n_try, alphas_small, deadline, iters_per_alpha=800)
            if c_opt2 < c_opt:

            h_opt, c_opt = coord_cd(h_opt, n_try, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 4: Gradient-directed perturbations + CD
    n_cur = len(best_h)
    for r in range(25):
        if time.time() > deadline - 60:
            break
        if r < 10:
            hp = gradient_perturb(best_h, n_cur, 5, rng, deadline)
        elif r < 18:
            s = 0.002 + 0.003 * (r - 10)
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        else:
            hp = best_h.copy()
            blk = max(10, n_cur // 15)
            start = rng.randint(0, n_cur - blk)
            hp[start:start + blk] = np.clip(hp[start:start + blk] + rng.uniform(-0.04, 0.04, blk), 0, 1)
            hp = project_feasible(hp, n_cur)

            best_h, best_c = h_r, c_r

    # Phase 5: Final Nesterov + aggressive CD
    if time.time() < deadline - 40:
        alphas_final = [10000, 30000]
        hf, cf = nesterov_opt(best_h, n_cur, alphas_final, deadline, iters_per_alpha=1500)
        if cf < best_c:

    hfin, cfin = coord_cd(best_h, n_cur, max_passes=600, deadline=deadline, shuffle=True, rng=rng)
    if cfin < best_c:
        best_h, best_c = hfin, cfin

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
