```python
def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            h = np.clip(h + diff / n, 0, 1)
            continue
        h[active] += diff / active.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    """Evaluate spectral representation h(x) = 0.5 + sum of harmonics at n points."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += coeffs[2*(k-1)] * np.cos(k * np.pi * x) + coeffs[2*(k-1)+1] * np.sin(k * np.pi * x)
    return h

    h = eval_spectral(coeffs, n, K)
    h = np.clip(h, 0.0, 1.0)
    # Penalty for out-of-bounds
    penalty = np.sum(np.maximum(0, h - 1)**2 + np.maximum(0, -h)**2) * 100
    c = get_correlations(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    lse = m + (1.0/alpha) * np.log(np.sum(np.exp(shifted)))
    return lse + penalty

def spectral_optimize(h_init, n, K_list, deadline):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    x = np.linspace(0, 2, n, endpoint=False)
    
    # Initial spectral coefficients via least squares
    for K in K_list:
        if time.time() > deadline - 60:
            break
        # Get initial coefficients from current best
        c_init = np.zeros(2 * K)
        for k in range(1, K + 1):

            c_init[2*(k-1)+1] = np.mean((best_h - 0.5) * np.sin(k * np.pi * x))

                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_h = h_new
                    best_c = c_new
            except Exception:
                pass
    return best_h, best_c
```

= dx * (sum_k w[n-1+k] - sum_k w[n-1+k]*h[p-k]*[0<=p-k<n] - sum_k w[n-1+k]*h[p+k]*[0<=p+k<n])

The first term is just sum(w) * dx = dx (since weights are normalized).

The second and third terms are convolutions. Let me use FFT:
- For the second term: (w * h)[p] where the convolution is sum_k w[k]*h[p-k]

```python
    # = sum_i h[i] * w_ext[i + n - 1 - p]
    # This is like a cross-correlation: corr(h, w_ext)[n-1-p]

    # = sum_i w_ext[i] * h[i + p - len(w_ext) + len(h) - 1]... this is getting confusing.
    
    # Let me just use: np.correlate(h, w_ext, 'full') for the second term
    # And np.correlate(w_ext, h, 'full') for the third term (with appropriate indexing)
    
    # Actually, let me verify:
    # np.correlate(a, b, 'full')[k] = sum_i a[i+k] * b[i] (for the "full" mode)
    # Wait no. For full mode with len(a)=M, len(b)=N:
    # result has length M+N-1
    # result[k] = sum_i a[i+k-(N-1)] * b[i] for valid i

    # result[k] = sum_i h[i + k - (2n-2)] * w_full[i] for valid i
    # Hmm, this doesn't match what I want.
    
    # Let me just use the simple O(n^2) approach but vectorized with broadcasting:
    # For n up to 2000, this is 4*10^6 operations per gradient eval, which is fine.
    
    # Create a matrix where entry [p, k] represents the contribution of lag k to position p
    # k ranges from -(n-1) to n-1, index k_idx = k + n - 1 ranges from 0 to 2n-2
    
    p = np.arange(n)
    k = np.arange(-(n-1), n)  # shape (2n-1,)
    # p[k_row] - k[col] and p[k_row] + k[col]
    pk_minus = p[:, None] - k[None, :]  # (n, 2n-1)
    pk_plus = p[:, None] + k[None, :]   # (n, 2n-1)
    mask_minus = (pk_minus >= 0) & (pk_minus < n)  # (n, 2n-1)
    mask_plus = (pk_plus >= 0) & (pk_plus < n)
    
    # grad = dx * (1 - sum_k w[k]*h[p-k]*mask_minus - sum_k w[k]*h[p+k]*mask_plus)
    # The h[p-k] values: for each p, we need h[p-k] for valid k
    # Use advanced indexing:
    # h_minus[p, k_idx] = h[p - k[k_idx]] if mask_minus[p, k_idx] else 0
    # This requires clipping indices...
    
    # Alternative: use np.take with mode='clip' and then mask
    idx_minus = np.clip(pk_minus, 0, n-1)
    idx_plus = np.clip(pk_plus, 0, n-1)
    h_minus = h[idx_minus]  # (n, 2n-1)
    h_plus = h[idx_plus]
    
    term1 = h_minus * weights[None, :] * mask_minus
    term2 = h_plus * weights[None, :] * mask_plus
    
    grad = dx * (1.0 - term1.sum(axis=1) - term2.sum(axis=1))
    grad += 2 * lambda_penalty * (h.sum() - n/2)
    return grad
```

OK this is getting very long. Let me simplify my approach.

For the actual optimization, let me use a simpler but effective strategy:

1. **L-BFGS-B with smoothed max**: Use L-BFGS-B with the log-sum-exp objective and a penalty for the sum constraint. The gradient can be computed in O(n*2n) = O(n^2) which is fine for n≤2000.

2. **Multi-scale**: Start at n=200, optimize, then interpolate to n=400, optimize, etc.

3. **Coordinate descent polish**: After L-BFGS-B, use the exact coordinate descent from the previous code.

4. **Anti-symmetric perturbation**: Add a step that specifically targets the lags with highest correlation.

Let me write a cleaner version:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        na = active.sum()
        if na == 0:
            h = np.clip(h + diff/n, 0, 1)
            continue
        h[active] += diff / na
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    w = np.exp(s - s.max())
    w /= w.sum()
    val = m + np.log(w.sum()) / alpha  # = m + lse(c)/alpha, where lse is log-sum-exp of shifted
    # Actually: lse(c) = m + log(sum(exp(alpha*(c-m))))/alpha
    # So val = m + log(sum(exp(alpha*(c-m))))/alpha
    # = m + log(w.sum())/alpha + ... hmm let me redo
    # lse_approx = m + (1/alpha)*log(sum(exp(alpha*(c-m))))
    # = m + (1/alpha)*log(sum(exp(s)))
    # = m + (1/alpha)*(s.max() + log(sum(exp(s - s.max()))))
    # Since s = alpha*(c-m), s.max() = alpha*(c.max()-m) = 0
    # So lse_approx = m + (1/alpha)*log(sum(exp(s)))
    
    lse_approx = m + np.log(np.sum(np.exp(s))) / alpha
    penalty = lam * (h.sum() - n/2)**2
    val = lse_approx + penalty
    
    # Gradient
    # dw/dc_j = alpha * w_j (for the lse part, the gradient of lse_approx w.r.t. c_j is w_j)
    # Actually d/dc_j [m + (1/alpha)*log(sum(exp(alpha*(c-m))))] 
    # If m = c_{j*} (the max), then dm/dc_j = 1 if j=j*, 0 otherwise
    # This is non-differentiable at the kinks. For simplicity, use the smooth approximation:
    # gradient of lse_approx w.r.t. c_j = w_j (the softmax weights)
    # (This is the standard result for log-sum-exp)
    
    # So grad_lse_h[p] = sum_j w_j * dc_j/dh[p]
    # dc_j/dh[p] = dx * [(1-h[p-k_j])*I(0<=p-k_j<n) - h[p+k_j]*I(0<=p+k_j<n)]
    # where k_j = j-(n-1)
    
    # Vectorized computation:
    p = np.arange(n)
    k = np.arange(-(n-1), n)
    pm = p[:, None] - k[None, :]
    pp = p[:, None] + k[None, :]
    mask_m = (pm >= 0) & (pm < n)
    mask_p = (pp >= 0) & (pp < n)
    idx_m = np.clip(pm, 0, n-1)
    idx_p = np.clip(pp, 0, n-1)
    hm = h[idx_m]  # (n, 2n-1)
    hp = h[idx_p]
    
    # sum_j w_j * [(1-h[p-k_j])*I - h[p+k_j]*I]
    # = sum_j w_j - sum_j w_j*h[p-k_j]*I_m - sum_j w_j*h[p+k_j]*I_p
    t1 = (hm * w[None, :] * mask_m).sum(axis=1)
    t2 = (hp * w[None, :] * mask_p).sum(axis=1)
    
    grad = dx * (1.0 - t1 - t2) + 2 * lam * (h.sum() - n/2)
    return val, grad

def lbfgsb_optimize(h0, n, alpha, lam, maxiter, deadline):
    def fun(x):
        return lse_obj_and_grad(x, n, alpha, lam)[0]
    def jac(x):
        return lse_obj_and_grad(x, n, alpha, lam)[1]
    
    bounds = [(0, 1)] * n
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12,
                   'maxls': 50, 'disp': False})
    return res.x

def coord_descent_fast(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c_full = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        order = rng.permutation(n)
        improved = False
        for p in order:
            h0v = h[p]
            if h0v < 1e-12 or h0v > 1.0 - 1e-12:
                continue
            # Find best value for h[p]
            # c_full[j] changes linearly with h[p]: c_j = a_j * h[p] + b_j
            # where a_j = dx * [(1-h[p-k_j]) - h[p+k_j]] (for affected j)
            # and b_j = c_j - a_j * h0v
            # We want to minimize max_j (a_j * x + b_j) over x in [0,1]
            # Also c_0 (j=n-1) has a special term: h[p]*(1-h[p]) contributes
            
            k = np.arange(-(n-1), n)
            j = n - 1 + k  # indices in c_full
            
            pm = p - k
            pp = p + k
            mask_m = (pm >= 0) & (pm < n)
            mask_p = (pp >= 0) & (pp < n)
            
            # a_j = dx * [(1-h[p-k_j])*mask_m - h[p+k_j]*mask_p]
            idx_m = np.clip(pm, 0, n-1)
            idx_p = np.clip(pp, 0, n-1)
            a = (2.0/n) * ((1.0 - h[idx_m]) * mask_m - h[idx_p] * mask_p)
            
            # But for j = n-1 (k=0): a = dx * (1-h[p] - h[p]) = dx*(1-2*h[p])
            # And c_{n-1} includes the term h[p]*(1-h[p]) which is quadratic
            # Let's handle k=0 specially
            # c_{n-1} = sum_i h[i]*(1-h[i]) * dx
            # d c_{n-1}/d h[p] = dx * (1-2*h[p])
            # So a_{n-1} = dx*(1-2*h[p]) and b_{n-1} = c_{n-1} - a_{n-1}*h[p]
            # But h[p]*(1-h[p]) = h[p] - h[p]^2
            # c_{n-1} includes this term, so c_{n-1} = (sum_{i≠p} h[i]*(1-h[i]) + h[p] - h[p]^2) * dx
            # If we change h[p] to x: c_{n-1}(x) = (sum_{i≠p} h[i]*(1-h[i]) + x - x^2) * dx
            # This is quadratic in x, not linear!
            # So the CD approach needs to handle this.
            
            # For k≠0, c_j is linear in h[p]: c_j(x) = a_j*x + b_j
            # For k=0: c_{n-1}(x) = c_{n-1} + dx*(x - h[p]) - dx*(x^2 - h[p]^2)
            # = c_{n-1} + dx*(x-h[p]) - dx*(x-h[p])(x+h[p])
            # Let d = x - h[p]: c_{n-1}(x) = c_{n-1} + dx*d - dx*d*(x+h[p])
            # = c_{n-1} + dx*d*(1 - x - h[p])
            # This is quadratic: c_{n-1} + dx*d - dx*(x^2 - h[p]^2)
            # = c_{n-1} - dx*(x^2 - h[p]^2) + dx*(x - h[p])
            
            # For simplicity, let's just use bisection on x in [0,1]:
            # f(x) = max_j c_j(x) where c_j(x) = a_j*x + b_j for j≠n-1,
            # and c_{n-1}(x) = c_{n-1} + dx*(x-h[p] - x^2 + h[p]^2)
            
            # Use bisection: f(x) is convex? Not necessarily. But it's unimodal in practice.
            # Let's just try a few candidate values:
            # The optimal x satisfies: the max of the linear functions is minimized
            # This is a linear programming problem in 1D: minimize t s.t. a_j*x+b_j <= t, 0<=x<=1
            # The optimal x is where two constraints are tight (or at a boundary)
            
            # Simple approach: bisection 30 iterations
            lo, hi = 0.0, 1.0
            def f_of_x(x):
                vals = a * x
                # Add b terms
                # c_j = a_j * h[p] + b_j, so b_j = c_j - a_j * h[p]
                b = c_full - a * h0v
                vals = a * x + b
                # Fix k=0 term (j = n-1):
                k0_idx = n - 1  # index where k=0
                vals[k0_idx] = c_full[k0_idx] + (2.0/n) * (x - h0v - x*x + h0v*h0v)
                return vals.max()
            
            # Bisection: find x that minimizes f_of_x
            # f_of_x is the max of linear (and one quadratic) functions => convex
            # So bisection on derivative works, or just ternary search
            best_x, best_v = h0v, f_of_x(h0v)
            for _ in range(40):
                x1 = lo + (hi - lo) / 3
                x2 = hi - (hi - lo) / 3
                v1, v2 = f_of_x(x1), f_of_x(x2)
                if v1 < v2:
                    hi = x2
                else:
                    lo = x1
                mid = (lo + hi) / 2
                vm = f_of_x(mid)
                if vm < best_v:
                    best_v, best_x = vm, mid
            
            if abs(best_x - h0v) > 1e-10:
                h[p] = best_x
                improved = True
        
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif not improved:
            break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        na = active.sum()
        if na == 0:
            h = np.clip(h + diff / n, 0, 1)
            continue
        h[active] += diff / na
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    w = np.exp(s - s.max())
    w /= w.sum()
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += lam * (h.sum() - n / 2.0) ** 2
    p = np.arange(n)
    k = np.arange(-(n-1), n)
    pm = p[:, None] - k[None, :]
    pp = p[:, None] + k[None, :]
    mm = (pm >= 0) & (pm < n)
    mp = (pp >= 0) & (pp < n)
    im = np.clip(pm, 0, n-1)
    ip = np.clip(pp, 0, n-1)
    t1 = (h[im] * w[None, :] * mm).sum(axis=1)
    t2 = (h[ip] * w[None, :] * mp).sum(axis=1)
    grad = dx * (1.0 - t1 - t2) + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgsb_step(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        def fun(x):
            return lse_obj_grad(x, n, alpha, 80.0)[0]
        def jac(x):
            return lse_obj_grad(x, n, alpha, 80.0)[1]
        try:
            res = minimize(fun, best_h, jac=jac, method='L-BFGS-B',
                           bounds=[(0,1)]*n,
                           options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
        except Exception:
            pass
    return best_h, best_c

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    mask = s != 0
    s = s[mask]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    all_m = np.arange(2 * n - 1)
    m_min, m_max = n - 1 + s[0], n - 1 + s[-1]
    unaff = (all_m < m_min) | (all_m > m_max)
    M_un = float(c[unaff].max()) if unaff.any() else 0.0
    c0b = c[n-1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_un > am + 1e-14:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        msl = float(a[active].min())
        xsl = float(a[active].max())
        if msl > 1e-12:
            hi = mid
        elif xsl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo+hi)*0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_un, c0b + x - x*x)
        if v < best_v:
            best_v, best_x = v, float(x)
    return best_x

def coord_descent(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10:
                h[p] = nv
                improved += 1
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
        elif improved == 0:
            break
    return h, best_c

def anneal_step(h, n, alpha, lam, lr, T, rng, deadline):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = c.max()
    s = alpha * (c - m)
    w = np.exp(s - s.max())
    w /= w.sum()
    topk = max(20, len(w) // 5)
    idx = np.argsort(w)[-topk:]
    ws = w[idx].copy()
    ws /= ws.sum()
    lags = idx - (n - 1)
    grad = np.zeros(n)
    for lag, wi in zip(lags, ws):
        if wi < 1e-12:
            continue
        p1, p2 = max(0, lag), min(n, n + lag)
        if p2 > p1:
            grad[p1:p2] += wi * (1.0 - h[p1-lag:p2-lag])
        p3, p4 = max(0, -lag), min(n, n - lag)
        if p4 > p3:
            grad[p3:p4] -= wi * h[p3+lag:p4+lag]
    grad *= (2.0 / n)
    hl = h - lr * grad + T * rng.randn(n) * 0.01
    hl = project_feasible(hl, n)
    return hl

            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [400, 800, 1000, 1200]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 2000, deadline - 80, seed + n_try)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new

    # Phase 5: Perturbation restarts with CD
    n_cur = len(best_h)
    for r in range(10):
        if time.time() > deadline - 40:
            break
        s = 0.008 + 0.02 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 1500, deadline - 30, seed + 200 + r * 7)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 6: Final CD
    if time.time() < deadline - 10:
        hf, cf = coord_descent(best_h, len(best_h), 5000, deadline - 5, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Check 1-h symmetry
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
```
