from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    lse = mx + np.log(np.sum(np.exp(c - mx)))
    return (lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    if unaff_max > result:

    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    exp_c = np.exp(c - mx)
    W = exp_c / exp_c.sum()  # softmax weights
    grad = np.zeros(n)
    for k in range(len(c)):

                bwd = -h[:n-nl]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):

    lse = mx + np.log(np.sum(np.exp(c - mx)))
    return (lse / alpha) * (2.0 / n)

            grad += w * (1.0 - 2.0 * h)
        elif l > 0:
            nl = l
            grad[:n-nl] += w * (1.0 - h[nl:])
            grad[nl:] += -w * h[:n-nl]
        else:
            nl = -l
            grad[:n-nl] += w * (1.0 - h[nl:])
            grad[nl:] += -w * h[:n-nl]
    return grad * dx / alpha
