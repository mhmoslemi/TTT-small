def simultaneous_cd(h, n, max_iter=500, deadline=None, omega=0.5):
    """Simultaneous coordinate descent (Jacobi-style)."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        dx_val = 2.0 / n
        
        # Compute optimal for each coordinate (independent, given other coords fixed)
        h_new = h.copy()
        for p in range(n):
            # The contribution to the active lag from coordinate p
            # is h[p]*(1-h[p+m_star-(n-1)]) + h[p-(m_star-(n-1))]*(1-h[p])
            # when indices are valid

            jb = p - m_star + (n - 1)

            grad = 0.0
            if 0 <= jf < n:
                grad += (1.0 - h[jf])
            if 0 <= jb < n:

            h_new[p] = h[p] - omega * grad * dx_val

            best_c = c_new
            h = h_new
        else:
            omega *= 0.8
            if omega < 0.01:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=200, deadline=None, rng=None, shuffle=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            mlo = max(0, n - 1 - p)
            mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
            ms = np.arange(mlo, mhi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            vec = np.zeros(len(ms))
            if np.any(vf):
                vec[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                vec[vb] -= h[jb[vb]]
            c_sub = c[mlo:mhi + 1]
            unaff = 0.0
            if mlo > 0:
                unaff = max(unaff, float(c[:mlo].max()))
            if mhi < 2 * n - 2:
                unaff = max(unaff, float(c[mhi + 1:].max()))
            lo, hi = -h0, 1.0 - h0

                active = vals >= mval - 1e-9
                vmin = float(vec[active].min())
                vmax = float(vec[active].max())
                if vmin > 1e-9:
                    hi = mid
                elif vmax < -1e-9:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            delta = (lo + hi) * 0.5
            if abs(delta) > 1e-12:
                darr = np.zeros(len(ms))
                if np.any(vf):
                    darr[vf] += delta * (1.0 - h[jf[vf]])
                if np.any(vb):
                    darr[vb] -= delta * h[jb[vb]]
                c[ms] += darr
                h[p] = np.clip(h[p] + delta, 0.0, 1.0)
                improved += 1
        if improved == 0:

        if p_idx % 20 == 19:
            c = np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:

def subgrad_step(h, n, lr, c=None):
    if c is None:
        c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    sg = np.zeros(n)
    for i in range(n):
        j = i + m_star - (n - 1)
        if 0 <= j < n:
            sg[i] += (1.0 - h[j])
        ip = i - m_star + (n - 1)
        if 0 <= ip < n:
            sg[i] -= h[ip]
    return sg
