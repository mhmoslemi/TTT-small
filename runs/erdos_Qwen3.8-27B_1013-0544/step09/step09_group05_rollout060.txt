  This is the same as the original integral but with shift -k instead of k. Since we take the max over all k, the C_5 value is the same.
  
  So we can exploit this symmetry: average h with its "dual" to get a symmetric solution that has the same C_5.
  
  h_sym(x) = (h(x) + 1 - h(2-x)) / 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True, nbisect=35):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad_c = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        kf = k - (n - 1)
        for i in range(n):
            jf = i + kf
            jb = i - kf
            if 0 <= jf < n:
                grad_c[k] += (1.0 - x[jf])
            if 0 <= jb < n:
                grad_c[k] -= x[jb]
    if kf == 0:
        for i in range(n):
            grad_c[k] += (1.0 - 2.0 * x[i])
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        kf = k - (n - 1)
        for i in range(n):
            jf = i + kf
            if 0 <= jf < n:
                grad[i] += weights[k] * (1.0 - x[jf]) * dx
            jb = i - kf
            if 0 <= jb < n:
                grad[i] -= weights[k] * x[jb] * dx
        if kf == 0:
            for i in range(n):
                grad[i] += weights[k] * (1.0 - 2.0 * x[i]) * dx
    return lse_obj(x, n, alpha), grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def symmetrize(h, n):
    x = np.linspace(0, 2, n)
    h_rev = np.interp(x, x[::-1], h)
    h_sym = 0.5 * (h + 1.0 - h_rev)
    return project_feasible(h_sym, n)

def subgrad_ada(h, n, steps, lr_base, deadline, rng):
    h = h.copy()
    grad_sq_cum = np.zeros(n)
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = c.argmax()
        k = idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            jf = i + k
            jb = i - k
            if 0 <= jf < n:
                grad[i] += (1.0 - h[jf])
            if 0 <= jb < n:
                grad[i] -= h[jb]
        if k == 0:
            for i in range(n):
                grad[i] += (1.0 - 2.0 * h[i])
        grad -= np.mean(grad)
        grad_sq_cum += grad ** 2
        lr = lr_base / np.sqrt(grad_sq_cum + 1e-8)
        h = project_feasible(h - lr * grad, n)
    return h
