def project_feasible(h, n):

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def update_c(c, h, p, new_val, n):
    """Update correlation c after changing h[p] to new_val. Returns new c."""
    delta = new_val - h[p]
    if abs(delta) < 1e-16:
        return c
    c = c.copy()
    # Affected m range
    m_lo = min(p, n-1-p)
    m_hi = n-1+p

        if abs(x_new - h0) > 1e-13:

def slsqp_solve(h, n, deadline):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        alpha = 2000.0
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        alpha = 2000.0
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = alpha * np.exp(shifted - mx)
        lse_grad = np.sum(weights)
        grad = np.zeros(n)
        for k in range(len(c)):
            w = (1.0 + (weights[k] / lse_grad)) * (2.0 / n)
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = i + k - (n - 1)
                if 0 <= j < n:
                    grad[i] += w * (1.0 - x[j] - x[i] + 2.0 * x[i] * (1.0 - x[j]))
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < compute_c5(h, n):

    return h, compute_c5(h, n)

                if c_curr < best_c - 1e-9:
                    best_h, best_c = h_curr, c_curr
                    stuck_count = 1
                elif stuck_count > 30:
                    break

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        remaining = deadline - time.time() - 3
        max_p = max(100, int(remaining / 0.05))
        hf, cf = coord_descent(best_h, n, max_p, deadline, None)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
