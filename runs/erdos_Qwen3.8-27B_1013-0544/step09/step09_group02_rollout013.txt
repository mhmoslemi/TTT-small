def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h

            for _ in range(3):
                for p in block:
                    d = exact_coord_delta(h, p, c, n)
                    if abs(d) > 1e-13:
                        apply_delta_to_corr(c, h, p, d, n)
                        h[p] = np.clip(h[p] + d, 0.0, 1.0)
                        improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n_init = 600

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD to convergence on parent
    h1, c1 = coord_cd(best_h, n_init, max_passes=300, deadline=deadline, shuffle=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP with multiple alphas (moderate to high)
    alphas_phase2 = [2000, 5000, 15000, 50000]
    for a in alphas_phase2:
        if time.time() > deadline - 200:
            break
        h_s = slsqp_optimize(best_h, n_init, a, 800, deadline)
        c_s = compute_c5(h_s, n_init)
        h_s, c_s = coord_cd(h_s, n_init, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # Phase 3: Multi-resolution (coarse to fine)
    x_cur = np.linspace(0, 2, len(best_h))
    for n_try in [200, 300, 400, 500, 700, 900]:
        if time.time() > deadline - 250:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 4: Block CD for escaping local optima
    n_cur = len(best_h)
    for bs in [2, 3, 4]:
        if time.time() > deadline - 200:
            break
        h_b, c_b = block_cd(best_h, n_cur, block_size=bs, max_passes=80, deadline=deadline, rng=rng)
        if c_b < best_c:
            best_h, best_c = h_b, c_b

    # Phase 5: Simulated annealing style perturbations + CD
    n_cur = len(best_h)
    temps = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
    for s in temps:
        if time.time() > deadline - 150:
            break
        for attempt in range(3):
            if time.time() > deadline - 100:
                break
            pert = rng.normal(0, s, n_cur)
            hp = project_feasible(best_h + pert, n_cur)
            h_r, c_r = coord_cd(hp, n_cur, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Phase 6: Focused CD on active lags
    n_cur = len(best_h)
    if time.time() < deadline - 100:
        h_f, c_f = focused_cd(best_h, n_cur, top_k=30, max_passes=300, deadline=deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 7: Final SLSQP with very high alpha + CD
    if time.time() < deadline - 60:
        for a in [100000, 300000]:
            if time.time() > deadline - 30:
                break
            h_s = slsqp_optimize(best_h, len(best_h), a, 1000, deadline)
            c_s = compute_c5(h_s, len(best_h))
            h_s, c_s = coord_cd(h_s, len(best_h), max_passes=300, deadline=deadline, shuffle=True, rng=rng)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Phase 8: Final CD polish
    n_cur = len(best_h)
    h_final, c_final = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
    if c_final < best_c:
        best_h, best_c = h_final, c_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
