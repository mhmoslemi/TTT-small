        return

    c[n-1] += new_val * (1.0 - new_val) - old_val * (1.0 - old_val)

    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    c[m] += delta * a
