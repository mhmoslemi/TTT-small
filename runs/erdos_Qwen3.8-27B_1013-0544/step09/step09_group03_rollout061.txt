```python
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
```

```python
def subgradient(h, n, top_k=5):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_max = c.max()
    # Active set: indices where c is close to max
    active = np.where(c >= m_max - 1e-10)[0]
    # Limit to top_k for stability
    if len(active) > top_k:
        top_idx = np.argsort(-c[active])[:top_k]
        active = active[top_idx]
    
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)  # actual shift
        # term 1: (1-h[p-s]) for p-s in [0,n)
        p_range = np.arange(max(0, s), min(n, n + s))  # p such that 0 ≤ p-s < n
        # Actually: 0 ≤ p-s < n → s ≤ p < n+s, and 0 ≤ p < n → 0 ≤ p < n
        # So p ∈ [max(0,s), min(n, n+s)) = [max(0,s), n)
        p_lo = max(0, s)
        p_hi = min(n, n + s)  # = n since s can be negative
        # Wait, if s < 0: p-s ≥ 0 → p ≥ s (automatically true since p ≥ 0 > s)
        # p-s < n → p < n+s. If s < 0, n+s < n, so p < n+s.
        # So for s < 0: p ∈ [0, n+s)
        # For s ≥ 0: p ∈ [s, n)
        if s >= 0:
            p_lo, p_hi = s, n
        else:
            p_lo, p_hi = 0, n + s
        if p_hi > p_lo:
            grad[p_lo:p_hi] += (1.0 - h[p_lo - s : p_hi - s])
        
        # term 2: -h[p+s] for p+s in [0,n)
        # 0 ≤ p+s < n → -s ≤ p < n-s, and 0 ≤ p < n
        # If s ≥ 0: p ∈ [0, n-s)
        # If s < 0: p ∈ [-s, n)
        if s >= 0:
            p_lo2, p_hi2 = 0, n - s
        else:
            p_lo2, p_hi2 = -s, n
        if p_hi2 > p_lo2:
            grad[p_lo2:p_hi2] -= h[p_lo2 + s : p_hi2 + s]
    
    grad /= len(active)
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    mask = s != 0
    s = s[mask]
    if len(s) == 0 or h0 < 1e-12 or h0 > 1.0 - 1e-12:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m].copy() - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:

                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)
