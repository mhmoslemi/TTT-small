def block_search(h, n, deadline, rng):

    h = h.copy()
    best_c = compute_c5(h, n)

        h_new = h.copy()
        h_new[p:p+bs] += delta
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

                h = h_new
            T = max(0.0005, T * 0.995)
    return h, best_c

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            if tot < 1e-15:
