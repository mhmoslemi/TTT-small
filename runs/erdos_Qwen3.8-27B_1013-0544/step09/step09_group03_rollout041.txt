   c = np.correlate(x,1-x,'full')
   m = float(np.max(c))

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    if n <= 0:
        return 10.0
    return float(np.max(compute_corr(h)) * (2.0 / float(n)))

def sum_with_shift(h, shift):
    return float(np.sum(np.clip(h + shift, 0.0, 1.0)))

def project_uniform(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        if sum_with_shift(h, mid) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(12):
        res = target - float(h.sum())
        if abs(res) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            idx = int(np.nonzero(mask)[0][0])
            h[idx] = min(1.0, max(0.0, h[idx] + res))
        else:
            if res > 0.0:
                if np.any(h < 1.0):
                    idx = int(np.nonzero(h < 1.0)[0][0])
                else:
                    idx = 0
            elif res < 0.0:
                if np.any(h > 0.0):
                    idx = int(np.nonzero(h > 0.0)[0][0])
                else:
                    idx = 0
            else:
                break
            h[idx] = min(1.0, max(0.0, h[idx] + res))
    return h

def resample_h(h, n_new):
    n_old = int(len(h))
    if n_new == n_old:
        return np.array(h, dtype=float, copy=True)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def make_coord_data(n):
    data = []
    nm1 = n - 1
    for p in range(n):
        s_min = min(-p, p - nm1)
        s_max = max(nm1 - p, p)
        s = np.arange(s_min, s_max + 1, dtype=np.int32)
        s = s[s != 0]
        if s.size:
            m_all = nm1 - s
            ps = p + s
            pm = p - s
            mps = (ps >= 0) & (ps < n)
            mpm = (pm >= 0) & (pm < n)
            aff = mps | mpm
            m_aff = m_all[aff].astype(np.int32, copy=False)
            ps_aff = ps[mps].astype(np.int32, copy=False)
            pm_aff = pm[mpm].astype(np.int32, copy=False)
        else:
            m_aff = np.empty(0, dtype=np.int32)
            ps_aff = np.empty(0, dtype=np.int32)
            pm_aff = np.empty(0, dtype=np.int32)
        m_min = int(nm1 - s_max)
        m_max = int(nm1 - s_min)
        data.append((m_aff, ps_aff, pm_aff, m_min, m_max))
    return data

def get_coord_data(n, cache):
    if n in cache:
        return cache[n]
    d = make_coord_data(n)
    if len(cache) > 5:
        cache.clear()
    cache[n] = d
    return d

def coord_descent(h_in, data, n, max_passes, deadline, seed):
    h = project_uniform(np.array(h_in, dtype=float, copy=True), n)
    c = compute_corr(h)
    cur_max = float(np.max(c))
    best_h = h.copy()
    best_c5 = cur_max * (2.0 / float(n))
    no_imp = 0
    rng = np.random.RandomState(seed)
    tol = 1e-11
    areps = 1e-12
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        changed = 0
        for p in order:
            h0 = float(h[p])
            m_aff, ps_aff, pm_aff, m_min, m_max = data[p]
            la = m_aff.size
            if la == 0:
                continue
            a = np.empty(la, dtype=float)
            a.fill(0.0)
            if ps_aff.size:
                a += 1.0 - h[ps_aff]
            if pm_aff.size:
                a -= h[pm_aff]
            M = 0.0
            if m_min > 0:
                M = float(np.max(c[:m_min]))
            if m_max + 1 < c.size:
                mv = float(np.max(c[m_max + 1:]))
                if mv > M:
                    M = mv
            c0_base = float(c[n - 1] - h0 * (1.0 - h0))
            T0 = cur_max
            lo = M

                lo = c0_base
            if lo < 0.0:
                lo = 0.0
            if T0 - lo <= 1e-10:

            base = c[m_aff] - a * h0
            pos_mask = a > areps
            neg_mask = a < -areps
            pos_a = a[pos_mask]
            pos_b = base[pos_mask]
            neg_a = a[neg_mask]
            neg_b = base[neg_mask]
            zero_mask = ~(pos_mask | neg_mask)
            if zero_mask.any():

            else:
                zero_b_max = -1e100
            hi = T0

                feasible = False
                if mid + tol >= M and mid + tol >= zero_b_max and mid + tol >= c0_base:

                    Ub = 1.0
                    if neg_a.size:
                        Lb = max(Lb, float(np.max((mid - neg_b) / neg_a)))
                    if pos_a.size:
                        Ub = min(Ub, float(np.min((mid - pos_b) / pos_a)))
                    Alo = Lb if Lb > 0.0 else 0.0
                    Ahi = Ub if Ub < 1.0 else 1.0
                    if Ahi + tol >= Alo:
                        gv = mid - c0_base
                        if gv >= 0.25 - tol:
                            feasible = True
                        else:
                            if gv < 0.0:
                                gv = 0.0
                            rr = math.sqrt(max(0.0, 1.0 - 4.0 * gv))
                            r1 = 0.5 * (1.0 - rr)
                            r2 = 0.5 * (1.0 + rr)
                            if Alo <= r1 + tol or Ahi >= r2 - tol:
                                feasible = True
                if feasible:
                    hi = mid
                else:
                    lo = mid
            gap = T0 - hi
            if gap <= 1e-9:
                continue
            margin = min(1e-8, 0.25 * gap)
            tt = hi + max(1e-10, margin)
            Lb = 0.0
            Ub = 1.0
            feasible_b = (tt + tol >= M and tt + tol >= zero_b_max and tt + tol >= c0_base)
            if neg_a.size:
                Lb = max(Lb, float(np.max((tt - neg_b) / neg_a)))
            if pos_a.size:
                Ub = min(Ub, float(np.min((tt - pos_b) / pos_a)))
            Alo = Lb if Lb > 0.0 else 0.0
            Ahi = Ub if Ub < 1.0 else 1.0
            if Ahi + tol < Alo:
                feasible_b = False
            else:
                gv = tt - c0_base
                if gv < 0.25 - tol:
                    if gv < 0.0:
                        gv = 0.0
                    rr = math.sqrt(max(0.0, 1.0 - 4.0 * gv))
                    r1 = 0.5 * (1.0 - rr)
                    r2 = 0.5 * (1.0 + rr)
                    if Alo > r1 + tol and Ahi < r2 - tol:
                        feasible_b = False
            if not feasible_b:
                continue
            cand = [h0, 0.0, 1.0]
            gv = tt - c0_base
            if gv >= 0.25 - tol:
                if Alo <= Ahi + tol:
                    cand += [Alo, Ahi, 0.5 * (Alo + Ahi), float(np.clip(h0, Alo, Ahi))]
            else:
                if gv < 0.0:
                    gv = 0.0
                rr = math.sqrt(max(0.0, 1.0 - 4.0 * gv))
                r1 = 0.5 * (1.0 - rr)
                r2 = 0.5 * (1.0 + rr)
                B = min(Ahi, r1)
                if Alo <= B + tol:
                    cand += [Alo, B, 0.5 * (Alo + B), float(np.clip(h0, Alo, B)), r1]
                A = max(Alo, r2)
                if A <= Ahi + tol:
                    cand += [A, Ahi, 0.5 * (A + Ahi), float(np.clip(h0, A, Ahi)), r2]
            best_x = h0
            best_val = T0
            for xv in cand:
                xx = float(np.clip(xv, 0.0, 1.0))
                am = float(np.max(a * xx + base))
                gv2 = c0_base + xx - xx * xx
                val = M
                if am > val:
                    val = am
                if gv2 > val:
                    val = gv2
                if val < best_val - 1e-11:
                    best_val = val
                    best_x = xx
                elif abs(val - best_val) <= 1e-11 and abs(xx - h0) < abs(best_x - h0):
                    best_val = val
                    best_x = xx

                c[m_aff] = a * best_x + base
                c[n - 1] = c0_base + best_x - best_x * best_x
                cur_max = best_val + 1e-8
                changed += 1
        h = project_uniform(h, n)
        c = compute_corr(h)
        cur_max = float(np.max(c))
        c5 = cur_max * (2.0 / float(n))
        if c5 < best_c5 - 1e-13:
            best_h = h.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
            if c5 > best_c5 + 1e-10:
                h = best_h.copy()
                c = compute_corr(h)
                cur_max = float(np.max(c))
        if no_imp >= 8:
            break
    return best_h, best_c5

def sum_con(x, n):
    return float(x.sum()) - 0.5 * float(n)

def sum_con_jac(x, n):
    return np.ones(len(x), dtype=float)

def smooth_obj(x, n, alpha):
    c = compute_corr(x)
    m = float(np.max(c))
    e = np.exp(alpha * (c - m))
    return (m + float(np.log(np.sum(e))) / alpha) * (2.0 / float(n))

def smooth_grad(x, n, alpha):
    c = compute_corr(x)
    m = float(np.max(c))
    e = np.exp(alpha * (c - m))
    z = float(np.sum(e))
    if z <= 0.0 or not math.isfinite(z):
        return np.zeros(n, dtype=float)
    q = e / z
    grad = np.zeros(n, dtype=float)
    grad += q[n - 1] * (1.0 - 2.0 * x)
    for r in range(1, n):
        wp = float(q[n - 1 - r])
        if wp > 1e-15:
            grad[:n - r] += wp * (1.0 - x[r:])
            grad[r:] -= wp * x[:n - r]
        wm = float(q[n - 1 + r])
        if wm > 1e-15:
            grad[r:] += wm * (1.0 - x[:n - r])
            grad[:r] -= wm * x[r:]
    return (2.0 / float(n)) * grad

def slsqp_polish(h0, n, deadline, maxiter=50):
    h = project_uniform(np.array(h0, dtype=float, copy=True), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [
        {"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}
    ]
    alphas = (3000, 15000)
    for alpha in alphas:
        if time.time() > deadline:
            break
        rem = max(0.0, deadline - time.time())
        mi = min(maxiter, max(20, int(rem / 3)))
        try:
            res = minimize(
                smooth_obj,
                best_h,
                args=(n, alpha),
                jac=smooth_grad,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(mi), "ftol": 1e-13},
            )
            h_new = project_uniform(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c
