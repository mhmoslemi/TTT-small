import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):

    c_base = c[ms].copy()

    c_diag = 0.0

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    best_local = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                return h, best_local
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        cur_c = compute_c5(h, n)
        if cur_c < best_local:
            best_local = cur_c
    return h, best_local

def compute_weighted_grad(h, n, p):
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-12:
            continue
        w = p[m]
        k = m - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - k:hi1 - k + 1])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + k:hi2 + k + 1]
    grad -= np.mean(grad)
    return grad

def expected_grad_opt(h, n, outer_iters, inner_iters, alpha, deadline):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    p = np.ones(2 * n - 1) / (2 * n - 1)
    for it in range(outer_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        f = c * dx
        p = np.exp(alpha * (f - f.max()))
        p /= p.sum()
        lr = 0.05
        for _ in range(inner_iters):
            if time.time() > deadline:
                break
            grad = compute_weighted_grad(h, n, p)
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-14:
                break
            step = lr / gnorm
            h_new = project_feasible(h - step * grad, n)
            c5_new = compute_c5(h_new, n)
            if c5_new <= best_c:
                h = h_new
            else:
                h = project_feasible(h - 0.5 * step * grad, n)
        c5 = compute_c5(h, n)
        if c5 < best_c:
            best_c = c5
    return h, best_c

def fourier_init(n, harm=3, amp=0.3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, harm + 1):
        h += (amp / k) * np.sin(k * np.pi * x)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def pair_perturb(h, n, rng, strength):
    h = h.copy()
    i = rng.randint(n)
    j = rng.randint(n - 1)
    if j >= i:
        j += 1
    if i == j:
        return h
    delta = strength * (h[i] - h[j]) * rng.uniform(0.5, 1.5)
    h[i] -= delta
    h[j] += delta
    return project_feasible(h, n)
