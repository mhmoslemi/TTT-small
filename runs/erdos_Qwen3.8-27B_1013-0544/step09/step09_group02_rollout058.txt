def focused_gradient_step(h, n, top_k=10, lr=0.01):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # Find top_k shifts
    top_idx = np.argsort(c)[-top_k:]
    # Compute subgradient
    sg = np.zeros(n)
    for m_idx in top_idx:
        s = m_idx - (n - 1)
        for j in range(n):
            val = 0.0
            jp = j + s
            jm = j - s
            if 0 <= jp < n:
                val += h[jp]
            if 0 <= jm < n:
                val += h[jm]
            sg[j] -= val
    # Project gradient onto constraint (sum = n/2)
    sg -= sg.mean()
    # Gradient step
    h_new = project_feasible(h - lr * sg, n)
    return h_new

Δc = δ*(1-h[j+s]) - δ*h[j-s] = δ*(1 - h[j+s] - h[j-s])

To reduce c, I want Δc < 0, so I want (1 - h[j+s] - h[j-s]) < 0, i.e., h[j+s] + h[j-s] > 1.

So the best coordinate to increase is the one where h[j+s] + h[j-s] is largest.
The best coordinate to decrease is the one where h[j+s] + h[j-s] is smallest.

This gives a directed search!

But we also need to maintain the sum constraint, so increasing one coordinate means decreasing another.

Let me formalize this as a "pair update":
- Find (j1, j2) such that increasing h[j1] and decreasing h[j2] most reduces the max shift
- This is a 2-coordinate update

The change in c[m] for a pair update (increase h[j1] by δ, decrease h[j2] by δ):
Δc[m] = δ*(1-h[j1+s_m]-h[j1-s_m]) - δ*(1-h[j2+s_m]-h[j2-s_m])
= δ*((h[j2+s_m]+h[j2-s_m]) - (h[j1+s_m]+h[j1-s_m]))

To minimize the max, we want to reduce the active shifts. For an active shift s:
We want (h[j2+s]+h[j2-s]) - (h[j1+s]+h[j1-s]) < 0, i.e., h[j1+s]+h[j1-s] > h[j2+s]+h[j2-s].

    h = h.copy()
    c = c.copy()
    for _ in range(max_pairs):
        if deadline is not None and time.time() > deadline:
            break
        m_max = c.max()
        # Find active shifts
        active_mask = c >= m_max - 1e-6
        active_idx = np.where(active_mask)[0]
        
        # Build score vector
        score = np.zeros(n)
        for m_idx in active_idx:
            s = m_idx - (n - 1)
            for i in range(n):
                ip = i + s
                im = i - s
                if 0 <= ip < n:
                    score[i] += h[ip]
                if 0 <= im < n:
                    score[i] += h[im]
        
        # Find best pair: i with max score (h[i]<1), j with min score (h[j]>0)
        valid_i = np.where(h < 1 - 1e-6)[0]

            break

def softmax_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    weights = exp_vals / exp_vals.sum()
    
    obj = (m + (1.0/alpha) * (mx + np.log(exp_vals.sum()))) * (2.0/n)
    
    grad = np.zeros(n)
    for m_idx in range(2*n-1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        # Contribution: w * (1 - h[j+s] - h[j-s]) for each j
        # The "1" part is constant across all j for this m
        grad += w  # This adds w to all components

        hi2 = min(n, n + s)
