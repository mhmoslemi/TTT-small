def block_solution(B, max_iter=5000):
    # B blocks, each of width 2/B
    # h[i] = v[i//block_size] where block_size = n//B
    # Constraint: sum(v) * (2/B) = 1, so sum(v) = B/2
    # 0 <= v[i] <= 1
    
    from scipy.optimize import minimize
    
    def obj_and_grad(p, B):
        v = p  # B values
        # Project to feasible
        v = np.clip(v, 0, 1)
        diff = B/2 - v.sum()
        if abs(diff) > 1e-10:
            # Simple projection
            v = v + diff / B  # crude
            v = np.clip(v, 0, 1)
        
        # Compute C5 on a fine grid
        n = B * 10  # fine grid
        h = np.repeat(v, 10)  # each block repeated 10 times
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full')
        return float(c.max() * dx)
    
    # Start from uniform
    v0 = np.full(B, 0.5)
    
    res = minimize(obj_and_grad, v0, args=(B,), method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
    
    v = res.x
    # Project
    v = np.clip(v, 0, 1)
    v = v + (B/2 - v.sum()) / B
    v = np.clip(v, 0, 1)
    
    return v, B

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha)

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):

            return lse_obj(x, n, alpha)

            return x.sum() - n / 2.0

        def cons_jac(x):
            return np.ones(n)

            c_alt = compute_c5(h_alt_cd, n_init)
            if c_alt < best_c:
                best_h, best_c = h_alt_cd, c_alt

    # Phase 8: Final SLSQP + CD polish
    if time.time() < deadline - 20:
        dl8 = deadline - 15
        h_f1, c_f1 = slsqp_polish(best_h, n_init, [10000, 50000], 200, dl8)
        if c_f1 < best_c:
            best_h, best_c = h_f1, c_f1
        if time.time() < deadline - 10:
            h_f2 = quick_cd(best_h, n_init, max_passes=200, deadline=deadline - 5, top_k=30)
            c_f2 = compute_c5(h_f2, n_init)
            if c_f2 < best_c:
                best_h, best_c = h_f2, c_f2

    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, n_init)
    c_flip = compute_c5(h_flip, n_init)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
