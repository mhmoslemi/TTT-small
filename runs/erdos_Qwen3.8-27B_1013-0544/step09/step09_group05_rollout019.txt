def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * dx * (c - c.max())  # alpha * c * dx, shifted for numerical stability
    w = np.exp(ac)
    W = w.sum()
    w = w / W  # normalized weights
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += w[k] * (1.0 - h[lag:])
            grad[lag:] -= w[k] * h[:n-lag]

            grad[alag:] += w[k] * (1.0 - h[:n-alag])
            grad[:n-alag] -= w[k] * h[alag:]

def project_feasible(h, n):

            if tot < 1e-15:

```python
def smooth_grad_fast(h, n, alpha, top_k=50):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * c
    scaled -= scaled.max()
    w = np.exp(scaled)
    
    # Only keep top-k significant weights
    W = w.sum()
    if W < 1e-300:
        top_idx = np.argsort(c)[-top_k:]
    else:
        w /= W
        # Get top-k indices by weight
        top_idx = np.argsort(w)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n-lag]
        else:
            alag = -lag
            grad[alag:] += wk * (1.0 - h[:n-alag])
            grad[:n-alag] -= wk * h[alag:]
    
    return grad * dx
```

With top_k=50, the loop is at most 50 iterations instead of 1399. This should be about 28x faster.

Actually, for large alpha, only the top 1-3 lags have significant weight, so top_k=50 is more than enough.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_grad_fast(h, n, alpha, top_k=100):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * c
    scaled -= scaled.max()
    w = np.exp(scaled)
    W = w.sum()
    if W < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= W
    # Top-k indices
    if top_k < 2*n-1:
        top_idx = np.argpartition(w, -top_k)[-top_k:]
    else:
        top_idx = np.arange(2*n-1)
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)

        else:
            alag = -lag
            grad[alag:] += wk * (1.0 - h[:n-alag])
            grad[:n-alag] -= wk * h[alag:]
    return grad * dx

def fista_optimize(h_init, n, alpha, deadline, min_iters=100):
    dx = 2.0 / n
    L = 2.0 * dx
    step = 0.9 / L
    h = h_init.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if it < min_iters and it % 100 == 0:
            pass  # let it run
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        y = h + ((t-1.0)/t_new) * (y - h)
        y = np.clip(y, 0, 1)
        grad = smooth_grad_fast(y, n, alpha)
        x_new = y - step * grad
        x_new = project_feasible(x_new, n)
        t = t_new
        h = x_new
        c = compute_c5(h, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 5000:
            break
    return best_h, best_c

def fista_anneal(h_init, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 300, 1000, 3000, 10000]
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_alphas = len(alphas)
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        t_alloc = deadline - remaining / (n_alphas - i) - 5
        h, c = fista_optimize(h, n, alpha, t_alloc)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
