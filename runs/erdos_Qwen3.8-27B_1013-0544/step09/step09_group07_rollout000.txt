import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=100)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd_thresh(h, n, max_passes=500, deadline=None, eps_start=0.02, eps_end=0.002, max_considers=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        frac = p / max(max_passes - 1, 1)
        eps = eps_start + (eps_end - eps_start) * frac
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            c_range = c[m_lo:m_hi + 1]
            threshold = current_max - eps
            mask = c_range > threshold
            if not mask.any():
                am = int(np.argmax(c_range))
                mask[am] = True
            n_masked = mask.sum()
            if n_masked > max_considers:
                vals = c_range[mask]
                ks = min(max_considers, n_masked)
                top_k_idx = np.argsort(vals)[-ks:]
                new_mask = np.zeros_like(mask)
                new_mask[take_idx[mask].take(top_k_idx)] = True
                mask = new_mask
            ms = ms_all[mask]
            unaff_mask = ~mask
            unaff_extra = float(c_range[unaff_mask].max()) if unaff_mask.any() else 0.0
            if m_lo > 0:
                unaff_extra = max(unaff_extra, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_extra = max(unaff_extra, float(c[m_hi + 1:].max()))
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_extra, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_extra)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def take_idx(arr):
    return np.where(arr)[0]

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_polish(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        sg = np.zeros(n)
        for i in range(n):
            j = i + m_star - (n - 1)
            if 0 <= j < n:
                sg[i] += (1.0 - h[j])
            ip = i - m_star + (n - 1)
            if 0 <= ip < n:
                sg[i] -= h[ip]
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.04, 0.08)
        else:
            lr *= 0.6
            stall += 1
            if stall > 40 or lr < 1e-12:
                break
    return h, best_c

def make_sinusoidal(n, freq=1.0, phase=0.0, amp=0.5):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)
