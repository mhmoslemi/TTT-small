        d = float(np.clip(d, lo, hi))
        v = f(d)
        if v < best_val - 1e-15:
            best_val = v
            best_d = d
    
    # Refine with golden section
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.3 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(f, gl, gh, maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    
    return best_d
