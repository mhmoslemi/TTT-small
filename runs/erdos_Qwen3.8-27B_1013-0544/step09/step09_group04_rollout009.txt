import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):

    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,

            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, order='sequential'):

    for passed in range(max_passes):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        if order == 'proximity':
            # Sort by how close affected values are to max
            sens = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n-1-p)
                m_hi = min(2*n-2, n-1+p)
                seg = c[m_lo:m_hi+1]
                sens[p] = np.max(seg)  # higher = more likely to be binding
            idx = n_idx[np.argsort(-sens)]
        elif order == 'sensitivity':
            sens = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n-1-p)
                m_hi = min(2*n-2, n-1+p)
                seg = c[m_lo:m_hi+1]
                sens[p] = np.std(seg) if len(seg) > 1 else 0.0
            idx = n_idx[np.argsort(-sens)]
        else:
            idx = n_idx
        improved = False

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1]+1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:

        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:

                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):

    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'proximity':
            sens = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                sens[p] = np.max(c[m_lo:m_hi + 1])
            idx = n_idx[np.argsort(-sens)]
        elif order == 'sensitivity':
            sens = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                seg = c[m_lo:m_hi + 1]
                sens[p] = np.std(seg) if len(seg) > 1 else 0.0
            idx = n_idx[np.argsort(-sens)]
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:

def informed_restart(h, n, rng, deadline, cd_passes=15):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    m_off = k_star - (n - 1)
    if m_off >= 0:
        pairs = [(i, i + m_off) for i in range(n - m_off) if 0 <= i + m_off < n]
    else:
        m_off = -m_off
        pairs = [(i + m_off, i) for i in range(n - m_off) if 0 <= i < n]
    if not pairs:
        return h, compute_c5(h, n)
    h_new = h.copy()
    contribs = []
    for i, j in pairs:
        contribs.append((h[i] * (1.0 - h[j]), i))
    contribs.sort(key=lambda x: -x[0])
    n_perturb = min(30, len(contribs))
    for _, i in contribs[:n_perturb]:
        if h_new[i] > 0.5:
            h_new[i] = max(0.0, h_new[i] - rng.uniform(0.3, 0.8))
        else:
            h_new[i] = min(1.0, h_new[i] + rng.uniform(0.3, 0.8))
    h_new = project_feasible(h_new, n)
    h_opt, c_opt = coord_cd(h_new, n, cd_passes, deadline, order='proximity')
    return h_opt, c_opt

            h_r, c_r = informed_restart(best_h, n, rng, deadline, cd_passes=12)
        elif attempt % 3 == 1:
            h_r, c_r = block_flip_restart(best_h, n, rng, deadline, cd_passes=12)
        else:
            h_r = project_feasible(best_h + rng.normal(0, 0.08, n), n)
            h_r, c_r = coord_cd(h_r, n, 10, deadline, order='sensitivity')
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    if time.time() < deadline - 15:
        h_f, c_f = coord_cd(best_h, n, 100, deadline, order='sequential')
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
