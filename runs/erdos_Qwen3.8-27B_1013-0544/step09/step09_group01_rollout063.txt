import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(x, S):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def _lsq_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return (m + mx / alpha + np.log(np.sum(np.exp(sh - mx))) / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    bh, bc = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_lsq_obj, bh, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            nh = project_box_sum(res.x, n / 2.0)
            nc = compute_c5(nh, n)
            if nc < bc:
                bh, bc = nh, nc
        except Exception:
            pass
    return bh, bc

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    return result if unaff_max <= result else unaff_max

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_d = len(dp) > 0
            d_idx = int(dp[0]) if has_d else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, d2)
            for _ in range(gs_iters):

                    d_hi, d2, f2 = d2, d1, f1

            d_opt = (d_lo + d_hi) / 2.0

            f_cur = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, 0.0)
            if f_opt < f_cur - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_box_sum(h, n / 2.0)
    return h, compute_c5(h, n)

def block_cd(h, n, max_passes, deadline, B=4, gs_iters=45):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n - B + 1, B):
            if time.time() > deadline:
                break
            for p in range(start, min(start + B, n)):
                if time.time() > deadline:
                    break
                c = np.correlate(h, 1.0 - h, mode='full')
                h0 = h[p]
                if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                    continue
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                ms = np.arange(m_lo, m_hi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                dp = np.where(ms == (n - 1))[0]
                has_d = len(dp) > 0
                d_idx = int(dp[0]) if has_d else -1
                c_aff = c[ms]
                unaff = 0.0
                if m_lo > 0:
                    unaff = max(unaff, float(c[:m_lo].max()))
                if m_hi < 2 * n - 2:
                    unaff = max(unaff, float(c[m_hi + 1:].max()))
                d_lo, d_hi = -h0, 1.0 - h0
                d1 = d_hi - gr * (d_hi - d_lo)
                d2 = d_lo + gr * (d_hi - d_lo)
                f1 = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, d1)
                f2 = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, d2)
                for _ in range(gs_iters):
                    if d_hi - d_lo < 1e-15:

                    else:
                        d_lo, d1, f1 = d1, d2, f2
                        d2 = d_lo + gr * (d_hi - d_lo)

                d_opt = (d_lo + d_hi) / 2.0
                f_opt = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, d_opt)
                f_cur = _cd_fval(c_aff, a, d_idx, has_d, unaff, h0, 0.0)
                if f_opt < f_cur - 1e-15:
                    h[p] = h0 + d_opt
                    improved = True
        h = project_box_sum(h, n / 2.0)

def momentum_step(h, n, alpha, lr, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        shift = k - (n - 1)
        p_arr = np.arange(n)
        jf = p_arr + shift
        jb = p_arr - shift
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        if np.any(vf):
            grad[vf] += w[k] * (1.0 - h[jf[vf]])
        if np.any(vb):
            grad[vb] -= w[k] * h[jb[vb]]
    grad *= (2.0 / n)
    return h, grad

def anneal_phase(h, n, deadline, rng, t0=0.002, t1=0.00005, steps=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    for s in range(steps):
        if time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (s / max(1, steps))
        k = rng.randint(1, 3)
        idx = rng.choice(n, size=k, replace=False)
        sigma = 0.003 + 0.005 * T / t0
        h_new = h.copy()
        h_new[idx] += rng.normal(0, sigma, size=k)
        h_new = project_box_sum(h_new, n / 2.0)
        c_new = compute_c5(h_new, n)
        if c_new <= cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-10)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_h = h.copy()
                best_c = c_new
    return best_h, best_c
