        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    """Exact 1D minimization for coordinate p given correlation values c."""
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-12
    pos_m = a > 1e-12
    zero_m = np.abs(a) <= 1e-12
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and float(np.max(c_base[zero_m])) > L:
            feas = False
        if feas and np.any(neg_m):
            vals = (L - c_base[neg_m]) / a[neg_m]
            x_lo = max(x_lo, float(np.max(vals)))
        if feas and np.any(pos_m):
            vals = (L - c_base[pos_m]) / a[pos_m]
            x_hi = min(x_hi, float(np.min(vals)))
        if x_lo > x_hi + 1e-14:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo, 0.0) > r1 and min(x_hi, 1.0) < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = abs(h0 - r1)
                d2 = abs(h0 - r2)
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-12:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, deadline, order='sequential'):
    c = np.correlate(h, 1.0 - h, mode='full')
    n_idx = np.arange(n)
    if order == 'random':
        np.random.shuffle(n_idx)
    for p in n_idx:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        x_new = solve_1d(c, h, p, n)
        if abs(x_new - h0) > 1e-13:
            # Update c incrementally
            delta = x_new - h0
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            a = np.zeros(len(ms))
            if np.any(jf >= 0) and np.any(jf < n):
                pass
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            di = np.where(ms == (n - 1))[0]
            if len(di) > 0:
                c[di] -= delta * delta
            h[p] = x_new
    return h, c

def cd_loop(h, n, max_passes, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max()) * (2.0 / n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, c = cd_pass(h, n, deadline)
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max()) * (2.0 / n)
        if cur < best_c:
            best_c = cur
        else:
            break
    return h, best_c

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = alpha * corr
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    # Gradient: w[m] * dc[m]/dh[p] for each p
    # dc[m]/dh[p] = dx * [(1-h[p+lag]) * 1{valid} - h[p-lag] * 1{valid}]
    # lag = m - (n-1)
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    for m in range(2*n-1):
        if w[m] < 1e-10:
            continue
        lag = lags[m]
        # First factor: p is first index, need 0<=p<n and 0<=p+lag<n
        # p ranges over [0, n-1] intersect [-lag, n-1-lag]
        p_lo = max(0, -lag)
        p_hi = min(n-1, n-1-lag)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += w[m] * dx * (1.0 - h[p_lo+lag:p_hi+1+lag])
        # Second factor: p is second index, p = i+lag, i ranges over [0,n-1]
        # So p ranges over [lag, n-1+lag] intersect [0, n-1]
        p_lo2 = max(0, lag)
        p_hi2 = min(n-1, n-1+lag)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= w[m] * dx * h[p_lo2-lag:p_hi2+1-lag]
    return grad

def proximal_gradient(h, n, deadline, alpha_start=50, alpha_end=2000, n_steps=200):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ds = np.linspace(np.log(alpha_start), np.log(alpha_end), n_steps)
    for si in range(n_steps):
        if time.time() > deadline:
            break
        alpha = np.exp(ds[si])
        # Backtracking line search
        eta = 1.0
        for _ in range(30):
            grad = smooth_gradient(h, n, alpha)
            h_trial = h - eta * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c:
                best_h = h_trial.copy()
                best_c = c_trial
                h = h_trial
                break
            eta *= 0.5
            if eta < 1e-10:
                break
        # If we improved, reset eta somewhat
    return best_h, best_c

def structured_perturb(h, n, rng, scale):
    """Create a perturbed version with structured noise."""
    h_new = h.copy()
    # Add low-frequency perturbation
    k = rng.randint(2, 5)
    freq = k * np.pi
    noise = scale * np.sin(freq * np.arange(n) / n * 2)
    h_new += noise
    # Add random perturbation
    h_new += rng.normal(0, scale * 0.5, n)
    return project_feasible(h_new, n)

            break

        h = project_feasible(h, n)
        cur = compute_c5(h, n)
        if cur >= best_c:
            break
        best_c = cur

def subgradient_step(h, n, step_size):

    c_max = c.max()
    active = c > c_max - 1e-8
    # Average gradient over active lags
    grad = np.zeros(n)
    dx = 2.0/n
    lags = np.arange(2*n-1) - (n-1)

        if not active[m]:
            continue
        w = 1.0 / active.sum()
        lag = lags[m]
        # dc[dm]/dh[p] for this m
        # First factor: p in [max(0,-lag), min(n-1, n-1-lag)]
        p_lo = max(0, -lag)
        p_hi = min(n-1, n-1-lag)
        if p_lo <= p_hi:

        # Second factor: p in [max(0,lag), min(n-1, n-1+lag)]
        p_lo2 = max(0, lag)
        p_hi2 = min(n-1, n-1+lag)
        if p_lo2 <= p_hi2:

    h_new = h - step_size * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1]+1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-12
    pos_m = a > 1e-12
    zero_m = np.abs(a) <= 1e-12
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feas = True
        if np.any(zero_m) and float(np.max(c_base[zero_m])) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-14:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo,0.0) > r1 and min(x_hi,1.0) < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                if abs(h0 - r1) < abs(h0 - r2):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-12:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass_vec(h, n, deadline, ordering):
    c = np.correlate(h, 1.0 - h, mode='full')
    for p in ordering:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        x_new = solve_1d(c, h, p, n)
        d = x_new - h0
        if abs(d) > 1e-13:
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += d * a
            di = np.where(ms == (n - 1))[0]
            if len(di) > 0:
                c[di] -= d * d
            h[p] = x_new
    return h

def cd_loop(h, n, max_passes, deadline, seed=0):
    rng = np.random.RandomState(seed)
    n_idx = np.arange(n)
    best_c = compute_c5(h, n)
    for psv in range(max_passes):
        if time.time() > deadline:
            break
        if psv % 4 == 0:
            ordering = n_idx
        elif psv % 4 == 1:
            ordering = n_idx.copy()
            rng.shuffle(ordering)
        elif psv % 4 == 2:
            ordering = n_idx[::-1]
        else:
            ordering = n_idx[::2]; ordering = np.concatenate([ordering, n_idx[1::2]])
            rng.shuffle(ordering)
        h = cd_pass_vec(h, n, deadline, ordering)
        h = project_feasible(h, n)
        cur = compute_c5(h, n)
        if cur >= best_c - 1e-15:

        best_c = cur
    return h, best_c

def subgradient_step(h, n, step_size):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    active = c > c_max - 1e-6
    cnt = active.sum()
    if cnt == 0:
        return h
    w = active.astype(float) / cnt
    grad = np.zeros(n)
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    for m in range(2 * n - 1):
        if not active[m]:
            continue
        lag = lags[m]
        p_lo = max(0, -lag)
        p_hi = min(n - 1, n - 1 - lag)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += w[m] * dx * (1.0 - h[p_lo+lag:p_hi+1+lag])
        p_lo2 = max(0, lag)
        p_hi2 = min(n - 1, n - 1 + lag)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= w[m] * dx * h[p_lo2-lag:p_hi2+1-lag]
    h_new = project_feasible(h - step_size * grad, n)
    return h_new

def subgrad_cd(h, n, deadline, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    for it in range(500):
        if time.time() > deadline:

        h = subgradient_step(h, n, step)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:

            h = best_h.copy()
        if step < 1e-5:

        h2, c2 = cd_loop(h, n, 5, min(time.time() + 5, deadline), seed=seed+it)
        if c2 < best_c:

    return best_h, best_c
