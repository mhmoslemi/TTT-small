def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return best_h, best_c

        def _cf(x, n_=n):
            return x.sum() - n_ / 2.0
        def _cj(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_sine_init(n, freq, phase=0.0, amp=0.3):
    i = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP + CD on parent (with bug fix)
    h1, c1 = slsqp_opt(h_best, n, [10000, 100000], 200, deadline)
    h1, c1 = exact_cd(h1, n, 200, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Multi-n exploration with structured inits
    n_list = [350, 400, 450, 500, 550, 600, 800]
    if n not in n_list:
        n_list.insert(0, n)
    for n_try in n_list:
        if time.time() > deadline - 300:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_init = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        inits = [h_init]
        for freq in [3, 5, 7, 11, 17]:
            inits.append(make_sine_init(n_try, freq, amp=0.25))
            inits.append(make_sine_init(n_try, freq, phase=np.pi / 4, amp=0.35))
        for hi in inits:
            if time.time() > deadline - 150:
                break
            hc = hi.copy()
            hc, cc = slsqp_opt(hc, n_try, [5000, 50000], 200, deadline)
            hc, cc = exact_cd(hc, n_try, 150, deadline)
            if cc < best_c:
                h_best, best_c = hc, cc

    # Phase 3: Extended CD on current best
    n_cur = len(h_best)
    h3, c3 = exact_cd(h_best, n_cur, 500, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Annealing-style perturbation + CD
    for r in range(15):
        if time.time() > deadline - 60:
            break
        s = max(0.002, 0.025 * (1.0 - r / 15.0) ** 2)
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        ho, co = exact_cd(hp, len(h_best), 120, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 5: Final SLSQP + CD polish
    if time.time() < deadline - 15:
        n_f = len(h_best)
        hf, cf = slsqp_opt(h_best, n_f, [100000, 500000], 300, deadline)
        hf, cf = exact_cd(hf, n_f, 300, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    # Phase 6: One last CD pass
    if time.time() < deadline - 3:
        hf, cf = exact_cd(h_best, len(h_best), 100, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
