import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_lo = min(-p, p - (n - 1))
    s_hi = max((n - 1) - p, p)
    s = np.arange(s_lo, s_hi + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_lo
    m_max = n - 1 + s_hi
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-13:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def binary_push(h, n, max_rounds=30, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        h_work = h.copy()
        c = np.correlate(h_work, 1.0 - h_work, mode='full')
        M = c.max()
        any_change = False
        order = rng.permutation(n)
        for p in order:
            v = h_work[p]
            if v < 1e-10 or v > 1.0 - 1e-10:
                continue
            s_lo = min(-p, p - (n - 1))
            s_hi = max((n - 1) - p, p)
            s_vals = np.arange(s_lo, s_hi + 1)
            s_vals = s_vals[s_vals != 0]
            if len(s_vals) == 0:
                continue
            m_aff = n - 1 + s_vals
            a = np.zeros(len(s_vals))
            ps = p + s_vals
            pm = p - s_vals
            vps = (ps >= 0) & (ps < n)
            vpm = (pm >= 0) & (pm < n)
            a[vps] += (1.0 - h_work[ps[vps]])
            a[vpm] -= h_work[pm[vpm]]
            c_aff = c[m_aff]
            c0_aff = c_aff - v * a
            c1_aff = c_aff + (1.0 - v) * a
            c_s0 = c[n - 1] - v * (1.0 - v)
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[m_aff] = False
            mask[n - 1] = False
            M_unaff = c[mask].max() if mask.any() else 0.0
            M_0 = max(M_unaff, float(c0_aff.max()), c_s0)
            M_1 = max(M_unaff, float(c1_aff.max()), c_s0)
            if M_0 < M - 1e-12:

            best_c = c_new
            best_h = h_work.copy()
            h = h_work
        if not any_change:
            break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=800, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x.copy(), n)
        new_c = compute_c5(new_h, n)
        return new_h, new_c
    except Exception:
        return h, compute_c5(h, n)
