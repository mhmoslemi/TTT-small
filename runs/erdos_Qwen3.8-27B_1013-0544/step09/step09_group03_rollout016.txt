def top_k_gradient(h, n, K, power):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top K indices
    top_idx = np.argsort(c)[::-1][:K]
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)  # shift

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

    return h, best_c

def top_k_gradient(h, n, K=10, power=4):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:K]
    weights = power * (c[top_idx] ** (power - 1))
    grad = np.zeros(n)
    for ki, w in zip(top_idx, weights):
        m = ki - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            s = -m
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    return grad, c

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = 1.0
    x_prev = h.copy()
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + ((t - 1.0) / t_new) * (h - x_prev)
        y = np.clip(y, 0.0, 1.0)

        # Line search / step

        x_new = project_feasible(x_new, n)
        # Check improvement

        if c_new < best_c:

                lr *= 0.8
            else:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def top_k_gradient(h, n, K, power):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:K]
    weights = power * np.maximum(c[top_idx], 1e-10) ** (power - 1)
    grad = np.zeros(n)
    for ki, w in zip(top_idx, weights):
        m = int(ki) - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            if n - m > 0:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
        else:
            s = -m
            if n - s > 0:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
    return grad, c

def grad_search(h, n, K, power, max_iter, lr_init, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    t = 1.0
    x_prev = h.copy()
    stall_count = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        y = h + ((t - 1.0) / (t + (4-t))) * (h - x_prev) if t > 1 else h.copy()
        y = np.clip(y, 0.0, 1.0)
        grad_y, c_y = top_k_gradient(y, n, K, power)
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-12:
            break
        x_new = y - lr * grad_y
        x_new = project_feasible(x_new, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 200:
            lr *= 0.5
            stall_count = 0
            h = best_h + rng.normal(0, 0.01, n)
            h = project_feasible(h, n)
            x_prev = h.copy()
            t = 1.0
            continue
        if c_new > best_c * 1.02 and it > 50:
            lr *= 0.85
        elif it % 100 == 99 and c_new <= best_c * 1.001:
            lr = min(lr * 1.03, 0.05)
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_prev = h
        h = x_new
    return best_h, best_c
