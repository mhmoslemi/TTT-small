    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Sequential exact CD (many passes)
    remaining = deadline - time.time() - 200
    n_passes = max(20, int(remaining / 2.5))
    h2, c2 = exact_cd(best_h, n_init, n_passes, deadline, order='sequential', rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Peak-priority CD
    if time.time() < deadline - 150:
        h3, c3 = exact_cd(best_h, n_init, 40, deadline, order='peak', rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Binary push + CD
    if time.time() < deadline - 120:
        hb = binary_refine(best_h, n_init, deadline, rng)
        c_val = compute_c5(hb, n_init)
        if c_val < best_c:
            best_h, best_c = hb, c_val
        h4, c4 = exact_cd(best_h, n_init, 30, deadline, order='sequential', rng=rng)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Random restart perturbations
    if time.time() < deadline - 60:
        n = len(best_h)
        for r in range(5):
            if time.time() > deadline - 40:
                break
            s = 0.02 + 0.02 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h5, c5 = exact_cd(hp, n, 25, deadline, order='peak', rng=rng)
            if c5 < best_c:
                best_h, best_c = h5, c5

    # Phase 6: Try nearby n values
    n_cur = len(best_h)
    for n_try in [n_cur - 20, n_cur + 20, n_cur - 50, n_cur + 50, 500, 900]:
        if time.time() > deadline - 80:
            break
        if n_try <= 100 or n_try == n_cur or n_try % 2 != 0:
            continue
        x_orig = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h6, c6 = exact_cd(h_int, n_try, 25, deadline, order='sequential', rng=rng)
        h6, c6 = exact_cd(h6, n_try, 25, deadline, order='peak', rng=rng)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final polish
    if time.time() < deadline - 10:
        n = len(best_h)
        max_passes = max(5, int((deadline - time.time() - 5) / 2.0))
        hf, cf = exact_cd(best_h, n, max_passes, deadline, order='sequential', rng=rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
