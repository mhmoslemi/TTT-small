    if fwd_lo <= fwd_hi:
        ks = np.arange(fwd_lo, fwd_hi+1)
        idx = p - ks + n - 1
        old_vals = h[idx].copy()
        # Fix idx == p
        mask_p = (idx == p)
        old_vals[mask_p] -= d  # h[p] is now h_old[p] + d, so h_old[p] = h[p] - d
        c[ks] += d * (1.0 - old_vals)
    # Backward: k ∈ [n-1-p, 2n-2-p] ∩ [0, 2n-2], idx = p+k-n+1
    # c[k] -= d * old_h[idx]
    bwd_lo = max(0, n-1-p)
    bwd_hi = min(2*n-2, 2*n-2-p)
    if bwd_lo <= bwd_hi:
        ks = np.arange(bwd_lo, bwd_hi+1)
        idx = p + ks - n + 1
        old_vals = h[idx].copy()
        mask_p = (idx == p)
        old_vals[mask_p] -= d
        c[ks] -= d * old_vals
    # Quadratic at k=n-1
    c[n-1] -= d * d

def exact_cd(h, n, c, max_passes, deadline, order='seq'):
    """Exact coordinate descent with incremental correlation update."""
    n_idx = np.arange(n)
    best_c = float(c.max()) * (2.0/n)
    best_h = h.copy()
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        if order == 'random':
            idx = n_idx.copy()
            np.random.shuffle(idx)
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = solve_1d(c, h, p, n)
            d = new_val - h0
            if abs(d) > 1e-14:
                h[p] = new_val
                apply_delta_to_corr(c, h, p, d, n)
                improved = True
        if not improved:
            break
        # Project to maintain sum constraint
        diff = n/2.0 - h.sum()
        if abs(diff) > 1e-12:
            # Simple: add uniform correction to interior points
            interior = (h > 1e-6) & (h < 1.0 - 1e-6)
            if interior.sum() > 0:
                h[interior] += diff / interior.sum()
                h = np.clip(h, 0, 1)
                # Recompute c after projection
                c = compute_corr(h)
        new_max = float(c.max()) * (2.0/n)
        if new_max < best_c:
            best_c = new_max
            best_h = h.copy()
    return best_h, best_c, c
