import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def block_lp_update(h, n, block_start, block_size, dx):
    """Optimize a block of consecutive coordinates using LP."""
    B = block_size
    p = block_start
    block_idx = np.arange(p, p + B)
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Build linear coefficients a[m, j] for j in block
    # c[m] = sum_j h[j] * (1-h[j+m-(n-1)])
    # For j in block, a[m, j_idx] = 1 - h[j+m-(n-1)] (if valid)
    # background[m] = c[m] - sum_{j in block} h[j] * a[m, j_idx]
    
    A = np.zeros((2*n-1, B))
    for j_idx, j in enumerate(block_idx):
        for m in range(2*n-1):
            idx2 = j + m - (n-1)
            if 0 <= idx2 < n:
                A[m, j_idx] = 1.0 - h[idx2]
    
    bg = c - h[block_idx] @ A.T  # background for each lag
    
    # LP: min t
    # s.t. A @ h_block + bg <= t  =>  A @ h_block - t <= -bg
    #      0 <= h_block <= 1
    #      sum(h_block) = target_block
    
    target_block = h[block_idx].sum()  # preserve current sum
    
    # Variables: [h_block (B), t (1)]
    c_obj = np.zeros(B + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A h_block - t <= -bg
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = -bg
    
    # Bounds: 0 <= h_block <= 1, t unbounded below (or t >= 0)
    bounds = [(0, 1)] * B + [(0, None)]
    
    # Equality: sum h_block = target_block
    A_eq = np.zeros((1, B + 1))
    A_eq[0, :B] = 1.0
    b_eq = np.array([target_block])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if res.success:
        h_new = h.copy()
        h_new[block_idx] = np.clip(res.x[:B], 0, 1)
        h_new = project_feasible(h_new, n)
        new_c = compute_c5(h_new, n)
        return h_new, new_c
    return h, compute_c5(h, n)

                h_block, lp_t = block_lp(best_h, n, p, B)
                if lp_t * dx < best_c:
                    c_new = compute_c5(h_block, n)
                    if c_new < best_c - 1e-12:
                        best_h, best_c = h_block, c_new
                        improved = True
            if not improved:
                break
    # Phase 3: SLSQP progressive alpha
    if time.time() < deadline - 250:
        for alpha in [30000, 150000, 600000, 2000000]:
            if time.time() > deadline - 250:
                break
            h_s, c_s = slsqp_run(best_h, n, alpha, 400)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
    # Phase 4: More block LP after SLSQP
    if time.time() < deadline - 150:
        for B in [8, 12, 18]:
            if time.time() > deadline - 150:
                break
            for sweep in range(5):
                if time.time() > deadline - 150:
                    break
                improved = False
                for p in range(0, n - B + 1):
                    if time.time() > deadline - 150:
                        break
                    h_block, lp_t = block_lp(best_h, n, p, B)
                    if lp_t * dx < best_c:
                        c_new = compute_c5(h_block, n)
                        if c_new < best_c - 1e-12:
                            best_h, best_c = h_block, c_new
                            improved = True
                if not improved:
                    break
    # Phase 5: Stochastic restarts
    if time.time() < deadline - 80:
        for r in range(10):
            if time.time() > deadline - 80:
                break
            s = 0.001 + 0.002 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            hc, cc = multi_cd(hp, n, 60, deadline - 80, 10)
            if cc < best_c:
                best_h, best_c = hc, cc
            # Also try block LP on perturbed
            if time.time() < deadline - 80:
                for p in range(0, n - 6, 6):
                    h_block, lp_t = block_lp(hc, n, p, 6)
                    if lp_t * dx < cc:
                        c_new = compute_c5(h_block, n)
                        if c_new < best_c:
                            best_h, best_c = h_block, c_new
    # Phase 6: Final SLSQP + deep CD
    if time.time() < deadline - 30:
        h_f, c_f = slsqp_run(best_h, n, 5000000, 600)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 2:
        h_f, c_f = multi_cd(best_h, n, 300, deadline - 1, 20)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h, best_c = try_symmetry(best_h, n)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
