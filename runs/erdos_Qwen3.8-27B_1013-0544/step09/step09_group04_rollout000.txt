def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:

                           bounds=bounds, constraints=cons,

            new_h = project_feasible(res.x, n)

            if new_c < best_c:

For a pair (p, q) with p < q:
- The affected correlations are those that involve either p or q
- The 2D problem is: minimize max of (affected correlations) over (h[p], h[q]) ∈ [0,1]²
- This can be solved by bisection on the target value L, then checking feasibility in 2D

The feasibility check in 2D is:
For each affected correlation m:
c[m] + a_m[0]*x + a_m[1]*y + δ_{m,n-1}*(x² if p is diagonal, y² if q is diagonal) ≤ L

This is a system of quadratic inequalities in (x, y). The feasible region is an intersection of half-planes (from linear constraints) and a half-plane from the parabolic constraint (if diagonal is involved).
