def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff/tot)*room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h - (diff/tot)*h, 0.0, 1.0)
    return h

def exact_cd(h, n, max_passes, deadline, rng=None):

    c = np.correlate(h, 1.0 - h, mode='full')
    if rng is None:
        rng = np.random.RandomState(42)
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue

            m_hi = 2*n - 2 - p

            a = np.zeros(len(ms))
            for i, m in enumerate(ms):

            L_hi = float(np.max(c))
            for _ in range(50):

                x_lo, x_hi = 0.0, 1.0

                    if feasible and np.any(neg):

                    if feasible and np.any(pos):

                    r2 = (1.0 + np.sqrt(disc))/2.0
                    if x_lo < r1 and x_hi > r2:

                    elif x_lo >= r1 and x_hi <= r2:

                improved = True
        if not improved: break

def anneal_cd(h, n, max_iters, deadline, rng, T0=0.01, T_end=0.0001):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T0
    for it in range(max_iters):
        if time.time() > deadline: break
        # Random perturbation
        scale = T
        h_new = h + rng.normal(0, scale, n)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Accept/reject
        delta = c_new - compute_c5(h, n)
        if delta < 0 or rng.random() < np.exp(-delta/T):
            h = h_new
        T = max(T_end, T * 0.999)
        # Periodic CD polish
        if it % 50 == 0 and it > 0:
            h, c = exact_cd(h, n, 5, deadline)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c
