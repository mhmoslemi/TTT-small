    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    # Phase 1: Coarse optimization at reduced resolution
    n_coarse = max(100, n // 3)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = project_feasible(np.interp(x_coarse, np.linspace(0, 2, n), best_h), n_coarse)
    h_coarse, c_coarse = coord_descent(h_coarse, n_coarse, 500, t0 + 60, seed)
    # Interpolate back
    x_fine = np.linspace(0, 2, n)
    h_interp = project_feasible(np.interp(x_fine, x_coarse, h_coarse), n)
    h_interp, c_interp = coord_descent(h_interp, n, 300, t0 + 120, seed + 1)
    if c_interp < best_c:
        best_h, best_c = h_interp, c_interp

    # Phase 2: Heavy 1D CD
    dl2 = min(t0 + 400, deadline - 250)
    h2, c2 = coord_descent(best_h, n, 3000, dl2, seed)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: 2D block moves
    dl3 = min(t0 + 550, deadline - 200)
    h3, c3 = block2d_fast(best_h, n, dl3, n_passes=40)
    if c3 < best_c:
        best_h, best_c = h3, c2 if c2 < c3 else c3
        best_h = best_h if c3 <= c2 else h2
    # Re-do CD after block moves
    h3b, c3b = coord_descent(best_h, n, 500, min(t0 + 620, deadline - 180), seed + 10)
    if c3b < best_c:
        best_h, best_c = h3b, c3b

    # Phase 4: SLSQP polish
    dl4 = min(t0 + 720, deadline - 150)
    if time.time() < dl4:
        h4, c4 = slsqp_run(best_h, n, [5000, 50000, 200000], 400)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_descent(best_h, n, 400, deadline - 130, seed + 20)
        if c4b < best_c:
            best_h, best_c = h4b, c4b

    # Phase 5: Resolution hopping
    x_orig = np.linspace(0, 2, len(best_h))
    n_trials = [n - 2, n + 2, n // 2, n * 3 // 2]
    n_trials = [nt for nt in n_trials if 100 <= nt <= 2000 and nt != len(best_h)]
    for n_try in n_trials:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 600, deadline - 100, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Multi-restart with varied perturbations
    n_cur = len(best_h)
    for r in range(15):
        if time.time() > deadline - 60:
            break
        s = 0.003 + 0.02 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 300, deadline - 50, seed + 300 + r * 13)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 7: Binary push
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        thresh = np.where(best_h > 0.5, 1.0, 0.0)
        hb = project_feasible(thresh, n_cur)
        h_o, c_o = coord_descent(hb, n_cur, 300, deadline - 30, seed + 600)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 8: Final extensive CD
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        hf, cf = coord_descent(best_h, n_cur, 5000, deadline - 5, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Check 1-h symmetry
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
