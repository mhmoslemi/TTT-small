def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

    dx = 2.0 / n

    best_f = compute_c5(x, n)

        if deadline is not None and time.time() > deadline:
            break
        # Compute gradient at z
        grad = compute_gradient(z, n, alpha)
        # Gradient step

        # Project

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        x = x_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha > 1e8:
        return m * (2.0 / n)
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _smooth_obj_grad(x, n, alpha):
    """Gradient of smoothed objective (for potential use)."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha > 1e8:
        # Subgradient: pick the argmax
        idx = np.argmax(c)
        s = idx - (n - 1)
        grad = np.zeros(n)
        # ∂c[idx]/∂h[j] = (1-h)[j+s]*1[valid] - h[j-s]*1[valid]
        for j in range(n):
            js = j + s
            if 0 <= js < n:
                grad[j] += (1.0 - x[js])
            jm = j - s
            if 0 <= jm < n:
                grad[j] -= x[jm]
        return grad * (2.0 / n)
    shifted = alpha * (c - m)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    grad = np.zeros(n)
    for s_idx in range(2 * n - 1):
        s = s_idx - (n - 1)
        w_s = w[s_idx]
        if w_s < 1e-12:
            continue
        j1lo = max(0, -s)
        j1hi = min(n, n - s)
        if j1lo < j1hi:
            grad[j1lo:j1hi] += w_s * (1.0 - x[j1lo + s:j1hi + s])
        j2lo = max(0, s)
        j2hi = min(n, n + s)
        if j2lo < j2hi:
            grad[j2lo:j2hi] -= w_s * x[j2lo - s:j2hi - s]
    return grad * (2.0 / n)

def slsqp_optimize(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fista_optimize(h, n, alpha, max_iter=3000, deadline=None):
    """FISTA on smoothed objective."""
    x = h.copy()
    z = h.copy()
    t_k = 1.0
    L = 2.0 * n  # Lipschitz estimate
    best_x = x.copy()
    best_c = compute_c5(x, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        grad = _smooth_obj_grad(z, n, alpha)
        x_new = z - grad / L
        x_new = project_feasible(x_new, n)
        # Backtracking: check if objective decreased
        f_curr = _smooth_obj(x_new, n, alpha)
        f_z = _smooth_obj(z, n, alpha)
        if f_curr > f_z + 0.5 * L * np.sum((x_new - z)**2):
            L *= 1.5
            x_new = z - grad / L
            x_new = project_feasible(x_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = project_feasible(z, n)
        x = x_new.copy()
        t_k = t_new
        c_curr = compute_c5(x, n)
        if c_curr < best_c - 1e-12:
            best_c = c_curr
            best_x = x.copy()
        if it > 50 and abs(c_curr - best_c) < 1e-10:
            break
    return best_x, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=70):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_descent(h, n, max_iter=5000, step_init=0.1, deadline=None):
    """Subgradient method with diminishing step size."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_max = c.max()
        # Active set: shifts where c is within epsilon of max
        eps = 0.01 * c_max + 0.001
        active = c >= c_max - eps
        # Compute subgradient: average of gradients at active shifts
        grad = np.zeros(n)
        n_active = active.sum()
        for m in np.where(active)[0]:
            s = m - (n - 1)
            for j in range(n):
                js = j + s
                if 0 <= js < n:
                    grad[j] += (1.0 - h_curr[js])
                jm = j - s
                if 0 <= jm < n:
                    grad[j] -= h_curr[jm]
        grad /= n_active
        # Step
        step = step_init / (1.0 + 0.01 * it)
        h_new = h_curr - step * grad * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def subgradient_descent(h, n, max_iter=5000, step_init=0.05, deadline=None):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_max = c.max()
        eps = max(0.005 * c_max, 0.01)
        active = np.where(c >= c_max - eps)[0]
        # Compute subgradient (vectorized)
        grad = np.zeros(n)
        nm1 = n - 1
        for m in active:
            s = m - nm1
            # Forward: (1-h)[j+s] for valid j
            j1lo = max(0, -s)
            j1hi = min(n, n - s)
            if j1lo < j1hi:
                grad[j1lo:j1hi] += (1.0 - h_curr[j1lo+s:j1hi+s])
            # Backward: -h[j-s] for valid j
            j2lo = max(0, s)
            j2hi = min(n, n + s)
            if j2lo < j2hi:
                grad[j2lo:j2hi] -= h_curr[j2lo-s:j2hi-s]
        grad /= len(active)
        step = step_init / (1.0 + 0.005 * it)
        h_new = h_curr - step * grad * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

One more thought: the "binary local search" idea. For the binary case with n/2 ones, I can do a more efficient local search:

Given a binary sequence, the correlation is:
c[m] = sum_i h[i] * (1-h)[i+s] where s = m-(n-1)

If I swap positions i (1→0) and j (0→1), the change in c[m] for shift s is:
- At position i: was contributing h[i]*(1-h[i+s]) = 1*(1-h[i+s]), now contributes 0
- At position j: was contributing h[j]*(1-h[j+s]) = 0, now contributes (1-h[j+s])
- Also, for any position k where k+s = i or k+s = j:
  - If k+s = i (k = i-s): was contributing h[k]*(1-h[i]) = h[k]*1 = h[k], now contributes h[k]*0 = 0
  - If k+s = j (k = j-s): was contributing h[k]*(1-h[j]) = h[k]*1 = h[k], now contributes h[k]*0 = 0

Wait, I need to be more careful. Let me define:
c[m] = sum_i h[i] * (1 - h[i+s]) where s = m-(n-1)

After swapping h[i] and h[j] (where h[i]=1, h[j]=0):
- h[i] becomes 0, h[j] becomes 1
- Change in c[m]:
  - Term i: was 1*(1-h[i+s]), now 0*(1-h[i+s]). Change: -(1-h[i+s])
  - Term j: was 0*(1-h[j+s]), now 1*(1-h[j+s]). Change: +(1-h[j+s])
  - Term k where k+s = i (i.e., k = i-s): was h[k]*(1-1) = 0, now h[k]*(1-0) = h[k]. Change: +h[k] (if k ≠ i,j)
  - Term k where k+s = j (i.e., k = j-s): was h[k]*(1-0) = h[k], now h[k]*(1-1) = 0. Change: -h[k] (if k ≠ i,j)

So Δc[m] = -(1-h[i+s]) + (1-h[j+s]) + h[i-s] - h[j-s]
= -1 + h[i+s] + 1 - h[j+s] + h[i-s] - h[j-s]
= h[i+s] - h[j+s] + h[i-s] - h[j-s]

(with appropriate zero-padding for out-of-bounds indices)

This is O(1) per (i, j, m) triple! So I can compute the effect of a swap on all m values in O(n) time.

For a local search, I want to find the swap (i, j) that minimizes max_m c[m]. This is still hard in general, but I can do a greedy approach:
1. Find the worst shift m* (where c is maximized)
2. Find swaps that reduce c[m*]
3. Among those, pick the one that also doesn't increase other c[m] too much

This is essentially what a "1-swap" local search does.
