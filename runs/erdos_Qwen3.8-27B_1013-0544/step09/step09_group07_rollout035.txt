```python
def smooth_obj_v2(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)
```

Actually, I can use a different strategy: since f(d) is convex (max of convex functions), I can use a subgradient method or bisection on the subgradient to find the minimum.

The subgradient of f(d) at point d is:
- If the max is achieved by a unique function i: subgradient is a_coeff[i] - 2*d*(diag==i)
- If max is achieved by multiple functions: convex hull of their subgradients
- If unaff_max is the active constraint: subgradient is {0}

So I can use a bisection method:
- At d, compute which function(s) are active
- If the active function has positive subgradient, decrease d
- If negative, increase d
- If zero (or 0 is in the subgradient), we're at the minimum

This is O(n) per evaluation and requires O(log(1/eps) * n) total. Much better than O(n^2)!

```python
def solve_1d_bisection(c_aff, a_coeff, diag_local, unaff_max, lo, hi, tol=1e-12, max_iter=200):
    """Find min of max(c_aff + d*a_coeff - d^2*(i==diag), unaff_max) using subgradient bisection."""
    k = len(a_coeff)
    
    def eval_at(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        val = float(nv.max())
        if val < unaff_max:
            return unaff_max, 0.0  # subgradient is 0 (flat region)
        # Find active set
        active = np.where(nv >= val - 1e-14)[0]
        if len(active) == 1:
            i = active[0]
            sg = a_coeff[i] - (2.0 * d if diag_local == i else 0.0)
            return val, sg
        else:
            # Subgradient is interval [min, max] of active subgradients
            sgs = a_coeff[active] - (2.0 * d if diag_local in active else 0.0)
            # Wait, need to be more careful
            sgs = np.array([a_coeff[i] - (2.0 * d if diag_local == i else 0.0) for i in active])
            return val, (sgs.min(), sgs.max())
    
    # Bisection on subgradient
    d_lo, d_hi = lo, hi
    for _ in range(max_iter):
        d_mid = 0.5 * (d_lo + d_hi)
        val, sg = eval_at(d_mid)
        if isinstance(sg, tuple):
            sg_min, sg_max = sg
            if sg_min > 1e-12:
                d_hi = d_mid  # function is increasing, go left
            elif sg_max < -1e-12:
                d_lo = d_mid  # function is decreasing, go right
            else:
                break  # 0 is in subgradient, optimal
        else:
            if sg > 1e-12:
                d_hi = d_mid
            elif sg < -1e-12:
                d_lo = d_mid
            else:
                break
        if d_hi - d_lo < tol:
            break
    
    d_best = 0.5 * (d_lo + d_hi)
    
    # Verify and refine with golden section in a small neighborhood
    def obj(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        return max(float(nv.max()), unaff_max)
    
    # Check boundaries and a few points
    candidates = [lo, hi, d_best, 0.0]
    best_d = min(max(d_best, lo), hi)
    best_val = obj(best_d)
    for cd in candidates:
        cd = float(np.clip(cd, lo, hi))
        v = obj(cd)
        if v < best_val:
            best_val = v
            best_d = cd
    
    # Golden section refinement
    margin = (hi - lo) * 0.05
    gl = max(lo, best_d - margin)
    gh = min(hi, best_d + margin)
    if gh - gl > 1e-14:
        try:
            res = golden(obj, gl, gh, maxiter=50)
            if res.fun < best_val:
                best_d = float(res.x)
                best_val = float(res.fun)
        except Exception:
            pass
    
    return best_d
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_vec(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    """Vectorized 1D solver: min_d max(c_aff[d]*1 + a_coeff[d]*d - (d^2 if diag, unaff_max)"""
    k = len(a_coeff)
    if k == 0:
        return 0.0
    
    # Build candidate points
    candidates = []
    
    # Boundary points
    candidates.append(lo)
    candidates.append(hi)
    if lo <= 0.0 <= hi:
        candidates.append(0.0)
    
    # Pairwise intersections of linear parts
    # For i < j: c_aff[i] + d*a_coeff[i] = c_aff[j] + d*a_coeff[j]
    # => d = (c_aff[j] - c_aff[i]) / (a_coeff[i] - a_coeff[j])
    # But if one of them is the diagonal, it's (c_aff[i] + d*a_i - d^2) vs (c_aff[j] + d*a_j)
    # or both could be diagonal (but only one can be)
    
    for i in range(k):
        for j in range(i + 1, k):
            di = 1.0 if (diag_local is not None and diag_local == i) else 0.0
            dj = 1.0 if (diag_local is not None and diag_local == j) else 0.0
            # f_i(d) = c_aff[i] + a_coeff[i]*d - di*d^2
            # f_j(d) = c_aff[j] + a_coeff[j]*d - dj*d^2
            # f_i - f_j = (c_aff[i]-c_aff[j]) + (a_coeff[i]-a_coeff[j])*d - (di-dj)*d^2 = 0
            a0 = di - dj  # coefficient of d^2
            a1 = a_coeff[i] - a_coeff[j]
            a2 = c_aff[i] - c_aff[j]
            if abs(a0) < 1e-14:
                if abs(a1) > 1e-14:
                    d = -a2 / a1
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = a1*a1 - 4.0*a0*a2
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    d1 = (-a1 + sq) / (2.0 * a0)
                    d2 = (-a1 - sq) / (2.0 * a0)
                    if lo < d1 < hi:
                        candidates.append(d1)
                    if lo < d2 < hi:
                        candidates.append(d2)
    
    # Intersections with unaff_max
    for i in range(k):
        di = 1.0 if (diag_local is not None and diag_local == i) else 0.0
        # c_aff[i] + a_coeff[i]*d - di*d^2 = unaff_max
        if di == 0:
            if abs(a_coeff[i]) > 1e-14:
                d = (unaff_max - c_aff[i]) / a_coeff[i]
                if lo < d < hi:
                    candidates.append(d)
        else:
            # -d^2 + a_coeff[i]*d + (c_aff[i] - unaff_max) = 0
            # d^2 - a_coeff[i]*d + (unaff_max - c_aff[i]) = 0
            disc = a_coeff[i]**2 - 4.0*(unaff_max - c_aff[i])
            if disc >= 0.0:
                sq = np.sqrt(disc)
                for d in [(a_coeff[i] + sq)/2.0, (a_coeff[i] - sq)/2.0]:
                    if lo < d < hi:
                        candidates.append(d)
    
    if diag_local is not None:
        # Vertex of the quadratic: d = a_coeff[diag_local] / 2
        d_vertex = a_coeff[diag_local] / 2.0
        if lo < d_vertex < hi:
            candidates.append(d_vertex)
    
    # Evaluate at all candidates
    def obj(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        return max(float(nv.max()), unaff_max)
    
    best_d = lo
    best_val = obj(lo)
    if hi > lo:
        v_hi = obj(hi)
        if v_hi < best_val:
            best_val = v_hi
            best_d = hi
    
    for d in candidates:
        v = obj(d)
        if v < best_val - 1e-15:
            best_val = v
            best_d = d
    
    # Golden section refinement
    if hi - lo > 1e-10:
        margin = (hi - lo) * 0.1
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    
    return best_d
```

```python
def smooth_obj(x, n, alpha):

    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
```
