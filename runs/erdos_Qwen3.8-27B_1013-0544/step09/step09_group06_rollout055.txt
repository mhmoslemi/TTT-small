def smooth_c5(h, n):
    """Compute the smoothed C5 (log-sum-exp approximation) and its gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    dx = 2.0 / n
    return c * dx, c

def smooth_obj_and_grad(h, n, p):
    """Log-sum-exp smoothed objective and gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    pc = p * c
    pc_shifted = pc - pc.max()
    exp_pc = np.exp(pc_shifted)
    w = exp_pc / exp_pc.sum()  # softmax weights, shape (2n-1,)
    
    # Build gradient
    lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # grad[p] = dx * sum_k w_k * [I(0≤p+lag_k<n)*(1-h[p+lag_k]) - I(0≤p-lag_k<n)*h[p-lag_k]]
    
    # Vectorize: for each p, compute sum over k
    # term1[p] = sum_k w_k * I(0≤p+lag_k<n) * (1-h[p+lag_k])
    # term2[p] = sum_k w_k * I(0≤p-lag_k<n) * h[p-lag_k]
    
    P = np.arange(n)  # (n,)
    L = lags  # (2n-1,)
    
    # term1: p+L in [0,n)
    idx1 = P[:, None] + L[None, :]  # (n, 2n-1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    vals1 = np.where(mask1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    term1 = (vals1 * w[None, :]).sum(axis=1)
    
    # term2: p-L in [0,n)
    idx2 = P[:, None] - L[None, :]  # (n, 2n-1)
    mask2 = (idx2 >= 0) & (idx2 < n)
    vals2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    term2 = (vals2 * w[None, :]).sum(axis=1)
    
    grad = dx * (term1 - term2)
    obj = dx * (c.max() + np.log(exp_pc.sum()) / p)
    return obj, grad

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, p, lam):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    pc = p * c
    pc_max = pc.max()
    exp_pc = np.exp(pc - pc_max)
    exp_sum = exp_pc.sum()
    w = exp_pc / exp_sum
    
    lags = np.arange(2*n-1) - (n-1)
    P = np.arange(n)
    
    idx1 = P[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    vals1 = np.where(mask1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    term1 = (vals1 * w[None, :]).sum(axis=1)
    
    idx2 = P[:, None] - lags[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    vals2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    term2 = (vals2 * w[None, :]).sum(axis=1)
    
    grad = dx * (term1 - term2)
    s = h.sum() - n/2.0
    grad += 2.0 * lam * s * np.ones(n)
    obj = dx * (pc_max + np.log(exp_sum) / p) + lam * s * s
    return obj, grad

def lbfgs_optimize(h, n, p, lam, deadline, maxiter=200):

    def obj_only(x):
        return smooth_obj_and_grad(x, n, p, lam)[0]
    def grad_only(x):
        return smooth_obj_and_grad(x, n, p, lam)[1]
    
    try:
        res = minimize(obj_only, h, jac=grad_only, method='L-BFGS-B',
                       bounds=[(0,1)]*n,

                                'maxls': 50})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _eval_1d(d, c_top, a_coeff, diag_idx, unaff_max, dx):
    vals = c_top + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max) * dx

def cd_topk(h, n, max_passes, deadline, top_k):
    h = h.copy()
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max()) * dx
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_top = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                lag = int(lags[top_idx[idx]])
                jb = p - lag
                jf = p + lag
                t1 = (1.0 - h[jb]) if 0 <= jb < n else 0.0
                t2 = h[jf] if 0 <= jf < n else 0.0
                a_coeff[idx] = t1 - t2
                if lag == 0:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            a, b = lo, hi
            gr = (np.sqrt(5) - 1) / 2
            for _ in range(35):
                d1 = b - gr * (b - a)
                d2 = a + gr * (b - a)
                v1 = _eval_1d(d1, c_top, a_coeff, diag_idx, unaff_max, dx)
                v2 = _eval_1d(d2, c_top, a_coeff, diag_idx, unaff_max, dx)
                if v1 < v2:
                    b = d2
                else:
                    a = d1
            d_best = (a + b) / 2.0
            v_best = _eval_1d(d_best, c_top, a_coeff, diag_idx, unaff_max, dx)
            if v_best < current_max - 1e-14:
                h[p] = float(np.clip(h0 + d_best, 0.0, 1.0))
                improved = True
                current_max = v_best
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _smooth_grad(h, n, p, lam):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    pc = p * c
    pc_max = pc.max()
    exp_pc = np.exp(pc - pc_max)
    exp_sum = exp_pc.sum()
    w = exp_pc / exp_sum
    lags = np.arange(2 * n - 1) - (n - 1)
    P = np.arange(n)
    idx2 = P[:, None] - lags[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    vals2 = np.where(mask2, 1.0 - h[np.clip(idx2, 0, n - 1)], 0.0)
    term1 = (vals2 * w[None, :]).sum(axis=1)
    idx1 = P[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    vals1 = np.where(mask1, h[np.clip(idx1, 0, n - 1)], 0.0)
    term2 = (vals1 * w[None, :]).sum(axis=1)
    grad = dx * (term1 - term2)
    s = h.sum() - n / 2.0
    grad += 2.0 * lam * s
    obj = dx * (pc_max + np.log(exp_sum) / p) + lam * s * s
    return obj, grad

def lbfgs_optimize(h, n, p, lam, deadline, maxiter=300):
    try:
        def obj_fn(x):
            return _smooth_grad(x, n, p, lam)[0]
        def grad_fn(x):
            return _smooth_grad(x, n, p, lam)[1]
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxls': 40})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_complement(h, n):
    h_c = project_feasible(1.0 - h, n)
    c_c = compute_c5(h_c, n)
    c_h = compute_c5(h, n)
    if c_c < c_h:
        return h_c, c_c
    return h, c_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: CD with increasing top_k
    for tk in [5, 10, 20, 40]:
        if time.time() > deadline - 500:
            break
        h2, c2 = cd_topk(best_h, n, 400, deadline - 500, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 2: L-BFGS-B with increasing sharpness
    for p_val, lam in [(50, 100.0), (200, 200.0), (1000, 500.0), (3000, 1000.0)]:
        if time.time() > deadline - 300:
            break
        h2, c2 = lbfgs_optimize(best_h, n, p_val, lam, deadline - 300, 250)
        if c2 < best_c:
            best_h, best_c = h2, c2
        if time.time() < deadline - 250:
            h3, c3 = cd_topk(best_h, n, 200, deadline - 250, 20)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 3: Multi-resolution refinement
    x_cur = np.linspace(0, 2, len(best_h))
    grid_list = [300, 400, 500, 700, 800]
    for n_try in grid_list:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h2, c2 = cd_topk(h_int, n_try, 250, deadline - 120, 20)
        h2, c2 = lbfgs_optimize(h2, n_try, 500, 500.0, deadline - 120, 150)
        h3, c3 = cd_topk(h2, n_try, 150, deadline - 120, 15)
        if c3 < c2:
            h2, c2 = h3, c3
        h2, c2 = try_complement(h2, n_try)
        if c2 < best_c:
            best_h, best_c = h2, c2
            x_cur = x_new

    # Phase 4: Stochastic perturbations with structure
    n_cur = len(best_h)
    for r in range(20):
        if time.time() > deadline - 60:
            break
        s = 0.002 + 0.003 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h2, c2 = cd_topk(hp, n_cur, 150, deadline - 60, 12)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 5: Final deep optimization
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        h2, c2 = lbfgs_optimize(best_h, n_cur, 5000, 2000.0, deadline - 40, 200)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 5:
        h2, c2 = cd_topk(best_h, n_cur, 800, deadline, 30)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h, best_c = try_complement(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
