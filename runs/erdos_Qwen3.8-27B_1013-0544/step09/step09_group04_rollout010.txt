def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    """Single pass of coordinate descent."""
    h = h.copy()
    n_idx = np.arange(n)
    if order == 'rand':
        idx = n_idx.copy()
        np.random.shuffle(idx)
    elif order == 'sensitivity':

        sens = np.zeros(n)
        for p in range(n):
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            seg = c[m_lo:m_hi + 1]
            sens[p] = seg.std()
        idx = n_idx[np.argsort(-sens)]
    else:
        idx = n_idx

            continue
        x_new = solve_1d(h, p, n)

            improved = True
    return h, improved

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def solve_1d(h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c = np.correlate(h, 1.0 - h, mode='full')
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    L_lo = unaff
    L_hi = float(c.max())
    for _ in range(80):
        if L_hi - L_lo < 1e-18:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feas = True
        zero_m = np.abs(a) <= 1e-15
        neg_m = a < -1e-15
        pos_m = a > 1e-15
        if np.any(zero_m) and float(c_base[zero_m].max()) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo, 0.0) > r1 and min(x_hi, 1.0) < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, deadline, order, rng):
    h = h.copy()
    n_idx = np.arange(n)
    if order == 'rand':
        idx = n_idx.copy()
        rng.shuffle(idx)
    elif order == 'sens':
        c = np.correlate(h, 1.0 - h, mode='full')
        sens = np.zeros(n)
        for p in range(n):
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            sens[p] = c[m_lo:m_hi+1].std()
        idx = n_idx[np.argsort(-sens)]
    else:
        idx = n_idx
    improved = False
    for p in idx:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        x_new = solve_1d(h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
            improved = True
    return h, improved

def block_cd(h, n, deadline, rng):
    """Update pairs of adjacent variables simultaneously via grid search."""
    h = h.copy()
    improved = False
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    for p in range(0, n - 1, 2):
        if time.time() > deadline:
            break
        h0, h1 = h[p], h[p+1]
        if (h0 < 1e-13 or h0 > 1.0-1e-13) and (h1 < 1e-13 or h1 > 1.0-1e-13):
            continue
        # Grid search over 2D
        best_val = cur_max
        best_dp, best_dq = 0.0, 0.0
        gp = np.linspace(-h0, 1.0-h0, 12)
        gq = np.linspace(-h1, 1.0-h1, 12)
        for dp in gp:
            for dq in gq:
                if abs(dp) < 1e-14 and abs(dq) < 1e-14:
                    continue
                h_test = h.copy()
                h_test[p] = np.clip(h0 + dp, 0, 1)
                h_test[p+1] = np.clip(h1 + dq, 0, 1)
                c_test = np.correlate(h_test, 1.0 - h_test, mode='full')
                val = float(c_test.max())
                if val < best_val - 1e-15:
                    best_val = val
                    best_dp, best_dq = dp, dq
        if abs(best_dp) > 1e-13 or abs(best_dq) > 1e-13:
            h[p] = np.clip(h0 + best_dp, 0, 1)
            h[p+1] = np.clip(h1 + best_dq, 0, 1)
            improved = True
    return h, improved

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-17})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
