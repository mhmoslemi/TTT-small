    m = c.max()

    mx = shifted.max()

    w = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))

        grad[p] = 2.0 / n * np.dot(w[ms], contrib)
    return val, grad

    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:

            break

            v, _ = smooth_obj_and_grad(x, n, alpha)
            return v

            return g

            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-13 or h0 > 1.0 - 1e-13: return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    cb = c[ms]
    di = np.where(ms == (n-1))[0]
    hd = len(di) > 0
    if hd:
        cd = cb[di[0]]
        a[di] = 0.0
    unaff = 0.0
    if m_min > 0: unaff = float(np.max(c[:m_min]))
    if m_max < 2*n-2: unaff = max(unaff, float(np.max(c[m_max+1:])))
    Llo, Lhi = unaff, float(np.max(c))
    nm = a < -1e-15
    pm = a > 1e-15
    zm = np.abs(a) <= 1e-15
    for _ in range(45):
        if Lhi - Llo < 1e-14: break
        L = (Llo + Lhi) * 0.5
        xl, xh = 0.0, 1.0
        ok = True
        if ok and np.any(zm) and np.any(cb[zm] > L): ok = False
        if ok and np.any(nm): xl = max(xl, float(np.max((L - cb[nm]) / a[nm])))
        if ok and np.any(pm): xh = min(xh, float(np.min((L - cb[pm]) / a[pm])))
        if xl > xh + 1e-14: ok = False
        if ok and hd:
            Cq = L - cd + h0*(1.0-h0)
            dsc = 1.0 - 4.0*Cq
            if dsc < 0: ok = False
            else:
                r1 = (1.0 - np.sqrt(dsc)) * 0.5
                r2 = (1.0 + np.sqrt(dsc)) * 0.5
                if max(xl,0.0) > r2 or min(xh,1.0) < r1: ok = False
        if ok: Lhi = L
        else: Llo = L
    L = Lhi
    xl, xh = 0.0, 1.0
    if np.any(nm): xl = max(xl, float(np.max((L - cb[nm]) / a[nm])))
    if np.any(pm): xh = min(xh, float(np.min((L - cb[pm]) / a[pm])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if hd:
        Cq = L - cd + h0*(1.0-h0)
        dsc = 1.0 - 4.0*Cq
        if dsc >= 0:
            r1 = (1.0 - np.sqrt(dsc)) * 0.5
            r2 = (1.0 + np.sqrt(dsc)) * 0.5
            if xl < r1 and xh > r2:
                if h0 - r1 <= r2 - h0: xh = min(xh, r1)
                else: xl = max(xl, r2)
            else:
                xl, xh = max(xl, r1), min(xh, r2)
    if xl > xh: return h0
    return float(np.clip((xl + xh) * 0.5, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for pas in range(max_passes):
        if time.time() > deadline: break
        if rng is not None and rng.rand() < 0.3:
            order = rng.permutation(n)
        elif pas % 2 == 0:
            order = range(n)
        else:
            order = range(n-1, -1, -1)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                d = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb): a[vb] -= h[jb[vb]]
                c[ms] += d * a
                di = np.where(ms == (n-1))[0]

                improved = True

        if np.max(np.abs(h2 - h)) > 1e-14:

        h = h2

def smooth_fn(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return (m + mx + np.log(np.sum(np.exp(s - mx)))) / alpha * (2.0/n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        co = np.zeros(len(ms))
        if np.any(vf): co[vf] += (1.0 - x[jf[vf]])
        if np.any(vb): co[vb] -= x[jb[vb]]
        di = np.where(ms == (n-1))[0]
        if len(di) > 0: co[di] = (1.0 - 2.0 * x[p])
        g[p] = dx * np.dot(w[ms], co)
    return g

def sq_sum_eq(x, n):
    return x.sum() - n / 2.0

def sq_ones(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    bh, bc = h.copy(), compute_c5(h, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_sum_eq, 'jac': sq_ones, 'args': (n)}]
    for alpha in alphas:
        if time.time() > deadline: break
        try:
            res = minimize(smooth_fn, bh, args=(n, alpha), jac=lambda x, nn, aa: smooth_jac(x, nn, aa),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc: bh, bc = nh, nc
        except: pass
    return bh, bc
