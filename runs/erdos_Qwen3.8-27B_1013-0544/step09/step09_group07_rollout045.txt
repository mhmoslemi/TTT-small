    if diag_local is not None:

        return 0.0

            dd = 0.0

            if diag_local == j:

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(0.0, lo, hi))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def peak_subgrad(h, n, top_m=5, max_iter=200, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        sorted_idx = np.argsort(c)[::-1][:top_m]
        weights = np.zeros(len(sorted_idx))
        for i, m in enumerate(sorted_idx):
            weights[i] = max(0.0, c[m] - c[sorted_idx[-1]]) + 0.1
        weights /= weights.sum()
        sg = np.zeros(n)
        for i, m in enumerate(sorted_idx):
            k = m - (n - 1)
            w = weights[i]
            for j in range(n):
                fwd = j - k
                if 0 <= fwd < n:
                    sg[j] += w * (1.0 - h[fwd])
                bwd = j + k
                if 0 <= bwd < n:
                    sg[j] -= w * h[bwd]
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.08, 0.15)
        else:
            lr *= 0.6
            stall += 1
            if stall > 40 or lr < 1e-13:
                break
    return h, best_c

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    shift = alpha * c
    mx = shift.max()
    lse = mx / alpha + np.log(np.sum(np.exp(shift - mx))) / alpha
    return lse

def lse_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    shift = alpha * c * dx
    mx = shift.max()
    probs = np.exp(shift - mx)
    probs /= probs.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = probs[m] * dx
        k = m - (n - 1)
        for j in range(n):
            fwd = j - k
            if 0 <= fwd < n:
                grad[j] += w * (1.0 - x[fwd])
            bwd = j + k
            if 0 <= bwd < n:
                grad[j] -= w * x[bwd]
    return grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_sin(n, freq, phase=0.0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.35 * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def make_multi_sin(n, freqs, amps):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for f, a in zip(freqs, amps):
        h += a * np.sin(np.pi * f * x)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: coordinate descent
    tl = deadline - 250
    if time.time() < tl:
        h1, c1 = coord_cd(best_h, n, max_passes=200, deadline=tl, top_k=30)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: peak-leveling subgradient
    if time.time() < deadline - 200:
        h2, c2 = peak_subgrad(best_h, n, top_m=5, max_iter=150, deadline=deadline - 180)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = coord_cd(best_h, n, max_passes=100, deadline=deadline - 160, top_k=30)
        if c2b < best_c:
            best_h, best_c = h2b, c2b

    # Phase 3: SLSQP
    if time.time() < deadline - 150:
        h3, c3 = slsqp_polish(best_h, n, [50, 200, 1000], 300, deadline - 130)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: more coordinate descent
    if time.time() < deadline - 100:
        h4, c4 = coord_cd(best_h, n, max_passes=300, deadline=deadline - 80, top_k=40)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: peak subgrad again
    if time.time() < deadline - 70:
        h5, c5v = peak_subgrad(best_h, n, top_m=8, max_iter=100, deadline=deadline - 55)
        if c5v < best_c:
            best_h, best_c = h5, c5v

    # Phase 6: try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 500, 700, 800]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=120, deadline=deadline - 60, top_k=30)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = np.linspace(0, 2, n_try)

    # Phase 7: diversified structural restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        restarts = []
        for freq in [1.5, 2.0, 2.5, 3.0]:
            for ph in [0, 1.0, 2.0]:
                restarts.append(make_sin(n_cur, freq, ph))
        restarts.append(make_multi_sin(n_cur, [1.0, 3.0], [0.2, 0.15]))
        restarts.append(make_multi_sin(n_cur, [2.0, 5.0], [0.15, 0.1]))
        for h_r in restarts[:6]:
            if time.time() > deadline - 40:
                break
            hr, cr = coord_cd(h_r, n_cur, max_passes=80, deadline=deadline - 35, top_k=25)
            if cr < best_c:
                best_h, best_c = hr, cr

    # Phase 8: small perturbations from best
    n_cur = len(best_h)
    if time.time() < deadline - 50:
        for r in range(8):
            if time.time() > deadline - 35:
                break
            s = 0.002 + 0.005 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            ho, co = coord_cd(hp, n_cur, max_passes=60, deadline=deadline - 30, top_k=30)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 9: final SLSQP + coordinate descent
    if time.time() < deadline - 25:
        h9, c9 = slsqp_polish(best_h, n_cur, [500, 2000, 10000], 300, deadline - 15)
        if c9 < best_c:
            best_h, best_c = h9, c9
    if time.time() < deadline - 10:
        h10, c10 = coord_cd(best_h, n_cur, max_passes=500, deadline=deadline - 5, top_k=50)
        if c10 < best_c:
            best_h, best_c = h10, c10

    # Phase 10: final polish with peak subgrad
    n_fin = len(best_h)
    if time.time() < deadline - 5:
        hf, cf = peak_subgrad(best_h, n_fin, top_m=3, max_iter=50, deadline=deadline - 3)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
