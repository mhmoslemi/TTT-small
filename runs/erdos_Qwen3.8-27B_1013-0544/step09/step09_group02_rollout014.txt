def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            tot = room.sum()

            if tot < 1e-15:

        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def apply_delta(h, p, d, c, n):
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    delta = np.zeros(len(ms))
    if np.any(vf):
        delta[vf] += d * (1.0 - h[jf[vf]])
    if np.any(vb):
        delta[vb] -= d * h[jb[vb]]
    c[ms] += delta
    h[p] = np.clip(h[p] + d, 0.0, 1.0)

def coord_cd(h, n, max_passes=500, deadline=None, rng=None, shuffle=True):
    h = h.copy()
    c = corr_full(h)
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                apply_delta(h, p, d, c, n)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = corr_full(h)
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def topk_sg_step(h, n, K=5, lr=0.04):
    h = h.copy()
    c = corr_full(h)
    top_idx = np.argsort(c)[-K:]
    sg = np.zeros(n)
    for k_val in top_idx:
        shift = int(k_val) - (n - 1)
        if shift >= 0:
            for i in range(n - shift):
                sg[i] += (1.0 - h[i + shift])
        else:
            s2 = -shift
            for i in range(s2, n):
                sg[i] -= h[i - s2]
    sg /= K
    return project_feasible(h - lr * sg, n)

def sum_eq_fun(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = corr_full(x)
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alpha, maxiter=500, deadline=None):
    best_c = compute_c5(h, n)
    try:
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
        res = minimize(lse_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            return new_h, new_c
    except Exception:
        pass
    return h, best_c

def basin_jump(h, n, jump, deadline, rng, cd_passes=60):
    h_new = project_feasible(h + rng.normal(0, jump, n), n)
    return coord_cd(h_new, n, max_passes=cd_passes, deadline=deadline, rng=rng)

def coarse_refine(h, n, n_coarse, deadline, rng):
    x_old = np.linspace(0, 2, n)
    x_coarse = np.linspace(0, 2, n_coarse)
    hc = project_feasible(np.interp(x_coarse, x_old, h), n_coarse)
    hc_opt, _ = coord_cd(hc, n_coarse, max_passes=400, deadline=deadline, rng=rng)
    hf = project_feasible(np.interp(x_old, x_coarse, hc_opt), n)
    return coord_cd(hf, n, max_passes=150, deadline=deadline, rng=rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_n = n_init
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Polish initial
    h1, c1 = coord_cd(best_h, best_n, max_passes=300, deadline=deadline, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Coarse-to-fine (try several coarse sizes)
    for nc in [100, 150, 200, 250]:
        if time.time() > deadline - 350:
            break
        h_r, c_r = coarse_refine(best_h, best_n, nc, deadline, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 3: Try different n values
    for n_try in [800, 500, 1000]:
        if time.time() > deadline - 250:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = coord_cd(h_int, n_try, max_passes=200, deadline=deadline, rng=rng)
        if c_t < best_c:
            best_h, best_c, best_n = h_t, c_t, n_try

    # Phase 4: Top-K subgradient cycling + CD
    n_cur = len(best_h)
    for K in [3, 5, 10, 20]:
        if time.time() > deadline - 200:
            break
        for _ in range(3):
            if time.time() > deadline - 200:
                break
            h_sg = topk_sg_step(best_h, n_cur, K=K, lr=0.03)
            h_cd, c_cd = coord_cd(h_sg, n_cur, max_passes=100, deadline=deadline, rng=rng)
            if c_cd < best_c:
                best_h, best_c = h_cd, c_cd

    # Phase 5: Basin jumping with structured perturbations
    n_cur = len(best_h)
    freqs = [2, 3, 5, 7, 11]
    for r in range(40):
        if time.time() > deadline - 50:
            break
        jump = 0.004 + 0.003 * (r % 4)
        base_pert = rng.normal(0, jump, n_cur)
        if r % 5 == 0:
            freq = freqs[(r // 5) % len(freqs)]
            base_pert += 0.015 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur)
        elif r % 5 == 1:
            freq = freqs[(r // 5) % len(freqs)]
            base_pert += 0.015 * np.cos(2 * np.pi * freq * np.arange(n_cur) / n_cur)
        h_new = project_feasible(best_h + base_pert, n_cur)
        h_j, c_j = coord_cd(h_new, n_cur, max_passes=80, deadline=deadline, rng=rng)
        if c_j < best_c:
            best_h, best_c = h_j, c_j

    # Phase 6: SLSQP polish
    n_cur = len(best_h)
    if time.time() < deadline - 60:
        for alpha in [20000, 80000]:
            if time.time() > deadline - 40:
                break
            h_s, c_s = slsqp_opt(best_h, n_cur, alpha, maxiter=400, deadline=deadline)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
            h_cd, c_cd = coord_cd(best_h, n_cur, max_passes=200, deadline=deadline, rng=rng)
            if c_cd < best_c:
                best_h, best_c = h_cd, c_cd

    # Phase 7: Extended final CD
    n_cur = len(best_h)
    hf, cf = coord_cd(best_h, n_cur, max_passes=800, deadline=deadline, rng=rng)
    if cf < best_c:
        best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
