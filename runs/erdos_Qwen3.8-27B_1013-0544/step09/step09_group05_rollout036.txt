def active_set_perturb(h, n, rng, num_steps=20):
    h = h.copy()
    for s in range(num_steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        # Active set: lags within 1% of max (or top 10)
        tol = 0.01 * c_max
        active = np.where(c >= c_max - tol)[0]
        # Or just top k
        if len(active) > 20:
            top_indices = np.argsort(c)[-20:]
            active = top_indices
        
        # Compute gradient direction that reduces active lags
        # ∂c[t]/∂h[j] = (1-h[j+k]) for j as first index + (-h[j-k]) for j as second index
        # where k = t - (n-1)
        grad = np.zeros(n)
        for t in active:
            k = t - (n - 1)
            # For each j, compute ∂c[t]/∂h[j]
            # j as first index: i=j, need 0≤j<n and 0≤j+k<n
            valid_j_first = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            if np.any(valid_j_first):
                js = np.where(valid_j_first)[0]

def correlation_gradient(h, n, active_lags):
    """Compute sum of gradients of active correlation entries w.r.t. h."""
    grad = np.zeros(n)
    for t in active_lags:
        k = t - (n - 1)
        # Term 1: (1-h[j+k]) for valid j
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            js = np.arange(lo1, hi1)
            grad[js] += (1.0 - h[js + k])
        # Term 2: -h[j-k] for valid j
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            js = np.arange(lo2, hi2)
            grad[js] += (-h[js - k])
    return grad

    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p]
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_active(h, n, deadline, max_steps=40):
    h = h.copy()
    for s in range(max_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        active = np.argsort(c)[-12:]
        grad = np.zeros(n)
        for t in active:
            k = int(t) - (n - 1)
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            if hi1 > lo1:
                js = np.arange(lo1, hi1)
                grad[js] += (1.0 - h[js + k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if hi2 > lo2:
                js = np.arange(lo2, hi2)
                grad[js] += (-h[js - k])
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        for lr in [0.15, 0.08, 0.04, 0.02, 0.01]:
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = float(np.correlate(h_new, 1.0 - h_new, mode='full').max())
            if c_new < c_max - 1e-12:
                h = h_new
                improved = True
                break
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: CD to convergence from parent
    h1, c1 = coord_descent(best_h, n, 400, deadline, rng, random_order=True)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Basin hopping via subgradient + CD
    for iteration in range(40):
        if time.time() > deadline - 30:
            break
        h_sg, c_sg = subgrad_active(best_h, n, deadline, max_steps=35)
        h_o, c_o = coord_descent(h_sg, n, 200, deadline, rng, random_order=True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
        if time.time() > deadline - 60:
            break
        hp = h_sg + rng.normal(0, 0.04, n)
        hp = project_feasible(hp, n)
        h_o, c_o = coord_descent(hp, n, 150, deadline, rng, random_order=True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 3: Diverse structured restarts
    if time.time() < deadline - 50:
        n_restarts = int(min(25, max(5, (deadline - time.time()) / 20)))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            strat = r % 6
            if strat == 0:
                hp = best_h + rng.normal(0, 0.06, n)
            elif strat == 1:
                hp = best_h.copy()
                nb = rng.randint(2, 6)
                cuts = np.sort(rng.choice(range(1, n), nb - 1, replace=False))
                bounds = np.concatenate([[0], cuts, [n]])
                for bi in range(nb):
                    if rng.rand() < 0.5:
                        hp[bounds[bi]:bounds[bi+1]] = 1.0 - hp[bounds[bi]:bounds[bi+1]]
            elif strat == 2:
                f = int(rng.randint(1, 14))
                ph = rng.uniform(0, 6.2832)
                x = np.linspace(0, 2, n)
                hp = best_h + 0.12 * rng.rand() * np.sin(f * np.pi * x + ph)
            elif strat == 3:
                s = 0.4 + 1.2 * rng.rand()
                hp = 0.5 + s * (best_h - 0.5)
            elif strat == 4:
                ns = n // 2
                x_full = np.linspace(0, 2, n)
                x_half = np.linspace(0, 2, ns)
                h_dn = project_feasible(np.interp(x_half, x_full, best_h), ns)
                h_dn, _ = coord_descent(h_dn, ns, 40, deadline, rng, random_order=True)
                hp = np.interp(x_full, x_half, h_dn)
            else:
                nmod = n // 3
                idx = rng.choice(n, nmod, replace=False)
                hp = best_h.copy()
                hp[idx] = rng.beta(2, 2, nmod)
            hp = project_feasible(hp, n)
            h_o, c_o = coord_descent(hp, n, 150, deadline, rng, random_order=True)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 4: Try nearby n values
    if time.time() < deadline - 40:
        for n_try in [n - 5, n + 5, n - 10, n + 10]:
            if time.time() > deadline - 20:
                break
            if n_try < 100:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h_opt, c_opt = coord_descent(h_int, n_try, 150, deadline, rng, random_order=True)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 5: Final intensive CD polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        hf, cf = coord_descent(best_h, n_f, 600, deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 6: One more subgradient cycle on best
    if time.time() < deadline - 5:
        h_sg, c_sg = subgrad_active(best_h, len(best_h), deadline, max_steps=25)
        h_o, c_o = coord_descent(h_sg, len(best_h), 200, deadline, rng, random_order=True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
