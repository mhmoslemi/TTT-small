import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-14:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
    return np.clip(h_p, 0.0, 1.0)

def cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, rng=None, shuffle=True):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if rng is None:
        rng = np.random.RandomState(42)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        order = rng.permutation(n) if shuffle else np.arange(n)
        for idx in order:
            if time.time() > deadline:
                break
            p_i = int(idx)
            h0 = h[p_i]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p_i)
            m_hi = min(2 * n - 2, n - 1 + p_i)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p_i + ms - (n - 1)
            jb = p_i - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_diag = len(dp) > 0
            diag_idx = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()])
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-16:
                h[p_i] = h0 + d_opt
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    dx = 2.0 / n
    def smooth_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * dx
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def _fun(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / alpha) * dx
        try:
            res = minimize(_fun, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                        'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def basin_hopping(h, n, max_iter, deadline, rng):
    best_h, best_c = h.copy(), compute_c5(h, n)
    temp = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        perturb_type = rng.randint(3)
        if perturb_type == 0:
            hp = project_feasible(best_h + rng.normal(0, temp, n), n)
        elif perturb_type == 1:
            x = np.linspace(0, 2, n)
            freq = 1.0 + 8.0 * rng.random()
            amp = 0.01 + 0.05 * rng.random()
            hp = project_feasible(best_h + amp * np.sin(2 * np.pi * freq * x), n)
        else:
            mask = rng.random(n) < 0.1
            shift = rng.normal(0, temp, n) * mask
            hp = project_feasible(best_h + shift, n)
        h_o, c_o = coord_descent(hp, n, 15, deadline, gs_iters=35, rng=rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            temp = max(0.005, temp * 0.85)
        else:
            accept_prob = np.exp(-(c_o - best_c) / (temp * 0.01))
            if rng.random() < accept_prob:
                best_h, best_c = h_o, c_o
            temp = max(0.005, temp * 1.05)
    return best_h, best_c
