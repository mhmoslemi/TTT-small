    for _pass in range(max_passes):

            break

        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    lx = shifted.max()
    lse = lx + np.log(np.sum(np.exp(shifted - lx)))
    obj = (m + lse / alpha) * (2.0 / n)
    w = np.exp(shifted - lx)
    w = w / w.sum()
    grad = np.zeros(n)
    for j in range(n):
        m1_lo = max(0, n - 1 - j)
        m1_hi = min(2 * n - 2, 2 * n - 2 - j)
        if m1_lo <= m1_hi:
            ms1 = np.arange(m1_lo, m1_hi + 1)
            idx1 = j + ms1 - (n - 1)
            grad[j] += np.sum(w[ms1] * (1.0 - x[idx1]))
        m2_lo = max(0, j - (n - 1))
        m2_hi = min(2 * n - 2, j)
        if m2_lo <= m2_hi:
            ms2 = np.arange(m2_lo, m2_hi + 1)
            idx2 = j - ms2 + (n - 1)
            grad[j] -= np.sum(w[ms2] * x[idx2])
    grad *= (2.0 / n)
    return obj, grad

            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass

def structured_perturb(h, n, rng, n_freqs):
    x = np.arange(n)
    pert = np.zeros(n)
    for freq in range(1, n_freqs + 1):

        b = rng.uniform(-0.04, 0.04)
        pert += a * np.cos(2.0 * np.pi * freq * x / n) + b * np.sin(2.0 * np.pi * freq * x / n)
    return project_feasible(h + pert, n)

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: establish baseline with CD
    h1, c1 = coord_descent(best_h, n, 150, deadline, rng, True)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: L-BFGS-B with increasing sharpness
    h2, c2 = lbfgs_run(best_h, n, [200, 1000, 5000, 20000], 400, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: CD polish
    h3, c3 = coord_descent(best_h, n, 200, deadline, rng, True)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: try different n via interpolation
    n = len(best_h)
    if time.time() < deadline - 150:
        x_orig = np.linspace(0, 2, n)
        for n_try in [500, 840]:
            if time.time() > deadline - 60:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_opt, c_opt = coord_descent(h_int, n_try, 100, deadline, rng, True)
            h_opt, c_opt = lbfgs_run(h_opt, n_try, [1000, 5000], 300, deadline)
            h_opt, c_opt = coord_descent(h_opt, n_try, 100, deadline, rng, True)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                x_orig = x_new
                n = n_try
    # Phase 5: structured and random restarts
    n = len(best_h)
    if time.time() < deadline - 80:
        n_rest = max(3, int((deadline - time.time() - 40) / 90))
        for r in range(n_rest):
            if time.time() > deadline - 30:
                break
            if r % 3 == 0:
                hp = structured_perturb(best_h, n, rng, 3 + r)
            elif r % 3 == 1:
                s = 0.015 + 0.015 * r
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            else:
                hp = structured_perturb(best_h, n, rng, 6 + r)
            ho, co = coord_descent(hp, n, 80, deadline, rng, True)
            if time.time() < deadline - 15:
                ho, co = lbfgs_run(ho, n, [2000, 10000], 200, deadline)
                ho, co = coord_descent(ho, n, 80, deadline, rng, True)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 6: final heavy CD polish
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = coord_descent(best_h, n, 300, deadline, rng, False)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 7: last L-BFGS + CD
    if time.time() < deadline - 8:
        n = len(best_h)
        hf, cf = lbfgs_run(best_h, n, [5000, 50000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        if time.time() < deadline - 3:
            hf, cf = coord_descent(best_h, len(best_h), 100, deadline)
            if cf < best_c:
                best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
