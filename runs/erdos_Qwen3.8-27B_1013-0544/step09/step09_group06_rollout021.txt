       dx = 2.0 / n
       return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

       return np.correlate(h, 1.0 - h, mode='full')

   def compute_gradients(h, active_lags, n):
       """Compute ∂c[m]/∂h[p] for m in active_lags"""
       result = np.zeros((len(active_lags), n))
       for idx, m in enumerate(active_lags):
           shift = m - (n - 1)
           # (1 - h[p+shift]) for valid p+shift
           for p in range(n):
               val = 0.0
               if 0 <= p + shift < n:
                   val += 1.0 - h[p + shift]
               if 0 <= p - shift < n:
                   val -= h[p - shift]
               result[idx, p] = val
       return result

       h = np.clip(h.copy(), 0.0, 1.0)

           if diff > 0:
               room = 1.0 - h

               if tot < 1e-15:
                   break
               h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_correlation(h, n)) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_for_shift(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += 1.0 - h[s:]
        if s > 0:
            grad[s:] -= h[:n - s]
    else:
        ss = -s
        grad[ss:] += 1.0 - h[:n - ss]
        grad[:n - ss] -= h[ss:]
    return grad

def lp_step(h, n, active_idx):
    c = compute_correlation(h, n)
    c_top = c[active_idx]
    K = len(active_idx)
    G = np.zeros((K, n))
    for i in range(K):
        s = int(active_idx[i]) - (n - 1)
        G[i] = grad_for_shift(h, s, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for i in range(K):
        rhs = c_top[i] - G[i] @ h
        cons.append(G[i] @ h_var <= t_var - rhs)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iter=50000, abstol=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.array(h_var.value).flatten()

def lp_active_set(h, n, deadline, max_rounds=300):
    c = compute_correlation(h, n)
    current_max = float(c.max())
    active_k = 5
    no_improve = 0
    for rnd in range(max_rounds):

        top_idx = np.argsort(c)[::-1][:active_k]

        if h_new is None:

            no_improve += 1
            if no_improve > 15:

            continue
        h_new = project_feasible(h_new, n)

                h = h_try
                current_max = c_try
                c = compute_correlation(h, n)

        if not improved:

    return h, compute_c5(h, n)

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_correlation(h, n)
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = compute_correlation(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
