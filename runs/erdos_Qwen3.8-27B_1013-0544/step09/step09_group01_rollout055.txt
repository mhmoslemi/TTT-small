    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0

            tot = room.sum()

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

        c = np.correlate(h, 1.0 - h, mode='full')
        h_before = h.copy()
        h = cd_sweep(h, n, c, gs_iters)
        h = project_feasible(h, n)
        if np.max(np.abs(h - h_before)) < 1e-14:
            break
    return h, compute_c5(h, n)

def boundary_push(h, n, threshold):
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP to refine starting point
    h1, c1 = slsqp_optimize(best_h, n, [200, 1000, 5000, 50000, 200000], 400)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Long coarse CD
    h2, c2 = coord_descent(best_h, n, 60, deadline - 350, gs_iters=35)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Try alternative n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [550, 650, 750, 850]:
        if time.time() > deadline - 500:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_optimize(h_int, n_try, [2000, 20000, 200000], 350)
        h_o, c_o = coord_descent(h_o, n_try, 40, deadline - 500, gs_iters=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 4: Boundary push attempts
    if time.time() < deadline - 250:
        n_cur = len(best_h)
        for thresh in [0.003, 0.008, 0.015, 0.03, 0.06]:
            if time.time() > deadline - 250:
                break
            hp = boundary_push(best_h, n_cur, thresh)
            h_o, c_o = coord_descent(hp, n_cur, 30, deadline - 250, gs_iters=45)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Small perturbation restarts
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 150
        n_restarts = max(2, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 150:
                break
            s = 0.005 + 0.015 * (r / max(1, n_restarts))
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 35, deadline - 150, gs_iters=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Fine CD (tight golden section)
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 60
        max_passes = max(8, int(remaining / 5.0))
        h_f, c_f = coord_descent(best_h, n_cur, max_passes, deadline - 60, gs_iters=80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 7: SLSQP final polish
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [500000, 2000000, 5000000], 1000)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 8: Ultra-fine CD
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 20, deadline - 2, gs_iters=120)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
