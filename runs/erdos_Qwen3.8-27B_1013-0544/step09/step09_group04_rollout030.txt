def cd_pass(h, n, order='sequential', deadline=None):
    """One pass of coordinate descent."""
    h = h.copy()
    n_idx = np.arange(n)
    
    if order == 'random':
        idx = n_idx.copy()
        np.random.shuffle(idx)
    elif order == 'priority':
        c = np.correlate(h, 1.0 - h, mode='full')
        # Priority: coordinates that appear in the top lags
        top_lags = np.argsort(c)[-min(50, 2*n-1):]
        # Count how many top lags each coordinate appears in
        priority = np.zeros(n)
        for m in top_lags:
            k = m - (n-1)
            lo = max(0, -k)
            hi = min(n, n - k)
            for i in range(lo, hi):
                priority[i] += 1
        idx = n_idx[np.argsort(-priority)]
    else:
        idx = n_idx
    
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    
    for p in idx:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        x_new = solve_coord(c, h, p, n)
        if abs(x_new - h0) > 1e-13:
            h[p] = x_new
            improved = True
    
    if improved:
        h = project_feasible(h, n)
    return h, improved

def multi_cd(h, n, max_passes, deadline, order='sequential'):
    """Multiple passes of coordinate descent."""
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, improved = cd_pass(h, n, order, deadline)
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_init(n, seed):
    """Create a smooth initial guess."""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)

    amps = rng.uniform(0.1, 0.4, size=5)

        g += a * np.sin(2*np.pi*f*x + ph)

def periodic_init(n, period, seed):

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n, lam=0.0):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_aff = c[ms].copy()
    di = (ms == (n - 1))
    has_di = np.any(di)
    c_diag = 0.0
    if has_di:
        c_diag = c_aff[np.where(di)[0][0]]
        c_aff[di] = 0.0
        a[di] = 0.0
    unaff_max = 0.0
    if ms[0] > 0:
        unaff_max = float(c[:ms[0]].max())
    if ms[-1] < 2 * n - 2:
        unaff_max = max(unaff_max, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    b = a + lam
    d_const = c_aff - h0 * a

    L_hi = cur_max + abs(lam)

        if np.any(neg_b):

        x_hi_v = min(x_hi_v, 1.0)

            disc = p_coef * p_coef - 4 * q_coef

                r1 = (-p_coef - sq) / 2
                r2 = (-p_coef + sq) / 2

                x2 = h0 + r2
                if x_lo > x1 and x_hi_v < x2:
                    feasible = False
        if feasible:

        else:
            L_lo = T

    x_lo = 0.0

    pos_b = b > 1e-15
    neg_b = b < -1e-15

    if np.any(neg_b):
        x_lo = max(x_lo, float(np.max((T - d_const[neg_b]) / b[neg_b])))
    x_lo = max(x_lo, 0.0)
    x_hi_v = min(x_hi_v, 1.0)
    if has_di:
        p_coef = -(1 - 2 * h0 + lam)
        q_coef = T - c_diag - lam * h0
        disc = p_coef * p_coef - 4 * q_coef

def cd_pass(h, n, order='sequential', lam=0.0, deadline=None, rng=None):
    h = h.copy()
    n_idx = np.arange(n)
    if order == 'random' and rng is not None:
        idx = n_idx.copy()
        rng.shuffle(idx)
    elif order == 'priority':
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[-min(80, 2 * n - 1):]
        priority = np.zeros(n)
        for m in top:
            k = m - (n - 1)
            lo = max(0, -k)
            hi = min(n, n - k)
            priority[lo:hi] += 1.0
        idx = n_idx[np.argsort(-priority)]
    else:
        idx = n_idx
    c = np.correlate(h, 1.0 - h, mode='full')
    improved = False
    for p in idx:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        x_new = solve_coord(c, h, p, n, lam)
        if abs(x_new - h0) > 1e-13:
            h[p] = x_new
            improved = True
    if improved and lam == 0.0:
        h = project_feasible(h, n)
    return h, improved

def multi_cd(h, n, max_passes, deadline, order='sequential', lam=0.0, rng=None):
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h, improved = cd_pass(h, n, order, lam, deadline, rng)
        if not improved:
            break
    return h
