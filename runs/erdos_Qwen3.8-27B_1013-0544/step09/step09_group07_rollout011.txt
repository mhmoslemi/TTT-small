    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max weights
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / exp_c.sum()  # weights for each shift
    
    # Compute gradient
    # For each shift m (0 to 2n-2) and each index i:
    # d(c[m])/d(h[i]) = (1-h[j_f]) * [0<=j_f<n] + (-h[j_b]) * [0<=j_b<n]
    # where j_f = i + m - (n-1), j_b = i - m + (n-1)

            continue
        weight = p[m] / n  # the 1/n factor from the smooth max

            jb = i - m + (n-1)
            if 0 <= jb < n:

    return grad

def smooth_gradient(h, n, alpha):
    """Compute gradient of smooth max approximation of C5."""
    dx = 2.0 / n
    raw_c = np.correlate(h, 1.0 - h, mode='full')
    scaled = raw_c * dx
    s_max = scaled.max()
    exp_scaled = np.exp(alpha * (scaled - s_max))
    p = exp_scaled / exp_scaled.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in range(2*n-1):
        if p[m] < 1e-14:
            continue
        w = p[m]
        lo_f = max(0, n-1-m)
        hi_f = min(n-1, 2*n-2-m)
        if lo_f <= hi_f:
            jf = np.arange(lo_f, hi_f+1) + m - (n-1)
            grad[lo_f:hi_f+1] += w * one_minus_h[jf]
        lo_b = max(0, m-n+1)
        hi_b = min(n-1, m+n-1)
        if lo_b <= hi_b:
            jb = np.arange(lo_b, hi_b+1) - m + (n-1)
            grad[lo_b:hi_b+1] -= w * h[jb]
    
    return dx * grad

def nesterov_smooth_opt(h, n, alphas, max_iter=2000, deadline=None, momentum0=0.9):
    """Nesterov momentum optimization of smooth max."""
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        momentum = momentum0
        lr = 0.5 / alpha  # initial learning rate
        for it in range(max_iter):
            if deadline is not None and time.time() > deadline:
                return h, best_c
            
            # Nesterov lookahead
            y = h + momentum * v
            y_clipped = np.clip(y, 0.0, 1.0)
            
            # Gradient at y (or at h for simplicity)
            g = smooth_gradient(y_clipped, n, alpha)
            
            # Gradient step
            v = momentum * v - lr * g
            h_new = h + v
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
            else:
                # Backtrack
                lr *= 0.7
                momentum = momentum * 0.95
                if lr < 1e-10:
                    break
            
            # Adaptive momentum
            momentum = min(momentum * 0.9999 + 0.00005, 0.95)
    
    return h, best_c

The problem is min_h max_m f_m(h) where f_m(h) = c[m]*dx.

The Moreau envelope of max_m f_m is:
F_μ(h) = min_v { max_m f_m(v) + (1/2μ)||v-h||² }

The gradient of F_μ at h is (v*(h) - h)/μ where v*(h) = argmin.

This is equivalent to a proximal step: v*(h) = prox_{μ * max_m f_m}(h).

Since max_m f_m is not easily prox-able, we can approximate it with the smooth max and then the proximal step becomes a regularized quadratic problem.
