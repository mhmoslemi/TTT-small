def project_feasible(h, n):

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            # Actually, distribute proportionally to available room

    if has_diag:

    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=80, random_order=False, rng=None):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    order = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            rng.shuffle(order)
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_eval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def reflect_complement(h):
    return 1.0 - h[::-1].copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except NameError:
        n = 700
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Check symmetry: try reflected complement
    h_ref = project_feasible(reflect_complement(best_h), n)
    c_ref = compute_c5(h_ref, n)
    if c_ref < best_c:
        best_h, best_c = h_ref, c_ref

    # Phase 1: Aggressive CD with random order
    h1, c1 = coord_descent(best_h, n, 150, deadline, gs_iters=80, random_order=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Iterated local search with diverse perturbations
    remaining = deadline - time.time()
    n_restarts = max(5, min(40, int(remaining / 60)))
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        s = 0.015 + 0.04 * (r / max(1, n_restarts - 1))
        strategy = r % 5
        if strategy == 0:
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
        elif strategy == 1:
            x = np.linspace(0, 2, n)
            freq = 1.5 + 4.0 * (r / max(1, n_restarts))
            hp = project_feasible(best_h + 0.04 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 6.28)), n)
        elif strategy == 2:
            hp = best_h.copy()
            idx = rng.choice(n, size=min(30, n // 5), replace=False)
            for i in idx:
                hp[i] = 1.0 - hp[i]
            hp = project_feasible(hp, n)
        elif strategy == 3:
            hp = reflect_complement(best_h) + rng.normal(0, 0.02, n)
            hp = project_feasible(hp, n)
        else:
            hp = best_h.copy()
            bs = rng.randint(0, n - 80)
            bl = 30 + rng.randint(0, 60)
            hp[bs:bs + bl] = rng.uniform(0.2, 0.8, bl)
            hp = project_feasible(hp, n)
        h_o, c_o = coord_descent(hp, n, 60, deadline, gs_iters=70, random_order=True, rng=rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            # Also check symmetry of the new best
            h_ref2 = project_feasible(reflect_complement(best_h), n)
            c_ref2 = compute_c5(h_ref2, n)
            if c_ref2 < best_c:
                best_h, best_c = h_ref2, c_ref2

    # Phase 3: Multi-resolution
    if time.time() < deadline - 120:
        x_orig = np.linspace(0, 2, len(best_h))
        n_list = [500, 550, 600, 750, 800, 850, 900]
        n_list = [x for x in n_list if x != len(best_h)]
        for n_try in n_list:
            if time.time() > deadline - 60:
                break
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_o, c_o = coord_descent(h_int, n_try, 80, deadline, gs_iters=75, random_order=True, rng=rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_orig = x_new

    # Phase 4: Final polish - many CD passes
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 5
        max_passes = max(20, int(remaining / 1.5))
        h_f, c_f = coord_descent(best_h, n_cur, max_passes, deadline, gs_iters=90, random_order=True, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final symmetry check
    h_ref = project_feasible(reflect_complement(best_h), len(best_h))
    c_ref = compute_c5(h_ref, len(best_h))
    if c_ref < best_c:
        best_h, best_c = h_ref, c_ref

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
