def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # Sharpen for numerical stability
    c_shifted = alpha * (c - c.max())
    log_sum_exp = c_shifted.max() + np.log(np.sum(np.exp(c_shifted - c_shifted.max())))
    w = np.exp(c_shifted - log_sum_exp)  # normalized weights
    g = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        # Forward: i + k - (n-1) in [0, n-1] => i in [max(0, n-1-k), min(n-1, 2n-1-k)]
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 1 - k)
        if i_lo <= i_hi:
            j = np.arange(i_lo, i_hi + 1) + k - (n - 1)
            g[i_lo:i_hi+1] += wk * (1.0 - h[j])
        # Backward: i - k + (n-1) in [0, n-1] => i in [max(0, k-n+1), min(n-1, k)]
        i_lo2 = max(0, k - n + 1)
        i_hi2 = min(n - 1, k)
        if i_lo2 <= i_hi2:
            j2 = np.arange(i_lo2, i_hi2 + 1) - k + (n - 1)
            g[i_lo2:i_hi2+1] -= wk * h[j2]
    g *= dx
    return g, c

        if rng is not None:
            rng.shuffle(order)

def smooth_grad_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_ = c - c.max()
    lse = np.log(np.sum(np.exp(alpha * c_)))
    f = (c.max() + lse / alpha) * dx
    w = np.exp(alpha * c_ - lse)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 5e-7:
            continue
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 1 - k)
        if i_lo <= i_hi:
            j = i_lo + k - (n - 1)
            g[i_lo:i_hi+1] += wk * (1.0 - h[j:j + (i_hi - i_lo + 1)])
        i_lo2 = max(0, k - n + 1)
        i_hi2 = min(n - 1, k)
        if i_lo2 <= i_hi2:
            j2 = i_lo2 - k + (n - 1)
            g[i_lo2:i_hi2+1] -= wk * h[j2:j2 + (i_hi2 - i_lo2 + 1)]
    g *= dx
    return g, f, c

def gd_nesterov(h0, n, alpha, lr0, max_iter, deadline, momentum=0.85):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        y = h + momentum * v
        g, _, _ = smooth_grad_obj(y, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = momentum * v - (lr / gn) * g
        h_new = project_feasible(y + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.3)
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= 0.7
            if lr < 1e-5:
                break
            v *= 0.5
        h = h_new
    return best_h, best_c

def coord_cd(h0, n, max_passes=150, deadline=None, rng=None):
    h = h0.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            h0v = h[p]
            if h0v <= 1e-10 or h0v >= 1.0 - 1e-10:
                continue
            k_lo = max(0, n - 1 - p)
            k_hi = min(2 * n - 2, n - 1 + p)
            nk = k_hi - k_lo + 1
            if nk <= 0:
                continue
            c_aff = c[k_lo:k_hi+1]
            fwd = np.zeros(nk)
            bwd = np.zeros(nk)
            for idx in range(nk):
                k = k_lo + idx
                jf = p + k - (n - 1)
                jb = p - k + (n - 1)
                if 0 <= jf < n:
                    fwd[idx] = 1.0 - h[jf]
                if 0 <= jb < n:
                    bwd[idx] = h[jb]
            vec = fwd - bwd
            unaff = 0.0
            if k_lo > 0:
                unaff = max(unaff, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff = max(unaff, float(c[k_hi+1:].max()))
            lo, hi = -h0v, 1.0 - h0v
            for _ in range(35):
                mid = (lo + hi) * 0.5
                vals = c_aff + mid * vec
                mval = float(vals.max())
                if mval <= unaff + 1e-13:
                    lo = hi = mid
                    break
                active = vals >= mval - 1e-10
                vmin = float(vec[active].min())
                vmax = float(vec[active].max())
                if vmin > 1e-10:
                    hi = mid
                elif vmax < -1e-10:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            delta = (lo + hi) * 0.5
            if abs(delta) > 1e-12:
                c[k_lo:k_hi+1] += delta * vec
                h[p] = np.clip(h0v + delta, 0.0, 1.0)
                improved = True
        if p_idx % 8 == 7:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def upscale(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 600
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    print(f"Initial: n={n}, C5={best_c:.6f}")

    # Phase 1: Nesterov GD at current n with increasing alpha
    alphas = [40, 80, 160, 400, 1000, 3000, 8000, 20000]
    for alpha in alphas:
        if time.time() > deadline - 200:
            break
        h_new, c_new = gd_nesterov(best_h, n, alpha, lr0=0.12,
                                    max_iter=2500, deadline=deadline - 50)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            print(f"  GD n={n} a={alpha}: C5={best_c:.6f}")

    # Phase 2: CD at current n
    h_cd, c_cd = coord_cd(best_h, n, max_passes=200, deadline=deadline - 80, rng=rng)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
        print(f"  CD n={n}: C5={best_c:.6f}")

    # Phase 3: Upscale to higher resolution
    n2 = min(n * 2, 1200)
    if n2 > n and time.time() < deadline - 300:
        h2 = project_feasible(upscale(best_h, n, n2), n2)
        for alpha in [200, 800, 3000, 10000]:
            if time.time() > deadline - 120:
                break
            h_new, c_new = gd_nesterov(h2, n2, alpha, lr0=0.08,
                                        max_iter=2000, deadline=deadline - 40)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                n = n2
                print(f"  GD n={n2} a={alpha}: C5={best_c:.6f}")
            h2 = h_new
        h_cd2, c_cd2 = coord_cd(best_h, n, max_passes=150, deadline=deadline - 40, rng=rng)
        if c_cd2 < best_c:
            best_c = c_cd2
            best_h = h_cd2
            print(f"  CD n={n2}: C5={best_c:.6f}")

    # Phase 4: Downscale exploits - try smaller n for different structure
    if time.time() < deadline - 80:
        n3 = max(400, n // 2)
        if n3 < n:
            h3 = project_feasible(upscale(best_h, n, n3), n3)
            h3c, c3 = coord_cd(h3, n3, max_passes=100, deadline=deadline - 40, rng=rng)
            if c3 < best_c:
                best_c = c3
                best_h = h3c
                n = n3
                print(f"  CD n={n3}: C5={best_c:.6f}")

    # Phase 5: Random restarts with CD
    for r in range(12):
        if time.time() > deadline - 30:
            break
        s = 0.005 + 0.003 * r
        hp = project_feasible(best_h + rng.normal(0, s, len(best_h)), len(best_h))
        h_r, c_r = coord_cd(hp, len(best_h), max_passes=60, deadline=deadline - 15, rng=rng)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
            print(f"  Restart {r}: C5={best_c:.6f}")

    # Phase 6: Final polish
    h_f, c_f = coord_cd(best_h, len(best_h), max_passes=300, deadline=deadline, rng=rng)
    if c_f < best_c:
        best_c = c_f
        best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    print(f"Final: n={len(best_h)}, C5={best_c:.6f}")
    return best_h, best_c, len(best_h)
