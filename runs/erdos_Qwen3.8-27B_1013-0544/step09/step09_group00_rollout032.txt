            if x_lo <= r1 and x_hi >= r2:
                if abs(h0 - r1) <= abs(h0 - r2):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            elif x_hi >= r2:
                x_lo = max(x_lo, r2)
    
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_step(h, n, p, c):
    """Update coordinate p and return new h[p] and updated c."""
    h0 = h[p]
    xn = solve_1d(c, h, p, n)
    delta = xn - h0
    if abs(delta) < 1e-14:
        return h0, False
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    h[p] = xn
    return xn, True

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if h[p] < 1e-14 or h[p] > 1.0 - 1e-14:
                continue
            _, did = cd_step(h, n, p, c)
            if did:
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h_init, n, alphas, maxiter, deadline):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x, n_=n, a_=alpha):
            return smooth_obj(x, n_, a_)
        def cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0
        def cons_jac(x, n_=n):
            return np.ones(n_)
        
        try:
            res = minimize(obj, best_h, args=(), method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_to(h, n_new):
    """Interpolate h from its current grid to a new grid of size n_new."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

    c = np.correlate(x, 1.0 - x, mode='full')
    # Softmax weights
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    w = exp_c / exp_c.sum()  # softmax weights
    
    # Gradient: (2/n) * sum_m w[m] * dc[m]/dx[j]
    # dc[m]/dx[j] for lag m (stored at index m+n-1):
    # = (1-x[j+m']) * [valid] - x[j-m''] * [valid]
    # where m' = m-(n-1) is the actual lag

    for m in range(2*n-1):
        if w[m] < 1e-15:
            continue
        lag = m - (n-1)
        # Contribution from x[i] term: (1-x[i+lag]) for i where 0<=i<n and 0<=i+lag<n
        # Contribution from (1-x[i+lag]) term: -x[i] for i where 0<=i+lag<n and 0<=i<n
        # i.e., for j = i: grad[j] += (1-x[j+lag]) if 0<=j+lag<n
        #      for j = i+lag: grad[j] -= x[j-lag] if 0<=j-lag<n

            valid_i = np.arange(0, n-lag)
            grad[valid_i] += w[m] * (1.0 - x[valid_i + lag])
            # j = i + lag ranges from lag to n-1
            valid_j = np.arange(lag, n)
            grad[valid_j] -= w[m] * x[valid_j - lag]
        else:
            # lag < 0, let lag2 = -lag > 0
            lag2 = -lag
            # i ranges from lag2 to n-1
            valid_i = np.arange(lag2, n)
            grad[valid_i] += w[m] * (1.0 - x[valid_i - lag2])
            # j = i + lag = i - lag2 ranges from 0 to n-1-lag2
            valid_j = np.arange(0, n - lag2)
            grad[valid_j] -= w[m] * x[valid_j + lag2]
    
    obj = (2.0 / n) * (max_c + np.log(exp_c.sum()) / alpha)
    grad *= (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c_arr, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c_arr[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c_arr[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c_arr[m_max + 1:])))
    cur_max = float(np.max(c_arr))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    if has_diag:
        zero_mask[diag_idx] = False
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc < 0:
                feasible = False
            else:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) <= r1 or min(x_hi, 1.0) >= r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if abs(h0 - r1) <= abs(h0 - r2):
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        s = m - (n - 1)
        if s == 0:
            grad += wm * (1.0 - 2.0 * x)
        elif s > 0:
            grad[:n - s] += wm * (1.0 - x[s:])
            grad[s:] -= wm * x[:n - s]
        else:
            t = -s
            grad[t:] += wm * (1.0 - x[:n - t])
            grad[:n - t] -= wm * x[t:]
    obj = (2.0 / n) * (max_c + np.log(exp_c.sum()) / alpha)
    grad *= (2.0 / n)
    if mu > 0:
        obj += mu * (x.sum() - n / 2.0) ** 2
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return obj, grad

def slsqp_opt(h_init, n, alphas, maxiter, deadline, use_grad=True):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        if use_grad:
            def obj(x, n_=n, a_=alpha):
                return smooth_obj_grad(x, n_, a_, 0.0)[0]
            def objg(x, n_=n, a_=alpha):
                return smooth_obj_grad(x, n_, a_, 0.0)
            def cf(x, n_=n):
                return x.sum() - n_ / 2.0
            def cj(x, n_=n):
                return np.ones(n_)
            try:
                res = minimize(obj, best_h, jac=objg, method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                              options={'maxiter': maxiter, 'ftol': 1e-16})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
        else:
            def obj_pen(x, n_=n, a_=alpha):
                o, g = smooth_obj_grad(x, n_, a_, 1e6)
                return o
            def obj_pen_g(x, n_=n, a_=alpha):
                o, g = smooth_obj_grad(x, n_, a_, 1e6)
                return o, g
            try:
                res = minimize(obj_pen, best_h, jac=obj_pen_g, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c

def lbfgs_penalty(h_init, n, alphas, mus, maxiter, deadline):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        for mu in mus:
            if time.time() > deadline:
                break
            def obj(x, n_=n, a_=alpha, m_=mu):
                return smooth_obj_grad(x, n_, a_, m_)[0]
            def objg(x, n_=n, a_=alpha, m_=mu):
                return smooth_obj_grad(x, n_, a_, m_)
            try:
                res = minimize(obj, best_h, jac=objg, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c

def interp_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def perturb_lowfreq(h, n, rng, scale=0.03):
    """Add low-frequency sinusoidal perturbation."""
    noise = np.zeros(n)
    idx = np.arange(n)
    for k in range(1, 8):
        noise += (rng.normal() * np.sin(2 * np.pi * k * idx / n) +
                  rng.normal() * np.cos(2 * np.pi * k * idx / n)) / k
    return project_feasible(h + scale * noise, n)

def perturb_block(h, n, rng):
    """Flip a random contiguous block."""
    hp = h.copy()
    length = rng.randint(3, max(4, n // 20))
    start = rng.randint(0, n - length)
    hp[start:start + length] = 1.0 - hp[start:start + length]
    return project_feasible(hp, n)
