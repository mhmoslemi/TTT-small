    for _ in range(100):

        if d > 0:

            if s < 1e-14: break

    return np.clip(h, 0.0, 1.0)

    if dl is not None:

    return max(v.max(), um)

        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 7: Perturbation restarts
    n_cur = len(best_h)
    d7 = deadline - 40
    for r in range(8):
        if time.time() > d7:
            break
        s = 0.002 + 0.008*r
        hp = proj(best_h + rng.normal(0, s, n_cur), n_cur)
        ho, co = cd(hp, n_cur, passes=150, dl_=d7, tk=30)
        if co < best_c:
            best_h, best_c = ho, ko if False else co

    # Phase 8: Final SLSQP with high alpha
    if time.time() < deadline - 25:
        h8, c8 = slsqp(best_h, n_cur, [5000, 20000, 50000], 400, deadline-20)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 8:
        h9, c9 = cd(best_h, n_cur, passes=1200, dl_=deadline-4, tk=40)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Try 1-h symmetry
    n_fin = len(best_h)
    h_flip = proj(1.0-best_h, n_fin)
    c_flip = c5val(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = proj(best_h, n_fin)
    best_c = c5val(best_h, n_fin)
    return best_h, best_c, n_fin
