        t_k = t_new
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Phase 0: Generate initial candidates
    candidates = []
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        candidates.append((project_feasible(h_init, n_init), n_init))
    except Exception:
        n_init = 700
        candidates.append((project_feasible(np.full(n_init, 0.5), n_init), n_init))

    # Add Fourier-based initializations
    for nt in [3, 5, 7]:
        candidates.append((fourier_init(n_init, nt, rng), n_init))
    # Add binary initializations
    for _ in range(3):
        candidates.append((binary_init(n_init, rng), n_init))

    # Phase 1: CD on each candidate, keep best
    best_h = candidates[0][0].copy()
    best_c = compute_c5(best_h, best_h.shape[0])
    best_n = best_h.shape[0]

    for h_c, n_c in candidates:
        if time.time() > deadline - 300:
            break
        h_o, c_o = coord_descent(h_c, n_c, 50, min(deadline - 250, t0 + 30), rng, random_order=True)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_c

    # Phase 2: Try different n values via interpolation
    n_list = [350, 500, 700, 840, 1050, 1400]
    n_list = [x for x in n_list if x != best_n]
    x_best = np.linspace(0, 2, best_n, endpoint=False)
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        x_try = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_try, x_best, best_h, period=2), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 80, min(deadline - 100, time.time() + 60), rng, random_order=True)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_try
            x_best = np.linspace(0, 2, best_n, endpoint=False)

    # Phase 3: Perturbation + re-optimization loop
    n = best_n
    noise_scales = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
    for scale in noise_scales:
        if time.time() > deadline - 60:
            break
        for attempt in range(3):
            if time.time() > deadline - 40:
                break
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_o, c_o = coord_descent(hp, n, 60, min(deadline - 30, time.time() + 40), rng, random_order=True)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 4: Binary rounding heuristic
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        if time.time() > deadline - 30:
            break
        h_bin = (best_h >= thresh).astype(float)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin
            # Polish the binary solution
            h_o, c_o = coord_descent(h_bin, n, 80, min(deadline - 20, time.time() + 30), rng, random_order=True)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Heavy CD polish
    if time.time() < deadline - 15:
        remaining = deadline - time.time() - 5
        max_p = max(30, int(remaining / 0.3))
        h_o, c_o = coord_descent(best_h, n, max_p, deadline - 3, rng, random_order=True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 6: Final sequential CD passes
    if time.time() < deadline - 5:
        h_o, c_o = coord_descent(best_h, n, 20, deadline - 1, rng, random_order=False)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
