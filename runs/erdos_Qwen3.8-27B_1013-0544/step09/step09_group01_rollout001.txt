def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-13:

            tot = room.sum()

        else:
            tot = h.sum()

    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
