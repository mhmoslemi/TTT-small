    S = h[p] + h[q]

        mhi = min(2*n-2, max(2*n-2-i, i+n-1))

        if np.any(vb): vec[vb] -= h[jb[vb]]
        return mlo, mhi, vec
    
    mlo_p, mhi_p, vec_p = get_vec(p)
    mlo_q, mhi_q, vec_q = get_vec(q)

    mlo = min(mlo_p, mlo_q)
    mhi = max(mhi_p, mhi_q)

    if mhi < 2*n-2: unaff = max(unaff, float(c[mhi+1:].max()))

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h

        if c < best_c:
            best_h, best_c = h.copy(), c

    # Phase 4: Try different grid resolutions
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in [450, 550, 700, 850]:
        if time.time() > deadline - 180:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        ht, ct = coord_cd(h_int, n_try, max_passes=120, deadline=deadline, rng=rng, shuffle=True)
        ht, ct = pair_cd(ht, n_try, max_passes=80, deadline=deadline, rng=rng, shuffle=True)
        if ct < best_c:
            best_h, best_c = ht.copy(), ct
            x_best = x_new

    # Phase 5: Random restarts on best solution
    for r in range(8):
        if time.time() > deadline - 100:
            break
        s = 0.003 + 0.005 * r
        hp = project_feasible(best_h + rng.normal(0, s, len(best_h)), len(best_h))
        hr, cr = coord_cd(hp, len(best_h), max_passes=80, deadline=deadline, rng=rng, shuffle=True)
        hr, cr = pair_cd(hr, len(best_h), max_passes=60, deadline=deadline, rng=rng, shuffle=True)
        if cr < best_c:
            best_h, best_c = hr.copy(), cr

    # Phase 6: Final intensive polish
    n_f = len(best_h)
    h_f, c_f = coord_cd(best_h, n_f, max_passes=250, deadline=deadline, rng=rng, shuffle=True)
    if c_f < best_c:
        best_h, best_c = h_f.copy(), c_f
    h_f, c_f = pair_cd(best_h, n_f, max_passes=150, deadline=deadline, rng=rng, shuffle=True)
    if c_f < best_c:
        best_h, best_c = h_f.copy(), c_f
    h_f, c_f = coord_cd(best_h, n_f, max_passes=150, deadline=deadline, rng=rng, shuffle=True)
    if c_f < best_c:
        best_h, best_c = h_f.copy(), c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
