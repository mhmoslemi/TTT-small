import time

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_of_c(c, alpha):
    m = c.max()
    if alpha < 1e-10:
        return m
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

    c = np.correlate(x, 1.0 - x, mode='full')
    smax = lse_of_c(c, alpha)
    s = x.sum() - n / 2.0
    return smax * (2.0 / n) + mu * s * s

    c = np.correlate(x, 1.0 - x, mode='full')
    smax = lse_of_c(c, alpha)
    s = x.sum() - n / 2.0
    # Penalty for variance of top-K correlation values
    idx = np.argsort(c)[-K:]

    return smax * (2.0 / n) + mu * s * s + var_pen

    return x.sum() - n / 2.0

    return np.ones(n)

import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)

            tot = h.sum()
            if tot < 1e-15:

    m = c.max()
    if alpha < 1e-6:
        return m
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

    c = np.correlate(x, 1.0 - x, mode='full')

    return np.ones(n)

    c = np.correlate(x, 1.0 - x, mode='full')

    smax = lse_of_c(c, alpha)
    s = x.sum() - n / 2.0

                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    tol = 1e-8
    active = np.where(c >= cmax - tol)[0]
    # Compute subgradient
    grad = np.zeros(n)
    for m in active:
        # d c[m] / d h[p] = 1 - 2*h[p + m - (n-1)] if valid
        # Actually from earlier derivation:
        # h[p] appears in c[m] at position p + m - (n-1) in the other factor
        # dc[m]/dh[p] = (1 - h[p+m-(n-1)]) - h[p+m-(n-1)] if p+m-(n-1) in [0,n)
        #              = 1 - 2*h[p+m-(n-1)]
        # But also h[p] can appear as the "other" factor:
        # h[i] * (1 - h[i+m-(n-1)]), if i+m-(n-1) = p, then i = p+m-(n-1)
        # dc/dh[p] from this term: -h[i] = -h[p+m-(n-1)]
        # Wait, both terms give the same index? Let me re-derive.
        
        # c[m] = sum_i h[i] * (1 - h[i+m-(n-1)])  (only valid i)
        # = sum_i h[i] - sum_i h[i] * h[i+m-(n-1)]
        
        # dc[m]/dh[p]:
        # From first term: 1 if p is a valid i (always true)
        # From second term: -h[p+m-(n-1)] if p+m-(n-1) is valid AND p is a valid i in the sum
        #                    -h[p+(n-1)-m] if p+(n-1)-m is valid AND p+(n-1)-m is a valid i
        
        # Wait, I need to be more careful. The sum is:
        # c[m] = sum_{i: 0<=i<n, 0<=i+m-(n-1)<n} h[i] * (1 - h[i+m-(n-1)])
        
        # Let's define the valid range: i_lo = max(0, n-1-m), i_hi = min(n-1, n-1+m)
        # Wait: i+m-(n-1) >= 0 means i >= n-1-m
        # i+m-(n-1) < n means i < n+m-(n-1) = m+1
        # 0 <= i < n is always the base constraint
        
        # So valid i: max(0, n-1-m) <= i <= min(n-1, m)
        
        i_lo = max(0, n - 1 - m)
        i_hi = min(n - 1, m)
        
        for p in range(n):
            # h[p] appears in c[m] if p is in [i_lo, i_hi] (first factor)
            if i_lo <= p <= i_hi:
                j = p + m - (n - 1)
                # From first sum: contributes 1
                # From second sum: contributes -h[j]
                grad[p] += 1.0 - h[j]
            # h[p] appears as the "second" factor if there exists i in [i_lo, i_hi]
            # such that i + m - (n-1) = p, i.e., i = p + (n-1) - m
            i2 = p + (n - 1) - m
            if i2 >= i_lo and i2 <= i_hi:
                # The term is h[i2] * (1 - h[p]), derivative wrt h[p] is -h[i2]
                grad[p] += -h[i2]
    
    # Take negative step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
