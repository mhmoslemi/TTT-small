            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        if np.any(active):
            vmin = float(vec[active].min())
            vmax = float(vec[active].max())
        else:
            vmin, vmax = 0.0, 0.0
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=100, deadline=None, shuffle=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def targeted_subgrad(h, n, max_iter=200, deadline=None, top_k=8):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.04
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_k:]
        sg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                for i in range(n - off):
                    sg[i] += (1.0 - h[i + off])
            else:
                o = -off
                for i in range(o, n):
                    sg[i] += (1.0 - h[i - o])
            if off >= 0:
                for i in range(off, n):
                    sg[i] -= h[i - off]
            else:
                o = -off
                for i in range(n - o):
                    sg[i] -= h[i + o]
        sg /= len(idx)
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-10:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD on parent
    h1, c1 = coord_cd(best_h, n_init, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Tournament over n values
    x_parent = np.linspace(0, 2, len(best_h), endpoint=False)
    candidates = [200, 250, 300, 400, 500, 700, 800, 900, 1000, 1200, 1440, 1600]
    results = [(best_c, best_h)]
    for n_try in candidates:
        if time.time() > deadline - 350:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_parent, best_h), n_try)
        h_q, c_q = coord_cd(h_int, n_try, max_passes=35, deadline=deadline, shuffle=True, rng=rng)
        results.append((c_q, h_q))
        # Also try random init for diversity
        if time.time() < deadline - 300:
            h_rand = project_feasible(0.5 + 0.25 * rng.randn(n_try), n_try)
            h_rq, c_rq = coord_cd(h_rand, n_try, max_passes=25, deadline=deadline, shuffle=True, rng=rng)
            if c_rq < best_c:
                results.append((c_rq, h_rq))

    results.sort(key=lambda x: x[0])

    # Phase 3: Thorough optimization on top candidates
    for c_val, h_val in results[:3]:
        if time.time() > deadline - 200:
            break
        if c_val >= best_c + 0.005:
            continue
        n_val = len(h_val)
        h_th, c_th = coord_cd(h_val, n_val, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if c_th < best_c:
            best_h, best_c = h_th, c_th
        if c_th < best_c + 0.002 and time.time() < deadline - 120:
            h_sg, c_sg = targeted_subgrad(h_th, n_val, max_iter=100, deadline=deadline, top_k=5)
            if c_sg < best_c:
                best_h, best_c = h_sg, c_sg
            h_cd2, c_cd2 = coord_cd(best_h, len(best_h), max_passes=100, deadline=deadline, shuffle=True, rng=rng)
            if c_cd2 < best_c:
                best_h, best_c = h_cd2, c_cd2

    # Phase 4: SLSQP
    if time.time() < deadline - 100:
        h_sl, c_sl = slsqp_polish(best_h, len(best_h), [3000, 10000, 50000], 500, deadline)
        if c_sl < best_c:
            best_h, best_c = h_sl, c_sl

    # Phase 5: Blur basin hopping
    n_cur = len(best_h)
    for sigma in [3, 6, 10, 16]:
        if time.time() > deadline - 60:
            break
        ksize = max(3, int(5 * sigma))
        kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / sigma) ** 2)
        kernel /= kernel.sum()
        h_blur = np.convolve(best_h, kernel, mode='same')
        h_blur = project_feasible(h_blur, n_cur)
        h_b, c_b = coord_cd(h_blur, n_cur, max_passes=60, deadline=deadline, shuffle=True, rng=rng)
        if c_b < best_c:
            best_h, best_c = h_b, c_b
            n_cur = len(best_h)

    # Phase 6: Random perturbations with varying scale
    for r in range(20):
        if time.time() > deadline - 40:
            break
        s = 0.001 + 0.002 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = coord_cd(hp, n_cur, max_passes=40, deadline=deadline)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            n_cur = len(best_h)

    # Phase 7: Final targeted subgradient + CD
    if time.time() < deadline - 30:
        h_tg, c_tg = targeted_subgrad(best_h, n_cur, max_iter=150, deadline=deadline, top_k=3)
        if c_tg < best_c:
            best_h, best_c = h_tg, c_tg
        hf, cf = coord_cd(best_h, len(best_h), max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if cf < best_c:

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
