    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,

            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass

def block_cd(h, n, block_size, max_passes, deadline):
    """Block coordinate descent: optimize blocks of coordinates together."""
    h = h.copy()
    best_c = compute_c5(h, n)
    n_blocks = n // block_size + (1 if n % block_size else 0)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for b in range(n_blocks):
            if time.time() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)

                continue

            dx = 1.0 - h_block  # 1 - h for the block
            
            def block_obj(xb, h_full=h, start=start, end=end, n=n):
                h_full[start:end] = xb
                return compute_c5(h_full, n)

    target = n / 2.0

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return np.ones(n)

                mask = (rng.rand(n_cur) < 0.1).astype(float)
                hp = project_feasible(best_h + 0.3 * mask - 0.15, n_cur)
            else:
                hp = project_feasible(best_h * (1.0 + 0.02 * np.sin(np.pi * x)), n_cur)
            h_o, c_o = slsqp_optimize(hp, n_cur, [5000, 30000], 250)
            h_o, c_o = coord_descent(h_o, n_cur, 40, deadline, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final CD polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 50, deadline, gs_iters=65)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [500000, 2000000], 600)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, len(best_h)
