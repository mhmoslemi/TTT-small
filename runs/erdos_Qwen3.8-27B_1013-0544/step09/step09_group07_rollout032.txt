def smooth_gradient(h, n, beta):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # Softmax weights
    c_max = c.max()
    weights = np.exp(beta * (c - c_max))
    weights /= weights.sum()
    # Gradient
    n_full = 2*n - 1
    grad = np.zeros(n)
    for m_idx in range(n_full):
        w = weights[m_idx]
        if w < 1e-10:
            continue
        m = m_idx - (n-1)
        # ∂c[m]/∂h[j] for all j
        if m == 0:
            grad += w * (1 - 2*h)
        else:
            if m > 0:
                # (1 - h[j+m]) for j = 0..n-1-m
                grad[:n-m] += w * (1 - h[m:])
                # -h[j-m] for j = m..n-1
                grad[m:] -= w * h[:n-m]
            else:
                mm = -m
                # (1 - h[j+m]) = (1 - h[j-mm]) for j = mm..n-1
                grad[mm:] += w * (1 - h[:n-mm])
                # -h[j-m] = -h[j+mm] for j = 0..n-1-mm
                grad[:n-mm] -= w * h[mm:]
    return grad
