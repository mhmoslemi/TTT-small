import numpy as np
from scipy.optimize import golden
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_ineq_matrix(g, n):
    """Build the (2n-1) x n matrix where row k has g at appropriate shifted positions."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        offset = k - (n - 1)
        # A[k, i] = g[i + offset] if 0 <= i + offset < n
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            A[k, i_start:i_end] = g[i_start + offset:i_end + offset]
    return A

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    
    def obj(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        return max(float(nv.max()), unaff_max)
    
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    
    best_d = 0.0
    best_val = obj(0.0)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj(dc)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj, gl, gh, maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=500, deadline=None, top_k=50, seed=0):

    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = rng.permutation(n)
        
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:

            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)

    for it in range(max_iters):

minimize t
subject to: for all k, sum_i h_new[i] * g[i+k-(n-1)] ≤ t
            sum(h_new) = n/2
            0 ≤ h_new ≤ 1

This gives the minimum possible max correlation when g is fixed. After solving, I set g = 1-h_new and repeat.

This is correct as a block coordinate descent step. The LP is valid because when g is fixed, each c[k] is linear in h_new.

import numpy as np
from scipy.optimize import golden
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_ineq_matrix(g, n):
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        offset = k - (n - 1)
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            A[k, i_start:i_end] = g[i_start + offset:i_end + offset]
    return A

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = _eval_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = _eval_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(lambda d: _eval_1d(d, c_aff, a_coeff, diag_local, unaff_max), gl, gh, maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
            except Exception:
                pass
    return best_d

def _eval_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_cd(h, n, max_passes=500, deadline=None, top_k=50, seed=0):
    h = h.copy()
    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if improved:
            h = project_feasible(h, n)
            new_c = compute_c5(h, n)
            if new_c < best_c:
                best_c = new_c
        else:
            break
    return h, best_c

def lp_block_cd(h, n, max_iters=20, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        g = 1.0 - h
        A = build_ineq_matrix(g, n)
        hv = cp.Variable(n)
        tv = cp.Variable()
        constraints = [A @ hv <= tv, cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1]
        prob = cp.Problem(cp.Minimize(tv), constraints)
        solved = False
        for solver in ['CBC', 'ECOS', 'PDLP', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or hv.value is None:
            break
        new_h = project_feasible(np.asarray(hv.value).flatten(), n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-13:
            h = new_h
            best_c = new_c
        else:
            break
    return h, best_c
