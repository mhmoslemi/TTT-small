import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = alpha * (c - cmax)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    val = (cmax + np.log(w.sum()) + mx) / alpha * (2.0 / n)
    grad = np.zeros(n)
    for p in range(n):
        m_lo = max(0, p - n + 1)
        m_hi = min(2 * n - 2, n - 1 + p)
        ms = np.arange(m_lo, m_hi + 1)
        j1 = p - ms + n - 1
        j2 = p + ms - n + 1
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        contrib = np.zeros(len(ms))
        if np.any(v1):
            contrib[v1] += (1.0 - h[j1[v1]])
        if np.any(v2):
            contrib[v2] -= h[j2[v2]]
        grad[p] = np.dot(w[ms], contrib)
    return val, grad * (2.0 / n)

def lbfgs_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def _f(x, n_=n, a_=alpha):
            return smooth_obj_and_grad(x, n_, a_)[0]
        def _g(x, n_=n, a_=alpha):
            return smooth_obj_and_grad(x, n_, a_)[1]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_f, best_h, jac=_g, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        if time.time() > deadline - 200:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = lbfgs_opt(h_int, n_try, [5000, 50000, 200000], 150, deadline)
        h_o, c_o = exact_cd(h_o, n_try, 200, deadline)
        if c_o < best_c - 1e-7:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 4: Shake and settle with L-BFGS-B + CD
    n = len(h_best)
    for r in range(10):
        if time.time() > deadline - 60:
            break
        s = 0.005 + 0.015 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = lbfgs_opt(hp, n, [10000, 50000, 200000], 150, deadline)
        h_o, c_o = exact_cd(h_o, n, 150, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Final polish with focused CD
    if time.time() < deadline - 10:
        h_f, c_f = focused_cd(h_best, n, 200, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
        h_f, c_f = exact_cd(h_best, n, 100, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
