            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-13:
            break
        active = vals >= (am - 1e-12)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-11:
            hi = mid
        elif max_sl < -1e-11:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=2000, deadline=None, seed=0, patience=30):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-12:
            best_c = new_c
            stall = 0
        else:
            stall += 1
            if improved == 0 and stall > patience:
                break
    return h, best_c

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    lse = mx + np.log(weights.sum())
    obj = (m + lse / alpha) * (2.0 / n)
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            idx = np.arange(n)
            g = (1.0 - 2.0 * x)
            gi = g.copy()
        else:
            ps = np.arange(n)
            ps_pos = ps + s
            valid_pos = (ps_pos >= 0) & (ps_pos < n)
            ps_neg = ps - s
            valid_neg = (ps_neg >= 0) & (ps_neg < n)
            g = np.zeros(n)
            g[valid_pos] += (1.0 - x[np.clip(ps_pos[valid_pos], 0, n-1)])
            g[valid_neg] -= x[np.clip(ps_neg[valid_neg], 0, n-1)]
        grad += weights[k] * * g
    grad *= (2.0 / n)
    return obj, grad

def fista_smooth(h, n, alpha, maxiter=500, deadline=None):
    x = h.copy()
    y = h.copy()
    t = 1.0
    L = 4.0 * n / (2.0 / n)
    step = 1.0 / (4.0 * n * n / (2.0))
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        c_y = np.correlate(y, 1.0 - y, mode='full')
        m_y = c_y.max()
        shifted = alpha * (c_y - m_y)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(c_y)):
            if w[k] < 1e-14:
                continue
            s = k - (n - 1)
            if s == 0:
                g = (1.0 - 2.0 * y)
            else:
                g = np.zeros(n)
                if s > 0:
                    g[:n-s] += (1.0 - y[s:])
                    g[s:] -= y[:n-s]
                else:
                    g[-s:] += (1.0 - y[:s])
                    g[:s] -= y[-s:]
            grad += w[k] * g
        grad *= (2.0 / n)
        step = 1.0 / (2.0 * alpha * n / (2.0 * n))
        step = 1.0 / (alpha * n)
        x_new = x - step * grad
        x_new = project_feasible(x_new, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + (t / t_new - 1.0) * (x_new - x)
        x = x_new
        t = t_new
        c_val = compute_c5(x, n)
        if c_val < best_c:
            best_c = c_val
            best_h = x.copy()
    return best_h, best_c

def gen_structured_perturb(h, n, rng, mode):
    if mode == 0:
        return project_feasible(h + rng.normal(0, 0.02, n), n)
    elif mode == 1:
        f = np.fft.rfft(h - 0.5)
        freqs = np.arange(len(f))
        mask = (freqs > 1) & (freqs < n // 3)
        f[mask] += rng.normal(0, 0.01, mask.sum())
        hr = np.fft.irfft(f, n=n) + 0.5
        return project_feasible(hr, n)
    elif mode == 2:
        shift = rng.randint(1, max(2, n // 10))
        return project_feasible(np.roll(h, shift), n)
    elif mode == 3:
        size = rng.randint(n // 20, n // 5)
        start = rng.randint(0, n - size)
        hp = h.copy()
        hp[start:start+size] += 0.1
        return project_feasible(hp, n)
    else:
        amp = 0.05
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        hp = h + amp * np.sin(2 * np.pi * freq * x + phase)
        return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    best_n = n

    # Phase 1: Intensive CD on initial
    if time.time() < deadline - 200:
        h1, c1 = coord_descent(best_h, n, 3000, deadline - 190, seed)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: FISTA smooth approximation for broader search
    if time.time() < deadline - 180:
        for alpha in [500, 2000, 10000]:
            if time.time() > deadline - 170:
                break
            hf, cf = fista_smooth(best_h, n, alpha, 300, deadline - 165)
            hc, cc = coord_descent(hf, n, 500, deadline - 155, seed + 100)
            if cc < best_c:
                best_h, best_c, best_n = hc, cc, n

    # Phase 3: Diverse restarts
    n_cur = len(best_h)
    x_orig = np.linspace(0, 2, n_cur)
    for r in range(20):
        if time.time() > deadline - 120:
            break
        mode = r % 5
        hp = gen_structured_perturb(best_h, n_cur, rng, mode)
        ho, co = coord_descent(hp, n_cur, 600, deadline - 110, seed + 300 + r * 13)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = np.linspace(0, 2, n_cur)

    # Phase 4: Try different resolutions
    n_trials = [500, 550, 580, 620, 650, 700, 750, 800, 850, 900, 1000]
    for n_try in n_trials:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 800, deadline - 75, seed + n_try)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_try
            x_orig = x_new

    # Phase 5: Binary-structured restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        for attempt in range(5):
            if time.time() > deadline - 55:
                break
            thresh = 0.3 + 0.1 * attempt
            hb = np.where(best_h > thresh, 1.0, 0.0)
            hb = project_feasible(hb, n_cur)
            if hb.sum() < 1 or hb.sum() > n_cur - 1:
                hb = project_feasible(np.clip(best_h + 0.3, 0, 1), n_cur)
            h_o, c_o = coord_descent(hb, n_cur, 500, deadline - 50, seed + 800 + attempt)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final extended CD
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        hf, cf = coord_descent(best_h, n_cur, 5000, deadline - 3, seed + 9999, patience=50)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Check 1-h complement
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
