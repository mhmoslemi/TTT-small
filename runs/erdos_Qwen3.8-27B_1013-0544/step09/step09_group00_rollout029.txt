import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-12 or h0 > 1.0 - 1e-12:
        return h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]

        c_diag = c_base[diag_idx[0]]
    unaff = 0.0
    if m_min > 0:

        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max

                elif x_lo < r2 and x_hi > r1:

        if feasible:
            L_hi = L
        else:

    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nm = a < -1e-14
    pm = a > 1e-14
    if np.any(nm):
        x_lo = max(x_lo, float(np.max(h0 + (L - c_base[nm]) / a[nm])))
    if np.any(pm):

def cd_update(h, c, p, n):
    h0 = h[p]
    xn = solve_1d(c, h, p, n)
    if abs(xn - h0) < 1e-12:
        return False
    delta = xn - h0
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    if len(diag_i) > 0:
        di = diag_i[0]
        c[ms[di]] += delta * (1.0 - 2.0 * h0) - delta * delta
        nz = np.ones(len(ms), dtype=bool)
        nz[di] = False
        c[ms[nz]] += delta * a[nz]
    else:
        c[ms] += delta * a
    h[p] = xn
    return True

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for pas in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if rng is not None and pas % 4 == 3:
            idx = rng.permutation(n)
        else:
            idx = range(n)
        for p in idx:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            if cd_update(h, c, p, n):
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)
