def coord_descent_fast(h, n, max_passes, deadline=None, seed=0):
    """Fast CD with incremental correlation update."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            # Compute A: change in correlation when h[p] changes by delta
            shift = np.arange(-(n-1), n)
            s1 = (p + shift >= 0) & (p + shift < n)
            s2 = (p - shift >= 0) & (p - shift < n)
            A = np.zeros(2*n - 1)
            A[s1] = 1.0 - h[p + shift[s1]]
            A[s2] -= h[p - shift[s2]]
            
            # Minimize max(c + delta * A) over delta in [-h0, 1-h0]
            lo, hi = -h0, 1.0 - h0
            for _ in range(45):
                mid = (lo + hi) * 0.5
                vals = c + mid * A
                am = vals.max()
                active = vals >= (am - 1e-13)
                if not active.any():
                    break
                mn = A[active].min()
                mx = A[active].max()
                if mn > 1e-12:
                    hi = mid
                elif mx < -1e-12:
                    lo = mid
                else:
                    break
            
            cands = np.array([lo, hi, (lo+hi)*0.5])
            best_d, best_v = 0.0, float('inf')
            for d in cands:
                d = float(np.clip(d, -h0, 1.0-h0))
                v = (c + d * A).max()
                if v < best_v:
                    best_v = v
                    best_d = d
            
            if abs(best_d) > 1e-12:
                h[p] += best_d
                h[p] = float(np.clip(h[p], 0.0, 1.0))
                actual_d = h[p] - h0
                if abs(actual_d) > 1e-14:
                    c += actual_d * A
                    improved = True
        
        if not improved:
            break
        
        new_max = c.max()
        if new_max < c_max:
            c_max = new_max
    
    return h, c_max * (2.0/n)

            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def cd_fast(h, n, max_passes, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    c_best = c.max()
    
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            shift = np.arange(-(n-1), n)
            s1 = (p + shift >= 0) & (p + shift < n)
            s2 = (p - shift >= 0) & (p - shift < n)
            A = np.zeros(2 * n - 1)
            A[s1] = 1.0 - h[p + shift[s1]]
            A[s2] -= h[p - shift[s2]]
            lo, hi = -h0, 1.0 - h0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = c + mid * A
                am = vals.max()
                active = vals >= (am - 1e-13)
                if active.any():
                    mn_sl = A[active].min()
                    mx_sl = A[active].max()
                    if mn_sl > 1e-12:
                        hi = mid
                    elif mx_sl < -1e-12:
                        lo = mid
                    else:
                        break
                else:
                    break
            cands = np.array([lo, hi, (lo + hi) * 0.5])
            best_d, best_v = 0.0, float('inf')
            for d in cands:
                d = float(np.clip(d, -h0, 1.0 - h0))
                v = float((c + d * A).max())
                if v < best_v:
                    best_v = v
                    best_d = d
            if abs(best_d) > 1e-12:
                new_val = float(np.clip(h0 + best_d, 0.0, 1.0))
                actual_d = new_val - h0
                if abs(actual_d) > 1e-14:
                    h[p] = new_val
                    c += actual_d * A
                    improved = True
        if not improved:
            break
        cm = c.max()
        if cm < c_best:
            c_best = cm
    
    return h, c_best * (2.0 / n)

def block_swap_perturb(h, n, rng, n_swaps=5):
    """Perturb h by swapping random contiguous blocks."""
    h = h.copy()
    for _ in range(n_swaps):
        # Pick two random positions and a random block length
        L = rng.randint(3, max(4, n // 10))
        i = rng.randint(0, n - L)
        j = rng.randint(0, n - L)
        if abs(i - j) < L:
            continue
        h[i:i+L], h[j:j+L] = h[j:j+L].copy(), h[i:i+L].copy()
    return h

def fourier_perturb(h, n, rng, n_modes=5, amp=0.1):
    """Add random low-frequency sinusoidal perturbations."""
    h = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(-amp, amp)
        h += a * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def cd_fast(h, n, max_passes, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    c_best = c.max()
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            sh = np.arange(-(n-1), n)
            m1 = (p + sh >= 0) & (p + sh < n)
            m2 = (p - sh >= 0) & (p - sh < n)
            A = np.zeros(2*n - 1)
            A[m1] = 1.0 - h[p + sh[m1]]
            A[m2] -= h[p - sh[m2]]
            lo, hi = -h0, 1.0 - h0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = c + mid * A
                am = vals.max()
                act = vals >= (am - 1e-13)
                if act.any():
                    if A[act].min() > 1e-12:
                        hi = mid
                    elif A[act].max() < -1e-12:
                        lo = mid
                    else:
                        break
                else:
                    break
            best_d, best_v = 0.0, float('inf')
            for d in (lo, hi, (lo+hi)*0.5):
                d = float(np.clip(d, -h0, 1.0-h0))
                v = float((c + d*A).max())
                if v < best_v:
                    best_v, best_d = v, d
            if abs(best_d) > 1e-12:
                nv = float(np.clip(h0 + best_d, 0.0, 1.0))
                ad = nv - h0
                if abs(ad) > 1e-14:
                    h[p] = nv
                    c += ad * A
                    improved = True
        if not improved:
            break
        if c.max() < c_best:
            c_best = c.max()
    return h, c_best * (2.0/n)

def soft_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp((c - m) / alpha)
    w = e / e.sum()
    # grad[i] = 1 - sum_k w[k]*h[i+k-(n-1)]*1(valid) - sum_k w[k]*h[i-k+(n-1)]*1(valid)
    conv1 = np.convolve(w[::-1], x, mode='full')
    conv2 = np.convolve(w, x, mode='full')
    term1 = conv1[n-1 : 2*n-1]
    term2 = conv2[n-1 : 2*n-1]
    grad = 1.0 - term1 - term2
    obj = (m + alpha * np.log(e.sum())) * (2.0/n)
    return obj, grad * (2.0/n)

def lbfgs_run(h, n, alphas, maxiter_each=200):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(soft_obj_grad, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14, 'gtol': 1e-8})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c
