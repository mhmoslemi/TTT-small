This is getting too complex. Let me just use a different strategy: compute the gradient using the subgradient of the top few k values.

**Simplified approach**: Instead of the full log-sum-exp gradient, use a "top-k" gradient:
- Compute all result[m] values
- Take the top K values (e.g., K=10 or K=50)
- Average their subgradients
- Use this as the effective gradient

This is much simpler to implement and should work well in practice.

For a given m*, the subgradient at h[p] is:
sg[p] = (1-h[p+m*-(n-1)]) * I[valid forward] - h[p-m*+(n-1)] * I[valid backward]

This can be computed in O(n) for each m*.

So the algorithm is:
1. Compute result = correlate(h, 1-h, mode='full')
2. Find top K m values
3. For each top m, compute subgradient (O(n))
4. Average to get effective gradient
5. Take a step: h_new = project(h - lr * grad)
6. Repeat

This is O(K*n + n²) per iteration (the n² from the correlation). With K=10, n=600, this is about 6000 + 360000 = 366000 operations per step. Very fast!

                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        free = np.where((h > threshold) & (h < 1.0 - threshold))[0]
        if len(free) == 0:
            break
    return h, compute_c5(h, n)

def try_resolutions(h_best, c_best, n_best, deadline, rng):
    x_best = np.linspace(0, 2, n_best)
    n_opts = [300, 400, 500, 600, 700, 800, 900, 1000]
    for n_try in n_opts:
        if time.time() > deadline - 100:
            break
        if n_try == n_best:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, h_best), n_try)
        h_opt, c_opt = smoothed_gd(h_int, n_try, 150, deadline, K=25, lr_start=0.03)
        h_opt, c_opt = coord_cd(h_opt, n_try, 150, deadline, rng, True)
        if c_opt < c_best:
            h_best, c_best = h_opt, c_opt
            n_best = n_try
            x_best = x_new
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n = 400

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Smoothed GD to escape local min
    if time.time() < deadline - 300:
        h1, c1 = smoothed_gd(best_h, n, 200, deadline, K=30, lr_start=0.03)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1b, c1b = coord_cd(best_h, n, 100, deadline, rng, True)
        if c1b < best_c:
            best_h, best_c = h1b, c1b

    # Phase 2: Full CD
    h2, c2 = coord_cd(best_h, n, 400, deadline, rng, True)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-resolution
    h3, c3, n3 = try_resolutions(best_h, best_c, len(best_h), deadline, rng)
    if c3 < best_c:
        best_h, best_c, n = h3, c3, n3

    # Phase 4: More smoothed GD with various K
    n_cur = len(best_h)
    if time.time() < deadline - 200:
        for K in [10, 50, 80]:
            if time.time() > deadline - 100:
                break
            h4, c4 = smoothed_gd(best_h, n_cur, 150, deadline, K=K, lr_start=0.02)
            if c4 < best_c:
                best_h, best_c = h4, c4
            h4b, c4b = coord_cd(best_h, n_cur, 100, deadline, rng, True)
            if c4b < best_c:
                best_h, best_c = h4b, c4b

    # Phase 5: Free-variable focused CD
    if time.time() < deadline - 100:
        for thresh in [0.1, 0.05, 0.02]:
            if time.time() > deadline - 50:
                break
            h5, c5 = free_var_cd(best_h, n_cur, 200, deadline, threshold=thresh)
            if c5 < best_c:
                best_h, best_c = h5, c5

    # Phase 6: Random perturbations
    n_cur = len(best_h)
    for r in range(25):
        if time.time() > deadline - 30:
            break
        s = 0.003 + 0.002 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        hr, cr = smoothed_gd(hp, n_cur, 80, deadline, K=20, lr_start=0.02)
        hr, cr = coord_cd(hr, n_cur, 60, deadline, rng, True)
        if cr < best_c:
            best_h, best_c = hr, cr

    # Phase 7: Binary snap + CD
    if time.time() < deadline - 60:
        hb = np.where(best_h > 0.5, 1.0, 0.0)
        hb = project_feasible(hb, len(best_h))
        hb, cb = coord_cd(hb, len(best_h), 200, deadline, rng, True)
        if cb < best_c:
            best_h, best_c = hb, cb

    # Phase 8: Final polish with long CD
    n_cur = len(best_h)
    h8, c8 = coord_cd(best_h, n_cur, 500, deadline, rng, True)
    if c8 < best_c:
        best_h, best_c = h8, c8

    # Phase 9: One more smoothed + CD cycle
    if time.time() < deadline - 20:
        h9, c9 = smoothed_gd(best_h, n_cur, 100, deadline, K=15, lr_start=0.01)
        if c9 < best_c:
            best_h, best_c = h9, c9
        h9b, c9b = coord_cd(best_h, n_cur, 200, deadline, rng, True)
        if c9b < best_c:
            best_h, best_c = h9b, c9b

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
