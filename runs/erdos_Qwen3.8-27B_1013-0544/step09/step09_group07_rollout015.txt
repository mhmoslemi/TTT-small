def obj_topk_smooth(x, n, k, alpha):
    """Smooth top-k objective: use soft weighting to approximate top-k average."""
    c = np.correlate(x, 1.0 - x, mode='full')
    # Find top-k threshold
    sorted_c = np.sort(c)[::-1]
    if k < len(sorted_c):
        threshold = sorted_c[k-1]
    else:
        threshold = sorted_c[-1]
    # Soft weighting: w_i = sigmoid(alpha * (c_i - threshold))

    w_sum = w.sum()
    if w_sum < 1e-300:
        return c.max() * (2.0 / n)
    weighted_mean = (w * c).sum() / w_sum
    return weighted_mean * (2.0 / n)
