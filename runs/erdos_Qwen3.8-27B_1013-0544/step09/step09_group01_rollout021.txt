import numpy as np
from scipy.optimize import minimize, line_search
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def p_norm_obj(h, n, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return dx * (np.sum(c**p))**(1.0/p)

def lse_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    if m < 0:
        m = 0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, obj_type, params, maxiter=500):
    """SLSQP with different objective types"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    
    if obj_type == 'lse':
        alphas = params
    else:  # p_norm
        ps = params
    
    if obj_type == 'lse':
        for alpha in alphas:
            try:
                res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-18})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    else:
        for p in ps:
            try:
                res = minimize(p_norm_obj, best_h, args=(n, p), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-18})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c

def subgradient_descent(h, n, max_iter, deadline, lr_init=0.01, momentum=0.9, top_k=5):
    """Subgradient descent with momentum, focusing on top-k correlation values"""
    h = h.copy().astype(float)
    dx = 2.0 / n
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(np.max(c) * dx)
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Find active set (top-k correlations)
        top_indices = np.argsort(c)[::-1][:top_k]
        # Weight by how close to max (softmax-like)
        weights = np.zeros(len(c))
        top_vals = c[top_indices]
        w_max = top_vals.max()
        if w_max > 0:
            weights[top_indices] = np.exp((top_vals - w_max) / (w_max * 0.1 + 1e-10))
            weights /= weights.sum()
        
        # Compute gradient of weighted sum of top correlations
        grad = np.zeros(n)
        for ki, idx in enumerate(top_indices):
            if weights[idx] < 1e-10:
                continue
            k = idx - (n - 1)  # the lag
            # h[p] contributes to c[idx] when:
            # 1) p is the "first" index: h[p] * (1-h[p+k]) where p+k is valid
            # 2) p is the "second" index: h[p-k] * (1-h[p]) where p-k is valid
            # Case 1: p + k in [0, n-1], i.e., -k <= p <= n-1-k
            p_lo1 = max(0, -k)
            p_hi1 = min(n-1, n-1-k)
            if p_hi1 >= p_lo1:
                ps = np.arange(p_lo1, p_hi1+1)
                grad[ps] += weights[idx] * (1.0 - h[ps + k])
            
            # Case 2: p - k in [0, n-1], i.e., k <= p <= n-1+k
            p_lo2 = max(0, k)
            p_hi2 = min(n-1, n-1+k)
            if p_hi2 >= p_lo2:
                ps = np.arange(p_lo2, p_hi2+1)
                grad[ps] += weights[idx] * (-h[ps - k])
        
        # Project gradient onto feasible direction (remove mean component)
        grad = grad - grad.mean()
        
        # Update with momentum
        v = momentum * v - lr * grad
        h = project_feasible(h + v, n)
        
        # Adaptive learning rate
        new_c = compute_c5(h, n)
        if new_c > cur_c * 1.01:
            lr *= 0.7
            v *= 0.5
            h = project_feasible(best_h, n)  # rollback
        elif it % 50 == 0 and new_c < cur_c:
            lr = min(lr * 1.05, 0.05)
    
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def anneal_restart(h, n, rng, deadline, temperature=0.05):
    """Simulated annealing step: randomly perturb and accept/reject"""
    h_new = h.copy()
    # Random swap of values
    idx = rng.choice(n, size=min(5, n), replace=False)
    for i in idx:
        h_new[i] += rng.uniform(-temperature, temperature)
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    return h_new, c_new

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def p_norm_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    return dx * (np.sum(c**p))**(1.0/p)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, obj_type, params, maxiter=500):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for param in params:
        if obj_type == 'p':
            func = p_norm_obj
        else:
            func = lse_obj
        try:

                           bounds=bounds, constraints=cons,

    return best_h, best_c

def subgrad_step(h, n, top_k, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    w_max = top_vals.max()
    if w_max <= 0:
        return np.zeros(n)
    weights = np.exp((top_vals - w_max) / (temp + 1e-10))
    weights /= weights.sum()
    grad = np.zeros(n)
    for ki, idx in enumerate(top_idx):
        if weights[ki] < 1e-12:
            continue
        k = idx - (n - 1)
        p_lo1 = max(0, -k)
        p_hi1 = min(n - 1, n - 1 - k)
        if p_hi1 >= p_lo1:
            ps = np.arange(p_lo1, p_hi1 + 1)
            grad[ps] += weights[ki] * (1.0 - h[ps + k])
        p_lo2 = max(0, k)
        p_hi2 = min(n - 1, n - 1 + k)
        if p_hi2 >= p_lo2:
            ps = np.arange(p_lo2, p_hi2 + 1)
            grad[ps] += weights[ki] * (-h[ps - k])
    return grad - grad.mean()

        remaining = deadline - time.time() - 100
        n_restarts = max(2, int(remaining / 100))
        for r in range(n_restarts):
            if time.time() > deadline - 100:
                break
            if r % 4 == 0:
                s = 0.015 + 0.02 * r / max(1, n_restarts)
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            elif r % 4 == 1:
                x = np.linspace(0, 2, n)
                freq = 2.0 + 3.0 * (r / max(1, n_restarts))
                hp = project_feasible(best_h + 0.035 * np.sin(np.pi * freq * x), n)
            elif r % 4 == 2:
                # Binary restart + refinement
                hp = make_binary_init(n, seed + 100 + r)
                h_b, c_b = slsqp_optimize(hp, n, 'p', [4, 16, 64], 250)
                h_b, c_b = slsqp_optimize(h_b, n, 'lse', [2000, 20000], 250)
                h_b, c_b = coord_descent(h_b, n, 40, deadline - 60, gs_iters=50)
                if c_b < best_c:
                    best_h, best_c = h_b, c_b
                continue
            else:
                # Block frequency perturbation
                x = np.linspace(0, 2, n)
                freq = 4.0 + 2.0 * ((r // 4) % 3)
                hp = project_feasible(best_h + 0.025 * np.cos(2 * np.pi * freq * x), n)
            h_o, c_o = slsqp_optimize(hp, n, 'p', [16, 64], 250)
            h_o, c_o = slsqp_optimize(h_o, n, 'lse', [5000, 50000], 250)
            h_o, c_o = coord_descent(h_o, n, 40, deadline - 60, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 7: Final high-precision polish
    if time.time() < deadline - 15:
        h_f, c_f = coord_descent(best_h, n, 100, deadline - 8, gs_iters=70)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 5:
        h_f, c_f = slsqp_optimize(best_h, n, 'lse', [100000, 500000, 2000000], 800)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
