def subgrad_c(h, n):

    Returns the subgradient vector and the current c_k values."""
    corr = compute_corr(h, n)
    dx = 2.0 / n
    target = corr.max()

    active = corr >= target - 1e-10

    weights /= weights.sum()

def subgrad_c_vec(h, n):
    """Vectorized subgradient of max_k c_k."""
    corr = compute_corr(h, n)
    dx = 2.0 / n
    target = corr.max()
    active = corr >= target - 1e-8
    if not active.any():
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    weights = active.astype(float)
    weights /= weights.sum()
    
    # For each k in active set, dc_k/dh[p] = (1-h[p+k-n+1])_valid - h[p-k+n-1]_valid
    # Let's define offset = k - n + 1, ranges from -(n-1) to n-1
    offsets = np.arange(len(corr)) - n + 1  # shape (2n-1,)
    
    sg = np.zeros(n)
    for k_idx in np.where(active)[0]:
        w = weights[k_idx] * dx
        off = offsets[k_idx]
        # First term: (1-h[p+off]) for p where 0 <= p+off < n
        p_start = max(0, -off)
        p_end = min(n, n - off)
        if p_start < p_end:
            sg[p_start:p_end] += w * (1.0 - h[p_start + off:p_end + off])
        # Second term: -h[p-off] for p where 0 <= p-off < n
        p_start2 = max(0, off)
        p_end2 = min(n, n + off)
        if p_start2 < p_end2:
            sg[p_start2:p_end2] -= w * h[p_start2 - off:p_end2 - off]
    return sg, float(target * dx)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):

            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]

            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0

                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:

        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def smooth_obj(x, n, alpha):
    c = compute_corr(x, n) * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return m + lse / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val, best_d = v, d
    return best_d, best_val

def cd_pass(h, n, top_k, order):
    corr = compute_corr(h, n)
    current_max = float(corr.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(corr)[::-1][:k]
    c_peak = corr[top_idx].astype(float)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[top_idx] = False
    unaff_max = float(corr[mask].max()) if mask.any() else 0.0
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        a_coeff = np.zeros(k)
        diag_idx = None
        for idx in range(k):
            m = int(top_idx[idx])
            jf = p + m - (n - 1)
            jb = p - m + (n - 1)
            af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            ab = (-h[jb]) if 0 <= jb < n else 0.0
            a_coeff[idx] = af + ab
            if m == n - 1:

        lo, hi = -h0, 1.0 - h0

    return h, improved, current_max

def multi_cd(h, n, max_passes, deadline, top_k, rng, recompute_freq=3):
    h = h.copy()
    no_improve = 0
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if p % recompute_freq == 0:
            order = rng.permutation(n)
        else:
            order = rng.permutation(n)

        if improved:
            no_improve = 0
            h = project_feasible(h, n)
        else:
            no_improve += 1
            if no_improve >= 5:
                break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = compute_corr(x, n) * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return m + lse / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def hinge_cd(h, n, t_hinge, deadline, rng, top_k=30):

    h = h.copy()
    for p in range(2000):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        mx = corr.max()
        active = corr >= mx - t_hinge
        k = min(top_k, int(active.sum()), 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:k]
        order = rng.permutation(n)
        c_peak = corr[top_idx].astype(float)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(corr[mask].max()) if mask.any() else 0.0
        improved = False
        for pidx in order:
            h0 = h[pidx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = pidx + m - (n - 1)
                jb = pidx - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:

            lo, hi = -h0, 1.0 - h0
            bd, bv = solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if bv < corr.max() - 1e-14:

                if improved:
                    corr = compute_corr(h, n)
                    mx = corr.max()
                    top_idx = np.argsort(corr)[::-1][:k]
                    c_peak = corr[top_idx].astype(float)
                    mask = np.ones(2 * n - 1, dtype=bool)
                    mask[top_idx] = False
                    unaff_max = float(corr[mask].max()) if mask.any() else 0.0
