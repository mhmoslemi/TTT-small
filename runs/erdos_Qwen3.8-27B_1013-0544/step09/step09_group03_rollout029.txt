def exact_coord_min(c, p, h, n):
    """Find optimal value for coordinate p given current correlation c."""
    h0 = h[p]

    s_max = (n - 1) - p

def dual_optimize(h, n, n_iter=100):
    """Iterative dual approach."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Initialize w proportional to softmax of c
    from scipy.special import softmax
    temp = 10.0
    w = softmax(c * temp)
    
    for it in range(n_iter):
        # Build Q_w matrix (sparse)
        # Q_w = sum_k w[k] * S_{s_k}
        # S_s is the shift matrix: (S_s)_{i,j} = 1 if j = i+s
        # h^T Q_w h = sum_k w[k] * sum_i h[i]*h[i+s_k]
        
        # Gradient of h^T Q_w h with respect to h: 2 * Q_w * h
        # (Q_w * h)[j] = sum_k w[k] * h[j+s_k] (with boundary handling)
        # This is a weighted sum of shifted h vectors
        
        Qh = np.zeros(n)
        for m in range(2*n-1):
            if w[m] < 1e-10:
                continue
            s = m - (n-1)
            # (S_s * h)[j] = h[j+s] if j+s valid
            if s >= 0:
                Qh[:n-s] += w[m] * h[s:]
            else:
                Qh[-s:] += w[m] * h[:n+s]
        
        # Gradient ascent step
        grad = 2.0 * Qh
        # Project: h_new = project(h + lr * grad)
        lr = 0.01
        h_new = project_feasible(h + lr * grad, n)
        
        # Update w
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        w = softmax(c_new * temp)
        h = h_new
    
    return h

        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_descent(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10:
                h[p] = nv
                improved += 1
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
        elif improved == 0:
            break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x.copy(), n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def momentum_subgrad(h, n, lr, beta=0.85, n_top=5, n_steps=200, deadline=None):
    v = np.zeros(n)
    h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for m in top_idx:
            s = m - (n - 1)
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            for i in range(i_lo, i_hi):
                g[i] += (1.0 - h[i + s])
                g[i + s] -= h[i]
        g *= (2.0 / n) / n_top
        v = beta * v + g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            lr *= 0.7
    return h, best_c

def pairwise_refine(h, n, n_pairs, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]
        if abs(hi - hj) < 1e-8:
            continue
        d = hj - hi
        if d > 0:
            lo_a = max(0.0, -hi / d, (hj - 1.0) / d)
            hi_a = min(1.0, (1.0 - hi) / d, hj / d)
        else:
            dn = -d
            lo_a = max(0.0, (hi - 1.0) / dn, -hj / dn)
            hi_a = min(1.0, hi / dn, (1.0 - hj) / dn)
        if lo_a >= hi_a:
            continue
        alphas = np.linspace(lo_a, hi_a, 7)
        cur_c = compute_c5(h, n)
        for alpha in alphas:
            h[i] = hi + alpha * (hj - hi)
            h[j] = hj + alpha * (hi - hj)
            c_try = compute_c5(h, n)
            if c_try < best_c:
                best_c = c_try
            else:
                h[i], h[j] = hi, hj
                break
        else:
            h[i] = hi + alphas[-1] * (hj - hi)
            h[j] = hj + alphas[-1] * (hi - hj)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n0 = 600

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Coarse-to-fine (build global structure)
    n_coarse = max(100, n0 // 6)
    if time.time() < deadline - 400:
        x_fine = np.linspace(0, 2, n0)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_coarse = project_feasible(np.interp(x_coarse, x_fine, best_h), n_coarse)
        h_coarse, c_coarse = coord_descent(h_coarse, n_coarse, 300, deadline - 380, seed + 1)
        h_up = project_feasible(np.interp(x_fine, x_coarse, h_coarse), n0)
        c_up = compute_c5(h_up, n0)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # Phase 2: Intensive coordinate descent
    if time.time() < deadline - 250:
        h2, c2 = coord_descent(best_h, n0, 800, deadline - 230, seed + 10)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP with high alpha
    if time.time() < deadline - 180:
        for alpha in [20000, 200000]:
            if time.time() > deadline - 160:
                break
            h3, c3 = slsqp_opt(best_h, n0, alpha, 200)
            if c3 < best_c:
                best_h, best_c = h3, c3
            h3b, c3b = coord_descent(best_h, n0, 300, deadline - 150, seed + alpha)
            if c3b < best_c:
                best_h, best_c = h3b, c3b

    # Phase 4: Multi-n exploration
    x_cur = np.linspace(0, 2, len(best_h))
    n_trials = [200, 300, 400, 500, 700, 800, 1000, 1200]
    for n_try in n_trials:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 500, deadline - 50, seed + n_try)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 5: Momentum subgradient
    if time.time() < deadline - 40:
        h5, c5 = momentum_subgrad(best_h, len(best_h), lr=0.02, beta=0.8,
                                  n_top=5, n_steps=300, deadline=deadline - 30)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Pairwise refinement
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h6, c6 = pairwise_refine(best_h, n_cur, 500, deadline - 10, seed + 777)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final intensive CD
    n_cur = len(best_h)
    hf, cf = coord_descent(best_h, n_cur, 3000, deadline - 5, seed + 999)
    if cf < best_c:
        best_h, best_c = hf, cf

    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
