from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def ps_obj(x, n, p):
    """Power-sum objective: dx * (sum c^p)^(1/p)"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    # Use log to avoid overflow
    log_c = np.log(np.maximum(c, 1e-30))
    m = log_c.max()
    val = m + np.log(np.sum(np.exp(log_c - m))) / p
    return np.exp(val) * dx

def lse_obj(x, n, alpha):
    """Log-sum-exp objective"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, objective='lse', maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            obj = lse_obj if objective == 'lse' else (lambda x, n, a: ps_obj(x, n, a))
            res = minimize(obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

    gr = (np.sqrt(5.0) - 1.0) / 2.0

            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) - d * d
    return max(float(np.max(vals)), unaff_max)

def coord_descent(h, n, max_passes, deadline, gs_iters=60, rng=None, random_order=False):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    order = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        curr_max = float(c.max())
        if random_order and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            if f_opt < curr_max - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, lr):
    """One step of projected subgradient on the max of correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    curr_max = float(c.max())
    # Find all lags within 1e-6 of the max (active set)
    active = np.where(c >= curr_max - 1e-6)[0]
    # Compute subgradient: sum of gradients over active lags
    sg = np.zeros(n)
    for lag_idx in active:
        k = lag_idx - (n - 1)  # actual lag
        if k == 0:
            sg += (1.0 - 2.0 * h)
        elif k > 0:
            # c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[p]: if p <= n-1-k, contribution is (1-h[p+k])
            #          if k <= p <= n-1, contribution is -h[p-k]
            for p in range(n):
                if p <= n - 1 - k:
                    sg[p] += (1.0 - h[p + k])
                if k <= p < n:
                    sg[p] += (-h[p - k])
        else:
            k_abs = -k
            for p in range(n):
                if p >= k_abs:
                    sg[p] += (1.0 - h[p - k_abs])
                if p < n - k_abs:
                    sg[p] += (-h[p + k_abs])
    # Normalize
    sg_norm = np.linalg.norm(sg)
    if sg_norm < 1e-15:
        return h
    step = h - lr * sg / sg_norm
    return project_feasible(step, n)

def project_feasible(h, n):
    target = n / 2.0
