            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

    s = alpha * (c - m)

    lse = mx + np.log(np.sum(np.exp(s - mx)))

            break

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

                continue

            if abs(da) > 1e-15:

                    candidates = np.append(candidates, d)
    if diag_idx is not None:

            if i == diag_idx:

            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C

                d1 = (-B + sq) / 2.0

                if lo <= d1 <= hi:

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=400):

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,

        return new_h, compute_c5(new_h, n)

            if time.time() > deadline:
                break
            init = x0.copy() if trial == 0 else np.clip(rng.randn(np_params) * 0.3, -0.8, 0.8)
            bounds_list = [(-0.8, 0.8)] * np_params
            try:
                de_res = differential_evolution(fourier_obj, bounds_list, args=(n, K),
                                                seed=seed + trial, maxiter=60, tol=1e-8,
                                                popsize=20, polish=True, init='sobol',
                                                update_interval=1, workers=1)
                if de_res.success or de_res.fun < best_c:
                    h_new = project_feasible(fourier_to_h(de_res.x, n, K), n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Initial CD from parent
    for tk in [15, 25, 40]:
        remaining = deadline - time.time()
        if remaining < 80:
            break
        passes = min(200, int(remaining * 1.5))
        h2, c2 = multi_cd(best_h, n, passes, deadline - 80, tk)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 2: Fourier global search for diverse starting points
    if time.time() < deadline - 350:
        fh, fc = fourier_global_search(n, seed, deadline - 350)
        if fh is not None and fc < best_c:
            best_c, best_h = fc, fh
        elif fh is not None:
            # Use Fourier result as perturbation base
            blend = project_feasible(0.5 * best_h + 0.5 * fh, n)
            hb, cb = multi_cd(blend, n, 100, deadline - 350, 25)
            if cb < best_c:
                best_c, best_h = cb, hb

    # Phase 3: SLSQP + CD alternating
    if time.time() < deadline - 250:
        for alpha in [2000, 10000, 40000, 100000, 300000]:
            if time.time() > deadline - 250:
                break
            h_s, c_s = slsqp_run(best_h, n, alpha, 350)
            if c_s < best_c:
                best_c, best_h = c_s, h_s
            h_f, c_f = try_symmetry(best_h, n)
            if c_f < best_c:
                best_c, best_h = c_f, h_f
            h2, c2 = multi_cd(best_h, n, 120, deadline - 250, 35)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 4: Deep CD
    if time.time() < deadline - 120:
        h3, c3 = multi_cd(best_h, n, 800, deadline - 120, 50)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 5: Multi-start with adaptive perturbation
    n_cur = len(best_h)
    if time.time() < deadline - 100:
        scales = [0.001, 0.002, 0.004, 0.006, 0.01, 0.015, 0.025, 0.04, 0.06]
        for s in scales:
            if time.time() > deadline - 100:
                break
            for trial in range(3):
                if time.time() > deadline - 100:
                    break
                noise = rng.normal(0, s, n_cur)
                hp = project_feasible(best_h + noise, n_cur)
                hc, cc = multi_cd(hp, n_cur, 60, deadline - 100, 20)
                hc, cc = try_symmetry(hc, n_cur)
                if cc < best_c:
                    best_c, best_h = cc, hc

    # Phase 6: Try different discretization sizes
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 650, 750, 900]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 150, deadline - 80, 25)
        h_o, c_o = try_symmetry(h_o, n_try)
        h_o, c_o = slsqp_run(h_o, n_try, 50000, 200)
        h_o, c_o = multi_cd(h_o, n_try, 80, deadline - 80, 30)
        h_o, c_o = try_symmetry(h_o, n_try)
        if c_o < best_c:
            best_c, best_h = c_o, h_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 7: Final polish
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        h_f, c_f = slsqp_run(best_h, n_cur, 500000, 400)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 2:
        h_f, c_f = multi_cd(best_h, n_cur, 1500, deadline, 55)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
