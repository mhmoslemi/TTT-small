    val = float(nv.max())

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):

    if k == 0:
        return 0.0
    cands = [0.0]

            dd = 0.0

            if diag_local == j: dd += 1.0

                    if lo < d < hi: cands.append(d)
            else:

                if disc >= 0:

                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: cands.append(d)

        for i in range(n):
            j = i + ms - (n-1)
            if 0 <= j < n: sg[i] += (1.0-h[j])
            ip = i - ms + (n-1)
            if 0 <= ip < n: sg[i] -= h[ip]
        d = -sg
        hi = 0.08
        for i in range(n):
            if d[i] > 0: hi = min(hi, (1-h[i])/d[i])
            elif d[i] < 0: hi = min(hi, -h[i]/d[i])
        if hi < 1e-12: break
        def ft(t):
            ht = np.clip(h + t*d, 0, 1)
            return compute_c5(ht, n)
        try:
            r = golden(ft, 0.0, hi, maxiter=25)
            hn = project_feasible(np.clip(h+r.x*d, 0, 1), n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-13:
                h, best_c = hn, cn
            else: break
        except: break
    return h, best_c

def slsqp_opt(h, n, alphas, maxiter_each=400, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    def sm_obj(x, nn, a):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        sh = a*(c-m)
        mx = sh.max()
        lse = mx + np.log(np.sum(np.exp(sh-mx)))
        return (m + lse/a)*(2.0/nn)
    def sc(x, nn): return x.sum()-nn/2.0
    for a in alphas:
        if deadline is not None and time.time() > deadline: break
        try:
            r = minimize(sm_obj, best_h, args=(n, a), method='SLSQP',
                         bounds=[(0,1)]*n,
                         constraints=[{'type':'eq','fun':sc,'args':(n,)}],
                         options={'maxiter':maxiter_each,'ftol':1e-16})
            nh = project_feasible(r.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

def anneal_step(h, n, rng, T, max_iters=200, deadline=None):
    h = h.copy()
    bc = compute_c5(h, n)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline: break
        k = rng.randint(2, 6)
        idxs = rng.choice(n, k, replace=False)
        delta = rng.uniform(-0.02, 0.02, k)
        h_new = h.copy()
        h_new[idxs] = np.clip(h_new[idxs]+delta, 0, 1)
        h_new = project_feasible(h_new, n)
        cn = compute_c5(h_new, n)
        if cn < bc or rng.rand() < np.exp(-(cn-bc)/max(T, 1e-10)):
            h, bc = h_new, cn
        T *= 0.97
    return h, bc

def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except:
        h_init = np.full(400, 0.5); n = 400
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent (random order, multiple passes)
    dl1 = min(deadline, t0 + 200)
    h1, c1 = coord_cd(best_h, n, 200, dl1, 30, rng)
    if c1 < best_c: best_h, best_c = h1, c1

    # Phase 2: Subgradient line search
    if time.time() < deadline - 300:
        h2, c2 = subgrad_ls(best_h, n, 300, deadline-280)
        if c2 < best_c: best_h, best_c = h2, c2

    # Phase 3: SLSQP
    if time.time() < deadline - 250:
        h3, c3 = slsqp_opt(best_h, n, [200, 1000, 5000, 20000], 300, deadline-230)
        if c3 < best_c: best_h, best_c = h3, c3

    # Phase 4: More coordinate descent
    if time.time() < deadline - 200:
        h4, c4 = coord_cd(best_h, n, 300, deadline-180, 35, rng)
        if c4 < best_c: best_h, best_c = h4, c4

    # Phase 5: Simulated annealing on the active region
    if time.time() < deadline - 150:
        h5, c5 = anneal_step(best_h, n, rng, 0.005, 300, deadline-130)
        if c5 < best_c: best_h, best_c = h5, c5
        if time.time() < deadline - 120:
            h5b, c5b = coord_cd(best_h, n, 200, deadline-100, 35, rng)
            if c5b < best_c: best_h, best_c = h5b, c5b

    # Phase 6: Try nearby n values
    n_base = len(best_h)
    x_orig = np.linspace(0, 2, n_base)
    for n_try in [n_base-5, n_base+5, n_base-10, n_base+10, n_base-20, n_base+20]:
        if n_try < 50 or time.time() > deadline - 80: break
        h_int = project_feasible(np.interp(np.linspace(0,2,n_try), x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, 150, deadline-60, 25, rng)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = np.linspace(0, 2, n_try)

    # Phase 7: Binary-based local search
    if time.time() < deadline - 60:
        nb = len(best_h)
        hb = make_binary_start(nb, rng)
        # Initialize binary from current best: round
        hb = (best_h > 0.5).astype(float)
        # Adjust count
        cnt = int(hb.sum())
        if cnt > nb//2:
            idxs1 = np.where(hb==1)[0]
            to_zero = rng.choice(idxs1, cnt-nb//2, replace=False)
            hb[to_zero] = 0
        elif cnt < nb//2:
            idxs0 = np.where(hb==0)[0]
            to_one = rng.choice(idxs0, nb//2-cnt, replace=False)
            hb[to_one] = 1
        hb2, cb2 = coord_cd(hb, nb, 200, deadline-40, 25, rng)
        if cb2 < best_c: best_h, best_c = hb2, cb2

    # Phase 8: Final SLSQP + coord
    n_fin = len(best_h)
    if time.time() < deadline - 40:
        h8, c8 = slsqp_opt(best_h, n_fin, [10000, 50000, 100000], 500, deadline-20)
        if c8 < best_c: best_h, best_c = h8, c8
    if time.time() < deadline - 10:
        h9, c9 = coord_cd(best_h, n_fin, 500, deadline-5, 40, rng)
        if c9 < best_c: best_h, best_c = h9, c9

    # Phase 9: 1-h symmetry check
    hf = project_feasible(1.0-best_h, n_fin)
    cf = compute_c5(hf, n_fin)
    if cf < best_c: best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
