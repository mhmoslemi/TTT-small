```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:

def gradient_cd(h, n, max_passes=5000, deadline=None, seed=42):

    h = h.copy()
    rng = np.random.RandomState(seed)
    eta = 0.1

        if deadline is not None and time.time() > deadline:
            break

        # Compute subgradient vector

        else:
```

```python
    # So: ac_full[m] = ifft(|fft(x)|²)[m] for m = 0, ..., n-1 (positive lags, including 0)
    #     ac_full[-m] = ifft(|fft(x)|²)[N-m] for m = 1, ..., n-1 (negative lags)
    
    # The "full" correlation in numpy is:
    # np.correlate(h, h, 'full') = [ac_full[-(n-1)], ..., ac_full[0], ..., ac_full[n-1]]
    # which has 2n-1 elements
    
    N = 2 * n
    H = np.fft.rfft(h, N)
    circ = np.fft.irfft(H * np.conj(H), N)
    full_ac = np.zeros(2 * n - 1)
    full_ac[n-1:] = circ[0:n]  # lags 0 to n-1
    full_ac[:n-1] = circ[N-1:0:-1]  # lags -(n-1) to -1
    return full_ac
```
