def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0

        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:

            break

                    delta[vb] -= d * h[jb[vb]]

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):

    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))

    # Phase 4: Multi-resolution
    x_cur = np.linspace(0, 2, len(best_h))
    for n_try in [200, 300, 400, 500, 700, 800, 900]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=60, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 5: Final CD polish
    n_final = len(best_h)
    h_f, c_f = coord_cd(best_h, n_final, max_passes=300, deadline=deadline, shuffle=True, rng=rng)
    if c_f < best_c:
        best_h, best_c = h_f, c_f

    # Phase 6: Tight restarts
    for r in range(12):
        if time.time() > deadline - 15:
            break
        sigma = 0.003 + 0.002 * r
        hp = project_feasible(best_h + rng.normal(0, sigma, n_final), n_final)
        h_r, c_r = coord_cd(hp, n_final, max_passes=40, deadline=deadline, shuffle=True, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 7: One more momentum pass on best
    if time.time() < deadline - 20:
        h_m, c_m = momentum_smooth(best_h, n_final, 500, 2.0, 0.95, 100, deadline)
        if c_m < best_c:
            best_h, best_c = h_m, c_m
        h_f2, c_f2 = coord_cd(best_h, n_final, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if c_f2 < best_c:
            best_h, best_c = h_f2, c_f2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
