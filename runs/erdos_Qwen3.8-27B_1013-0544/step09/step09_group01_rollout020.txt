                freq = 7.0 + 4.0 * r / 12.0
                hp = project_feasible(best_h + 0.015 * np.cos(2 * np.pi * freq * x), n_cur)
            else:
                mask = rng.rand(n_cur) > 0.7
                amp = rng.normal(0, 0.04, n_cur) * mask
                hp = project_feasible(best_h + amp, n_cur)
            h_new = slsqp_opt(hp, n_cur, 50000, maxiter=250)
            h_new, c_new = coord_descent(h_new, n_cur, 50, deadline, gs_iters=70)
            if c_new < best_c:
                best_h, best_c = h_new, c2

    # Phase 5: Grid size exploration
    if time.time() < deadline - 150:
        n_cur = len(best_h)
        for n_try in [n_cur - 40, n_cur + 40, 640, 760]:
            if time.time() > deadline - 120:
                break
            if n_try <= 0 or abs(n_try - n_cur) < 10:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h_o = slsqp_opt(h_int, n_try, 50000, maxiter=300)
            h_o, c_o = coord_descent(h_o, n_try, 60, deadline, gs_iters=80)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final high-precision CD polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 150, deadline, gs_iters=120)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final SLSQP micro-polish
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        h_f = slsqp_opt(best_h, n_cur, 5000000, maxiter=300)
        c_f = compute_c5(h_f, n_cur)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
