def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def golden_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi, iters=60):
    """Golden section search for 1D subproblem."""
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_1d(c, c_peak, a_coeff, diag_idx, unaff_max)
    fd = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_1d(c, c_peak, a_coeff, diag_idx, unaff_max)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
    mid = (a + b) / 2
    return mid, eval_1d(mid, c_peak, a_coeff, diag_idx, unaff_max)

    # First try candidate points (exact for piecewise linear parts)

                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da

                    candidates.append(d)

            if i == diag_idx:

            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo < d1 < hi:

                if lo < d2 < hi:

        b = coefs[2 * (k - 1) + 1]

            break
        s = 0.003 + 0.008 * r
        hp = project_feasible(best_h + rng.normal(0, s, nc), nc)
        hc, cc = multi_cd(hp, nc, 120, deadline - 120, 25)
        hc, cc = try_flip(hc, nc)
        if cc < best_c:
            best_h, best_c = hc, cc
    # Phase 5: Grid size exploration with CD
    xo = np.linspace(0.0, 2.0, len(best_h))
    for ntry in [400, 500, 550, 650, 750, 800]:
        if time.time() > deadline - 80:
            break
        if ntry == len(best_h):
            continue
        xn = np.linspace(0.0, 2.0, ntry)
        hi = project_feasible(np.interp(xn, xo, best_h), ntry)
        ho, co = multi_cd(hi, ntry, 150, deadline - 80, 25)
        ho, co = try_flip(ho, ntry)
        if co < best_c:
            best_h, best_c = ho, co
            xo = xn
    # Phase 6: Final deep CD
    if time.time() < deadline - 10:
        hf, cf = multi_cd(best_h, len(best_h), 2000, deadline - 10, 70)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h, best_c = try_flip(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
