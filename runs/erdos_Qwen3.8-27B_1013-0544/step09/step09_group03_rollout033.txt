import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0

            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    lse_val = m + np.log(sum_exp) / alpha
    p = exp_shifted / sum_exp
    
    # Compute gradient
    # c[m] = sum_i x[i] * (1-x[i + (m-(n-1))])
    # Let l = m - (n-1), so c[l+n-1] = sum_i x[i] * (1-x[i+l])
    # d/dx[j] of c[l+n-1] = (1-x[j+l])*I(0<=j+l<n) - x[j-l]*I(0<=j-l<n)
    
    grad = np.zeros(n)
    # For efficiency, use the structure
    # grad[j] = sum_l p[l+n-1] * [(1-x[j+l])*I(0<=j+l<n) - x[j-l]*I(0<=j-l<n)]
    
    # Let me compute this using the "sliding window" approach
    # For each lag l, the weight is p[l+n-1]
    # Contribution to grad[j] from lag l:
    #   If 0 <= j+l < n: add p[l+n-1] * (1-x[j+l])
    #   If 0 <= j-l < n: subtract p[l+n-1] * x[j-l]
    
    # Vectorized approach using convolution/correlation
    # First term: sum_l p[l+n-1] * (1-x[j+l]) * I(0<=j+l<n)
    #   = sum_l p[l+n-1] * I(0<=j+l<n) - sum_l p[l+n-1]*x[j+l]*I(0<=j+l<n)
    
    # For the second sum (the "correlation" part), I'll use a loop over lags
    # but vectorize over j. This is O(n^2) but with fast vectorized ops.
    
    # Alternative: use FFT for the correlation
    # sum_l w[l] * x[j+l] where w[l] = p[l+n-1] and valid range
    # This is the cross-correlation of w (padded) and x
    
    # Let me just do the loop - for n<=1000 it's fast enough
    for l in range(-(n-1), n):
        w = p[l + n - 1]
        if w < 1e-15:
            continue
        # Positions where j+l is valid: j in [max(0,-l), min(n-1, n-1-l)]
        j1 = max(0, -l)
        j2 = min(n-1, n-1-l)
        if j1 <= j2:
            grad[j1:j2+1] += w  # from the (1-x[j+l]) part: the 1
            grad[j1:j2+1] -= w * x[j1+l:j2+1+l]  # from the (1-x[j+l]) part: the -x[j+l]
        # Positions where j-l is valid: j in [max(0,l), min(n-1, n-1+l)]
        j3 = max(0, l)
        j4 = min(n-1, n-1+l)
        if j3 <= j4:
            grad[j3:j4+1] -= w * x[j3-l:j4+1-l]
    
    dx_scale = 2.0 / n
    return lse_val * dx_scale, grad * dx_scale

        best_imp = 0
        best_pair = None
        # Try random swaps

        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            # Swap h[i] and h[j]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c_new = compute_c5(h_bin, n)
            imp = best_c - c_new
            if imp > best_imp:
                best_imp = imp
                best_pair = (i, j)
                best_c_store = c_new
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        
        if best_pair is not None:
            i, j = best_pair
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            best_c = best_c_store
        else:
            break
    
    return h_bin, best_c

    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    lse_val = m + np.log(sum_exp) / alpha
    p = exp_s / sum_exp
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        w = p[l + n - 1]
        if w < 1e-15:
            continue
        j1 = max(0, -l)
        j2 = min(n - 1, n - 1 - l)
        if j1 <= j2:
            grad[j1:j2 + 1] += w
            grad[j1:j2 + 1] -= w * x[j1 + l:j2 + 1 + l]
        j3 = max(0, l)
        j4 = min(n - 1, n - 1 + l)
        if j3 <= j4:
            grad[j3:j4 + 1] -= w * x[j3 - l:j4 + 1 - l]
    dx_s = 2.0 / n
    return lse_val * dx_s, grad * dx_s

def lbfgs_refine(h, n, alphas, deadline, maxiter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq',
                 'fun': lambda x, n=n: x.sum() - n / 2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(smooth_obj_and_grad, best_h, args=(n, alpha),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': maxiter, 'gtol': 1e-9,
                                    'ftol': 1e-14, 'maxls': 40})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def block_perturb_cd(h, n, deadline, seed, n_blocks=20, block_sz=8, noise=0.15):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        start = rng.randint(0, max(1, n - block_sz))
        end = min(n, start + block_sz)
        hp = best_h.copy()
        hp[start:end] = np.clip(hp[start:end] + rng.normal(0, noise, end - start), 0, 1)
        hp = project_feasible(hp, n)
        # Local CD on the block: fix outside, optimize inside
        for it in range(80):
            if time.time() > deadline:
                break
            c = np.correlate(hp, 1.0 - hp, mode='full')
            for p in range(start, end):
                if p < 1 or p >= n - 1:
                    continue
                hv = hp[p]
                if hv < 1e-10 or hv > 1.0 - 1e-10:
                    continue
                nv = exact_coord_min(c, p, hp, n)
                if abs(nv - hv) > 1e-10:
                    hp[p] = nv
            hp = project_feasible(hp, n)
        # Also do a few passes over the whole thing
        hp, hc = coord_descent(hp, n, 100, deadline, seed + _ * 31)
        if hc < best_c:
            best_h, best_c = hp, hc
    return best_h, best_c

def multi_alpha_lbfgs(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000, 100000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq',
                 'fun': lambda x, n=n: x.sum() - n / 2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        # Try slightly perturbed starting point
        hp = project_feasible(best_h + rng.normal(0, 0.01, n), n)
        try:
            res = minimize(smooth_obj_and_grad, hp, args=(n, alpha),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': 150, 'gtol': 1e-9,
                                    'ftol': 1e-14, 'maxls': 40})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Exact coordinate descent (workhorse)
    if time.time() < deadline - 500:
        h1, c1 = coord_descent(best_h, n, 3000, deadline - 480, seed)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: L-BFGS-B with analytical gradients + CD polish
    if time.time() < deadline - 350:
        h2, c2 = multi_alpha_lbfgs(best_h, n, deadline - 330, seed)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n, 400, deadline - 320, seed + 1)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Block perturbation + local CD
    if time.time() < deadline - 250:
        h3, c3 = block_perturb_cd(best_h, n, deadline - 230, seed + 5,
                                   n_blocks=15, block_sz=10, noise=0.2)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 300, deadline - 220, seed + 6)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: More L-BFGS-B from slightly different angles
    if time.time() < deadline - 200:
        for r in range(3):
            if time.time() > deadline - 190:
                break
            s = 0.02 + 0.03 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h4, c4 = multi_alpha_lbfgs(hp, n, deadline - 185, seed + 50 + r * 10)
            if c4 < best_c:
                best_h, best_c = h4, c4
            h4, c4 = coord_descent(best_h, n, 300, deadline - 180, seed + 60 + r * 10)
            if c4 < best_c:
                best_h, best_c = h4, c4

    # Phase 5: Multi-restart with varied scales
    for r in range(6):
        if time.time() > deadline - 120:
            break
        s = 0.004 + 0.02 * r
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        h5, c5 = coord_descent(hp, n, 400, deadline - 100, seed + 200 + r * 7)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Try nearby n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [n - 5, n + 5, n + 10, n - 10]:
        if time.time() > deadline - 80:
            break
        if n_try < 100 or n_try > 2000:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h6, c6 = coord_descent(h_int, n_try, 500, deadline - 60, seed + n_try)
        if c6 < best_c:
            best_h, best_c = h6, c6
            x_orig = x_new
            n = n_try

    # Phase 7: Final extensive CD
    if time.time() < deadline - 10:
        n_f = len(best_h)
        hf, cf = coord_descent(best_h, n_f, 3000, deadline - 3, seed + 999)
        if cf < best_c:
            best_c = cf
            best_h = hf

    # Check 1-h symmetry
    n_f = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_f)
    c_flip = compute_c5(h_flip, n_f)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
