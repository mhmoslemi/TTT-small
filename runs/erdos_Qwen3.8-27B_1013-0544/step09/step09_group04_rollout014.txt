def cd_pass(h, n, order='sequential', skip_boundary=True):
    """One pass of coordinate descent. Returns updated h and improvement."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    n_idx = np.arange(n)
    if order == 'reverse':
        idx = n_idx[::-1]
    elif order == 'random':
        idx = n_idx.copy()
        np.random.shuffle(idx)
    else:
        idx = n_idx
    improved = 0
    for p in idx:
        h0 = h[p]
        if skip_boundary and (h0 < 1e-14 or h0 > 1.0 - 1e-14):
            continue
        x_new = solve_1d(c, h, p, n)
        if abs(x_new - h0) > 1e-14:
            h[p] = x_new
            improved += 1
    return h, improved

        if improved == 0:

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

def slsqp_phase(h, n, alphas, maxiter_each=300):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,

            if new_c < best_c:

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
