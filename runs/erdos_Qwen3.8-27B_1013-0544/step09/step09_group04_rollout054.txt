```python
def solve_pair_fast(c, d, diag_val, h0p, h0q, n):
    """Fast pair solver using vectorized operations."""
    lo = max(-h0p, h0q - 1.0)
    hi = min(1.0 - h0p, h0q)
    if lo >= hi - 1e-15:
        return 0.0
    
    diag = n - 1
    c_lin = np.delete(c, diag)  # c for r ≠ diag
    d_lin = np.delete(d, diag)  # d for r ≠ diag
    
    L_lo = 0.0
    L_hi = float(c.max())
    
    for _ in range(40):
        L = (L_lo + L_hi) / 2.0
        
        # Linear constraints
        pos = d_lin > 1e-15
        neg = d_lin < -1e-15
        zero = ~(pos | neg)
        
        x_lo = lo
        x_hi = hi
        
        if np.any(zero) and c_lin[zero].max() > L:
            L_lo = L
            continue
        
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((L - c_lin[pos]) / d_lin[pos])))
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((L - c_lin[neg]) / d_lin[neg])))
        
        # Quadratic constraint
        # -2δ² + 2(h0q-h0p)δ + (c[diag] - L) <= 0
        # 2δ² - 2(h0q-h0p)δ + (L - c[diag]) >= 0
        bq = 2.0 * (h0q - h0p)
        cq = L - c[diag]
        # 2δ² - bq*δ + cq >= 0
        disc = bq*bq - 8.0*cq
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (bq - sq) / 4.0
            r2 = (bq + sq) / 4.0
            # Infeasible for r1 < δ < r2
            if x_lo >= r2 or x_hi <= r1:
                pass  # OK
            elif x_lo <= r1 and x_hi >= r2:
                # Pick the side with more room
                if (r1 - x_lo) >= (x_hi - r2):
                    x_hi = r1
                else:
                    x_lo = r2
            else:
                # Partly overlaps
                if x_lo > r1 and x_lo < r2:
                    x_lo = r2
                if x_hi > r1 and x_hi < r2:
                    x_hi = r1
        
        if x_lo > x_hi + 1e-16:
            L_lo = L
        else:
            L_hi = L
    
    # Find optimal δ at L = L_hi
    L = L_hi
    x_lo = lo
    x_hi = hi
    pos = d_lin > 1e-15
    neg = d_lin < -1e-15
    if np.any(pos):
        x_hi = min(x_hi, float(np.min((L - c_lin[pos]) / d_lin[pos])))
    if np.any(neg):
        x_lo = max(x_lo, float(np.max((L - c_lin[neg]) / d_lin[neg])))
    
    # Quadratic
    bq = 2.0 * (h0q - h0p)
    cq = L - c[diag]
    disc = bq*bq - 8.0*cq
    if disc >= 0:
        sq = np.sqrt(disc)
        r1 = (bq - sq) / 4.0
        r2 = (bq + sq) / 4.0
        if x_lo <= r1 and x_hi >= r2:
            if (r1 - max(x_lo,lo)) >= (min(x_hi,hi) - r2):
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
        elif x_lo > r1 and x_hi < r2:
            return 0.0  # infeasible, no improvement
        else:
            x_lo = max(x_lo, r2) if x_lo > r1 else x_lo
            x_hi = min(x_hi, r1) if x_hi < r2 else x_hi
    
    x_lo = max(x_lo, lo)
    x_hi = min(x_hi, hi)
    if x_lo > x_hi:
        return 0.0
    return float((x_lo + x_hi) / 2.0)
```

Actually, the standard CD approach is: make one change at a time and recompute. But that's expensive. Alternative: make all changes for a "pass" and recompute c once at the end.

But the problem is that after changing h[p], the correlations change, which affects the optimal δ for subsequent (p,q) pairs. So the "greedy" approach of making changes in sequence is more accurate but slower.

A compromise: for each p, find the best single q and δ, apply it, then move to the next p. This is O(n) pair solves per pass, each O(n), totaling O(n²) per pass. For n=700, that's 490000 operations per pass, very fast.

Let me go with this approach:
```
For each pass:
    c = correlate(h, 1-h)
    For p in range(n):
        Find best (q, δ) for this p
        Apply the change
    Optionally: project_feasible(h)
```

For "find best q for p", I'll try a subset of q values:
- q = p+1, p-1 (neighbors)
- q = (p + n//2) % n (opposite)
- A few random q's
- The q with min h[q] (if h[p] > 0.5) or max h[q] (if h[p] < 0.5)

This gives about 10 candidates per p, so O(10n) pair solves per pass = O(n). Very fast.

1. Find the critical correlations (those near the max)
2. For each critical correlation r, identify which coordinates contribute most to it

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return (m + mx + np.log(np.sum(np.exp(s - mx)))) / alpha * (2.0/n)

def solve_1d(c, h, p, n):
    """Find optimal h[p] given all others fixed. Returns new value."""
    h0 = h[p]
    m_lo, m_hi = max(0, n-1-p), min(2*n-2, n-1+p)
    ms = np.arange(m_lo, m_hi+1)
    m = ms - (n-1)  # lag
    # Coefficients
    a = np.zeros(len(ms))
    for idx, (pi, mi) in enumerate(zip(ms, m)):
        # h[p] appears as first factor when i=p, i+m valid: m = pi-(n-1), check p+m in [0,n)
        # h[p] appears as second factor when i+m=p, i=p-m valid: check p-m in [0,n)
        if m_lo <= pi <= m_hi:
            pass  # already in range by construction
        # First factor: p+m valid?
        if 0 <= p + (pi-(n-1)) < n:
            a[idx] += (1.0 - h[p + pi - (n-1)])
        # Second factor: p-m valid? i.e., p-(pi-(n-1)) in [0,n)
        if 0 <= p - (pi-(n-1)) < n:
            a[idx] -= h[p - pi + (n-1)]
    
    # Diagonal term
    diag = n - 1
    di = (ms == diag)[0]  # first occurrence (there's only one)
    has_diag = (diag >= m_lo) and (diag <= m_hi)
    
    c_base = c[ms].copy()
    if has_diag:
        di_idx = np.where(ms == diag)[0][0]
        c_diag = c_base[di_idx]
        c_base[di_idx] = 0.0
        a[di_idx] = 0.0  # will handle separately
    
    # Unaffected correlations
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, c[:m_lo].max())
    if m_hi < 2*n-2:
        unaff = max(unaff, c[m_hi+1:].max())
    
    L_lo, L_hi = unaff, c.max()
    for _ in range(50):
        if L_hi - L_lo < 1e-17: break
        L = (L_lo + L_hi) / 2
        x_lo, x_hi = 0.0, 1.0
        # Linear constraints
        pos = a > 1e-15
        neg = a < -1e-15
        zero = ~(pos | neg)
        if np.any(zero) and c_base[zero].max() > L:
            L_lo = L; continue
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
        # Quadratic (diagonal)
        if has_diag:
            # c_diag + (x-h0)(1-x-h0) <= L
            # c_diag + x - h0 - x² + h0*x - x*h0 + h0² <= L  ... let me redo
            # (x-h0)(1-x-h0) = x - x² - h0 + h0*x = x(1-h0) - x² - h0
            # Wait: (x-h0)(1-x-h0) = x(1-x-h0) - h0(1-x-h0) = x - x² - xh0 - h0 + h0x + h0²
            # = x - x² - h0 + h0²
            # So: c_diag + x - x² - h0 + h0² <= L
            # => x² - x + (L - c_diag + h0 - h0²) >= 0
            # Quadratic: δ = x, a=1, b=-1, c = L - c_diag + h0 - h0²
            Q = L - c_diag + h0 - h0*h0
            # x² - x + Q >= 0
            disc = 1.0 - 4.0*Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl, xh = max(x_lo, 0.0), min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    L_lo = L; continue
                if xl <= r1 and xh >= r2:
                    if h0 - r1 < r2 - h0:
                        x_hi = min(x_hi, r1)
                    else:
                        x_lo = max(x_lo, r2)
                elif xl <= r1:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
        if x_lo > x_hi + 1e-16:
            L_lo = L
        else:
            L_hi = L
    
    # Find x at L = L_hi
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    pos = a > 1e-15
    neg = a < -1e-15
    if np.any(pos): x_hi = min(x_hi, float(np.min((L-c_base[pos])/a[pos])))
    if np.any(neg): x_lo = max(x_lo, float(np.max((L-c_base[neg])/a[neg])))
    if has_diag:
        Q = L - c_diag + h0 - h0*h0
        disc = 1.0 - 4.0*Q
        if disc >= 0:
            r1 = (1.0-np.sqrt(disc))/2.0
            r2 = (1.0+np.sqrt(disc))/2.0
            if x_lo > r1 and x_hi < r2: return h0
            if x_lo <= r1 and x_hi >= r2:
                if abs(h0-r1) < abs(h0-r2): x_hi = min(x_hi, r1)
                else: x_lo = max(x_lo, r2)
            elif x_lo <= r1: x_hi = min(x_hi, r1)
            else: x_lo = max(x_lo, r2)
    x_lo, x_hi = max(x_lo,0.0), min(x_hi,1.0)
    if x_lo > x_hi: return h0
    return float((x_lo+x_hi)/2.0)

def cd_pass(h, n, deadline, order='seq'):
    h = h.copy()
    for _ in range(10000):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        improved = False
        for p in range(n):
            h0 = h[p]

        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
