sum_i [h*[i]*(1-h*[i+m]) + (h[i]-h*[i])*(1-h*[i+m]) + h*[i]*(h*[i+m]-h[i+m])] ≤ t/dx

This is a linear constraint! So we can solve a sequence of LPs (successive linearization).

**Algorithm: Successive Linear Programming (SLP)**

1. Start with h₀ (from initial_h_values)
2. At each iteration k:
   a. Linearize the bilinear terms around h_k
   b. Solve: min t s.t. linearized constraints, 0≤h≤1, sum(h)=n/2
   c. Update h_{k+1} = solution
   d. Check convergence

This is essentially the Dinkelbach method or convex-concave procedure (CCP).

The LP has variables h[0..n-1] and t, with 2n-1 constraints (one per correlation shift). With n=600, that's 601 variables and ~1200 constraints. This should be very fast with CVXPY and a good solver.

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h, n):
    """Build G where G[m,j] = d(c[m])/d(h[j]) at current h."""
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        s = m - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:
            # First term: (1-h[j+s]) for j in [i_lo, i_hi]
            G[m, i_lo:i_hi+1] += (1.0 - h[i_lo+s:i_hi+1+s])
            # Second term: -h[j-s] for j in [i_lo+s, i_hi+s] ∩ [0, n-1]
            j_lo = max(0, i_lo + s)
            j_hi = min(n-1, i_hi + s)
            if j_lo <= j_hi:
                G[m, j_lo:j_hi+1] -= h[j_lo-s:j_hi-s+1]
    return G

def slp_active_set(h, n, max_iters=50, deadline=None, top_k=20, damping=1.0):
    """SLP with active set focusing on top-k correlations."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        c_full = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c_full.max())
        
        # Get top-k correlations
        top_indices = np.argsort(c_full)[-top_k:]
        top_indices = np.sort(top_indices)
        
        # Build gradient matrix for only these rows
        G = build_gradient_matrix(h, n)
        G_active = G[top_indices, :]
        c_active = c_full[top_indices]
        
        # LP: min t s.t. G_active @ dh <= t - c_active, -h <= dh <= 1-h, sum(dh)=0
        dh = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [
            dh >= -h,
            dh <= 1.0 - h,
            cp.sum(dh) == 0,
        ]
        for k in range(len(top_indices)):
            constraints.append(G_active[k] @ dh <= t - c_active[k])
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000, abstol=1e-10)
        except:
```

```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, golden

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_G_rows(h, n, m_indices):
    """Build rows of gradient matrix for specific correlation indices."""
    G = np.zeros((len(m_indices), n))
    for idx, m in enumerate(m_indices):
        s = m - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:
            G[idx, i_lo:i_hi+1] += (1.0 - h[i_lo+s:i_hi+1+s])
            j_lo = max(0, i_lo + s)
            j_hi = min(n-1, i_hi + s)
            if j_lo <= j_hi:
                G[idx, j_lo:j_hi+1] -= h[j_lo-s:j_hi-s+1]
    return G

def slp_optimize(h, n, max_iters=100, top_k=25, deadline=None, damping=1.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        c_full = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c_full.max())
        
        # Top-k indices
        top_idx = np.argsort(c_full)[-top_k:]
        
        G = build_G_rows(h, n, top_idx)
        c_active = c_full[top_idx]
        
        dh = cp.Variable(n)
        t_var = cp.Variable()
        
        cons = [
            dh >= -h,
            dh <= 1.0 - h,
            cp.sum(dh) == 0,
        ]
        for k in range(len(top_idx)):
            cons.append(G[k] @ dh <= t_var - c_active[k])
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000, abstol=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=20000)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        dh_val = np.asarray(dh.value).flatten()
        
        # Try full step first, then damped
        for damp in [damping, 0.7*damping, 0.4*damping, 0.2*damping]:
            h_new = np.clip(h + damp * dh_val, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                break
        else:
            break
        
        # Check convergence
        if abs(compute_c5(h, n) - c_max) < 1e-10:
            break
    
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [max(lo, min(hi, 0.0))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = candidates[0]
    best_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

        cur_max = float(c.max())

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:

            m_b_lo = max(0, idx)
            m_b_hi = min(2*n-2, idx+n-1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)

            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_i = np.argsort(vals)[-top_k:]
                top_i = np.sort(top_i)
                ms = ms_all[top_i]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):

            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]

            if m_hi < 2*n-2:

            diag_local = None
            for ii in range(len(ms)):

                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)

                    cur_max = min(cur_max, new_val)
```

```python
        c_full = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c_full.max())
        top_idx = np.argsort(c_full)[-top_k:]
        G = build_G_rows(h, n, top_idx)
        c_active = c_full[top_idx]
        dh = cp.Variable(n)
        t_var = cp.Variable()
        cons = [dh >= -h, dh <= 1.0 - h, cp.sum(dh) == 0]
        for k in range(len(top_idx)):
            cons.append(G[k] @ dh <= t_var - c_active[k])
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABINE]:
            try:
                prob.solve(solver=solver, max_iters=10000)
                solved = True
                break
            except Exception:
                continue
        if not solved or prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        dh_val = np.asarray(dh.value).flatten()
        accepted = False
        for damp in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_new = np.clip(h + damp * dh_val, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                accepted = True
                break
        if not accepted:
            break
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [max(lo, min(hi, 0.0))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = candidates[0]
    best_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        d = max(lo, min(hi, d))
        v = obj_1d(d, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = d
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
            except Exception:
                pass
    return best_d

def coord_descend(h, n, max_passes=200, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(max(0, n - 1 - idx), max(0, idx))
            m_hi = max(min(2*n-2, 2*n-2-idx), min(2*n-2, idx+n-1))
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_i = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_i]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < cur_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    cur_max = min(cur_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)
```
