            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smart_grad(h, n, alpha):
    """Analytical gradient of smooth-max correlation"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * c
    s -= s.max()
    ec = np.exp(s)
    w = ec / ec.sum()
    grad_h = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-16:
            continue
        m = k - n1
        if m >= 0:
            pe = n - m
            grad_h[:pe] += wi * (1.0 - h[m:m + pe])
            grad_h[m:] -= wi * h[:pe]
        else:
            ms = -m
            pe = n + m
            grad_h[ms:] += wi * (1.0 - h[:pe])
            grad_h[:pe] -= wi * h[ms:]
    val = (c.max() + np.log(ec.sum()) / alpha) * dx
    return val, grad_h * dx

def adam_optimize(h, n, deadline, alphas, lr0=0.005, beta1=0.8, beta2=0.999, eps=1e-8):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha_i = 0
    for _ in range(50000):
        if time.time() > deadline:
            break
        t += 1
        alpha = alphas[alpha_i % len(alphas)]
        if t % 500 == 0:
            alpha_i += 1
        val, g = smart_grad(h, n, alpha)
        if not np.all(np.isfinite(g)):
            continue
        lr = lr0 / (1.0 + 0.01 * t)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-peak CD
    for tk in [5, 10, 20, 30]:
        if time.time() > deadline - 400:
            break
        h2, c2 = multi_cd(best_h, n, 300, deadline - 400, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 2: Adam gradient optimization
    if time.time() < deadline - 350:
        alphas = [200, 500, 1000, 3000, 10000]
        h_a, c_a = adam_optimize(best_h, n, deadline - 350, alphas, lr0=0.003)
        if c_a < best_c:
            best_h, best_c = h_a, c_a

    # Phase 3: CD refinement after gradient
    if time.time() < deadline - 250:
        h3, c3 = multi_cd(best_h, n, 400, deadline - 250, 25)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: More Adam with different params
    if time.time() < deadline - 200:
        alphas2 = [5000, 20000, 50000]
        h_a2, c_a2 = adam_optimize(best_h, n, deadline - 200, alphas2, lr0=0.002)
        if c_a2 < best_c:
            best_h, best_c = h_a2, c_a2

    # Phase 5: Interleave CD and gradient
    if time.time() < deadline - 100:
        for round_i in range(4):
            if time.time() > deadline - 100:
                break
            h5, c5 = multi_cd(best_h, n, 100, deadline - 100, 30)
            if c5 < best_c:
                best_h, best_c = h5, c5
            if time.time() > deadline - 100:
                break
            h5b, c5b = adam_optimize(best_h, n, deadline - 100, [5000, 20000], lr0=0.001)
            if c5b < best_c:
                best_h, best_c = h5b, c5b

    # Phase 6: Stochastic perturbations
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        for r in range(15):
            if time.time() > deadline - 80:
                break
            s = 0.001 + 0.003 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            hc, cc = multi_cd(hp, n_cur, 150, deadline - 80, 15)
            if cc < best_c:
                best_h, best_c = hc, cc

    # Phase 7: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 700, 800, 1000]:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 200, deadline - 100, 15)
        h_o, c_o = try_symmetry(h_o, n_try)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new

    # Phase 8: Final refinement
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        h_f, c_f = adam_optimize(best_h, n_cur, deadline - 30, [10000, 50000, 100000], lr0=0.001)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 3:
        h_f, c_f = multi_cd(best_h, n_cur, 2000, deadline, 40)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
