import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def compute_smoothed_grad(h, n, alpha):
    """Compute gradient of log-sum-exp smoothing of max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # c[m] corresponds to shift k = m - (n-1)
    log_w = alpha * c * dx
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        sh = m - (n - 1)  # shift
        wm = w[m]
        if wm < 1e-12:
            continue
        # ∂c[m]/∂h[j] = (1-h[j-sh])·[0≤j-sh<n] - h[j+sh]·[0≤j+sh<n]
        # For the first term: j-sh valid means 0≤j-sh≤n-1, i.e., sh≤j≤sh+n-1
        #   and j in [0,n-1], so j in [max(0,sh), min(n-1, sh+n-1)]
        lo1 = max(0, sh)
        hi1 = min(n - 1, sh + n - 1)
        if hi1 >= lo1:
            grad[lo1:hi1+1] += wm * (1.0 - h[lo1-sh:hi1+1-sh])
        # For the second term: j+sh valid means 0≤j+sh≤n-1, i.e., -sh≤j≤n-1-sh
        #   and j in [0,n-1], so j in [max(0,-sh), min(n-1, n-1-sh)]
        lo2 = max(0, -sh)
        hi2 = min(n - 1, n - 1 - sh)
        if hi2 >= lo2:
            grad[lo2:hi2+1] -= wm * h[lo2+sh:hi2+1+sh]
    
    return dx * grad

def smoothed_gradient_descent(h, n, alphas, deadline, momentum=0.9, lr_scale=1.0):
    """Vanilla gradient descent with momentum on the smoothed objective."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    
    for alpha in alphas:
        lr = lr_scale / (alpha * 0.1 + 1.0)
        steps = 300
        for s in range(steps):
            if time.time() > deadline:
                break
            g = compute_smoothed_grad(h, n, alpha)
            v = momentum * v - lr * g
            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
            h = h_new
            # Decay learning rate
            lr *= 0.999
        print(f"  alpha={alpha}, c={best_c:.6f}")
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smoothed_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    lw = alpha * c * dx
    lw -= lw.max()
    w = np.exp(lw)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        sh = m - (n - 1)
        wm = w[m]
        if wm < 1e-12: continue
        lo1 = max(0, sh)
        hi1 = min(n-1, sh+n-1)
        if hi1 >= lo1:
            grad[lo1:hi1+1] += wm * (1.0 - h[lo1-sh:hi1+1-sh])
        lo2 = max(0, -sh)
        hi2 = min(n-1, n-1-sh)
        if hi2 >= lo2:
            grad[lo2:hi2+1] -= wm * h[lo2+sh:hi2+1+sh]
    return dx * grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n as 0-1 array."""
    out = np.zeros(n)
    for i in range(n):
        out[i] = bin(i).count('1') % 2
    return out

def sa_01(n, rng, deadline, n_steps=200000, T0=0.15, T1=0.0005):
    """Simulated annealing on binary patterns."""
    b = np.zeros(n)
    b[:n//2] = 1.0
    rng.shuffle(b)
    dx = 2.0/n
    c_curr = float(np.correlate(b, 1.0-b, 'full').max())
    c_best = c_curr
    b_best = b.copy()
    cooling = (T1/T0)**(1.0/n_steps)
    T = T0
    for step in range(n_steps):
        if time.time() > deadline: break
        T *= cooling
        ones = np.where(b==1.0)[0]
        zeros = np.where(b==0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        c_new = float(np.correlate(b, 1.0-b, 'full').max())
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta*dx/max(T,1e-10)):
            c_curr = c_new
            if c_new < c_best:
                c_best = c_new
                b_best = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    return b_best, c_best*dx

    """Find optimal h[p] given correlation c and other h values."""
    h0 = h[p]
    m_lo = max(0, n-1-p)
    m_hi = min(2*n-2, n-1+p)
    ms = np.arange(m_lo, m_hi+1)
    jf = p + ms - (n-1)
    jb = p - ms + (n-1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = (ms == (n-1))
    c_diag = 0.0
    if di.any():

        c_base[di] = 0.0

            x_hi = min(x_hi, float(np.min((L - c_base[pos_m])/a[pos_m])))
        if x_lo > x_hi + 1e-16: feas = False

            Q = L - c_diag + h0*(1.0-h0)

                r2 = (1.0 + np.sqrt(disc))/2.0
                xl, xh = max(x_lo,0.0), min(x_hi,1.0)
                if xl > r1 and xh < r2: feas = False
        if feas: L_hi = L
        else: L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m])/a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m])/a[pos_m])))
    x_lo, x_hi = max(x_lo,0.0), min(x_hi,1.0)
    if di.any():
        Q = L - c_diag + h0*(1.0-h0)
        disc = 1.0 - 4.0*Q
        if disc >= 0:
            r1 = (1.0-np.sqrt(disc))/2.0
            r2 = (1.0+np.sqrt(disc))/2.0
            if x_lo > r1 and x_hi < r2: return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0-r1 if x_lo <= r1 else 1e9
                d2 = r2-h0 if x_hi >= r2 else 1e9
                if d1 < d2: x_hi = min(x_hi, r1)
                else: x_lo = max(x_lo, r2)
            elif x_lo <= r1: x_hi = min(x_hi, r1)
            else: x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14: return h0
    return float(np.clip((x_lo+x_hi)/2.0, 0.0, 1.0))

def coord_cd(h, n, deadline, max_passes=100):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            x_new = solve_1d(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def grad_step(h, n, alpha, lr):
    """One gradient step on smoothed max correlation."""
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    lw = alpha * c * dx
    lw -= lw.max()
    w = np.exp(lw)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2*n-1):
        sh = m - (n-1)
        wm = w[m]
        if wm < 1e-12: continue
        lo1, hi1 = max(0,sh), min(n-1, sh+n-1)
        if hi1 >= lo1:
            grad[lo1:hi1+1] += wm * (1.0 - h[lo1-sh:hi1+1-sh])
        lo2, hi2 = max(0,-sh), min(n-1, n-1-sh)
        if hi2 >= lo2:
            grad[lo2:hi2+1] -= wm * h[lo2+sh:hi2+1+sh]
    return project_feasible(h + lr * dx * grad, n)

def smoothed_grad_desc(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [30, 20, 10, 5, 3, 2, 1.5]
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        lr = 0.8 / (alpha * 0.5 + 1)
        for s in range(200):
            if time.time() > deadline: break
            h_new = grad_step(h, n, alpha, lr)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h_init = np.full(500, 0.5)
    n = len(h_init)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Try 0-1 patterns
    # 1a: Thue-Morse
    tm = thue_morse(n)
    tm_proj = project_feasible(tm, n)
    c_tm = compute_c5(tm_proj, n)
    if c_tm < best_c:
        best_h, best_c = tm_proj, c_tm
    
    # 1b: Simulated annealing
    if time.time() < deadline - 200:
        b_sa, c_sa = sa_01(n, rng, deadline - 300, n_steps=200000)
        if c_sa < best_c:
            best_h, best_c = b_sa, c_sa
    
    # Phase 2: Continuous optimization from best 0-1 and from initial
    for start_h, start_n in [(best_h, n), (project_feasible(h_init, n), n)]:
        if time.time() > deadline - 150: break
        h_g, c_g = smoothed_grad_desc(start_h, start_n, deadline - 150)
        if c_g < best_c:
            best_h, best_c = h_g, c_g
    
    # Phase 3: Exact CD
    if time.time() < deadline - 50:
        h_c, c_c = coord_cd(best_h, len(best_h), deadline - 30, max_passes=80)
        if c_c < best_c:
            best_h, best_c = h_c, c_c
    
    # Phase 4: Try different n
    for n_try in [400, 500, 600, 800, 1000]:
        if time.time() > deadline - 80: break
        if n_try == len(best_h): continue
        x_o = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_n, x_o, best_h), n_try)
        h_t, c_t = coord_cd(h_int, n_try, deadline - 50, max_passes=30)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
