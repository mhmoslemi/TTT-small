Actually, for the Erdős overlap problem, the known constructions involve functions that are 1 on a set of measure 1 and 0 elsewhere, but structured so that the overlap with its translates is minimized. The key is that h should be such that h(x) and h(x+k) don't overlap much for any k.

This is related to the concept of "cross-intersecting" families or "difference sets." In the continuous case, we want a set A of measure 1 in [0,2] such that |A ∩ (A+k)| is small for all k.

If h is truly binary (0 or 1), then h(x)(1-h(x+k)) = 1 iff x ∈ A and x+k ∉ A, i.e., x ∈ A \ (A-k). So the overlap is |A \ (A-k)| = |A| - |A ∩ (A-k)| = 1 - |A ∩ (A-k)|.

Maximizing |A ∩ (A-k)| for all k would minimize the overlap. But we can't maximize it for all k simultaneously (that would require A to be invariant under all shifts, i.e., A = [0,2]).

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()

            if tot < 1e-14:

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):

    if diag_idx is not None:
        vals[diag_idx] -= d * d

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):

def smooth_grad_step(h, n, k_peaks, lr, max_iter, deadline):
    """Gradient step using softmax-weighted subgradient."""
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_sorted = np.sort(c)[::-1]
        kk = min(k_peaks, len(c_sorted))
        threshold = c_sorted[kk - 1]
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]
        # Softmax weights with temperature
        c_active = c[active_idx]
        temp = np.max(c_active) - np.min(c_active)
        if temp < 1e-10:
            w = np.ones(len(active_idx)) / len(active_idx)
        else:
            w = np.exp(10.0 * (c_active - c_active.max()) / temp)
            w /= w.sum()
        # Compute weighted gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, m in zip(w, active_idx):
            s = m - (n - 1)
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n - s] += wi * (1.0 - h[s:])
                grad[s:] += wi * (-h[:n - s])
            else:
                sa = -s
                grad[sa:] += wi * (1.0 - h[:n - sa])
                grad[:n - sa] += wi * (-h[sa:])
        grad *= dx
        # Step
        h = project_feasible(h - lr * grad, n)
    return h, compute_c5(h, n)

        c = np.correlate(h, 1.0 - h, mode='full')
        c_sorted = np.sort(c)[::-1]
        kk = min(k_peaks, len(c_sorted))
        threshold = c_sorted[kk - 1]
        active_idx = np.where(c >= threshold)[0]
        c_active = c[active_idx]
        spread = c_active.max() - c_active.min()
        if spread < 1e-10:
            w = np.ones(len(active_idx)) / len(active_idx)
        else:
            w = np.exp(20.0 * (c_active - c_active.max()) / spread)
            w /= w.sum()
        grad = np.zeros(n)
        for wi, m in zip(w, active_idx):
            s = m - (n - 1)
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n - s] += wi * (1.0 - h[s:])
                grad[s:] += wi * (-h[:n - s])
            else:
                sa = -s
                grad[sa:] += wi * (1.0 - h[:n - sa])
                grad[:n - sa] += wi * (-h[sa:])
        grad *= dx
        h = project_feasible(h - lr * grad, n)
    return h, compute_c5(h, n)
