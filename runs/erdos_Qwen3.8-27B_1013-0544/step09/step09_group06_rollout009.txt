```python
    if diag_idx is not None:
```

```python
    T = len(c_peak)
    # Generate all candidates vectorized
    candidates = [lo, hi]
    
    # Pair crossings
    for i in range(T):
        if i == diag_idx:
            continue
        da = a_coeff[i] - a_coeff[diag_idx if diag_idx is not None else 0]  # skip
    
    # Actually, let's just do it the simple way but with numpy
    # For non-diagonal pairs:
    if diag_idx is not None:
        idx_range = [i for i in range(T) if i != diag_idx]
    else:
        idx_range = list(range(T))
    
    # Pair crossings (non-diagonal)
    n_idx = len(idx_range)
    for ii in range(n_idx):
        i = idx_range[ii]
        for jj in range(ii + 1, n_idx):
            j = idx_range[jj]
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    
    # Diagonal crossings
    if diag_idx is not None:
        for i in idx_range:
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    
    # Unaff crossings
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    
    # Evaluate all candidates
    candidates = np.array(candidates)
    candidates = np.clip(candidates, lo, hi)
    
    vals = c_peak[None, :] + candidates[:, None] * a_coeff[None, :]
    if diag_idx is not None:
        vals[:, diag_idx] -= candidates ** 2
    obj = np.maximum(vals.max(axis=1), unaff_max)
    
    best_i = int(obj.argmin())
    return float(candidates[best_i]), float(obj[best_i])
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    if diag_idx is not None:
        idx_range = [i for i in range(T) if i != diag_idx]
    else:
        idx_range = list(range(T))
    n_idx = len(idx_range)
    for ii in range(n_idx):
        i = idx_range[ii]
        for jj in range(ii + 1, n_idx):
            j = idx_range[jj]
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in idx_range:
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    candidates = np.array(candidates)
    candidates = np.clip(candidates, lo, hi)
    vals = c_peak[None, :] + candidates[:, None] * a_coeff[None, :]
    if diag_idx is not None:
        vals[:, diag_idx] -= candidates ** 2
    obj = np.maximum(vals.max(axis=1), unaff_max)
    best_i = int(obj.argmin())
    return float(candidates[best_i]), float(obj[best_i])

def fast_multi_cd(h, n, max_passes, deadline, top_k, coord_order=None):
    h = h.copy()
    two_n_1 = 2 * n - 1
    all_mask_base = np.ones(two_n_1, dtype=bool)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, two_n_1)
        top_idx = np.argpartition(c, -k)[-k:]
        c_peak = c[top_idx].copy()
        all_mask_base[:] = True
        all_mask_base[top_idx] = False
        unaff_max = float(c[all_mask_base].max()) if all_mask_base.any() else 0.0
        m_vals = top_idx.astype(np.int64)
        improved = False
        if coord_order is not None:
            order = coord_order
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            jf = p + m_vals - (n - 1)
            jb = p - m_vals + (n - 1)
            a_coeff = np.zeros(k)
            mask_f = (jf >= 0) & (jf < n)
            mask_b = (jb >= 0) & (jb < n)
            a_coeff[mask_f] = 1.0 - h[jf[mask_f]]
            a_coeff[mask_b] += -h[jb[mask_b]]
            diag_mask = (m_vals == n - 1)
            diag_idx = int(np.where(diag_mask)[0][0]) if diag_mask.any() else None
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n = 600
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD with increasing top_k
    for tk in [20, 40, 60, 90, 120, 150]:
        remaining = deadline - time.time()
        if remaining < 120:
            break
        alloc = min(remaining * 0.22, 220)
        h2, c2 = fast_multi_cd(best_h, n, 100000, time.time() + alloc, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 2: Perturbation with shuffled coords
    remaining = deadline - time.time()
    if remaining > 120:
        n_cur = len(best_h)
        for r in range(50):
            if time.time() > deadline - 50:
                break
            s = 0.0005 * (1.0 + 3.0 * r / 50.0)
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            order = rng.permutation(n_cur)
            alloc = min(deadline - 50 - time.time(), 60)
            if alloc < 5:
                break
            h2, c2 = fast_multi_cd(hp, n_cur, 100000, time.time() + alloc, 70, coord_order=order)
            if c2 < best_c:
                best_h, best_c = h2, c2
    # Phase 3: Try different n values
    for n_try in [550, 575, 625, 650, 700, 400]:
        remaining = deadline - time.time()
        if remaining < 80:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        alloc = min(remaining * 0.18, remaining - 60, 100)
        if alloc < 10:
            break
        h2, c2 = fast_multi_cd(h_int, n_try, 100000, time.time() + alloc, 70)
        h2, c2_s = fast_multi_cd(h2, n_try, 100000, min(deadline - 50, time.time() + alloc * 0.5), 50)
        c2 = min(c2, c2_s)
        if c2_s < c2:
            c2 = c2_s
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 4: Final deep CD
    remaining = deadline - time.time()
    if remaining > 20:
        h2, c2 = fast_multi_cd(best_h, len(best_h), 1000000, deadline - 5, 150)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    # One more CD after symmetry
    remaining = deadline - time.time()
    if remaining > 10:
        h2, c2 = fast_multi_cd(best_h, len(best_h), 100000, deadline - 2, 100)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
