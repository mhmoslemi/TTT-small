import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-16:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-15:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
                    elif 0.0 >= r1:
                        pass
                    elif 1.0 <= r2:
                        pass
                    else:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)

                x_hi = min(x_hi, r1)
            elif xh2 > r2:
                x_lo = max(x_lo, r2)
            else:

                    if h0 - r1 <= r2 - h0:
                        x_hi = min(x_hi, r1 + 1e-9)
                    else:
                        x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, use_random=True):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if use_random and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones_like(x)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
