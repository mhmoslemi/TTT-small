import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c - alpha * c.max()
    w = np.exp(ac)
    w /= w.sum()
    grad = np.zeros(n)
    n_c = len(w)
    for mk in range(n_c):
        if w[mk] < 1e-7:
            continue
        k = mk - (n - 1)
        wi = w[mk]
        if k >= 0:
            if n - k > 0:
                grad[:n - k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:n - k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += wi * (1.0 - h[:n - kk])
                grad[:n - kk] -= wi * h[kk:]
    return grad * (2.0 / n)

def smooth_gd(h, n, alphas, steps_each=100, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        h_cur = best_h.copy()
        for step in range(steps_each):
            if deadline is not None and time.time() > deadline:
                break
            g = smooth_grad(h_cur, n, alpha)
            lr = 1.0

                h_new = project_feasible(h_cur - lr * g, n)

                    h_cur = h_new
                    improved = True
                    if c_new < best_c:
                        best_h, best_c = h_new.copy(), c_new
                    break
                lr *= 0.5
            if not improved:
                break
    return best_h, best_c

def basin_hopping(h, n, n_restarts=25, cd_passes=400, deadline=None, seed=0):

    best_h = h.copy()
    best_c = compute_c5(h, n)

            if p2 + bl <= n:

                h_p[p1:p1 + bl] = h_p[p2:p2 + bl]
