            h = np.clip(h + diff / tot * h, 0.0, 1.0)

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-16:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-15:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if x_lo >= r1 - 1e-14 and x_hi <= r2 + 1e-14:
                    if h0 - r1 <= r2 - h0:
                        x_hi = min(x_hi, r1 + 1e-9)
                    else:
                        x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float((x_lo + x_hi) / 2.0)

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fourier_eval(coeffs, n, M):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for m in range(M):
        h += coeffs[2*m] * np.cos((m+1)*np.pi*x)
        h += coeffs[2*m+1] * np.sin((m+1)*np.pi*x)
    return h

def smooth_c5(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(sh - sh.max())))
    return (lse / alpha) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n_init = 700
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    def try_update(h, n):
        nonlocal best_h, best_c
        c = compute_c5(h, n)
        if c < best_c:
            best_h, best_c = h.copy(), c
        return c

    # Phase 1: Fourier global search
    for M in [3, 4, 5, 6, 8]:
        if time.time() > deadline - 300:
            break
        n_vars = 2 * M
        for start_type in range(4):
            if time.time() > deadline - 300:
                break
            if start_type == 0:
                coeffs0 = np.zeros(n_vars)
            elif start_type == 1:
                coeffs0 = rng.normal(0, 0.15, n_vars)
            elif start_type == 2:
                coeffs0 = np.zeros(n_vars)
                coeffs0[0] = 0.35
                if n_vars > 2: coeffs0[2] = -0.15
            else:
                coeffs0 = np.zeros(n_vars)
                coeffs0[1] = 0.25
                if n_vars > 4: coeffs0[3] = -0.1
            def fobj(coeffs):
                h = fourier_eval(coeffs, n_init, M)
                h = np.clip(h, 0, 1)
                if abs(h.sum() - n_init/2.0) > 1.0:
                    return 9.99
                h = project_feasible(h, n_init)
                return smooth_c5(h, n_init, 500.0)
            try:
                res = minimize(fobj, coeffs0, method='Nelder-Mead',
                             options={'maxiter': 8000, 'xatol': 1e-11, 'fatol': 1e-14, 'adaptive': True})
                h_f = project_feasible(fourier_eval(res.x, n_init, M), n_init)
                c_f = compute_c5(h_f, n_init)
                if c_f < best_c:
                    best_h, best_c = h_f.copy(), c_f
            except Exception:
                pass

    # Phase 2: CD from best (random order, then sequential)
    if time.time() < deadline - 150:
        h_cd, c_cd = coord_descent(best_h, len(best_h), 250, deadline, rng, random_order=True)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    if time.time() < deadline - 50:
        h_cd, c_cd = coord_descent(best_h, len(best_h), 100, deadline, rng, random_order=False)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 3: CD from initial separately
    if time.time() < deadline - 150:
        h_i2 = project_feasible(h_init, n_init)
        h_cd2, c_cd2 = coord_descent(h_i2, n_init, 200, deadline, rng, random_order=True)
        if c_cd2 < best_c:
            best_h, best_c = h_cd2, c_cd2

    # Phase 4: Multi-n exploration
    n_cands = [400, 500, 600, 800, 840, 550, 650]
    n_cands = [x for x in n_cands if x != len(best_h)]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_cands:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, 200, deadline, rng, random_order=True)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            x_orig = x_new

    # Phase 5: Perturbation escapes
    n = len(best_h)
    if time.time() < deadline - 80:
        for r in range(6):
            if time.time() > deadline - 50:
                break
            sigma = 0.005 + 0.02 * r / 6.0
            hp = best_h + rng.normal(0, sigma, n)
            hp = project_feasible(hp, n)
            h_o, c_o = coord_descent(hp, n, 80, deadline, rng, random_order=True)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final heavy CD polish
    if time.time() < deadline - 5:
        n = len(best_h)
        rem = deadline - time.time() - 3
        max_p = max(50, int(rem / 0.04))
        hf, cf = coord_descent(best_h, n, max_p, deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, n, 50, deadline, rng, random_order=True)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
