        weights = np.exp(shifted - mx)
        weights /= weights.sum()
        grad = np.zeros(n)
        for p in range(n):
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            di = np.where(ms == (n - 1))[0]
            if len(di) > 0:
                a[di] = 1.0 - 2.0 * h[p]
            grad[p] = np.dot(weights[ms], a) * dx
        c0 = best_c
        improved = False
        for _ls in range(8):
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c0 - 1e-15:
                h = h_new
                best_c = c_new
                improved = True
                break
            lr *= 0.5
        if improved:
            lr = min(lr * 1.1, 2.0)
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 50:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n_init = 700
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP warmup
    h1, c1 = slsqp_optimize(best_h, n_init, [500, 2000, 8000], 200)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Main CD
    h2, c2 = cd_optimize(best_h, n_init, 60, deadline, rng, 'random')
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Subgradient polish
    if time.time() < deadline - 150:
        h3, c3 = subgrad_polish(best_h, n_init, 400, deadline, alpha=200.0)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: More CD after subgrad
    if time.time() < deadline - 120:
        h4, c4 = cd_optimize(best_h, n_init, 40, deadline, rng, 'random')
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Structured multi-restart
    n = n_init
    if time.time() < deadline - 80:
        freqs = [1.0, 2.0, 3.0, 5.0, 7.0, 11.0]
        amps = [0.04, 0.03, 0.02, 0.015, 0.01, 0.005]
        x_grid = np.linspace(0, 2, n)
        for f, amp in zip(freqs, amps):
            if time.time() > deadline - 40:
                break
            phase = rng.uniform(0, 2.0 * np.pi)
            pert = amp * np.sin(2.0 * np.pi * f * x_grid + phase)
            h_p = project_feasible(best_h + pert, n)
            h_opt, c_opt = cd_optimize(h_p, n, 8, deadline, rng, 'random')
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
    # Phase 6: Try nearby n
    for n_try in [n_init - 50, n_init + 50, n_init + 100, n_init - 100]:
        if time.time() > deadline - 30:
            break
        if n_try < 200 or n_try > 2000:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = cd_optimize(h_int, n_try, 8, deadline, rng, 'random')
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 7: Final sequential CD polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        h_f, c_f = cd_optimize(best_h, n_f, 20, deadline, rng, 'sequential')
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
