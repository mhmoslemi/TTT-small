def exact_coord_delta(h, p, c, n):

        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:

    if mhi < 2 * n - 2:

            hi = mid
        elif vmax < -1e-10:

        else:

            break

def binary_sa(h_bin, n, max_iter=50000, deadline=None, rng=None):
    """Simulated annealing on binary solutions with swap moves."""
    h = h_bin.copy()
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_c = cur_max * (2.0 / n)
    best_h = h.copy()
    
    T0 = 0.5  # initial "temperature" (in units of correlation change)
    T = T0
    cooling = 0.9995
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Random swap
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        if i1 == i0:
            continue
        
        # Perform swap
        h[i1] = 0
        h[i0] = 1
        
        # Recompute correlation
        c_new = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(c_new.max())
        
        # Accept or reject
        if new_max <= cur_max:
            # Always accept improvements
            cur_max = new_max
            c = c_new
            # Update ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max * (2.0 / n) < best_c:
                best_c = cur_max * (2.0 / n)
                best_h = h.copy()
        else:
            # Metropolis acceptance
            delta = new_max - cur_max
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                c = c_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                # Reject: undo swap
                h[i1] = 1
                h[i0] = 0
        
        T *= cooling
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_sa(h_bin, n, max_iter=100000, deadline=None, rng=None, T_init=1.5, cooling=0.9997):
    h = h_bin.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_c = cur_max * (2.0 / n)
    best_h = h.copy()
    T = T_init
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if rng.random() < 0.25:
            i1 = ones_list[rng.randint(len(ones_list))]
            if i1 > 0 and h[i1 - 1] == 0:
                i0 = i1 - 1
            elif i1 < n - 1 and h[i1 + 1] == 0:
                i0 = i1 + 1
            else:
                i0 = zeros_list[rng.randint(len(zeros_list))]
        else:
            i1 = ones_list[rng.randint(len(ones_list))]
            i0 = zeros_list[rng.randint(len(zeros_list))]
        if i1 == i0:
            continue
        h[i1] = 0
        h[i0] = 1
        c_new = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(c_new.max())
        if new_max <= cur_max:
            cur_max = new_max
            c = c_new
            ones_list.remove(i1)
            zeros_list.append(i1)
            zeros_list.remove(i0)
            ones_list.append(i0)
            if cur_max * (2.0 / n) < best_c:
                best_c = cur_max * (2.0 / n)
                best_h = h.copy()
        else:
            delta = new_max - cur_max
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                c = c_new
                ones_list.remove(i1)
                zeros_list.append(i1)
                zeros_list.remove(i0)
                ones_list.append(i0)
            else:
                h[i1] = 1
                h[i0] = 0
        T *= cooling
    return best_h, best_c
