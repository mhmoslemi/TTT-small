import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    d = np.where(ms == (n - 1))[0]
    hd = len(d) > 0
    c_diag = c_base[d[0]] if hd else 0.0
    if hd:
        a[d] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg = a < -1e-14
    pos = a > 1e-14

    Llo, Lhi = unaff, cur_max

            ok = False
        if ok and np.any(neg):
            xlo = max(xlo, float(np.max((L - c_base[neg]) / a[neg])))
        if ok and np.any(pos):
            xhi = min(xhi, float(np.min((L - c_base[pos]) / a[pos])))
        if xlo > xhi + 1e-12:

                r2 = (1.0 + np.sqrt(disc)) * 0.5
                if not (max(xlo, 0) < r1 and min(xhi, 1) > r2):
                    ok = False
        if ok:
            Lhi = L
        else:
            Llo = L
    L = Lhi
    xlo, xhi = 0.0, 1.0
    if np.any(neg):
        xlo = max(xlo, float(np.max((L - c_base[neg]) / a[neg])))
    if np.any(pos):
        xhi = min(xhi, float(np.min((L - c_base[pos]) / a[pos])))
    xlo, xhi = max(xlo, 0.0), min(xhi, 1.0)
    if hd:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if max(xlo, 0) < r1 and min(xhi, 1) > r2:
                if h0 - r1 <= r2 - h0:
                    xhi = min(xhi, r1)
                else:
                    xlo = max(xlo, r2)
            elif xlo >= r1 and xhi <= r2:
                if h0 - r1 <= r2 - h0:
                    xhi = min(xhi, r1)
                else:
                    xlo = max(xlo, r2)
    if xlo > xhi:
        return h0
    return float(np.clip(0.5 * (xlo + xhi), 0.0, 1.0))

def cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                di = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(di) > 0:
                    c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

    xs = np.linspace(0, 2, n_src)
    xd = np.linspace(0, 2, n_dst)
    return project_feasible(np.interp(xd, xs, h_src), n_dst)
