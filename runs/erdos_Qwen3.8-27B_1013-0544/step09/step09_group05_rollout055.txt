            ok = False
        elif ok and len(a_nd) > 0:
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            xl = 0.0
            xh = 1.0
            if neg.any():
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if pos.any():
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if xl > xh + 1e-15:
                ok = False
        if ok and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = max(0.0, float(np.max((T - c_nd[neg]) / a_nd[neg])) if neg.any() else 0.0)
                    xh = min(1.0, float(np.min((T - c_nd[pos]) / a_nd[pos])) if pos.any() else 1.0)
                    if xl > xh + 1e-15:
                        ok = False
                    elif xl >= r1 and xh <= r2:
                        ok = False
                else:
                    if (0.0 >= r1 and 1.0 <= r2) or not (0.0 >= r1 or 1.0 <= r2):
                        ok = False
        Lhi = T if ok else Llo
        Llo = T if not ok else Llo
    T = max(Lhi, float(np.max(c)))
    T = (Llo + Lhi) / 2.0
    xl, xh = 0.0, 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if neg.any():
            xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if pos.any():
            xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if xl < r1:
                xh = min(xh, r1)
            elif xh > r2:
                xl = max(xl, r2)
            else:
                mid = (r1 + r2) / 2.0
                if h0 - r1 <= r2 - h0:
                    xh = min(xh, r1 + 1e-9)
                else:
                    xl = max(xl, r2 - 1e-9)
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if xl > xh + 1e-10:
        return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def coord_descend(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if rng is not None else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter, deadline):
    bh, bc = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, bh, args=(n, alpha), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc:
                bh, bc = nh, nc
        except Exception:
            pass
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n0 = 700
    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Fourier global optimization
    xev = np.linspace(0, 2, 300)
    kev = np.linspace(0, 2, 150)
    for N in [12, 18, 25, 35]:
        if time.time() > deadline - 80:
            break
        c0 = extract_fourier(best_h, len(best_h), N)
        for s in range(12):
            if time.time() > deadline - 80:
                break
            x0 = c0 + (rng.randn(2 * N) * 0.04 / (1.0 + 0.3 * s)) if s > 0 else c0.copy()
            mu = 5.0
            for it in range(25):
                if time.time() > deadline - 80:
                    break
                res = minimize(fourier_obj, x0, args=(N, xev, kev, 150.0, mu),
                              method='L-BFGS-B', options={'maxiter': 200, 'ftol': 1e-14})
                x0 = res.x
                gv = g_from_coeffs(x0, N, xev)
                viol = float(np.max(np.maximum(0.0, np.abs(gv) - 0.5)))
                if viol < 1e-8:
                    break
                mu *= 4.0
                if mu > 1e7:
                    break
            xn = np.linspace(0, 2, n0)
            gn = g_from_coeffs(x0, N, xn)
            ht = project_feasible(0.5 + gn, n0)
            ct = compute_c5(ht, n0)
            if ct < best_c:
                best_h, best_c = ht, ct

    # Phase 2: SLSQP on best
    h2, c2 = slsqp(best_h, n0, [800, 5000, 30000], 300, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Heavy coordinate descent
    h3, c3 = coord_descend(best_h, n0, 250, deadline, rng)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Try different n
    n_cands = [500, 600, 800, 400]
    n_cands = [x for x in n_cands if x != n0]
    x_orig = np.linspace(0, 2, len(best_h))
    for nt in n_cands:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(x_new, x_orig, best_h), nt)
        ho, co = slsqp(hi, nt, [2000, 15000], 250, deadline)
        ho, co = coord_descend(ho, nt, 150, deadline, rng)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = np.linspace(0, 2, nt)

    # Phase 5: Random restarts
    n = len(best_h)
    if time.time() < deadline - 100:
        n_r = max(3, int((deadline - time.time() - 80) / 70))
        for r in range(n_r):
            if time.time() > deadline - 60:
                break
            s = 0.02 + 0.04 * (r / max(1, n_r - 1))
            hp = best_h + rng.normal(0, s, n) if r % 3 != 1 else np.full(n, 0.5) + rng.normal(0, 0.12, n)
            hp = project_feasible(hp, n)
            ho, co = slsqp(hp, n, [3000, 20000], 200, deadline)
            ho, co = coord_descend(ho, n, 80, deadline, rng)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        n = len(best_h)
        rem = deadline - time.time() - 5
        hp = coord_descend(best_h, n, max(30, int(rem / 0.05)), deadline, rng)
        if hp[1] < best_c:
            best_h, best_c = hp
        hp, cp = slsqp(best_h, n, [50000, 200000], 400, deadline)
        if cp < best_c:
            best_h, best_c = hp, cp
        hp, cp = coord_descend(best_h, len(best_h), 40, deadline)
        if cp < best_c:
            best_h, best_c = hp, cp

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
