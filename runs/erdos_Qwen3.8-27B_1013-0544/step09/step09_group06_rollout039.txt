```python
    """Optimize one coordinate at a time using exact 1D minimax."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')

                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            # Find the d that minimizes max(c_peak + d*a_coeff - d^2*delta_diag, unaff_max)
            # Candidate d values where the max changes
            candidates = [lo, hi]
            for i in range(top_k):
```

```python
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_1d_solve(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    """Solve the 1D minimax problem exactly."""
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = 0.0
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def coord_descent(h, n, max_passes, deadline, top_k=10):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]
        c_peak = c[top_idx].astype(float)
        improved = False
        # Randomize order
        order = np.random.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a_coeff = np.zeros(top_k)
            diag_idx = None
            for idx in range(top_k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            best_d, best_val = exact_1d_solve(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def simulated_annealing(h, n, deadline, n_iter=50000):

            current_c = new_c
            if current_c < best_c:
                best_h = h.copy()
                best_c = current_c
    return best_h, best_c
```

```python
def apply_two_moves(c, h, p, q, d1, d2, n):
    k_vals = np.arange(2*n-1)
    new_c = c.copy()
    
    # Move 1: h[p] += d1
    jf1 = p + k_vals - (n-1)
    jb1 = p - k_vals + (n-1)
    mf1 = (jf1 >= 0) & (jf1 < n)
    mb1 = (jb1 >= 0) & (jb1 < n)
    new_c[mf1] += d1 * (1.0 - h[jf1[mf1]])
    new_c[mb1] -= d1 * h[jb1[mb1]]
    new_c[n-1] -= d1*d1
    
    # Move 2: h[q] += d2
    jf2 = q + k_vals - (n-1)
    jb2 = q - k_vals + (n-1)
    mf2 = (jf2 >= 0) & (jf2 < n)
    mb2 = (jb2 >= 0) & (jb2 < n)
    new_c[mf2] += d2 * (1.0 - h[jf2[mf2]])
    new_c[mb2] -= d2 * h[jb2[mb2]]
    new_c[n-1] -= d2*d2
    
    # Cross terms
    k_cross1 = q - p + n - 1
    k_cross2 = p - q + n - 1
    new_c[k_cross1] -= d1 * d2
    new_c[k_cross2] -= d1 * d2
    
    return new_c
```
