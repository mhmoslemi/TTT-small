import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    if h.size == 0:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    hh = np.clip(h - lam, 0.0, 1.0)
    for _ in range(30):
        dd = target - hh.sum()
        if abs(dd) < 1e-13:
            break
        idx = np.flatnonzero((hh > 0.0) & (hh < 1.0))
        if idx.size == 0:
            break
        hh[idx] += dd / idx.size
        np.clip(hh, 0.0, 1.0, out=hh)
    dd = target - hh.sum()
    if abs(dd) > 1e-14:
        if dd > 0:
            idx = int(np.argmax(1.0 - hh))
            hh[idx] = min(1.0, hh[idx] + dd)
        else:
            idx = int(np.argmax(hh))
            hh[idx] = max(0.0, hh[idx] + dd)
    return hh


def weighted_grad(w, h, n, dx):
    L = 2 * n - 1
    off = n - 1
    need = max(8, 3 * n - 2)
    N = 1 << need.bit_length()
    fp = np.fft.rfft(w, N)
    fh = np.fft.rfft(h, N)
    corr = np.fft.irfft(np.conj(fp) * fh, N)
    conv = np.fft.irfft(fp * fh, N)
    ar = np.arange(n)
    idx = (ar - off) % N
    x1 = corr[idx]
    x2 = conv[off:off + n]
    pref = np.empty(L + 1)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    starts = off - ar
    ends = off + n - 1 - ar
    s1 = pref[ends + 1] - pref[starts]
    return dx * (s1 - x1 - x2)


def lbfgs_fun(x, n, alpha, lam_pen, dx, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    t = alpha * (c - m)
    e = np.exp(t)
    z = float(e.sum())
    w = e / z
    soft = m + np.log(z) / alpha
    s = float(x.sum())
    val = dx * soft + 0.5 * lam_pen * (s - target) * (s - target)
    g = weighted_grad(w, x, n, dx)
    if lam_pen:
        g += lam_pen * (s - target)
    return val, g


def refine_lbfgs(h, n, deadline, alphas, maxiter=80, lam_pen=100.0):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() > deadline - 5:
        return best_h, best_c
    dx = 2.0 / n
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        args = (n, float(alpha), float(lam_pen), dx, target)
        try:
            res = minimize(
                lbfgs_fun, best_h, args=args, jac=True, method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': int(maxiter),
                    'maxfun': int(max(10, 2 * maxiter)),
                    'ftol': 1e-15,
                    'gtol': 1e-9,
                    'maxls': 10
                }
            )
        except Exception:
            continue
        if res is None or res.x is None:
            continue
        if not np.all(np.isfinite(res.x)):
            continue
        nh = project_feasible(res.x, n)
        nc = compute_c5(nh, n)
        if nc < best_c - 1e-12:
            best_h = nh
            best_c = nc
    return best_h, best_c


def exact_update_p(h, c, p, n, a, c0, b, lin):
    off = n - 1
    h0 = float(h[p])
    a.fill(0.0)
    s1 = off - p
    a[s1:s1 + n] += 1.0 - h
    s2 = off + p - n + 1
    a[s2:s2 + n] -= h[::-1]
    a[off] = 0.0

    np.copyto(c0, c)
    c0[off] = 0.0
    b[:] = c0
    if h0 != 0.0:
        b -= h0 * a

    cur_max = float(c.max())
    pos = a > 1e-13
    neg = a < -1e-13
    zero = ~(pos | neg)
    zmax = float(c0[zero].max()) if zero.any() else 0.0
    Llo = max(0.0, zmax)
    Lhi = cur_max
    if Lhi <= Llo + 1e-12:
        return h0

    pi = np.nonzero(pos)[0]
    ni = np.nonzero(neg)[0]
    bpos = b[pi]
    apos = a[pi]
    bneg = b[ni]
    aneg = a[ni]

    for _ in range(36):
        if Lhi - Llo < 1e-11:
            break
        L = 0.5 * (Llo + Lhi)
        feas = zmax <= L + 1e-14
        lom = 0.0
        him = 1.0
        if feas and ni.size:
            rn = (L - bneg) / aneg
            if not np.all(np.isfinite(rn)):
                feas = False
            else:
                ot = float(np.max(rn))
                if ot > 1.0 + 1e-13:
                    feas = False
                else:
                    lom = max(0.0, ot)
        if feas and pi.size:
            rp = (L - bpos) / apos
            if not np.all(np.isfinite(rp)):
                feas = False
            else:
                ot = float(np.min(rp))
                if ot < -1e-13:
                    feas = False
                else:
                    him = min(1.0, ot)
        if feas and lom > him + 1e-13:
            feas = False
        if feas and L < 0.25 - 1e-15:
            disc = max(0.0, 1.0 - 4.0 * L)
            sq = np.sqrt(disc)
            r1 = 0.5 * (1.0 - sq)
            r2 = 0.5 * (1.0 + sq)
            if not ((lom <= min(him, r1) + 1e-13) or (max(lom, r2) <= him + 1e-13)):
                feas = False
        if feas:
            Lhi = L
        else:
            Llo = L

    L = min(cur_max, max(Lhi, Llo + 1e-13))
    lo = 0.0
    hi = 1.0
    if ni.size:
        rn = (L - bneg) / aneg
        if np.all(np.isfinite(rn)):
            lo = max(lo, float(np.max(rn)))
    if pi.size:
        rp = (L - bpos) / apos
        if np.all(np.isfinite(rp)):
            hi = min(hi, float(np.min(rp)))
    lo = max(0.0, lo)
    hi = min(1.0, hi)

    cand = [h0, 0.0, 1.0, lo, hi]
    r1 = None
    r2 = None
    if L < 0.25 - 1e-15:
        disc = max(0.0, 1.0 - 4.0 * L)
        sq = np.sqrt(disc)
        r1 = 0.5 * (1.0 - sq)
        r2 = 0.5 * (1.0 + sq)
        cand.append(r1)
        cand.append(r2)
        e1 = min(hi, r1)
        if lo <= e1:
            cand.append(0.5 * (lo + e1))
        e2 = max(lo, r2)
        if e2 <= hi:
            cand.append(0.5 * (e2 + hi))
    if lo <= hi:
        cand.append(0.5 * (lo + hi))

    best_x = h0
    best_val = cur_max
    for xx in cand:
        if xx is None:
            continue
        try:
            xx = float(xx)
        except Exception:
            continue
        if not np.isfinite(xx):
            continue
        if xx < 0.0:
            xx = 0.0
        elif xx > 1.0:
            xx = 1.0
        lin[:] = c0
        lin += (xx - h0) * a
        vv = float(lin.max())
        dv = xx * (1.0 - xx)
        if dv > vv:
            vv = dv
        if vv < best_val:
            best_val = vv
            best_x = xx

    if best_val < cur_max - 1e-9:
        return best_x
    return h0


def exact_cd(h, n, max_passes, deadline, order='seq'):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() > deadline - 10:
        return best_h, best_c

    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    a = np.zeros(L)
    c0 = np.zeros(L)
    b = np.zeros(L)
    lin = np.zeros(L)
    base_idx = np.arange(n)

    for _ in range(int(max_passes)):
        if time.time() > deadline - 18:
            break
        if order == 'sens':
            sens = np.empty(n)
            for p in base_idx:
                seg = c[off - p:off - p + n]
                sens[p] = seg.std()
            order_idx = base_idx[np.argsort(-sens)]
        elif order == 'rand':
            order_idx = base_idx.copy()
            np.random.shuffle(order_idx)
        else:
            order_idx = base_idx

        old_h = h.copy()
        old_c = c.copy()
        old_cabs = float(old_c.max())
        improved = False
        stopped = False
        cnt = 0
        for p in order_idx:
            cnt += 1
            if cnt % 32 == 0 and time.time() > deadline - 12:
                stopped = True
                break
            x = exact_update_p(h, c, p, n, a, c0, b, lin)
            if abs(x - h[p]) > 1e-14:
                hd = x - h[p]
                h[p] = x
                c += hd * a
                c[off] = x * (1.0 - x)
                improved = True
        if stopped:
            break
        if not improved:
            break

        hp = project_feasible(h, n)
        cp = np.correlate(hp, 1.0 - hp, mode='full')
        cp_abs = float(cp.max()) * dx
        if cp_abs < best_c - 1e-12:
            best_h = hp.copy()
            best_c = cp_abs
        if cp_abs > old_cabs * dx + 1e-7:
            break
        h = hp
        c = cp

    return best_h, best_c


def interpolate_h(h_src, n):
    h_src = np.asarray(h_src, dtype=float)
    n0 = len(h_src)
    if n0 == n:
        return h_src.copy()
    xs = np.linspace(0.0, 2.0, n0)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, xs, h_src)


def sym_refl(h):
    return 0.5 * (h + h[::-1])


def sym_anti(h, n):
    if n % 2:
        return None
    m = n // 2
    left = 0.5 * (h[:m] + 1.0 - h[m:2 * m])
    out = np.empty(n)
    out[:m] = left
    out[m:2 * m] = 1.0 - left
    return out


def best_variant(h, n):
    v = interpolate_h(h, n)
    v = project_feasible(v, n)
    bv = v.copy()
    bc = compute_c5(v, n)

    v = sym_refl(v)
    v = project_feasible(v, n)
    c = compute_c5(v, n)
    if c < bc:
        bv = v.copy()
        bc = c

    va = sym_anti(v, n)
    if va is not None:
        va = project_feasible(va, n)
        ca = compute_c5(va, n)
        if ca < bc:
            bv = va.copy()
            bc = ca
    return bv, bc


def polish_n(h, n, deadline, quick=False):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if quick:

        p1 = 10
        p2 = 7
    else:
        alphas = [40, 150, 600, 2000, 8000]
        mi = 110 if n <= 750 else 65
        p1 = 30
        p2 = 12

    if time.time() < deadline - 30:
        h1, c1 = refine_lbfgs(best_h, n, deadline, alphas, mi, 100.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 20:
        h1, c1 = exact_cd(best_h, n, p1, deadline, 'seq')
        if c1 < best_c:
            best_h, best_c = h1, c1

    if (not quick) and time.time() < deadline - 45:
        h1, c1 = refine_lbfgs(best_h, n, deadline, [2000, 8000, 20000], 35, 100.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 18:
        h1, c1 = exact_cd(best_h, n, p2, deadline, 'sens')
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 12:
        h1, c1 = exact_cd(best_h, n, p2, deadline, 'seq')
        if c1 < best_c:

    return best_h, best_c
