    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lfgs_optimize(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_and_grad, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=20, reverse_odd=True):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    order = list(range(n))
    for p_idx in range(max_passes):
        if time.time() > deadline:
            break
        if reverse_odd and p_idx % 2 == 1:
            order = list(reversed(range(n)))
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-13:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B on original n
    t_phase = min(deadline, t0 + budget_s * 0.15)
    h1, c1 = lfgs_optimize(best_h, n, [1000, 5000, 20000, 100000], 600, t_phase)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: CD on original n (bulk of time)
    t_cd_end = min(deadline, t0 + budget_s * 0.45)
    h2, c2 = coord_descent(best_h, n, 200, t_cd_end, gs_iters=20)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Try n=500
    if time.time() < deadline - 120:
        n2 = 500
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h_i = project_feasible(np.interp(x_new, x_old, best_h), n2)
        h3, c3 = lfgs_optimize(h_i, n2, [5000, 50000, 200000], 500, None)
        t_sub = min(deadline, time.time() + budget_s * 0.2)
        h3, c3 = coord_descent(h3, n2, 100, t_sub, gs_iters=22)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Try n=1000
    if time.time() < deadline - 120:
        n3 = 1000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n3)
        h_i = project_feasible(np.interp(x_new, x_old, best_h), n3)
        h4, c4 = lfgs_optimize(h_i, n3, [5000, 50000], 400, None)
        t_sub = min(deadline, time.time() + budget_s * 0.15)
        h4, c4 = coord_descent(h4, n3, 60, t_sub, gs_iters=20)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Restart with structured perturbation
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        for r in range(4):
            if time.time() > deadline - 40:
                break
            if r == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.015, n_cur), n_cur)
            elif r == 1:
                hp = project_feasible(best_h + 0.04 * np.sin(2 * np.pi * 3.0 * x / 2.0), n_cur)
            elif r == 2:
                bp = np.zeros(n_cur)
                blk = n_cur // 7
                for bi in range(0, n_cur, 2 * blk):
                    bp[bi:min(bi + blk, n_cur)] = 0.05
                hp = project_feasible(best_h + bp, n_cur)
            else:
                hp = project_feasible(best_h + 0.03 * np.cos(np.pi * 7.0 * x / 2.0), n_cur)
            h5, c5v = lfgs_optimize(hp, n_cur, [10000, 100000], 400, None)
            t_sub = min(deadline, time.time() + 60)
            h5, c5v = coord_descent(h5, n_cur, 40, t_sub, gs_iters=20)
            if c5v < best_c:
                best_h, best_c = h5, c5v
    # Phase 6: Final high-precision CD
    if time.time() < deadline - 3:
        h6, c6 = coord_descent(best_h, len(best_h), 100, deadline, gs_iters=25)
        if c6 < best_c:
            best_h, best_c = h6, c6
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
