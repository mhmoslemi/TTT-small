    n = h.size
    vh = 1.0 - h
    Ls = np.arange(-(n-1), n)  # length 2n-1
    # compute C[L] = sum_i h[i] vh[i+L]
    c = np.empty(2*n-1)
    for kk, L in enumerate(Ls):
        if L >= 0:
            i0, i1 = 0, n - L        # i in [0, n-L), vh at i+L in [L, n)
            seg = h[i0:i1] * vh[i0+L:i1+L]
        else:
            Lp = -L
            i0, i1 = Lp, n           # i in [Lp, n), vh at i+L = i-Lp in [0, n-Lp)
            seg = h[i0:i1] * vh[i0+L:i1+L]  # i+L ranges [0, n-Lp)
        c[kk] = seg.sum()
    # smooth max
    m = c.max()
    e = np.exp((c - m)/alpha)
    se = e.sum(); s = e/se
    smoothmax = (m + alpha*np.log(se))*dx
    # gradient d(smoothmax)/dh
    g = np.zeros(n)

            g[j0:j1] += w * vh[j0+L:j1+L]      # vh term, j in [0,n-L)

            g[j0:j1] += w * vh[j0+L:j1+L]

def smoothmax_jac(h, alpha, dx):
    n = h.size
    vh = 1.0 - h
    c = np.empty(2*n-1)
    cidx = np.arange(2*n-1) - (n-1)   # L values
    for kk in range(2*n-1):
        L = cidx[kk]
        if L >= 0:
            a0 = 0; a1 = n - L
            c[kk] = (h[a0:a1] * vh[a0+L:a1+L]).sum()
        else:
            Lp = -L
            a0 = Lp; a1 = n
            c[kk] = (h[a0:a1] * vh[a0+L:a1+L]).sum()
    m = c.max()
    e = np.exp((c - m)/alpha)
    se = e.sum()
    s = e/se
    smoothmax = (m + alpha*np.log(se)) * dx
    g = np.zeros(n)
    sdx = s*dx
    for kk in range(2*n-1):
        w = sdx[kk]
        if w <= 0.0: 
            continue
        L = cidx[kk]
        if L >= 0:
            g[:n-L] += w          # base window [0,n-L)
            g[:n-L] -= w*h[L:]    # term2: i in [0,n-L), h[i+L]=h[L:n]

            g[Lp:]    += w          # window [Lp,n)

def solve1d(cvals, coef, diag_i, unaff, lo, hi):
    # value_k(d)=cvals[k]+coef[k]*d + (-d^2 if k==diag_i)
    cands = [lo, hi, unaff if False else lo]
    cands = [lo, hi]
    T=len(cvals)
    for i in range(T):
        for j in range(i+1,T):
            if i==diag_i or j==diag_i: continue
            da = coef[i]-coef[j]
            if abs(da)>1e-15:
                d=(cvals[j]-cvals[i])/da
                if lo<=d<=hi: cands.append(d)
    if diag_i is not None:
        for i in range(T):
            if i==diag_i: continue
            # cvals[i]+coef[i]d = cvals[diag]+coef[diag]d - d^2
            # => d^2 + (coef[i]-coef[diag]) d + (cvals[i]-cvals[diag]) = 0
            A=1.0; B=coef[i]-coef[diag_i]; Cc=cvals[i]-cvals[diag_i]
            disc=B*B-4*Cc
            if disc>=0:
                sq=np.sqrt(disc)
                d1=(-B+sq)/2; d2=(-B-sq)/2
                if lo<=d1<=hi: cands.append(d1)
                if lo<=d2<=hi: cands.append(d2)
    for i in range(T):
        if abs(coef[i])>1e-15:
            d=(unaff-cvals[i])/coef[i]
            if lo<=d<=hi: cands.append(d)
    bestd=lo; bestv=float('inf')
    for d in cands:
        d=float(min(max(d,lo),hi))
        vals=cvals+coef*d
        if diag_i is not None: vals[diag_i]-=d*d
        v=float(max(vals.max(), unaff))
        if v<bestv: bestv=v; bestd=d
    return bestd, bestv
