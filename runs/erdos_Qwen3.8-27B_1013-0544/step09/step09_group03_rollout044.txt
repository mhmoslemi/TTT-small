def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = [lo, hi, (lo + hi) * 0.5, 0.0, 1.0]
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=500, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def sum_eq_fun(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_polish(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def block_expand(block_vals, n, n_blocks):
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * n // n_blocks
        end = (b + 1) * n // n_blocks
        h[start:end] = block_vals[b]
    return project_feasible(h, n)

def block_obj_wrapper(block_vals, n, n_blocks):
    h = block_expand(block_vals, n, n_blocks)
    return compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick CD to establish baseline
    if time.time() < deadline - budget_s * 0.85:
        best_h, best_c = coord_descent(best_h, n, 300, deadline - budget_s * 0.85, seed)

    # Phase 2: Aggressive multi-resolution structural search
    low_res = [24, 32, 40, 48, 64, 80, 96, 128]
    for n_low in low_res:
        if time.time() > deadline - budget_s * 0.60:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_low = np.linspace(0, 2, n_low)
        h_low = project_feasible(np.interp(x_low, x_orig, best_h), n_low)

        # Optimize from parent
        h_a, c_a = coord_descent(h_low, n_low, 300, deadline - budget_s * 0.58, seed + n_low)

        # Try random binary
        h_rb = np.where(rng.rand(n_low) < 0.5, 1.0, 0.0)
        h_rb = project_feasible(h_rb, n_low)
        h_b, c_b = coord_descent(h_rb, n_low, 200, deadline - budget_s * 0.58, seed + 500 + n_low)

        # Try anti-periodic constraint: h[i] + h[i+n/2] = 1
        if n_low % 2 == 0:
            h_anti = np.zeros(n_low)
            h_anti[:n_low//2] = rng.rand(n_low // 2)
            h_anti[n_low//2:] = 1.0 - h_anti[:n_low//2]
            h_anti = project_feasible(h_anti, n_low)
            h_c, c_c = coord_descent(h_anti, n_low, 150, deadline - budget_s * 0.58, seed + 900 + n_low)
        else:
            h_c, c_c = h_a, c_a

        best_local = min((h_a, c_a), (h_b, c_b), (h_c, c_c), key=lambda t: t[1])

        # Check flip
        hf = project_feasible(1.0 - best_local[0], n_low)
        cf = compute_c5(hf, n_low)
        if cf < best_local[1]:
            best_local = (hf, cf)

        # Expand to current resolution
        h_exp = project_feasible(np.interp(x_orig, x_low, best_local[0]), len(best_h))
        h_exp, c_exp = coord_descent(h_exp, len(best_h), 200, deadline - budget_s * 0.55, seed + 1500 + n_low)
        if c_exp < best_c:
            best_h, best_c = h_exp, c_exp

    # Phase 3: Block-structure global search
    n_cur = len(best_h)
    for nb in [6, 8, 10, 12, 16, 20]:
        if time.time() > deadline - budget_s * 0.42:
            break
        bounds = [(0.0, 1.0)] * nb
        try:
            res = differential_evolution(block_obj_wrapper, bounds, args=(n_cur, nb),
                                          maxiter=30, popsize=12, seed=seed + nb,
                                          tol=1e-6, polish=False, init='sobol')
            h_bl = block_expand(res.x, n_cur, nb)
            h_bl, c_bl = coord_descent(h_bl, n_cur, 150, deadline - budget_s * 0.40, seed + 3000 + nb)
            if c_bl < best_c:
                best_h, best_c = h_bl, c_bl
                n_cur = len(best_h)
        except Exception:
            pass

    # Phase 4: SLSQP with various alpha
    if time.time() < deadline - budget_s * 0.30:
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - budget_s * 0.28:
                break
            h_s = slsqp_polish(best_h, n_cur, alpha, 200)
            h_s, c_s = coord_descent(h_s, n_cur, 150, deadline - budget_s * 0.26, seed + 4000 + alpha)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
                n_cur = len(best_h)

    # Phase 5: Diverse restarts
    n_cur = len(best_h)
    for r in range(40):
        if time.time() > deadline - budget_s * 0.12:
            break
        strat = r % 7
        if strat == 0:
            scale = 0.008 + 0.01 * (r // 7)
            h_p = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
        elif strat == 1:
            nb = rng.randint(3, 8)
            bs = n_cur // nb
            idx = rng.randint(0, nb)
            h_p = best_h.copy()
            h_p[idx*bs:(idx+1)*bs] = 1.0 - h_p[idx*bs:(idx+1)*bs]
            h_p = project_feasible(h_p, n_cur)
        elif strat == 2:
            # Multi-scale smoothing + noise
            sz = max(3, n_cur // rng.randint(10, 40))
            h_sm = uniform_filter1d(best_h, size=sz)
            h_p = project_feasible(h_sm + rng.normal(0, 0.03, n_cur), n_cur)
        elif strat == 3:
            # Oscillatory perturbation
            x = np.linspace(0, 2, n_cur)
            freq = rng.uniform(3, 20)
            amp = rng.uniform(0.04, 0.12)
            h_p = project_feasible(best_h + amp * np.sin(np.pi * freq * x + rng.uniform(0, 6.28)), n_cur)
        elif strat == 4:
            # Stochastic binary from parent probabilities
            h_p = np.where(rng.rand(n_cur) < best_h, 1.0, 0.0)
            h_p = project_feasible(h_p, n_cur)
        elif strat == 5:
            # Piecewise constant from random partition
            n_bl = rng.randint(5, 15)
            bounds_pt = np.sort(rng.rand(n_bl - 1))
            vals = rng.rand(n_bl)
            h_p = np.zeros(n_cur)
            edges = np.concatenate([[0], bounds_pt, [1.0]])
            for bi in range(n_bl):
                h_p[int(edges[bi]*n_cur):int(edges[bi+1]*n_cur)] = vals[bi]
            h_p = project_feasible(h_p, n_cur)
        else:
            # Random low-res structure expanded
            n_lr = rng.randint(16, 40)
            h_lr = rng.rand(n_lr)
            h_lr = project_feasible(h_lr, n_lr)
            x_lr = np.linspace(0, 2, n_lr)
            x_cr = np.linspace(0, 2, n_cur)
            h_p = project_feasible(np.interp(x_cr, x_lr, h_lr), n_cur)

        h_o, c_o = coord_descent(h_p, n_cur, 120, deadline - budget_s * 0.10, seed + 10000 + r * 17)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            n_cur = len(best_h)

    # Phase 6: Resolution upgrade
    for n_try in [800, 1000, 1200, 1500]:
        if time.time() > deadline - budget_s * 0.04:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_hi = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_hi, c_hi = coord_descent(h_hi, n_try, 400, deadline - budget_s * 0.03, seed + 50000 + n_try)
        if c_hi < best_c:
            best_h, best_c = h_hi, c_hi

    # Phase 7: Final heavy CD + flip check
    n_fin = len(best_h)
    best_h, best_c = coord_descent(best_h, n_fin, 3000, deadline - 2, seed + 99999)

    h_fl = project_feasible(1.0 - best_h, n_fin)
    c_fl = compute_c5(h_fl, n_fin)
    if c_fl < best_c:
        best_h, best_c = h_fl, c_fl

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
