        b = params[2*(k-1)+1]

    return h

def smooth_obj(params, n, K, alpha):
    """Smooth objective for Fourier parameters."""
    h = fourier_h(params, n, K)
    h = np.clip(h, 0.0, 1.0)
    # Enforce sum constraint softly
    target = n / 2.0
    penalty_alpha = 100.0
    pen = penalty_alpha * (h.sum() - target)**2
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n) + pen

            best_val, best_d = v, d
    return best_d, best_val

def cd_run(h, n, top_k, deadline, order=None):
    h = h.copy()
    if order is None:
        order = list(range(n))
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peaks = c[top_idx].astype(float)
        mask = np.ones(2*n-1, dtype=bool)
        mask[top_idx] = False
        unaff_max = float(c[mask].max()) if mask.any() else 0.0
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0-1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n-1)
                jb = p - m + (n-1)
                af = (1.0-h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n-1:
                    diag_idx = idx
            best_d, best_val = solve_1d(c_peaks, a_coeff, diag_idx, unaff_max, -h0, 1.0-h0)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fourier_to_h(params, n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        h += params[2*(k-1)] * np.cos(np.pi*k*x)
        h += params[2*(k-1)+1] * np.sin(np.pi*k*x)
    return h
