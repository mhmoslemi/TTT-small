def coord_descent(h, n, max_passes, deadline, random_order=False, rng=None):

        improved = False

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m <= 0:
        return 0.0
    shifted = alpha * (c - m)
    sm = shifted.max()
    lse = sm + np.log(np.sum(np.exp(shifted - sm)))
    return (m + lse / alpha) * (2.0 / n)


def prox_grad_step(h, n, alpha, lr, deadline):
    """One proximal gradient step with smooth max."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    weights = np.exp(alpha * (c - m))
    weights /= weights.sum()
    
    # Gradient of c[k] w.r.t. h[p]
    # c[k] = sum_{valid i} h[i]*(1-h[i+k]) where k = idx - (n-1)
    # d c[k] / d h[p] = (1-h[p+k]) [if 0<=p+k<n] - h[p-k] [if 0<=p-k<n]
    #                  + (1-2*h[p]) [if k=0, i.e., p+k=p and p-k=p]
    # Wait, when k=0: c[0] = sum h[i]*(1-h[i]), dc[0]/dh[p] = (1-h[p]) - h[p] = 1-2h[p]
    # For k≠0: dc[k]/dh[p] = (1-h[p+k]) - h[p-k] (with bounds)
    
    # Actually for the full correlation, the "zero lag" is at index n-1.
    # Let me compute the gradient of the weighted sum:
    # grad[p] = sum_k weights[k] * dc[k]/dh[p]
    
    # dc[k]/dh[p] for offset k = idx-(n-1):
    #   = (1-h[p+k]) [if 0<=p+k<n] + (-h[p-k]) [if 0<=p-k<n]
    #   but when k=0, both terms give (1-h[p]) and -h[p], total = 1-2h[p] ✓
    
    grad = np.zeros(n)
    for idx in range(len(c)):
        if weights[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        for p in range(n):
            g = 0.0
            if 0 <= p + k < n:
                g += (1.0 - h[p + k])
            if 0 <= p - k < n:
                g -= h[p - k]
            grad[p] += weights[idx] * g
    # Remove mean to handle the sum constraint
    grad -= grad.mean()
    h = project_feasible(h - lr * grad, n)
    return h
