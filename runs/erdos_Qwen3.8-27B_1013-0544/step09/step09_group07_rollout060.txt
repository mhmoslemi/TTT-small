    best_val = vals[best_i]
    best_d = xs[best_i]
    # Refine with golden section in a neighborhood
    if 0 < best_i < n_grid - 1:
        gl = xs[best_i - 1]
        gh = xs[best_i + 1]
    else:
        gl, gh = lo, hi
    gr = (np.sqrt(5) - 1) / 2.0
    x1 = gh - gr * (gh - gl)
    x2 = gl + gr * (gh - gl)
    f1 = _obj_val(x1, c0, a, di, unaff)
    f2 = _obj_val(x2, c0, a, di, unaff)
    for _ in range(20):
        if f1 < f2:
            gh, x2, f2 = x2, x1, f1
            x1 = gh - gr * (gh - gl)
            f1 = _obj_val(x1, c0, a, di, unaff)
        else:
            gl, x1, f1 = x1, x2, f2
            x2 = gl + gr * (gh - gl)
            f2 = _obj_val(x2, c0, a, di, unaff)
    d_mid = (gl + gh) / 2.0
    v_mid = _obj_val(d_mid, c0, a, di, unaff)
    if v_mid < best_val:
        best_d, best_val = d_mid, v_mid
    return best_d, best_val

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h = np.clip(h + (d / t) * r, 0, 1)
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h = np.clip(h + (d / t) * h, 0, 1)
    return h

def coord_cd(h, n, max_passes=200, deadline=None, top_k=40):
    h = h.copy()
    n1 = n - 1
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, n1 - idx)
            m_hi = min(2 * n - 2, max(n1 + idx, 2 * n - 2 - idx))
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                idxs = np.argsort(c[ms_all])[-top_k:]
                ms = ms_all[np.sort(idxs)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff = float(c[skipped].max()) if skipped else 0.0
            else:
                ms = ms_all
                unaff = 0.0
            a = np.zeros(len(ms))
            jf = idx + ms - n1
            jb = idx - ms + n1
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c0 = c[ms]
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            di = -1
            for ii in range(len(ms)):
                if ms[ii] == n1:
                    di = ii
                    break
            lo, hi = -h0, 1.0 - h0
            best_d, new_val = _solve_1d(c0, a, di, unaff, lo, hi)
            if new_val < cur - 1e-12:
                h[idx] = float(np.clip(h0 + best_d, 0, 1))
                improved = True
                cur = new_val
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def make_sinusoidal(n, freq=2.0, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def make_block(n, frac=0.5):
    h = np.zeros(n)
    h[:int(n * frac)] = 1.0
    return project_feasible(h, n)

def subgradient_step(h, n, lr):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    offset = m_star - (n - 1)
    sg = np.zeros(n)
    for i in range(n):
        j1 = i - offset
        if 0 <= j1 < n:
            sg[i] += (1.0 - h[j1])
        j2 = i + offset
        if 0 <= j2 < n:
            sg[i] -= h[j2]
    return project_feasible(h - lr * sg, n)
