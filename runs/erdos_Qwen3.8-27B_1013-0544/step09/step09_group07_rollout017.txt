import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    v = float(nv.max())
    return v if v > unaff_max else unaff_max

def _golden_wrap(d, c_aff, a_coeff, diag_local, unaff_max):
    return obj1d(d, c_aff, a_coeff, diag_local, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    cands = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        cands.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            cands.append(d)
    best_d, best_v = 0.0, obj1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v:
            best_v, best_d = v, dc
    margin = min((hi - lo) * 0.12, abs(best_d) * 0.35 + (hi - lo) * 0.02)
    gl, gh = max(lo, best_d - margin), min(hi, best_d + margin)
    if gh - gl > 1e-12:
        try:
            res = golden(_golden_wrap, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=40)
            if res.fun < best_v:
                best_d, best_v = float(res.x), float(res.fun)
        except Exception:
            pass
    return best_d

def coord_cd(h, n, max_passes=500, deadline=None, top_k=30, gs_top=20, rng=None):
    h = h.copy()
    no_improve = 0
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(corr.max())
        improved = False
        active_thresh = current_max * 0.999
        active = np.where(corr >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        if len(active) > 25:
            active = active[np.argsort(corr[active])[-25:]]
        subgrad = np.zeros(n)
        for m in active:
            s = int(m) - (n - 1)
            if s >= 0:
                subgrad[:n - s] += (1.0 - h[s:])
                subgrad[s:] -= h[:n - s]
            else:
                ss = -s
                subgrad[ss:] += (1.0 - h[:n - ss])
                subgrad[:n - ss] -= h[ss:]
        subgrad /= max(len(active), 1)
        gs_idx = np.argsort(np.abs(subgrad))[-gs_top:]
        remaining = np.setdiff1d(np.arange(n), gs_idx)
        n_rand = min(top_k - len(gs_idx), len(remaining))
        if n_rand > 0 and rng is not None:
            rand_idx = rng.choice(remaining, size=n_rand, replace=False)
            idxs = np.concatenate([gs_idx, rand_idx])
        else:
            idxs = gs_idx
        for idx in idxs:
            if deadline is not None and time.time() > deadline:
                break
            idx = int(idx)
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_c1_lo = max(0, n - 1 - idx)
            m_c1_hi = min(2 * n - 2, n - 1 + idx)
            m_c2_lo = max(0, idx)
            m_c2_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_c1_lo, m_c2_lo)
            m_hi = max(m_c1_hi, m_c2_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = corr[ms_all]
                top_pos = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_pos)]
                skipped_mask = np.ones(len(ms_all), dtype=bool)
                skipped_mask[np.sort(top_pos)] = False
                unaff_extra = float(corr[ms_all[skipped_mask]].max()) if skipped_mask.any() else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            s_vals = ms - (n - 1)
            a_coeff = np.zeros(len(ms))
            fwd = (idx + s_vals >= 0) & (idx + s_vals < n)
            a_coeff[fwd] += (1.0 - h[idx + s_vals[fwd]])
            bwd = (idx - s_vals >= 0) & (idx - s_vals < n)
            a_coeff[bwd] -= h[idx - s_vals[bwd]]
            c_aff = corr[ms].copy()
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            if m_lo > 0:
                unaff_extra = max(unaff_extra, float(corr[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_extra = max(unaff_extra, float(corr[m_hi + 1:].max()))
            lo, hi = -h0, 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_extra, lo, hi)
            if abs(best_d) > 1e-10:
                new_val = obj1d(best_d, c_aff, a_coeff, diag_local, unaff_extra)
                if new_val < current_max - 1e-13:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        if improved:
            h = project_feasible(h, n)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = alpha * (c - m)
    s = np.clip(s, -60, 60)
    lse = m + np.log(np.exp(s).sum()) / alpha
    return lse * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def sa_perturb(h, n, rng, n_trials=200, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        scale = 0.005 + 0.01 * rng.random()
        hp = project_feasible(h + rng.normal(0, scale, n), n)
        ho, co = coord_cd(hp, n, max_passes=40, deadline=deadline, top_k=20, gs_top=15, rng=rng)
        if co < best_c:
            best_h, best_c = ho, co
            h = best_h.copy()
    return best_h, best_c
