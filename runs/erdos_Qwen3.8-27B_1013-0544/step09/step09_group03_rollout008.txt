    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()

            tot = h.sum()
            if tot < 1e-14:

                    args=(n_de, cd_p), seed=seed + n_de * 10 + trial,
                    popsize=pop, maxiter=mx, tol=1e-7, polish=False,
                    mutation=(0.5, 1.0), recombination=0.8
                )
                h_de = project_feasible(result.x.copy(), n_de)
                c_de, _ = coord_descent(h_de, n_de, 200, deadline - 370, seed + 500 + n_de)
                h_up = upsample(project_feasible(result.x.copy(), n_de), n_de, n_init)
                h_pol, c_pol = coord_descent(h_up, n_init, 400, deadline - 360, seed + 600 + n_de + trial)
                if c_pol < best_c:
                    best_h, best_c = h_pol, c_pol
            except Exception:
                pass

    # Phase 3: Long CD at original resolution
    h3, c3 = coord_descent(best_h, len(best_h), 3000, deadline - 200, seed + 2000)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Structured sine perturbations + CD
    n_cur = len(best_h)
    x_grid = np.linspace(0, 2, n_cur)
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12, 14]:
        if time.time() > deadline - 120:
            break
        for amp in [0.08, 0.15, 0.25, 0.35, 0.45]:
            if time.time() > deadline - 110:
                break
            for phase in [0.0, np.pi / 2]:
                pert = best_h + amp * np.sin(np.pi * freq * x_grid + phase)
                hp = project_feasible(pert, n_cur)
                h_p, c_p = coord_descent(hp, n_cur, 200, deadline - 100, seed + 3000 + freq * 100 + int(amp * 100) + int(phase * 10))
                if c_p < best_c:
                    best_h, best_c = h_p, c_p

    # Phase 5: Reheating - perturb + CD multiple times
    for r in range(8):
        if time.time() > deadline - 80:
            break
        s = 0.01 + 0.01 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = coord_descent(hp, n_cur, 400, deadline - 70, seed + 4000 + r * 13)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 6: Binary-push restart
    if time.time() < deadline - 60:
        thresh = np.where(best_h > 0.5, 1.0, 0.0)
        hb = project_feasible(thresh, n_cur)
        h_b, c_b = coord_descent(hb, n_cur, 500, deadline - 50, seed + 5000)
        if c_b < best_c:
            best_h, best_c = h_b, c_b

    # Phase 7: Final extensive CD
    h7, c7 = coord_descent(best_h, len(best_h), 5000, deadline - 5, seed + 6000)
    if c7 < best_c:
        best_h, best_c = h7, c7

    # Check 1-h symmetry
    n_final = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
