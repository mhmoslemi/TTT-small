def grad_descent_lse(h, n, alpha, steps, deadline, lr=0.005):

                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    global _con_n
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        _con_n = n
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad, args=(n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def grad_step(h, n, alpha, steps, deadline, lr=0.01):
    h = h.copy()
    for s in range(steps):
        if time.time() > deadline:
            break
        g = lse_grad(h, n, alpha)
        f0 = lse_obj(h, n, alpha)
        step = lr
        for _ in range(15):
            h_try = np.clip(h - step * g, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            f1 = lse_obj(h_try, n, alpha)
            if f1 < f0:
                h = h_try
                break
            step *= 0.5
        else:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP with increasing alpha
    h1, c1 = slsqp_opt(h_best, n, [100, 500, 2000, 10000], 300, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Exact coordinate descent
    h2, c2 = exact_cd(h_best, n, 200, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Randomized CD for escape
    h3, c3 = rand_cd(h_best, n, 150, deadline, seed=seed + 7)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Multi-n exploration
    n_list = [500, 700, 900, 1100]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = slsqp_opt(h_int, n_try, [500, 5000], 200, deadline)
        h_o, c_o = exact_cd(h_o, n_try, 120, deadline)
        if c_o < best_c - 1e-9:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 5: Gradient descent refinement
    if time.time() < deadline - 80:
        for alph in [200, 1000, 5000]:
            if time.time() > deadline - 40:
                break
            h_g = grad_step(h_best, n, alph, 200, deadline, lr=0.02)
            h_g, c_g = exact_cd(h_g, n, 100, deadline)
            if c_g < best_c:
                h_best, best_c = h_g, c_g

    # Phase 6: Shake and settle (diverse perturbations)
    for r in range(6):
        if time.time() > deadline - 40:
            break
        s = 0.015 + 0.015 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_opt(hp, n, [1000, 10000], 150, deadline)
        h_o, c_o = exact_cd(h_o, n, 100, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 7: Final CD polish
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, n, 200, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
