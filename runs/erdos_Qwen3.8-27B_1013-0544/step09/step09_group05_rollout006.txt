        elif r % 5 == 2:
            sigma = 0.02 + 0.06 * r / max(1, n_restarts)
            h_p = best_h + rng.normal(0, sigma, n)
        elif r % 5 == 3:
            h_p = fourier_perturb(best_h, n, rng, 3, 0.08)
        else:
            idx = rng.choice(n, size=max(5, n // 15), replace=False)
            h_p = best_h.copy()
            h_p[idx] = rng.uniform(0, 1, len(idx))
        h_p = project_feasible(h_p, n)
        h_o, c_o = coord_descent(h_p, n, 20, t_end, rng, True, 15)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 5: Final polish with higher precision
    if time.time() < deadline - 10:
        best_h, best_c = coord_descent(best_h, n, 80, deadline, rng, False, 28)
    if time.time() < deadline - 3:
        best_h, best_c = coord_descent(best_h, n, 20, deadline, rng, True, 30)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
