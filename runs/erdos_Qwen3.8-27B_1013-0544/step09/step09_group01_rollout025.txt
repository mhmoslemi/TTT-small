                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-12:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    if time.time() > deadline:
        return h, compute_c5(h, n)
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < compute_c5(h, n):
            return new_h, new_c
    except Exception:
        pass
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coarse CD for fast initial improvement
    h1, c1 = coord_descent(best_h, n, 40, deadline, random_order=False, gs_iters=45, seed_val=0)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP with increasing alpha
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > deadline - 200:
            break
        h2, c2 = slsqp_run(best_h, n, alpha, 400, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Fine CD
    h3, c3 = coord_descent(best_h, n, 40, deadline, random_order=False, gs_iters=80, seed_val=1)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Multi-resolution - go up then back
    if time.time() < deadline - 250:
        n2 = n * 2
        x1 = np.linspace(0, 2, n)
        x2 = np.linspace(0, 2, n2)
        h2_init = project_feasible(np.interp(x2, x1, best_h), n2)
        h2, c2 = coord_descent(h2_init, n2, 25, deadline, random_order=False, gs_iters=50, seed_val=2)
        h_back = project_feasible(np.interp(x1, x2, h2), n)
        h_back, c_back = coord_descent(h_back, n, 30, deadline, random_order=False, gs_iters=80, seed_val=3)
        if c_back < best_c:
            best_h, best_c = h_back, c_back

    # Phase 5: Random-order CD to escape local optima
    if time.time() < deadline - 150:
        h5, c5 = coord_descent(best_h, n, 30, deadline, random_order=True, gs_iters=60, seed_val=seed)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: High-alpha SLSQP
    if time.time() < deadline - 100:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 60:
                break
            h6, c6 = slsqp_run(best_h, n, alpha, 500, deadline)
            if c6 < best_c:
                best_h, best_c = h6, c6

    # Phase 7: Diverse restarts
    if time.time() < deadline - 80:
        x = np.linspace(0, 2, n)
        remaining = deadline - time.time() - 50
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            if r % 5 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.015, n), n)
            elif r % 5 == 1:
                freq = 1.0 + 1.3 * r
                hp = project_feasible(best_h + 0.04 * np.sin(2 * np.pi * freq * x / 2.0), n)
            elif r % 5 == 2:
                mask = rng.random(n) < 0.08
                hp = best_h.copy()
                hp[mask] = np.clip(hp[mask] + 0.2 * rng.randn(mask.sum()) - 0.1, 0, 1)
                hp = project_feasible(hp, n)
            elif r % 5 == 3:
                # Anti-symmetric perturbation
                hp = project_feasible(0.85 * best_h + 0.15 * (1.0 - best_h)[::-1], n)
            else:
                # Periodic structure
                period = max(2, n // (3 + r % 4))
                pat = np.tile(np.array([0.7, 0.3]), period)
                pat = pat[:n]
                hp = project_feasible(0.8 * best_h + 0.2 * pat, n)
            h_o, c_o = coord_descent(hp, n, 20, deadline, random_order=(r % 2 == 0), gs_iters=50, seed_val=100+r)
            if time.time() < deadline - 20 and c_o < best_c + 0.001:
                h_o, c_o = slsqp_run(h_o, n, 50000, 200, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 8: Final CD polish
    if time.time() < deadline - 10:
        remaining = deadline - time.time() - 5
        max_passes = max(5, int(remaining / 3.5))
        h_f, c_f = coord_descent(best_h, n, max_passes, deadline, random_order=False, gs_iters=100, seed_val=999)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
