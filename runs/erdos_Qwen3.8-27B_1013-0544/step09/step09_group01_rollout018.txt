    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    n_lags = len(c)
    n_top = max(5, int(top_frac * n_lags))
    for pass_idx in range(max_passes):
        if time.time() > deadline:
            break
        # Find top lags
        top_lags = np.argsort(c)[-n_top:]
        improved = False
        for lag in top_lags:
            m = int(lag - (n-1))  # actual lag
            if m == 0:
                continue
            # Try pairs (i, i+m) or (i+m, i)
            for _ in range(pairs_per_lag):

                    h[j] -= d_opt
                    improved = True
            # Recompute c
            c = np.correlate(h, 1.0-h, mode='full')
        if not improved:
            break
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-15:

            hp = project_feasible(best_h + pert, n_cur)
            hp, cp = coord_descent(hp, n_cur, 60, deadline - 35, gs_iters=50)
            if cp < best_c:
                best_h, best_c = hp, cp

    # Phase 5: Final high-precision CD polish
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 500, deadline - 3, gs_iters=65)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
