            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_step(h, n, steps, lr, momentum, rng, deadline):
    h = h.copy()
    v = np.zeros(n)
    beta = momentum
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        max_c = c.max()
        active = np.where(np.abs(c - max_c) < 1e-10)[0]
        p = active[rng.randint(len(active))]
        grad = np.zeros(n)
        for i in range(n):
            jf = i + p - (n - 1)
            jb = i - p + (n - 1)
            if 0 <= jf < n:
                grad[i] += (1.0 - h[jf])
            if 0 <= jb < n:
                grad[i] -= h[jb]
        if p == n - 1:
            grad += (1.0 - 2.0 * h)
        grad = grad - np.mean(grad)
        v = beta * v - lr * grad
        h = project_feasible(h + v, n)
    return h

def make_fourier_init(n, a1, a2, a3, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1 * np.sin(np.pi * x + phase) + a2 * np.sin(2 * np.pi * x) + a3 * np.sin(3 * np.pi * x + phase)
    return project_feasible(h, n)

def make_structured_init(n, kind):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = np.where(x < 1.0, 1.0, 0.0)
    elif kind == 1:
        h = np.abs(x - 1.0)
    elif kind == 2:
        h = 0.5 + 0.45 * np.sin(np.pi * x)
    elif kind == 3:
        h = 0.5 + 0.45 * np.sin(2 * np.pi * x)
    elif kind == 4:
        h = np.minimum(x, 2.0 - x)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = make_structured_init(700, 2)
        n_init = 700
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Quick CD on initial
    h1, c1 = coord_descent(best_h, n_init, 30, deadline, rng, random_order=True)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: SLSQP warm-start
    if time.time() < deadline - 60:
        h2, c2 = slsqp_run(best_h, n_init, [2000, 20000], 300, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n_init, 50, deadline, rng, random_order=True)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Annealing - structured perturbations + CD
    n = n_init
    x_grid = np.linspace(0, 2, n, endpoint=False)
    n_anneal = max(5, int((deadline - time.time() - 120) / 40))
    for a in range(n_anneal):
        if time.time() > deadline - 80:
            break
        temp = 0.15 * (0.7 ** a)
        freq = rng.randint(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.8) * temp
        pert = amp * np.sin(freq * np.pi * x_grid + phase)
        hp = project_feasible(best_h + pert, n)
        h_a, c_a = coord_descent(hp, n, 40, deadline, rng, random_order=True)
        if c_a < best_c:
            best_h, best_c = h_a, c_a
    # Phase 4: Multi-n exploration
    n_cands = [300, 400, 500, 800, 1000]
    n_cands = [x for x in n_cands if x != n_init and x <= 1000]
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in n_cands:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        max_p_small = 200 if n_try <= 500 else 80
        h_o, c_o = coord_descent(h_int, n_try, max_p_small, deadline, rng, random_order=True)
        if c_o < compute_c5(h_int, n_try):
            # Interpolate back to n_init scale for comparison is not needed; compare directly
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_orig = np.linspace(0, 2, n_try, endpoint=False)
    # Phase 5: Subgradient with momentum for global exploration
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        n_sg = max(3, int((deadline - time.time() - 60) / 30))
        for s in range(n_sg):
            if time.time() > deadline - 40:
                break
            lr = 0.02 * (0.8 ** s)
            steps = max(50, int(n_cur * 2))
            hp = best_h + rng.normal(0, 0.05 * (0.8 ** s), n_cur)
            hp = project_feasible(hp, n_cur)
            h_s, _ = subgrad_step(hp, n_cur, steps, lr, 0.9, rng, deadline)
            h_s, c_s = coord_descent(h_s, n_cur, 30, deadline, rng, random_order=True)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
    # Phase 6: Final heavy CD polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 5
        max_p = max(30, int(remaining / max(0.02, 0.005 * n_cur / 700)))
        hf, cf = coord_descent(best_h, n_cur, max_p, deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 7: One final SLSQP + CD
    if time.time() < deadline - 8:
        n_cur = len(best_h)
        hf, cf = slsqp_run(best_h, n_cur, [50000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, n_cur, 20, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
