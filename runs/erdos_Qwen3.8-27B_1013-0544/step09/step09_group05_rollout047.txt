    c = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(c))
    max_c = c[max_idx]
    s = max_idx - (n - 1)  # shift
    grad = np.zeros(n)
    for i in range(n):
        # forward: i + s in [0, n)
        if 0 <= i + s < n:
            grad[i] += (1.0 - h[i + s])
        # backward: i - s in [0, n)
        if 0 <= i - s < n:
            grad[i] -= h[i - s]
    return max_c, max_idx, grad

    best_h = h.copy()
