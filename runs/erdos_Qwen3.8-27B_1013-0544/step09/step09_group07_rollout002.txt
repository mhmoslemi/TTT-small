import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):

                        if lo < d < hi: candidates.append(d)

def topk_gradient_step(h, n, k=5, lr=0.02):
    """Take a step using combined gradient from top-k correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for mk in top_k:
        s = mk - (n-1)
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                grad[i] += (1.0 - h[j])
            ip = i - s
            if 0 <= ip < n:
                grad[i] -= h[ip]
    return h - lr * grad

def thue_morse_binary(n):
    """Generate Thue-Morse based binary sequence."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = c % 2
    # Adjust to have exactly n//2 ones
    n_ones = tm.sum()
    target = n // 2
    if n_ones > target:
        excess = np.where(tm == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(excess)
        tm[excess[:n_ones-target]] = 0
    elif n_ones < target:
        deficit = np.where(tm == 0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(deficit)
        tm[deficit[:target-n_ones]] = 1
    return tm.astype(float)

def block_flip_move(h, n, block_sizes=(3,5,7,11)):

            if c_try < best_c - 1e-12:

    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter=400, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline: break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c: best_h, best_c = new_h, new_c
        except: pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0: return 0.0
    candidates = [max(lo, min(hi, 0.0))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: candidates.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: candidates.append(d)
    best_d, best_val = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = max(lo, min(hi, d))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15: best_val, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span*0.1, abs(best_d)*0.3 + span*0.03)
        gl, gh = max(lo, best_d-margin), min(hi, best_d+margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_val - 1e-15: best_d, best_val = float(res.x), float(res.fun)
            except: pass
    return best_d

    h = h.copy()
    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    cur_topk = top_k
    stall = 0
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for idx in order:
            if deadline is not None and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12: continue
            m_f_lo = max(0, n-1-idx); m_f_hi = min(2*n-2, 2*n-2-idx)
            m_b_lo = max(0, idx); m_b_hi = min(2*n-2, idx+n-1)
            m_lo, m_hi = min(m_f_lo, m_b_lo), max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi+1)
            if len(ms_all) > cur_topk:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-cur_topk:]
                ms = ms_all[np.sort(top_idx)]
                skipped = [m for m in ms_all if m not in set(top_idx.tolist())]
                unaff_extra = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff_extra = ms_all, 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1); jb = idx - ms + (n-1)
            vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
            if np.any(vf): a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a_coeff[vb] -= h[jb[vb]]

def topk_grad_step(h, n, k=5, lr=0.02):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for mk in top_idx:
        s = int(mk) - (n-1)
        for i in range(n):
            j = i + s
            if 0 <= j < n: grad[i] += (1.0 - h[j])
            ip = i - s
            if 0 <= ip < n: grad[i] -= h[ip]
    return h - lr * grad

def block_flips(h, n, sizes=(3,5,7,11), trials=4, seed=7):
    best_h, best_c = h.copy(), compute_c5(h, n)
    rng = np.random.RandomState(seed)
    for b in sizes:
        for _ in range(trials):
            if n <= b: continue
            start = rng.randint(0, n-b)
            h_try = h.copy()
            h_try[start:start+b] = 1.0 - h_try[start:start+b]
            h_try = project_feasible(h_try, n)
            ct = compute_c5(h_try, n)
            if ct < best_c - 1e-12: best_h, best_c = h_try.copy(), ct
    return best_h, best_c

def thue_morse(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n_ones, target = int(tm.sum()), n // 2
    rng = np.random.RandomState(0)
    if n_ones > target:
        idx = np.where(tm == 1)[0]; rng.shuffle(idx)
        tm[idx[:n_ones-target]] = 0
    elif n_ones < target:
        idx = np.where(tm == 0)[0]; rng.shuffle(idx)
        tm[idx[:target-n_ones]] = 1
    return tm

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max(); shifted = alpha * (c - m)
    mx = shifted.max()
    return (m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha) * (2.0/n)

def sum_con(x, n): return x.sum() - n/2.0
def sum_jac(x, n): return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter=400, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline: break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=[(0.0,1.0)]*n,
                           constraints=[{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}],
                           options={'maxiter':maxiter,'ftol':1e-18})
            nh = project_feasible(res.x, n); nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c
