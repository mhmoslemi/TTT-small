def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqq_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n_init = 600
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP warm-up
    h1, c1 = slsqq_run(best_h, n_init, [2000, 20000], 300, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Heavy coordinate descent
    h2, c2 = coord_descent(best_h, n_init, 80, deadline, rng, True, 70)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Kick-and-polish cycles
    for cycle in range(10):
        if time.time() > deadline - 80:
            break
        sigma = 0.008 + 0.035 * (cycle / 10.0)
        h_k = best_h + rng.normal(0, sigma, n_init)
        h_k = project_feasible(h_k, n_init)
        h_p, c_p = coord_descent(h_k, n_init, 35, deadline, rng, True, 70)
        if c_p < best_c:
            best_h, best_c = h_p, c_p
    # Phase 4: Try different resolutions
    n_cands = [400, 500, 700, 840, 1000, 1200, 1400]
    n_cands = [x for x in n_cands if x != n_init]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_cands:
        if time.time() > deadline - 60:
            break
        x_new = np.linspace(0, 2, n_try)
        h_try = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_try, n_try, 40, deadline, rng, True, 70)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)
    # Phase 5: SLSQP on best + CD polish
    if time.time() < deadline - 30:
        n = len(best_h)
        hf, cf = slsqq_run(best_h, n, [50000, 200000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, n, 40, deadline, rng, False, 50)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 6: Final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        remaining = max(5, deadline - time.time() - 2)
        max_p = max(20, int(remaining / 0.04))
        hf, cf = coord_descent(best_h, n, max_p, deadline, rng, False, 40)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
