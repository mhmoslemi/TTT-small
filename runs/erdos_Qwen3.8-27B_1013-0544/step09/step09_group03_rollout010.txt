def projected_gradient_min(h, n, alpha, max_iters=5000, deadline=None):

    h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.1 / n  # initial step size
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Smooth max gradient (using top few active)
        m = c.max()
        sorted_idx = np.argsort(c)[::-1]
        top_k = min(5, len(c))
        active = sorted_idx[:top_k]
        alpha_c = alpha * (c[active] - m)
        w = np.exp(alpha_c)
        w /= w.sum()
        
        # Compute gradient
        # grad[p] = dx * sum_{k in active} w_k * ((1-h)[p-(k-(n-1))] - h[p+(k-(n-1))])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx, k in enumerate(active):
            lag = k - (n - 1)
            # (1-h)[p-lag] for p in [lag, n-1] if lag > 0, or [0, n-1-lag] if lag < 0
            p_lag = np.arange(max(0, lag), min(n, n + lag))
            # h[p+lag] for p in [0, n-1-lag] if lag >= 0, or [0-lag, n-1] if lag < 0
            p_plus = np.arange(max(0, -lag), min(n, n - lag))
            grad[p_lag] += w[idx] * (1.0 - h)[p_lag - lag]
            grad[p_plus] -= w[idx] * h[p_plus + lag]
        grad *= dx
        
        # Line search
        for _ in range(20):
            h_new = project_feasible(h - eta * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c or c_new < compute_c5(h, n):
                break
            eta *= 0.5
        
        h = h_new
        current_c = compute_c5(h, n)
        if current_c < best_c:
            best_c = current_c
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):

    s_arr = s_arr[s_arr != 0]
    if len(s_arr) == 0: return h0
    ps = p + s_arr; pm = p - s_arr
    a = np.zeros(len(s_arr))
    vp = (ps >= 0) & (ps < n); vm = (pm >= 0) & (pm < n)
    a[vp] += (1.0 - h[ps[vp]]); a[vm] -= h[pm[vm]]
    m = n - 1 + s_arr
    b = c[m] - a * h0
    m_lo = n - 1 + s_arr.min(); m_hi = n - 1 + s_arr.max()
    all_m = np.arange(2 * n - 1)
    unaff = (all_m < m_lo) | (all_m > m_hi)
    M_unaff = float(c[unaff].max()) if unaff.any() else 0.0
    c0_base = c[n-1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14: break
        active = vals >= (am - 1e-13)
        if not active.any(): break
        mns, mxs = float(a[active].min()), float(a[active].max())
        if mns > 1e-12: hi = mid
        elif mxs < -1e-12: lo = mid
        else: break
    cands = np.array([lo, hi, (lo+hi)*0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        x = float(np.clip(x, 0.0, 1.0))
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x*x)
        if v < best_v: best_v, best_x = v, float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy(); best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n); improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10: continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10: h[p] = nv; improved += 1
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c: best_c = nc
        elif improved == 0: break
    return h, best_c

def proj_grad_descent(h, n, alpha, max_iters=3000, deadline=None):
    h = h.copy(); best_c = compute_c5(h, n); dx = 2.0/n; eta = 1.0/n
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        idx = np.argsort(c)[::-1][:8]
        ac = alpha * (c[idx] - m); w = np.exp(ac); w /= w.sum()
        grad = np.zeros(n)
        for ii in range(len(idx)):
            lag = idx[ii] - (n - 1)
            plo = max(0, lag); phi = min(n, n + lag)
            grad[plo:phi] += w[ii] * (1.0 - h)[plo:phi - lag] if lag >= 0 else (1.0 - h)[plo - lag:phi - lag]
            pml = max(0, -lag); phh = min(n, n - lag)
            grad[pml:phh] -= w[ii] * h[pml + lag:phh + lag]
        grad *= dx
        for _ in range(25):
            hn = project_feasible(h - eta * grad, n)
            cn = compute_c5(hn, n)
            if cn < best_c: break
            eta *= 0.5
        else: break
        h = hn
        cn = compute_c5(h, n)
        if cn < best_c: best_c = cn
        if it > 100 and cn >= best_c: best_c = min(best_c, cn)
    return h, best_c

def targeted_swap(h, n, max_iters=80000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy(); best_c = compute_c5(h, n)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(c)); lag = k_max - (n - 1)
        if lag >= 0:
            ov = h[:n-lag] * (1.0 - h[lag:])
            idx_i = np.arange(n - lag)
        else:
            ov = h[-lag:] * (1.0 - h[:n+lag])
            idx_i = np.arange(-lag, n)
        if ov.max() < 0.005: break
        wts = ov ** 3 + 1e-8
        wts /= wts.sum()
        pos = rng.choice(len(wts), p=wts)
        i1 = int(idx_i[pos]); i2 = i1 + abs(lag)
        h[i1], h[i2] = h[i2], h[i1]
        cn = compute_c5(h, n)
        if cn < best_c - 1e-13: best_c = cn
        else: h[i1], h[i2] = h[i2], h[i1]
    return h, best_c

def random_swap(h, n, max_iters=50000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy(); best_c = compute_c5(h, n)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline: break
        i, j = rng.randint(0, n, 2)
        if i == j or abs(h[i] - h[j]) < 0.05: continue
        h[i], h[j] = h[j], h[i]
        cn = compute_c5(h, n)
        if cn < best_c - 1e-13: best_c = cn
        else: h[i], h[j] = h[j], h[i]
    return h, best_c

def structured_init(n, variant=0):
    x = np.arange(n)
    if variant == 0:
        g = np.cos(2 * np.pi * x * x / (2 * n + 1))
    elif variant == 1:
        g = np.sin(2 * np.pi * x / max(n // 3, 1))
    elif variant == 2:
        p = max(int(n * 0.618), 2)
        g = np.sign(np.sin(2 * np.pi * x / p))
    else:
        g = np.zeros(n); g[x % 2] = 1.0
    g -= g.mean()
    return project_feasible(0.5 + 0.5 * g / (np.abs(g).max() + 1e-10), n)

def slsqp_smooth(h, n, alphas, maxit=300):
    best_h = h.copy(); best_c = compute_c5(h, n)
    for alpha in alphas:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            sh = alpha * (c - c.max()); mx = sh.max()
            return (c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0/n)
        cons = [{'type':'eq','fun':lambda x: x.sum()-n/2.0,'jac':lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n,
                          constraints=cons, options={'maxiter':maxit,'ftol':1e-16})
            nh = project_feasible(res.x, n); nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy(); n = len(h0)
    except:
        h0 = np.full(500, 0.5); n = 500
    h0 = project_feasible(h0, n)
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    
    # Phase 1: CD on initial
    if time.time() < deadline - 350:
        h1, c1 = coord_descent(h0, n, 3000, deadline - 340, seed)
        if c1 < best_c: best_h, best_c = h1, c1
    
    # Phase 2: Projected gradient
    if time.time() < deadline - 250:
        h2, c2 = proj_grad_descent(best_h, n, 5000, 2000, deadline - 240)
        if c2 < best_c: best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n, 1000, deadline - 230, seed+1)
        if c2 < best_c: best_h, best_c = h2, c2
    
    # Phase 3: Targeted swap
    if time.time() < deadline - 180:
        h3, c3 = targeted_swap(best_h, n, 80000, deadline - 170, seed+2)
        if c3 < best_c: best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 1500, deadline - 160, seed+3)
        if c3 < best_c: best_h, best_c = h3, c3
    
    # Phase 4: SLSQP smooth
    if time.time() < deadline - 100:
        h4, c4 = slsqp_smooth(best_h, n, [5000, 50000, 200000], 200)
        if c4 < best_c: best_h, best_c = h4, c4
        h4, c4 = coord_descent(best_h, n, 800, deadline - 90, seed+4)
        if c4 < best_c: best_h, best_c = h4, c4
    
    # Phase 5: Random swap exploration
    if time.time() < deadline - 60:
        h5, c5v = random_swap(best_h, n, 50000, deadline - 55, seed+5)
        if c5v < best_c: best_h, best_c = h5, c5v
        h5, c5v = coord_descent(best_h, n, 1000, deadline - 50, seed+6)
        if c5v < best_c: best_h, best_c = h5, c5v
    
    # Phase 6: Try structured inits at same n
    for v in range(3):
        if time.time() > deadline - 40: break
        hs = structured_init(n, v)
        hs, cs = coord_descent(hs, n, 500, deadline - 35, seed+10+v)
        if cs < best_c: best_h, best_c = hs, cs
    
    # Phase 7: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 480, 550, 650, 800, 1000]:
        if time.time() > deadline - 20: break
        x_new = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = coord_descent(hi, n_try, 600, deadline - 15, seed+n_try)
        if co < best_c:
            best_h, best_c = ho, co; x_orig = x_new
    
    # Phase 8: Final CD + flip check
    n_f = len(best_h)
    hf, cf = coord_descent(best_h, n_f, 2000, deadline - 3, seed+99)
    if cf < best_c: best_h, best_c = hf, cf
    hfl = project_feasible(1.0 - best_h, n_f); cfl = compute_c5(hfl, n_f)
    if cfl < best_c: best_h, best_c = hfl, cfl
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
