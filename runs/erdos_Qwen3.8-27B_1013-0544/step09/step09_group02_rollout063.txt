import time

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    cond = u - cssv / np.arange(1, n + 1) > 0
    rho = np.max(np.where(cond)[0]) + 1 if np.any(cond) else 1
    theta = (cssv[rho-1] if rho > 0 else 0) / rho
    return np.maximum(v - theta, 0)

def proj_feasible(h, n):
    """Project h onto {0≤x≤1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Binary search for threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_new = np.clip(h - mid, 0.0, 1.0)
        s = h_new.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    """Compute correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_grad(h, lam, n):

    g = np.zeros(n)

        hi2 = min(n, n - l)
        if lo2 < hi2:

    return g

def cp_minmax(h, n, alpha, beta, max_iter, deadline=None):

        if c_val < best_c:

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0

        unaff = max(unaff, float(c[:mlo].max()))

def fourier_optimize(h, n, n_harm, max_iter, deadline):

    X = np.ones((n, 2 * n_harm + 1))
    for k in range(1, n_harm + 1):

    alpha = 800.0
    def obj(params):
        z = X @ params
        z = np.clip(z, -50, 50)
        h_f = 1.0 / (1.0 + np.exp(-z))
        c = np.correlate(h_f, 1.0 - h_f, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        lse = m + np.log(np.sum(np.exp(shifted - m))) / alpha
        pen = 200.0 * (h_f.sum() - n / 2.0) ** 2
        return (m + lse) * (2.0 / n) + pen
    try:
        res = minimize(obj, params0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
        z = np.clip(X @ res.x, -50, 50)

        return h, compute_c5(h, n)
    c_new = compute_c5(h_new, n)
    if c_new > compute_c5(h, n):
        return h, compute_c5(h, n)
    return h_new, c_new
