def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break

        def obj(x, n_=n, a=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = a * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / a) * (2.0 / n_)
        
        def cons_fun(x, n_=n):
            return x.sum() - n_ / 2.0

            return np.ones(n_)

                           bounds=bounds, constraints=cons,

            if new_c < best_c:

            pass

    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max(h0 + (L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min(h0 + (L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_base[diag_idx[0]]
            b = 1.0 - 2.0 * h0
            disc = b * b - 4.0 * Cq
            if disc >= 0:
                r1 = (b - np.sqrt(disc)) / 2.0
                r2 = (b + np.sqrt(disc)) / 2.0
                lo = max(x_lo, 0.0)
                hi = min(x_hi, 1.0)
                if lo > h0 + r1 and hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max(h0 + (L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min(h0 + (L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_base[diag_idx[0]]
        b = 1.0 - 2.0 * h0
        disc = b * b - 4.0 * Cq
        if disc >= 0:
            r1 = (b - np.sqrt(disc)) / 2.0
            r2 = (b + np.sqrt(disc)) / 2.0
            fb_lo, fb_hi = h0 + r1, h0 + r2
            best_x = h0
            seg_hi = min(x_hi, fb_lo)
            if x_lo <= seg_hi + 1e-14:
                cand = float(np.clip(h0, x_lo, seg_hi))
                best_x = cand
            seg_lo = max(x_lo, fb_hi)
            if seg_lo <= x_hi + 1e-14:
                cand = float(np.clip(h0, seg_lo, x_hi))
                if abs(cand - h0) < abs(best_x - h0):
                    best_x = cand
            return float(np.clip(best_x, 0.0, 1.0))
        else:
            return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                if np.any(vf):
                    c[ms[vf]] += delta * (1.0 - h[jf[vf]])
                if np.any(vb):
                    c[ms[vb]] += delta * (-h[jb[vb]])
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] += (xn - xn * xn) - (h0 - h0 * h0)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
    return h, best_c

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_smooth_obj, best_h, args=(n, alpha),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP polish
    h1, c1 = slsqp_opt(h_best, n, [5000, 50000], 200, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Heavy coordinate descent
    h2, c2 = exact_cd(h_best, n, 200, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Try complement as restart
    n = len(h_best)
    h_comp = project_feasible(1.0 - h_best, n)
    h3, c3 = exact_cd(h_comp, n, 100, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Targeted perturbations
    for r in range(5):
        if time.time() > deadline - 300:
            break
        scale = 0.003 + 0.008 * r
        mask = (h_best > 0.1) & (h_best < 0.9)
        hp = h_best.copy()
        hp[mask] += rng.normal(0, scale, mask.sum())
        hp = project_feasible(hp, n)
        ho, co = exact_cd(hp, n, 80, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 5: Resolution sweep
    for n_try in [500, 600, 800, 1000]:
        if time.time() > deadline - 200:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = slsqp_opt(h_int, n_try, [20000], 100, deadline)
        ho, co = exact_cd(ho, n_try, 100, deadline)
        if co < best_c - 1e-7:
            h_best, best_c = ho, co
            n = n_try

    # Phase 6: Final CD polish
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 150, deadline)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
