    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))
    for _ in range(55):
        if L_hi - L_lo < 1e-16:
            break
        T = (L_lo + L_hi) / 2.0
        ok = True
        if unaff_max > T + 1e-16:
            ok = False
        elif ok and len(a_nd) > 0:
            xl = 0.0
            xh = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if xl > xh + 1e-15:
                ok = False
        if ok and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    xl = 0.0
                    xh = 1.0
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-15:
                        ok = False
                    elif xl >= r1 and xh <= r2:
                        ok = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        ok = False
        if ok:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = list(np.random.permutation(n)) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def aperiodic_obj(hh, n2, alpha):
    h = np.concatenate([hh, 1.0 - hh])
    n = 2 * n2
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_aperiodic(hh, n2, alphas, maxiter_each, deadline):
    best = hh.copy()
    best_c = compute_c5(np.concatenate([best, 1.0 - best]), 2 * n2)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n2
        try:
            res = minimize(aperiodic_obj, best, args=(n2, alpha), method='SLSQP',
                           bounds=bounds, options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_hh = np.clip(res.x, 0.0, 1.0)
            full = np.concatenate([new_hh, 1.0 - new_hh])
            new_c = compute_c5(full, 2 * n2)
            if new_c < best_c:
                best, best_c = new_hh, new_c
        except Exception:
            pass
    full = np.concatenate([best, 1.0 - best])
    return full, best_c

def coord_descent_aperiodic(hh, n2, max_passes, deadline, rng=None):
    best_hh = hh.copy()
    best_c = compute_c5(np.concatenate([best_hh, 1.0 - best_hh]), 2 * n2)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        n = 2 * n2
        h = np.concatenate([best_hh, 1.0 - best_hh])
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = list(np.random.permutation(n2)) if rng is not None else range(n2)
        for p in order:
            if time.time() > deadline:
                break
            h0 = best_hh[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                best_hh[p] = x_new
                best_hh[n2 + p] = 1.0 - x_new if (n2 + p) < len(best_hh) else 1.0 - x_new
                improved = True
                h[p] = x_new
                h[n2 + p] = 1.0 - x_new
                c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    full = np.concatenate([best_hh, 1.0 - best_hh])
    return full, compute_c5(full, 2 * n2)

def fourier_init(n, mode=0, rng=None):
    x = np.linspace(0, 2, n, endpoint=False)
    if mode == 0:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.25 * np.sin(np.pi * x) - 0.15 * np.sin(2 * np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.35 * np.sin(np.pi * x)
    else:
        h = 0.5 * np.ones(n)
    return project_feasible(h, n)

def aperiodic_init(n2, mode=0, rng=None):
    x = np.linspace(0, 1, n2, endpoint=False)
    if mode == 0:
        hh = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    elif mode == 1:
        hh = 0.5 + 0.35 * np.sin(np.pi * x)
    elif mode == 2:
        hh = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    else:
        hh = np.full(n2, 0.5)
    return np.clip(hh, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = fourier_init(700, mode=0)
        n_init = 700

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD polish on initial
    h1, c1 = coord_descent(best_h, n_init, 80, deadline, rng, random_order=True)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP with increasing alpha
    h2, c2 = slsqp_run(best_h, n_init, [200, 2000, 20000, 200000], 400, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    h3, c3 = coord_descent(best_h, n_init, 150, deadline, rng, random_order=True)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 3: Antiperiodic optimization (halved dimensionality)
    if time.time() < deadline - 60:
        n2_try = n_init // 2
        # Try antiperiodic with current best as seed
        hh_seed = best_h[:n2_try].copy()
        full_seed = np.concatenate([hh_seed, 1.0 - hh_seed])
        se_c = compute_c5(full_seed, n_init)
        # Optimize in antiperiodic space
        h_ap, c_ap = slsqp_aperiodic(hh_seed, n2_try, [500, 5000, 50000], 400, deadline)
        if c_ap < best_c:
            best_h, best_c = h_ap, c_ap
        # CD in antiperiodic space
        if c_ap < best_c + 0.001:
            hh_cur = best_h[:n2_try].copy()
            h_ap2, c_ap2 = coord_descent_aperiodic(hh_cur, n2_try, 100, deadline, rng)
            if c_ap2 < best_c:
                best_h, best_c = h_ap2, c_ap2

    # Phase 4: Try different n values
    x_cur = np.linspace(0, 2, len(best_h), endpoint=False)
    n_cands = [200, 400, 500, 800, 1000, 1400, 1600]
    n_cands = [x for x in n_cands if x != n_init]
    for n_try in n_cands:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [2000, 20000], 300, deadline)
        h_opt, c_opt = coord_descent(h_opt, n_try, 120, deadline, rng, random_order=True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 5: Perturbation search
    n = len(best_h)
    for trial in range(40):
        if time.time() > deadline - 40:
            break
        strat = trial % 5
        if strat == 0:
            hp = best_h + rng.normal(0, 0.015, n)
        elif strat == 1:
            hp = best_h + rng.normal(0, 0.04, n)
        elif strat == 2:
            hp = best_h.copy()
            idx = rng.choice(n, size=min(15, n), replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif strat == 3:
            k = rng.randint(2, 8)
            hp = best_h + 0.03 * np.sin(k * np.pi * np.arange(n) / n + rng.uniform(0, 6.28))
        else:
            hp = fourier_init(n, mode=rng.randint(4))
        hp = project_feasible(hp, n)
        h_o, c_o = slsqp_run(hp, n, [5000, 50000], 200, deadline)
        h_o, c_o = coord_descent(h_o, n, 60, deadline, rng, random_order=True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            n = len(best_h)

    # Phase 6: Final heavy CD
    if time.time() < deadline - 5:
        n = len(best_h)
        rem = max(10, int(deadline - time.time() - 3))
        hf, cf = coord_descent(best_h, n, max(30, rem // 3), deadline, rng, random_order=False)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 7: One more SLSQP at very high alpha + CD
    if time.time() < deadline - 3:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [500000, 2000000], 500, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, len(best_h), 20, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
