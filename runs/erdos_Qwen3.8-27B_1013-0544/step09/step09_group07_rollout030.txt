import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0: return 0.0
    cands = [lo, hi, 0.0]
    # Crossings between pairs
    kl = min(k, 50)
    for i in range(kl):
        for j in range(i+1, kl):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: cands.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: cands.append(d)
    best_d, best_v = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v - 1e-15:
            best_v, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        mg = min(span*0.15, abs(best_d)*0.5 + span*0.05)
        gl, gh = max(lo, best_d-mg), min(hi, best_d+mg)
        if gh - gl > 1e-12:
            try:
                r = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if r.fun < best_v - 1e-15: best_d, best_v = float(r.x), float(r.fun)
            except: pass
    return best_d

def coord_cd(h, n, max_passes=500, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        cur_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            m_fl, m_fh = max(0, n-1-idx), min(2*n-2, 2*n-2-idx)
            m_bl, m_bh = max(0, idx), min(2*n-2, idx+n-1)
            m_lo, m_hi = min(m_fl, m_bl), max(m_fh, m_bh)
            ms_all = np.arange(m_lo, m_hi+1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff = ms_all, 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf>=0)&(jf<n)
            vb = (jb>=0)&(jb<n)
            a_coeff[vf] += (1.0-h[jf[vf]])
            a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            if m_lo > 0: unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff = max(unaff, float(c[m_hi+1:].max()))
            dl = None
            for ii in range(len(ms)):
                if ms[ii] == n-1: dl = ii; break
            lo, hi = -h0, 1.0-h0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha*(c-m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted-mx)))
    return (m + lse/alpha)*(2.0/n)

def sum_con(x, n): return x.sum() - n/2.0
def sum_con_jac(x, n): return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline: break
        bounds = [(0.0, 1.0)]*n
        cons = [{'type':'eq', 'fun':sum_con, 'jac':sum_con_jac, 'args':(n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter':maxiter_each, 'ftol':1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

```python
import numpy as np
from scipy.optimize import minimize, golden, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0: return 0.0
    cands = [lo, hi, 0.0]
    kl = min(k, 50)
    for i in range(kl):
        for j in range(i+1, kl):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: cands.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: cands.append(d)
    best_d, best_v = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v - 1e-15: best_v, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        mg = min(span*0.15, abs(best_d)*0.5 + span*0.05)
        gl, gh = max(lo, best_d-mg), min(hi, best_d+mg)
        if gh - gl > 1e-12:
            try:
                r = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if r.fun < best_v - 1e-15: best_d, best_v = float(r.x), float(r.fun)
            except: pass
    return best_d

def coord_cd(h, n, max_passes=500, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        cur_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            m_fl, m_fh = max(0, n-1-idx), min(2*n-2, 2*n-2-idx)
            m_bl, m_bh = max(0, idx), min(2*n-2, idx+n-1)
            m_lo, m_hi = min(m_fl, m_bl), max(m_fh, m_bh)
            ms_all = np.arange(m_lo, m_hi+1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff = ms_all, 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf>=0)&(jf<n)
            vb = (jb>=0)&(jb<n)
            a_coeff[vf] += (1.0-h[jf[vf]])
            a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            if m_lo > 0: unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff = max(unaff, float(c[m_hi+1:].max()))
            dl = None

            lo, hi = -h0, 1.0-h0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha*(c-m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted-mx)))
    return (m + lse/alpha)*(2.0/n)

def sum_con(x, n): return x.sum() - n/2.0
def sum_con_jac(x, n): return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=500, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline: break
        bounds = [(0.0, 1.0)]*n
        cons = [{'type':'eq', 'fun':sum_con, 'jac':sum_con_jac, 'args':(n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter':maxiter_each, 'ftol':1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

        improved_this_iter = False
        for sf in step_fracs:
            h_try = project_feasible(h + sf * (h_lp - h), n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_h, best_c = h_try, c_try
                h = h_try
                improved_this_iter = True
                break
        if not improved_this_iter:
            # Even the LP solution itself might be good (if linearization was accurate)
            h_try = project_feasible(h_lp, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_h, best_c = h_try, c_try
                h = h_try
            else:
                # No improvement, break
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except:
        h_init = np.full(500, 0.5)
        n = 500
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Coordinate descent
    if time.time() < deadline - 500:
        h1, c1 = coord_cd(best_h, n, max_passes=300, deadline=deadline-480, top_k=30)
        if c1 < best_c: best_h, best_c = h1, c1
    
    # Phase 2: LP-based linearized min-max
    if time.time() < deadline - 400:
        h2, c2 = lp_phase(best_h, n, max_iter=200, deadline=deadline-380)
        if c2 < best_c: best_h, best_c = h2, c2
        # Follow up with coordinate descent
        if time.time() < deadline - 340:
            h2b, c2b = coord_cd(best_h, n, max_passes=100, deadline=deadline-320, top_k=35)
            if c2b < best_c: best_h, best_c = h2b, c2b
    
    # Phase 3: SLSQP with gradual alpha
    if time.time() < deadline - 200:
        h3, c3 = slsqp_polish(best_h, n, [50, 100, 300, 1000, 3000], 400, deadline-180)
        if c3 < best_c: best_h, best_c = h3, c3
    
    # Phase 4: More coordinate descent
    if time.time() < deadline - 120:
        h4, c4 = coord_cd(best_h, n, max_passes=400, deadline=deadline-100, top_k=40)
        if c4 < best_c: best_h, best_c = h4, c4
    
    # Phase 5: LP again (refined)
    if time.time() < deadline - 80:
        h5, c5 = lp_phase(best_h, n, max_iter=100, deadline=deadline-60)
        if c5 < best_c: best_h, best_c = h5, c5
        if time.time() < deadline - 40:
            h5b, c5b = coord_cd(best_h, n, max_passes=100, deadline=deadline-30, top_k=40)
            if c5b < best_c: best_h, best_c = h5b, c5b
    
    # Phase 6: Multi-scale
    n_cur = len(best_h)
    x_orig = np.linspace(0, 2, n_cur)
    for n_try in [400, 500, 700, 1000]:
        if time.time() > deadline - 40: break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=100, deadline=deadline-30, top_k=25)
        if cn < best_c:
            best_h, best_c, n_cur = hn, cn, n_try
            x_orig = x_new
    
    # Phase 7: Final SLSQP + CD
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        h7, c7 = slsqp_polish(best_h, n_cur, [1000, 5000, 20000], 400, deadline-10)
        if c7 < best_c: best_h, best_c = h7, c7
    if time.time() < deadline - 3:
        h8, c8 = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline-2, top_k=50)
        if c8 < best_c: best_h, best_c = h8, c8
    
    # Symmetry
    h_flip = project_feasible(1.0-best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur
