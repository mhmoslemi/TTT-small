def smooth_minmax_obj(x, n, alpha, n_lags):
    c = np.correlate(x, 1.0 - x, mode='full')
    # Use soft-max over top lags
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:n_lags]
    mx = top.max()
    lse = mx + np.log(np.sum(np.exp(top - mx))) / n_lags
    return (mse + lse / alpha) * (2.0 / n)

            if tot < 1e-14:

def slsqp_refine(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        def sfun(x):
            return x.sum() - n / 2.0
        def sjac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    # Phase 4: More LSE with higher alphas
    t4 = deadline - 120
    if time.time() < t4:
        h4, c4 = lse_descent(best_h, n, [500, 2000], 30, t4)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4b, c4b = coord_cd(best_h, n, max_passes=100, deadline=t4, top_k=60)
        if c4b < best_c:
            best_h, best_c = h4b, c4b
    # Phase 5: Try different n
    t5 = deadline - 60
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 700, 900, 1200]:
        if time.time() > t5:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=100, deadline=t5, top_k=40)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new
    # Phase 6: Multi-restart
    n_cur = len(best_h)
    t6 = deadline - 30
    for r in range(10):
        if time.time() > t6:
            break
        s = 0.003 + 0.012 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        ho, co = coord_cd(hp, n_cur, max_passes=60, deadline=t6, top_k=50)
        if co < best_c:
            best_h, best_c = ho, co
    # Phase 7: Final LSE + coord
    if time.time() < deadline - 20:
        h7, c7 = lse_descent(best_h, n_cur, [1000, 5000], 20, deadline - 15)
        if c7 < best_c:
            best_h, best_c = h7, c7
    if time.time() < deadline - 5:
        h8, c8 = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline - 3, top_k=70)
        if c8 < best_c:
            best_h, best_c = h8, c8
    # Symmetry
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
