    T = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-10)
                else:
                    x_lo = max(x_lo, r2 - 1e-10)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float((x_lo + x_hi) / 2.0)

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, 10):
        amp = rng.uniform(0.03, 0.12) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.randint(1, 5) * k
        h += amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def perturbed_copy(h, n, rng, strategy, intensity):
    h_p = h.copy()
    if strategy == 0:
        h_p += rng.normal(0, intensity, n)
    elif strategy == 1:
        tails = np.where((h_p < 0.15) | (h_p > 0.85))[0]
        if len(tails) > 0:
            h_p[tails] += rng.normal(0, intensity * 2, len(tails))
    elif strategy == 2:
        idx = rng.permutation(n)[:max(1, n // 10)]
        h_p[idx] = 1.0 - h_p[idx]
    elif strategy == 3:
        i = rng.choice(n)
        j = rng.choice(n)
        h_p[i], h_p[j] = h_p[j], h_p[i]
    else:
        half = n // 2
        idx = rng.permutation(n)
        h_p[idx[:half]] += intensity * 0.5
        h_p[idx[half:]] -= intensity * 0.5
    return project_feasible(h_p, n)

def slsqp_quick(h, n, alpha, maxiter, deadline):
    if time.time() > deadline - 10:
        return h, compute_c5(h, n)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
        return (m + lse / alpha) * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Initial coordinate descent
    h, c = coord_descent(best_h, n_init, 150, deadline, rng, random_order=True)
    if c < best_c:
        best_h, best_c = h, c

    # Phase 2: SLSQP polish with smooth max
    h, c = slsqp_quick(best_h, len(best_h), 3000, 150, deadline)
    if c < best_c:
        best_h, best_c = h, c
    h, c = coord_descent(best_h, len(best_h), 100, deadline, rng)
    if c < best_c:
        best_h, best_c = h, c

    # Phase 3: Iterated perturbation + CD (main improvement phase)
    n = len(best_h)
    for it in range(30):
        if time.time() > deadline - 60:
            break
        strategy = it % 5
        intensity = 0.008 + 0.004 * (it % 8)
        h_pert = perturbed_copy(best_h, n, rng, strategy, intensity)
        max_pass = 60 + 20 * (it % 3)
        h_new, c_new = coord_descent(h_pert, n, max_pass, deadline, rng, random_order=(it % 2 == 0))
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            n = len(best_h)

    # Phase 4: Try different n values
    n_cands = [350, 500, 840, 1000]
    for n_try in n_cands:
        if time.time() > deadline - 120:
            break
        n_cur = len(best_h)
        if n_try == n_cur:
            continue
        x_orig = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_new, c_new = coord_descent(h_int, n_try, 120, deadline, rng, random_order=True)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            # Quick SLSQP polish on new n
            if time.time() < deadline - 30:
                h_tmp, c_tmp = slsqp_quick(best_h, n_try, 5000, 100, deadline)
                if c_tmp < best_c:
                    best_h, best_c = h_tmp, c_tmp
                h_tmp, c_tmp = coord_descent(best_h, n_try, 80, deadline, rng)
                if c_tmp < best_c:
                    best_h, best_c = h_tmp, c_tmp

    # Phase 5: Fourier restarts
    if time.time() < deadline - 80:
        for fi in range(4):
            if time.time() > deadline - 40:
                break
            n_f = 700
            h_f = fourier_init(n_f, seed + 200 + fi * 13)
            h_new, c_new = coord_descent(h_f, n_f, 120, deadline, rng, random_order=True)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 6: Final polish
    n = len(best_h)
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, n, 200, deadline, rng, random_order=False)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 3:
        h_f, c_f = coord_descent(best_h, len(best_h), 50, deadline, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
