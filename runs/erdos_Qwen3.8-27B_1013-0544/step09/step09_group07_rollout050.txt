def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha)

def make_sinusoidal(n, freq=1.0, phase=0.0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.5 * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def make_trapezoid(n, duty=0.5):
    """Trapezoidal wave."""
    x = np.linspace(0, 2, n)
    period = 1.0 / duty  # in units of [0,2]
    t = (x / period) % 1.0
    h = np.where(t < 0.5, 1.0, 0.0)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    def f1d(d):
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        return max(float(nv.max()), unaff_max)
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d_c = dc / da
                    if lo < d_c < hi:
                        candidates.append(d_c)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d_c in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d_c < hi:
                            candidates.append(d_c)
    best_d = 0.0
    best_val = f1d(0.0)
    for d_c in candidates:
        dc = float(np.clip(d_c, lo, hi))
        v = f1d(dc)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(f1d, gl, gh, maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            c_aff = c[ms].copy()
            diag_local = None
            for ii in range(len(ms)):
                m = ms[ii]
                jf = idx + m - (n - 1)
                jb = idx - m + (n - 1)
                if jf == idx:
                    diag_local = ii
                    a_coeff[ii] = 1.0 - 2.0 * h0
                else:
                    if 0 <= jf < n:
                        a_coeff[ii] += (1.0 - h[jf])
                    if 0 <= jb < n:
                        a_coeff[ii] -= h[jb]

            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                nv = c_aff + best_d * a_coeff

                    nv[diag_local] -= best_d * best_d

        current_max = float(c.max())
        improved = False

            if deadline is not None and time.time() > deadline:
                break
            i, j = rng.choice(n, 2, replace=False)
            if h[i] < 1e-12 or h[j] < 1e-12 or h[i] > 1 - 1e-12 or h[j] > 1 - 1e-12:
                continue
            s = h[i] + h[j]
            lo = max(-s, s - 2.0)
            hi = min(2.0 - s, s)
            if hi - lo < 1e-10:
                continue
            def f_pair(d, i=i, j=j, s=s, h_ref=h, n_ref=n):
                h_t = h_ref.copy()
                h_t[i] = (s + d) / 2.0
                h_t[j] = (s - d) / 2.0
                return float(np.max(np.correlate(h_t, 1.0 - h_t, mode='full'))) * (2.0 / n_ref)
            try:

                    h[i] = (s + res.x) / 2.0

                    improved = True

            except Exception:

        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        if not improved:

    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alphas, deadline=None, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
