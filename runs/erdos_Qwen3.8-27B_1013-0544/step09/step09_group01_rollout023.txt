            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_idx = -1
            has_diag = False
            if (n - 1) >= m_lo and (n - 1) <= m_hi:
                has_diag = True
                diag_idx = (n - 1) - m_lo
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = float(c.max())
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                c[ms] += d_opt * a
                if has_diag:
                    c[diag_idx] -= d_opt * d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP (brief)
    phase1_end = t0 + 100
    if time.time() < phase1_end:
        h1, c1 = slsqp_optimize(best_h, n, [500, 5000, 50000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Fast CD forward
    phase2_end = t0 + 400
    if time.time() < phase2_end:
        h2, c2 = fast_cd(best_h, n, 500, phase2_end, gs_iters=45, reverse=False)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Fast CD reverse
    phase3_end = t0 + 600
    if time.time() < phase3_end:
        h3, c3 = fast_cd(best_h, n, 500, phase3_end, gs_iters=50, reverse=True)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-restart with perturbations
    phase4_end = deadline - 60
    n_cur = len(best_h)
    x = np.linspace(0, 2.0, n_cur)
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() > phase4_end:
            break
        if r % 4 == 0:
            s = 0.005 + 0.02 * (r / n_restarts)
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        elif r % 4 == 1:
            freq = 3.0 + 2.0 * r
            amp = 0.02 + 0.02 * (r / n_restarts)
            hp = project_feasible(best_h + amp * np.sin(np.pi * freq * x), n_cur)
        elif r % 4 == 2:
            freq = 5.0 + 3.0 * r
            hp = project_feasible(best_h + 0.015 * np.cos(2.0 * np.pi * freq * x), n_cur)
        else:
            perm = rng.permutation(n_cur)
            hp = best_h.copy()
            delta = 0.01 * rng.uniform(-1, 1, n_cur)
            hp = np.clip(hp + delta, 0, 1)
            hp = project_feasible(hp, n_cur)
        h_o, c_o = fast_cd(hp, n_cur, 200, phase4_end, gs_iters=40, reverse=(r % 2 == 1))
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 5: Final polish with high-iteration CD
    if time.time() < deadline - 3:
        h_f, c_f = fast_cd(best_h, n_cur, 300, deadline - 3, gs_iters=60, reverse=False)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
