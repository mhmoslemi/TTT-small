    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq',
                 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_step(h, n, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c > cmax - 0.02 * cmax
    n_active = active.sum()
    if n_active == 0:
        return h
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - lr * grad, 0.0, 1.0)
    return project_feasible(h, n)

def pair_swap(h, n, attempts, deadline, rng):
    h = h.copy()
    c_cur = compute_c5(h, n)
    for _ in range(attempts):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        if c_new >= c_cur - 1e-14:
            h[i], h[j] = h[j], h[i]
        else:
            c_cur = c_new
    return h, c_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP with multiple smoothness levels
    h1, c1 = slsqp_opt(h_best, n, [5000, 20000, 100000, 500000], 300, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Fixed-order exact CD
    h2, c2 = exact_cd(h_best, n, 500, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Random-order CD to escape local optima
    h3, c3 = exact_cd(h_best, n, 300, deadline, random_order=True, rng=rng)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Subgradient descent + CD polish
    if time.time() < deadline - 200:
        for lr in [0.02, 0.01, 0.005, 0.002]:
            if time.time() > deadline - 100:
                break
            hs = h_best.copy()
            for _ in range(20):
                if time.time() > deadline - 100:
                    break
                hs = subgrad_step(hs, n, lr, deadline)
            ho, co = exact_cd(hs, n, 200, deadline)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 5: Multi-n exploration
    n_list = [700, 550, 800, 450, 900]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = slsqp_opt(h_int, n_try, [10000, 50000], 150, deadline)
        ho, co = exact_cd(ho, n_try, 200, deadline)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    # Phase 6: Shake and settle with diverse noise
    n = len(h_best)
    for s in [0.005, 0.01, 0.015, 0.03, 0.05, 0.02, 0.04]:
        if time.time() > deadline - 40:
            break
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        ho, co = slsqp_opt(hp, n, [20000, 100000], 150, deadline)
        ho, co = exact_cd(ho, n, 200, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 7: Pair swap local search
    if time.time() < deadline - 20:
        h_best, best_c = pair_swap(h_best, n, 3000, deadline, rng)

    # Phase 8: Final polish
    if time.time() < deadline - 5:
        n = len(h_best)
        hf, cf = exact_cd(h_best, n, 300, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
