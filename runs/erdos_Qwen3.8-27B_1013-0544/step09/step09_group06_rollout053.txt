    c_peak = c[top_idx].astype(float)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[top_idx] = False

    n1 = n - 1

    for p in range(n):

2. Add a constraint that the true max ≤ t + ε for some small ε

    # The second part is: for each m, the vector h0 shifted and masked.

    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        # First term: (1-h0[j]) where j = k+m-n1, valid k
        for k in range(n):
            j = k + m - n1
            if 0 <= j < n:
                A[m, k] += (1.0 - h0[j])
            j2 = k - m + n1
            if 0 <= j2 < n:
                A[m, k] += (-h0[j2])
    
    # Constant term: c[m] = sum_k h0[k]*h0[k+m-n1] for valid k
    c = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        s = 0.0
        for k in range(n):
            j = k + m - n1
            if 0 <= j < n:
                s += h0[k] * h0[j]
        c[m] = s

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    constraints.append(cp.A @ (h_var * dx) + c * dx <= t_var)
    # Wait, A @ h * dx componentwise... let me use proper notation
    constraints.append(cp.multiply(A, h_var) @ ... )

    # = A[m,:] @ h + c[m] <= t/dx

    # Let me reformulate: A @ h + c <= t/dx
    # => A @ h - (t/dx) * ones <= -c
    # This is still not standard because t appears with different coefficients.
    
    # Standard form: [A, -ones] @ [h; t/dx] <= -c
    # Let's use s = t/dx as the variable
    s_var = cp.Variable()
    A_ext = np.hstack([A, -np.ones((2*n-1, 1))])
    constraints = [A_ext @ cp.hstack([h_var, s_var]) <= -c,
                   cp.sum(h_var) == n/2.0,
                   h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(s_var), constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.GLOP)
    except:
        prob.solve(solver=cp.CBC)
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1), s_var.value * dx
    return h0, None

max( max_i (c_i + a_i_p * d_p + a_i_q * d_q),  c_diag + ... ,  U )

where the max is over shifts. Each shift gives a PLANE in (d_p, d_q) space (or a paraboloid for the diagonal shift). We want to find the point in the rectangle [d_p_min, d_p_max] × [d_q_min, d_q_max] that minimizes the max.

This is a 2D minimax of planes (and one paraboloid). The optimal point is at:
- A vertex of the rectangle
- An intersection of two planes (and the intersection point is below all other planes)
- An intersection of a plane and the paraboloid
- A point where the paraboloid is tangent to the upper envelope

The number of candidate points is O(k²) for the plane-plane intersections, O(k) for plane-paraboloid, plus 4 corners. For k=30, that's about 450 + 30 + 4 ≈ 484 candidates. Each requires O(k) evaluation. Total: O(k³) ≈ 27K per block update.
