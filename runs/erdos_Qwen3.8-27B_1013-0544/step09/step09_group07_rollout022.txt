def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    """Solve min_d max(unaff_max, max_i(c_aff[i] + d*a_coeff[i] - d² if i==diag_local else 0))"""
    k = len(a_coeff)
    if k == 0:
        return 0.0
    
    def f(d):
        vals = c_aff + d * a_coeff
        if diag_local is not None:
            vals[diag_local] -= d * d
        return max(float(vals.max()), unaff_max)
    
    # Build candidates from pairwise intersections
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    candidates.append((-da + sq) / (2.0 * dd))
                    candidates.append((-da - sq) / (2.0 * dd))
        # Also: intersection with unaff_max (c_aff[i] + d*a_coeff[i] - d²*(diag) = unaff_max)
        if i == diag_local or (diag_local is None and False):
            pass
    # Intersection of each piece with unaff_max

        if v < best_val:

            best_d = d

            m_lo = min(m_f_lo, m_b_lo)

So dc[o]/dh[idx] = (1-h[idx+s]) * I(0≤idx+s<n) - h[idx-s] * I(0≤idx-s<n)

where s = o - (n-1).

Actually, when s=0: c[n-1] = sum_i h[i](1-h[i])
If we change h[idx] by d: the term h[idx](1-h[idx]) becomes (h[idx]+d)(1-h[idx]-d) = h[idx] - h[idx]² + d(1-2h[idx]) - d²

So the change is d(1-2h[idx]) - d².
The linear part d(1-2h[idx]) is captured by the a_coeff (since (1-h[idx]) - h[idx] = 1-2h[idx]).
The quadratic part -d² is the diag_local term.

So the model is: c[o] + d * a_coeff[o] - d² * I(o == n-1)

    return (m + lse / alpha) * (2.0 / n)

    if time.time() < deadline - 20:
        for r in range(4):
            if time.time() > deadline - 15:
                break
            s = 0.002 + 0.005 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            ho, co = coord_cd(hp, n_cur, 100, deadline - 12, top_k=60, adapt_topk=False)
            if co < best_c:
                best_h, best_c = ho, c6
    # Final: CD polish
    if time.time() < deadline - 5:
        h8, c8 = coord_cd(best_h, n_cur, 500, deadline - 3, top_k=100, adapt_topk=False)
        if c8 < best_c:
            best_h, best_c = h8, c8
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
