def proj(z,n):
    z=np.asarray(z,dtype=float)
    z=np.clip(z,0.0,1.0)
    target=0.5*n
    s=z.sum()
    if abs(s-target) < 1e-12:
        return z
    # fast simple correction if works
    diff=target-s
    if abs(diff) <= n*1e-9:
        h=z+diff/n
        if h.min() >= -1e-12 and h.max() <= 1+1e-12:

    for _ in range(45):

        h=np.clip(z+mid,0.0,1.0)
        if h.sum() < target:

import numpy as np
import time, math

def nP2(m):
    x = max(1, int(m))
    return 1 << (x - 1).bit_length()

def proj(z, n):
    z = np.asarray(z, dtype=float)
    z = np.clip(z, 0.0, 1.0)
    target = 0.5 * n
    s = float(z.sum())
    if abs(s - target) < 1e-12:
        return z
    h = z + (target - s) / n
    if h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        h = np.clip(h, 0.0, 1.0)
        if abs(float(h.sum()) - target) < 1e-9:
            return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        h = np.clip(z + mid, 0.0, 1.0)
        if float(h.sum()) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(z + hi, 0.0, 1.0)

def compute_f(h):
    n = len(h)
    L = nP2(2 * n - 1)
    H = np.fft.rfft(h, L)
    P = np.fft.irfft(H * np.conj(H), L)[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    tot = pref[-1]
    idx = n - np.arange(n)
    return tot - pref[:-1] - P, pref[idx] - P

def c5_f(h):
    fL, fR = compute_f(h)
    return float(np.maximum(fL, fR).max()) * 2.0 / len(h)

def c5_direct(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / len(h))

def interaction_H(H, ws, ker, L, n):
    if n <= 1:
        return np.zeros(1)
    ker.fill(0.0)
    ker[:n - 1][::-1] = ws
    ker[n:] = ws
    V = np.fft.rfft(ker, L)
    c = np.fft.irfft(H * V, L)
    return c[n - 1:2 * n - 1]

def _eval_cd(d, fL, fR, bL, bR):
    val = fL[0] + d * bL[0] - d * d
    if fL.size > 1:
        mm = np.maximum(fL[1:] + d * bL[1:], fR[1:] + d * bR[1:]).max()
        if mm > val:
            val = mm
    return float(val)
