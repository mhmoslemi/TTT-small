def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):

    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]

                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            # Golden section search
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                # Update c for affected entries
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size, top_k=20):
    """Take a projected subgradient step on the active lags."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    # Find active lags
    active = np.where(c > c_max - 1e-6 * c_max)[0]
    if len(active) > top_k:
        # Keep only top k
        idx = np.argsort(c[active])[::-1][:top_k]
        active = active[idx]
    
    # Compute gradient of each active lag
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)  # actual shift
        if m == 0:
            grad += 1.0 - 2.0 * h
        else:
            # dc[k]/dh[p] = (1-h[p+m]) for valid p+m, minus h[p-m] for valid p-m
            if m > 0:
                # h[p] appears as first factor when p+m < n: contribution (1-h[p+m])
                valid = np.arange(n - m)
                grad[valid] += (1.0 - h[valid + m])
                # h[p] appears as second factor when p-m >= 0: contribution -h[p-m]
                valid2 = np.arange(m, n)
                grad[valid2] -= h[valid2 - m]
            else:
                m2 = -m
                valid = np.arange(m2, n)
                grad[valid] += (1.0 - h[valid - m2])
                valid2 = np.arange(0, n - m2)
                grad[valid2] -= h[valid2 + m2]
    
    grad /= len(active)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=18, rng=None, perturb_every=0):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for pth in range(max_passes):

        c = np.correlate(h, 1.0 - h, mode='full')

            if time.time() > deadline:

            m_lo = max(0, n - 1 - p)

            ms = np.arange(m_lo, m_hi + 1)

            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))

            d_lo = -h0

def subgradient_line_search(h, n, top_k=15, max_steps=30, deadline=None):
    """Use subgradient direction + line search to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_steps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(best_h, 1.0 - best_h, mode='full')
        c_max = c.max()
        threshold = c_max - 0.02 * c_max
        active = np.where(c >= threshold)[0]
        if len(active) > top_k:
            top_idx = np.argsort(c[active])[::-1][:top_k]

        grad = np.zeros(n)
        for k in active:

            if m == 0:
                grad += 1.0 - 2.0 * best_h
            elif m > 0:
                v1 = np.arange(n - m)
                grad[v1] += (1.0 - best_h[v1 + m])
                v2 = np.arange(m, n)
                grad[v2] -= best_h[v2 - m]
            else:

                grad[v1] += (1.0 - best_h[v1 - m2])
                v2 = np.arange(0, n - m2)
                grad[v2] -= best_h[v2 + m2]
        grad /= len(active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:

        direction = -grad / grad_norm
        # Line search
        best_t = 0.0

        t_vals = np.concatenate([np.array([0.0]), 0.15 / (1.0 + 0.3 * it) * np.array([1.0, 0.5, 0.25, 0.1])])

                best_t = t
        if best_t > 0:
            best_h = project_feasible(best_h + best_t * direction, n)
            best_c = best_f
        else:
            break
    return best_h, best_c
