    shifted = alpha * (c - m)

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=70, shuffle=False):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    order = list(range(n))
    for ppass in range(max_passes):
        if time.time() > deadline:
            break
        if shuffle:
            np.random.shuffle(order)
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
                c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

opt_h = None

def run(seed=42, budget_s=1100, **kwargs):
    global opt_h
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Progressive SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for idx_a, a in enumerate(alphas):
        if time.time() > deadline - 350:
            break
        opt_h = h.copy()
        if idx_a > 0 and idx_a % 3 == 0:
            opt_h = project_feasible(h + rng.normal(0, 0.005, n), n)
        h_new = slsqp_opt(opt_h, n, a, 500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    # Phase 2: CD with sequential order
    if time.time() < deadline - 200:
        h2, c2 = coord_descent(best_h, n, 120, deadline, gs_iters=75, shuffle=False)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: High-alpha SLSQP polish
    if time.time() < deadline - 100:
        opt_h = best_h.copy()
        h3 = slsqp_opt(opt_h, n, 10000000, 1000)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: CD with shuffled order for different convergence path
    if time.time() < deadline - 80:
        h4, c4 = coord_descent(best_h, n, 80, deadline, gs_iters=80, shuffle=True)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Final SLSQP + CD polish
    if time.time() < deadline - 30:
        opt_h = best_h.copy()
        h5 = slsqp_opt(opt_h, n, 20000000, 500)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    if time.time() < deadline - 5:
        h6, c6 = coord_descent(best_h, n, 60, deadline, gs_iters=90, shuffle=False)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
