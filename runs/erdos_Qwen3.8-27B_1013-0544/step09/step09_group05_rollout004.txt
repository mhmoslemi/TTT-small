```python
def compute_lse_grad(h, n, alpha):
    """Compute gradient of smooth max-correlation objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = (c - m) / alpha
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    # Gradient of max term:
    # The smooth max is approximately at the max, so grad ≈ (1/n_lags) * sum of active grads
    # But with LSE weighting:
    g = np.zeros(n)
    for idx in range(len(c)):
        if weights[idx] < 1e-10:
            continue
        s = idx - (n - 1)
        # p ∈ S: 0 ≤ p < n and 0 ≤ p+s < n
        p_lo1 = max(0, -s)
        p_hi1 = min(n-1, n-1-s)
        # p-s ∈ S: 0 ≤ p-s < n and 0 ≤ p < n (second always true)
        # so p-s ∈ [0, n-1] means p ∈ [s, s+n-1], intersect with [0, n-1]
        p_lo2 = max(0, s)
        p_hi2 = min(n-1, s+n-1)
        
        # I(p ∈ S)
        if p_lo1 <= p_hi1:
            g[p_lo1:p_hi1+1] += weights[idx] * 1.0
            # -h[p+s] for p in S
            g[p_lo1:p_hi1+1] -= weights[idx] * h[p_lo1+s:p_hi1+s+1]
        
        # -h[p-s] for p-s ∈ S
        if p_lo2 <= p_hi2:
            g[p_lo2:p_hi2+1] -= weights[idx] * h[p_lo2-s:p_hi2-s+1]
    
    g *= (2.0 / n)  # dx factor
    return g
```

This is a 1D root-finding problem, solvable by bisection or by sorting.

```python
def project_simplex_box(h, target):
    """Project h onto {x : 0≤x≤1, sum(x)=target}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

                        xl2 = max(xl2, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh2 = min(xh2, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    if xl2 > xh2 + 1e-15:
                        feasible = False
                    elif xl2 >= r1 - 1e-14 and xh2 <= r2 + 1e-14:
                        feasible = False
                else:
                    if (0.0 >= r1 and 1.0 <= r2):
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    Cq = T - c_diag + h0 - h0*h0
    disc = 1.0 - 4.0*Cq
    if disc >= 0:
        r1 = (1.0 - np.sqrt(disc)) / 2.0
        r2 = (1.0 + np.sqrt(disc)) / 2.0
        if x_lo < r1:
            x_hi = min(x_hi, r1)
        elif x_hi > r2:
            x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            h_new = solve_coord(c, h, p, n)
            if abs(h_new - h0) > 1e-12:
                delta = h_new - h0
                s_arr = np.arange(-(n-1), n)
                vf = (p + s_arr >= 0) & (p + s_arr < n)
                c[vf] += delta * (1.0 - h[p + s_arr[vf]])
                vb = (p - s_arr >= 0) & (p - s_arr < n)
                c[vb] -= delta * h[p - s_arr[vb]]
                h[p] = h_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(c) * (2.0/n))

def lse_smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_run(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        def sum_fun(x, nn):
            return x.sum() - nn / 2.0
        def sum_jac(x, nn):
            return np.ones(nn)
        try:
            res = minimize(lse_smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=[{'type':'eq','fun':sum_fun,'jac':sum_jac,'args':(n,)}],
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def greedy_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    positions = list(range(n))
    rng.shuffle(positions)
    target = n // 2
    for i in range(n):
        p = positions[i]
        if sum(h) >= target:
            h[p] = 0
        else:
            h[p] = 1
    return project_feasible(h, n)

def rounding_scan(h, n, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(c)
    for _ in range(3):
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            for trial in [0.0, 1.0]:
                if abs(trial - h0) < 1e-14:
                    continue
                delta = trial - h0
                s_arr = np.arange(-(n-1), n)
                vf = (p + s_arr >= 0) & (p + s_arr < n)
                c_new = c.copy()
                c_new[vf] += delta * (1.0 - h[p + s_arr[vf]])
                vb = (p - s_arr >= 0) & (p - s_arr < n)
                c_new[vb] -= delta * h[p - s_arr[vb]]
                if np.max(c_new) < cur_max - 1e-12:
                    s = h.sum()
                    new_sum = s + delta
                    if abs(new_sum - n/2.0) < 0.51:
                        h[p] = trial
                        c = c_new
                        cur_max = np.max(c)
                        improved = True
                        break
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP global optimization
    if time.time() < deadline - 200:
        h1, c1 = slsqp_run(best_h, n, [200, 2000, 20000], 300, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Heavy coordinate descent
    if time.time() < deadline - 100:
        h2, c2 = coord_descent(best_h, n, 200, deadline, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
        best_c = compute_c5(best_h, n)

    # Phase 3: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    n_cands = [n+2, n-2, n+10, n-10, n+20]
    n_cands = [x for x in n_cands if x > 100 and x != n and x % 2 == 0]
    for n_try in n_cands[:3]:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_run(h_int, n_try, [2000, 20000], 200, deadline)
        h_o, c_o = coord_descent(h_o, n_try, 150, deadline, rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            n = n_try
            x_orig = np.linspace(0, 2, n)

    # Phase 4: Rounding scan on best
    if time.time() < deadline - 30:
        h_r, c_r = rounding_scan(best_h, len(best_h), deadline)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 5: Final coordinate descent polish
    if time.time() < deadline - 10:
        n = len(best_h)
        remaining = deadline - time.time() - 3
        max_p = max(30, int(remaining / max(0.05, (35.0*n) * 1e-5)))
        hf, cf = coord_descent(best_h, n, max_p, deadline, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 6: Another SLSQP + CD
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [50000, 200000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_descent(best_h, n, 50, deadline, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
