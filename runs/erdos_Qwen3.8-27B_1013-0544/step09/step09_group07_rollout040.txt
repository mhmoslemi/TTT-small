def proj_exact(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    lse = shifted.max() + np.log(np.sum(np.exp(shifted)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad_fast(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo <= i_hi:
            g[i_lo:i_hi+1] += w[m] * (1.0 - h[i_lo+s:i_hi+s+1])
        i_lo2 = max(0, s)
        i_hi2 = min(n - 1, n - 1 + s)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2+1] += w[m] * (-h[i_lo2-s:i_hi2-s+1])
    obj = (c_max + np.log(w.sum()) / alpha) * (2.0 / n)
    return g, obj

def fista_smoothed(h, n, alpha, lr, n_iter, deadline=None):
    h = h.copy()
    t = 1.0
    y = h.copy()
    g_y, _ = smooth_grad_fast(y, n, alpha)
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        v = y - lr * g_y
        x_new = proj_exact(v, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        y = np.clip(y, -0.5, 1.5)
        h = x_new
        g_y, _ = smooth_grad_fast(y, n, alpha)
        t = t_new
    return h

def solve_1d_fast(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    best_d = 0.0
    best_val = max(float((c_aff + a_coeff * 0.0).max()), unaff_max)
    if diag_local is not None:
        best_val = max(float((c_aff - 0.0).max()), unaff_max)
    nv = c_aff.copy()
    v0 = float(nv.max())
    best_val = max(v0, unaff_max)
    lo_c = min(max(lo, 0.0), hi)
    hi_c = max(min(hi, 1.0), 0.0)
    for d_try in [lo_c, hi_c, 0.0]:
        nv = c_aff + d_try * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d_try * d_try
        v = max(float(nv.max()), unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = d_try
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                def obj_at(d):
                    nv = c_aff + d * a_coeff
                    if diag_local is not None:
                        nv[diag_local] -= d * d
                    return max(float(nv.max()), unaff_max)
                res = golden(obj_at, gl, gh, maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                skipped = set(ms_all.tolist()) - set(ms.tolist())
                unaff_extra = float(c[list(skipped)].max()) if skipped else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d_fast(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                nv = c_aff + best_d * a_coeff
                if diag_local is not None:
                    nv[diag_local] -= best_d * best_d
                new_val = max(float(nv.max()), unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = proj_exact(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def make_alternating(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    return proj_exact(h, n)

def make_sinusoidal(n, freq=1.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x)
    return proj_exact(h, n)

def make_fourier_init(n, coeffs_c, coeffs_d):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k, (c, s) in enumerate(zip(coeffs_c, coeffs_d), 1):
        h += c * np.cos(np.pi * k * x) + s * np.sin(np.pi * k * x)
    return proj_exact(h, n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_phase(h, n, alphas, maxiter, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = proj_exact(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = proj_exact(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick coordinate descent
    t1 = deadline - 400
    if time.time() < t1:
        h1, c1 = coord_cd(best_h, n, max_passes=150, deadline=t1, top_k=30)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: FISTA with smoothed objective (key innovation)
    t2 = deadline - 200
    if time.time() < t2:
        for alpha in [200, 500, 1500, 5000, 15000]:
            if time.time() > t2:
                break
            lr = 0.8 / (alpha * 2.5)
            h_f = fista_smoothed(best_h, n, alpha, lr, 3000, t2)
            h_f = proj_exact(h_f, n)
            c_f = compute_c5(h_f, n)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Phase 3: SLSQP polish
    t3 = deadline - 150
    if time.time() < t3:
        h3, c3 = slsqp_phase(best_h, n, [500, 2000, 10000], 300, t3)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: More coordinate descent
    t4 = deadline - 100
    if time.time() < t4:
        h4, c4 = coord_cd(best_h, n, max_passes=400, deadline=t4, top_k=35)
        if c4 < best_c:
            best_h, best_c = h4, c3 if False else c4

    # Phase 5: FISTA again with higher alpha
    t5 = deadline - 80
    if time.time() < t5:
        for alpha in [20000, 50000, 200000]:
            if time.time() > t5:
                break
            lr = 0.8 / (alpha * 2.5)
            h_f = fista_smoothed(best_h, n, alpha, lr, 2000, t5)
            h_f = proj_exact(h_f, n)
            c_f = compute_c5(h_f, n)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Phase 6: Try different n values
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [400, 800, 1000]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = proj_exact(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=80, deadline=deadline - 60, top_k=25)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 7: Alternative starts
    if time.time() < deadline - 60:
        for n_alt in [n, 500, 300]:
            if time.time() > deadline - 40:
                break
            h_alt = make_alternating(n_alt)
            ha, ca = coord_cd(h_alt, n_alt, max_passes=80, deadline=deadline - 30, top_k=25)
            if ca < best_c:
                best_h, best_c = ha, ca
            if time.time() > deadline - 25:
                break
            h_sin = make_sinusoidal(n_alt, 1.5)
            hs, cs = coord_cd(h_sin, n_alt, max_passes=60, deadline=deadline - 20, top_k=25)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 8: Fourier-based initialization
    if time.time() < deadline - 50:
        for n_f in [n, 500]:
            if time.time() > deadline - 35:
                break
            for (cc, cd) in [([0.15, 0.05, 0.02], [0.1, 0.04, 0.01]),
                            ([0.1, 0.08], [0.12, 0.03]),
                            ([0.2, 0.03], [0.05, 0.15])]:
                if time.time() > deadline - 30:
                    break
                h_f = make_fourier_init(n_f, cc, cd)
                hf_, cf_ = coord_cd(h_f, n_f, max_passes=60, deadline=deadline - 25, top_k=25)
                if cf_ < best_c:
                    best_h, best_c = hf_, cf_

    # Phase 9: Multi-restart with noise
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        for r in range(4):
            if time.time() > deadline - 25:
                break
            s = 0.002 + 0.01 * r
            hp = proj_exact(best_h + rng.normal(0, s, n_cur), n_cur)
            # Quick FISTA then coord
            lr = 0.8 / (500 * 2.5)
            h_f = fista_smoothed(hp, n_cur, 500, lr, 500, deadline - 20)
            h_f = proj_exact(h_f, n_cur)
            ho, co = coord_cd(h_f, n_cur, max_passes=100, deadline=deadline - 15, top_k=30)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 10: Final SLSQP + coord
    if time.time() < deadline - 20:
        h7, c7 = slsqp_phase(best_h, n_cur, [10000, 50000], 400, deadline - 15)
        if c7 < best_c:
            best_h, best_c = h7, c7
    if time.time() < deadline - 5:
        h8, c8 = coord_cd(best_h, n_cur, max_passes=500, deadline=deadline - 3, top_k=40)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Try 1-h symmetry
    n_fin = len(best_h)
    h_flip = proj_exact(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = proj_exact(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
