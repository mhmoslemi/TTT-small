def exact_cd(h, n, max_passes, deadline, n_bisect=35):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for pas in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n, n_bisect)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if h.sum() != n / 2.0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def fourier_init(n, rng, K=8, scale=0.3):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = scale / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * k * x / 1.0 + phase)
    h = project_feasible(h, n)
    return h

def sinusoidal_perturb(h, n, rng, freq_range=(2, 15), scale=0.05):
    x = np.linspace(0, 2, n)
    f = rng.uniform(freq_range[0], freq_range[1])
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(scale * 0.5, scale)
    pert = h + amp * np.sin(np.pi * f * x + phase)
    return project_feasible(pert, n)

def gaussian_perturb(h, n, rng, scale=0.02):
    pert = h + rng.normal(0, scale, n)
    return project_feasible(pert, n)

def swap_perturb(h, n, rng, n_swaps=5):
    h = h.copy()
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return project_feasible(h, n)

def subgrad_step(h, n, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    active = c > cm - 0.05 * cm
    n_active = active.sum()
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        mask = active[ms]
        if mask.any():
            grad[p] = np.sum(contrib[mask]) / mask.sum()
    h = np.clip(h - lr * grad, 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_best)
    except Exception:
        h_best = np.full(350, 0.5)
        n_parent = 350
    h_best = project_feasible(h_best, n_parent)
    best_c = compute_c5(h_best, n_parent)
    best_h = h_best.copy()
    best_n = n_parent

    # Phase 1: CD on parent
    h1, c1 = exact_cd(h_best, n_parent, 200, min(deadline, t0 + budget_s * 0.25), 35)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, n_parent

    # Phase 2: Try multiple n values with diverse inits
    n_list = [200, 250, 300, 350, 400, 450, 500, 550, 600, 650]
    n_list = [x for x in n_list if x != best_n]
    n_list.sort(key=lambda x: abs(x - best_n))
    n_list = n_list[:6]

    x_fine = np.linspace(0, 2, len(best_h))
    for n_try in n_list:
        if time.time() > deadline - 60:
            break
        time_per_n = (deadline - time.time()) / max(1, (len(n_list) - n_list.index(n_try)))
        n_bis = max(20, min(40, int(n_try * 0.06)))
        # Init by interpolation
        x_new = np.linspace(0, 2, n_try)
        h_init = project_feasible(np.interp(x_new, x_fine, best_h), n_try)
        h_o, c_o = exact_cd(h_init, n_try, 200, min(deadline, t0 + time.time() + time_per_n * 0.7), n_bis)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_try
        # Also try a Fourier init for this n
        if time.time() < deadline - 60:
            h_f = fourier_init(n_try, rng, K=6, scale=0.25)
            h_f2, c_f2 = exact_cd(h_f, n_try, 150, min(deadline, t0 + time.time() + time_per_n * 0.3), n_bis)
            if c_f2 < best_c:
                best_h, best_c, best_n = h_f2, c_f2, n_try

    # Phase 3: Perturbation + CD from best
    n = best_n
    perturbations = [
        ("gauss1", lambda: gaussian_perturb(best_h, n, rng, 0.01)),
        ("gauss2", lambda: gaussian_perturb(best_h, n, rng, 0.03)),
        ("sin1", lambda: sinusoidal_perturb(best_h, n, rng, (3, 10), 0.04)),
        ("sin2", lambda: sinusoidal_perturb(best_h, n, rng, (5, 20), 0.06)),
        ("swap1", lambda: swap_perturb(best_h, n, rng, 8)),
        ("gauss3", lambda: gaussian_perturb(best_h, n, rng, 0.005)),
    ]
    for pname, pfunc in perturbations:
        if time.time() > deadline - 30:
            break
        h_p = pfunc()
        # Subgradient warm-up
        h_pg = subgrad_step(h_p, n, 0.003, deadline)
        h_o, c_o = exact_cd(h_pg, n, 100, min(deadline, t0 + budget_s - 20), 30)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n

    # Phase 4: Final polish
    if time.time() < deadline - 5:
        n = best_n
        h_f, c_f = exact_cd(best_h, n, 300, deadline, 30)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
