def block_cd(h, n, block_size, top_k, deadline, max_passes=200):
    """Block coordinate descent: update block_size coordinates at a time."""
    h = h.copy()
    rng = np.random.RandomState(42)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[top_idx] = False
        unaff = float(c[mask].max()) if mask.any() else 0.0
        improved = False
        # Random blocks
        for _b in range(n // block_size + 1):
            if time.time() > deadline:
                break
            start = rng.randint(0, n - block_size + 1)
            block = list(range(start, start + block_size))
            # Compute gradient w.r.t. each variable in block
            h_block = h[block].copy()
            # Use Nelder-Mead on the block
            def block_obj(hb):
                h_new = h.copy()
                for i, p in enumerate(block):
                    h_new[p] = hb[i]
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                return max(float(c_new[top_idx].max()), unaff)
            
            res = minimize(block_obj, h_block, method='Nelder-Mead',
                          options={'maxiter': 50, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < cur_max - 1e-12:
                for i, p in enumerate(block):
                    h[p] = float(np.clip(res.x[i], 0.0, 1.0))
                h = project_feasible(h, n)
                cur_max = res.fun
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

For a symmetric tridiagonal (or higher-band) matrix, the eigenvalues can be negative.

Actually, S_d (with 1s on d-th superdiagonal) is NOT symmetric. S_d + S_d^T is symmetric with 1s on the d-th super and sub diagonals. For d=1, this is the adjacency matrix of a path graph, which has eigenvalues 2cos(πj/(n+1)) for j=0..n, some of which are negative.

So B_λ might NOT be PSD, meaning the inner problem might be non-convex!

This complicates things. Let me abandon the QP approach and stick with direct optimization.

**Revised plan**: Focus on what's most likely to give the small improvement needed:

1. Start from initial_h_values
2. Do intensive single-coordinate CD (proven to work well)
3. Add a "nudging" phase: randomly perturb pairs of coordinates in opposite directions (preserving the sum) and see if CD from the perturbed point gives a better result
4. Use different top_k values aggressively
5. Try multiple n values

The key difference from before: I'll use a more systematic "pair perturbation" approach and more aggressive CD.

Let me also think about the "pair perturbation" more carefully:
- Pick two random indices p, q
- Change h[p] += ε, h[q] -= ε (preserving sum)
- Clip to [0,1]
- Run a few rounds of CD
- If improved, keep; otherwise, revert

This is essentially a "2-opt" type local search applied to our problem.

Let me also add: "structured perturbation" where I perturb a contiguous block of coordinates in a smooth pattern (like a Gaussian bump), which might be more effective than random point perturbations for a function that should be somewhat smooth.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, n, top_k, deadline, max_passes=500):

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())

        top_idx = np.argsort(c)[::-1][:k]

        mask[top_idx] = False
        unaff = float(c[mask].max()) if mask.any() else 0.0
        improved = False

                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue

            xr = np.linspace(max(lo, best_d - eps), min(hi, best_d + eps), 50)
            for d in xr:
                v = c_peak + d * a_coeff
                if diag_idx is not None:
                    v[diag_idx] -= d * d
                val = max(float(v.max()), unaff)
                if val < best_v:
                    best_v, best_d = val, d
            if best_v < cur_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved, cur_max = True, best_v
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - x
            tot = room.sum()
            if tot > 1e-14:
                x = x + (diff / tot) * room
        else:
            tot = x.sum()
            if tot > 1e-14:
                x = x + (diff / tot) * x
    x = np.clip(x, 0, 1)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

def smooth_lbfgs(h, n, alpha, deadline, maxiter=300):
    def obj(x):
        return smooth_obj(np.asarray(x, dtype=float), n, alpha)
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 5000})
    except Exception:
        return h, compute_c5(h, n)
    new_h = project_feasible(res.x, n)
    return new_h, compute_c5(new_h, n)

def pair_perturb_cd(h, n, eps, rng, top_k, deadline, max_tries=50):
    h = h.copy()
    for _ in range(max_tries):
        if time.time() > deadline:
            break
        p = rng.randint(0, n)
        q = rng.randint(0, n)
        if p == q:
            continue
        hp = h.copy()
        hp[p] += eps
        hp[q] -= eps
        if hp[p] < 0 or hp[p] > 1 or hp[q] < 0 or hp[q] > 1:
            continue
        hp = project_feasible(hp, n)
        hc, cc = cd_optimize(hp, n, top_k, deadline, 50)
        if cc < compute_c5(h, n) - 1e-14:
            h = hc
    return h, compute_c5(h, n)

def bump_perturb_cd(h, n, width, amplitude, rng, top_k, deadline, max_tries=30):
    h = h.copy()
    base_c = compute_c5(h, n)
    for _ in range(max_tries):
        if time.time() > deadline:
            break
        center = rng.randint(0, n)
        x = np.arange(n)
        bump = amplitude * np.exp(-0.5 * ((x - center) / max(width, 1)) ** 2)
        hp = h + bump
        hp = project_feasible(hp, n)
        hc, cc = cd_optimize(hp, n, top_k, deadline, 80)
        if cc < base_c - 1e-14:
            h = hc
            base_c = cc
    return h, compute_c5(h, n)
