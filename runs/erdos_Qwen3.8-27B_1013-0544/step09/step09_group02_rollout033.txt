   import numpy as np
   from scipy.optimize import minimize

   def compute_c5(h, n):
       dx = 2.0 / n
       return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

   def project_feasible(h, n):
       target = n / 2.0
       h = np.clip(h.copy(), 0.0, 1.0)
       for _ in range(200):
           diff = target - h.sum()
           if abs(diff) < 1e-14:
               break
           if diff > 0:
               room = 1.0 - h
               tot = room.sum()
               if tot < 1e-15:
                   break
               h += (diff / tot) * room
           else:
               tot = h.sum()
               if tot < 1e-15:
                   break
               h += (diff / tot) * h
       return np.clip(h, 0.0, 1.0)

   def exact_coord_delta(h, p, c, n):
       h0 = h[p]
       if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
           return 0.0
       mlo = max(0, n - 1 - p)
       mhi = min(2 * n - 2, p + n - 1)
       ms = np.arange(mlo, mhi + 1)
       jf = p + ms - (n - 1)
       jb = p - ms + (n - 1)
       vf = (jf >= 0) & (jf < n)
       vb = (jb >= 0) & (jb < n)
       vec = np.zeros(len(ms))
       if np.any(vf):
           vec[vf] += 1.0 - h[jf[vf]]
       if np.any(vb):
           vec[vb] -= h[jb[vb]]
       c_sub = c[mlo:mhi + 1]
       unaff = 0.0
       if mlo > 0:
           unaff = max(unaff, float(c[:mlo].max()))
       if mhi < 2 * n - 2:
           unaff = max(unaff, float(c[mhi + 1:].max()))
       lo, hi = -h0, 1.0 - h0
       for _ in range(45):
           mid = (lo + hi) * 0.5
           vals = c_sub + mid * vec
           mval = float(vals.max())
           if mval <= unaff + 1e-14:
               lo = hi = mid
               break
           active = vals >= mval - 1e-10
           vmin = float(vec[active].min())
           vmax = float(vec[active].max())
           if vmin > 1e-10:
               hi = mid
           elif vmax < -1e-10:
               lo = mid
           else:
               lo = hi = mid
               break
       return (lo + hi) * 0.5
