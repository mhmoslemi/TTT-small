import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_lo = max(0 - p, p - (n - 1))
    s_hi = min(n - 1 - p, p)
    s = np.arange(s_lo, s_hi + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s[0]
    m_max = n - 1 + s[-1]
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else -1.0
    c0 = c[n - 1]
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        c0_val = c0 - h0 * (1.0 - h0) + mid * (1.0 - mid)
        if am >= M_unaff and am >= c0_val:
            active = vals >= (am - 1e-13)
            if not active.any():
                break
            if float(a[active].min()) > 1e-12:
                hi = mid
            elif float(a[active].max()) < -1e-12:
                lo = mid
            else:
                break
        elif c0_val > M_unaff + 1e-13:
            lo = mid if mid < 0.5 else hi
            if mid < 0.5:
                lo = mid
            else:
                hi = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0, 0.5])
    best_x, best_v = h0, float('inf')
    for x in cands:
        x = float(np.clip(x, 0.0, 1.0))
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0 - h0 * (1 - h0) + x * (1 - x))
        if v < best_v:
            best_v, best_x = v, x
    return best_x

def coord_descent(h, n, max_passes=2000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = get_corr(h)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            nv = exact_coord_min(c, p, h, n)
            if abs(nv - h0) > 1e-10:
                h[p] = nv
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def binary_swap_eval(h, n, i, j):
    """Evaluate change in max correlation for swapping h[i] 1->0, h[j] 0->1."""
    c = get_corr(h)
    c_max = c.max()
    # Try the swap
    h[i], h[j] = h[j], h[i]
    new_c = get_corr(h)
    new_max = float(new_c.max())
    h[i], h[j] = h[j], h[i]
    return new_max * (2.0 / n)

def binary_local_search(h, n, max_iters=20000, deadline=None, seed=0, anneal=True):

    h = np.where(h > 0.5, 1.0, 0.0)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()

    T = 0.01
    stall = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        delta = new_c - best_c
        accept = False
        if delta < -1e-15:
            accept = True
        elif anneal and T > 1e-6:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            best_c = new_c
            best_h = h.copy()
            ones = [x for x in ones if x != i]
            ones.append(j)
            zeros = [x for x in zeros if x != j]
            zeros.append(i)
            stall = 0
            T *= 0.999
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
            T *= 0.9995
        if stall > 500:
            stall = 0
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = get_corr(x)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = get_corr(x)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for idx in np.where(weights > 1e-10)[0]:
        w = weights[idx]
        k = idx - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            mask1 = np.arange(n) < (n - k)
            mask2 = np.arange(n) >= k
            grad[mask1] += w * (1.0 - x[mask1 + k])
            grad[mask2] -= w * x[mask2 - k]
        else:
            ka = -k
            mask1 = np.arange(n) < (n - ka)
            mask2 = np.arange(n) >= ka
            grad[mask1] += w * (1.0 - x[mask1 + ka])
            grad[mask2] -= w * x[mask2 - ka]
    return grad * dx

def slsqp_refine(h, n, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, 5000), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x.copy(), n)
    except Exception:
        return h

        if co < best_c:
            best_h, best_c = ho, co
        # Quick binary polish on best
        if time.time() < deadline - 15:
            hp2, cp2 = binary_local_search(best_h, n_cur, 10000, deadline - 12,
                                           seed + 500 + r * 7, anneal=True)
            if cp2 < best_c:
                best_h, best_c = hp2, cp2

    # Phase 6: Try alternate n
    n_cur = len(best_h)
    x_old = np.linspace(0, 2, n_cur)
    for n_try in [n_cur - 20, n_cur + 20, n_cur + 50]:
        if time.time() > deadline - 40:
            break
        if n_try < 100:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        to = time.time() + min(deadline - time.time() - 15, 60)
        ho, co = coord_descent(h_int, n_try, 500, to, seed + n_try)
        if co < best_c:
            best_h, best_c = ho, co

    # Final CD polish
    if time.time() < deadline - 3:
        n_f = len(best_h)
        hf, cf = coord_descent(best_h, n_f, 2000, deadline - 2, seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Check 1-h symmetry
    n_f = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_f)
    c_flip = compute_c5(h_flip, n_f)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
