def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):

        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)

                improved = True

        h = project_feasible(h, n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        
        def smooth_obj(x, n=n, alpha=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        
        def sum_con(x, n=n):
            return x.sum() - n / 2.0
        def sum_con_jac(x, n=n):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def flatten_top_obj(x, n, gamma, tol):
    """Minimize max + gamma * variance of values near max"""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    top = c[c >= m - tol]
    if len(top) > 1:
        return (m + gamma * (top.max() - top.min())) * (2.0 / n)
    return m * (2.0 / n)

For a periodic signal with period p, autocorr[d] is periodic in d with period p.

We want to maximize min_d autocorr[d].

By Parseval: sum_d autocorr[d] = sum_i g[i]² (the total energy).
So the average of autocorr[d] (over valid d) is related to the energy.

To maximize the minimum, we want the autocorrelation to be as "flat" as possible (all values equal to the average).

This is the "Golay sequence" / "Barker sequence" problem in disguise! We want a sequence with flat autocorrelation.

For real-valued sequences with zero mean and bounded amplitude, the optimal solution has autocorrelation as close to constant as possible.

**Key insight for the algorithm**: Instead of minimizing the max of c[k], we can equivalently:
1. Work with g = h - 0.5
2. Compute the autocorrelation of g
3. Try to make it as flat as possible (minimize the variance, or minimize max - min)

This suggests a different optimization objective: minimize variance of autocorrelation, or minimize (max_autocorr - min_autocorr).

Let me implement this. The advantage is that this objective is smoother and might be easier to optimize.

Actually, let me combine both: minimize max(c) + λ * variance_of_top_lags.

Or better: since C5 = 0.5 - min_d autocorr[d] (in the continuous case, approximately), we want to maximize min_d autocorr[d]. This is equivalent to minimizing -min_d autocorr[d] = min_d (-autocorr[d]).

A smooth approximation: minimize log(sum_d exp(-β*autocorr[d])) / β (this is a smooth min).

Or equivalently: minimize log(sum_d exp(β*c_k[d])) / β (smooth max of c).

These are the same thing (just a sign change and offset). So the log-sum-exp approach in the previous code is already optimal in this sense.

1. **Better SLSQP parameterization**: Use a wider range of alphas, and in some cases, use a "top-k focus" objective.

2. **Top-k focus objective**: Instead of log-sum-exp over all lags, focus on the top k lags:
   obj = (1/k) * sum_{top k} c[k] * dx
   This is a linear objective in c, which is a quadratic objective in h. We can solve it with a standard QP or just gradient descent.

3. **Iterative reweighting**: 
   - Start with uniform weights on all lags
   - Solve the weighted quadratic
   - Update weights: p_k ∝ exp(β * c_k)
   - Repeat
   
   This is the "exponential weights" or "follow-the-regularized-leader" algorithm for min-max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = c_base[diag_idx[0]] if has_diag else 0.0
    if has_diag:
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    L_lo = unaff_max
    L_hi = float(np.max(c))
    for _ in range(52):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-16:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-15:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl, xh = 0.0, 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    if xl > max(xh, 0.0) + 1e-15 or (max(xl,0) >= r1 and min(xh,1) <= r2):
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo, x_hi = 0.0, 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def reweighting_cd(h, n, deadline, beta, n_outer, n_inner):
    h = h.copy()
    for it in range(n_outer):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_shift = c - c.max()
        p = np.exp(beta * c_shift)
        p /= p.sum()
        for _ in range(n_inner):
            if time.time() > deadline:
                break
            improved = False
            for pidx in range(n):
                if time.time() > deadline:
                    break
                h0 = h[pidx]
                if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                    continue
                lo_a = max(0, n - 1 - pidx)
                hi_a = min(2 * n - 2, n - 1)
                lo_b = max(0, pidx)
                hi_b = min(2 * n - 2, pidx + n - 1)
                m_lo = min(lo_a, lo_b)
                m_hi = max(hi_a, hi_b)
                ms = np.arange(m_lo, m_hi + 1)
                jf = pidx + ms - (n - 1)
                jb = pidx - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                pq = p[ms]
                diag = ms == (n - 1)
                a_lin = pq * a
                quad_coef = float(np.sum(pq[diag])) if np.any(diag) else 0.0
                lin_coef = float(np.sum(a_lin))
                if quad_coef > 1e-14:
                    x_new = -lin_coef / (2.0 * quad_coef)
                    x_new = float(np.clip(x_new, 0.0, 1.0))
                else:
                    if abs(lin_coef) < 1e-14:
                        x_new = h0
                    else:
                        x_new = 0.0 if lin_coef > 0 else 1.0
                if abs(x_new - h0) > 1e-12:
                    h[pidx] = x_new
                    improved = True
            h = project_feasible(h, n)
            if not improved:
                break
    return h, compute_c5(h, n)

def _smooth_obj(x, n, alpha, top_k):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        shifted = alpha * (c_top - c_top.max())
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        val = (c_top.max() + lse / alpha) * (2.0 / n)
    else:
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        val = (m + lse / alpha) * (2.0 / n)
    return val

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha_list, top_k_list, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    for i in range(len(alpha_list)):
        if time.time() > deadline - 5:
            break
        alpha = alpha_list[i]
        tk = top_k_list[i] if i < len(top_k_list) else 0
        for _attempt in range(2):
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(_smooth_obj, best_h, args=(n, alpha, tk),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_each, 'ftol': 1e-14})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c
