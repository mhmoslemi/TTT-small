    """Solve min_d max(m with c_peak + d*a_coeff - d^2*1[m=diag], unaff_max) using golden section"""
    def obj(d):
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        return max(float(vals.max()), unaff_max)
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    while b - a > tol:
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj(d_pt)
    return (a + b) / 2, obj((a + b) / 2)

    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = len(c)

def pnorm_penalty_grad(x, n, p, lam):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = len(c)
    # Gradient of (sum(c^p)/M)^(1/p) = (p-1)/p * (sum(c^p)/M)^(1/p - 1) * (p/M) * c^(p-1)
    # = (p-1)/p * (sum(c^p)/M)^((1-p)/p) * c^(p-1)
    # For large p, this is approximately: c^(p-1) / (max(c)^(p-1)) * (p-1)/p * max(c)
    # But let's just compute it directly

    sum_p = np.sum(cn ** p)
    if sum_p < 1e-30:
        base_grad = np.zeros(M)
    else:
        factor = (p - 1.0) / p * (sum_p / M) ** ((1.0 - p) / p) / M * p
        # factor * cn^(p-1)
        base_grad = factor * np.power(cn, p - 1)
    # Now compute grad wrt h
    offsets = np.arange(M)
    grad = np.zeros(n)

        grad[j] = dx * val
    # Penalty gradient
    s = x.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    v = float(vals.max())
    if unaff_max > v:
        v = unaff_max
    return v

        return lo, _obj1d(lo, c_peak, a_coeff, diag_idx, unaff_max)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc1 = _obj1d(c1, c_peak, a_coeff, diag_idx, unaff_max)

    w = base * np.power(cn, p - 1)

    obj = (sp / M) ** (1.0 / p) + lam * s * s

        if vb.any():

            if cc < best_c:
                best_h, best_c = hc, cc

    # Phase 6: Different n values
    x_orig = np.linspace(0, 2, len(best_h))
    n_list = [400, 500, 600, 700, 800]
    for nt in n_list:
        if time.time() > deadline - 40:
            break
        if nt == len(best_h):
            continue
        xn = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(xn, x_orig, best_h), nt)
        ho, co = cd_opt(hi, nt, 100, deadline - 40, 30)
        ho, co = try_flip(ho, nt)
        if co < best_c:
            best_h, best_c = ho, co

    # Phase 7: Final polish
    n_fin = len(best_h)
    if time.time() < deadline - 20:
        h2, c2 = cd_opt(best_h, n_fin, 1000, deadline - 20, n_fin * n_fin)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 2:
        h2, c2 = cd_opt(best_h, n_fin, 500, deadline, n_fin * n_fin)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h, best_c = try_flip(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
