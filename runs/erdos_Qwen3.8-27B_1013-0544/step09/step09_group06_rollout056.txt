import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(y, n):
    t = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(y - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c
    am = ac.max()
    ea = np.exp(ac - am)
    ws = ea.sum()
    w = ea / ws
    val = dx * (am + np.log(ws)) / alpha
    grad = np.empty(n)
    for p in range(n):
        s1 = np.dot(w[:p+1][::-1], 1.0 - h[:p+1])
        s2 = np.dot(w[:n-p], h[p:])
        grad[p] = dx * (s1 - s2)
    return val, grad

def pg_optimize(h, n, alpha, steps, lr, deadline):
    h = h.copy().astype(float)
    for _ in range(steps):
        if time.time() > deadline:
            break
        val, grad = smooth_obj_grad(h, n, alpha)
        h = project_box_sum(h - lr * grad, n)
    return h, compute_c5(h, n)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    cands = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    cands.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    cands.append(d1)
                if lo <= d2 <= hi:
                    cands.append(d2)
        dd = a_coeff[diag_idx]
        disc = dd * dd - 4.0 * (unaff_max - c_peak[diag_idx])
        if disc >= 0:
            sq = np.sqrt(disc)
            d1 = (dd + sq) / 2.0
            d2 = (dd - sq) / 2.0
            if lo <= d1 <= hi:
                cands.append(d1)
            if lo <= d2 <= hi:
                cands.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                cands.append(d)
    best_d = lo
    best_v = float('inf')
    for d in cands:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_v:
            best_v = v
            best_d = d
    return best_d, best_v

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        k_eff = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k_eff]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            ac = np.zeros(k_eff)
            di = None
            for idx in range(k_eff):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                ac[idx] = af + ab
                if m == n - 1:
                    di = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            um = float(c[mask].max()) if mask.any() else 0.0
            bd, bv = solve_1d_minimax(c_peak, ac, di, um, -h0, 1.0 - h0)
            if bv < cur - 1e-14:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                improved = True
                cur = bv
        if not improved:
            break
        h = project_box_sum(h, n)
    return h, compute_c5(h, n)
