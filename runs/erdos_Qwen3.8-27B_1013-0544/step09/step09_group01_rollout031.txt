import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def get_worst_lag_grad(h, n):
    """Get gradient of the worst bilinear form at h"""
    c = np.correlate(h, 1.0 - h, mode='full')
    worst_idx = c.argmax()
    m = worst_idx - (n - 1)  # the lag
    
    # Gradient of B(m) = sum_i h[i]*(1-h[i+m]) w.r.t. h
    # = for each i: (1-h[i+m]) if moving h[i] up, -h[i-m] if moving h[i] up (from the backward part)
    g = np.zeros(n)
    if m >= 0:
        # B(m) = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # d/dh[j] = (1-h[j+m]) for j <= n-1-m, plus -h[j-m] for j >= m
        forward = np.arange(0, n - m)
        g[forward] += (1.0 - h[forward + m])
        backward = np.arange(m, n)
        g[backward] -= h[backward - m]
    else:
        m_abs = -m
        # B(-m) = sum_{i=m}^{n-1} h[i]*(1-h[i-m]) = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        forward = np.arange(m_abs, n)
        g[forward] += (1.0 - h[forward - m_abs])
        backward = np.arange(0, n - m_abs)
        g[backward] -= h[backward + m_abs]
    return g, worst_idx, c

def subgradient_step(h, n, step, top_k=5):
    """Take a subgradient step to reduce the max correlation"""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k worst lags
    top_indices = np.argsort(c)[-top_k:]
    # Average gradient over top-k lags
    g = np.zeros(n)
    for idx in top_indices:
        m = idx - (n - 1)
        if m >= 0:
            forward = np.arange(0, n - m)
            g[forward] += (1.0 - h[forward + m])
            backward = np.arange(m, n)
            g[backward] -= h[backward - m]
        else:
            m_abs = -m
            forward = np.arange(m_abs, n)
            g[forward] += (1.0 - h[forward - m_abs])
            backward = np.arange(0, n - m_abs)
            g[backward] -= h[backward + m_abs]
    g /= len(top_indices)
    # Remove mean to preserve sum constraint
    g -= g.mean()
    h_new = h - step * g
    return project_feasible(h_new, n)

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lag_gradient(h, n, m):
    g = np.zeros(n)
    if m == 0:
        g = 1.0 - 2.0 * h
    elif m > 0:
        fwd = np.arange(0, n - m)
        g[fwd] += (1.0 - h[fwd + m])
        bwd = np.arange(m, n)
        g[bwd] -= h[bwd - m]
    else:
        ma = -m
        fwd = np.arange(ma, n)
        g[fwd] += (1.0 - h[fwd - ma])
        bwd = np.arange(0, n - ma)
        g[bwd] -= h[bwd + ma]
    return g

def subgradient_optimize(h, n, max_iters, deadline, top_k=8):
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    step = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_full = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(c_full)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            g += lag_gradient(h_cur, n, idx - (n - 1))
        g /= len(top_idx)
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        step_eff = step / g_norm
        h_new = project_feasible(h_cur - step_eff * g, n)
        new_c = compute_c5(h_new, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            h_cur = h_new
            step = min(step * 1.02, 0.5)
        else:
            step *= 0.6
            if step < 1e-6:
                break
    return h_cur, best_c

def slp_cvxpy(h, n, max_iters, deadline, top_m=15):
    import cvxpy as cp
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    dx = 2.0 / n
    target_sum = n / 2.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_lin = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_val = c_lin * dx
        cur_max = float(c_val.max())
        if cur_max < best_c:
            best_c = cur_max
        top_lags = np.argsort(c_lin)[-top_m:]
        x = cp.Variable(n)
        constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
        for idx in top_lags:
            a = lag_gradient(h_cur, n, idx - (n - 1))
            target = cur_max - 0.001
            b_const = (target - float(c_val[idx])) / dx + float(a @ h_cur)
            constraints.append(a @ x <= b_const)
        obj = cp.Minimize(cp.sum_squares(x - h_cur))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-9)
            if x.value is not None:
                h_new = project_feasible(np.array(x.value).flatten(), n)
                new_c = compute_c5(h_new, n)
                if new_c < best_c:
                    best_c = new_c
                    h_cur = h_new
        except Exception:
            break
    return h_cur, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=60):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def mirror_descent_explore(h, n, max_iters, deadline, top_k=6):
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    step = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_full = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(c_full)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            g += lag_gradient(h_cur, n, idx - (n - 1))
        g /= len(top_idx)
        h_raw = h_cur * np.exp(-step * g)
        h_raw = h_raw / h_raw.sum() * (n / 2.0)
        h_new = project_feasible(h_raw, n)
        new_c = compute_c5(h_new, n)
        if new_c < best_c:
            best_c = new_c
            h_cur = h_new
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.7
            if step < 1e-5:
                break
    return h_cur, best_c
