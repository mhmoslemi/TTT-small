import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def compute_subgradient(h, n, top_frac=0.2):
    """Compute subgradient of the max overlap objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    thresh = m * (1.0 - top_frac)
    weights = np.clip(c - thresh, 0.0, None)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = (c == m).astype(float)
        wsum = weights.sum()
    weights /= wsum
    
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wval = weights[n - 1 + d]
        if wval < 1e-14:
            continue
        # First term: (1-h[p+d]) for valid p
        p_lo = max(0, -d)
        p_hi = min(n, n - d)
        if p_lo < p_hi:
            grad[p_lo:p_hi] += wval * (1.0 - h[p_lo + d:p_hi + d])
        # Second term: -h[p-d] for valid p (except d=0)
        if d != 0:
            p_lo2 = max(0, d)
            p_hi2 = min(n, n + d)
            if p_lo2 < p_hi2:
                grad[p_lo2:p_hi2] -= wval * h[p_lo2 - d:p_hi2 - d]
        else:
            # d=0: need (1-2h[p]) total, already added (1-h[p]), need -h[p]
            grad[p_lo:p_hi] -= wval * h[p_lo:p_hi]
    return grad, c, m

def subgrad_descent(h, n, max_steps, deadline, initial_step=0.05, decay=0.999):
    """Subgradient descent with adaptive step size."""
    h = h.copy()
    step = initial_step
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for s in range(max_steps):
        if time.time() > deadline:
            break
        grad, c, m = compute_subgradient(h, n, top_frac=0.15)
        
        # Adaptive step: use Barzilai-Borwein or simple decay
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            # Keep step or increase slightly
            step = min(step * 1.02, 0.1)
        else:
            step *= decay
        
        h = h_new
    return best_h, best_c

def subgrad_descent(h, n, n_steps, deadline, step0=0.04, top_frac=0.12):
    h = h.copy()
    step = step0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(n_steps):
        if time.time() > deadline:
            break
        grad, c, m = subgrad_compute(h, n, top_frac)
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            step = min(step * 1.015, 0.1)
        else:
            step *= 0.97
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Exact CD to local convergence
    h1, c1 = exact_cd(h_best, n, 80, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Subgradient exploration + CD cycles
    for rnd in range(12):
        if time.time() > deadline - 250:
            break
        tf = 0.08 + 0.03 * (rnd % 3)
        h_sg, c_sg = subgrad_descent(h_best, n, 150, min(deadline - 100, time.time() + 40), step0=0.03, top_frac=tf)
        h_cd, c_cd = exact_cd(h_sg, n, 40, deadline)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd
        h_cd2, c_cd2 = exact_cd(h_best, n, 15, deadline)
        if c_cd2 < best_c:
            h_best, best_c = h_cd2, c_cd2

    # Phase 3: Multi-n exploration
    n_list = [400, 550, 650, 800, 900, 350, 500]
    n_list = [x for x in n_list if x != n]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = subgrad_descent(h_int, n_try, 120, deadline, step0=0.04)
        h_o, c_o = exact_cd(h_o, n_try, 60, deadline)
        if c_o < best_c - 1e-7:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 4: Structured restarts (thresholding + antipodal)
    n = len(h_best)
    for r in range(12):
        if time.time() > deadline - 120:
            break
        if r % 3 == 0:
            hp = h_best.copy()
            hp[hp > 0.90] = 1.0
            hp[hp < 0.10] = 0.0
            hp = project_feasible(hp, n)
        elif r % 3 == 1:
            scale = 0.03 + 0.015 * r
            hp = project_feasible(h_best + rng.normal(0, scale, n), n)
        else:
            hp = project_feasible(1.0 - h_best + rng.normal(0, 0.02, n), n)
        h_o, c_o = exact_cd(hp, n, 35, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Final intensive CD polish
    if time.time() < deadline - 10:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 60, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
        h_f2, c_f2 = exact_cd(h_best, n, 30, deadline)
        if c_f2 < best_c:
            h_best, best_c = h_f2, c_f2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
