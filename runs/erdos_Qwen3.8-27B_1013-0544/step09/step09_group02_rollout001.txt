def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        mx = 700
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=600, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_init(n, method, rng):
    x = np.linspace(0, 2, n)
    if method == 0:
        h = 0.5 + 0.4 * np.sin(np.pi * x + rng.uniform(0, 2*np.pi))
    elif method == 1:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x + 0.7)
    elif method == 2:
        h = 0.5 + 0.35 * np.cos(np.pi * x + rng.uniform(0, np.pi))
    elif method == 3:
        h = np.where(np.abs(x - 1.0) < 0.4, 0.9, 0.1)
    elif method == 4:
        h = np.where((np.abs(x - 0.4) < 0.25) | (np.abs(x - 1.6) < 0.25), 0.9, 0.1)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(600, 0.5)
        n_parent = 600

    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h, n_parent)

    # Phase 1: Coarse exploration from structured inits
    coarse_n = 50
    best_coarse = (None, float('inf'))
    for m in range(6):
        if time.time() > deadline - 400:
            break
        h_c = structured_init(coarse_n, m, rng)
        h_opt, c_opt = coord_cd(h_c, coarse_n, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_coarse[1]:
            best_coarse = (h_opt.copy(), c_opt)

    # Interpolate coarse to parent n
    if best_coarse[0] is not None:
        x_coarse = np.linspace(0, 2, coarse_n)
        h_interp = project_feasible(np.interp(np.linspace(0, 2, n_parent), x_coarse, best_coarse[0]), n_parent)
        h_opt, c_opt = coord_cd(h_interp, n_parent, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 2: CD on current best
    h2, c2 = coord_cd(best_h, n_parent, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-resolution upward
    for n_try in [750, 900]:
        if time.time() > deadline - 200:
            break
        if n_try <= len(best_h):
            continue
        x_cur = np.linspace(0, 2, len(best_h))
        h_int = project_feasible(np.interp(np.linspace(0, 2, n_try), x_cur, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=250, deadline=deadline, shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Multi-shift subgradient + CD
    n_cur = len(best_h)
    if time.time() < deadline - 120:
        for ks in [3, 5, 8]:
            h4, c4 = sg_descent(best_h, n_cur, 200, deadline, k_shifts=ks)
            if c4 < best_c:
                best_h, best_c = h4, c4
            h4b, c4b = coord_cd(best_h, n_cur, max_passes=200, deadline=deadline, shuffle=True, rng=rng)
            if c4b < best_c:
                best_h, best_c = h4b, c4b

    # Phase 5: SLSQP with increasing alpha
    n_cur = len(best_h)
    if time.time() < deadline - 100:
        h5, c5 = slsqp_polish(best_h, n_cur, [8000, 30000, 120000], 600, deadline)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: CD after SLSQP
    n_cur = len(best_h)
    h6, c6 = coord_cd(best_h, n_cur, max_passes=600, deadline=deadline, shuffle=True, rng=rng)
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 7: Random perturbations + CD (finer steps)
    n_cur = len(best_h)
    for r in range(25):
        if time.time() > deadline - 25:
            break
        s = 0.001 + 0.003 * r * 0.5
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = coord_cd(hp, n_cur, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 8: Final SLSQP + CD
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        hf, cf = slsqp_polish(best_h, n_cur, [400000, 1000000], 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
