```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def compute_R(h, n):
    """Compute R(k) = ∫h(x)h(x+k)dx for all lags"""
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_R(h, n, lag):
    """Compute ∂R(k)/∂h(x) for given lag. Returns array g where g[x] = h(x+lag) + h(x-lag)"""
    g = np.zeros(n)
    for x in range(n):
        xf = x + lag
        xb = x - lag
        if 0 <= xf < n:
            g[x] += h[xf]
        if 0 <= xb < n:
            g[x] += h[xb]
    return g

def lp_step(h, n, tol=0.01, max_active=40):
    """One LP iteration of the cutting plane method."""
    dx = 2.0 / n
    R = compute_R(h, n)
    min_R = R.min()
    
    # Select active lags (those near the minimum)
    active_mask = R >= min_R + tol
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > max_active:
        # Take the most active (closest to min)
        active_idx = active_idx[np.argsort(R[active_idx])[:max_active]]
    
    if len(active_idx) == 0:
        return h
    
    # Build LP
    # Variables: [h_new (n), t (1)]
    # Minimize: -t => c = [0,...,0, -1]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active m: -g_m @ h_new + t <= h @ g_m - R[m]
    A_rows = []
    b_rows = []
    
    for m in active_idx:
        lag = int(m) - (n - 1)
        g = gradient_R(h, n, lag)
        row = np.zeros(n + 1)
        row[:n] = -g
        row[n] = 1.0  # coefficient of t
        A_rows.append(row)
        b_rows.append(h @ g - R[m])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h_new <= 1, t unbounded
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = project_feasible(res.x[:n], n)
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def compute_R(h, n):
    dx = 2.0 / n
    return np.correlate(h, h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_R(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += h[lag:]
        g[lag:] += h[:n - lag]
    else:
        l = -lag
        g[l:] += h[:n - l]
        g[:n - l] += h[l:]
    return g

def lp_heuristic(h, n, tol=0.01, max_active=30, step=1.0):
    R = compute_R(h, n)
    min_R = R.min()
    active = np.where(R >= min_R + tol)[0]
    if len(active) == 0:
        active = np.array([np.argmin(R)])
    if len(active) > max_active:
        active = active[np.argsort(R[active])[:max_active]]
    A_rows, b_rows = [], []
    for m in active:
        lag = int(m) - (n - 1)
        g = gradient_R(h, n, lag)
        row = np.zeros(n + 1)
        row[:n] = -g
        row[n] = 1.0
        A_rows.append(row)
        b_rows.append(h @ g - R[m])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = -1.0
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_lp = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible((1 - step) * h + step * h_lp, n)
            return h_new
    except Exception:
        pass
    return h

def fw_step(h, n, temperature=0.05):
    R = compute_R(h, n)
    min_R = R.min()
    weights = np.exp(-(R - min_R) / temperature)
    weights /= weights.sum()
    s = np.zeros(n)
    for m in range(len(R)):
        if weights[m] < 1e-6:
            continue
        lag = m - (n - 1)
        s += weights[m] * gradient_R(h, n, lag)
    k = int(n // 2)
    threshold = np.partition(s, n - k)[n - k]
    v = (s >= threshold).astype(float)
    if v.sum() > k:
        idx = np.where(v == 1)[0]
        vals = s[idx]
        remove = idx[np.argsort(vals)[:int(v.sum() - k)]]
        v[remove] = 0.0
    elif v.sum() < k:
        idx = np.where(v == 0)[0]
        vals = s[idx]
        add = idx[np.argsort(vals)[::-1][:int(k - v.sum())]]
        v[add] = 1.0
    return v

def fw_iterate(h, n, gamma0=1.0, max_iters=50, deadline=0, temperature=0.05):
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = fw_step(h, n, temperature)
        gamma = gamma0 / (it + 2)
        h_new = project_feasible((1 - gamma) * h + gamma * v, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            h = h_new
        else:
            gamma /= 2
            h_new = project_feasible((1 - gamma) * h + gamma * v, n)
            h = h_new
    return h

def cd_refine(h, n, max_passes, deadline, top_k=15):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo, hi = -h0, 1.0 - h0
            best_d, best_val = _solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _eval_d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def _solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d, best_val = lo, float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = _eval_d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val, best_d = v, d
    return best_d, best_val

def smooth_step(h, n, window):
    if window <= 1:
        return h
    kernel = np.ones(window) / window
    h_pad = np.pad(h, window // 2, mode='edge')
    h_smooth = np.convolve(h_pad, kernel, mode='valid')[:n]
    return project_feasible(h_smooth, n)

def slsqp_run(h, n, alpha, maxiter=300):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        active = np.where(np.abs(c - m) < 1e-6)[0]
        grad = np.zeros(n)
        for m_idx in active:
            lag = int(m_idx) - (n - 1)
            g = gradient_R(x, n, lag)
            g2 = np.zeros(n)
            if lag >= 0:
                g2[:n-lag] = x[lag:]
                g2[lag:] = x[:n-lag]
            else:
                l = -lag
                g2[l:] = x[:n-l]
                g2[:n-l] = x[l:]
            grad += (g - 2 * g2)
        grad /= len(active)
        return -grad * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_jac, args=(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

            best_c, best_h = c_o, h_o
            x_orig = x_new

    # Final CD
    n_final = len(best_h)
    if time.time() < deadline - 3:
        h_f, c_f = cd_refine(best_h, n_final, 800, deadline - 1, top_k=30)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, n_final)
    c_flip = compute_c5(h_flip, n_final)
    if c_flip < best_c:
        best_c, best_h = c_flip, h_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
