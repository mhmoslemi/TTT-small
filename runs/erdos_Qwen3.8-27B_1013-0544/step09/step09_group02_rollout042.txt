def box_sum_project(y, n):
    target = 0.5 * n
    y = np.asarray(y, float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    tau = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(55):
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-13:
            return x
        if s > target: lo = tau
        else: hi = tau
        tau = 0.5 * (lo + hi)
    x = np.clip(y - tau, 0.0, 1.0)
    for _ in range(3):
        res = target - x.sum()
        if abs(res) < 1e-13: break
        mask = (x < 1.0 - 1e-10) if res > 0 else (x > 1e-10)
        if np.any(mask):
            x[mask] += res / np.count_nonzero(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

    C=np.correlate(h,1.0-h,mode='full')*dx
    m=2*n-1
    A=np.empty((m,n+1),float)
    b=np.empty(m,float)
    for r in range(m):
        off = r-(n-1)
        g=np.zeros(n)
        if off == 0:
            g[:] = 1.0 - 2.0*h
        elif off > 0:
            g[:n-off] += 1.0 - h[off:]
            g[off:] -= h[:n-off]
        else:
            d=-off
            g[d:] += 1.0 - h[:n-d]
            g[:n-d] -= h[d:]
        gd = dx*g
        A[r,:n]=gd
        A[r,n]=-1.0
        b[r]=C[r] - float(np.dot(gd,h))
    obj=np.zeros(n+1); obj[-1]=1.0
    Aeq=np.zeros((1,n+1)); Aeq[0,:n]=1.0

    try:

        if sol.status==0:

    except Exception:
        pass
    return None, False

import time
import numpy as np
from scipy.optimize import linprog


def box_sum_project(y, n):
    target = 0.5 * n
    y = np.asarray(y, float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    tau = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(55):
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-13:
            return x
        if s > target:
            lo = tau
        else:
            hi = tau
        tau = 0.5 * (lo + hi)
    x = np.clip(y - tau, 0.0, 1.0)
    for _ in range(3):
        res = target - x.sum()
        if abs(res) < 1e-13:
            break
        mask = (x < 1.0 - 1e-10) if res > 0 else (x > 1e-10)
        if np.any(mask):
            x[mask] += res / np.count_nonzero(mask)
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def smooth_fg(h, alpha, n):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(C.max())
    sh = alpha * (C - m)
    ex = np.exp(sh)
    s = float(ex.sum())
    if s <= 0.0:
        p = np.zeros(len(C))
        p[int(np.argmax(C))] = 1.0
    else:
        p = ex / s
    F = m + np.log(s) / alpha
    g = np.zeros(n)
    w0 = p[n - 1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        a = p[n - 1 + d]
        b = p[n - 1 - d]
        if a <= 1e-15 and b <= 1e-15:
            continue
        w = a + b
        if a > 1e-15:
            g[:n - d] += a
        if b > 1e-15:
            g[d:] += b
        if w > 0.0:
            g[:n - d] -= w * h[d:]
            g[d:] -= w * h[:n - d]
    return F, dx * g, m


def proj_gradient_smooth(h0, n, alpha, iters, deadline, step0):
    x = box_sum_project(h0, n)
    F, g, cm = smooth_fg(x, alpha, n)
    bx = x.copy()
    bcm = cm
    step = step0
    for _ in range(iters):
        if time.time() > deadline:
            break
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        old = step
        cs = step
        ok = False
        for _l in range(8):
            xn = box_sum_project(x - cs * g, n)
            Fn, gnv, cnm = smooth_fg(xn, alpha, n)
            if Fn <= F + 1e-4 * cs * gn * gn + 1e-13 and cnm <= bcm + 1e-7:
                x, F, g = xn, Fn, gnv
                if cnm < bcm:
                    bcm = cnm
                    bx = xn.copy()
                ok = True
                break
            cs *= 0.5
        if not ok:
            step *= 0.5
            if step < 1e-6:
                break
            continue
        step = old * 1.25 if cs > old * 0.99 else max(cs * 1.1, 1e-6)
        if step > 2.5:
            step = 2.5
    return bx, bcm


def coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0, None, None
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(ms.shape, float)
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    csub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    mval = float(csub.max())
    if mval <= unaff + 1e-12:
        return 0.0, ms, vec
    act = csub >= mval - 1e-9
    vmin = float(vec[act].min())
    vmax = float(vec[act].max())
    if vmin <= 1e-9 and vmax >= -1e-9:
        return 0.0, ms, vec
    lo = -h0
    hi = 1.0 - h0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        vals = csub + mid * vec
        mv = float(vals.max())
        if mv <= unaff + 1e-12:
            lo = hi = mid
            break
        act = vals >= mv - 1e-9
        vmin = float(vec[act].min())
        vmax = float(vec[act].max())
        if vmin > 1e-9:
            hi = mid
        elif vmax < -1e-9:
            lo = mid
        else:
            lo = hi = mid
            break
    d = 0.5 * (lo + hi)
    if abs(d) < 1e-14:
        return 0.0, ms, vec
    return float(d), ms, vec


def coord_cd(h0, n, max_passes, deadline, shuffle, rng):
    h = box_sum_project(h0, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        imp = False
        for p in order:
            d, ms, vec = coord_delta(h, p, c, n)
            if abs(d) > 1e-12:
                c[ms] += d * vec
                hp = h[p] + d
                h[p] = 1.0 if hp > 1.0 else (0.0 if hp < 0.0 else hp)
                imp = True
        if not imp:
            break
        h = box_sum_project(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h)


def lp_direction(h, n):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    m = 2 * n - 1
    A = np.empty((m, n + 1), float)
    b = np.empty(m, float)
    for r in range(m):
        off = r - (n - 1)
        g = np.zeros(n)
        if off == 0:
            g[:] = 1.0 - 2.0 * h
        elif off > 0:
            g[:n - off] += 1.0 - h[off:]
            g[off:] -= h[:n - off]
        else:
            d = -off
            g[d:] += 1.0 - h[:n - d]
            g[:n - d] -= h[d:]
        gd = dx * g
        A[r, :n] = gd
        A[r, n] = -1.0
        b[r] = C[r] - float(np.dot(gd, h))
    obj = np.zeros(n + 1)
    obj[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bee = np.array([0.5 * n])
    bounds = np.empty((n + 1, 2))
    bounds[:, 0] = 0.0
    bounds[:, 1] = 1.0
    bounds[-1, 0] = -1.0
    try:
        sol = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=bee, bounds=bounds,
                      method='highs', options={'presolve': True})
        if sol.status == 0:
            return box_sum_project(sol.x[:n], n), True
    except Exception:
        pass
    return None, False


def lp_polish(h0, n, max_iter, deadline):
    h = box_sum_project(h0, n)
    cm = compute_c5(h)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        hlin, ok = lp_direction(h, n)
        if not ok:
            break
        dvec = hlin - h
        if float(np.linalg.norm(dvec)) < 1e-8:
            break
        improved = False
        for beta in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            cand = box_sum_project(h + beta * dvec, n)
            cc = compute_c5(cand)
            if cc < cm - 1e-12:
                h = cand
                cm = cc
                improved = True
                break
        if not improved:
            break
    return h, cm
