    m = c.max()

    mx = shifted.max()
    weights = np.exp(shifted - mx)

        if w < 1e-15:
            continue
        m_shift = k - (n-1)

        i_lo = max(0, -m_shift)
        i_hi = min(n-1, n-1-m_shift)
        if i_lo > i_hi:

        # For j in [i_lo, i_hi]:
        # dc[k]/dh[j] = (1-h[j+m_shift]) - h[j-m_shift] (with boundaries)

        jf_hi = min(n - 1, n - 1 - m)

        if jb_lo <= jb_hi:
            grad[jb_lo:jb_hi+1] -= w * x[jb_lo-m:jb_hi-m+1]
    grad *= dx
    if penalty > 0:
        s = x.sum() - n / 2.0
        obj += 0.5 * penalty * s * s
        grad += penalty * s
    return obj, grad

def lbfgs_smooth(h, n, alphas, penalty, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        def func(x):
            o, _ = lse_obj_grad(x, n, alpha, penalty)
            return o

            return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def lse_obj(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    z = alpha * (c - c_max)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    obj = (c_max + z_max + np.log(exp_z.sum()) / alpha) * (2.0 / n)
    s = x.sum() - n / 2.0
    return obj + 0.5 * penalty * s * s

def lse_grad(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = c.max()
    z = alpha * (c - c_max)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    total = exp_z.sum()
    weights = exp_z / total
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m = k - (n - 1)
        jf_lo = max(0, -m)
        jf_hi = min(n - 1, n - 1 - m)
        if jf_lo <= jf_hi:
            grad[jf_lo:jf_hi + 1] += w * (1.0 - x[jf_lo + m:jf_hi + m + 1])
        jb_lo = max(0, m)
        jb_hi = min(n - 1, n - 1 + m)
        if jb_lo <= jb_hi:
            grad[jb_lo:jb_hi + 1] -= w * x[jb_lo - m:jb_hi - m + 1]
    grad *= dx
    s = x.sum() - n / 2.0
    grad += penalty * s
    return grad

def lbfgs_opt(h, n, alphas, penalty, maxiter):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad, args=(n, alpha, penalty),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'gtol': 1e-8})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
