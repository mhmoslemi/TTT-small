        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_fun_grad(x, n, alpha, penalty_mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = alpha * (c - m)
    mx = s.max()
    exp_s = np.exp(s - mx)
    w = exp_s / exp_s.sum()
    lse = (mx + np.log(exp_s.sum())) / alpha
    dx = 2.0 / n
    val = (m + lse) * dx + penalty_mu * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    n_active = min(50, len(c))
    active = np.argsort(w)[::-1][:n_active]
    w_act = w[active]
    for idx in range(n_active):
        mi = int(active[idx])
        wt = float(w_act[idx]) * dx
        if wt < 1e-14:
            continue
        lo1 = max(0, n - 1 - mi)
        hi1 = min(n, n + mi - 1)
        if hi1 > lo1:
            grad[lo1:hi1] += wt * (1.0 - x[lo1 + mi - (n - 1):hi1 + mi - (n - 1)])
        lo2 = max(0, mi - n + 1)
        hi2 = min(n, mi + n - 1)
        if hi2 > lo2:
            grad[lo2:hi2] += wt * (-x[lo2 - mi + (n - 1):hi2 - mi + (n - 1)])
    grad += 2.0 * penalty_mu * (x.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    penalty_mu = 10.0
    def fun(x):
        return smooth_fun_grad(x, n, alpha, penalty_mu)[0]
    def grad(x):
        return smooth_fun_grad(x, n, alpha, penalty_mu)[1]
    try:
        res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

def try_reflection_sym(h, n):
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    h_sym = project_feasible(h_sym, n)
    c = compute_c5(h_sym, n)
    if c < compute_c5(h, n):
        return h_sym, c
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: CD with growing top_k
    for tk in [5, 10, 20, 40, 60]:
        if time.time() > deadline - 350:
            break
        h2, c2 = multi_cd(best_h, n, 300, deadline - 350, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 2: L-BFGS-B with increasing smoothness
    if time.time() < deadline - 300:
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 300:
                break
            h_s, c_s = lbfgs_opt(best_h, n, alpha, 400)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
            if time.time() < deadline - 300:
                h2, c2 = multi_cd(best_h, n, 200, deadline - 300, 40)
                if c2 < best_c:
                    best_h, best_c = h2, c2

    # Phase 3: High top_k CD
    if time.time() < deadline - 150:
        for tk in [60, 100, 150]:
            if time.time() > deadline - 150:
                break
            h3, c3 = multi_cd(best_h, n, 500, deadline - 150, tk)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 4: Higher alpha L-BFGS-B
    if time.time() < deadline - 200:
        h_s, c_s = lbfgs_opt(best_h, n, 500000, 500)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
        if time.time() < deadline - 200:
            h2, c2 = multi_cd(best_h, n, 300, deadline - 200, 60)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 5: Stochastic perturbations
    if time.time() < deadline - 100:
        for r in range(25):
            if time.time() > deadline - 100:
                break
            s = 0.001 + 0.002 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            hc, cc = multi_cd(hp, n, 200, deadline - 100, 30)
            if cc < best_c:
                best_h, best_c = hc, cc

    # Phase 6: Different discretizations
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 550, 700, 800]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 300, deadline - 80, 50)
        h_o, c_o = try_symmetry(h_o, n_try)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new

    # Phase 7: Final deep CD
    n_cur = len(best_h)
    if time.time() < deadline - 10:
        h_f, c_f = multi_cd(best_h, n_cur, 3000, deadline, 80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final checks
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h, best_c = try_reflection_sym(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
