import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.1, abs(best_d) * 0.3 + span * 0.03)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

def coord_descend(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                top_idx = np.sort(top_idx)
                ms = ms_all[top_idx]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

    v = np.zeros(n)
    momentum = 0.9

        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Softmax weights
        w = np.exp(alpha * (c - c.max()))
        w /= w.sum()
        # Gradient of weighted sum of c_k
        grad = np.zeros(n)
        for k in range(len(c)):

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    cands = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: cands.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2.0*dd), (-da-sq)/(2.0*dd)]:
                        if lo < d < hi: cands.append(d)
    best_d, best_v = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v - 1e-15:
            best_v, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        m = min(span*0.1, abs(best_d)*0.3 + span*0.03)
        gl, gh = max(lo, best_d-m), min(hi, best_d+m)
        if gh - gl > 1e-12:
            try:
                r = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                if r.fun < best_v - 1e-15: best_d = float(r.x)
            except: pass
    return best_d

def coord_descend(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    for p in range(max_passes):
        if deadline and time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        cm = float(c.max())
        imp = False
        for idx in range(n):
            if deadline and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            mflo = max(0, n-1-idx)
            mfhi = min(2*n-2, 2*n-2-idx)
            mblo = max(0, idx)
            mbhi = min(2*n-2, idx+n-1)
            mlo, mhi = min(mflo, mblo), max(mfhi, mbhi)
            ms_all = np.arange(mlo, mhi+1)
            if len(ms_all) > top_k:
                ti = np.argsort(c[ms_all])[-top_k:]
                ms = np.sort(ms_all[ti])
                unaff = float(c[ms_all[~np.isin(ms_all, ms)]].max()) if len(ms_all)>top_k else 0.0
            else:
                ms, unaff = ms_all, 0.0
            a = np.zeros(len(ms))
            jf, jb = idx+ms-(n-1), idx-ms+(n-1)
            vf, vb = (jf>=0)&(jf<n), (jb>=0)&(jb<n)
            if np.any(vf): a[vf] += 1.0-h[jf[vf]]
            if np.any(vb): a[vb] -= h[jb[vb]]
            ca = c[ms]
            um = unaff
            if mlo > 0: um = max(um, float(c[:mlo].max()))
            if mhi < 2*n-2: um = max(um, float(c[mhi+1:].max()))
            dl = None
            for ii in range(len(ms)):
                if ms[ii] == n-1: dl = ii; break
            bd = solve_1d(ca, a, dl, um, -h0, 1.0-h0)
            if abs(bd) > 1e-12:
                nv = obj_1d(bd, ca, a, dl, um)
                if nv < cm - 1e-14:
                    h[idx] = float(np.clip(h0+bd, 0, 1))
                    imp = True
                    cm = min(cm, nv)
        h = project_feasible(h, n)
        if not imp: break
    return h, compute_c5(h, n)

def subgrad_mom(h, n, max_it=400, deadline=None):

            else:

        v = mom*v - lr*g
        hnew = project_feasible(h + v, n)
        cn = compute_c5(hnew, n)
        if cn < bc - 1e-12:

            lr = min(lr*1.02, 0.06)
        else:

            lr *= 0.7

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    cands = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: cands.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi: cands.append(d)
    best_d, best_v = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v - 1e-15:
            best_v, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        m = min(span * 0.1, abs(best_d) * 0.3 + span * 0.03)
        gl, gh = max(lo, best_d - m), min(hi, best_d + m)
        if gh - gl > 1e-12:
            try:
                r = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                if r.fun < best_v - 1e-15: best_d = float(r.x)
            except Exception:
                pass
    return best_d

def coord_descend(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    for p in range(max_passes):
        if deadline and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        imp = False
        for idx in range(n):
            if deadline and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            mflo = max(0, n - 1 - idx)
            mfhi = min(2 * n - 2, 2 * n - 2 - idx)
            mblo = max(0, idx)
            mbhi = min(2 * n - 2, idx + n - 1)
            mlo, mhi = min(mflo, mblo), max(mfhi, mbhi)
            ms_all = np.arange(mlo, mhi + 1)
            if len(ms_all) > top_k:
                ti = np.argsort(c[ms_all])[-top_k:]
                ms = np.sort(ms_all[ti])
                mask = np.zeros(len(ms_all), dtype=bool)
                mask[ti] = True
                unaff = float(c[ms_all[~mask]].max()) if (~mask).any() else 0.0
            else:
                ms, unaff = ms_all, 0.0
            a = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb): a[vb] -= h[jb[vb]]
            ca = c[ms]
            um = unaff
            if mlo > 0: um = max(um, float(c[:mlo].max()))
            if mhi < 2 * n - 2: um = max(um, float(c[mhi + 1:].max()))
            dl = None
            for ii in range(len(ms)):
                if ms[ii] == n - 1: dl = ii; break
            bd = solve_1d(ca, a, dl, um, -h0, 1.0 - h0)
            if abs(bd) > 1e-12:
                nv = obj_1d(bd, ca, a, dl, um)
                if nv < cm - 1e-14:
                    h[idx] = float(np.clip(h0 + bd, 0, 1))
                    imp = True
                    cm = min(cm, nv)
        h = project_feasible(h, n)
        if not imp: break
    return h, compute_c5(h, n)
