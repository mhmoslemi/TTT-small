import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(lo, lo, hi)), float(np.clip(hi, lo, hi)), 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d = 0.0
    best_val = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val = v
            best_d = dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.12, abs(best_d) * 0.4 + span * 0.04)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    return best_d

    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False

        if rng is not None:
            rng.shuffle(order)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:

            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)

            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)

                unaff_extra = 0.0

            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == (n - 1):
                    diag_local = ii
                    break
            lo = -h0
            hi = 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff_max)
                if new_val < current_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    current_max = min(current_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    sh = alpha*(c-m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh-mx)))
    return (m + lse/alpha)*(2.0/n)

def slsqp(h, n, alphas, deadline=None, maxiter=300):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in alphas:
        if deadline is not None and time.time() > deadline: break
        try:
            res = minimize(smooth_obj, best_h, args=(n,a), method='SLSQP',
                bounds=[(0,1)]*n,
                constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0,'jac':lambda x: np.ones(len(x))}],
                options={'maxiter':maxiter,'ftol':1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0: return 0.0
    cands = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i+1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi: cands.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2.0*dd), (-da-sq)/(2.0*dd)]:
                        if lo < d < hi: cands.append(d)
    best_d = 0.0
    best_v = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_v - 1e-15: best_v, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        m = min(span*0.12, abs(best_d)*0.4 + span*0.04)
        gl, gh = max(lo, best_d-m), min(hi, best_d+m)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_v - 1e-15: best_d, best_v = float(res.x), float(res.fun)
            except: pass
    return best_d

def coord_cd(h, n, max_passes=200, deadline=None, top_k=40, rng=None):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline: break
        c = np.correlate(h, 1.0-h, mode='full')
        cur = float(c.max())
        improved = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for idx in order:
            if deadline is not None and time.time() > deadline: break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            m_lo = min(max(0, n-1-idx), idx)
            m_hi = max(min(2*n-2, 2*n-2-idx), idx+n-1)
            ms_all = np.arange(m_lo, m_hi+1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                ti = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[ti]
                ms_set = set(ms.tolist())
                sk = [m for m in ms_all if m not in ms_set]
                unaff = float(c[sk].max()) if len(sk) > 0 else 0.0
            else:
                ms = ms_all; unaff = 0.0
            a_c = np.zeros(len(ms))
            jf = idx + ms - (n-1)
            jb = idx - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a_c[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a_c[vb] -= h[jb[vb]]
            c_a = c[ms].copy()
            u = unaff
            if m_lo > 0: u = max(u, float(c[:m_lo].max()))
            if m_hi < 2*n-2: u = max(u, float(c[m_hi+1:].max()))
            dl = None
            for ii in range(len(ms)):
                if ms[ii] == n-1: dl = ii; break
            bd = solve_1d(c_a, a_c, dl, u, -h0, 1.0-h0)
            if abs(bd) > 1e-12:
                nv = obj_1d(bd, c_a, a_c, dl, u)
                if nv < cur - 1e-14:
                    h[idx] = float(np.clip(h0+bd, 0.0, 1.0))
                    improved = True
                    cur = min(cur, nv)
        h = project_feasible(h, n)
        if not improved: break
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    cands = [lo, hi, 0.0]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        cands.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            cands.append(d)
    best_d = 0.0
    best_v = obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in cands:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)

    if best_v > unaff_max + 1e-14:

            m = min(span * 0.12, abs(best_d) * 0.4 + span * 0.04)
            gl, gh = max(lo, best_d - m), min(hi, best_d + m)

                    res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if res.fun < best_v - 1e-15:
                        best_d, best_v = float(res.x), float(res.fun)
                except Exception:
                    pass
    return best_d

def coord_cd(h, n, max_passes=200, deadline=None, top_k=40, rng=None):
    h = h.copy()
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        improved = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_lo = min(max(0, n - 1 - idx), idx)
            m_hi = max(min(2 * n - 2, 2 * n - 2 - idx), idx + n - 1)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                ti = np.sort(np.argsort(vals)[-top_k:])
                ms = ms_all[ti]
                ms_set = set(ms.tolist())
                sk = [m for m in ms_all if m not in ms_set]
                unaff = float(c[sk].max()) if len(sk) > 0 else 0.0
            else:
                ms = ms_all

            a_c = np.zeros(len(ms))

            jb = idx - ms + (n - 1)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    w = exp_sh / exp_sh.sum()
    grad = np.zeros(n)
    for j in range(n):
        mm_lo_f = max(0, n - 1 - j)
        mm_hi_f = min(2 * n - 2, 2 * n - 2 - j)
        if mm_lo_f <= mm_hi_f:
            jf = j + np.arange(mm_lo_f, mm_hi_f + 1) - (n - 1)
            grad[j] += np.dot(w[mm_lo_f:mm_hi_f + 1], 1.0 - x[jf])
        mm_lo_b = max(0, j - n + 1)
        mm_hi_b = min(2 * n - 2, j)
        if mm_lo_b <= mm_hi_b:
            jb = j - np.arange(mm_lo_b, mm_hi_b + 1) + (n - 1)
            grad[j] += np.dot(w[mm_lo_b:mm_hi_b + 1], -x[jb])
    return (2.0 / n) * grad

def slsqp_polish(h, n, alphas, deadline=None, maxiter=200):
    from scipy.optimize import minimize
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, a), method='SLSQP',
                           jac=lambda x, nn, aa: smooth_grad(x, nn, aa),
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: x.sum() - n / 2.0,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c
