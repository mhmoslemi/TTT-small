The problem is: min_h max_k c_k(h)
This is equivalent to: min_h g(h) where g(h) = max_k c_k(h)

The subgradient of g at h is: ∂g(h) = conv{∇c_k(h) : k ∈ A(h)} where A(h) = {k : c_k(h) = max}

So the subgradient is the convex hull of the gradients of the active constraints.

A natural algorithm is:
1. h_{t+1} = Π_feasible(h_t - α_t * ∇̄ c_{k_t}(h_t))
where k_t is a random active constraint and ∇̄ is the gradient.

A better approach: **Use the average subgradient**:
1. Find all active constraints A
2. Compute g_avg = (1/|A|) Σ_{k∈A} ∇c_k(h)
3. h_{t+1} = Π_feasible(h_t - α_t * g_avg)

This should converge faster because it uses more information.

    for t in range(steps):

            break
        c = np.correlate(h, 1-h, mode='full')
        T = c.max()
        # Active set (with tolerance)
        tol = max(1e-10, 0.01 * T)
        active = np.where(c >= T - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        grad = np.zeros(n)
        for k in active:
            j = k - (n-1)
            # d(c[k])/dh[p] = (1-h[p+j]) [if p+j valid] - h[p-j] [if p-j valid]
            # Vectorized:

            grad[mask1] += (1 - h[valid1[mask1]])

def subgrad_avg(h, n, max_steps, deadline, rng):
    h = h.copy()
    for t in range(max_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        T = c.max()
        tol = max(1e-8, 0.03 * T)
        active = np.where(c >= T - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 15:
            idx = rng.choice(len(active), 15, replace=False)
            active = active[idx]
        grad = np.zeros(n)
        for k in active:
            j = k - (n - 1)
            idx1 = np.arange(n) + j
            m1 = (idx1 >= 0) & (idx1 < n)
            grad[m1] += (1.0 - h[idx1[m1]])
            idx2 = np.arange(n) - j
            m2 = (idx2 >= 0) & (idx2 < n)
            grad[m2] -= h[idx2[m2]]
        grad /= len(active)
        grad -= np.mean(grad)
        lr = 0.03 / (1 + 0.05 * t)
        h_new = project_feasible(h - lr * grad, n)
        if compute_c5(h_new, n) < compute_c5(h, n):
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD on initial
    t1_end = min(t0 + 160, deadline - 500)
    h1, c1 = coord_descent(best_h, n, 500, t1_end, rng, random_order=True)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Perturb + subgrad + CD cycles
    t2_end = min(t0 + 420, deadline - 300)
    for round_i in range(8):
        if time.time() > t2_end:
            break
        s = 0.01 + 0.015 * round_i
        if round_i % 3 == 1:
            hp = best_h + rng.normal(0, s, n)
        elif round_i % 3 == 2:
            freq = rng.uniform(2, 12)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n)
            hp = best_h + s * np.sin(freq * np.pi * x + phase)
        else:
            hp = best_h.copy()
            block = n // 4
            start = rng.randint(0, max(1, n - block))
            hp[start:start + block] = rng.uniform(0, 1, block)
        hp = project_feasible(hp, n)
        hp = subgrad_avg(hp, n, 15, min(time.time() + 20, t2_end - 20), rng)
        t_sub = min(time.time() + 80, t2_end - 10)
        h_cd, c_cd = coord_descent(hp, n, 200, t_sub, rng, random_order=True)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
    # Phase 3: Different n values
    t3_end = min(t0 + 700, deadline - 200)
    x_cur = np.linspace(0, 2, n)
    for n_try in [400, 600, 800, 1000, 700]:
        if time.time() > t3_end:
            break
        if n_try == n:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        t_sub = min(time.time() + 100, t3_end - 20)
        h_opt, c_opt = coord_descent(h_int, n_try, 200, t_sub, rng, random_order=True)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            n = n_try
            x_cur = x_new
    # Phase 4: Final heavy CD polish
    n = len(best_h)
    remaining = deadline - time.time()
    max_p = max(50, int(remaining / 1.5))
    hf, cf = coord_descent(best_h, n, max_p, deadline, rng, random_order=False)
    if cf < best_c:
        best_h, best_c = hf, cf
    # One more random-order pass if time
    if time.time() < deadline - 20:
        hf2, cf2 = coord_descent(best_h, n, 50, deadline, rng, random_order=True)
        if cf2 < best_c:
            best_h, best_c = hf2, cf2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
