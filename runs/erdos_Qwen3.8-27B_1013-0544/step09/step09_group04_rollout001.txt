import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order='sequential'):
    h = h.copy()
    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'random':
            idx = n_idx.copy()
            np.random.shuffle(idx)
        elif order == 'hot':
            hot = c >= c.max() - 0.05 * c.max()
            order_score = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                order_score[p] = hot[m_lo:m_hi+1].sum()
            idx = n_idx[np.argsort(-order_score)]
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:

    best_h = project_feasible(h_init, n_init)

    # Phase 1: Quick CD convergence
    h1, c1 = exact_cd(best_h, n_init, 15, deadline)
    if c1 < best_c:

    # Phase 2: Hot-shift ordered CD
    if time.time() < deadline - 100:
        h2, c2 = exact_cd(best_h, n_init, 15, deadline, order='hot')
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2b, c2b = exact_cd(best_h, n_init, 10, deadline)
        if c2b < best_c:
            best_h, best_c = h2b, c2b

    n = n_init
    # Phase 3: Structured restarts
    for r in range(15):
        if time.time() > deadline - 150:
            break
        strategy = r % 5
        if strategy == 0:
            s = 0.01 + 0.015 * (r // 5)
            hp = best_h + rng.normal(0, s, n)
        elif strategy == 1:
            hp = best_h.copy()
            n_flip = max(5, n // 10)
            idx = rng.choice(n, n_flip, replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif strategy == 2:
            hp = best_h.copy()
            block_size = max(3, n // 15)
            for b in range(n // block_size):
                if rng.random() < 0.4:
                    s0 = b * block_size
                    s1 = min(s0 + block_size, n)
                    hp[s0:s1] = rng.choice([0.0, 1.0], s1 - s0)
        elif strategy == 3:
            hp = best_h.copy()
            n_swap = max(3, n // 12)
            idx = rng.choice(n, 2 * n_swap, replace=False)
            for i in range(0, 2 * n_swap, 2):
                hp[idx[i]], hp[idx[i+1]] = hp[idx[i+1]], hp[idx[i]]
        else:
            hp = best_h.copy()
            quarter = n // 4
            seg = rng.randint(0, 4)
            s0 = seg * quarter
            s1 = s0 + quarter
            hp[s0:s1] = 1.0 - hp[s0:s1]
        hp = project_feasible(hp, n)
        hc, cc = exact_cd(hp, n, 15, deadline)
        if cc < best_c:
            best_h, best_c = hc, cc
            hc2, cc2 = exact_cd(best_h, n, 15, deadline, order='hot')
            if cc2 < best_c:
                best_h, best_c = hc2, cc2
            hc3, cc3 = exact_cd(best_h, n, 10, deadline)

            ho2, co2 = exact_cd(best_h, n, 10, deadline, order='hot')

                best_h, best_c = ho2, co2

    for r in range(8):
        if time.time() > deadline - 80:
            break
        s = 0.005 + 0.01 * r
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        hc, cc = exact_cd(hp, n, 12, deadline)
        if cc < best_c:
            best_h, best_c = hc, cc

    if time.time() < deadline - 5:

        hf, cf = exact_cd(best_h, n, max_p, deadline)

            best_h, best_c = hf, cf
        hf2, cf2 = exact_cd(best_h, n, 5, deadline, order='hot')
        if cf2 < best_c:
            best_h, best_c = hf2, cf2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
