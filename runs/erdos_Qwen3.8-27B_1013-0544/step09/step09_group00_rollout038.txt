Let me also think about whether the coordinate descent is truly converging to the local minimum. The bisection finds the smallest L such that there exists x ∈ [0,1] with all constraints satisfied. This should give the optimal 1D solution for that coordinate. So if we cycle through all coordinates and keep improving, we should converge to a coordinate-wise minimum.

The issue is that coordinate-wise minimum ≠ global minimum. To escape, we need perturbations or a different optimization method.

**Key improvement idea**: Instead of Gaussian perturbations for "shaking", use **structured perturbations** that move blocks of coordinates together. This preserves more of the structure while still allowing escape from local minima.

**Another idea**: Use **simulated annealing** style: accept worse solutions with some probability, allowing the search to escape local minima.

**Yet another idea**: The problem has a natural "temperature" parameter. We can define:
f_α(h) = (1/α) * log(Σ_k exp(α * c[k]))

For small α, this is nearly convex (the log-sum-exp is convex in c, and c is bilinear in h, so the composition is not necessarily convex, but it's "smoother"). For large α, it approaches the max.

By solving a sequence of problems with increasing α (continuation method), we can navigate the landscape more effectively.

The gradient of f_alpha with respect to h[p] is:
(1/alpha) * dx * [Σ_k w_k * ∂c[k]/∂h[p]] / Σ_k w_k

where w_k = exp(alpha * c[k]).

This is a weighted average of the partial derivatives, with weights proportional to exp(alpha * c[k]). As alpha → ∞, the weights concentrate on the maximum, recovering the subgradient of max.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    """Find optimal h[p] given all others fixed. Returns new value."""
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_pos = np.where(ms == (n - 1))[0]
    has_diag = len(diag_pos) > 0
    if has_diag:
        c_diag = c_base[diag_pos[0]]
        a[diag_pos] = 0.0  # diagonal handled separately
    
    # Unaffected max
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    
    cur_max = float(np.max(c))
    
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    
    # Bisection on L
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            feasible = False
        
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        if x_lo > x_hi + 1e-13:
            feasible = False
        
        if feasible and has_diag:
            # x(1-x) constraint: c_diag + delta*(1-2*h0) - delta^2 <= L
            # Let delta = x - h0
            # c_diag + (x-h0)(1-2*h0) - (x-h0)^2 <= L
            # = c_diag + x(1-2*h0) - h0(1-2*h0) - (x^2 - 2*x*h0 + h0^2) <= L
            # = c_diag + x - 2*x*h0 - h0 + 2*h0^2 - x^2 + 2*x*h0 - h0^2 <= L
            # = c_diag - h0 + h0^2 + x - x^2 <= L
            # So: x - x^2 <= L - c_diag + h0 - h0^2
            Cq = L - c_diag + h0 - h0 * h0
            # Need x(1-x) <= Cq, i.e., x^2 - x + Cq >= 0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                # x must be in [0, r1] U [r2, 1] intersected with [x_lo, x_hi]
                if not (x_hi <= r1 or x_lo >= r2):
                    feasible = False
            else:
                feasible = False
        
        if feasible:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi  # optimal L
    
    # Find x for this L
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            # Prefer the endpoint closer to h0
            if x_lo >= r1 and x_hi <= r2:
                # Current x is in forbidden zone, move to nearest boundary
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif not (x_hi <= r1 or x_lo >= r2):
                # Interval overlaps forbidden zone
                if x_lo < r1 and x_hi > r2:
                    # Choose side with more room
                    if r1 - x_lo >= x_hi - r2:
                        x_hi = min(x_hi, r1)
                    else:
                        x_lo = max(x_lo, r2)
                elif x_lo < r1:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def smooth_obj_and_grad(x, n, alpha):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    exp_s = np.exp(shifted)
    ss = exp_s.sum()
    lse = m + np.log(ss) / alpha
    dx = 2.0 / n
    # Gradient
    w = exp_s / ss  # softmax weights
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        dpos = ms == (n - 1)
        contrib[dpos] = (1.0 - 2.0 * h[p])
        grad[p] = np.dot(w[ms], contrib) * dx
    return lse * dx, grad

def lbfgs_opt(h, n, deadline, alphas=(500, 2000, 10000, 50000)):

    for alpha in alphas:
        if time.time() > deadline:

                # Add penalty

                val += pen_param * s * s
                g += 2.0 * pen_param * s * np.ones(n)
                return val, g
            try:
                res = minimize(obj_grad, h_curr, jac=True, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': 200, 'gtol': 1e-10})

                if c_curr < best_c:
                    best_h, best_c = h_curr.copy(), c_curr
                pen_param *= 10
            except Exception:
                break
    return best_h, best_c

        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif not (x_hi <= r1 or x_lo >= r2):
                if x_lo < r1 and x_hi > r2:
                    if r1 - x_lo >= x_hi - r2:
                        x_hi = min(x_hi, r1)
                    else:
                        x_lo = max(x_lo, r2)
                elif x_lo < r1:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                dpos = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(dpos) > 0:
                    c[ms[dpos[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj_grad(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    es = np.exp(sh)
    ss = es.sum()
    val = (m + np.log(ss) / alpha) * (2.0 / n)
    w = es / ss
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        grad[p] = np.dot(w[ms], contrib) * (2.0 / n)
    return val, grad

def obj_grad_pen(x, n, alpha, pen):
    v, g = smooth_obj_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    v += pen * s * s
    g = g + 2.0 * pen * s
    return v, g

def lbfgs_opt(h, n, deadline, alphas=(1000, 5000, 20000)):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        pen = 200.0
        h_curr = best_h.copy()
        for _ in range(3):
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(obj_grad_pen, h_curr, args=(n, alpha, pen),
                             jac=True, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 250, 'gtol': 1e-14})
                h_curr = project_feasible(res.x, n)
                c_curr = compute_c5(h_curr, n)
                if c_curr < best_c:
                    best_c = c_curr
                    best_h = h_curr.copy()
            except Exception:
                pass
            pen *= 5.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    h_best, c = exact_cd(h_best, n, 30, deadline)
    if c < best_c:
        best_c = c

    h_best, c = lbfgs_opt(h_best, n, deadline, alphas=(1000, 5000, 20000))
    if c < best_c:
        best_c = c

    h_best, c = exact_cd(h_best, n, 60, deadline)
    if c < best_c:
        best_c = c

    for n_try in [350, 500, 800, 1000]:
        if time.time() > deadline - 250:
            break
        if n_try == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(xn, xo, h_best), n_try)
        ho, co = exact_cd(hi, n_try, 20, deadline)
        ho, co = lbfgs_opt(ho, n_try, deadline, alphas=(2000, 10000))
        ho, co = exact_cd(ho, n_try, 40, deadline)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    for r in range(50):
        if time.time() > deadline - 80:
            break
        strategy = r % 4
        if strategy == 0:
            h_r = project_feasible(h_best + rng.normal(0, 0.01 + 0.003 * r, n), n)
        elif strategy == 1:
            h_r = h_best.copy()
            for _ in range(3):
                s = rng.randint(0, max(1, n - 30))
                l = rng.randint(10, min(40, n - s))
                h_r[s:s + l] += rng.uniform(-0.1, 0.1, l)
            h_r = project_feasible(h_r, n)
        elif strategy == 2:
            h_r = h_best.copy()
            idx = rng.choice(n, size=max(1, n // 5), replace=False)
            h_r[idx] = 1.0 - h_r[idx]
            h_r = project_feasible(h_r, n)
        else:
            h_r = h_best.copy()
            shift = rng.randint(5, min(30, n // 4))
            h_r[-shift:] = h_best[:shift] + rng.normal(0, 0.02, shift)
            h_r = project_feasible(h_r, n)
        ho, co = exact_cd(h_r, n, 12, deadline)
        ho, co = lbfgs_opt(ho, n, deadline, alphas=(5000,))
        ho, co = exact_cd(ho, n, 25, deadline)
        if co < best_c:
            h_best, best_c = ho, co

    if time.time() < deadline - 3:
        h_best, c = exact_cd(h_best, n, 100, deadline)
        if c < best_c:
            best_c = c

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
