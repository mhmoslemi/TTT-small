    return np.clip(h, 0.0, 1.0)

def exact_cd_step(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def apply_cd(h, p, d, c, n):
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    delta = np.zeros(len(ms))
    if np.any(vf):
        delta[vf] += d * (1.0 - h[jf[vf]])
    if np.any(vb):
        delta[vb] -= d * h[jb[vb]]
    c[ms] += delta
    h[p] = np.clip(h[p] + d, 0.0, 1.0)

def run_cd(h, n, max_passes=300, deadline=None, rng=None, order='alt'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if order == 'fwd':
            idx = np.arange(n)
        elif order == 'bwd':
            idx = np.arange(n - 1, -1, -1)
        elif order == 'alt':
            idx = np.arange(n) if p_idx % 2 == 0 else np.arange(n - 1, -1, -1)
        else:
            idx = rng.permutation(n)
        improved = False
        for p in idx:
            d = exact_cd_step(h, p, c, n)
            if abs(d) > 1e-13:
                apply_cd(h, p, d, c, n)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def lse_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    mx = c.max()
    return mx + np.log(np.sum(np.exp(alpha * (c - mx)))) / alpha

def sum_eq_fun(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
        try:
            res = minimize(lse_objective, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD from parent
    h1, c1 = run_cd(best_h, n_init, 400, deadline, rng, 'alt')
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: 1-h symmetry
    h_comp = project_feasible(1.0 - best_h, n_init)
    h2, c2 = run_cd(h_comp, n_init, 300, deadline, rng, 'alt')
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-resolution sweep
    x_cur = np.linspace(0, 2, len(best_h))
    n_list = [300, 350, 400, 450, 500, 550, 600, 650, 700, 750, 800, 850, 900, 950]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = run_cd(h_int, n_try, 250, deadline, rng, 'alt')
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = np.linspace(0, 2, n_opt if False else n_try)
            x_cur = np.linspace(0, 2, n_try)

    # Phase 4: Random binary restarts
    n_cur = len(best_h)
    for ri in range(12):
        if time.time() > deadline - 100:
            break
        n_try = n_cur
        h_bin = np.zeros(n_try)
        ones = rng.choice(n_try, n_try // 2, replace=False)
        h_bin[ones] = 1.0
        h_bin = project_feasible(h_bin, n_try)
        h_r, c_r = run_cd(h_bin, n_try, 300, deadline, rng, 'rand')
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            n_cur = n_try

    # Phase 5: Perturbation + CD (annealing schedule)
    n_cur = len(best_h)
    for r in range(20):
        if time.time() > deadline - 80:
            break
        sigma = 0.02 * (1.0 - r / 20.0) + 0.002
        hp = project_feasible(best_h + rng.normal(0, sigma, n_cur), n_cur)
        h_r, c_r = run_cd(hp, n_cur, 200, deadline, rng, 'alt')
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 6: SLSQP polish
    if time.time() < deadline - 60:
        h6, c6 = slsqp_polish(best_h, len(best_h), [5000, 20000, 100000], 500, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final CD polish
    n_final = len(best_h)
    h7, c7 = run_cd(best_h, n_final, 500, deadline, rng, 'alt')
    if c7 < best_c:
        best_h, best_c = h7, c7

    # Phase 8: Another SLSQP with very high alpha
    if time.time() < deadline - 20:
        h8, c8 = slsqp_polish(best_h, n_final, [200000, 500000], 600, deadline)
        if c8 < best_c:
            best_h, best_c = h8, c8
        h8b, c8b = run_cd(best_h, n_final, 300, deadline, rng, 'bwd')
        if c8b < best_c:
            best_h, best_c = h8b, c8b

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
