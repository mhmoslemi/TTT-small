import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_sb(x, n):
    """Project onto {h: sum(h)=n/2, 0<=h<=1} via bisection"""
    target = n / 2.0
    lo, hi = x.min() - 0.5, x.max() + 0.5
    for _ in range(80):
        tau = (lo + hi) * 0.5
        s = np.clip(x - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes, deadline, shuffle, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):
                    delta[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    delta[vb] -= d * h[jb[vb]]
                c[ms] += delta
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 10 == 9:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def topk_subgrad(h, n, k, deadline, max_iter=3000, lr_init=0.03):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Top-k subgradient
        topk = np.argpartition(c, -k)[-k:]
        sg = np.zeros(n)
        for m in topk:
            # dc[m]/dh[p] = (1-h[p-m+(n-1)])*I(valid) - h[p+m-(n-1)]*I(valid)
            p_lo = max(0, m - (n - 1))
            p_hi = min(n - 1, m)
            if p_lo <= p_hi:
                idx = np.arange(p_lo, p_hi + 1)
                sg[idx] += 1.0 - h[idx - m + (n - 1)]
            q_lo = max(0, (n - 1) - m)
            q_hi = min(n - 1, 2 * n - 2 - m)
            if q_lo <= q_hi:
                idx = np.arange(q_lo, q_hi + 1)
                sg[idx] -= h[idx + m - (n - 1)]
        sg /= k
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_h = h.copy()
            best_c = c_new
            stall = 0
            lr = min(lr * 1.01, 0.08)
        else:
            lr *= 0.9
            stall += 1
            if stall > 80 or lr < 1e-10:
                break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15
    
    try:

def penalty_smooth_obj(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    obj = (m + lse / alpha) * (2.0 / n)
    penalty = lam * (x.sum() - n / 2.0) ** 2
    return obj + penalty

def penalty_smooth_jac(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of soft-max objective
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        offset = k - (n - 1)
        # First term: (1-h[p-offset]) for valid p
        p_lo = max(0, k - (n-1))
        p_hi = min(n-1, k)
        if p_lo <= p_hi:
            idx = np.arange(p_lo, p_hi+1)
            grad[idx] += w[k] * (1.0 - h[idx - offset])
        # Second term: -h[p+offset] for valid p
        q_lo = max(0, (n-1) - k)

            idx = np.arange(q_lo, q_hi+1)
            grad[idx] -= w[k] * h[idx + offset]
    grad *= (2.0 / n)
    
    # Add penalty gradient
    grad += 2 * lam * (x.sum() - n/2.0)
    return grad
