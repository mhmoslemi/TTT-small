```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    # Same as before - find optimal value for coordinate p
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_correlation(h, n)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c
```

```python
def eval_swap_delta(h, i, j, delta, corr, n):
    """Evaluate corr + delta_change for a swap h[i]+=delta, h[j]-=delta."""
    # Compute the change in each correlation value
    # For h[p] += delta:
    # corr[n-1] changes by delta - delta^2 (approximately delta for small delta)
    # corr[n-1+l] for l>0 changes by: delta*(1-h[p-l]) * [l<=p] - delta*h[p+l] * [p+l<n]
    # corr[n-1-l] for l>0 changes by: delta*(1-h[p+l]) * [p+l<n] - delta*h[p-l] * [l<=p]
    
    # Actually for exact: at center, corr[n-1] = sum h[k](1-h[k])
    # If h[i] changes to h[i]+delta: new center = old center - h[i](1-h[i]) + (h[i]+delta)(1-h[i]-delta)
    # = old center + delta - delta^2
    # If h[j] changes to h[j]-delta: new center += -delta - delta^2 = -delta + delta^2
    # Total center change: -2*delta^2 (approximately 0 for small delta)
    
    # For the non-center terms, the change is linear in delta
    # Let's compute the coefficients
    # For h[i] += delta:
    ai = np.zeros(2*n-1)  # coefficient of delta for each corr[m]
    # center: ai[n-1] = 1 (ignoring delta^2)
    ai[n-1] = 1
    # m = n-1+l, l >= 1: ai[n-1+l] += (1-h[i-l])*[l<=i] - h[i+l]*[i+l<n]
    for l in range(1, n):
        if l <= i:
            ai[n-1+l] += (1 - h[i-l])
        if i+l < n:
            ai[n-1+l] -= h[i+l]
        if i+l < n:
            ai[n-1-l] += (1 - h[i+l])
        if l <= i:
            ai[n-1-l] -= h[i-l]
    
    # For h[j] -= delta: negate the above with i replaced by j
    aj = np.zeros(2*n-1)
    aj[n-1] = 1
    for l in range(1, n):
        if l <= j:
            aj[n-1+l] += (1 - h[j-l])
        if j+l < n:
            aj[n-1+l] -= h[j+l]
        if j+l < n:
            aj[n-1-l] += (1 - h[j+l])
        if l <= j:
            aj[n-1-l] -= h[j-l]
    
    # Total change: ai * delta - aj * delta = (ai - aj) * delta
    # New corr: corr + (ai - aj) * delta - 2*delta^2 (center correction)
    # For small delta, ignore the -2*delta^2
    return corr + (ai - aj) * delta
```

```python
def swap_coeff(h, p, n):
    """Compute the coefficient vector for changing h[p] by +delta."""
    coeff = np.zeros(2*n-1)
    coeff[n-1] = 1.0  # center term (delta - delta^2, approximate as delta)
    
    # For m = n-1+l (l >= 1):
    # coefficient += (1-h[p-l]) if l <= p, minus h[p+l] if p+l < n
    l = np.arange(1, n)
    # m = n-1+l
    # term1: (1-h[p-l]) for l <= p, i.e., p-l >= 0
    mask1 = l <= p
    idx1 = n - 1 + l[mask1]
    coeff[idx1] += (1 - h[p - l[mask1]])
    
    # term2: -h[p+l] for p+l < n
    mask2 = p + l < n
    idx2 = n - 1 + l[mask2]
    coeff[idx2] -= h[p + l[mask2]]
    
    # For m = n-1-l (l >= 1):
    # coefficient += (1-h[p+l]) if p+l < n, minus h[p-l] if l <= p

    mask4 = l <= p
    idx4 = n - 1 - l[mask4]
    coeff[idx4] -= h[p - l[mask4]]
    
    return coeff
```

```python
def compute_swap_coeffs(h, n):
    """For each position p, compute the vector of how corr changes when h[p] increases by 1.
    Returns a matrix of shape (n, 2n-1)."""
    coeffs = np.zeros((n, 2*n-1))
    for p in range(n):
        coeffs[p, n-1] = 1.0  # center (approx)
        l = np.arange(1, n)
        # m = n-1+l
        mask_right = l <= p
        coeffs[p, n-1+l[mask_right]] += (1 - h[p - l[mask_right]])
        mask_right2 = (p + l) < n
        coeffs[p, n-1+l[mask_right2]] -= h[p + l[mask_right2]]
        # m = n-1-l
        mask_left = (p + l) < n
        coeffs[p, n-1-l[mask_left]] += (1 - h[p + l[mask_left]])
        mask_left2 = l <= p
        coeffs[p, n-1-l[mask_left2]] -= h[p - l[mask_left2]]
    return coeffs
```
