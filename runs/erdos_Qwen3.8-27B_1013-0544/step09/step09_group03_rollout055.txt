def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)

    h = h.copy()
    best_c = compute_c5(h, n)

def pair_local_search(h, n, max_iters, deadline, seed):
    """2-coordinate local search: transfer mass between pairs."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for iteration in range(max_iters):
        if time.time() > deadline:
            break
        # Select pairs: one high, one low (to create more binary-like structure)
        # Or just random interior points
        interior = np.where((h > 0.001) & (h < 0.999))[0]
        if len(interior) < 2:
            break
        i, j = rng.choice(interior, size=2, replace=False)
        if i == j:
            continue
        
        c = np.correlate(h, 1.0 - h, mode='full')
        
        # Compute d[m] = change in c[m] per unit transfer from j to i
        d = np.zeros(2 * n - 1)
        for k in range(-(n - 1), n):
            val = 0.0
            if 0 <= i - k < n:
                val += (1.0 - h[i - k])
            if 0 <= i + k < n:
                val -= h[i + k]
            if 0 <= j - k < n:
                val -= (1.0 - h[j - k])
            if 0 <= j + k < n:
                val += h[j + k]
            d[n - 1 + k] = val
        
        # Find optimal t in [0, min(1-h[i], h[j])]
        t_hi = min(1.0 - h[i], h[j])
        if t_hi < 1e-12:
            continue
        
        # Ternary search for min of max(c + t*d)
        lo, hi2 = 0.0, t_hi
        for _ in range(30):
            mid1 = lo + (hi2 - lo) / 3
            mid2 = hi2 - (hi2 - lo) / 3
            if np.max(c + mid1 * d) < np.max(c + mid2 * d):
                hi2 = mid2
            else:
                lo = mid1
        
        t_opt = (lo + hi2) / 2
        f_new = float(np.max(c + t_opt * d))
        f_cur = float(c.max())
        
        if f_new < f_cur - 1e-12:
            h[i] += t_opt
            h[j] -= t_opt
            h = np.clip(h, 0, 1)
            new_c = compute_c5(h, n)
            if new_c < best_c:
                best_c = new_c
    
    h = project_feasible(h, n)
    return h, best_c

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    def smooth_obj(x, n, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    
    def sum_con(x, n):
        return x.sum() - n / 2.0
    
    def sum_con_jac(x, n):
        return np.ones(n)
    
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x.copy(), n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
