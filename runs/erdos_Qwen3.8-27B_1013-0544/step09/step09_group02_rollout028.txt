from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_fun(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    lse = mx + np.log(exp_s.sum())
    return (m + lse / alpha) * (2.0 / n), exp_s / exp_s.sum()

def lbfgs_phase(h, n, alphas, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        res = minimize(
            softmax_obj, best_h,
            jac=lambda x: softmax_grad(x, n, alpha),
            args=(n, alpha),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8}
        )
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    return best_h, best_c

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5
