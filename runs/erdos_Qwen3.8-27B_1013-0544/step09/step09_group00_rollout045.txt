def interp_to_n(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_int = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_int, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n_best = 700
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    # Phase 1: Primary CD on initial (most time budget)
    time_alloc_1 = min(500, (budget_s - 100) * 0.45)
    dl1 = t0 + time_alloc_1
    h1, c1 = exact_cd(h_best, n_best, 20000, dl1, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Focal perturbations and re-CO
    n = len(h_best)
    time_alloc_2 = min(300, (budget_s - 200) * 0.3)
    dl2 = dl1 + time_alloc_2
    scales = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
    for sc in scales:
        if time.time() > dl2 - 30:
            break
        best_c_local = best_c
        for attempt in range(3):
            if time.time() > dl2 - 10:
                break
            frac = 0.1 + 0.05 * attempt
            hp = focal_perturb(h_best, n, rng, sc, frac)
            h_o, c_o = exact_cd(hp, n, 5000, dl2, rng)
            if c_o < best_c_local:
                best_c_local = c_o
                h_best = h_o
        if best_c_local < best_c:
            best_c = best_c_local

    # Phase 3: Multi-n exploration
    n_candidates = [650, 750, 800, 600]
    time_per_n = max(40, (budget_s - 50) * 0.08)
    for nc in n_candidates:
        if time.time() > deadline - 60:
            break
        if nc == n:
            continue
        dl_nc = min(time.time() + time_per_n, deadline - 30)
        h_int = interp_to_n(h_best, n, nc)
        h_o, c_o = exact_cd(h_int, nc, 5000, dl_nc, rng)
        if c_o < best_c:
            h_best, best_c, n = h_o, c_o, nc

    # Phase 4: Final intensive polish on best
    n = len(h_best)
    if time.time() < deadline - 20:
        # One more round of tiny perturbations
        for sc in [0.001, 0.0005, 0.0003, 0.0002]:
            if time.time() > deadline - 15:
                break
            for attempt in range(2):
                if time.time() > deadline - 10:
                    break
                hp = focal_perturb(h_best, n, rng, sc, 0.05)
                h_o, c_o = exact_cd(hp, n, 3000, deadline, rng)
                if c_o < best_c:
                    h_best, best_c = h_o, c_o
        # Final CD polish
        h_f, c_f = exact_cd(h_best, n, 5000, deadline, rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
