        if abs(diff) < 1e-14:
            break
        if diff > 0:

            if d_hi - d_lo < 1e-15:
                continue
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

def subgradient_polish(h, n, n_steps, deadline, delta_frac=0.005):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    dx = 2.0 / n
    for s in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        threshold = c_max - delta_frac * c_max
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        grad = np.zeros(n)
        for k_idx in active:
            shift = int(k_idx) - (n - 1)
            i_min = max(0, -shift)
            i_max = min(n, n - shift)
            S = np.arange(i_min, i_max)
            if len(S) == 0:
                continue
            grad[S] += (1.0 - h[S + shift])
            S2 = S + shift
            S2_valid = S2[(S2 >= 0) & (S2 < n)]
            if len(S2_valid) > 0:
                grad[S2_valid] -= h[S2_valid - shift]
        grad /= len(active)
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.6
            if step < 1e-5:
                break
    return h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    alphas_p1 = [500, 2000, 10000, 100000]
    h1, c1 = slsqp_optimize(best_h, n, alphas_p1, 300)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: CD allowing bounds to move
    h2, c2 = fast_coord_descent(best_h, n, 80, deadline, gs_iters=55, allow_bounds=True)
    if c2 < best_c:
        best_h, best_c = h2, c1 if c2 < best_c else best_c

    # Phase 3: Subgradient + CD alternation
    for round_i in range(4):
        if time.time() > deadline - 60:
            break
        h3, c3 = subgradient_polish(best_h, n, 80, deadline, delta_frac=0.003)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h4, c4 = fast_coord_descent(best_h, n, 40, deadline, gs_iters=65, allow_bounds=True)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 4: Try different resolutions
    for n_try in [500, 600, 800, 900, 1000]:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_optimize(h_int, n_try, [2000, 20000, 200000], 200)
        h_o, c_o = fast_coord_descent(h_o, n_try, 50, deadline, gs_iters=55, allow_bounds=True)
        h_o, c_o = subgradient_polish(h_o, n_try, 40, deadline, delta_frac=0.003)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 5: Multi-restart with diverse perturbations
    remaining = deadline - time.time() - 30
    if remaining > 60:
        n_cur = len(best_h)
        x_grid = np.linspace(0, 2, n_cur)
        n_restarts = max(2, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            if r % 4 == 0:
                s = 0.02 + 0.02 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            elif r % 4 == 1:
                freq = 3.0 + 2.0 * ((r // 4) / max(1, n_restarts // 4))
                hp = project_feasible(best_h + 0.04 * np.sin(np.pi * freq * x_grid), n_cur)
            elif r % 4 == 2:
                freq = 5.0 + 3.0 * ((r // 4) / max(1, n_restarts // 4))
                hp = project_feasible(best_h + 0.03 * np.cos(2 * np.pi * freq * x_grid), n_cur)
            else:
                idx = rng.choice(n_cur, size=n_cur // 5, replace=False)
                hp = best_h.copy()
                hp[idx] = rng.uniform(0, 1, len(idx))
                hp = project_feasible(hp, n_cur)
            h_o, c_o = slsqp_optimize(hp, n_cur, [5000, 50000], 200)
            h_o, c_o = fast_coord_descent(h_o, n_cur, 30, deadline, gs_iters=50, allow_bounds=True)
            h_o, c_o = subgradient_polish(h_o, n_cur, 30, deadline, delta_frac=0.002)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final precision CD polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        remaining2 = deadline - time.time() - 8
        max_passes = max(10, int(remaining2 / 2.0))
        h_f, c_f = fast_coord_descent(best_h, n_cur, max_passes, deadline, gs_iters=80, allow_bounds=True)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final SLSQP micro-polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [500000, 2000000], 500)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final CD
    if time.time() < deadline - 2:
        h_f, c_f = fast_coord_descent(best_h, len(best_h), 20, deadline, gs_iters=70, allow_bounds=True)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
