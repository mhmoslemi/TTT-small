import numpy as np
import math
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * n
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += min(1.0, d / tot) * room
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h += max(-1.0, d / tot) * h
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.flatnonzero((h > 1e-12) & (h < 1.0 - 1e-12))
        if idx.size:
            h[int(idx[0])] += d
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_fg(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = float(c.max())
    e = np.exp(alpha * (c - mc))
    z = float(e.sum())
    if z <= 0.0:
        z = 1.0
    w = e / z
    g = np.zeros(n, dtype=float)
    cen = n - 1
    dx = 2.0 / n
    for m in range(2 * n - 1):
        wt = float(w[m])
        if wt <= 1e-12:
            continue
        d = m - cen
        L = n - abs(d)
        if L <= 0:
            continue
        if d >= 0:
            g[:L] += wt * (1.0 - h[d:d + L])
            g[d:] += -wt * h[:L]
        else:
            a = -d
            g[a:] += wt * (1.0 - h[:L])
            g[:L] += -wt * h[a:]
    return dx * (mc + math.log(z) / alpha), dx * g

def slsqp_f(x, n, alpha):
    return smooth_fg(x, n, alpha)[0]

def slsqp_g(x, n, alpha):
    return smooth_fg(x, n, alpha)[1]

def sum_eq(x, n):
    return x.sum() - 0.5 * n

def sum_eq_jac(x, n):
    return np.ones(n, dtype=float)

def slsqp_opt(h, n, alpha, maxiter=45):
    h = project_box_sum(h, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac, "args": (n,)}]
    try:
        r = minimize(slsqp_f, h, args=(n, alpha), method="SLSQP",
                     bounds=bnds, constraints=cons, jac=slsqp_g,
                     options={"maxiter": maxiter, "ftol": 1e-12})
        x = np.asarray(r.x, dtype=float)
        if not np.all(np.isfinite(x)):
            x = h
        else:
            x = project_box_sum(x, n)
    except Exception:
        x = h
    return x, compute_c5(x, n)

def env_at(delta, ha, hb, hx):
    i = int(np.searchsorted(hx, delta, side="right")) - 1
    if i < 0:
        i = 0
    elif i >= len(ha):

    return ha[i] * delta + hb[i]

def cd_full(h, n, deadline, max_passes, proj_every=30, seed=0, cap=0.35):
    h = project_box_sum(np.asarray(h, dtype=float), n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur
    rng = np.random.RandomState(seed)
    stale = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        h_start = h.copy()
        c_start = c.copy()
        start_cur = cur
        pass_best_h = h.copy()
        pass_best = cur
        for cnt, p in enumerate(order):
            if time.time() > deadline:
                break
            hv = float(h[p])
            lo = max(-hv, -cap)
            hi = min(1.0 - hv, cap)
            if hi - lo <= 1e-14:
                continue
            delta, bud, a = solve_coord(c, h, p, n, lo, hi)
            if abs(delta) > 1e-10 and bud < cur - 1e-9:
                c += delta * a
                c[n - 1] -= delta * delta
                nv = hv + delta
                if nv < 0.0:
                    nv = 0.0
                elif nv > 1.0:
                    nv = 1.0
                h[p] = nv
                cur = float(c.max())
                if (cnt + 1) % proj_every == 0 or cnt == n - 1:
                    h = project_box_sum(h, n)
                    c = np.correlate(h, 1.0 - h, mode="full")
                    cur = float(c.max())
                    if cur < pass_best - 1e-11:
                        pass_best = cur
                        pass_best_h = h.copy()
        h = project_box_sum(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        cur = float(c.max())
        if cur < pass_best:
            pass_best = cur
            pass_best_h = h.copy()
        if pass_best < best_c - 1e-11:
            h = pass_best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full")
            cur = float(c.max())
            best_h = h.copy()
            best_c = cur
            stale = 0
        else:
            h = h_start.copy()
            c = c_start.copy()
            cur = start_cur
            stale += 1
            if stale >= 3:
                break
    return best_h, best_c * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(600, 0.5, dtype=float)
    n = len(h)
    if n > 950:
        n = 900
        x0 = np.linspace(0.0, 2.0, len(h))
        x1 = np.linspace(0.0, 2.0, n)
        h = np.interp(x1, x0, h)
    h = project_box_sum(h, n)
    c = compute_c5(h, n)

    for vh in (1.0 - h, h[::-1], 0.5 * (h + h[::-1])):
        vh = project_box_sum(vh, n)
        vc = compute_c5(vh, n)
        if vc < c:
            h, c = vh, vc

    left = deadline - time.time()
    if left > 60:
        h2, c2 = cd_full(h, n, deadline - 260, 300, 30, seed, 0.35)
        if c2 < c:
            h, c = h2, c2

    if time.time() < deadline - 130:
        for alpha in (1000.0, 20000.0, 150000.0):
            if time.time() > deadline - 90:
                break
            hs, cs = slsqp_opt(h, n, alpha, 40)
            if cs < c:
                h, c = hs, cs
            if c <= 0.38080:
                break
            if time.time() > deadline - 130:
                break
            h2, c2 = cd_full(h, n, deadline - 130, 8, 30, seed + 1, 0.15)
            if c2 < c:
                h, c = h2, c2

    if time.time() < deadline - 70:
        rng = np.random.RandomState(seed + 7)
        variants = [1.0 - h, h[::-1], 0.5 * (h + h[::-1])]
        xb = np.linspace(0.0, 2.0, n)
        for sc in (0.02, 0.05, 0.10):
            hh = h.copy()
            for k in range(1, 6):
                hh += (sc / (k + 1.0)) * rng.normal(0.0, 1.0) * np.sin(math.pi * k * xb + rng.normal(0.0, 1.0))
            variants.append(hh)
        for vv in variants:
            if time.time() > deadline - 25:
                break
            vv = project_box_sum(vv, n)
            vc = compute_c5(vv, n)
            if vc < c:
                h, c = vv, vc
            else:
                rem = deadline - time.time()
                if rem > 25:
                    sub = min(deadline - 10, time.time() + max(10.0, 0.25 * rem))
                    h2, c2 = cd_full(vv, n, sub, 5, 30, seed + 5, 0.25)
                    if c2 < c:
                        h, c = h2, c2

    if time.time() < deadline - 15:
        h2, c2 = cd_full(h, n, deadline - 4, 150, 30, seed + 2, 0.12)
        if c2 < c:
            h, c = h2, c2

    if c > 0.38080 and time.time() < deadline - 25:
        hs, cs = slsqp_opt(h, n, 200000.0, 30)
        if cs < c:
            h, c = hs, cs
        if time.time() < deadline - 8:
            h2, c2 = cd_full(h, n, deadline - 2, 40, 30, seed + 3, 0.08)
            if c2 < c:
                h, c = h2, c2

    for vh in (1.0 - h, h[::-1]):
        vh = project_box_sum(vh, n)
        vc = compute_c5(vh, n)
        if vc < c:
            h, c = vh, vc
    h = project_box_sum(h, n)
    return h, compute_c5(h, n), n
