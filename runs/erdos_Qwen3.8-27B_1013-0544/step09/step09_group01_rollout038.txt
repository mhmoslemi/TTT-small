import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_fast(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    obj = dx * (m + (mx + np.log(w.sum())) / alpha)
    # Gradient
    q = (1.0 - h)[::-1]
    first = np.correlate(w, q, mode='full')[:n]
    second = np.correlate(w, h, mode='full')[:n][::-1]
    grad = dx * (first - second)
    return obj, grad

def projected_gradient(h, n, alpha, steps, lr, deadline):
    h = h.copy()
    t = 1.0
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(steps):
        if time.time() > deadline:
            break
        obj, grad = smooth_obj_and_grad(h, n, alpha)
        # Nesterov
        mu = (t - 1.0) / (t + 1.0 + np.sqrt(1.0 + 4.0 * t * t))
        # Actually standard Nesterov: t_new = (1+sqrt(1+4*t^2))/2, mu = (t-1)/t_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        y = h + mu * (h - h_prev)
        # Gradient step on y
        _, grad_y = smooth_obj_and_grad(y, n, alpha)
        h_new = y - lr * grad_y
        h_new = project_fast(h_new, n)
        h_prev = h
        h = h_new
        t = t_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_fast(h, n)
    return h, compute_c5(h, n)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)

    best_h = project_fast(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Projected gradient with Nesterov
    alphas = [50, 200, 1000, 5000, 20000]
    for alpha in alphas:
        if time.time() > deadline - 200:
            break
        steps = max(100, int((deadline - time.time() - 200) / 0.05))
        # Estimate step size
        h_test = best_h + 0.01 * rng.randn(n)
        h_test = project_fast(h_test, n)
        _, g1 = smooth_obj_and_grad(best_h, n, alpha)
        _, g2 = smooth_obj_and_grad(h_test, n, alpha)
        diff_g = np.linalg.norm(g1 - g2)
        diff_x = np.linalg.norm(h_test - best_h)
        L = max(diff_g / max(diff_x, 1e-10), 1.0)
        lr = 1.0 / L
        h1, c1 = projected_gradient(best_h, n, alpha, steps, lr, deadline - 200)
        if c1 < best_c:
            best_h, best_c = h1, c1
    
    # Phase 2: Coordinate descent
    h2, c2 = coord_descent(best_h, n, 40, deadline - 100, gs_iters=50)
    if c2 < best_c:
        best_h, best_c = h2, c2
    
    # Phase 3: Try different resolutions
    for n_try in [500, 800]:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        x_orig = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_fast(np.interp(x_new, x_orig, best_h), n_try)
        for alpha in [1000, 10000]:
            if time.time() > deadline - 150:

            _, g = smooth_obj_and_grad(h_int, n_try, alpha)
            L = np.linalg.norm(g) * 10 + 1
            lr = 1.0 / L

            h_o, c_o = coord_descent(h_o, n_try, 30, deadline - 100, gs_iters=45)
        if c_o < best_c:
