```python
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
```

```python
def solve_1d_linear_quad(c_aff, a_coeff, b_coeff, unaff_max, lo, hi):
    """Solve min_d max_k (c_aff[k] + a_coeff[k]*d + b_coeff[k]*d^2) subject to lo <= d <= hi"""
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [lo, hi, 0.0]
    # Pairwise intersections of affine-quadratic functions
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            db = b_coeff[i] - b_coeff[j]
            dc = c_aff[j] - c_aff[i]

                if abs(da) > 1e-14:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:

                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2*db), (-da - sq) / (2*db)]:
                        if lo < d < hi:
                            candidates.append(d)
    # Also find vertex of each quadratic (if it's within [lo, hi])
    for i in range(k):
        if abs(b_coeff[i]) > 1e-14:
            dv = -a_coeff[i] / (2 * b_coeff[i])

                if res.fun < best_val - 1e-15:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:

    return best_d
```

```python
# This is the "valid" part of the correlation of p and neg_2x
# np.correlate(p, neg_2x, 'valid') gives n + (2n-1) - 1 - (n-1) = ... 
# Actually, 'valid' mode: the output has length max(M, N) - min(M, N) + 1 where M, N are the
# lengths of the two arrays. p has length 2n-1, neg_2x has length n.
# So 'valid' gives (2n-1) - n + 1 = n elements.
```

Let me verify: np.correlate(a, v, 'valid')[j] = sum_i a[i+j] * v[i] where the sum is over valid i.
With a = p (length 2n-1) and v = neg_2x (length n):
result[j] = sum_{i=0}^{n-1} p[i+j] * neg_2x[i] = sum_{q=0}^{n-1} p[q+j] * neg_2x[q] ✓

So the gradient is:
```python
grad = (2.0/n) * np.correlate(p, (1.0 - 2.0*x), mode='valid')
```

This is O(n log n) with FFT or O(n²) directly. For n=600, O(n²) is fine.

```python
def slsqp_with_grad(h, n, alphas, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def obj_and_grad(x):
            dx = 2.0 / n
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            mx = c.max()
            shifted = alpha * (c - mx)
            sm = shifted.max()
            exp_s = np.exp(shifted - sm)
            total = exp_s.sum()
            p = exp_s / total
            val = mx + (sm + np.log(total)) / alpha
            neg_2x = 1.0 - 2.0 * x
            grad = dx * np.correlate(p, neg_2x, mode='valid')
            return val, grad
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_and_grad, best_h, jac=True, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
```
