from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):

def coord_descent(h, n, max_passes, deadline, rng, random_order=True):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
    return h, best_c

def pair_transfer(h, n, deadline, rng, n_ops=5000):
    h = h.copy()
    c_cur = compute_c5(h, n)
    deltas = np.array([0.02, 0.05, 0.1, 0.2, 0.4, -0.02, -0.05, -0.1, -0.2, -0.4])
    for _ in range(n_ops):
        if time.time() > deadline:
            break
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        for d in deltas:
            h_test = h.copy()
            h_test[i] = h[i] + d
            h_test[j] = h[j] - d
            if h_test[i] < -1e-12 or h_test[i] > 1 + 1e-12:
                continue
            if h_test[j] < -1e-12 or h_test[j] > 1 + 1e-12:
                continue
            h_test = project_feasible(h_test, n)
            c_new = compute_c5(h_test, n)
            if c_new < c_cur - 1e-14:
                h = h_test
                c_cur = c_new
                break
    return h, c_cur

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
