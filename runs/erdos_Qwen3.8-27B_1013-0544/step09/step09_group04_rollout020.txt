def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(h, c, p, n):

    h0 = h[p]

    m_hi = n - 1 + p

    jb = p - ms + (n - 1)

def exact_cd(h, n, deadline, max_passes=500):
    """Exact coordinate descent with full recompute per pass."""
    h = h.copy()
    for passed in range(max_passes):
        if time.time() > deadline - 5:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline - 5:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            x_new = solve_coord(h, c, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_phase(h, n, alphas, deadline, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline - 60:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            s = alpha * (c - m)
            s = s - s.max()
            return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
