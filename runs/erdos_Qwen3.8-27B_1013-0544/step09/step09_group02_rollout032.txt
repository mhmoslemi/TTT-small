def lse_bfgs(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda n=n: np.ones(n)}]
        try:
            res = minimize(lse_fg, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15, 'gtol': 1e-8})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
