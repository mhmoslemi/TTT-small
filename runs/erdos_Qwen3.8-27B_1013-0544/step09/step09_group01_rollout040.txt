import numpy as np
from scipy.optimize import minimize
import time, math


def _c5_exact(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def _proj_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = len(y)
    target = float(target)
    if target <= 0.0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = x < 1.0 - 1e-15
            if not m.any():
                break
            x[m] = np.clip(x[m] + diff / float(np.count_nonzero(m)), 0.0, 1.0)
        else:
            m = x > 1e-15
            if not m.any():
                break
            x[m] = np.clip(x[m] + diff / float(np.count_nonzero(m)), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - x)) if diff > 0 else int(np.argmin(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x


def _fft_len(m):
    m = int(m)
    return 1 if m <= 1 else 1 << (m - 1).bit_length()


def _grad_weights(p, h):
    n = len(h)
    L = 2 * n - 1
    off = n - 1
    size = _fft_len(L + n - 1)
    A1 = np.zeros(size); A1[:L] = p
    B1 = np.zeros(size); B1[:n] = 1.0 - h
    A2 = np.zeros(size); A2[:L] = p[::-1]
    B2 = np.zeros(size); B2[:n] = h
    g1 = np.fft.irfft(np.fft.rfft(A1) * np.fft.rfft(B1), size)[off:off + n]
    g2 = np.fft.irfft(np.fft.rfft(A2) * np.fft.rfft(B2), size)[off:off + n]
    return g1 - g2


def _smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    L = 2 * n - 1
    if alpha < 1.0:
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        val = (2.0 / n) * m
    else:
        shifted = alpha * (c - m)
        with np.errstate(under='ignore'):
            e = np.exp(shifted)
        S = float(e.sum())
        if S <= 0.0 or not np.isfinite(S):
            w = np.zeros(L)
            w[int(np.argmax(c))] = 1.0
            val = (2.0 / n) * m
        else:
            w = e / S
            val = (2.0 / n) * (m + math.log(S) / alpha)
    if not np.isfinite(val):
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        val = (2.0 / n) * m
    return float(val), (2.0 / n) * _grad_weights(w, x)


def _lbfgs_fun(x, n, alpha, lam):
    val, grad = _smooth_val_grad(x, n, alpha)
    s = float(x.sum() - 0.5 * n)
    if lam != 0.0:
        val += lam * s * s
        grad += 2.0 * lam * s
    return val, grad


def _lbfgs_optimize(h, n, alpha, lam, maxiter, deadline):
    if time.time() > deadline - 8:
        return h, _c5_exact(h, n)
    maxiter = int(min(maxiter, max(40, (deadline - time.time()) * 0.45)))
    if maxiter <= 0:
        return h, _c5_exact(h, n)
    try:
        res = minimize(
            _lbfgs_fun, h, args=(n, alpha, lam),
            method='L-BFGS-B', jac=True,
            bounds=[(0.0, 1.0)] * n,
            options={
                'maxiter': maxiter,
                'maxfun': maxiter * 3,
                'maxls': 40,
                'ftol': 1e-11,
                'gtol': 1e-5,
                'maxcor': 50,
            }
        )
        x = np.asarray(res.x, dtype=float)
    except Exception:
        return h, _c5_exact(h, n)
    if not np.all(np.isfinite(x)):
        return h, _c5_exact(h, n)
    x = _proj_box_sum(x, 0.5 * n)
    return x, _c5_exact(x, n)


def _lin_pair(h, p, q, k, n):
    L = len(k)
    lin = np.zeros(L, dtype=float)
    idx = p - k
    m = (idx >= 0) & (idx < n)
    if m.any():
        lin[m] += 1.0 - h[idx[m]]
    idx = p + k
    m = (idx >= 0) & (idx < n)
    if m.any():
        lin[m] -= h[idx[m]]
    idx = q - k
    m = (idx >= 0) & (idx < n)
    if m.any():
        lin[m] -= 1.0 - h[idx[m]]
    idx = q + k
    m = (idx >= 0) & (idx < n)
    if m.any():
        lin[m] += h[idx[m]]
    return lin


def _pair_fval(c, lin, d, m0, m1, m2, temp):
    d = float(d)
    np.multiply(lin, d, out=temp)
    temp += c
    temp[m0] = c[m0] + d * lin[m0] - 2.0 * d * d
    temp[m1] = c[m1] + d * lin[m1] + d * d
    if m2 != m1:
        temp[m2] = c[m2] + d * lin[m2] + d * d
    return float(np.max(temp))


def _pair_cd(h, n, offsets, time_s, deadline, gs_iters=24, step_limit=0.1):
    h = h.copy()
    stop = min(deadline, time.time() + float(time_s))
    L = 2 * n - 1
    k = np.arange(-n + 1, n, dtype=int)
    m0 = n - 1
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    temp = np.empty(L, dtype=float)
    offs = []
    for o in offsets:
        o = int(o) % n
        if 0 < o < n and o not in offs:
            offs.append(o)
    for _ in range(100000):
        if time.time() > stop:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for off in offs:
            if time.time() > stop:
                break
            for p in range(n):
                if time.time() > stop:
                    break
                q = (p + off) % n
                hp = h[p]
                hq = h[q]
                dlo = max(-hp, hq - 1.0, -step_limit)
                dhi = min(1.0 - hp, hq, step_limit)
                if dhi - dlo < 1e-14:
                    continue
                lin = _lin_pair(h, p, q, k, n)
                m1 = m0 + p - q
                m2 = m0 + q - p
                f0 = float(np.max(c))
                dl, dh = dlo, dhi
                d1 = dh - gr * (dh - dl)
                d2 = dl + gr * (dh - dl)
                f1 = _pair_fval(c, lin, d1, m0, m1, m2, temp)
                f2 = _pair_fval(c, lin, d2, m0, m1, m2, temp)
                for _ in range(gs_iters):
                    if dh - dl < 1e-15:
                        break
                    if f1 < f2:
                        dh = d2
                        d2 = d1
                        f2 = f1
                        d1 = dh - gr * (dh - dl)
                        f1 = _pair_fval(c, lin, d1, m0, m1, m2, temp)
                    else:
                        dl = d1
                        d1 = d2
                        f1 = f2
                        d2 = dl + gr * (dh - dl)
                        f2 = _pair_fval(c, lin, d2, m0, m1, m2, temp)
                dopt = 0.5 * (dl + dh)
                bestd = 0.0
                bestf = f0
                for cand in (dopt, 0.5 * (dlo + dhi), dlo, dhi):
                    fv = _pair_fval(c, lin, float(cand), m0, m1, m2, temp)
                    if fv < bestf:
                        bestf = fv
                        bestd = float(cand)
                if bestf < f0 - 1e-12 and bestd != 0.0:
                    d = bestd
                    h[p] += d
                    h[q] -= d
                    c += d * lin
                    c[m0] -= 2.0 * d * d
                    c[m1] += d * d
                    if m2 != m1:
                        c[m2] += d * d
                    improved = True
        if not improved:
            break
    h = _proj_box_sum(h, 0.5 * n)
    return h, _c5_exact(h, n)

def _interp_to_n(h, n, n2):
    x1 = np.linspace(0.0, 2.0, n)
    x2 = np.linspace(0.0, 2.0, n2)
    y = np.interp(x2, x1, h)
    return _proj_box_sum(y, 0.5 * n2)
