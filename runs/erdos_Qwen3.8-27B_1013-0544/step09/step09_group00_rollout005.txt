def thue_morse_initial(n, rng):
    # Generate Thue-Morse sequence of length n
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    return project_feasible(h, n)

def smooth_obj_topk(x, n, alpha, k_frac=0.1):
    """Smooth objective focusing on top-k correlations"""
    c = np.correlate(x, 1.0 - x, mode='full')
    sorted_c = np.sort(c)[::-1]
    k = max(1, int(k_frac * len(c)))
    top_k = sorted_c[:k]
    # Use log-sum-exp on top k as smooth max
    mx = top_k.max()
    lse = mx + np.log(np.sum(np.exp(alpha * (top_k - mx)))) / alpha
    return lse * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline, k_fracs=[0.1, 0.3, 0.5]):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        for kf in k_fracs:
            if time.time() > deadline:
                break
            bounds = [(0.0, 1.0)] * n
            def _cons_fun(x, n_=n):
                return x.sum() - n_ / 2.0
            def _cons_jac(x, n_=n):
                return np.ones(n_)
            cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
            try:
                res = minimize(smooth_obj_topk, best_h, args=(n, alpha, kf),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': maxiter, 'ftol': 1e-16})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c

        h_soft = h.copy()
        # Push toward binary
        for i in range(n):

    return best_h, best_c

        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = randomized_cd(h_int, n_try, 80, deadline, rng)
        if c_o < best_c:
            h_best, best_c, n = h_o, c_o, n_try
    
    # Phase 4: Try Thue-Morse based starts
    if time.time() < deadline - 150:
        for n_try in [n, 500, 800]:
            if time.time() > deadline - 100:
                break
            h_tm = thue_morse_initial(n_try, rng)
            h_o, c_o = randomized_cd(h_tm, n_try, 100, deadline, rng)
            if c_o < best_c:
                h_best, best_c, n = h_o, c_o, n_try
            # Also try reversed Thue-Morse
            h_rev = h_tm[::-1].copy()
            h_rev = project_feasible(h_rev, n_try)
            h_o2, c_o2 = randomized_cd(h_rev, n_try, 50, deadline, rng)
            if c_o2 < best_c:
                h_best, best_c, n = h_o2, c_o2, n_try
    
    # Phase 5: Annealing
    if time.time() < deadline - 100:
        n = len(h_best)
        h5, c5 = anneal_cd(h_best, n, deadline, rng)
        if c5 < best_c:
            h_best, best_c = h5, c5
    
    # Phase 6: Final intensive CD
    if time.time() < deadline - 5:
        n = len(h_best)
        remaining = deadline - time.time()
        # Estimate how many passes we can do
        # Each pass is roughly O(n^2) for the bisection
        h_f, c_f = randomized_cd(h_best, n, 200, deadline, rng)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
    
    # Final check
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo = unaff
    if np.any(zero_mask):
        L_lo = max(L_lo, float(np.max(c_base[zero_mask])))
    L_hi = cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                x_hi = min(x_hi, r1) if (h0 - r1) <= (r2 - h0) else x_hi
                x_lo = max(x_lo, r2) if (h0 - r1) > (r2 - h0) else x_lo
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if (h0 - r1) <= (r2 - h0) else x_hi
                x_lo = max(x_lo, r2) if (h0 - r1) > (r2 - h0) else x_lo
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def randomized_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def thue_morse_initial(n, rng):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    if tm.sum() > n / 2:
        tm = 1 - tm
    h = tm.astype(float)
    return project_feasible(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    sorted_c = np.sort(c)[::-1]
    k = max(1, int(0.2 * len(c)))
    top_k = sorted_c[:k]
    mx = top_k.max()
    lse = mx + np.log(np.sum(np.exp(alpha * (top_k - mx)))) / alpha
    return lse * (2.0 / n)

def slsqp_opt(h, n, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in [3000, 15000, 80000]:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def _cf(x, n_=n):
            return x.sum() - n_ / 2.0
        def _cj(x, n_=n):
            return np.ones(n_)
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def anneal_cycle(h, n, deadline, rng):
    best_h, best_c = h.copy(), compute_c5(h, n)
    temps = [0.08, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.004]
    for temp in temps:
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, temp, n)
        h_pert = project_feasible(h_pert, n)
        h_new, c_new = randomized_cd(h_pert, n, 80, deadline, rng)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        # Binary push variant
        h_bin = h.copy()
        h_bin[h_bin > 0.5] = np.minimum(1.0, h_bin[h_bin > 0.5] + temp * 2)
        h_bin[h_bin <= 0.5] = np.maximum(0.0, h_bin[h_bin <= 0.5] - temp * 2)
        h_bin = project_feasible(h_bin, n)
        h_new2, c_new2 = randomized_cd(h_bin, n, 50, deadline, rng)
        if c_new2 < best_c:
            best_h, best_c = h_new2, c_new2
    return best_h, best_c

    rng = np.random.RandomState(seed)
