import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_top_m_vec(h, n, m):

    top_idx = np.argsort(c)[-m:]
    offsets = top_idx - (n - 1)
    
    grad = np.zeros(n)
    for offset in offsets:
        if offset == 0:
            grad += (1 - 2*h)
        elif offset > 0:
            # i - offset >= 0 means i >= offset
            # i - offset < n means i < n + offset (always true if offset >= 0 and i < n)
            grad[offset:] += (1 - h[:n-offset])

            grad[:n-offset] -= h[offset:]

            grad[:n-o] += (1 - h[o:])
            # i + offset = i - o >= 0 means i >= o

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    """Solve 1D min: min_d max(c_aff + d*a_coeff, unaff_max) with optional -d^2 term."""
    k = len(a_coeff)
    candidates = [lo, hi, 0.0]
    # Add crossing points
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da*da + 4.0*dd*dc
                if disc >= 0:

                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    # Also add crossing with unaff_max

                a2 = 1.0

                c2 = unaff_max - c_aff[i]

    return best_d

            c = old_c.copy()

Hmm, but the issue is that after projection, c changes in a way that's not just a single-index update. The projection changes all h values (by a small amount to maintain the sum constraint). So the "incremental" update becomes inaccurate.

One approach: instead of projecting after each step, only project periodically (every K steps). The sum constraint deviation accumulates slowly if the changes are small.

Another approach: adjust only h[idx] and one other index to maintain the sum constraint. This is a "two-index" SA.

Let me try the two-index approach:
- Pick two random indices i and j
- Propose: h[i] += delta, h[j] -= delta (maintains sum)
- This keeps the sum constraint satisfied without needing projection
- The only remaining constraint is 0 <= h[k] <= 1

This is much cleaner! Let me implement this:

```python
def sa_two_index(h, n, t0=0.005, t_end=1e-8, max_iter=50000, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c = float(c.max()) * (2.0/n)
    best_h = h.copy()
    c_max = float(c.max())
    rng = np.random.RandomState(42)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        T = t0 * (t_end/t0) ** (it / max(1, max_iter))
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        h_i = h[i]
        h_j = h[j]
        # Max delta limited by bounds
        max_d = min(0.05, 1.0 - h_i, h_j)
        if max_d < 1e-10:
            continue
        delta = rng.uniform(-max_d, max_d)
        if abs(delta) < 1e-12:
            continue
        
        # Change: h[i] += delta, h[j] -= delta
        # Delta_c: for each offset, the change in c[n-1+offset]
        # h[i] changes by +delta, h[j] changes by -delta
        # Effect on c at offset:
        # From h[i]: +delta * (1-h[i-offset]) - delta*h[i+offset]
        # From h[j]: -delta * (1-h[j-offset]) + delta*h[j+offset]
        
        offsets = np.arange(-(n-1), n)
        dc = np.zeros(2*n-1)
        
        # From i:
        j1 = i - offsets
        m1 = (j1 >= 0) & (j1 < n)
        dc[m1] += delta * (1 - h[j1[m1]])
        j2 = i + offsets
        m2 = (j2 >= 0) & (j2 < n)
        dc[m2] += -delta * h[j2[m2]]
        
        # From j:
        j1 = j - offsets
        m1 = (j1 >= 0) & (j1 < n)
        dc[m1] += -delta * (1 - h[j1[m1]])
        j2 = j + offsets
        m2 = (j2 >= 0) & (j2 < n)
        dc[m2] += delta * h[j2[m2]]
        
        c_new = c + dc
        new_max = float(c_new.max())
        delta_c5 = (new_max - c_max) * dx
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            h[i] += delta
            h[j] -= delta
            c = c_new
            c_max = new_max
            new_c5 = c_max * dx
            if new_c5 < best_c:
                best_c = new_c5
                best_h = h.copy()
        # else: reject, h unchanged
    
    return best_h, best_c
```
