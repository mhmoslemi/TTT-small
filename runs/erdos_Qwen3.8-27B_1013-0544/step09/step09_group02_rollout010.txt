import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]

    if mhi < 2 * n - 2:

    lo, hi = -h0, 1.0 - h0

def apply_delta(h, c, p, d, n):
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)

def coord_cd(h, n, max_passes=200, deadline=None, rng=None, shuffle=True):

def block_cd(h, n, block_size=2, max_passes=100, deadline=None, rng=None):
    """Block coordinate descent: optimize blocks of size `block_size` at a time."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(0, n - block_size + 1)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for start in order:
            # Try to improve the block h[start:start+block_size]
            h_block = h[start:start+block_size].copy()
            # Use Nelder-Mead on the block
            def block_obj(x):
                h_trial = h.copy()
                h_trial[start:start+block_size] = np.clip(x, 0, 1)
                h_trial = project_feasible(h_trial, n)
                # Check if the block values are still what we set
                if not np.allclose(h_trial[start:start+block_size], np.clip(x, 0, 1), atol=1e-10):
                    return best_c + 1.0  # penalty for projection changing the block
                return compute_c5(h_trial, n)
            
            from scipy.optimize import minimize
            try:
                res = minimize(block_obj, h_block, method='Nelder-Mead',
                              options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
                new_block = np.clip(res.x, 0, 1)
                h_trial = h.copy()
                h_trial[start:start+block_size] = new_block
                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5(h_trial, n)
                if c_trial < best_c - 1e-10:
                    h = h_trial
                    best_c = c_trial
                    improved = True
            except Exception:
                pass
        if not improved:
            break
        h = project_feasible(h, n)
    return h, best_c

            h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h_opt, c_opt = coord_cd(h_int, n_try, max_passes=80, deadline=deadline, rng=rng)
            if c_opt < best_c:
                # Could use this solution directly
                best_h = h_opt
                best_c = c_opt
                n = n_try
                x = np.arange(n) / n
                # Also try interpolating back to a standard n
                x_std = np.linspace(0, 2, 600)
                h_back = project_feasible(np.interp(x_std, x_new, h_opt), 600)
                h_b, c_b = coord_cd(h_back, 600, max_passes=100, deadline=deadline, rng=rng)
                if c_b < best_c:
                    best_h, best_c, n = h_b, c_b, 600
                    x = np.arange(n) / n
                    x_new = np.linspace(0, 2, n)

    # Phase 5: Final refinement - CD with many passes
    n = len(best_h)
    h_final, c_final = coord_cd(best_h, n, max_passes=400, deadline=deadline, rng=rng)
    if c_final < best_c:
        best_h, best_c = h_final, c_final

    # Phase 6: One more round of small perturbations + CD
    if time.time() < deadline - 20:
        for _ in range(30):
            if time.time() > deadline - 5:
                break
            s = rng.uniform(0.001, 0.008)
            h_p = project_feasible(best_h + s * rng.standard_normal(n), n)
            h_new, c_new = coord_cd(h_p, n, max_passes=50, deadline=deadline, rng=rng)
            if c_new < best_c - 1e-10:
                best_h, best_c = h_new, c_new

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
