from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    # Soft max
    scaled = alpha * c
    mx = scaled.max()
    return (mx + np.log(np.sum(np.exp(scaled - mx)))) / alpha * dx

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * c
    mx = scaled.max()
    exps = np.exp(scaled - mx)
    weights = exps / exps.sum()
    obj = (mx + np.log(exps.sum())) / alpha * dx
    
    grad = np.zeros(n)
    for p in range(n):
        k_min = n - 1 - p
        k_max = n - 1 + p
        ls = np.arange(k_min, k_max + 1)
        jf = p + ls - (n - 1)
        jb = p - ls + (n - 1)

        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib[vf] += (1.0 - x[jf[vf]])
        contrib[vb] -= x[jb[vb]]
        grad[p] = np.dot(weights[ls], contrib) * dx
    return obj, grad

    """Objective with penalty for sum constraint."""
    obj, grad = smooth_obj_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    obj += mu * s * s
    grad += 2 * mu * s * np.ones(n)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def get_weights(c, alpha):
    """Compute softmax weights for the smooth max."""
    scaled = alpha * c
    mx = scaled.max()
    exps = np.exp(scaled - mx)
    return exps / exps.sum()

def compute_grad(h, n, weights):
    """Compute gradient of the weighted sum of correlation terms."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        k_min = n - 1 - p
        k_max = n - 1 + p
        ls = np.arange(k_min, k_max + 1)
        jf = p + ls - (n - 1)
        jb = p - ls + (n - 1)
        contrib = np.zeros(len(ls))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib[vf] += (1.0 - h[jf[vf]])
        contrib[vb] -= h[jb[vb]]
        grad[p] = np.dot(weights[ls], contrib)
    return grad * dx

def fista_opt(h, n, alpha, steps, deadline, lr=None):
    """FISTA (Nesterov accelerated gradient) on the smooth objective with projection."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if lr is None:
        # Lipschitz constant estimate
        # The gradient is Lipschitz with constant ~ 4*(2n-1)*dx / alpha (rough)
        lr = 1.0 / (4.0 * (2 * n) / alpha)
        lr = min(lr, 0.5)
    
    h = h.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(steps):
        if time.time() > deadline:
            break
        c_y = np.correlate(y, 1.0 - y, mode='full')
        weights = get_weights(c_y, alpha)
        grad = compute_grad(y, n, weights)
        
        # Gradient step
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        # Note: we don't project onto sum constraint here;
        # instead we'll handle it separately
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        t = t_new
        h = h_new
        
        if it % 50 == 0:
            h_proj = project_feasible(h, n)
            c_val = compute_c5(h_proj, n)
            if c_val < best_c:
                best_c = c_val
                best_h = h_proj.copy()
    
    h_final = project_feasible(h, n)
    c_final = compute_c5(h_final, n)
    if c_final < best_c:
        best_h, best_c = h_final, c_final
    return best_h, best_c


def solve_1d_bisect(c, h, p, n):
    """Solve the 1D subproblem for coordinate p using bisection on L."""
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi >= r2:
                x_lo = max(x_lo, r2)
            elif x_lo <= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))


def exact_cd(h, n, max_passes, deadline):
    """Exact coordinate descent with incremental correlation update."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:

            xn = solve_1d_bisect(c, h, p, n)

def lbfgs_smooth(h, n, alpha, deadline, maxiter=3000):
    """L-BFGS-B optimization of the smooth objective."""
    def fun(x):

        scaled = alpha * c
        mx = scaled.max()
        exps = np.exp(scaled - mx)
        s = exps.sum()
        f = (mx + np.log(s)) / alpha * (2.0 / n)
        # Add penalty for sum constraint

            jb = p - ls + (n - 1)

            vb = (jb >= 0) & (jb < n)
            contrib[vf] += (1.0 - x[jf[vf]])
            contrib[vb] -= x[jb[vb]]
            grad[p] = np.dot(weights[ls], contrib)
        grad *= (2.0 / n)
        grad += 2e6 * (x.sum() - n / 2.0)
        return f, grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 2, 'ftol': 1e-18, 'gtol': 1e-8})
    return project_feasible(res.x, n)

    f += pen

    for p in range(n):

        k_max = n - 1 + p

def lbfgs_smooth(h, n, alpha, deadline, maxiter=3000):

def _sigmoid_fun(t, n, alpha, lam):
    h = 1.0 / (1.0 + np.exp(-t))

    scaled = alpha * c

    pen = lam * (h.sum() - n / 2.0) ** 2
    f += pen

    # dh/dt = h*(1-h)
    weights = exps / s
    grad_h = np.zeros(n)
    for p in range(n):
        k_min = n - 1 - p
        k_max = n - 1 + p

        grad_h[p] = np.dot(weights[ls], contrib)
    grad_h *= (2.0 / n)
    grad_h += 2 * lam * (h.sum() - n / 2.0)
    
    grad_t = grad_h * h * (1.0 - h)
    return f, grad_t

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _fb_fun(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * c
    mx = scaled.max()
    exps = np.exp(scaled - mx)
    s = exps.sum()
    f = (mx + np.log(s)) / alpha * dx
    pen = 1e5 * (x.sum() - n / 2.0) ** 2
    f += pen
    weights = exps / s
    grad = np.zeros(n)
    for p in range(n):
        k_min = n - 1 - p
        k_max = n - 1 + p
        ls = np.arange(k_min, k_max + 1)
        jf = p + ls - (n - 1)
        jb = p - ls + (n - 1)
        contrib = np.zeros(len(ls))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib[vf] += (1.0 - x[jf[vf]])
        contrib[vb] -= x[jb[vb]]
        grad[p] = np.dot(weights[ls], contrib)
    grad *= dx
    grad += 2e5 * (x.sum() - n / 2.0)
    return f, grad

def _sig_fun(t, n, alpha, lam):
    h = 1.0 / (1.0 + np.exp(-np.clip(t, -50, 50)))
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * c
    mx = scaled.max()
    exps = np.exp(scaled - mx)
    s = exps.sum()
    f = (mx + np.log(s)) / alpha * dx
    f += lam * (h.sum() - n / 2.0) ** 2
    weights = exps / s
    grad_h = np.zeros(n)
    for p in range(n):
        k_min = n - 1 - p
        k_max = n - 1 + p
        ls = np.arange(k_min, k_max + 1)
        jf = p + ls - (n - 1)
        jb = p - ls + (n - 1)
        contrib = np.zeros(len(ls))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib[vf] += (1.0 - h[jf[vf]])
        contrib[vb] -= h[jb[vb]]
        grad_h[p] = np.dot(weights[ls], contrib)
    grad_h *= dx
    grad_h += 2.0 * lam * (h.sum() - n / 2.0)
    grad_t = grad_h * h * (1.0 - h)
    return f, grad_t

def lbfgs_direct(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_fb_fun, h, args=(n, alpha), method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter * 2, 'ftol': 1e-18, 'gtol': 1e-8})
    if time.time() > deadline:
        return h
    return project_feasible(res.x, n)

def lbfgs_sigmoid(h, n, alpha, maxiter, deadline, lam=100.0):
    t0 = np.clip(np.log(h / (1.0 - h + 1e-15) + 1e-15), -30, 30)
    res = minimize(_sig_fun, t0, args=(n, alpha, lam), method='L-BFGS-B', jac=True,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 2, 'ftol': 1e-15, 'gtol': 1e-6})
    if time.time() > deadline:
        return h
    h_new = 1.0 / (1.0 + np.exp(-res.x))
    return project_feasible(h_new, n)

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_hi > r2:
                x_hi = r1
            elif x_lo < r1:
                x_lo = r2
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisection_safe(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def solve_1d_bisection_safe(c, h, p, n):
    try:
        return solve_1d_bisect(c, h, p, n)
    except Exception:
        return h[p]
