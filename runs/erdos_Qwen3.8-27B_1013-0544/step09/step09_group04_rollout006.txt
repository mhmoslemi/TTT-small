            break
        w /= wsum
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            wm = w[m]
            if wm < 1e-10:
                continue
            shift = m - (n - 1)
            if m == n - 1:
                lo, hi = 0, n - 1
                grad[lo:hi+1] += wm * (1.0 - 2.0 * h[lo:hi+1])
            else:
                lo1 = max(0, -shift)
                hi1 = min(n - 1, n - 1 - shift)
                if lo1 <= hi1:
                    grad[lo1:hi1+1] += wm * (1.0 - h[lo1+shift:hi1+1+shift])
                lo2 = max(0, shift)
                hi2 = min(n - 1, n - 1 + shift)
                if lo2 <= hi2:
                    grad[lo2:hi2+1] += wm * (-h[lo2-shift:hi2+1-shift])
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            if it % 200 == 0 and it > 0:
                eta = min(eta * 1.1, 0.5)
        else:
            h = h_new
            eta *= 0.7
    return best_h, best_c

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feas and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-16:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = abs(h0 - r1)
                d2 = abs(r2 - h0)
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-14:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_cd(h, n, max_passes, deadline, order='sequential'):
    h = h.copy()
    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'random':
            idx = n_idx.copy()
            idx = np.random.permutation(idx)
        elif order == 'sensitivity':
            sens = np.zeros(n)
            for p in range(n):
                m_lo = max(0, n - 1 - p)
                m_hi = min(2 * n - 2, n - 1 + p)
                seg = c[m_lo:m_hi + 1]
                sens[p] = np.max(seg) if len(seg) > 0 else 0.0
            idx = n_idx[np.argsort(-sens)]
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: SLSQP warmup
    h1, c1 = slsqp_run(best_h, n_init, [500, 3000, 15000], 300)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Projected gradient with cycling alpha
    for alpha in [200, 500, 1000, 3000]:
        if time.time() > deadline - 400:
            break
        n_iter = max(100, int((deadline - time.time() - 350) / 0.003))
        n_iter = min(n_iter, 5000)
        h2, c2 = projected_gradient(best_h, n_init, alpha, n_iter, 0.03, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Exact CD multiple orderings
    for _ in range(3):
        if time.time() > deadline - 200:
            break
        h3, c3 = coord_cd(best_h, n_init, 30, deadline, order='sequential')
        if c3 < best_c:
            best_h, best_c = h3, c3
    if time.time() < deadline - 100:
        h3b, c3b = coord_cd(best_h, n_init, 15, deadline, order='sensitivity')
        if c3b < best_c:
            best_h, best_c = h3b, c3b
    # Phase 4: Try other n values
    for n_try in [800, 1000, 900]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        h_int = project_feasible(interpolate_h(best_h, len(best_h), n_try), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [2000, 10000], 200)
        if time.time() < deadline - 60:
            h_pg, c_pg = projected_gradient(h_opt, n_try, 1000, 2000, 0.03, deadline)
            if c_pg < c_opt:
                h_opt, c_opt = h_pg, c_pg
        if time.time() < deadline - 40:
            h_cd, c_cd = coord_cd(h_opt, n_try, 15, deadline, order='sequential')
            if c_cd < c_opt:
                h_opt, c_opt = h_cd, c_cd
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 5: Interpolate back to best n for final polish
    n_best = len(best_h)
    best_h = project_feasible(best_h, n_best)
    # Phase 6: Final projected gradient + CD polish
    if time.time() < deadline - 20:
        h6, c6 = projected_gradient(best_h, n_best, 5000, 2000, 0.01, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6
    if time.time() < deadline - 10:
        max_passes = max(5, int((deadline - time.time() - 3) / 2.0))
        h7, c7 = coord_cd(best_h, n_best, max_passes, deadline, order='sequential')
        if c7 < best_c:
            best_h, best_c = h7, c7
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
