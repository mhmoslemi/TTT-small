import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 40:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_exact(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p)
    m_hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[di[0]]
        c_base[di] = 0.0
        a[di] = 0.0
    unaff = 0.0
    if ms[0] > 0:
        unaff = max(unaff, float(c[:ms[0]].max()))
    if ms[-1] < 2 * n - 2:
        unaff = max(unaff, float(c[ms[-1] + 1:].max()))
    cur_max = float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    L_lo = unaff
    L_hi = cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        feas = True
        if np.any(zero_m) and np.max(c_base[zero_m]) > L:
            feas = False
        if feasiblepseudo := feas:
            if np.any(neg_m):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feas and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-15:
            feas = False
        if feas and has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl = max(x_lo, 0.0)
                xh = min(x_hi, 1.0)
                if xl > r1 and xh < r2:
                    feas = False
        if feas:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = h0 - r1 if x_lo <= r1 else 1e9
                d2 = r2 - h0 if x_hi >= r2 else 1e9
                if d1 < d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-13:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None, order='sequential'):
    h = h.copy()
    n_idx = np.arange(n)
    for passed in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'random' and rng is not None:
            idx = n_idx.copy()
            rng.shuffle(idx)
        else:
            idx = n_idx
        improved = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            x_new = solve_1d_exact(c, h, p, n)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def upsample(h, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_new = np.interp(x_to, x_from, h)
    return project_feasible(h_new, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
