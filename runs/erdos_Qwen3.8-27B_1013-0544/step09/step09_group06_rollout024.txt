import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B + sq) / 2.0, (-B - sq) / 2.0]:
                    if lo <= d <= hi:
                        candidates.append(d)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        vals = c_peak + d * a_coeff
        if diag_idx is not None:
            vals[diag_idx] -= d * d
        v = max(float(vals.max()), unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

```python
def lbfgs_run(h, n, alpha, maxiter):
    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse / alpha) * (2.0 / n)
    
    def gradient(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        # Gradient of soft-max
        probs = np.exp(alpha * (c - c.max()))
        probs /= probs.sum()
        # d/dx_j of c[m] for each m
        # This is expensive to compute exactly, so we use the probabilities as weights
        grad = np.zeros(n)
        for m_idx in range(len(c)):
            if probs[m_idx] < 1e-10:
                continue
            lag = m_idx - (n - 1)
            # ∂c[m_idx]/∂h[j] for all j
            for j in range(n):
                val = 0.0
                jf = j - lag
                if 0 <= jf < n:
                    val += (1.0 - x[jf])
                jb = j + lag
                if 0 <= jb < n:
                    val -= x[jb]
                grad[j] += probs[m_idx] * val
        return grad * (2.0 / n)
    ...
```

The gradient computation is O(n * 2n) = O(n^2) per step, which for n=600 is 720K operations. With 500 iterations, that's 360M operations. This might be too slow in Python (due to the inner loop).

Let me vectorize the gradient computation:

For a given lag, ∂c[m]/∂h[j] = (1-h[j-lag]) * [0 ≤ j-lag < n] - h[j+lag] * [0 ≤ j+lag < n]

For a fixed lag, this is:
- The first term is a shifted version of (1-h): specifically, (1-h)[j-lag] for valid j
- The second term is a shifted version of h: specifically, h[j+lag] for valid j

So ∂c[m]/∂h = shifted(1-h, -lag) - shifted(h, lag) (with zero padding)

This can be computed efficiently using numpy slicing!

```python
def compute_c5_grad(c, h, n, probs):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for m_idx, w in enumerate(probs):
        if w < 1e-12:
            continue
        lag = m_idx - (n - 1)
        # First term: (1-h) shifted by -lag
        # (1-h)[j-lag] for j such that 0 ≤ j-lag < n
        # i.e., lag ≤ j < n+lag
        start1 = max(0, lag)
        end1 = min(n, n + lag)
        if end1 > start1:
            grad[start1:end1] += w * (1.0 - h[start1-lag:end1-lag])
        # Second term: -h[j+lag] for j such that 0 ≤ j+lag < n
        # i.e., -lag ≤ j < n-lag
        start2 = max(0, -lag)
        end2 = min(n, n - lag)
        if end2 > start2:
            grad[start2:end2] -= w * h[start2+lag:end2+lag]
    return grad
```

This is O(n * |active|) where |active| is the number of non-zero probabilities. If we use a small number of active constraints (top-k), this is fast.

Actually, for the soft-max with high alpha, only the top few correlations have significant probability. So |active| is small, and the gradient is fast.

```python
def lbfgs_softmax(h, n, alpha, maxiter):
    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        return (m + lse / alpha) * (2.0 / n)
    
    def gradient(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        probs = np.exp(s - s.max())
        probs /= probs.sum()
        # Only compute gradient for significant probabilities
        threshold = probs.max() * 1e-6
        active = np.where(probs > threshold)[0]
        grad = np.zeros(n)
        for m_idx in active:
            w = probs[m_idx]
            if w < 1e-12:
                continue
            lag = m_idx - (n - 1)
            # ∂c[m_idx]/∂h[j] = (1-h[j-lag])*[valid] - h[j+lag]*[valid]
            start1 = max(0, lag)
            end1 = min(n, n + lag)
            if end1 > start1:
                grad[start1:end1] += w * (1.0 - x[start1-lag:end1-lag])
            start2 = max(0, -lag)
            end2 = min(n, n - lag)
            if end2 > start2:
                grad[start2:end2] -= w * x[start2+lag:end2+lag]
        return grad * (2.0 / n)
    
    def constraint(x):
        return x.sum() - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, jac=gradient, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)
```

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        lse = mx + np.log(np.sum(np.exp(s - mx)))
        smooth_max = (m + lse / alpha) * (2.0 / n)
        penalty = rho * (x.sum() - n / 2.0) ** 2
        return smooth_max + penalty
    
    def gradient(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        probs = np.exp(s - s.max())
        probs /= probs.sum()
        threshold = probs.max() * 1e-6
        active = np.where(probs > threshold)[0]
        grad = np.zeros(n)
        for m_idx in active:
            w = probs[m_idx]
            if w < 1e-12:
                continue
            lag = m_idx - (n - 1)
            start1 = max(0, lag)
            end1 = min(n, n + lag)
            if end1 > start1:
                grad[start1:end1] += w * (1.0 - x[start1-lag:end1-lag])
            start2 = max(0, -lag)
            end2 = min(n, n - lag)
            if end2 > start2:
                grad[start2:end2] -= w * x[start2+lag:end2+lag]
        grad *= (2.0 / n)
        # Add penalty gradient
        grad += 2 * rho * (x.sum() - n / 2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, jac=gradient, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
