def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0, unaff_max
    
    def f(d):
        vals = c_aff + d * a_coeff
        if diag_local is not None:
            vals[diag_local] -= d * d
        return max(float(vals.max()), unaff_max)
    
    candidates = [lo, hi]
    if lo < 0 < hi:
        candidates.append(0.0)
    
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-14:
                if abs(da) > 1e-14:
                    d = dc / da
                    if lo <= d <= hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= -1e-14:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    d1 = (-da + sq) / (2.0 * dd)
                    d2 = (-da - sq) / (2.0 * dd)
                    if lo <= d1 <= hi:
                        candidates.append(d1)
                    if lo <= d2 <= hi:
                        candidates.append(d2)
    
    # Also add the vertex of each quadratic (where derivative = 0)
    for i in range(k):
        if diag_local == i:
            d_vertex = a_coeff[i] / 2.0  # derivative: a[i] - 2d = 0 => d = a[i]/2
            if lo <= d_vertex <= hi:
                candidates.append(d_vertex)
    
    best_d = lo
    best_val = f(lo)
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = f(d)
        if v < best_val:
            best_val = v
            best_d = d
    
    # Golden section refinement
    span = hi - lo
    if span > 1e-12:
        margin = min(span * 0.1, abs(best_d) * 0.3 + span * 0.05)
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(f, gl, gh, maxiter=80)
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
    
    return best_d, best_val

def coord_descent(h, n, deadline, top_k_start=20, top_k_end=50, max_passes=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    total_passes = max_passes
    n_active = 2 * n - 1
    
    for p in range(total_passes):
        if time.time() > deadline:
            break
        
        # Adaptive top_k
        frac = p / max(total_passes - 1, 1)
        top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        total_change = 0.0
        
        for idx in range(n):
            if time.time() > deadline:
                break
            
            h0 = h[idx]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            
            m_lo = min(idx, n - 1 - idx)
            m_hi = min(max(idx + n - 1, 2 * n - 2 - idx), 2 * n - 2)
            
            ms_all = np.arange(m_lo, m_hi + 1)
            
            if len(ms_all) > top_k:
                # Use argpartition for efficiency
                part = np.argpartition(c[ms_all], -top_k)[-top_k:]
                ms = ms_all[np.sort(part)]
                ms_set = set(ms.tolist())
                all_m_set = set(ms_all.tolist())
                skipped = np.array([m for m in ms_all if m not in ms_set])
                unaff_extra = float(c[skipped].max()) if len(skipped) > 0 else 0.0
            else:
                ms = ms_all
                unaff_extra = 0.0
            
            a = np.zeros(len(ms))
            i_vals = idx + n - 1 - ms
            vf = (i_vals >= 0) & (i_vals < n)
            if np.any(vf):
                a[vf] += (1.0 - h[i_vals[vf]])
            j_vals = idx + ms - (n - 1)
            vb = (j_vals >= 0) & (j_vals < n)
            if np.any(vb):
                a[vb] -= h[j_vals[vb]]
            
            c_aff = c[ms]
            
            unaff_max = unaff_extra
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            diag_local = None
            n1 = n - 1
            if m_lo <= n1 <= m_hi:
                # Check if n-1 is in ms
                pos = np.searchsorted(ms, n1)
                if pos < len(ms) and ms[pos] == n1:
                    diag_local = pos
            
            lo = -h0
            hi = 1.0 - h0
            
            best_d, best_val = solve_1d(c_aff, a, diag_local, unaff_max, lo, hi)
            
            if abs(best_d) > 1e-13:
                h[idx] = h0 + best_d
                improved = True
                total_change += abs(best_d)
                current_max = min(current_max, best_val)
        
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        
        if new_c < best_c:
            best_c = new_c
        if not improved:
            break
        if total_change < 1e-12:
            break
    
    return h, compute_c5(h, n)

def slsqp_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(slsqp_smooth, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    # Phase 2: SLSQP polish

            best_h, best_c = h2, c2
