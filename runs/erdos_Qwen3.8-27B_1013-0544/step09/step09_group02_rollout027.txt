        h_final, c_final = coord_cd(h_cd, n_try, 200, deadline, True, rng)
        if c_final < best_c:
            best_h, best_c = h_final, c_final
            x_cur = x_new

    # Phase 4: Targeted perturbations + CD at best resolution
    n_cur = len(best_h)
    for r in range(25):
        if time.time() > deadline - 60:
            break
        s = 0.002 + 0.004 * (r % 6)
        hp = targeted_perturb(best_h, n_cur, rng, s)
        hp = project_feasible(best_h + rng.normal(0, s * 0.3, n_cur), n_cur)
        h_r, c_r = coord_cd(hp, n_cur, 80, deadline, True, rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
            n_cur = len(best_h)

    # Phase 5: Final SLSQP with very high alpha + CD
    if time.time() < deadline - 40:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 30:
                break
            hs, cs = slsqp_with_grad(best_h, n_cur, alpha, 800, deadline)
            if cs < best_c:
                best_h, best_c = hs, cs
        hf, cf = coord_cd(best_h, n_cur, 500, deadline, True, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 6: One more SLSQP round
    if time.time() < deadline - 15:
        hs, cs = slsqp_with_grad(best_h, n_cur, 5000000, 1000, deadline)
        if cs < best_c:
            best_h, best_c = hs, cs
        hf, cf = coord_cd(best_h, n_cur, 300, deadline, True, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
