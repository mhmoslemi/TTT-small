    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):

def logsumexp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
    m = c.max()
    if m < 1e-15: return 0.0
    s = alpha * (c - m)
    smax = s.max()
    return m + (smax + np.log(np.exp(s - smax).sum())) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(150):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-13:
            lo = hi = mid

        active = vals >= mval - 1e-9

            lo = mid
        else:

            break

        for p in order:

                mhi = min(2*n-2, max(2*n-2-p, p+n-1))
                ms = np.arange(mlo, mhi+1)
                jf = p + ms - (n-1)
                jb = p - ms + (n-1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))

def logsumexp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    return m + (smax + np.log(np.exp(s - smax).sum())) / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(logsumexp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_polish(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.06
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Use weighted subgradient over top few lags
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]
        weights = np.zeros(len(c))
        for idx in top_idx:
            weights[idx] = c[idx] - c.min() + 1.0
        weights /= weights.sum()
        sg = np.zeros(n)
        for k in top_idx:
            w = weights[k]
            for i in range(n):
                j = int(i + k - (n-1))
                if 0 <= j < n:
                    sg[i] += w * (1.0 - h[j])
                ip = int(i - k + (n-1))
                if 0 <= ip < n:
                    sg[i] -= w * h[ip]
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.75
            stall += 1
            if stall > 35 or lr < 1e-10:
                break
    return h, best_c

def downsample_upsample(h, n_orig, n_coarse, deadline, rng):
    x_fine = np.linspace(0, 2, n_orig)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = project_feasible(np.interp(x_coarse, x_fine, h), n_coarse)
    h_opt, c_opt = coord_cd(h_coarse, n_coarse, max_passes=80, deadline=deadline, shuffle=True, rng=rng)
    x_new = np.linspace(0, 2, n_orig)
    h_up = project_feasible(np.interp(x_new, x_coarse, h_opt), n_orig)
    return h_up
