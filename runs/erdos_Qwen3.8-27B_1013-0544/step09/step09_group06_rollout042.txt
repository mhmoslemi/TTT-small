def exact_cd(h,n,deadline,max_passes=100,start_seed=1,project_every=25,update_tol=1e-11):

    c=c_count(h,n)
    c=np.maximum(c,0.0)
    target=0.5*n; dx=2.0/n; diag=n-1; M=2*n-1
    best_c=compute_c5(h,n); best_h=h.copy()
    cand_h=None; cand_c=best_c
    rng=np.random.RandomState(start_seed)
    no_improve_streak=0
    updates_since_proj=0
    total_upd=0
    for pass_i in range(max_passes):
       if time.time() > deadline: break
       if pass_i == 0: order = np.arange(n)
       elif pass_i == 1: order = np.arange(n-1,-1,-1)
       elif pass_i % 4 == 0: order = np.arange(n)
       else: order = rng.permutation(n)
       pass_improved=False
       for p in order:
          if time.time() > deadline: break

def solve_1d_full(c, a, diag, lo, hi, base_cmax):
    if base_cmax <= 1e-15:

    neg_idx = np.flatnonzero(neg)
    pos_a = aw[pos_idx]
    pos_c = c[pos_idx]
    neg_a = aw[neg_idx]
    neg_c = c[neg_idx]
    zero_mask = ~(pos | neg)
    zero_mask[diag] = False
    if zero_mask.any():
        zero_max = float(c[zero_mask].max())
    else:
        zero_max = -1e100

import numpy as np
import time


def c_count(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(c_count(h, n).max()) * (2.0 / float(n))


def project_sum_box(x, n):
    target = 0.5 * float(n)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if abs(float(x.sum()) - target) < 1e-15:
        return x
    lo = -float(x.max()) - 1.0
    hi = 1.0 - float(x.min()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)


def solve_1d_full(c, a, diag, lo, hi, base_cmax):
    if base_cmax <= 1e-15:
        return 0.0, float(base_cmax)

    aw = a.copy()
    aw[diag] = 0.0
    eps = 1e-14

    pos = aw > eps
    neg = aw < -eps
    pos_idx = np.flatnonzero(pos)
    neg_idx = np.flatnonzero(neg)
    pos_a = aw[pos_idx]
    pos_c = c[pos_idx]
    neg_a = aw[neg_idx]
    neg_c = c[neg_idx]

    zero_mask = ~(pos | neg)
    zero_mask[diag] = False
    zero_max = float(c[zero_mask].max()) if zero_mask.any() else -1e100

    ad = float(a[diag])
    cd = float(c[diag])
    tol = 1e-13

    Tlo = 0.0
    Thi = float(base_cmax)

    for _ in range(48):
        T = 0.5 * (Tlo + Thi)
        ok = T >= zero_max - tol
        if ok:
            L = lo
            U = hi
            if pos_idx.size:
                ub = float(np.min((T - pos_c) / pos_a))
                if ub < U:
                    U = ub
            if neg_idx.size:
                lb = float(np.max((T - neg_c) / neg_a))
                if lb > L:
                    L = lb
            if L <= U + tol:
                disc = ad * ad - 4.0 * (T - cd)
                if disc > 0.0:
                    sq = float(np.sqrt(disc))
                    r1 = 0.5 * (ad - sq)
                    r2 = 0.5 * (ad + sq)
                    if L > r1 + tol and U < r2 - tol:
                        ok = False
        if ok:
            Thi = T
        else:
            Tlo = T
        if Thi - Tlo < 1e-12 * max(1.0, Thi):
            break

    T = Thi
    L = lo
    U = hi
    if pos_idx.size:
        ub = float(np.min((T - pos_c) / pos_a))
        if ub < U:
            U = ub
    if neg_idx.size:
        lb = float(np.max((T - neg_c) / neg_a))
        if lb > L:
            L = lb
    if L > U:
        mid = 0.5 * (L + U)
        L = U = mid

    cands = [L, U, 0.5 * (L + U), 0.0, -0.5 * ad, lo, hi]
    disc = ad * ad - 4.0 * (T - cd)
    if disc > 0.0:
        sq = float(np.sqrt(disc))
        cands.extend([0.5 * (ad - sq), 0.5 * (ad + sq)])

    best_d = 0.0

        vals[diag] -= d * d
        v = float(vals.max())

            best_d = d

                    if upd_since >= 80 or abs(sum_cur - target) > 0.45:
                        h = project_sum_box(h, n)
                        c = np.maximum(c_count(h, n), 0.0)
                        sum_cur = float(h.sum())
                        upd_since = 0
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-14:
                            best_h = h.copy()
                            best_c = cv
                        if cv < cand_c - 1e-14:
                            cand_h = h.copy()
                            cand_c = cv

        if pass_improved:
            h = project_sum_box(h, n)
            c = np.maximum(c_count(h, n), 0.0)
            sum_cur = float(h.sum())
            upd_since = 0
            no_improve = 0
            cv = compute_c5(h, n)
            if cv < best_c - 1e-14:
                best_h = h.copy()
                best_c = cv
            if cv < cand_c - 1e-14:
                cand_h = h.copy()
                cand_c = cv
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    if cand_h is not None:
        hp = project_sum_box(cand_h, n)
        cv = compute_c5(hp, n)
        if cv < best_c - 1e-14:
            best_h = hp.copy()
            best_c = cv

    hp = project_sum_box(h, n)
    cv = compute_c5(hp, n)
    if cv < best_c - 1e-14:
        best_h = hp
        best_c = cv
    return best_h, best_c


def interpolate_h(h, n_old, n_new):
    if n_old == n_new:
        return project_sum_box(h, n_new)
    x_old = np.linspace(0.0, 2.0, int(n_old))
    x_new = np.linspace(0.0, 2.0, int(n_new))
    return project_sum_box(np.interp(x_new, x_old, h), n_new)


def smooth_noise(n, scale, rng):
    k = max(3, min(21, int(n) // 30 if int(n) >= 30 else 3))
    noise = rng.normal(0.0, scale, n)
    kernel = np.ones(k) / float(k)
    return np.convolve(noise, kernel, mode="same")


def fourier_start(n, rng, scale=0.2):
    x = np.linspace(0.0, 2.0, n)
    v = np.full(n, 0.5)
    modes = max(4, n // 20)
    for m in range(1, modes + 1):
        amp = float(rng.uniform(0.0, scale))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        v += amp * np.cos(np.pi * m * x + phase)
    return project_sum_box(v, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = int(h0.size)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_sum_box(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    dl = t0 + min(380.0, max(80.0, 0.38 * float(budget_s)))
    if dl > deadline:
        dl = deadline
    h1, c1 = exact_cd(h0, n0, dl, 220, seed + 1)
    if c1 < best_c:
        best_h, best_c = h1, c1

    if time.time() < deadline - 160.0:
        hc = project_sum_box(1.0 - best_h, len(best_h))
        h1, c1 = exact_cd(hc, len(best_h), deadline - 160.0, 70, seed + 2)
        if c1 < best_c:
            best_h, best_c = h1, c1

    cur_n = len(best_h)
    for scale in (0.015, 0.04, 0.09, 0.18):
        if time.time() > deadline - 130.0:
            break
        alloc = min(80.0, max(25.0, 0.06 * float(budget_s)))
        dlp = min(deadline - 110.0, time.time() + alloc)
        hp = project_sum_box(best_h + scale * smooth_noise(cur_n, 1.0, rng), cur_n)
        ht, ct = exact_cd(hp, cur_n, dlp, 45, seed + 17)
        if ct < best_c:
            best_h, best_c = ht, ct

    for n_c in (120, 160, 200, 260, 320, 420):
        if n_c >= len(best_h):
            continue
        if time.time() > deadline - 100.0:
            break
        passes = 140 if n_c <= 150 else 90 if n_c <= 250 else 55
        base = interpolate_h(best_h, len(best_h), n_c)
        h1, c1 = exact_cd(base, n_c, deadline - 80.0, passes, seed + 101)
        if c1 < best_c:
            best_h, best_c = h1, c1

        if time.time() < deadline - 70.0:
            h2 = fourier_start(n_c, rng, 0.18)
            h2, c2 = exact_cd(h2, n_c, deadline - 70.0, max(25, passes // 2), seed + 201)
            if c2 < best_c:
                best_h, best_c = h2, c2

        if time.time() < deadline - 60.0:
            h3 = project_sum_box((rng.rand(n_c) < 0.5).astype(float), n_c)
            h3, c3 = exact_cd(h3, n_c, deadline - 60.0, max(25, passes // 2), seed + 301)
            if c3 < best_c:
                best_h, best_c = h3, c3

    if len(best_h) < min(800, max(400, n0)) and time.time() < deadline - 90.0:
        n_up = min(800, max(len(best_h) * 2, min(n0, 600)))
        hu = interpolate_h(best_h, len(best_h), n_up)
        hu, cu = exact_cd(hu, n_up, deadline - 70.0, 60, seed + 401)
        if cu < best_c:
            best_h, best_c = hu, cu

    best_h = project_sum_box(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
