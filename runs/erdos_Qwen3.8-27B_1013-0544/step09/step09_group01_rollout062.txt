    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    # numerically stable log-sum-exp
    shifted = alpha * (c - c_max)
    log_sum_exp = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    # gradient
    w = np.exp(shifted) / np.sum(np.exp(shifted))  # shape (2n-1,)
    grad = np.zeros(n)
    for p in range(n):

        # forward: k such that 0 <= p+k-(n-1) < n => k in [n-1-p, n-1+p]

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def fast_cd(h, n, max_passes, deadline, gs_iters=40):

    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0

            break

    if time.time() < deadline - 3:
        h_f, c_f = lbfgs(best_h, len(best_h), [500000, 2000000], [0.0, 0.0], 300, deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
