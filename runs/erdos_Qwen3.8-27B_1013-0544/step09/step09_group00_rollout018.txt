            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def smooth_obj(x, n, beta):

    mx = c.max()
    shifted = beta * c
    s = shifted - shifted.max()
    lse = shifted.max() + np.log(np.sum(np.exp(s)))
    return (lse / beta) * (2.0 / n)

def exact_cd(h, n, max_passes, deadline):

                c_diag = None

                if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
                    feasible = False

def block_cd(h, n, block_size, n_blocks, deadline, rng):
    """Block coordinate descent: optimize blocks of consecutive coordinates jointly."""
    h = h.copy()
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        # Choose random blocks
        starts = rng.randint(0, n - block_size + 1, size=20)
        for start in starts:
            if time.time() > deadline:
                break
            idx = np.arange(start, start + block_size)
            fixed_sum = h.sum() - h[idx].sum()
            target_block = n / 2.0 - fixed_sum
            if target_block < 0 or target_block > block_size:
                continue
            
            def block_obj(x):
                h_in = h.copy()
                h_in[idx] = x
                return compute_c5(h_in, n)
            
            x0 = h[idx].copy()
            bounds = [(0.0, 1.0)] * block_size
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_block,
                     'jac': lambda x: np.ones(block_size)}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
                if res.fun < compute_c5(h, n) - 1e-12:
                    h[idx] = np.clip(res.x, 0, 1)
                    h = project_feasible(h, n)
            except:
                pass
    return h, compute_c5(h, n)

def anneal_block(h, n, T, n_steps, deadline, rng):
    """Simulated annealing on random blocks."""
    h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T_cur = T * (0.95 ** step)
        if T_cur < 1e-8:
            break
        # Pick a random block
        bs = rng.randint(2, 6)
        start = rng.randint(0, max(1, n - bs))
        idx = np.arange(start, min(start + bs, n))
        # Propose perturbation
        h_new = h.copy()
        h_new[idx] = np.clip(h[idx] + rng.normal(0, 0.05, len(idx)), 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T_cur):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
    return h, best_c

def slsqp_opt(h, n, betas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for beta in betas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        
        def _cons_fun(x):
            return x.sum() - n / 2.0
        def _cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c
