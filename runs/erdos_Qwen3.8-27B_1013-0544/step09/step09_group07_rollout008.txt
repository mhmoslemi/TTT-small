def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):

            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i: dd -= 1.0
            if diag_local == j: dd += 1.0
            if abs(dd) < 1e-12:

                    d = dc / da
                    if lo < d < hi: candidates.append(d)
            else:

                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da+sq)/(2*dd), (-da-sq)/(2*dd)]:
                        if lo < d < hi: candidates.append(d)
    best_d, best_val = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)

def slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse/alpha) * (2.0/n)

def slsqp_polish(h, n, alphas, maxiter=300, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline: break
        try:
            res = minimize(slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':lambda x,*a: x.sum()-a[0]/2,'args':(n,)}],
                          options={'maxiter':maxiter,'ftol':1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c: best_h, best_c = new_h, new_c
        except: pass
    return best_h, best_c

For the full correlation c[k] = Σ_i h[i] * (1-h[i + (k-(n-1))])

Let lag = k - (n-1), so k = n-1+lag, and lag ranges from -(n-1) to (n-1).

c[n-1+lag] = Σ_i h[i] * (1-h[i+lag]) where the sum is over valid i (0 ≤ i < n and 0 ≤ i+lag < n).

∂c[n-1+lag]/∂h[j]:
- h[j] appears as the first factor when i=j: contributes (1-h[j+lag]) if j+lag is valid
- h[j] appears as the second factor's argument when i+lag=j, i.e., i=j-lag: contributes -h[j-lag] if j-lag is valid

So:
∂c[n-1+lag]/∂h[j] = (1-h[j+lag])*[0≤j+lag<n] - h[j-lag]*[0≤j-lag<n]

The weighted gradient:
grad[j] = Σ_lag weights[n-1+lag] * [(1-h[j+lag])*[0≤j+lag<n] - h[j-lag]*[0≤j-lag<n]]

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    k = len(a_coeff)
    if k == 0:
        return 0.0
    candidates = [float(np.clip(0.0, lo, hi))]
    for i in range(k):
        for j in range(i + 1, k):
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0
            if abs(dd) < 1e-12:
                if abs(da) > 1e-12:
                    d = dc / da
                    if lo < d < hi:
                        candidates.append(d)
            else:
                disc = da * da + 4.0 * dd * dc
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    for d in [(-da + sq) / (2.0 * dd), (-da - sq) / (2.0 * dd)]:
                        if lo < d < hi:
                            candidates.append(d)
    best_d, best_val = 0.0, obj_1d(0.0, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        dc = float(np.clip(d, lo, hi))
        v = obj_1d(dc, c_aff, a_coeff, diag_local, unaff_max)
        if v < best_val - 1e-15:
            best_val, best_d = v, dc
    span = hi - lo
    if span > 1e-10:
        margin = min(span * 0.15, abs(best_d) * 0.5 + span * 0.05)
        gl, gh = max(lo, best_d - margin), min(hi, best_d + margin)
        if gh - gl > 1e-12:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=60)
                if res.fun < best_val - 1e-15:
                    best_d, best_val = float(res.x), float(res.fun)
            except Exception:
                pass
    return best_d

def coord_cd(h, n, max_passes=300, deadline=None, top_k=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for idx in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_f_lo = max(0, n - 1 - idx)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - idx)
            m_b_lo = max(0, idx)
            m_b_hi = min(2 * n - 2, idx + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms_all = np.arange(m_lo, m_hi + 1)
            if len(ms_all) > top_k:
                vals = c[ms_all]
                top_idx = np.argsort(vals)[-top_k:]
                ms = ms_all[np.sort(top_idx)]
                ms_set = set(ms.tolist())
                skipped = [m for m in ms_all if m not in ms_set]
                unaff = float(c[skipped].max()) if skipped else 0.0
            else:
                ms, unaff = ms_all, 0.0
            a_coeff = np.zeros(len(ms))
            jf = idx + ms - (n - 1)
            jb = idx - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms].copy()
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            diag_local = None
            for ii in range(len(ms)):
                if ms[ii] == n - 1:
                    diag_local = ii
                    break
            lo, hi_ = -h0, 1.0 - h0
            best_d = solve_1d_exact(c_aff, a_coeff, diag_local, unaff, lo, hi_)
            if abs(best_d) > 1e-12:
                new_val = obj_1d(best_d, c_aff, a_coeff, diag_local, unaff)
                if new_val < cur_max - 1e-14:
                    h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                    improved = True
                    cur_max = min(cur_max, new_val)
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def tilt_minmax(h, n, max_iter=400, tau_init=4.0, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    tau = tau_init
    step = 0.04
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_s = c * (2.0 / n)
        c_max = float(c_s.max())
        if c_max < best_c - 1e-14:
            best_c, best_h = c_max, h.copy()
        weights = np.exp(tau * (c_s - c_max))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights /= w_sum
        grad = np.zeros(n)
        omh = 1.0 - h
        sig = weights > 1e-14
        lags = np.where(sig)[0] - (n - 1)
        for lag in lags:
            w = weights[n - 1 + lag]
            if w < 1e-14:
                continue
            if lag >= 0:
                grad[:n - lag] += w * omh[lag:]
                grad[lag:] -= w * h[:n - lag]
            else:
                grad[-lag:] += w * omh[:n + lag]
                grad[:n + lag] -= w * h[-lag:]
        grad -= grad.mean()
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max:
            h = h_new
            step = min(step * 1.03, 0.15)
        else:
            step *= 0.65
            if step < 1e-9:
                break
        tau = max(tau * 1.002, 1.0)
        if tau > 15.0:
            tau = 15.0
    return best_h, best_c

def fft_flatten(h, n, alpha=0.4, max_iter=80, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        H = np.fft.rfft(h)
        power = np.abs(H) ** 2
        mean_p = power[1:].mean()
        new_power = power.copy()
        new_power[1:] = (1 - alpha) * power[1:] + alpha * mean_p
        phase = np.angle(H)
        new_H = np.sqrt(new_power) * np.exp(1j * phase)
        new_h = np.fft.irfft(new_H, n=n)
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c - 1e-14:
            best_c, best_h, h = c_new, new_h.copy(), new_h
        else:
            for mix_val in [0.5, 0.3, 0.15]:
                mixed = project_feasible((1 - mix_val) * h + mix_val * new_h, n)
                c_mix = compute_c5(mixed, n)
                if c_mix < best_c - 1e-14:
                    best_c, best_h, h = c_mix, mixed.copy(), mixed
                    break
            else:
                break
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_polish(h, n, alphas, maxiter=300, deadline=None):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - len(x) / 2.0}],
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
