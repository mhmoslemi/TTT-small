    # First term: (1-h[p-m+n-1]) for m where 0 <= p-m+n-1 <= n-1

            x_lo = max(x_lo, float(np.max((L - c_base[negm]) / a[negm])))
        if np.any(posm):
            x_hi = min(x_hi, float(np.min((L - c_base[posm]) / a[posm])))
        if x_lo > x_hi + 1e-14:
            L_lo = L
            continue
        if has_diag:
            Q = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Q
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if max(x_lo, 0.0) > r1 and min(x_hi, 1.0) < r2:
                    L_lo = L
                    continue
        L_hi = L
    
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    negm = a < -1e-14
    posm = a > 1e-14
    if np.any(negm):
        x_lo = max(x_lo, float(np.max((L - c_base[negm]) / a[negm])))
    if np.any(posm):
        x_hi = min(x_hi, float(np.min((L - c_base[posm]) / a[posm])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Q = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Q
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_hi < r2:
                return h0
            if x_lo <= r1 and x_hi >= r2:
                d1 = (h0 - r1) if x_lo <= r1 else 1e9
                d2 = (r2 - h0) if x_hi >= r2 else 1e9
                if d1 <= d2:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo <= r1:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-13:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
