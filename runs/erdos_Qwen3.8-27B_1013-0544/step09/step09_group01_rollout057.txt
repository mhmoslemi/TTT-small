def compute_smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    # Compute softmax weights
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    w = exp_ac / exp_ac.sum()
    
    # Compute PartA and PartB
    # PartA[j] = np.correlate(w, h, 'full')[2n-2-j] for j=0..n-1
    corr_wh = np.correlate(w, h, mode='full')
    PartA = corr_wh[n-1:2*n-1][::-1]
    
    # PartB[j] = np.convolve(w, h, 'full')[j+n-1] for j=0..n-1
    conv_wh = np.convolve(w, h, mode='full')
    PartB = conv_wh[n-1:2*n-1]
    
    # W_valid[j]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    m_lo = np.maximum(0, j_arr - n + 1)
    m_hi = np.minimum(n, j_arr + n)
    W_valid = w_cum[m_hi - j_arr + n - 1] - w_cum[m_lo - j_arr + n - 1]
    
    grad = (W_valid - PartA - PartB) / alpha
    return grad, c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, n, alpha, beta=0.0):
    c = np.correlate(x, 1.0 - x, mode='full')
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    w = exp_ac / exp_ac.sum()
    
    obj = (ac_max + np.log(exp_ac.sum())) / alpha * (2.0 / n)
    
    if beta > 0:
        s = x.sum() - n / 2.0
        obj += beta * s * s
    
    # Gradient
    corr_wh = np.correlate(w, x, mode='full')
    PartA = corr_wh[n-1:2*n-1][::-1]
    conv_wh = np.convolve(w, x, mode='full')
    PartB = conv_wh[n-1:2*n-1]
    
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    m_lo = np.maximum(0, j_arr - n + 1)
    m_hi = np.minimum(n, j_arr + n)
    W_valid = w_cum[m_hi - j_arr + n - 1] - w_cum[m_lo - j_arr + n - 1]
    
    grad = (W_valid - PartA - PartB) / alpha * (2.0 / n)
    
    if beta > 0:
        grad += 2.0 * beta * (x.sum() - n / 2.0)
    
    return obj, grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(
                smooth_obj_and_grad, best_h,
                args=(n, alpha, beta),
                method='L-BFGS-B',
                bounds=bounds,
                jac=True,
                options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-14}
            )
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False

            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):

            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0

            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):

                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)

                improved = True
        if not improved:

        h = project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _smooth_obj_and_grad(x, n, alpha, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    w = exp_ac / exp_ac.sum()
    dx = 2.0 / n
    obj = (ac_max + np.log(exp_ac.sum())) / alpha * dx
    if beta > 0:
        s = x.sum() - n / 2.0
        obj += beta * s * s
    corr_wh = np.correlate(w, x, mode='full')
    PartA = corr_wh[n-1:2*n-1][::-1]
    conv_wh = np.convolve(w, x, mode='full')
    PartB = conv_wh[n-1:2*n-1]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    m_lo = np.maximum(0, j_arr - n + 1)
    m_hi = np.minimum(n, j_arr + n)
    W_valid = w_cum[m_hi - j_arr + n - 1] - w_cum[m_lo - j_arr + n - 1]
    grad = (W_valid - PartA - PartB) / alpha * dx
    if beta > 0:
        grad += 2.0 * beta * (x.sum() - n / 2.0)
    return obj, grad

def lbfgsb_optimize(h, n, alphas, maxiter_each=300, beta=100.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_and_grad, best_h, args=(n, alpha, beta),
                           method='L-BFGS-B', bounds=bounds, jac=True,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, rng=None):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        count = 0
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
                count += 1
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
