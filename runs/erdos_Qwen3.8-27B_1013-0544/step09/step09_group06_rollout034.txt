```python
def solve_1d_minimax(c_peak, a_coeff, has_diag, diag_idx, unaff_max, lo, hi):
    """Solve min_d max(max_i(c_i + d*a_i - d^2*diag_i), unaff_max) for d in [lo,hi]"""
    T = len(c_peak)
    candidates = [lo, hi, 0.0]
    
    # Intersections of linear pieces (ignoring quadratic for now)
    for i in range(T):

                continue

            if abs(da) > 1e-15:

                if lo <= d <= hi:

                d2 = (-B - sq) / (2*A)
                for d in [d1, d2]:
                    if lo <= d <= hi:

            C = unaff_max - c_peak[i]

def cd_pass(h, n, top_k, deadline, max_passes=100):
    """Coordinate descent with exact 1D minimax."""
    h = h.copy()
    total_lags = 2 * n - 1
    dx = 2.0 / n
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        idx = np.argsort(c)[::-1][:top_k]
        
        improved = False
        # Process coordinates in order of gradient magnitude
        # Compute a rough gradient for ordering
        grad_norms = np.zeros(n)
        for ii in range(top_k):
            m = idx[ii]
            shift = m - (n - 1)
            if shift >= 0:
                valid = slice(0, n - shift)
                grad_norms[valid] += np.abs(1 - h[shift:n] - h[:n-shift])
            else:
                s2 = -shift
                valid = slice(s2, n)
                grad_norms[valid] += np.abs(1 - h[:n-s2] - h[s2:])
        
        order = np.argsort(-grad_norms)
        
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            a_coeff = np.zeros(top_k)
            has_diag = False
            diag_idx = -1
            for ii in range(top_k):
                m = int(idx[ii])
                shift = m - (n - 1)
                if shift == 0:
                    a_coeff[ii] = 1.0 - 2.0 * h0
                    has_diag = True
                    diag_idx = ii
                else:
                    val = 1.0
                    if 0 <= p - shift < n:
                        val -= h[p - shift]
                    if 0 <= p + shift < n:
                        val -= h[p + shift]
                    a_coeff[ii] = val
            
            mask = np.ones(total_lags, dtype=bool)
            mask[idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c[idx], a_coeff, has_diag, diag_idx, unaff_max, lo, hi)
            
            if best_val < current_max - 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        
        if not improved:
            break
        h = project_feasible(h, n)
    
    return h, compute_c5(h, n)

def gradient_descent(h, n, alpha, lr, max_iter, deadline):
    """Simple gradient descent with top-k subgradient."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Top-k weights
        k = min(20, 2*n-1)
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
        s = alpha * c_top
        s_max = s.max()
        w = np.exp(s - s_max)
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for ii in range(k):
            m = int(idx[ii])
            shift = m - (n - 1)
            wm = w[ii]
            if shift >= 0:
                grad[:n - shift] += wm * (1.0 - h[shift:n] - h[:n - shift])
            else:
                s2 = -shift
                grad[s2:] += wm * (1.0 - h[:n - s2] - h[s2:])
        
        # Update
        h = project_feasible(h - lr * grad, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        # Adaptive learning rate
        lr *= 0.999
    
    return best_h, best_c

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig
```

```python
def compute_gradient_fast(h, w, n):
    # w has length 2n-1
    cumsum_w = np.cumsum(w)[:n]  # length n
    conv_h_w = np.convolve(h, w, 'full')[:n]  # length n
    corr_h_w = np.correlate(h, w, 'full')[:n]  # length n
    return cumsum_w - conv_h_w - corr_h_w
```

This is O(n log n) using FFT-based convolution! Much faster than the O(nnz(w) * n) loop.

But wait - there's a subtlety. The `w` here is the softmax weight over ALL lags (length 2n-1), not just the top-k. If I use the full softmax, all entries of w are non-zero (though very small for non-top lags). The convolution/correlation will handle this fine.

Actually, for numerical stability, I should use the top-k approach: set w[m] = 0 for non-top lags. Then the convolution/correlation naturally handles the bounds.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                return np.clip(h, 0.0, 1.0)
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                return np.clip(h, 0.0, 1.0)
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_fast(h, w, n):
    """Compute gradient using FFT-based convolution."""
    cumsum_w = np.cumsum(w)[:n]
    conv_h_w = np.convolve(h, w, 'full')[:n]
    corr_h_w = np.correlate(h, w, 'full')[:n]
    return cumsum_w - conv_h_w - corr_h_w

def gradient_descent(h, n, alpha, lr0, max_iter, deadline, top_k=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    total_lags = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:top_k]
        
        w = np.zeros(total_lags)
        c_top = c[idx]
        s = alpha * c_top
        s_max = s.max()
        w_top = np.exp(s - s_max)
        w_top /= w_top.sum()
        w[idx] = w_top
        
        grad = compute_gradient_fast(h, w, n)
        
        # Adaptive step size
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = project_feasible(h - lr * grad, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c:
            best_c = c5_new
            best_h = h_new.copy()
        
        # Accept if not too bad (allows some non-monotone progress)
        if c5_new <= compute_c5(h, n) + 0.001:
            h = h_new
        else:
            h = project_feasible(h - lr * 0.5 * grad, n)
    
    return best_h, best_c
