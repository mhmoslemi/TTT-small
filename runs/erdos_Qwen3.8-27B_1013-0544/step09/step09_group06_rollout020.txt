def subgradient_step(h, n, step_size, top_k):
    # Compute correlations
    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top contributing shifts
    top_idx = np.argsort(c)[::-1][:top_k]
    # Compute subgradient for these shifts
    g = np.zeros(n)
    for idx in top_idx:
        # shift = idx - (n-1)
        shift = idx - (n - 1)
        for j in range(n):
            jf = j + shift  # forward
            jb = j - shift  # backward
            term_f = (1.0 - h[jf]) if 0 <= jf < n else 0.0
            term_b = -h[jb] if 0 <= jb < n else 0.0
            g[j] += term_f + term_b
    # Normalize
    g_norm = np.linalg.norm(g)
    if g_norm > 1e-15:
        h_new = h - step_size * g / g_norm
    else:
        h_new = h.copy()
    return np.clip(h_new, 0, 1)

            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max):
    vals = c_peak + d * a_coeff
    if diag_idx is not None:
        vals[diag_idx] -= d * d
    return max(float(vals.max()), unaff_max)

def solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C
            if disc >= 0:
                sq = np.sqrt(disc)
                d1 = (-B + sq) / 2.0
                d2 = (-B - sq) / 2.0
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    for i in range(T):
        if abs(a_coeff[i]) > 1e-15:
            d = (unaff_max - c_peak[i]) / a_coeff[i]
            if lo <= d <= hi:
                candidates.append(d)
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        v = eval_1d(d, c_peak, a_coeff, diag_idx, unaff_max)
        if v < best_val:
            best_val = v
            best_d = d
    return best_d, best_val

def multi_cd(h, n, max_passes, deadline, top_k):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        c_peak = c[top_idx].astype(float)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-8 or h0 > 1.0 - 1e-8:
                continue
            a_coeff = np.zeros(k)
            diag_idx = None
            for idx in range(k):
                m = int(top_idx[idx])
                jf = p + m - (n - 1)
                jb = p - m + (n - 1)
                af = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                ab = (-h[jb]) if 0 <= jb < n else 0.0
                a_coeff[idx] = af + ab
                if m == n - 1:
                    diag_idx = idx
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[top_idx] = False
            unaff_max = float(c[mask].max()) if mask.any() else 0.0
            lo = -h0
            hi = 1.0 - h0
            best_d, best_val = solve_1d_minimax(c_peak, a_coeff, diag_idx, unaff_max, lo, hi)
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_refine(h, n, deadline, max_iter=800):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h_old = h.copy()
    step = 0.005
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            shift = int(idx) - (n - 1)
            for j in range(n):
                jf = j + shift
                jb = j - shift
                tf = (1.0 - h[jf]) if 0 <= jf < n else 0.0
                tb = -h[jb] if 0 <= jb < n else 0.0
                g[j] += tf + tb
        g /= top_k
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            h_new = h - step * g / gn
        else:
            break
        h_new = project_feasible(h_new, n)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))
        if c_new < c_max:
            h = h_new
            if c_new * (2.0/n) < best_c:
                best_c = c_new * (2.0 / n)
                best_h = h.copy()
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
        if it % 40 == 39:
            h, c_cd = multi_cd(h, n, 15, deadline, 12)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h.copy()
    return best_h, best_c

def spectral_rebalance(h, n, n_modes=12, strength=0.08):
    H = np.fft.fft(h)
    mag = np.abs(H)
    non_dc = np.arange(1, n // 2 + 1)
    order = non_dc[np.argsort(mag[non_dc])[::-1]]
    for i in range(min(n_modes, len(order))):
        m = order[i]
        H[m] *= (1.0 - strength)
        H[n - m] = np.conj(H[m])
    h_new = np.real(np.fft.ifft(H))
    return project_feasible(h_new, n)

def try_symmetry(h, n):
    h_flip = project_feasible(1.0 - h, n)
    c_flip = compute_c5(h_flip, n)
    c_orig = compute_c5(h, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(600, 0.5)
        n = 600
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: Progressive multi-peak CD
    for tk in [8, 15, 25, 35, 45, 55]:
        if time.time() > deadline - 350:
            break
        h2, c2 = multi_cd(best_h, n, 400, deadline - 350, tk)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 2: Subgradient with CD interleaving
    if time.time() < deadline - 250:
        h2, c2 = subgrad_refine(best_h, n, deadline - 250, 600)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Spectral rebalancing + CD
    if time.time() < deadline - 200:
        for str_val in [0.04, 0.08, 0.12, 0.18]:
            if time.time() > deadline - 200:
                break
            h2 = spectral_rebalance(best_h, n, n_modes=10, strength=str_val)
            h2, c2 = multi_cd(h2, n, 300, deadline - 200, 25)
            h2, c2 = try_symmetry(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
    # Phase 4: Deep CD with very large top_k
    if time.time() < deadline - 150:
        h2, c2 = multi_cd(best_h, n, 800, deadline - 150, 60)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 5: Stochastic perturbations + CD
    if time.time() < deadline - 100:
        for r in range(10):
            if time.time() > deadline - 100:
                break
            s = 0.002 + 0.003 * r
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            hp, c2 = multi_cd(hp, n, 200, deadline - 100, 20)
            if c2 < best_c:
                best_h, best_c = hp, c2
    # Phase 6: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [650, 700, 750, 800, 850]:
        if time.time() > deadline - 70:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = multi_cd(h_int, n_try, 250, deadline - 70, 20)
        h_o, c_o = try_symmetry(h_o, n_try)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new
    # Phase 7: Final deep refinement
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        h_f, c_f = multi_cd(best_h, n_cur, 2000, deadline - 3, 50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h, best_c = try_symmetry(best_h, len(best_h))
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
