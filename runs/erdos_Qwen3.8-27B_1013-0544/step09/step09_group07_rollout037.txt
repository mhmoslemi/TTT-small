import numpy as np
import math
import time

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h):
    n = len(h)
    c = corr_raw(h)
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(np.sum(room))
            if s > 1e-14:
                h = h + (diff / s) * room
        else:
            s = float(np.sum(h))
            if s > 1e-14:
                h = h + (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def make_plan(n):
    idx = np.arange(n)
    d = np.maximum(idx, n - 1 - idx)
    return d, n - 1 - d, 2 * d + 1

def value_1d(d, c_aff, a_aff, diag, U):
    if d == 0.0:
        m = float(np.max(c_aff))
    else:
        v = c_aff + d * a_aff
        v[diag] -= d * d
        m = float(np.max(v))
    return U if m < U else m

def interval_pre(V, U, lo, hi, c_zero_max, c_pos, a_pos, c_neg, a_neg, c0, a0):
    tol = 1e-12
    if U > V + tol:
        return None
    if c_zero_max is not None and c_zero_max > V + tol:
        return None
    L = lo
    R = hi
    if c_pos.size:
        b = float(np.min((V - c_pos) / a_pos))
        if b < R:
            R = b
    if c_neg.size:
        b = float(np.max((V - c_neg) / a_neg))
        if b > L:
            L = b
    if L > R + tol:
        return None
    disc = a0 * a0 - 4.0 * (V - c0)
    r1 = None
    r2 = None
    if disc >= 0.0:
        sq = math.sqrt(disc)
        r1 = 0.5 * (a0 - sq)
        r2 = 0.5 * (a0 + sq)
        if L > r1 + tol and R < r2 - tol:
            return None
    return (L, R, r1, r2)

def min_max_1d(c_aff, a_aff, diag, U, lo, hi):
    K = c_aff.size
    if K == 1:
        c_line = np.empty(0, dtype=float)
        a_line = np.empty(0, dtype=float)
    else:
        mask = np.ones(K, dtype=bool)
        mask[diag] = False
        c_line = c_aff[mask]
        a_line = a_aff[mask]

    absa = np.abs(a_line)
    zero = absa <= 1e-14
    c_zero_max = float(np.max(c_line[zero])) if zero.any() else None

    pos = a_line > 1e-14
    c_pos = c_line[pos] if pos.any() else np.empty(0, dtype=float)
    a_pos = a_line[pos] if pos.any() else np.empty(0, dtype=float)

    neg = a_line < -1e-14
    c_neg = c_line[neg] if neg.any() else np.empty(0, dtype=float)
    a_neg = a_line[neg] if neg.any() else np.empty(0, dtype=float)

    c0 = float(c_aff[diag])
    a0 = float(a_aff[diag])

    v_hi = min(
        value_1d(0.0, c_aff, a_aff, diag, U),
        value_1d(lo, c_aff, a_aff, diag, U),
        value_1d(hi, c_aff, a_aff, diag, U)
    )
    v_lo = U if U > 0.0 else 0.0

    if v_hi > v_lo + 1e-12:
        for _ in range(34):
            mid = 0.5 * (v_lo + v_hi)
            if interval_pre(mid, U, lo, hi, c_zero_max, c_pos, a_pos, c_neg, a_neg, c0, a0) is not None:
                v_hi = mid
            else:
                v_lo = mid

    V = v_hi + 1e-10
    res = interval_pre(V, U, lo, hi, c_zero_max, c_pos, a_pos, c_neg, a_neg, c0, a0)

    if res is None:
        cands = [0.0, lo, hi]
    else:
        L, R, r1, r2 = res
        cands = [lo, hi, 0.0, L, R, 0.5 * (L + R)]
        if r1 is not None:
            cands.append(r1)
            cands.append(r2)
            lh = min(R, r1)
            if L <= lh:
                cands.append(0.5 * (L + lh))
            rl = max(L, r2)
            if rl <= R:
                cands.append(0.5 * (rl + R))

    best_d = 0.0
    best_score = 1e100
    for dc in cands:
        if not (lo - 1e-10 <= dc <= hi + 1e-10):
            continue
        dc = float(np.clip(dc, lo, hi))
        val = value_1d(dc, c_aff, a_aff, diag, U)
        score = val + 1e-10 * abs(dc)
        if score < best_score:
            best_score = score
            best_d = dc
    return best_d, value_1d(best_d, c_aff, a_aff, diag, U)

def smooth_score(c, alpha=30.0):
    m = float(np.max(c))
    if not math.isfinite(m):
        return 1e100
    z = np.exp(alpha * (c - m))
    return m + float(np.log(np.sum(z))) / alpha

def coordinate_descent(h, n, deadline, max_passes=5000, seed=0):
    h = project_feasible(h.copy(), n)
    c = corr_raw(h)
    current_max = float(np.max(c))
    score = smooth_score(c)

    best_h = h.copy()
    best_max = current_max
    darr, mstart, mlen = make_plan(n)
    nshift = 2 * n - 1
    no_improve = 0

    for p in range(max_passes):
        if time.time() >= deadline:
            break
        if p < 6:
            margin = 0.0
        else:
            margin = current_max * min(0.25, 0.004 * (p - 6) + 1e-9)

        if p % 4 == 0:
            order = np.random.RandomState(seed + 1000 * p).permutation(n)
        else:
            order = range(n)

        improved = False
        for i in order:
            if time.time() >= deadline:
                break
            x = float(h[i])
            lo = -x
            hi = 1.0 - x
            mst = int(mstart[i])
            K = int(mlen[i])
            end = mst + K

            ca = float(np.max(c[mst:end]))
            if ca < current_max - margin - 1e-12:
                continue

            U = 0.0
            if mst > 0:
                U = max(U, float(np.max(c[:mst])))
            if end < nshift:
                U = max(U, float(np.max(c[end:])))

            Di = int(darr[i])
            s = np.arange(-Di, Di + 1, dtype=np.int64)
            a = np.zeros(K, dtype=float)

            ip = i + s
            vm = (ip >= 0) & (ip < n)
            if vm.any():
                a[vm] += 1.0 - h[ip[vm]]

            im = i - s
            vm2 = (im >= 0) & (im < n)
            if vm2.any():
                a[vm2] -= h[im[vm2]]

            d, _ = min_max_1d(c[mst:end], a, Di, U, lo, hi)
            if abs(d) < 1e-14:
                continue

            for dtry in (d, 0.5 * d):
                if abs(dtry) < 1e-14:
                    continue
                hc = h.copy()
                hc[i] = float(x + dtry)
                hc = project_feasible(hc, n)
                cn = corr_raw(hc)
                nm = float(np.max(cn))

                if nm < current_max - 1e-12:
                    h = hc
                    c = cn
                    current_max = nm
                    score = smooth_score(cn)
                    improved = True
                    break

                if nm < current_max + 1e-10:
                    ns = smooth_score(cn)
                    if ns < score - 1e-6:
                        h = hc
                        c = cn
                        current_max = nm
                        score = ns
                        improved = True
                        break

        if current_max < best_max - 1e-12:
            best_max = current_max
            best_h = h.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 12 and margin > 0.2 * max(1e-12, current_max):
                break

        if current_max <= 0.0:
            break

    h_out = project_feasible(best_h, n)
    c_out = corr_raw(h_out)
    return h_out, float(np.max(c_out))

def make_alternating(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    return project_feasible(h, n)

def make_sinusoidal(n, freq=1.0, phase=0.0):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.5 * np.sin(math.pi * freq * x + phase)
    return project_feasible(h, n)

def make_random_smooth(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for _ in range(12):
        h += rng.uniform(0.02, 0.15) * np.sin(
            math.pi * float(rng.randint(1, 4)) * x + rng.uniform(0.0, 2.0 * math.pi)
        )
    return project_feasible(h, n)

def interpolate_to(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    return project_feasible(np.interp(x_new, x_old, h_src), n_dst)
