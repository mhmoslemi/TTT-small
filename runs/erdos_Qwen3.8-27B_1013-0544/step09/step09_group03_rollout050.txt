    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m = n - 1 + s
    b = c[m] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes=1000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
        elif improved == 0:
            break
    return h, best_c

def lse_objective(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lse_gradient(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for d in range(2 * n - 1):
        k = d - (n - 1)
        wd = w[d] * dx
        if k >= 0:
            i = np.arange(n - k)
            grad[i] += (1.0 - x[i + k]) * wd
            grad[i + k] += -x[i] * wd
        else:
            i = np.arange(-k, n)
            grad[i] += (1.0 - x[i + k]) * wd
            grad[i + k] += -x[i] * wd
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_smooth(h, n, alpha, mu, maxiter=200, deadline=None):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h, args=(n, alpha, mu),
                      method='L-BFGS-B', jac=lse_gradient,
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
        new_h = project_feasible(res.x.copy(), n)
        return new_h
    except Exception:
        return h

def subgradient_method(h, n, n_iter=5000, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Compute all overlaps
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        
        # Compute subgradient of g_{k*}
        grad = np.zeros(n)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] += (1.0 - h[i + lag]) * dx
            grad[i + lag] += -h[i] * dx
        else:
            i = np.arange(-lag, n)
            grad[i] += (1.0 - h[i + lag]) * dx
            grad[i + lag] += -h[i] * dx
        
        # Multiply by some factor to balance with the sum constraint
        # Use a "combined" subgradient
        step = 1.0 / np.sqrt(it + 1) * 0.5
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        new_c = compute_c5(h_new, n)
        if new_c < best_c:
            best_c = new_c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c
