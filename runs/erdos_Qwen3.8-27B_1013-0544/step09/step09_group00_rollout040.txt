```python
import numpy as np
import time
from scipy.optimize import minimize

_SPEC_K = 0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = int(len(h))
    if n <= 0:
        return 1.0
    return float(np.max(compute_corr(h)) * (2.0 / n))


def project_sum_box(z, target, n):
    z = np.asarray(z, dtype=float)
    n = int(n)
    target = float(target)
    if n <= 0:
        return z.copy()
    lo = float(np.min(z)) - float(n) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - hi, 0.0, 1.0)
    for _ in range(6):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            if add > 0.0:
                h[idx] += add
        else:
            idx = int(np.argmin(h))
            add = max(diff, -h[idx])
            if add < 0.0:
                h[idx] += add
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_maxc(c, alpha):
    m = float(np.max(c))
    if m <= 0.0:
        return 0.0
    z = alpha * (c - m)
    zm = float(np.max(z))
    return float(m + (zm + np.log(np.sum(np.exp(z - zm)))) / alpha)


def grad_weights(h, w, topK=150):
    n = int(len(h))
    g = np.zeros(n, dtype=float)
    center = n - 1
    tol = 1e-14
    idx = np.nonzero(w > tol)[0]
    if len(idx) == 0:
        return g
    if topK and len(idx) > topK:
        keep = np.argpartition(w, -topK)[-topK:]
    else:
        keep = idx
    for idd in keep:
        wv = float(w[idd])
        if wv <= tol:
            continue
        d = int(idd) - center
        ad = d if d >= 0 else -d
        if ad >= n:
            continue
        li = n - ad
        if d >= 0:
            g[0:li] += wv * (1.0 - h[d:d + li])
            g[d:d + li] -= wv * h[0:li]
        else:
            e = -d
            g[e:e + li] += wv * (1.0 - h[0:li])
            g[0:li] -= wv * h[e:e + li]
    return g


def lbfgs_fun(h, n, alpha, mu, topK):
    c = compute_corr(h)
    val = smooth_maxc(c, alpha)
    s = float(h.sum()) - 0.5 * float(n)
    return float(val + mu * s * s)


def lbfgs_jac(h, n, alpha, mu, topK):
    c = compute_corr(h)
    m = float(np.max(c))
    z = alpha * (c - m)
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    w = e / e.sum()
    g = grad_weights(h, w, topK)
    s = float(h.sum()) - 0.5 * float(n)
    g += 2.0 * mu * s
    return g


def lbfgs_polish(h, n, deadline, maxiter=35, alphas=(25, 80), mus=(30, 150), topK=130):
    h = project_sum_box(h, 0.5 * float(n), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() >= deadline:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * int(n)
    for alpha in alphas:
        for mu in mus:
            if time.time() > deadline:
                break
            try:
                res = minimize(
                    lbfgs_fun,
                    best_h,
                    args=(int(n), float(alpha), float(mu), int(topK)),
                    jac=lbfgs_jac,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": int(maxiter), "ftol": 1e-12, "gtol": 1e-7},
                )
                nh = project_sum_box(res.x, 0.5 * float(n), n)
                nc = compute_c5(nh, n)
                if nc < best_c:
                    best_h = nh
                    best_c = nc
            except Exception:
                pass
    return best_h, best_c


def pg_polish(h, n, deadline, max_steps=120, step0=0.008, alphas=(25, 80, 160), topK=130):
    h = project_sum_box(h, 0.5 * float(n), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / float(n)
    for alpha in alphas:
        step = step0
        stall = 0
        for _ in range(int(max_steps)):
            if time.time() > deadline:
                break
            c = compute_corr(h)
            cm = float(np.max(c))
            if cm * dx >= best_c + 1e-12:
                pass
            z = alpha * (c - cm)
            zmax = float(np.max(z))
            e = np.exp(z - zmax)
            w = e / e.sum()
            g = grad_weights(h, w, topK)
            gn = float(np.linalg.norm(g))
            if gn < 1e-11:
                break
            improved = False
            t = step
            for __ in range(9):
                hn = project_sum_box(h - t * g, 0.5 * float(n), n)
                cn = compute_c5(hn, n)
                if cn < best_c - 1e-13:
                    h = hn
                    best_h = h.copy()
                    best_c = cn
                    step = min(step * 1.4, 0.035)
                    improved = True
                    break
                t *= 0.5
            if improved:
                stall = 0
            else:
                step *= 0.5
                stall += 1
                if stall > 6 or step < 1e-8:
                    break
    return best_h, best_c


def solve_coord(c, h, p, n):
    h0 = float(h[p])
    center = n - 1
    m_min = center - p
    end = m_min + n
    base = c[m_min:end]
    a = np.zeros(n, dtype=float)
    if p + 1 < n:
        a[p + 1:] = 1.0 - h[p + 1:n]
    if p > 0:
        a[:p] = -h[:p]
    diag = p
    eps = 1e-14
    neg = a < -eps
    pos = a > eps
    zero = (~neg) & (~pos)
    zero[diag] = False
    any_neg = bool(np.any(neg))
    any_pos = bool(np.any(pos))
    any_zero = bool(np.any(zero))
    if any_neg:
        a_neg = a[neg]
        b_neg = base[neg] - a_neg * h0
    else:
        a_neg = b_neg = np.empty(0)
    if any_pos:
        a_pos = a[pos]
        b_pos = base[pos] - a_pos * h0
    else:
        a_pos = b_pos = np.empty(0)
    if any_zero:
        base_zero = base[zero]
    else:
        base_zero = np.empty(0)
    diag_base = float(base[diag])
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if end < c.size:
        unaff = max(unaff, float(np.max(c[end:])))
    cur_max = float(np.max(c))
    lo = unaff
    hi = max(cur_max + 1.0, lo + 1.0)
    for _ in range(24):
        mid = 0.5 * (lo + hi)
        ok = True
        xl = 0.0
        xh = 1.0
        if ok and any_zero:
            if float(np.max(base_zero)) > mid + 1e-12:
                ok = False
        if ok and any_neg:
            t = float(np.max((mid - b_neg) / a_neg))
            if t > xl:
                xl = t
        if ok and any_pos:
            t = float(np.min((mid - b_pos) / a_pos))
            if t < xh:
                xh = t
        if xl > 1.0 + 1e-12 or xh < -1e-12:
            ok = False
        if ok:
            xl = max(0.0, xl)
            xh = min(1.0, xh)
            if xl > xh + 1e-12:
                ok = False
        if ok:
            Cx = mid - diag_base + h0 - h0 * h0
            if Cx < -1e-12:
                ok = False
            elif Cx < 0.25:
                disc = 1.0 - 4.0 * Cx
                if disc < 0.0:
                    ok = False
                else:
                    rt = float(np.sqrt(disc))
                    r1 = 0.5 - 0.5 * rt
                    r2 = 0.5 + 0.5 * rt
                    ok1 = (max(xl, 0.0) <= min(xh, r1) + 1e-12)
                    ok2 = (max(xl, r2) <= min(xh, 1.0) + 1e-12)
                    if not (ok1 or ok2):
                        ok = False
        if ok:
            hi = mid
        else:
            lo = mid
    L = hi
    xl = 0.0
    xh = 1.0
    if any_zero and float(np.max(base_zero)) > L + 1e-12:
        return h0
    if any_neg:
        t = float(np.max((L - b_neg) / a_neg))
        if t > xl:
            xl = t
    if any_pos:
        t = float(np.min((L - b_pos) / a_pos))
        if t < xh:
            xh = t
    xl = max(0.0, xl)
    xh = min(1.0, xh)
    if xl > xh + 1e-12:
        return h0
    Cx = L - diag_base + h0 - h0 * h0
    if Cx < 0.0:
        Cx = 0.0
    intervals = []
    if Cx >= 0.25:
        intervals.append((xl, xh))
    else:
        disc = max(0.0, 1.0 - 4.0 * Cx)
        rt = float(np.sqrt(disc))
        r1 = 0.5 - 0.5 * rt
        r2 = 0.5 + 0.5 * rt
        lo1 = max(xl, 0.0)
        hi1 = min(xh, r1)
        if lo1 <= hi1 + 1e-12:
            intervals.append((lo1, hi1))
        lo2 = max(xl, r2)
        hi2 = min(xh, 1.0)
        if lo2 <= hi2 + 1e-12:
            intervals.append((lo2, hi2))
    if not intervals:
        return h0
    best_x = h0
    best_d = 2.0
    for a1, a2 in intervals:
        mid = 0.5 * (a1 + a2)
        d = abs(mid - h0)
        if d < best_d:
            best_d = d
            best_x = mid
    return float(np.clip(best_x, 0.0, 1.0))


def coord_polish(h, n, deadline, max_passes=35):
    h = project_sum_box(h, 0.5 * float(n), n)
    c = compute_corr(h)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    center = n - 1
    stagnant = 0
    for pss in range(int(max_passes)):
        if time.time() > deadline:
            break
        changed = False
        order = range(n) if (pss % 2 == 0) else range(n - 1, -1, -1)
        for p in order:
            if time.time() > deadline:
                break
            h0 = float(h[p])
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            x = solve_coord(c, h, p, n)
            delta = x - h0
            if abs(delta) > 1e-13:
                if p + 1 < n:
                    c[center + 1:center + n - p] += delta * (1.0 - h[p + 1:n])
                if p > 0:
                    c[center - p:center] += -delta * h[:p]
                c[center] += (x - x * x) - (h0 - h0 * h0)
                h[p] = x
                changed = True
        h = project_sum_box(h, 0.5 * float(n), n)
        c = compute_corr(h)
        cn = compute_c5(h, n)
        if cn < best_c - 1e-13:
            best_h = h.copy()
            best_c = cn
            stagnant = 0
        else:
            if not changed:
                stagnant += 1
            if stagnant > 10:
                break
    return best_h, best_c


def make_spectral_grid(K, nk=601, nx=201):
    k = np.linspace(0.0, 2.0, nk)
    L = 2.0 - k
    B = np.empty((nk, K), dtype=float)
    for j in range(1, K + 1):
        B[:, j - 1] = 0.5 * L * np.cos(np.pi * j * k) - np.sin(np.pi * j * k) / (2.0 * np.pi * j)
    base = 0.25 * L
    x = np.linspace(0.0, 2.0, nx)
    X = np.empty((nx, K), dtype=float)
    for j in range(1, K + 1):
        X[:, j - 1] = np.cos(np.pi * j * x)
    return base, B, X


def spectral_obj(v, base, B, X, alpha, K):
    a = v[:K]
    G = base - B.dot(a)
    m = float(np.max(G))
    if m < 0.0:
        m = 0.0
    z = alpha * (G - m)
    zm = float(np.max(z))
    val = m + (zm + float(np.log(np.sum(np.exp(z - zm))))) / alpha
    neg = np.maximum(-G, 0.0)
    ns = float(np.sum(neg))
    if ns > 0.0:
        val += 0.25 * ns
    return float(val)


def spectral_norm_con(a):
    k = _SPEC_K
    if k <= 0:
        return 0.5
    return 0.5 - float(np.linalg.norm(a[:k]))


def _spectral_key(item):
    return item[0]


def spectral_candidates(seed=42, max_k=5, max_time_s=140.0):
    t0 = time.time()
    tend = t0 + float(max_time_s)
    rng = np.random.RandomState(int(seed))
    out_scores = []
    for K in range(1, int(max_k) + 1):
        if time.time() > tend:
            break
        base, B, X = make_spectral_grid(K)
        _SPEC_K = K
        cons = [{"type": "ineq", "fun": spectral_norm_con}]
        bounds = [(-0.30, 0.30)] * K
        starts = [np.zeros(K)]
        a1 = np.zeros(K)
        a1[0] = 0.35
        starts.append(a1)
        a2 = np.zeros(K)
        a2[0] = 0.30
        if K > 1:
            a2[1] = 0.05
        starts.append(a2)
        starts.append(rng.normal(0.0, 0.08, K))
        for a in starts:
            if time.time() > tend:
                break
            a = np.asarray(a, dtype=float)
            nov = float(np.linalg.norm(a))
            if nov > 0.45:
                a = a * (0.45 / nov)
            try:
                r = minimize(
                    spectral_obj,
                    a,
                    args=(base, B, X, 30.0, K),
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 130, "ftol": 1e-9},
                )
                x = r.x[:K].astype(float)
                nov = float(np.linalg.norm(x))
                if nov > 0.5:
                    x = x * (0.5 / nov)
                val = spectral_obj(x, base, B, X, 30.0, K)
                if np.all(np.isfinite(x)) and val < 0.45:
                    out_scores.append((val, x))
            except Exception:
                pass
    if not out_scores:
        for K in [1, 2, 3]:
            a = np.zeros(K)
            a[0] = 0.35
            if K > 1:
                a[1] = 0.04
            base, B, X = make_spectral_grid(K)
            out_scores.append((spectral_obj(a, base, B, X, 30.0, K), a))
    out_scores.sort(key=_spectral_key)
    out = []
    for val, a in out_scores:
        a = np.asarray(a, dtype=float)
        dup = False
        for b in out:
            if len(a) == len(b) and np.allclose(a, b, atol=1e-4):
                dup = True
                break
        if not dup:
            out.append(a)
            if len(out) >= 5:
                break
    return out


def h_from_coeffs(coeffs, n):
    n = int(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n, dtype=float)
    for j, cj in enumerate(coeffs, start=1):
        cj = float(cj)
        if abs(cj) > 1e-12:
            h += cj * np.cos(np.pi * j * x)
    return project_sum_box(h, 0.5 * float(n), n)


def make_n_list(base_n):
    raw = [int(base_n), 512, 384, 640, int(round(int(base_n) * 0.8)), int(round(int(base_n) * 1.1))]
    vals = []
    for m in raw:
        m = int(m)
        if m % 2:
            m += 1
        if 128 <= m <= 1000 and m not in vals:
            vals.append(m)
    return vals[:5]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 3.0
    rng = np.random.RandomState(int(seed))

    has_parent = False
    try:
        ih = initial_h_values
        if ih is None:
            raise ValueError
        h_best = np.asarray(ih, dtype=float).copy()
        if len(h_best) < 32:
            raise ValueError
        n_best = int(len(h_best))
        has_parent = True
    except Exception:
        n_best = 512
        if budget > 500.0:
            coeff = np.array([0.35, 0.04, 0.0, 0.0], dtype=float)
            h_best = h_from_coeffs(coeff, n_best)
        else:
            h_best = 0.5 * np.ones(n_best, dtype=float)

    h_best = project_sum_box(h_best, 0.5 * float(n_best), n_best)
    best_c = compute_c5(h_best, n_best)

    coeffs = None
    if not has_parent:
        if time.time() < deadline - 60.0:
            coeffs = spectral_candidates(seed + 7, max_k=4, max_time_s=min(80.0, deadline - time.time() - 10.0))
            if coeffs:
                for n_try in make_n_list(512)[:3]:
                    for cc in coeffs[:3]:
                        hh = h_from_coeffs(cc, n_try)
                        cc5 = compute_c5(hh, n_try)
                        if cc5 < best_c:
                            best_c, h_best, n_best = cc5, hh, n_try
                        if cc5 < 0.34:
                            break

    if has_parent:
        end = min(deadline, t0 + 110.0)
        h1, c1 = lbfgs_polish(h_best, n_best, end, maxiter=28, alphas=(30,), mus=(50, 200), topK=130)
        if c1 < best_c:
            h_best, best_c = h1, c1
        end = min(deadline, t0 + 230.0)
        h1, c1 = pg_polish(h_best, n_best, end, max_steps=130, step0=0.008, alphas=(35, 110), topK=130)
        if c1 < best_c:
            h_best, best_c = h1, c1
        end = min(deadline, t0 + 430.0)
        h1, c1 = coord_polish(h_best, n_best, end, max_passes=26)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if coeffs is None and time.time() < deadline - 180.0:
        ct0 = time.time()
        coeffs = spectral_candidates(seed + 11, max_k=5, max_time_s=min(150.0, deadline - ct0 - 60.0))

    if coeffs is not None and len(coeffs) > 0 and time.time() < deadline - 70.0:
        ns = make_n_list(n_best)
        for n_try in ns:
            if time.time() > deadline - 55.0:
                break
            n_end = min(deadline - 25.0, time.time() + 80.0)
            local_best_h = None
            local_best_c = best_c
            for cc in coeffs[:4]:
                if time.time() > n_end:
                    break
                hh = h_from_coeffs(cc, n_try)
                cc5 = compute_c5(hh, n_try)
                if cc5 > best_c + 0.006 and cc5 > local_best_c + 0.006:
                    continue
                e = min(n_end, time.time() + 18.0)
                if e - time.time() > 3.0:
                    hh, cc5 = lbfgs_polish(hh, n_try, e, maxiter=18, alphas=(60,), mus=(120,), topK=120)
                e = min(n_end, time.time() + 12.0)
                if e - time.time() > 3.0:
                    hh, cc5 = pg_polish(hh, n_try, e, max_steps=70, step0=0.006, alphas=(90,), topK=120)
                if cc5 < local_best_c:
                    local_best_c = cc5
                    local_best_h = hh
            if local_best_h is not None and local_best_c < best_c:
                h_best, best_c, n_best = local_best_h, local_best_c, n_try

    if time.time() < deadline - 55.0:
        for sc in (0.02, 0.05):
            if time.time() > deadline - 30.0:
                break
            hh = project_sum_box(h_best + rng.normal(0.0, sc, n_best), 0.5 * float(n_best), n_best)
            e = min(deadline - 12.0, time.time() + 35.0)
            if e - time.time() > 4.0:
                hh, cc5 = pg_polish(hh, n_best, e, max_steps=90, step0=0.007, alphas=(80,), topK=130)
            e = min(deadline - 6.0, time.time() + 20.0)
            if e - time.time() > 4.0:
                hh, cc5 = coord_polish(hh, n_best, e, max_passes=10)
            if cc5 < best_c:
                h_best, best_c, n_best = hh, cc5, int(len(hh))

    if time.time() < deadline - 10.0:
        e = deadline - 7.0
        h1, c1 = lbfgs_polish(h_best, n_best, e, maxiter=18, alphas=(80,), mus=(80, 250), topK=130)
        if c1 < best_c:
            h_best, best_c = h1, c1

    if time.time() < deadline - 3.0:
        e = deadline - 1.5
        h1, c1 = pg_polish(h_best, n_best, e, max_steps=50, step0=0.005, alphas=(120,), topK=130)
        if c1 < best_c:
            h_best, best_c = h1, c1

    h_best = project_sum_box(h_best, 0.5 * float(len(h_best)), len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, float(best_c), int(len(h_best))
```
