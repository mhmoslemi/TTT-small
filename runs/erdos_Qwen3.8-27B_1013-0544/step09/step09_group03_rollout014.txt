import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def exact_coord_min(c, p, h, n):
    h0 = h[p]
    s_min = p - (n - 1)
    s_max = (n - 1) - p
    if s_max <= s_min:
        return h0
    s = np.arange(s_min, s_max + 1)
    s = s[s != 0]
    if len(s) == 0:
        return h0
    ps = p + s
    pm = p - s
    a = np.zeros(len(s))
    vps = (ps >= 0) & (ps < n)
    vpm = (pm >= 0) & (pm < n)
    a[vps] += (1.0 - h[ps[vps]])
    a[vpm] -= h[pm[vpm]]
    m_idx = n - 1 + s
    b = c[m_idx] - a * h0
    m_min = n - 1 + s_min
    m_max = n - 1 + s_max
    all_m = np.arange(2 * n - 1)
    unaff_mask = (all_m < m_min) | (all_m > m_max)
    M_unaff = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
    c0_base = c[n - 1] - h0 * (1.0 - h0)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        am = float(vals.max())
        if M_unaff > am + 1e-14:
            break
        active = vals >= (am - 1e-13)
        if not active.any():
            break
        min_sl = float(a[active].min())
        max_sl = float(a[active].max())
        if min_sl > 1e-12:
            hi = mid
        elif max_sl < -1e-12:
            lo = mid
        else:
            break
    cands = np.array([lo, hi, (lo + hi) * 0.5, 0.0, 1.0])
    best_x, best_v = h0, float('inf')
    for x in cands:
        v = float((a * x + b).max())
        v = max(v, M_unaff, c0_base + x - x * x)
        if v < best_v:
            best_v = v
            best_x = float(x)
    return best_x

def coord_descent(h, n, max_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            new_val = exact_coord_min(c, p, h, n)
            if abs(new_val - h0) > 1e-10:
                h[p] = new_val
                improved += 1
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-12:
            best_c = new_c
            stall = 0
        else:
            stall += 1
        if improved == 0 or stall > 20:
            break
    return h, best_c

def pair_optimize(h, n, c, i, j):
    """Try optimizing the pair (i,j) jointly by searching over h[i] with h[j] adjusted for sum."""
    hi, hj = h[i], h[j]
    total = h.sum()
    best_c_val = c.max()
    best_hi, best_hj = hi, hj
    # Try several values for h[i], adjust h[j] to preserve sum
    for trial_hi in np.linspace(0, 1, 9):
        trial_hj = hj + (hi - trial_hi)
        if trial_hj < 0 or trial_hj > 1:
            continue
        h[i], h[j] = trial_hi, trial_hj
        c_new = compute_corr(h, n)
        cv = c_new.max()
        if cv < best_c_val:
            best_c_val = cv
            best_hi, best_hj = trial_hi, trial_hj
    h[i], h[j] = best_hi, best_hj
    return best_c_val
