        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD on parent resolution
    h1, c1 = coord_cd(best_h, n_init, max_passes=150, deadline=deadline, shuffle=True, rng=rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Multi-resolution upsampling with CD
    x_cur = np.linspace(0, 2, len(best_h))
    target_ns = [800, 1000, 1200, 1500, 2000]
    for n_try in target_ns:
        if time.time() > deadline - 300:
            break
        if abs(n_try - len(best_h)) < 30:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        passes = max(60, int((deadline - time.time() - 200) * 2))
        h_opt, c_opt = coord_cd(h_int, n_try, max_passes=passes, deadline=deadline,
                                shuffle=True, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 3: Multi-scale perturbations at current resolution
    n_cur = len(best_h)
    if time.time() < deadline - 120:
        h3, c3 = multi_perturb_cd(best_h, n_cur, deadline - 10, rng, n_trials=60)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: SLSQP on reduced grid, then interpolate back
    if time.time() < deadline - 100:
        n_slsqp = min(n_cur, 800)
        x_full = np.linspace(0, 2, n_cur)
        if n_slsqp < n_cur:
            x_s = np.linspace(0, 2, n_slsqp)
            h_s = project_feasible(np.interp(x_s, x_full, best_h), n_slsqp)
            h_s, c_s = slsqp_polish(h_s, n_slsqp, [8000, 30000, 120000], 250, deadline)
            if c_s < best_c:
                best_h = project_feasible(np.interp(x_full, x_s, h_s), n_cur)
                best_c = compute_c5(best_h, n_cur)
        else:
            h_s, c_s = slsqp_polish(best_h, n_cur, [8000, 30000, 120000], 250, deadline)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Phase 5: Final aggressive CD
    n_cur = len(best_h)
    h5, c5 = coord_cd(best_h, n_cur, max_passes=400, deadline=deadline, shuffle=True, rng=rng)
    if c5 < best_c:
        best_h, best_c = h5, c5

    # Phase 6: Final perturbations + CD
    if time.time() < deadline - 30:
        h6, c6 = multi_perturb_cd(best_h, n_cur, deadline - 5, rng, n_trials=25)
        if c6 < best_c:
            best_h, best_c = h6, c6
        h7, c7 = coord_cd(best_h, n_cur, max_passes=200, deadline=deadline)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
