            break

        if shuffle and rng is not None:
            rng.shuffle(order)
        improved = False

            if abs(d) > 1e-14:

                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                delta = np.zeros(len(ms))
                if np.any(vf):

                if np.any(vb):

            break

def block_cd_2d(h, n, deadline=None, rng=None, max_iter=100):
    """Update pairs of coordinates jointly using 2D optimization."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = compute_c5(h, n)
    
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Pick a random pair
        p1 = rng.randint(n)
        p2 = rng.randint(n)
        if p1 == p2:
            continue
        
        # Compute the 2D subproblem
        # c_new(k) = c(k) + d1 * vec1(k) + d2 * vec2(k) + d1*d2 * cross(k)
        # where vec1, vec2 are the contributions of p1, p2
        # and cross is the interaction term
        
        # For simplicity, use a grid search over d1, d2
        # Range: d1 in [-h[p1], 1-h[p1]], d2 in [-h[p2], 1-h[p2]]
        
        # Compute vec1 and vec2 (same as in exact_coord_delta)
        def get_vec(p):
            mlo = max(0, n - 1 - p)
            mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
            ms = np.arange(mlo, mhi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            vec = np.zeros(len(ms))
            if np.any(vf):
                vec[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                vec[vb] -= h[jb[vb]]
            return mlo, mhi, vec
        
        mlo1, mhi1, vec1 = get_vec(p1)
        mlo2, mhi2, vec2 = get_vec(p2)
        
        # Find the overlap range
        mlo = max(mlo1, mlo2, 0)
        mhi = min(mhi1, mhi2, 2 * n - 2)
        
        # For the overlap region, we have c + d1*vec1 + d2*vec2
        # For non-overlap, just c + d1*vec1 or c + d2*vec2
        # For simplicity, do a coarse grid search
        
        n_grid = 10
        d1_vals = np.linspace(-h[p1], 1.0 - h[p1], n_grid)
        d2_vals = np.linspace(-h[p2], 1.0 - h[p2], n_grid)
        
        c_sub = c[mlo:mhi+1].copy()
        v1 = vec1[mlo-mlo1:mhi-mlo1+1]
        v2 = vec2[mlo-mlo2:mhi-mlo2+1]
        
        best_d1, best_d2, best_obj = 0, 0, c.max()
        
        for d1 in d1_vals:
            for d2 in d2_vals:
                vals = c_sub + d1 * v1 + d2 * v2
                # Also need to check non-overlap regions
                obj = max(vals.max(), 
                         c[:mlo].max() if mlo > 0 else 0,
                         c[mhi+1:].max() if mhi < 2*n-2 else 0)
                if obj < best_obj:
                    best_obj = obj
                    best_d1, best_d2 = d1, d2
        
        if best_obj < best_c - 1e-14:
            # Apply the update
            h[p1] += best_d1
            h[p2] += best_d2
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            best_c = compute_c5(h, n)
    
    return h, best_c
