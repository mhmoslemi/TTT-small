- For coordinate p, the number of active values it affects is the sum of active[n-1-p : n-1+p+1]
- This can be computed in O(1) per coordinate using a prefix sum

def sens_order(c, threshold):
    n = len(c) // 2  # c has length 2*dim - 1, so dim = (len(c)+1)//2
    # Actually c has length 2*dim-1 where dim = n
    # Let me just use the length
    L = len(c)
    dim = (L + 1) // 2  # n
    active = (c > threshold).astype(float)
    # Prefix sum
    ps = np.concatenate([[0], np.cumsum(active)])
    sens = np.zeros(dim)
    for p in range(dim):
        lo = max(0, dim-1-p)
        hi = min(L-1, dim-1+p)
        sens[p] = ps[hi+1] - ps[lo]
    return sens

import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * r, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + d / s * h, 0.0, 1.0)
    return h

def solve1(c, h, p, n):
    h0 = h[p]
    lo = max(0, n - 1 - p)
    hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(lo, hi + 1)

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

    if np.any(vb):

def sens_prefix(c, thresh, n):
    L = len(c)
    act = (c > thresh).astype(np.float64)
    ps = np.concatenate([[0.0], np.cumsum(act)])
    s = np.empty(n)
    for p in range(n):
        lo = max(0, n - 1 - p)
        hi = min(L - 1, n - 1 + p)
        s[p] = ps[hi + 1] - ps[lo]
    return s

def cd(h, n, deadline, order='seq', maxp=10000, rng=None):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(maxp):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if order == 'rand':
            if rng is not None:
                idx = rng.permutation(n)
            else:
                idx = np.random.permutation(n)
        elif order == 'sens':
            s = sens_prefix(c, best * 0.93, n)
            idx = np.argsort(-s)
        elif order == 'hybrid':
            s = sens_prefix(c, best * 0.90, n)
            top_k = max(10, n // 3)
            top = np.argsort(-s)[:top_k]
            rest = np.setdiff1d(np.arange(n), top)
            idx = np.concatenate([top, rest])
            if rng is not None:
                order2 = rng.permutation(len(idx))
            else:
                order2 = np.random.permutation(len(idx))
            idx = idx[order2]
        else:
            idx = np.arange(n)
        imp = False
        for p in idx:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve1(c, h, p, n)
            if abs(xn - h0) > 1e-14:
                h[p] = xn
                imp = True
        if not imp:
            break
        h = project(h, n)
        nc = compute_c5(h, n)
        if nc < best:
            best = nc
    return h, best
