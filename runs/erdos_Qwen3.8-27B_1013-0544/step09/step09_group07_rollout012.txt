def compute_gradient_c(h, N):
    """Returns array of shape (2N-1, N) where grad[m,k] = dc_m/dh_k"""
    grad = np.zeros((2*N-1, N))
    # For each m, valid k range:
    # Term 1 valid: 0 <= k-m+N-1 < N => m-N+1 <= k <= m-1 => k in [max(0,m-N+1), min(N-1,m-1)]
    # Term 2 valid: 0 <= k+m-N+1 < N => N-1-m <= k <= N-1+m-N+1 => k in [max(0,N-1-m), min(N-1,m)]
    
    for m in range(2*N-1):
        k1_lo = max(0, m - N + 1)
        k1_hi = min(N-1, m - 1)
        if k1_lo <= k1_hi:
            j1 = np.arange(k1_lo, k1_hi+1) - m + N - 1
            grad[m, k1_lo:k1_hi+1] += (1.0 - h[j1])
        
        k2_lo = max(0, N - 1 - m)
        k2_hi = min(N-1, m)
        if k2_lo <= k2_hi:
            j2 = np.arange(k2_lo, k2_hi+1) + m - N + 1
            grad[m, k2_lo:k2_hi+1] -= h[j2]

    grad = compute_gradient_c(h, N)
    
    h_var = cp.Variable(N)

    constraints = [

        return h

The objective is max_m Σ_i h[j_i^m](1-h[i^m]). 

At the optimum, we want all c_m to be as small as possible. The key trade-off is:
- Making h[i] closer to 0 reduces c_m for shifts where i is the "source" (h[i] multiplies (1-h[j]))
- Making h[i] closer to 1 reduces c_m for shifts where i is the "target" ((1-h[i]) is small)

So the optimal h should make each h[i] either 0 or 1, depending on which role (source or target) matters more for the problematic shifts.

**Algorithm: Role-based assignment**
1. Identify the "problematic" shifts (those with high correlation)
2. For each coordinate h[i], compute:

   - cost_1(i) = increase in max correlation if we set h[i] = 1
3. Assign h[i] = argmin(cost_0, cost_1)
4. Re-optimize

This is essentially what coordinate descent does, but in a more global way.

def sca_lp_step(h, N):
    """One SCA iteration: linearize and solve LP."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Build gradient matrix (sparse structure)
    # grad[m, k] = (1-h[k-m+N-1]) if valid - h[k+m-N+1] if valid
    hvar = cp.Variable(N)
    t = cp.Variable()
    cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == N / 2.0]
    for m in range(2 * N - 1):
        # Build linearized constraint: c[m] + grad[m] @ (hvar - h) <= t
        # grad[m,k] = (1-h[k-m+N-1])*[valid] - h[k+m-N+1]*[valid]
        k = np.arange(N)
        j1 = k - m + N - 1
        j2 = k + m - N + 1
        v1 = (j1 >= 0) & (j1 < N)
        v2 = (j2 >= 0) & (j2 < N)
        g = np.zeros(N)
        if np.any(v1):
            g[v1] += (1.0 - h[j1[v1]])
        if np.any(v2):
            g[v2] -= h[j2[v2]]
        cons.append(c[m] + g @ (hvar - h) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.CLARABEL)
        if prob.status == 'optimal' and hvar.value is not None:
            return project_feasible(np.array(hvar.value), N)
    except Exception:
        pass
    return h

        else:

        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        hn, cn = coord_cd(h_int, n_try, max_passes=80, deadline=deadline - 140, top_k=35)
        if cn < best_c:
            best_h, best_c = hn, cn
            x_orig = x_new

    # Phase 5: Multi-restart
    N_cur = len(best_h)
    if time.time() < deadline - 100:
        for r in range(8):
            if time.time() > deadline - 85:
                break
            s = 0.002 + 0.015 * r
            hp = project_feasible(best_h + rng.normal(0, s, N_cur), N_cur)
            ho, co = coord_cd(hp, N_cur, max_passes=100, deadline=deadline - 80, top_k=40)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Binary projection attempts
    if time.time() < deadline - 80:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, N_cur)
        hb, cb = coord_cd(h_bin, N_cur, max_passes=100, deadline=deadline - 70, top_k=40)
        if cb < best_c:
            best_h, best_c = hb, cb

    # Phase 7: SCA on best
    if time.time() < deadline - 60:
        h7, c7 = sca_phase(best_h, N_cur, max_iters=50, deadline=deadline - 50, damp=0.3)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Phase 8: Final coordinate descent
    if time.time() < deadline - 10:
        h8, c8 = coord_cd(best_h, N_cur, max_passes=500, deadline=deadline - 5, top_k=50)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, N_cur)
    c_flip = compute_c5(h_flip, N_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, N_cur)
    best_c = compute_c5(best_h, N_cur)
    return best_h, best_c, N_cur
