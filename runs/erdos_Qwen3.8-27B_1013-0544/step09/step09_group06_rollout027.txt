                except Exception:
                    return h, compute_c5(h, n)

            return h, compute_c5(h, n)

        x_new = project_feasible(x_new, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def correlations(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, n, m):
    offset = m - (n - 1)
    g = np.zeros(n)
    if offset == 0:
        g[:] = 1.0 - 2.0 * h
    elif offset > 0:
        g[:n - offset] += (1.0 - h[offset:])
        g[n - offset:] += 1.0
        g[offset:] -= h[:n - offset]
    else:
        k = -offset
        g[k:] += (1.0 - h[:n - k])
        g[:k] += 1.0
        g[:n - k] -= h[k:]
    return g

def solve_1d(c_peak, a_coeff, diag_idx, unaff_max, lo, hi):
    T = len(c_peak)
    candidates = [lo, hi]
    for i in range(T):
        for j in range(i + 1, T):
            if i == diag_idx or j == diag_idx:
                continue
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_peak[j] - c_peak[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    if diag_idx is not None:
        for i in range(T):
            if i == diag_idx:
                continue
            B = -(a_coeff[diag_idx] - a_coeff[i])
            C = -(c_peak[diag_idx] - c_peak[i])
            disc = B * B - 4.0 * C

                d1 = (-B + sq) / 2.0

                if lo <= d1 <= hi:
                    candidates.append(d1)

                    candidates.append(d2)
    for i in range(T):

            d = (unaff_max - c_peak[i]) / a_coeff[i]

    best_d = lo

        v = max(float(vals.max()), unaff_max)
        if v < best_val:

            best_d = d
    return best_d, best_val
