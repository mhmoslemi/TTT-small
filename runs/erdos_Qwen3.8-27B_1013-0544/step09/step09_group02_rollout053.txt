def subgradient_momentum(h, n, max_iter=5000, deadline=None):
    h = h.copy()
    v = np.zeros(n)  # momentum
    alpha = 0.1  # step size
    beta = 0.9  # momentum coefficient
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(c))
        d_star = k_max - (n - 1)
        
        # Compute subgradient
        g = np.zeros(n)
        # h[p] appears in h[i+d] when i = p-d: contributes (1-h[p-d])
        # h[p] appears in (1-h[i]) when i = p: contributes -h[p+d]
        idx_fwd = np.arange(n) - d_star  # p - d_star
        idx_bwd = np.arange(n) + d_star  # p + d_star
        valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
        valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
        g[valid_fwd] += (1.0 - h[idx_fwd[valid_fwd]])
        g[valid_bwd] -= h[idx_bwd[valid_bwd]]
        
        # Nesterov momentum
        v = beta * v - alpha * g

            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_coord_delta(h, p, c, n):
    h0 = h[p]
    if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
        return 0.0
    mlo = max(0, n - 1 - p)
    mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    vec = np.zeros(len(ms))
    if np.any(vf):
        vec[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        vec[vb] -= h[jb[vb]]
    c_sub = c[mlo:mhi + 1]
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    lo, hi = -h0, 1.0 - h0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = c_sub + mid * vec
        mval = float(vals.max())
        if mval <= unaff + 1e-14:
            lo = hi = mid
            break
        active = vals >= mval - 1e-10
        vmin = float(vec[active].min())
        vmax = float(vec[active].max())
        if vmin > 1e-10:
            hi = mid
        elif vmax < -1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_cd(h, n, max_passes=100, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for p_idx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            d = exact_coord_delta(h, p, c, n)
            if abs(d) > 1e-13:
                mlo = max(0, n - 1 - p)
                mhi = min(2 * n - 2, max(2 * n - 2 - p, p + n - 1))
                ms = np.arange(mlo, mhi + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                darr = np.zeros(len(ms))
                if np.any(vf):
                    darr[vf] += d * (1.0 - h[jf[vf]])
                if np.any(vb):
                    darr[vb] -= d * h[jb[vb]]
                c[ms] += darr
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        if p_idx % 5 == 4:
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_topk(h, n, max_iter=2000, deadline=None, top_k=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.06
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for ki in top_idx:
            d_s = ki - (n - 1)
            ifb = np.arange(n) - d_s
            ibb = np.arange(n) + d_s
            vf = (ifb >= 0) & (ifb < n)
            vb = (ibb >= 0) & (ibb < n)
            g[vf] += (1.0 - h[ifb[vf]])
            g[vb] -= h[ibb[vb]]
        g /= top_k
        h_new = project_feasible(h - alpha * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            alpha = min(alpha * 1.003, 0.3)
            stall = 0
        else:
            alpha *= 0.85
            stall += 1
            if stall > 60 or alpha < 1e-10:
                break
    return best_h, compute_c5(best_h, n)

def antisym_construct(g, n):
    h = np.zeros(n)
    h[:n // 2] = g
    h[n // 2:] = 1.0 - g
    return h

def antisym_optimize(g, n, max_iter=1000, deadline=None):
    g = np.clip(g.copy(), 0.0, 1.0)
    best_c = compute_c5(antisym_construct(g, n), n)
    best_g = g.copy()
    alpha = 0.04
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h_s = antisym_construct(g, n)
        c = np.correlate(h_s, 1.0 - h_s, mode='full')
        k_max = int(np.argmax(c))
        d_s = k_max - (n - 1)
        g_grad = np.zeros(n // 2)
        for p in range(n // 2):
            gr = 0.0
            q = p
            if 0 <= q - d_s < n:
                gr += (1.0 - h_s[q - d_s])
            if 0 <= q + d_s < n:
                gr -= h_s[q + d_s]
            q2 = p + n // 2
            if 0 <= q2 - d_s < n:
                gr += (1.0 - h_s[q2 - d_s]) * (-1.0)
            if 0 <= q2 + d_s < n:
                gr -= h_s[q2 + d_s] * (-1.0)
            g_grad[p] = gr
        g_new = np.clip(g - alpha * g_grad, 0.0, 1.0)
        h_new = antisym_construct(g_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_g = g_new.copy()
            best_c = c_new
            g = g_new
            alpha = min(alpha * 1.003, 0.2)
            stall = 0
        else:
            alpha *= 0.85
            stall += 1
            if stall > 50 or alpha < 1e-10:
                break
    return best_g, best_c

def block_flip(h, n, rng, block_frac=0.15):
    h = h.copy()
    bsize = max(5, n // 10)
    start = rng.randint(0, n - bsize)
    h[start:start + bsize] = 1.0 - h[start:start + bsize]
    return project_feasible(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Subgradient top-k
    h1, c1 = subgrad_topk(best_h, n_init, max_iter=2000, deadline=deadline, top_k=3)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Exact CD
    h2, c2 = coord_cd(best_h, n_init, max_passes=200, deadline=deadline, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Anti-symmetric track
    if n_init % 2 == 0 and time.time() < deadline - 150:
        g_init = (best_h[:n_init // 2] + (1.0 - best_h[n_init // 2:])) / 2.0
        g_init = np.clip(g_init, 0.0, 1.0)
        g_opt, c_sym = antisym_optimize(g_init, n_init, max_iter=800, deadline=deadline)
        h_sym = antisym_construct(g_opt, n_init)
        c_sym = compute_c5(h_sym, n_init)
        if c_sym < best_c:
            best_h, best_c = h_sym, c_sym

    # Phase 4: Multi-resolution
    x_cur = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 500, 600, 700, 800, 1000, 1200]:
        if time.time() > deadline - 150:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_opt, c_opt = subgrad_topk(h_int, n_try, max_iter=800, deadline=deadline, top_k=3)
        h_opt, c_opt = coord_cd(h_opt, n_try, max_passes=100, deadline=deadline, rng=rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_cur = x_new

    # Phase 5: Block flip perturbations + reoptimize
    n_cur = len(best_h)
    for r in range(12):
        if time.time() > deadline - 80:
            break
        hp = block_flip(best_h, n_cur, rng)
        h_r, c_r = coord_cd(hp, n_cur, max_passes=60, deadline=deadline, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 6: Random noise perturbations
    for r in range(10):
        if time.time() > deadline - 60:
            break
        s = 0.001 + 0.004 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = subgrad_topk(hp, n_cur, max_iter=500, deadline=deadline, top_k=3)
        h_r, c_r = coord_cd(h_r, n_cur, max_passes=80, deadline=deadline, rng=rng)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 7: SLSQP polish
    if time.time() < deadline - 50:
        h5, c5 = slsqp_polish(best_h, len(best_h), [5000, 20000, 80000], 400, deadline)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 8: Final CD
    h6, c6 = coord_cd(best_h, len(best_h), max_passes=400, deadline=deadline, rng=rng)
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 9: Final subgradient
    if time.time() < deadline - 10:
        h7, c7 = subgrad_topk(best_h, len(best_h), max_iter=1000, deadline=deadline, top_k=5)
        if c7 < best_c:
            best_h, best_c = h7, c7
        h8, c8 = coord_cd(best_h, len(best_h), max_passes=200, deadline=deadline, rng=rng)
        if c8 < best_c:
            best_h, best_c = h8, c8

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
