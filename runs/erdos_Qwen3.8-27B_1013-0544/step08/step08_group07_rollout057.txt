import time

    return float(np.correlate(h, 1.0 - h, mode='full').max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    lse = smax + np.log(np.sum(np.exp(s - smax)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def lbfgs_step(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:

                    if -v < d < 1.0 - v:

            # Top few line intersections

                    x1, x2 = top_idx[ii], top_idx[jj]
                    denom = a[x2] - a[x1]
                    if abs(denom) > 1e-12:
                        d = (c_aff[x1] - c_aff[x2]) / denom
                        if -v < d < 1.0 - v:
                            cands.append(d)
            
            best_d, best_val = 0.0, curr_max
            for d in cands:
                d2 = max(-v, min(1.0 - v, d))
                if abs(d2) < 1e-14:
                    continue
                nv = c_aff + d2 * a
                if has_d:
                    nv[di] -= d2 * d2
                val = max(unaff, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d2
            
            if abs(best_d) > 1e-14 and best_val < curr_max - 1e-15:
                h[p] = v + best_d
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if has_d:
                    c_prev[diag] -= best_d * best_d
        
        if not improved:
            break
        h = project_feasible(h, n)
        c_prev = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    lse = sm + np.log(np.sum(np.exp(s - sm)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def cd_fast(h, n, max_passes, deadline, threshold=0.03, top_k=12):
    h = h.copy()
    c_prev = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        curr_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                return h, compute_c5(h, n)
            v = h[p]
            if v < 1e-10 or v > 1.0 - 1e-10:
                continue
            j_lo = max(0, min(p - 1, n - 1 - p))
            j_hi = min(2 * n - 2, p + n - 1)
            if j_lo >= j_hi:
                continue
            ms = np.arange(j_lo, j_hi + 1)
            i1 = p - ms + (n - 1)
            i2 = p + ms - (n - 1)
            a = np.zeros(len(ms))
            vf = (i1 >= 0) & (i1 < n)
            a[vf] += (1.0 - h[i1[vf]])
            vb = (i2 >= 0) & (i2 < n)
            a[vb] -= h[i2[vb]]
            c_aff = c_prev[ms]
            unaff = curr_max
            if j_lo > 0:
                t = float(c_prev[:j_lo].max())
                if t > unaff:
                    unaff = t
            if j_hi < 2 * n - 2:
                t = float(c_prev[j_hi + 1:].max())
                if t > unaff:
                    unaff = t
            diag = n - 1
            has_d = j_lo <= diag <= j_hi
            di = diag - j_lo if has_d else -1
            cands = [-v, 1.0 - v]
            thr = unaff - threshold
            act_mask = c_aff > thr
            act_idx = np.where(act_mask)[0]
            for idx in act_idx:
                if abs(a[idx]) > 1e-12:
                    d = (unaff - c_aff[idx]) / a[idx]
                    if -v < d < 1.0 - v:
                        cands.append(d)
            if len(act_idx) > 1:
                si = act_idx[np.argsort(c_aff[act_idx])][-top_k:]
                for ii in range(len(si)):
                    for jj in range(ii + 1, len(si)):
                        x1, x2 = si[ii], si[jj]
                        denom = a[x2] - a[x1]
                        if abs(denom) > 1e-12:
                            d = (c_aff[x1] - c_aff[x2]) / denom
                            if -v < d < 1.0 - v:
                                cands.append(d)
            best_d, best_val = 0.0, curr_max

                    continue

                if has_d:
                    nv[di] -= d2 * d2
                val = max(unaff, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d2
            if abs(best_d) > 1e-13 and best_val < curr_max - 1e-15:
                h[p] = v + best_d
                improved = True
                c_prev[ms] = c_aff + best_d * a
                if has_d:
                    c_prev[diag] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        c_prev = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_shake(h, n, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    active = np.where(c > mx - 1e-6)[0]
    sg = np.zeros(n)
    for k in active:
        for p in range(n):
            d1 = p - k + (n - 1)
            d2 = p + k - (n - 1)
            if 0 <= d1 < n:
                sg[p] += 1.0 - h[d1]
            if 0 <= d2 < n:
                sg[p] -= h[d2]
    sg /= len(active)
    h2 = project_feasible(h - lr * sg, n)
    return h2

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        hh = 0.5 + 0.35 * np.sin(2 * np.pi * x)
    elif kind == 1:

    elif kind == 2:
        hh = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif kind == 3:
        hh = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x + 1.0)
    else:
        hh = rng.uniform(0, 1, n)
    return project_feasible(hh, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    deadline = t0 + budget_s - 5

    dl1 = min(deadline, t0 + 120)
    h1, c1 = lbfgs_opt(best_h, n, [500, 5000, 50000, 200000], 600, dl1)
    if c1 < best_c:

    # Phase 2: Coordinate descent (main)
    dl2 = min(deadline, t0 + 500)
    h2, c2 = cd_fast(best_h, n, 400, dl2)
    if c2 < best_c:
        best_h, best_c = h2, c1 if False else c2

    # Phase 3: Subgradient shake + re-optimize (escape local opt)
    for shake_round in range(5):
        if time.time() > deadline - 200:
            break
        lr = 0.002 / (shake_round + 1)
        h_sh = subgrad_shake(best_h, n, lr, deadline)
        h_sh, c_sh = cd_fast(h_sh, n, 150, min(deadline - 50, time.time() + 120))
        if c_sh < best_c:

    for n_try in [350, 450, 600]:

        x_big = np.linspace(0, 2, len(best_h), endpoint=False)

        h_t = project_feasible(np.interp(x_t, x_big, best_h) + 0.03 * rng.randn(n_try), n_try)
        h_t, c_t = lbfgs_opt(h_t, n_try, [5000, 50000], 400, min(deadline - 30, time.time() + 60))
        h_t, c_t = cd_fast(h_t, n_try, 150, min(deadline - 10, time.time() + 80))
        if c_t < best_c:

        if cf < best_c:

    best_c = compute_c5(best_h, len(best_h))
