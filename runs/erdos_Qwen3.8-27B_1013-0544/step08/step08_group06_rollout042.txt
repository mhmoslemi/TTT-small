    h = project_feasible(np.array(h, dtype=float).copy(), n)
    target = 0.5 * n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = build_A(h, n)
    diag = n - 1
    mN = 2 * n - 1
    idx = np.arange(n)
    cur = float(corr.max())
    best_c = cur * dx
    best_h = h.copy()
    s = float(h.sum())
    pcount = 0
    stall = 0
    while time.time() < deadline:
        pcount += 1
        order = idx if (pcount % 2 == 0) else idx[::-1]
        accepted = 0
        for p in order:
            if time.time() > deadline: break
            old = float(h[p])
            if old <= 1e-12:
                lo, hi = 0.0, 1.0
            elif old >= 1.0 - 1e-12:
                lo, hi = -old, 0.0
            else:
                lo, hi = -old, 1.0 - old
            if hi - lo < 1e-12: continue
            off = p if p < diag - p else diag - p
            start = off
            end = mN - 1 - off
            unaff = 0.0
            if start > 0:
                u = float(corr[:start].max())
                if u > unaff: unaff = u
            if end < mN - 1:
                u = float(corr[end+1:].max())
                if u > unaff: unaff = u
            if unaff >= cur - 1e-10:
                continue
            ca = corr[start:end+1]
            aa = A[p, start:end+1]
            dpos = diag - start
            G = 64 if grid <= 0 else int(grid)
            dg = np.linspace(lo, hi, int(G))
            vals = ca[None, :] + aa[None, :] * dg[:, None]
            vals[:, dpos] -= dg[:, None] ** 2
            maxvals = vals.max(axis=1)
            if unaff > 0.0:
                maxvals = np.maximum(unaff, maxvals)
            im = int(np.argmin(maxvals))
            bd = float(dg[im])
            bv = float(maxvals[im])
            if refine and 0 < im < int(G)-1 and bv < cur + 1.0:
                x0=float(dg[im-1]); x1=bd; x2=float(dg[im+1])
                y0=float(maxvals[im-1]); y1=bv; y2=float(maxvals[im+1])
                den = y0*(x1-x2)+y1*(x2-x0)+y2*(x0-x1)
                if abs(den) > 1e-14:
                    xp = 0.5*(y0*(x1-x2)+y1*(x2-x0)+y2*(x0-x1))/den
                    if x0 < xp < x2:
                        tmp = ca + aa*xp
                        tmp[dpos] -= xp*xp
                        v = float(tmp.max())
                        if unaff > v: v = unaff
                        if v < bv:
                            bv=v; bd=xp
                if bv < cur - 1e-10 and (x2 - x0) > 1e-9:
                    left=x0; right=x2
                    for k in range(1, 9):
                        cand = left + (right-left)*(k/9.0)
                        tmp = ca + aa*cand
                        tmp[dpos] -= cand*cand
                        v = float(tmp.max())
                        if unaff > v: v = unaff
                        if v < bv:
                            bv=v; bd=cand
            if bv < cur - 1e-10 and abs(bd) > 1e-11:
                d = bd
                corr[start:end+1] += d * aa
                corr[diag] -= d*d
                A[idx, idx - p + diag] -= d
                A[idx, p - idx + diag] -= d
                h[p] = float(np.clip(old + d, 0.0, 1.0))

                accepted += 1

def project_feasible(h,n):
    target = 0.5*n
    h = np.clip(h.astype(float, copy=True), 0.0, 1.0)
    for _ in range(200):
       diff = target - h.sum()
       if abs(diff) < 1e-13: break
       if diff > 0:
          room = 1.0 - h
          tot = room.sum()
          if tot <= 1e-15: break
          h += (diff/tot)*room
       else:
          tot = h.sum()
          if tot <= 1e-15: break
          h += (diff/tot)*h
    h = np.clip(h,0,1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
       if diff > 0:
          idx = int(np.argmax(1.0-h))
          h[idx] = min(1.0, h[idx]+diff)
       else:
          idx = int(np.argmax(h))
          h[idx] = max(0.0, h[idx]-diff)
    return h

import numpy as np
import time
import math


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] - diff)
    return h


def build_A(h, n):
    mN = 2 * n - 1
    diag = n - 1
    A = np.zeros((n, mN), dtype=float)
    for p in range(n):
        lo = p if p < diag - p else diag - p
        hi = 2 * diag - lo
        ms = np.arange(lo, hi + 1)
        q = p + ms - diag
        r = p + diag - ms
        a = np.zeros(len(ms), dtype=float)
        vq = (q >= 0) & (q < n)
        a[vq] += 1.0 - h[q[vq]]
        vr = (r >= 0) & (r < n)
        a[vr] -= h[r[vr]]
        A[p, lo:hi + 1] = a
    return A


def fast_cd(h, n, deadline, grid=48, refine=True, drift_tol=0.3):
    h = project_feasible(h, n)
    target = 0.5 * n
    dx = 2.0 / n
    diag = n - 1
    mN = 2 * n - 1
    idx = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = build_A(h, n)
    cur = float(corr.max())
    best_raw = cur
    best_c = cur * dx
    best_h = h.copy()
    s = float(h.sum())
    stall = 0
    pcount = 0
    G = max(16, int(grid))

    while time.time() < deadline:
        pcount += 1
        accepted = 0
        order = idx if (pcount % 2 == 0) else idx[::-1]
        for p in order:
            if time.time() > deadline:
                break
            old = float(h[p])
            if old <= 1e-12:
                lo, hi = 0.0, 1.0
            elif old >= 1.0 - 1e-12:
                lo, hi = -old, 0.0
            else:
                lo, hi = -old, 1.0 - old
            if hi - lo < 1e-12:
                continue

            off = p if p < diag - p else diag - p
            start = off
            end = mN - 1 - off
            unaff = 0.0
            if start > 0:
                u = float(corr[:start].max())
                if u > unaff:
                    unaff = u
            if end < mN - 1:
                u = float(corr[end + 1:].max())
                if u > unaff:
                    unaff = u
            if unaff >= cur - 1e-10:
                continue

            ca = corr[start:end + 1]
            aa = A[p, start:end + 1]
            dpos = diag - start
            dg = np.linspace(lo, hi, G)
            vals = ca[None, :] + aa[None, :] * dg[:, None]
            vals[:, dpos] -= dg * dg
            maxvals = vals.max(axis=1)
            if unaff > 0.0:

            bd = float(dg[im])
            bv = float(maxvals[im])

                x2 = float(dg[im + 1])
                y0 = float(maxvals[im - 1])
                y1 = bv

                den = y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)

                    if x0 < xp < x2:

                        tmp[dpos] -= xp * xp
                        v = float(tmp.max())

                nv = old + bd

                    nv = 0.0

                A[idx, idx - p + diag] -= d

                cur = float(corr.max())
                accepted += 1

                        s = float(h.sum())

                        A = build_A(h, n)
                        cur = float(corr.max())

        if time.time() > deadline:
            break

        if accepted > 0:
            h = project_feasible(h, n)
            s = float(h.sum())
            corr = np.correlate(h, 1.0 - h, mode='full')
            A = build_A(h, n)
            cur = float(corr.max())
            if cur < best_raw - 1e-12:
                best_raw = cur
                best_c = cur * dx

            elif cur > best_raw + 1.0:
                h = best_h.copy()
                s = float(h.sum())
                corr = np.correlate(h, 1.0 - h, mode='full')
                A = build_A(h, n)
                cur = float(corr.max())
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)


def make_binary(h, n):
    target = n // 2
    hb = (h > 0.5).astype(float)
    cnt = int(hb.sum())
    if cnt > target:
        idx = np.where(hb > 0.5)[0]
        order = idx[np.argsort(h[idx])]
        hb[order[:cnt - target]] = 0.0
    elif cnt < target:
        idx = np.where(hb == 0.0)[0]
        order = idx[np.argsort(h[idx])[::-1]]
        hb[order[:target - cnt]] = 1.0
    return hb


def binary_structures(h, n):
    ones = []
    zeros = []
    pos_o = [-1] * n
    pos_z = [-1] * n
    for k in range(n):
        if h[k] > 0.5:
            pos_o[k] = len(ones)
            ones.append(k)
        else:
            pos_z[k] = len(zeros)
            zeros.append(k)
    return ones, zeros, pos_o, pos_z


def apply_flip(corr, h, k, d, fm, sm, diag):
    fq = fm[k]
    sq = sm[k]
    corr[fq] += d * (1.0 - h)
    corr[sq] -= d * h
    corr[diag] -= d * d
    h[k] = h[k] + d


def build_flip_indices(n):
    diag = n - 1
    q = np.arange(n, dtype=np.int32)
    fm = [np.asarray(diag + k - q, dtype=np.int32) for k in range(n)]
    sm = [np.asarray(diag - k + q, dtype=np.int32) for k in range(n)]
    return fm, sm, diag


def binary_sa(h, n, deadline, rng, tau0=4.0, tau1=0.05):
    if n % 2 != 0:
        return h, compute_c5(h, n)
    h = make_binary(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur = float(corr.max())
    best_raw = cur
    best_h = h.copy()
    fm, sm, diag = build_flip_indices(n)
    ones, zeros, pos_o, pos_z = binary_structures(h, n)
    T = tau0
    it = 0
    stall = 0

    while time.time() < deadline:
        it += 1
        if not ones or not zeros:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        apply_flip(corr, h, i, -1.0, fm, sm, diag)
        apply_flip(corr, h, j, 1.0, fm, sm, diag)
        new = float(corr.max())
        delta = new - cur

        accept = delta < -1e-12
        if not accept and delta <= T:
            prob = 0.5 if delta < 1e-12 else math.exp(-delta / T)
            accept = rng.random_sample() < prob

        if accept:
            pi = pos_o[i]
            last_o = ones[-1]
            ones[pi] = last_o
            pos_o[last_o] = pi
            ones.pop()
            pos_o[i] = -1
            pos_z[i] = len(zeros)
            zeros.append(i)

            pj = pos_z[j]
            last_z = zeros[-1]
            zeros[pj] = last_z
            pos_z[last_z] = pj
            zeros.pop()
            pos_z[j] = -1
            pos_o[j] = len(ones)
            ones.append(j)

            cur = new
            stall = 0
            if cur < best_raw - 1e-12:
                best_raw = cur
                best_h = h.copy()

            apply_flip(corr, h, j, -1.0, fm, sm, diag)

            stall += 1
            if stall > 20000:
                h = best_h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur = float(corr.max())
                ones, zeros, pos_o, pos_z = binary_structures(h, n)
                T = max(T, 1.0)
                stall = 0

        T = max(tau1, T * 0.9997)
        if it > 200000 and T <= tau1 * 1.1:
            break

    return best_h, best_raw * dx


def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)
