        if np.clip(z - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        if res > 0:
            m = y < 1.0
            r = (1.0 - y[m]).sum()
            if r > 1e-12: y[m] += (res/r)*(1.0-y[m])
        else:
            m = y > 0.0
            r = y[m].sum()
            if r > 1e-12: y[m] += (res/r)*y[m]
    return np.clip(y, 0.0, 1.0)

def get_f_offsets(h, n, active_lags):
    """Compute f_d(p) for active lags. Returns dict: lag_offset -> array of f values."""
    off = n - 1
    result = {}
    for k in active_lags:
        d = k - off
        f = np.zeros(n)
        if d >= 0:
            m = n - d
            f[:m] = (1.0 - h[d:]) - np.concatenate([h[:m], np.zeros(d)])[:m]
            # f[p] = (1-h[p+d]) if p+d<n else 0) - (h[p-d]) if p-d>=0 else 0
            f[:m] += np.concatenate([np.zeros(d), h[:m]])[:m] * 0  # fix
            f = np.zeros(n)
            f[:m] += (1.0 - h[d:])
            f[d:] -= h[:m]
        else:
            e = -d
            m = n - e
            f[e:] += (1.0 - h[:m])
            f[:m] -= h[e:]
        result[d] = f
    return result

def pair_move_eval(h, c, n, i, j, t):
    """Evaluate new c after h[i]+=t, h[j]-=t. Returns new c array and new max."""
    cn = c.copy()
    off = n - 1
    old_i, old_j = h[i], h[j]
    for k in range(2*n-1):
        d = k - off
        if d >= 0:
            m = n - d
            # check if i or j are in [0, m-1] or [d, n-1]
            delta = 0.0
            if i < m: delta += t * (1.0 - old_j) if (i+d==j) else t*(1.0-h[i+d]) if i+d<n else 0.0
            # This is getting complicated, just recompute affected terms
            pass
    # Simpler: just recompute full correlation
    hi = h.copy()
    hi[i] += t
    hi[j] -= t
    cn = np.correlate(hi, 1.0-hi, mode='full')
    return cn, float(cn.max())

def exact_pair_search(h, n, i, j, c, deadline):
    """Find optimal t for pair (i,j) using bisection on the max correlation."""
    a = h[i]; b = h[j]
    tlo = max(-a, -(1.0-b))
    thi = min(1.0-a, b)
    if thi - tlo < 1e-12:
        return 0.0
    # Sample to find approximate best
    best_t = 0.0
    best_val = float(c.max())
    for _ in range(20):
        t = tlo + (thi-tlo)*np.random.random()
        hi = h.copy(); hi[i]=a+t; hi[j]=b-t
        cn = np.correlate(hi, 1.0-hi, mode='full')
        v = float(cn.max())
        if v < best_val:
            best_val = v; best_t = t
    return best_t

def intensive_pair_search(h, n, budget, rng):
    """Intensive pair search with systematic pair selection."""
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    best_c = float(c.max())
    off = n - 1
    t_start = time.time()
    no_improve = 0
    max_no_improve = 200
    step = 0.01
    while time.time() < t_start + budget:
        cur_max = float(c.max())
        # Find active lags
        tol = max(0.01, 0.005*cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        # Compute f for the most active lag
        kk = active[0]
        d = kk - off
        f = np.zeros(n)
        if d >= 0:
            m = n - d
            f[:m] += (1.0 - h[d:])
            f[d:] -= h[:m]
        else:
            e = -d
            m = n - e
            f[e:] += (1.0 - h[:m])
            f[:m] -= h[e:]
        # Find pairs with large |f[i]-f[j]|
        # Top candidates: high f and low f
        sorted_idx = np.argsort(f)
        n_try = min(50, n)
        candidates = []
        for idx in range(min(20, n_try)):
            candidates.append((sorted_idx[idx], sorted_idx[n-1-idx]))
        # Also random pairs
        for _ in range(20):
            candidates.append((rng.randint(n), rng.randint(n)))
        best_move = None
        best_new_max = cur_max
        for (i, j) in candidates:
            if i == j: continue
            a = h[i]; b = h[j]
            # Determine sign: we want f[i]-f[j] to have opposite sign to desired direction
            # If f[i] < f[j], increasing h[i] decreases c at this lag
            tlo = max(-a, -(1.0-b))
            thi = min(1.0-a, b)
            if thi - tlo < 1e-12: continue
            # Quick evaluation at a few points
            for frac in [0.25, 0.5, 0.75]:
                t = tlo + frac*(thi-tlo)
                # Clamp
                t = max(-a, min(1.0-a, t))
                t = max(-(1.0-b), min(b, t))
                if abs(t) < 1e-12: continue
                hi = h.copy()
                hi[i] = a + t
                hi[j] = b - t
                cn = np.correlate(hi, 1.0-hi, mode='full')
                v = float(cn.max())
                if v < best_new_max - 1e-10:
                    best_new_max = v
                    best_move = (i, j, t)
        if best_move is not None:
            i, j, t = best_move
            h[i] += t
            h[j] -= t
            c = np.correlate(h, 1.0-h, mode='full')
            no_improve = 0
            step = min(0.05, step*1.5)
        else:
            no_improve += 1
            step = max(0.001, step*0.8)
            if no_improve > max_no_improve:
                # Perturb and reset
                h = project_feasible(h + rng.normal(0, 0.05, n), n)
                c = np.correlate(h, 1.0-h, mode='full')
                no_improve = 0
                max_no_improve = 100
    return h, compute_c5(h, n)

def grad_smooth(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
    m = c.max()
    z = alpha*(c - m)
    p = np.exp(z - z.max()); p /= p.sum()
    g = np.zeros(n)
    off = n-1
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-13: continue
        dd = k - off
        if dd >= 0:
            g[dd:] += w*(1.0-h[:n-dd])
            g[:n-dd] -= w*h[dd:]
        else:
            e = -dd
            g[:n-e] += w*(1.0-h[e:])
            g[e:] -= w*h[:n-e]
    return g

def obj_smooth(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
    m = c.max()
    z = alpha*(c-m)
    return m + np.log(np.exp(z).sum())/alpha

def sum_con(x, n):
    return float(x.sum() - 0.5*n)

def sum_jac(x, n):
    return np.ones_like(x)

def interp_h(h, m):
    n = len(h)
    if n == m: return project_feasible(h.copy(), m)
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_short(h, n, deadline, rng):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    # Symmetry
    hs = project_feasible(0.5*(h+h[::-1]), n)
    if compute_c5(hs, n) < c: h, c = hs, compute_c5(hs, n)
    bounds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]
    if time.time() < deadline - 10:
        try:
            res = minimize(obj_smooth, h, args=(n, 200.0), method='SLSQP',
                          jac=lambda x,nn,aa: grad_smooth(x,nn,aa),
                          bounds=bounds, constraints=cons,
                          options={'maxiter':50,'ftol':1e-12,'disp':False})
            x = np.asarray(res.x)
            if np.all(np.isfinite(x)):
                x = project_feasible(x, n)
                if compute_c5(x,n) < c: h, c = x, compute_c5(x,n)
        except: pass
    # Pair search
    rem = deadline - time.time()
    if rem > 5 and n <= 600:
        h, c = intensive_pair_search(h, n, rem-2, rng)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 10 or not np.all(np.isfinite(h0)): raise ValueError
    except:
        x = np.linspace(0, 2, 200, endpoint=False)
        h0 = np.clip(0.5 + 0.2*np.sin(np.pi*x) + 0.1*np.cos(2*np.pi*x), 0, 1)
        h0 = project_feasible(h0, len(h0))
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Quick symmetry
    hs = project_feasible(0.5*(best_h+best_h[::-1]), n0)
    if compute_c5(hs,n0) < best_c: best_h, best_c = hs, compute_c5(hs,n0)
    hc = 1.0-best_h
    if compute_c5(hc,n0) < best_c: best_h, best_c = hc, compute_c5(hc,n0)
    
    # Phase 1: SLSQP polish on initial
    p1_dead = min(hard - 120, t0 + 80)
    if p1_dead > t0 + 20:
        best_h, best_c = optimize_short(best_h, n0, p1_dead, rng)
    
    # Phase 2: Intensive pair search at current resolution
    p2_dead = min(hard - 80, t0 + 500)
    if p2_dead > time.time() + 20:
        best_h, best_c = intensive_pair_search(best_h, len(best_h), p2_dead - time.time(), rng)
    
    # Phase 3: Try higher resolution
    for m in [600, 800, 1000]:
        if time.time() > hard - 40: break
        if m <= len(best_h): continue
        rem = hard - time.time()
        if rem < 50: break
        dl = time.time() + min(40, 0.3*rem)
        if dl < time.time() + 10: continue
        hm = interp_h(best_h, m)
        hm, cm = optimize_short(hm, m, dl, rng)
        if cm < best_c:
            best_h, best_c = hm, cm
    
    # Phase 4: Final pair search
    if time.time() < hard - 10:
        rem = hard - time.time() - 5
        if rem > 10 and len(best_h) <= 800:
            best_h, best_c = intensive_pair_search(best_h, len(best_h), rem, rng)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
