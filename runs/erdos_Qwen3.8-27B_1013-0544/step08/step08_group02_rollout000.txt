        return None
    us = np.linspace(lo - hi, hiT - hi, max(3, int(grid)))
    B = A[i] - A[j]
    lines = c + us[:, None] * B[None, :]
    off = n - 1
    u2 = us * us
    lines[:, off] -= 2.0 * u2
    lines[:, off + i - j] += u2
    lines[:, off + j - i] += u2
    vals = lines.max(axis=1)
    idx = int(np.argmin(vals))
    return float(us[idx]), float(vals[idx])

def pair_search_sa(h, n, max_tries, grid, deadline, rng, T0=0.004, T1=0.00005):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    cur_max = float(c.max())
    best_max = cur_max
    best_h = h.copy()
    accept_count = 0
    stall = 0

    for trial in range(max_tries):
        if time.time() > deadline:
            break
        frac = trial / max(1, max_tries)
        T = T0 * (T1 / T0) ** frac
        if stall > 200:
            T = max(T, T0 * 0.3)  # reheat
            stall = 0

        cur_max = float(c.max())
        if cur_max < best_max - 1e-10:
            best_max = cur_max
            best_h = h.copy()
            accept_count += 1
            stall = 0
        else:
            stall += 1

        if stall > max(800, max_tries // 4):
            break

        tol = max(0.008, 0.0015 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size > 0 and rng.random() < 0.7:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - (n - 1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    i = int(rng.randint(n)); j = int(rng.randint(n))
                    if i == j: j = (j + 1) % n
                else:
                    b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    i = int(rng.randint(n)); j = int(rng.randint(n))
                    if i == j: j = (j + 1) % n
                else:
                    b = e + int(rng.randint(m)); i = b - e; j = b
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j: j = (j + 1) % n
            if i == j: continue

        result = eval_pair(h, c, A, i, j, n, grid)
        if result is None:
            continue
        u, new_max = result
        if abs(u) < 1e-15:
            continue

        delta = new_max - cur_max
        if delta < -1e-12 or (T > 1e-12 and rng.random() < np.exp(-max(0, delta) / T)):
            old = h[i] + h[j]
            h[i] = np.clip(h[i] + u, 0.0, 1.0)
            h[j] = old - h[i]
            B = A[i] - A[j]
            u2 = u * u
            c = c + u * B
            off = n - 1
            c[off] -= 2.0 * u2
            c[off + i - j] += u2
            c[off + j - i] += u2
            if accept_count % 30 == 0 and accept_count > 0:
                c = np.correlate(h, 1.0 - h, mode='full')
                A = compute_A(h, n)

    if best_max <= float(c.max()):
        return best_h, best_max
    return h, float(c.max())

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_at(n, start_h, deadline, rng, quick=False):
    h = project_feasible(start_h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c: h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c: h, c = hc, cs
    hsc = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    cs = compute_c5(hsc, n)
    if cs < c: h, c = hsc, cs

    if not quick and time.time() < deadline - 20.0:
        plans = [(30, 30), (200, 25), (1000, 20), (5000, 15)]
        h, c = slsqp_opt(h, n, deadline, plans)
    elif time.time() < deadline - 5.0:
        plans = [(50, 20), (500, 15)]
        if n <= 250:
            plans.append((2000, 12))
        h, c = slsqp_opt(h, n, deadline, plans)

    rem = deadline - time.time()
    if rem > 3.0:
        mt = int(min(12000 if not quick else 4000, max(300, rem * 30)))
        grid = 50 if n <= 300 else 35
        h, c = pair_search_sa(h, n, mt, grid, deadline, rng)

    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        rng2 = np.random.RandomState(seed + 1)
        x = np.linspace(0, 2, 200, endpoint=False)
        h0 = np.clip(0.5 + 0.3*np.sin(np.pi*x) + 0.1*np.sin(2*np.pi*x) + rng2.normal(0,0.05,200), 0, 1)
        h0 = project_feasible(h0, 200)

    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    if hard - t0 < 25.0:
        return best_h, best_c, n0

    # Phase 1: Optimize at parent resolution
    dl1 = hard - max(150.0, 0.3 * float(budget_s))
    best_h, best_c = optimize_at(n0, best_h, dl1, rng, False)

    # Phase 2: Try smaller resolutions (fewer lags -> potentially lower max)
    small_ns = [100, 128, 150, 200, 256]
    for m in small_ns:
        if time.time() > hard - 40.0:
            break
        rem = hard - time.time()
        if rem < 30.0:
            break
        dl = time.time() + min(50.0, 0.2 * rem)
        if dl < time.time() + 8.0:
            continue
        ht, ct = optimize_at(m, interp_h(best_h, m), dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 3: Try larger resolutions
    for m in [640, 800]:
        if time.time() > hard - 35.0:
            break
        rem = hard - time.time()
        if rem < 50.0:
            break
        dl = time.time() + min(45.0, 0.18 * rem)
        ht, ct = optimize_at(m, interp_h(best_h, m), dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 4: Perturbation restarts at best resolution
    n_best = len(best_h)
    for s in [0.03, 0.06, 0.12, 0.25]:
        if time.time() > hard - 25.0:
            break
        hp = project_feasible(best_h + rng.normal(0.0, s, n_best), n_best)
        dl = min(hard - 5.0, time.time() + 35.0)
        ht, ct = optimize_at(n_best, hp, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 5: Final intensive pair search
    if time.time() < hard - 10.0:
        rem = hard - time.time()
        mt = int(min(20000, max(500, rem * 35)))
        grid = 50 if len(best_h) <= 300 else 35
        best_h, best_c = pair_search_sa(best_h, len(best_h), mt, grid, hard - 2.0, rng,
                                         T0=0.003, T1=0.00003)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
