            h_o, c_o = slsqp_optimize(hp, n, [50000, 500000], 200, deadline)
            rem = deadline - time.time() - 20
            mp = max(3, int(rem / 3.5))
            h_o, c_o = cd_exact(h_o, n, mp, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final polishing
    if time.time() < deadline - 15:
        rem = deadline - time.time() - 10
        mp = max(5, int(rem / 3.0))
        hf, cf = cd_exact(best_h, n, mp, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
