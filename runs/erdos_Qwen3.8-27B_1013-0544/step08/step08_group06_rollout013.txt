def project_feasible(h, n):

            if tot < 1e-15:

            h += (diff / tot) * h

def smooth_max(c, n, alpha):
    """Smooth approximation of max(c) * dx"""
    m = c.max()

    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, n, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max of correlation w.r.t. h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    p = exp_shifted / exp_shifted.sum()  # softmax weights over lags
    
    # p has length 2n-1, indexed 0..2n-2
    # p[k] corresponds to lag m = k - (n-1)
    # gradient[j] = dx * sum_k p[k] * dc[k]/dh[j]
    # dc[k]/dh[j] = (1-h[j+m])*I[valid1] - h[j-m]*I[valid2]
    # where m = k-(n-1), valid1: 0<=j<n and 0<=j+m<n
    #                        valid2: 0<=j-m<n and 0<=j<n
    
    grad = np.zeros(n)
    # For each lag k (0 to 2n-2):
    # m = k - (n-1)
    # j ranges over: for valid1: max(0,-m) <= j <= min(n-1, n-1-m)
    #                for valid2: max(0,m) <= j <= min(n-1, n-1+m)
    
    for k in range(2 * n - 1):
        m = k - (n - 1)
        pk = p[k]
        if pk < 1e-15:
            continue
        
        # valid1: j in [max(0,-m), min(n-1, n-1-m)]
        j_lo1 = max(0, -m)
        j_hi1 = min(n - 1, n - 1 - m)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            grad[js] += pk * (1.0 - h[js + m])
        
        # valid2: j in [max(0,m), min(n-1, n-1+m)]
        j_lo2 = max(0, m)
        j_hi2 = min(n - 1, n - 1 + m)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            grad[js] -= pk * h[js - m]
    
    return dx * grad

def projected_gradient_descent(h, n, alpha, max_iter=5000, deadline=None, lr=None):
    """Projected gradient descent with barzilai-borwein step size."""
    h = h.copy()
    dx = 2.0 / n
    c0 = compute_c5(h, n)
    
    if lr is None:
        lr = 0.1 / alpha
    
    h_prev = h.copy()
    grad_prev = smooth_max_grad(h, n, alpha)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        h_new = project_feasible(h - lr * grad_prev, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new >= c0:
            # Try sub-Gibanov / backtracking
            for _ in range(10):
                lr *= 0.5
                h_new = project_feasible(h - lr * grad_prev, n)
                c_new = compute_c5(h_new, n)
                if c_new < c0:
                    break
            if c_new >= c0:
                break
        
        # Barzilai-Borwein step size update
        wh = h_new - h
        h = h_new
        h_prev_new = h_new
        grad_new = smooth_max_grad(h, n, alpha)
        
        # BB step
        dy = grad_new - grad_prev
        num = np.dot(wh, np.dot(wh, dy))
        den = np.dot(dy, dy)
        if den > 1e-15 and abs(num) > 1e-15:
            lr_new = num / den
            lr = 0.1 * lr_new + 0.9 * lr  # damped BB
            lr = np.clip(lr, 1e-6 / alpha, 1.0 / alpha)
        
        h_prev = h_prev_new
        grad_prev = grad_new
        c0 = c_new
    
    return h, c0

def coord_cd(h, n, max_passes=3000, deadline=None, grid_size=100):

    c = np.correlate(h, 1.0 - h, mode='full')

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        
        def obj_fn(x, n=n, alpha=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        
        try:
            res = minimize(obj_fn, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, n, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    es = np.exp(shifted - mx)
    p = es / es.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m_k = k - (n - 1)
        pk = p[k]
        if pk < 1e-10:
            continue
        j_lo1 = max(0, -m_k)
        j_hi1 = min(n - 1, n - 1 - m_k)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            grad[js] += pk * (1.0 - h[js + m_k])
        j_lo2 = max(0, m_k)
        j_hi2 = min(n - 1, n - 1 + m_k)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            grad[js] -= pk * h[js - m_k]
    return dx * grad

def proj_grad_descent(h, n, alpha, max_iter=5000, deadline=None):
    h = h.copy()
    c0 = compute_c5(h, n)
    lr = 0.5 / alpha
    grad_prev = smooth_max_grad(h, n, alpha)
    h_prev = h.copy()
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h_new = project_feasible(h - lr * grad_prev, n)
        c_new = compute_c5(h_new, n)
        if c_new >= c0:
            for _ in range(15):
                lr *= 0.5
                h_new = project_feasible(h - lr * grad_prev, n)
                c_new = compute_c5(h_new, n)
                if c_new < c0:
                    break
            if c_new >= c0:
                break
        wh = h_new - h
        h = h_new
        grad_new = smooth_max_grad(h, n, alpha)
        dy = grad_new - grad_prev
        num = np.dot(wh, np.dot(wh, dy))
        den = np.dot(dy, dy)
        if den > 1e-15 and abs(num) > 1e-15:
            lr_bb = num / den
            if lr_bb > 0:
                lr = 0.3 * lr_bb + 0.7 * lr
                lr = np.clip(lr, 1e-8, 2.0 / alpha)
        grad_prev = grad_new
        h_prev = h
        c0 = c_new
    return h, c0

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if p % 3 == 0:
            idx_order = range(n)
        elif p % 3 == 1:
            idx_order = range(n - 1, -1, -1)
        else:
            idx_order = sorted(range(n), key=lambda i: (i * 7919) % n)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-10:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])

                if abs(denom) > 1e-15:

                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:

            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c[ms] + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2

                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        def obj_fn(x, n=n, alpha=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(obj_fn, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
