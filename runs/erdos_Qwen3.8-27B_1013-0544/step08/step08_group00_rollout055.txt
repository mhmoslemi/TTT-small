def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_phase(h, n, alphas, iters):
    best = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        b = [(0.0, 1.0)] * n
        c = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
        try:
            r = minimize(lse_obj, best, args=(n, a), method='SLSQP',
                         bounds=b, constraints=c,
                         options={'maxiter': iters, 'ftol': 1e-16})
            nh = project_feasible(r.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best, best_c = nh, nc
        except Exception:
            pass
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_phase(h, n, alphas, iters, deadline=None):
    best = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if deadline is not None and time.time() > deadline - 10:
            break
        b = [(0.0, 1.0)] * n
        c = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
        try:
            r = minimize(lse_obj, best, args=(n, a), method='SLSQP',
                         bounds=b, constraints=c,
                         options={'maxiter': iters, 'ftol': 1e-15})
            nh = project_feasible(r.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best, best_c = nh, nc
        except Exception:
            pass
    return best, best_c

def fourier_perturb(h, n, rng, amp_scale=0.03):
    x = np.linspace(0, 2, n)
    n_modes = min(8, n // 4)
    freqs = rng.choice(range(1, n // 2 + 1), size=n_modes, replace=False)
    phases = rng.uniform(0, 2 * np.pi, n_modes)
    amps = amp_scale * rng.uniform(0.3, 1.0, n_modes)
    pert = np.zeros(n)
    for f, ph, am in zip(freqs, phases, amps):
        pert += am * np.cos(np.pi * f * x + ph)
    return project_feasible(h + pert, n)
