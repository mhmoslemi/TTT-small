import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    z = np.clip(z, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) < 1e-12:
        return z
    # Bisection for threshold
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = np.clip(z - mid, 0, 1).sum()
        if s > target: lo = mid
        else: hi = mid
    y = np.clip(z - 0.5*(lo+hi), 0, 1)
    res = target - y.sum()
    if res > 1e-12:
        mask = y < 1.0
        room = (1.0-y)[mask]
        if room.sum() > 1e-12:
            y[mask] += res * room / room.sum()
    elif res < -1e-12:
        mask = y > 0.0
        room = y[mask]
        if room.sum() > 1e-12:
            y[mask] += res * room / room.sum()
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5*n)

def sum_jac(x, n):
    return np.ones_like(x)

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1-x, mode='full')
    m = c.max()
    z = alpha*(c - m)
    zm = z.max()
    e = np.exp(z - zm)
    return (m + np.log(e.sum())/alpha) * (2.0/n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1-x, mode='full')
    m = c.max()
    z = alpha*(c-m)
    zm = z.max()
    e = np.exp(z-zm)
    p = e / e.sum()
    off = n-1
    g = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-14: continue
        d = k - off
        if d >= 0:
            g[d:] += p[k]*(1-x[:n-d])
            g[:n-d] -= p[k]*x[d:]
        else:
            e2 = -d
            g[:n-e2] += p[k]*(1-x[e2:])
            g[e2:] -= p[k]*x[:n-e2]
    return (2.0/n)*g

def slsqp_optimize(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]
    for alpha, mx in [(20,30),(100,30),(500,30),(2000,25),(10000,20),(50000,15),(200000,10)]:
        if time.time() > deadline - 2: break
        try:

                          jac=smooth_grad, bounds=bounds, constraints=cons,
                          options={'maxiter':mx,'ftol':1e-14,'disp':False})
            x = project_feasible(res.x, n)
            cc = compute_c5(x, n)
            if cc < best_c:

    return h, best_c

        k2 = n-1+j-i

        q = np.zeros(2*n-1)

        if 0 <= k2 < 2*n-1: q[k2] = 1.0

        hi = 1.0 - h[j] - 1e-14
        if hi - lo < 1e-10: continue

        c1 = b - gr*(b-a)
        c2 = a + gr*(b-a)
        f1 = (c + c1*s + c1*c1*q).max()
        f2 = (c + c2*s + c2*c2*q).max()
        for _ in range(45):

                c1 = b - gr*(b-a)
                f1 = (c + c1*s + c1*c1*q).max()
            else:

                c2 = a + gr*(b-a)
                f2 = (c + c2*s + c2*c2*q).max()

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0, 1)
    res = target - y.sum()
    if res > 1e-12:
        m = y < 1.0
        r = (1.0 - y)[m]
        if r.sum() > 1e-12:
            y[m] += res * r / r.sum()
    elif res < -1e-12:
        m = y > 0.0
        r = y[m]
        if r.sum() > 1e-12:

    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    z = alpha * (c - m)
    ez = np.exp(z - z.max())
    return (m + np.log(ez.sum()) / alpha) * (2.0 / n)

    m = c.max()

    off = n - 1
