```python
def sqp_solve(h, n, max_iter=50, active_set=None):
    for it in range(max_iter):
        c = np.correlate(h, 1-h, mode='full')
        if active_set is None:
            threshold = c.max() - 0.01 * c.max()
            active_set = set(np.where(c >= threshold)[0])
        
        # Compute gradients for active constraints
        grads = []
        for m in active_set:
            g = np.zeros(n)
            # c[m] = sum_i h[i]*(1-h[i+m-(n-1)]) for valid i
            offset = m - (n-1)
            for i in range(n):
                j = i + offset
                if 0 <= j < n:
                    g[i] += (1 - h[j])
                    g[j] -= h[i]
            grads.append(g)
        
        # Solve LP: min t s.t. c_m(h) + g_m^T (h-h) <= t for m in active_set
        # Simplified: min t s.t. g_m^T h <= t - c_m(h) + g_m^T h for m in active_set
        # Or: t >= c_m(h) + g_m^T (h_new - h) for each m
        
        # Using CVXPY
        import cvxpy as cp
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
        for m in active_set:
            c_val = c[m]
            g = grads[m]
            constraints.append(t_var >= c_val + g @ (h_var - h))
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)
        
        if h_var.value is None:
            break
        new_h = np.clip(h_var.value, 0, 1)
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        if new_c < compute_c5(h, n) - 1e-12:
            h = new_h
            # Update active set
            c = np.correlate(h, 1-h, mode='full')
            threshold = c.max() - 0.005 * c.max()
            active_set = set(np.where(c >= threshold)[0])
        else:
            break
    return h
```

**Let me try a different algorithmic approach: Frank-Wolfe (Conditional Gradient) Method**

The problem: min_h f(h) = max_k c_k(h) s.t. h ∈ P where P = {h: 0≤h≤1, sum h = n/2}

Frank-Wolfe:
1. Compute subgradient g of f at current h (gradient of active c_k)
2. Find s ∈ argmin_{s∈P} g^T s (this is a linear program over P, which is easy)
3. Update: h ← (1-γ)h + γs where γ is a step size

The LP in step 2: min g^T s s.t. 0≤s≤1, sum s = n/2
This is solved by: sort g, set s[i]=1 for the smallest g values, s[i]=0 for the largest, and one fractional value to satisfy the sum constraint.

This is O(n log n) per iteration! Very fast.

The issue is that f(h) = max_k c_k(h) is not differentiable, and Frank-Wolfe converges slowly (O(1/t)). But with the "away steps" variant or "shrinking" variant, it can be much faster.

**Combined algorithm:**
1. Use Frank-Wolfe for a few iterations (fast, gives a good descent direction)
2. Use L-BFGS-B with high alpha for refinement
3. Use coordinate descent for final polish
4. Repeat with perturbations

```python
def frank_wolfe_step(h, n, alpha=1.0):
    """One Frank-Wolfe step for min max_k c_k."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    # Compute gradient of c[k_star]
    g = np.zeros(n)
    offset = k_star - (n - 1)
    for i in range(n):
        j = i + offset
        if 0 <= j < n:
            g[i] += (1.0 - h[j])
            g[j] -= h[i]
    # Solve min g^T s s.t. 0<=s<=1, sum s = n/2
    # Solution: sort g, assign 1 to smallest, 0 to largest, one fractional
    order = np.argsort(g)
    s = np.zeros(n)
    target = n / 2.0
    remaining = target
    for idx in range(n):
        if remaining >= 1.0:
            s[order[idx]] = 1.0
            remaining -= 1.0
        elif remaining > 0:
            s[order[idx]] = remaining
            remaining = 0.0
        else:
            break
    # Update
    new_h = (1.0 - alpha) * h + alpha * s
    return project_feasible(new_h, n)

    """Frank-Wolfe optimization with decaying step size."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, iters + 1):
        if deadline is not None and time.time() > deadline:

            if c_trial < best_c_fw:
                best_c_fw = c_trial
                best_h_fw = h_trial
        h = best_h_fw
        if best_c_fw < best_c:
            best_c = best_c_fw
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    cl = c - c.max()
    lse = c.max() + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * cl)))
    return lse * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_quick(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, max_passes=200, deadline=None, rng=None, fine_pts=60):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full').copy()
        current_max = float(c_prev.max())
        improved = False
        order = rng.permutation(n) if rng is not None else range(n)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(max(0, n - 1 - p), p)
            m_hi = max(min(2 * n - 2, 2 * n - 2 - p), min(2 * n - 2, p + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            A = np.zeros(len(ms))
            if vf.any():
                A[vf] += (1.0 - h[jf[vf]])
            if vb.any():
                A[vb] -= h[jb[vb]]
            c_aff = c_prev[ms].copy()
            center = (ms == (n - 1))
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c_prev[m_hi + 1:].max()))
            best_d = 0.0
            best_val = current_max
            # Coarse
            d_coarse = np.linspace(-h0, 1.0 - h0, 25)
            for d in d_coarse:
                if abs(d) < 1e-14:
                    continue
                vals = c_aff + d * A
                if center.any():
                    vals = vals.copy()
                    vals[center] -= d * d
                val = max(unaff, float(vals.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            # Fine

        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def fw_step(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    offset = k_star - (n - 1)
    for i in range(n):
        j = i + offset
        if 0 <= j < n:
            g[i] += (1.0 - h[j])
            g[j] -= h[i]
    order_idx = np.argsort(g)
    s = np.zeros(n)
    remaining = n / 2.0
    for idx in range(n):
        if remaining >= 1.0:
            s[order_idx[idx]] = 1.0
            remaining -= 1.0
        elif remaining > 0:
            s[order_idx[idx]] = remaining
            remaining = 0.0
        else:
            break
    return s

def fw_optimize(h, n, iters=60, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, iters + 1):
        if deadline is not None and time.time() > deadline:
            break
        alpha = 2.0 / (t + 2.0)
        s = fw_step(h, n)
        new_h = project_feasible((1.0 - alpha) * h + alpha * s, n)
        h = new_h
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c
```
