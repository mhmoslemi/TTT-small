    c = np.correlate(h, 1.0 - h, mode='full')
    m = argmax(c)  # the maximizing shift
    center = n - 1
    g = np.zeros(n)
    s = center - m  # offset
    if s == 0:

        g[s:] -= h[:n-s]

        g[r:] += 1.0 - h[:n-r]
        g[:n-r] -= h[r:]
    return g, float(c.max() * (2.0/n))

    """Solve min g·x s.t. 0≤x≤1, sum(x)=n/2. Returns binary x."""
    k = n // 2

    x[idx] = 1.0
