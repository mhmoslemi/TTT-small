            m = y > 0.0
            r = y[m].sum()
            if r > 1e-14: y[m] += (res / r) * y[m]
    return np.clip(y, 0.0, 1.0)

def grad_w(x, n, p):
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-13: continue
        d = k - off
        m = n - abs(d)
        if d >= 0:
            g[:m] += w * (1.0 - x[d:d+m])
            g[d:d+m] -= w * x[:m]
        else:
            e = -d
            g[e:e+m] += w * (1.0 - x[:m])
            g[:m] -= w * x[e:e+m]
    return (2.0 / n) * g

def sm_w(c, alpha):
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - float(z.max()))
    return e / float(e.sum())

def smooth_opt(h, n, alphas, iters, deadline):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    v = np.zeros(n)
    for alpha in alphas:
        if time.time() > deadline - 3: break
        lr = 0.01
        for _ in range(iters):
            if time.time() > deadline - 3: break
            c = np.correlate(h, 1.0 - h, mode='full')
            p = sm_w(c, alpha)
            g = grad_w(h, n, p)
            v = 0.35 * v + 0.65 * g
            h = project_feasible(h - lr * v, n)
            cc = compute_c5(h, n)
            if cc < best_c:
                best_h, best_c = h.copy(), cc
    return best_h, best_c

def subgrad_opt(h, n, steps, deadline):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    off = n - 1
    for i in range(steps):
        if time.time() > deadline - 3: break
        c = np.correlate(h, 1.0 - h, mode='full')
        k = int(np.argmax(c))
        d = k - off
        m = n - abs(d)
        g = np.zeros(n)
        if d >= 0:
            g[:m] += 1.0 - h[d:d+m]
            g[d:d+m] -= h[:m]
        else:
            e = -d
            g[e:e+m] += 1.0 - h[:m]
            g[:m] -= h[e:e+m]
        g *= 2.0 / n
        lr = 0.006 / (1.0 + 0.2 * i / max(steps, 1))
        h = project_feasible(h - lr * g, n)
        cc = compute_c5(h, n)
        if cc < best_c:
            best_h, best_c = h.copy(), cc
    return best_h, best_c

def compute_B(h, n, i, j):
    off = n - 1
    B = np.zeros(2 * n - 1)
    for d in range(-i, n - i):
        k = off + d
        if 0 <= i + d < n: B[k] += (1.0 - h[i + d])
        if 0 <= i - d < n: B[k] -= h[i - d]
    for d in range(-j, n - j):
        k = off + d
        if 0 <= j + d < n: B[k] -= (1.0 - h[j + d])
        if 0 <= j - d < n: B[k] += h[j - d]
    return B

def pair_cd(h, n, tries, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max()) * 2.0 / n
    off = n - 1
    no_imp = 0
    for _ in range(tries):
        if time.time() > deadline - 3: break
        if no_imp > 200: break
        cur = float(c.max())
        tol = 0.004 * cur
        active = np.flatnonzero(c >= cur - tol)
        if len(active) == 0: active = np.array([off])
        kk = active[rng.randint(len(active))]
        d = kk - off
        e = abs(d)
        m = n - e
        if m <= 0: continue
        j = int(rng.randint(m))
        i = j + e
        B = compute_B(h, n, i, j)
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-14: continue
        a, b = lo, hi
        for _ in range(35):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if float(np.max(c + m1 * B)) < float(np.max(c + m2 * B)):
                b = m2
            else:
                a = m1
        u = 0.5 * (a + b)
        nm = float(np.max(c + u * B))
        if nm < cur - 1e-10 and abs(u) > 1e-13:
            h[i] += u
            h[j] -= u
            c = c + u * B
            no_imp = 0
            if nm * 2.0 / n < best_c:
                best_c = nm * 2.0 / n
        else:
            no_imp += 1
    return h, best_c

def optimize_instance(h, n, deadline, rng):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(1.0 - h, n)
    cs = compute_c5(hs, n)
    if cs < c: h, c = hs, cs
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c: h, c = hs, cs
    if time.time() < deadline - 60:
        alphas = [10, 30, 100, 300, 1000, 5000, 20000]
        iters = [40, 40, 30, 30, 25, 20, 15]
        h, c = smooth_opt(h, n, alphas, iters, deadline)
    if time.time() < deadline - 30:
        h, c = subgrad_opt(h, n, 300, deadline)
    if time.time() < deadline - 10:
        rem = deadline - time.time()
        tries = int(min(3000, max(300, rem * 30)))
        h, c = pair_cd(h, n, tries, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m: return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)): raise ValueError
    except Exception:
        h0 = random_start(500, rng)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    hc = 1.0 - best_h
    cc = compute_c5(hc, n0)
    if cc < best_c:
        best_h, best_c = hc, cc
    main_dl = hard - max(100.0, 0.2 * float(budget_s))
    if main_dl < t0 + 30: main_dl = hard
    h_opt, c_opt = optimize_instance(best_h, n0, main_dl, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt
    for m in [200, 300, 400, 600, 800]:
        if time.time() > hard - 30: break
        if m == n0: continue
        rem = hard - time.time()
        if rem < 50: break
        dl = time.time() + min(50.0, 0.25 * rem)
        ht, ct = optimize_instance(interp_h(best_h, m), m, dl, rng)
        if ct < best_c:
            best_h, best_c, n0 = ht, ct, m
    if time.time() < hard - 60:
        for s in [0.05, 0.1, 0.2]:
            if time.time() > hard - 30: break
            hp = project_feasible(best_h + rng.normal(0, s, len(best_h)), len(best_h))
            dl = min(hard - 10, time.time() + 40.0)
            ht, ct = optimize_instance(hp, len(best_h), dl, rng)
            if ct < best_c:
                best_h, best_c = ht, ct
    if time.time() < hard - 15:
        n = len(best_h)
        rem = hard - time.time()
        tries = int(min(2000, max(200, rem * 15)))
        best_h, best_c = pair_cd(best_h, n, tries, hard, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
