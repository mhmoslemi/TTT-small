            if tot < 1e-15:

            h += (diff / tot) * h

    if abs(leftover) > 1e-13:

        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += leftover / free.sum()
            h = np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, step_size, top_k=50):
    """One step of projected subgradient method on max of correlations."""
    c = compute_correl(h)
    cmax = c.max()
    # Find top-k lags
    idx = np.argsort(c)[::-1][:top_k]
    # Compute gradient for each top lag (weighted by proximity to max)
    weights = np.zeros(len(c))
    weights[idx] = (c[idx] - cmax + 1.0) / top_k  # simple weighting
    weights /= weights.sum()
    # Compute combined gradient

        w = weights[k_idx]

        # ∂c_r/∂h[j] = (1-h[j+r]) if j,j+r valid
        #             + (-h[j-r]) if j-r,j valid

        j_first = np.arange(i_lo, i_hi + 1)

        grad[j_first] += w * (1.0 - h[j_second])
        grad[j_second] += w * (-h[j_first])

    m = c.max()
    shifted = alpha * (c - m)
