        # This requires dC_k/dx_i for all k, i
        # Skip gradient, let SLSQP use numerical
        return None
    def sum_con(x):

        return np.ones(n)

        # For each zero position, compute A

Hmm, but this is still O(|ones| * |zeros| * n) per round, which for n=500 is about 250*250*1000 = 62.5M per round. That's too slow.

Let me limit the number of pairs we try. Instead of trying all pairs, we can:
1. Precompute A for all ones and zeros (O(n²) total)
2. For each pair (i,j), the update is c + A_j - A_i - 2D + corrections, and we just need to check if max(c_new) < max(c).

The key observation: max(c_new) >= max(c + A_j - A_i - 2D) (since the delta corrections are non-negative). So if max(c + A_j - A_i - 2D) >= max(c), the swap can't help (ignoring the small delta corrections).

Even better: we can precompute for each zero j, the vector B_j = A_j - 2D. Then for each pair (i,j), c_new ≈ c + B_j - A_i. We want max(c + B_j - A_i) < max(c).

This is still O(|ones| * |zeros| * n) if we naively compute the max for each pair.

For n=500, |ones| ≈ |zeros| ≈ 250, so 250*250*1000 = 62.5M per round. In numpy, if we can vectorize this... but it's hard to vectorize the max over the pair dimension.

Alternative: only try a subset of swaps per round. For example, for each one position i, try swapping with the "best" zero position (the one that most reduces the max). This is O(|ones| * |zeros| * n) per round but we can limit to, say, 10 swaps per round.

Actually, let me step back. The CD already pushes to binary. After CD converges, the solution is a local optimum for single-coordinate moves. To improve, we need multi-coordinate moves (swaps). But the number of possible swaps is O(n²), and each evaluation is O(n). So one full round of swap checking is O(n³) = 125M for n=500. In Python with numpy, this might take 1-5 seconds per round.

Let me try a more efficient approach: for the swap problem, note that:
c_new[k] = c[k] + A_j[k] - A_i[k] - 2*D[k] + corrections

The corrections are at most 2 (at specific positions). So approximately:
max(c_new) ≈ max_k (c[k] + A_j[k] - A_i[k] - 2*D[k])

We want this to be < max(c). This is equivalent to: for all k, c[k] + A_j[k] - A_i[k] - 2*D[k] < max(c).

Rearranging: A_j[k] - A_i[k] < max(c) - c[k] + 2*D[k] for all k.

This is a system of 2n-1 inequalities. Checking if it's satisfied takes O(n) per pair.

I think for practical purposes, the bit-swap approach with limited pairs per round is the way to go. Let me limit to, say, trying each one with the top 5 "best" zeros (based on some heuristic).

Actually, you know what, let me just keep it simple and efficient:

1. Run fast CD (binary-pushing) to convergence
2. Do a few rounds of random pair swaps (accept if improves)
3. Run CD again
4. Repeat

For the random swaps, each round tries O(n) random pairs instead of O(n²) all pairs. This is O(n²) per round (n pairs * O(n) evaluation each), which is fast.

Let me now write the final solution with all these ideas:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_A(h, p, n):
    """Compute the A vector for coordinate p."""
    n2 = 2*n-1
    m = np.arange(n2) - (n-1)
    A = np.zeros(n2)
    idx1 = p + m
    v1 = (idx1 >= 0) & (idx1 < n)
    A[v1] = (1.0 - h[idx1[v1]])
    idx2 = p - m
    v2 = (idx2 >= 0) & (idx2 < n)
    A[v2] -= h[idx2[v2]]
    return A

def binary_cd(h, n, max_passes=500, deadline=None, random_order=False, rng=None):
    """Fast coordinate descent that pushes to binary (exploiting concavity)."""
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    n2 = 2*n-1
    D = (np.arange(n2) == (n-1)).astype(float)
    m_all = np.arange(n2) - (n-1)
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        improved = False
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            
            A = compute_A(h, p, n)
            c0 = c - h0 * A + h0*h0 * D
            
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            current_max = float(np.max(c))
            
            if f0 < f1:
                if f0 < current_max - 1e-13:
                    t_new = 0.0
                    c += (t_new - h0) * A - (t_new*t_new - h0*h0) * D
                    h[p] = t_new
                    improved = True
            else:
                if f1 < current_max - 1e-13:
                    t_new = 1.0
                    c += (t_new - h0) * A - (t_new*t_new - h0*h0) * D
                    h[p] = t_new
                    improved = True
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full').astype(float)
        if not improved:
            break
    
    return h, compute_c5(h, n)

def random_swaps(h, n, max_rounds=20, deadline=None, rng=None):
    """Try random pair swaps to escape local optima."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full').astype(float)
    n2 = 2*n-1
    D = (np.arange(n2) == (n-1)).astype(float)
    m_all = np.arange(n2) - (n-1)
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        improved = False
        n_tries = min(len(ones) * len(zeros), 2000)
        for _ in range(n_tries):
            if deadline is not None and time.time() > deadline:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            A_i = compute_A(h, i, n)
            A_j = compute_A(h, j, n)
            
            c_new = c + A_j - A_i - 2.0 * D
            k1 = i - j + (n-1)
            if 0 <= k1 < n2:
                c_new[k1] += 1.0
            k2 = j - i + (n-1)
            if 0 <= k2 < n2:
                c_new[k2] += 1.0
            
            if float(np.max(c_new)) < float(np.max(c)) - 1e-13:
                c = c_new
                h[i] = 0.0
                h[j] = 1.0
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
        
        if not improved:
            break
    
    return h, compute_c5(h, n)

def interpolate_to_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slsqp_polish(h, n, p, maxiter=200, deadline=None):
    """SLSQP with p-norm objective."""
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        return float(np.sum(c**p) ** (1.0/p))
    def sum_con(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter, 'ftol':1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)
