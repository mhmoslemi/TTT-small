import numpy as np
from scipy.optimize import minimize
import time


def project_simplex(h, n):
    if n <= 0:
        return np.asarray(h, dtype=float).copy()
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return y


def compute_c5(h, n):
    if n <= 0:
        return 0.0
    return float(np.correlate(h, 1.0 - h, mode="full").max()) * (2.0 / float(n))


def make_binary(h, n):
    if n % 2:
        return None
    t = n // 2
    if t <= 0 or t >= n:
        b = np.zeros(n, float)
        b[:t] = 1.0
        return b
    idx = np.argpartition(h, t - 1)[:t]
    b = np.zeros(n, float)
    b[idx] = 1.0
    return b


def transfer_h(h, n, nt):
    if nt <= 0:
        return np.zeros(0, float)
    if nt == n:
        return project_simplex(h, nt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nt)
    return project_simplex(np.interp(y, x, h), nt)


def build_A(h, n):
    L = 2 * n - 1
    A = np.zeros((n, L), float)
    r = np.arange(n)
    c1 = r[:, None] + np.arange(n)[None, :]
    c2 = (n - 1 - r[:, None]) + np.arange(n)[None, :]
    om = 1.0 - h[::-1]
    A[r[:, None], c1] = om[None, :]
    A[r[:, None], c2] -= h[None, :]
    return A


def binary_full_improve(h, n, deadline):
    h = np.array(h, float)
    h[h > 0.5] = 1.0
    h[h < 0.5] = 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    best_h = h.copy()
    best_cm = cm
    diag = n - 1
    while time.time() < deadline:
        A = build_A(h, n)
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        subA = A[zeros]
        rows = np.arange(len(zeros))
        best_val = cm
        best_i = -1
        best_j = -1
        for i in ones:
            if time.time() > deadline:
                break
            cand = subA + (c - A[i])
            cand[:, diag] -= 2.0
            p1 = diag + i - zeros
            p2 = diag + zeros - i
            cand[rows, p1] += 1.0
            cand[rows, p2] += 1.0
            vals = cand.max(axis=1)
            k = int(np.argmin(vals))
            if vals[k] < best_val - 1e-8:
                best_val = float(vals[k])
                best_i = int(i)
                best_j = int(zeros[k])
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        if cm < best_cm - 1e-12:
            best_h = h.copy()
            best_cm = cm
    return best_h, best_cm * (2.0 / float(n))


def binary_run(h, n, deadline):
    b = make_binary(h, n)
    if b is None:
        return h, float(compute_c5(h, n))
    b = project_simplex(b, n)
    bh, bc = binary_full_improve(b, n, deadline)
    return bh, float(bc)


def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / float(n)
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx


def anti_h(v, n):
    m = n // 2
    h = np.empty(n, float)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h


def anti_v(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1::-1][:m])


def anti_smooth_val(v, n, alpha):
    h = anti_h(v, n)
    return smooth_val_grad(h, n, alpha)[0]


def anti_smooth_grad(v, n, alpha):
    h = anti_h(v, n)
    _, g = smooth_val_grad(h, n, alpha)
    m = n // 2
    mir = n - 1 - np.arange(m)
    return g[:m] - g[mir]


def anti_optimize(h0, n, alphas, deadline, rng=None):
    m = n // 2
    if m <= 0:
        return h0, float(compute_c5(h0, n))
    v0 = np.clip(anti_v(h0, n), 0.0, 1.0)
    best = anti_h(v0, n).copy()
    best_c = float(compute_c5(best, n))
    bnds = [(0.0, 1.0)] * m
    starts = [v0]
    if rng is not None:
        starts.append(np.clip(v0 + 0.25 * (rng.rand(m) - 0.5), 0.0, 1.0))
    for s in starts:
        vs = np.clip(np.asarray(s, float), 0.0, 1.0)
        for a in alphas:
            if time.time() > deadline:
                break
            try:
                res = minimize(
                    anti_smooth_val, vs, args=(n, a), jac=anti_smooth_grad,
                    method="L-BFGS-B", bounds=bnds,
                    options={"maxiter": 140, "ftol": 1e-13}
                )
            except Exception:
                continue
            xx = np.asarray(res.x, float)
            if np.all(np.isfinite(xx)):
                xx = np.clip(xx, 0.0, 1.0)
                hh = anti_h(xx, n)
                hh = np.clip(hh, 0.0, 1.0)
                cc = float(compute_c5(hh, n))
                if cc < best_c - 1e-12:
                    best = hh
                    best_c = cc
                    vs = np.clip(anti_v(hh, n), 0.0, 1.0)
    return best, best_c


def anti_axis_vals(c, Av, oldv, xs, diag, p1, p2):
    d = xs - oldv
    dd = d * d
    cand = c[None, :] + d[:, None] * Av[None, :]
    cand[:, diag] -= 2.0 * dd
    cand[:, p1] += dd
    cand[:, p2] += dd
    return cand.max(axis=1)


def anti_coord_cd(h0, n, max_passes, deadline, grid, refine):
    m = n // 2
    if m <= 0:
        return h0, float(compute_c5(h0, n))
    best = anti_h(np.clip(anti_v(h0, n), 0.0, 1.0), n).copy()
    c = np.correlate(best, 1.0 - best, mode="full")
    bc = float(c.max())
    L = 2 * n - 1
    diag = n - 1
    xs = np.linspace(0.0, 1.0, grid)
    Av = np.zeros(L, float)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h = best.copy()
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        om = 1.0 - h[::-1]
        improved = False
        for i in range(m):
            if time.time() > deadline:
                break
            mir = n - 1 - i
            old = float(h[i])
            Av.fill(0.0)
            Av[i:i + n] += om
            Av[diag - i:diag - i + n] -= h
            Av[mir:mir + n] -= om
            Av[i:i + n] += h
            p1 = diag + i - mir
            p2 = diag + mir - i
            vals = anti_axis_vals(c, Av, old, xs, diag, p1, p2)
            bi = int(np.argmin(vals))
            bu = float(xs[bi])
            bv = float(vals[bi])
            if bv < cm - 1e-12:
                sp = 1.0 / (grid - 1) if grid > 1 else 1.0
                lo = max(0.0, bu - sp)
                hi = min(1.0, bu + sp)
                if hi > lo:
                    xx = np.linspace(lo, hi, refine + 1)
                    mm = anti_axis_vals(c, Av, old, xx, diag, p1, p2)
                    bj = int(np.argmin(mm))
                    if mm[bj] < bv - 1e-12:
                        bu = float(xx[bj])
                        bv = float(mm[bj])
                if bv < cm - 1e-12:
                    d = bu - old
                    dd = d * d
                    c += d * Av
                    c[diag] -= 2.0 * dd
                    c[p1] += dd
                    c[p2] += dd
                    h[i] = bu
                    h[mir] = 1.0 - bu
                    om[diag - i] = 1.0 - bu
                    om[diag - mir] = bu
                    cm = float(c.max())
                    improved = True
        if improved:
            ch = float(c.max())
            if ch < bc - 1e-12:
                best = h.copy()
                bc = ch
            else:
                break
        else:
            break
    return best, bc * (2.0 / float(n))


def coord_eval(c, A, xs, old, diag):
    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, diag] -= d * d
    return cand.max(axis=1)


def coord_cd(h0, n, max_passes, deadline, grid, refine):
    best = project_simplex(h0, n)
    c = np.correlate(best, 1.0 - best, mode="full")
    bc = float(c.max())
    L = 2 * n - 1
    diag = n - 1
    xs = np.linspace(0.0, 1.0, grid)
    A = np.zeros(L, float)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        cm = float(c.max())
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            vals = coord_eval(c, A, xs, old, diag)
            bi = int(np.argmin(vals))
            bu = float(xs[bi])
            bv = float(vals[bi])
            if bv < cm - 1e-12:
                sp = 1.0 / (grid - 1) if grid > 1 else 1.0
                lo = max(0.0, bu - sp)
                hi = min(1.0, bu + sp)
                if hi > lo:
                    xx = np.linspace(lo, hi, refine + 1)
                    mm = coord_eval(c, A, xx, old, diag)
                    bj = int(np.argmin(mm))
                    if mm[bj] < bv - 1e-12:
                        bu = float(xx[bj])
                        bv = float(mm[bj])
                if bv < cm - 1e-12:
                    d = bu - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bu
                    om[diag - p] = 1.0 - bu
                    cm = float(c.max())
                    improved = True
        if not improved:
            break
        xp = project_simplex(x, n)
        cp = np.correlate(xp, 1.0 - xp, mode="full")
        cm2 = float(cp.max())
        if cm2 < bc - 1e-12:
            best = xp
            bc = cm2
        else:
            break
    return best, bc * (2.0 / float(n))
