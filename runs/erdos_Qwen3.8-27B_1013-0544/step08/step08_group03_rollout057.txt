    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_A_vec(h, n, p):
    A = np.zeros(2 * n - 1)
    n1 = n - 1
    k_lo = max(0, n1 - p)
    k_hi = min(2 * n - 2, 2 * n1 - p)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        js = p + ks - n1
        mask = (js >= 0) & (js < n) & (ks != n1)
        if np.any(mask):
            A[ks[mask]] += (1.0 - h[js[mask]])
    k_lo2 = p
    k_hi2 = min(2 * n - 2, p + n1)
    if k_lo2 <= k_hi2:
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        js2 = p - ks2 + n1
        mask2 = (js2 >= 0) & (js2 < n) & (ks2 != n1)
        if np.any(mask2):
            A[ks2[mask2]] += (-h[js2[mask2]])
    return A

def eval_f(c, A, h0, t, n):
    if t == h0:
        return float(c.max())
    vals = c + (t - h0) * A
    vals[n - 1] = c[n - 1] + (t - h0) - (t * t - h0 * h0)
    return float(np.max(vals))

def exact_cd(h, n, c, max_passes, deadline, grid_size=41):
    current_max = float(c.max())
    rng_local = np.random.RandomState(777)
    pass_count = 0
    for pass_num in range(max_passes):
        if deadline and time.time() > deadline:
            break
        order = rng_local.permutation(n)
        any_improve = False
        for p in order:
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            A = compute_A_vec(h, n, p)
            if np.abs(A).max() < 1e-15:
                continue
            # Build grid
            if pass_num < 3:
                grid = np.linspace(0.0, 1.0, grid_size)
            else:
                span = max(0.03, 3.0 / (pass_num + 2))
                lo = max(0.0, h0 - span)
                hi = min(1.0, h0 + span)
                grid = np.concatenate([np.array([0.0, 1.0]), np.linspace(lo, hi, grid_size - 2)])
            
            best_f = float('inf')
            best_t = h0
            for t in grid:
                if abs(t - h0) < 1e-15:
                    continue
                fval = eval_f(c, A, h0, t, n)
                if fval < best_f:
                    best_f = fval
                    best_t = t
            
            if best_f < current_max - 1e-14:
                c = c + (best_t - h0) * A
                c[n - 1] = c[n - 1] + (best_t - h0) - (best_t * best_t - h0 * h0)
                h[p] = best_t
                current_max = float(c.max())
                any_improve = True
        pass_count += 1
        if not any_improve:
            break
    return h, c, current_max, pass_count
