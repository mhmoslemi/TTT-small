def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n

        tot = float(cap.sum())

            h += (diff / tot) * cap
    else:

        if tot > 0:
            h += (diff / tot) * h
