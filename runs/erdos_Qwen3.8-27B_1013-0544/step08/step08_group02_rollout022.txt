        tau = 0.5 * (lo + hi)
        s = np.clip(z - tau, 0.0, 1.0).sum()

        else:
            hi = tau
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)

        room = y[mask]
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res/tot) * room

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return m + float(np.log(e.sum()))/alpha

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    # gradient of smooth max w.r.t. x
    # d(c[k])/dx[i] for each k, weighted by p[k]
    off = n - 1
    g = np.zeros(n)
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-15:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n-d])
            g[:n-d] -= w * x[d:]
        else:
            e2 = -d
            g[:n-e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n-e2]
    return (2.0/n) * g

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(smooth_val, work, args=(n, alpha), method='SLSQP',
                          jac=smooth_grad, bounds=bounds, constraints=cons,
                          options={'maxiter':mx, 'ftol':1e-14, 'disp':False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue

            c = compute_c5(x, n)
            if c < best_c:
                best_h, best_c = x, c
            work = x if c <= best_c+1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2*n-1))
    off = n-1
    for k in range(2*n-1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n-d]
            A[:n-d, k] -= h[d:]
        else:
            e2 = -d
            A[:n-e2, k] += 1.0 - h[e2:]
            A[e2:, k] -= h[:n-e2]
    return A
