    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + np.log(s) / alpha) * (2.0 / n), p, c

    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

    return smooth_components(x, n, alpha)[0]

    dev = x.sum() - 0.5 * n

    m = float(c.max())

    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + np.log(s) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):

    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:

    g += mu * dev

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e2 = -d
            A[:n - e2, k] += 1.0 - h[e2:]
            A[e2:, k] -= h[:n - e2]
    return A

    """Improved pair search with precomputed sensitivities."""
    if n < 2 or max_tries <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    accepts = 0
    no_improve = 0
    max_accept = min(200, max(60, max_tries // 4))
    
    for trial in range(max_tries):

        cur_max = float(c.max())

            j = int(rng.randint(n))
