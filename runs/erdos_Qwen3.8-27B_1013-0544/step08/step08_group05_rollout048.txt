    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(5.0, budget_s - 3.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = c5_val(h0, n0)
    n_best = n0
    # Symmetry averaging
    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + 1.0 - rev)):
        vv = project_simplex(v, n_best)
        cv = c5_val(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv
    # Phase 1: coordinate descent on main resolution
    t1 = min(deadline - 400.0, t0 + 180.0)
    if time.time() < t1:
        h1, c1 = coord_descent(best_h, n_best, max_passes=25, deadline=t1, grid=128)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: SLSQP polish
    t2 = min(deadline - 300.0, time.time() + 200.0)
    if time.time() < t2:
        h1, c1 = slsqp_polish(best_h, n_best, [5000, 50000, 500000], 150, t2)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 3: binary SA
    t3 = min(deadline - 200.0, time.time() + 250.0)
    if n_best % 2 == 0 and time.time() < t3:
        hb, cb = binary_sa(best_h, n_best, rng, t3, n_restarts=4)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
            # Polish after binary
            t3b = min(deadline - 100.0, time.time() + 80.0)
            if time.time() < t3b:
                h1, c1 = coord_descent(best_h, n_best, max_passes=8, deadline=t3b, grid=96)
                if c1 < best_c:
                    best_h, best_c = h1, c1
                h1, c1 = slsqp_polish(best_h, n_best, [10000, 100000], 100, t3b)
                if c1 < best_c:
                    best_h, best_c = h1, c1
    # Phase 4: try different resolution
    t4 = min(deadline - 120.0, time.time() + 120.0)
    for nt in [400, 350, 450, 600]:
        if time.time() > t4:
            break
        if nt % 2 or nt == n_best:
            continue
        ht = transfer_h(best_h, n_best, nt)
        cv = c5_val(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt
        chunk = min(deadline - 80.0, time.time() + 60.0)
        if time.time() < chunk:
            h1, c1 = coord_descent(ht, nt, max_passes=10, deadline=chunk, grid=96)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, nt
            h1, c1 = slsqp_polish(best_h, n_best, [20000], 80, chunk)
            if c1 < best_c:
                best_h, best_c = h1, c1
    # Phase 5: final polish
    t5 = deadline - 5.0
    if time.time() < t5:
        h1, c1 = coord_descent(best_h, n_best, max_passes=15, deadline=t5, grid=192)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 2.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline - 1.0, n_restarts=2)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
    final_h = project_simplex(best_h, n_best)
    final_c = c5_val(final_h, n_best)
    return final_h, float(final_c), int(n_best)
