import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    """Project h onto {h in [0,1], sum(h) = n/2}."""
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-15:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-15:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n)), c

def coord_optimize(hp, cseg, a, qconst, unaff, iters=20):
    """Find x in [0,1] minimizing max(unaff, qconst+x(1-x), max(cseg+a*x))."""
    tol = 1e-12
    m = a.size
    b = cseg - a * hp  # cseg_m + a_m*(x-hp) = b_m + a_m*x
    pos = a > 1e-10
    neg = a < -1e-10
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    qcur = qconst + hp * (1.0 - hp)
    
    # Bisection to find optimal t
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-10, unaff, qconst, qcur)
    if m:
        cm = float(np.max(cseg))
        if cm > high: high = cm
    high += 1e-8
    
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    
    for _ in range(iters):
        mid = 0.5 * (low + high)
        # Check feasibility: exists x in [0,1] with all constraints satisfied
        ll, lh = 0.0, 1.0
        if neg.any():
            v = float(np.max((mid - b_neg) / a_neg))
            if v > ll: ll = v
        if pos.any():
            v = float(np.min((mid - b_pos) / a_pos))
            if v < lh: lh = v
        feasible = False
        if ll <= 1.0 + tol and lh >= -tol:
            g = mid - qconst
            if g > 0.25:
                # x(1-x) <= g is always true for x in [0,1] when g >= 0.25
                feasible = (max(0.0, ll) <= min(1.0, lh) + tol)
            elif g >= -tol:
                if g < 0: g = 0.0
                disc = max(0.0, 1.0 - 4.0 * g)
                r1 = 0.5 - 0.5 * np.sqrt(disc)
                r2 = 0.5 + 0.5 * np.sqrt(disc)
                if max(0.0, ll) <= min(r1, lh) + tol:
                    feasible = True
                elif max(r2, ll) <= min(1.0, lh) + tol:
                    feasible = True
        if feasible:
            high = mid
        else:
            low = mid
    
    t = high
    # Find best x at optimal t
    best_x, best_f = hp, 0.0
    # Evaluate at candidate points
    cands = [0.0, 1.0, hp, 0.5]
    if m:

        for mi in range(min(m, 200)):
            if abs(a[mi]) > 1e-14:

                if -0.01 <= xv <= 1.01:
                    cands.append(xv)
    cands = np.clip(np.array(cands), 0.0, 1.0)
    
    def eval_at(x):
        val = max(unaff, qconst + x * (1.0 - x))
        if m:
            val = max(val, float(np.max(cseg + a * (x - hp))))
        return val
    
    best_f = float('inf')
    for x in cands:
        f = eval_at(float(x))
        if f < best_f:
            best_f = f
            best_x = float(x)
    return best_x, best_f

def cd_optimize(h, n, max_passes, deadline, project_every=10):
    """Coordinate descent with exact piecewise-linear optimization."""
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = coord_optimize(hp, cseg, asg, qconst, unaff, 20)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())

        return h
    if diff > 0:

        tot = float(cap.sum())

            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-15:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-13:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size: h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size: h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def coord_opt(hp, cseg, a, qconst, unaff):
    tol = 1e-12
    m = a.size
    b = cseg - a * hp if m else np.array([])
    pos = a > 1e-10 if m else np.array([], dtype=bool)
    neg = a < -1e-10 if m else np.array([], dtype=bool)
    zero = ~(pos | neg) if m else np.array([], dtype=bool)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-10, unaff, qcur)
    if m:
        cm = float(np.max(cseg))
        if cm > high: high = cm
    high += 1e-8
    a_p, a_n = a[pos], a[neg]
    b_p, b_n = b[pos], b[neg]
    for _ in range(24):
        mid = 0.5 * (low + high)
        ll, lh = 0.0, 1.0
        if a_n.size:
            v = float(np.max((mid - b_n) / a_n))
            if v > ll: ll = v
        if a_p.size:
            v = float(np.min((mid - b_p) / a_p))
            if v < lh: lh = v
        feas = False
        if ll <= 1.0 + tol and lh >= -tol:
            g = mid - qconst
            if g > 0.25:
                feas = max(0.0, ll) <= min(1.0, lh) + tol
            elif g >= -tol:
                g = max(0.0, g)
                disc = max(0.0, 1.0 - 4.0 * g)
                r1 = 0.5 - 0.5 * np.sqrt(disc)
                r2 = 0.5 + 0.5 * np.sqrt(disc)
                if max(0.0, ll) <= min(r1, lh) + tol: feas = True
                elif max(r2, ll) <= min(1.0, lh) + tol: feas = True
        if feas: high = mid
        else: low = mid
    t = high
    cands = np.array([0.0, 1.0, hp, 0.5, hp + 0.1, hp - 0.1])
    if 0 < m <= 400:
        valid = np.abs(a) > 1e-10
        if valid.any():
            bp = (t - b[valid]) / a[valid]
            bp = bp[(bp >= -0.01) & (bp <= 1.01)]
            cands = np.concatenate([cands, bp[:60]])
    cands = np.clip(cands, 0.0, 1.0)
    best_x, best_f = hp, float('inf')
    for x in cands:
        x = float(x)
        val = max(unaff, qconst + x * (1.0 - x))
        if m:
            val = max(val, float(np.max(cseg + a * (x - hp))))
        if val < best_f:
            best_f, best_x = val, x
    return best_x, best_f

def cd_opt(h, n, max_passes, deadline, proj_every=10):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j_arr = p + s
            i_arr = p - s
            a = np.zeros(seg, dtype=float)
            vj = (j_arr >= 0) & (j_arr < n)
            vi = (i_arr >= 0) & (i_arr < n)
            if np.any(vj): a[vj] += 1.0 - h[j_arr[vj]]
            if np.any(vi): a[vi] -= h[i_arr[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L: unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = coord_opt(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off: c[lo:center] += asg[:off] * dxv
                if hi > center: c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur, improved = bf, True
            if (p + 1) % proj_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c, best_h = cval, h.copy()
        if not improved: break
    return best_h, best_c

def subgrad_polish(h, n, deadline, steps=3000):
    h = project_feasible(h, n)
    best_c = compute_c(h, n)
    best_h = h.copy()
    center = n - 1
    for step in range(steps):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        s_star = m_star - center
        g = np.zeros(n)
        js = np.arange(n) + s_star
        jm = np.arange(n) - s_star
        v1 = (js >= 0) & (js < n)
        v2 = (jm >= 0) & (jm < n)
        if v1.any(): g[v1] += (1.0 - h[js[v1]])
        if v2.any(): g[v2] -= h[jm[v2]]
        alpha = 0.15 / (1.0 + 0.02 * step)
        h = project_feasible(h - alpha * g, n)
        cv = compute_c(h, n)
        if cv < best_c:
            best_c, best_h = cv, h.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    zv = alpha * c
    zm = float(zv.max())
    return ((zm + np.log(np.sum(np.exp(zv - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    zv = alpha * c
    zv = zv - zv.max()
    w = np.exp(zv)
    w /= w.sum()
    g = np.zeros(len(x))
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14: continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0 and n > s:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        elif s < 0 and n > -s:
            g[-s:] += wm * (1.0 - x[:-s])
            g[:-s] -= wm * x[-s:]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, deadline, alphas=(30, 300, 3000), maxiter=50):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline: break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          jac=smooth_jac, bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            x = project_feasible(res.x, n)
            cv = compute_c(x, n)
            if cv < best_c:
                best_h, best_c = x, cv
        except Exception:
            pass
    return best_h, best_c

def binary_init(h, n):
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        x = (np.arange(500) + 0.5) * (2.0 / 500)
        h0 = project_feasible(1.0 - np.abs(x - 1.0), 500)
    n0 = len(h0)
    if n0 % 2:
        n0 += 1
        h0 = resample_h(h0, n0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c(best_h, n0)
    # Phase 1: SLSQP warmstart
    if n0 <= 700 and time.time() < deadline - 200:
        h1, c1 = slsqp_opt(best_h, n0, deadline - 150, alphas=(30, 300, 3000), maxiter=40)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Main CD
    h1, c1 = cd_opt(best_h, n0, 30, deadline - 100, proj_every=8)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 3: Subgradient polish
    if time.time() < deadline - 60:
        h1, c1 = subgrad_polish(best_h, n0, deadline - 40, steps=2000)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 4: Multi-resolution refinement
    for nt in [n0, 400, 600, 300]:
        if time.time() > deadline - 120: break
        if nt == n0: continue
        h_r = resample_h(best_h, nt)
        h1, c1 = cd_opt(h_r, nt, 15, deadline - 80, proj_every=10)
        if c1 < best_c:
            best_h, best_c = h1, c1
            if time.time() < deadline - 60:
                h1, c1 = subgrad_polish(best_h, nt, deadline - 40, steps=1000)
                if c1 < best_c:
                    best_h, best_c = h1, c1
    # Phase 5: Perturbation restarts
    for r in range(12):
        if time.time() > deadline - 80: break
        if best_c <= 0.38070: break
        nt = n0 if r % 2 == 0 else 400
        base = resample_h(best_h, nt)
        if r % 4 == 0:
            h_p = project_feasible(base + rng.normal(0, 0.03, nt), nt)
        elif r % 4 == 1:
            h_p = binary_init(base, nt)
        elif r % 4 == 2:
            h_p = project_feasible(np.clip(base + 0.1 * rng.randn(nt), 0, 1), nt)
        else:
            h_p = base.copy()
            h_p[rng.randint(0, nt, 20)] = rng.uniform(0, 1, 20)
            h_p = project_feasible(h_p, nt)
        h1, c1 = cd_opt(h_p, nt, 10, deadline - 50, proj_every=12)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 6: Final polish
    if time.time() < deadline - 20:
        final_n = len(best_h)
        h1, c1 = cd_opt(best_h, final_n, 20, deadline - 5, proj_every=5)
        if c1 < best_c:
            best_h, best_c = h1, c1
        if time.time() < deadline - 10:
            h1, c1 = subgrad_polish(best_h, final_n, deadline - 2, steps=1000)
            if c1 < best_c:
                best_h, best_c = h1, c1
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
