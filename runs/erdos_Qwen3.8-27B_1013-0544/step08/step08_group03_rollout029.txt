def compute_c5(h, n):

def project_feasible(h, n):
    """Project h onto {x: 0≤x[i]≤1, sum(x)=n/2} using bisection on water level."""
    h = h.copy()
    target = n / 2.0
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Now adjust to hit target sum using water-filling
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Binary search for lambda such that sum(clip(h - lambda, 0, 1)) = target
    # If s > target, we need to decrease (lambda > 0)
    # If s < target, we need to increase (lambda < 0)
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0.0, 1.0)
        s_test = h_test.sum()
        if s_test > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Final check
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Adjust the most "interior" elements
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

    logsum = Tc_max / T + np.log(np.sum(np.exp(Tc - Tc_max)))
    obj = logsum * dx
    
    # Weights: w_s = T * exp(T*c_s) / sum(exp(T*c_s))
    exp_Tc = np.exp(Tc - Tc_max)

        if hi2 > lo2:

    grad *= dx
    return obj, grad

    """Projected gradient descent with backtracking."""
    h = h.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        obj, grad = smooth_c5_and_grad(h, n, T)
        
        # Check actual C5
        actual = compute_c5(h, n)
        if actual < best_obj - 1e-10:
            best_obj = actual
            best_h = h.copy()
        
        # Backtracking line search
        alpha = 0.01
        for _ in range(50):
            h_new = project_feasible(h - alpha * grad, n)
            obj_new, _ = smooth_c5_and_grad(h_new, n, T)
            if obj_new < obj - 1e-4 * alpha * np.dot(grad, h_new - h):
                break
            alpha *= 0.5
        
        h = h_new
        
        # Check convergence
        if it > 0 and abs(obj_prev - obj) < 1e-12:

        obj_prev = obj
    
    # Final check
    actual = compute_c5(h, n)
    if actual < best_obj:
        best_h, best_obj = h, actual
    return best_h, best_obj

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Bisection for water level
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h_test = np.clip(h - mid, 0.0, 1.0)
        if h_test.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(h - lam, 0.0, 1.0)
    # Fine adjustment
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * c
    Tc = Tc - Tc.max()
    exp_Tc = np.exp(Tc)
    sum_exp = exp_Tc.sum()
    obj = (Tc.max() / T + np.log(sum_exp)) * dx
    
    w = T * exp_Tc / sum_exp  # softmax weights times T
    
    # Gradient: for each p, grad[p] = dx * sum_k w[k] * dc_k/dh[p]
    # dc_k/dh[p] where k corresponds to shift s=k-(n-1):
    #   = (1-h[p+s])*[0<=p+s<n] - h[p-s]*[0<=p-s<n] (for s≠0)
    #   = 1 - 2*h[p] (for s=0, since it's h[p](1-h[p]))
    #     Actually for s=0: d(h[p]*(1-h[p]))/dh[p] = 1-2*h[p]
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-18:
            continue
        s = shifts[k]
        if s == 0:
            # Only affects h[p] for all p with 1-2h[p]
            # Wait no, for s=0: c_{n-1} = sum_i h[i](1-h[i])
            # dc/dh[p] = 1-2h[p]
            grad += wk * (1.0 - 2.0 * h)
        else:

            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if hi1 > lo1:

            # Term 2: -h[p-s] for p in [max(0,s), min(n, n+s))
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if hi2 > lo2:
                grad[lo2:hi2] -= wk * h[lo2-s:hi2-s]
    
    return obj, grad * dx

def coordinate_descent(h, n, max_passes=200, deadline=None):
    """Coordinate descent with exact 1D minimization."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = c.max()
        improved = False

            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            
            # Build A and D vectors for c_k_new = c_k + (t-h0)*A[k] - (t^2-h0^2)*D[k]
            A = np.zeros(2*n-1)
            D = np.zeros(2*n-1)

            fd = float(np.max(c + (d_pt-h0)*A - (d_pt**2-h0**2)*D))

                else:

                    d_pt = a + gr*(b-a)
                    fd = float(np.max(c + (d_pt-h0)*A - (d_pt**2-h0**2)*D))
            
            t_star = (a + b) * 0.5
            f_star = float(np.max(c + (t_star-h0)*A - (t_star**2-h0**2)*D))
            
            # Also check endpoints
            f_0 = float(np.max(c - h0*A - h0**2*D))  # t=0
            f_1 = float(np.max(c + (1-h0)*A - (1-h0**2)*D))  # t=1
            
            candidates = [(f_star, t_star), (f_0, 0.0), (f_1, 1.0)]
            f_best, t_best = min(candidates, key=lambda x: x[0])
            
            if f_best < current_max - 1e-14:
                h[p] = t_best
                c += (t_best - h0) * A - (t_best**2 - h0**2) * D
                current_max = float(c.max())
                improved = True
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:

        res = minimize(obj, x0, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50})
        if deadline is not None and time.time() > deadline:
            return project_feasible(x0, n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(x0, n)

def fista_optimize(h, n, T, max_iters=500, deadline=None):
    h = h.copy()
    z = h.copy()
    t_mom = 1.0
    L = max(5.0, 3.0 * T / n * 10)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(z, 1.0 - z, mode='full')
        Tc = T * c - T * c.max()
        exp_Tc = np.exp(Tc)
        w = T * exp_Tc / exp_Tc.sum()
        g = np.zeros(n)
        shifts = np.arange(2 * n - 1) - (n - 1)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-18:
                continue
            s = shifts[k]
            if s == 0:
                g += wk * (1.0 - 2.0 * z)
            else:
                lo1 = max(0, -s)
                hi1 = min(n, n - s)
                if hi1 > lo1:
                    g[lo1:hi1] += wk * (1.0 - z[lo1 + s:hi1 + s])
                lo2 = max(0, s)
                hi2 = min(n, n + s)
                if hi2 > lo2:
                    g[lo2:hi2] -= wk * z[lo2 - s:hi2 - s]
        g *= dx
        h_new = project_feasible(z - g / L, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        z = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        z = project_feasible(z, n)
        h = h_new
        t_mom = t_new
    return h
