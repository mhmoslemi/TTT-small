def project_simplex(h, n):

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s+n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                smooth_obj, best, args=(n, a), jac=smooth_grad,
                method="SLSQP", bounds=bounds, constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-15}
            )
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc

    b = np.zeros(n, dtype=float)

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-13) & (y < 1.0 - 1e-13)
        if mask.any():

            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s+n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-15})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc

def coord_cd(h, n, max_passes, deadline, grid=48, refine=20):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    tol = 1e-14
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A = np.zeros(L)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = float(c.max())
            d = xs0 - old
            cand = c[None, :] + d[:, None] * A[None, :]
            cand[:, n-1] -= d * d
            vals = cand.max(axis=1)
            i = int(np.argmin(vals))
            bx, bv = float(xs0[i]), float(vals[i])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                d2 = xx - old
                cand2 = c[None, :] + d2[:, None] * A[None, :]
                cand2[:, n-1] -= d2 * d2
                mm = cand2.max(axis=1)
                j = int(np.argmin(mm))
                if mm[j] < bv - tol:
                    bx, bv = float(xx[j]), float(mm[j])
                sp2 = (hi - lo) / refine
                lo2 = max(0.0, bx - sp2)
                hi2 = min(1.0, bx + sp2)
                yy = np.linspace(lo2, hi2, 13)
                d3 = yy - old
                cand3 = c[None, :] + d3[:, None] * A[None, :]
                cand3[:, n-1] -= d3 * d3
                ww = cand3.max(axis=1)
                k = int(np.argmin(ww))
                if ww[k] < bv - tol:
                    bx, bv = float(yy[k]), float(ww[k])
                d = bx - old
                c[p:p+n] += d * om
                c[st:st+n] -= d * x
                c[diag] -= d * d
                x[p] = bx
                om[diag - p] = 1.0 - bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def rand_10_swap(h, n, rng):
    i = int(rng.randint(n))
    for _ in range(8):
        i = int(rng.randint(n))
        if h[i] >= 0.5:
            break
    j = int(rng.randint(n))
    for _ in range(8):
        j = int(rng.randint(n))
        if h[j] < 0.5:
            break
    if i == j:
        i, j = int(np.argmax(h)), int(np.argmin(h))
    return i, j

def binary_trial_fast(c, h, om, i, j, n):
    diag = n - 1
    d = c.copy()
    d[i:i+n] -= om
    d[diag - i:diag - i + n] += h
    d[j:j+n] += om
    d[diag - j:diag - j + n] -= h
    d[diag] -= 2.0
    d[diag + i - j] += 1.0
    d[diag + j - i] += 1.0
    return d, float(d.max())

def binary_optimize(h_init, n, rng, deadline):
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = b.copy()

    bcm = float(bc_raw.max()) * dx
    stagnant = 0
    while time.time() < deadline and stagnant < 6:
        cur = best.copy()
        cur_c = bc_raw.copy()
        cur_m = float(cur_c.max())
        om = 1.0 - cur[::-1]
        plateau = 0
        trials = 0
        phase_improved = False
        while time.time() < deadline and plateau < 10000 and trials < 80000:
            i, j = rand_10_swap(cur, n, rng)
            if i == j:
                continue
            c2, nm = binary_trial_fast(cur_c, cur, om, i, j, n)
            trials += 1
            if nm < cur_m - 1e-10:
                cur_c, cur_m = c2, nm
                cur[i], cur[j] = 0.0, 1.0
                om[n-1-i], om[n-1-j] = 1.0, 0.0
                plateau = 0
                if nm * dx < bcm:
                    best, bc_raw, bcm = cur.copy(), cur_c.copy(), nm * dx
                    phase_improved = True
            else:
                plateau += 1
        stagnant = 0 if phase_improved else stagnant + 1
    return best, bcm

def transfer_h(h, n, n_tgt):
