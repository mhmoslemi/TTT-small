def exact_cd(h, n, max_passes=300, deadline=None, gs_iters=45, grid_res=60):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Compute a vector (length 2n-1)
            a = np.zeros(2*n-1)
            s = np.arange(-(n-1), n, dtype=np.int64)
            # i=p: valid when 0<=p+s<n
            valid_i = (p + s >= 0) & (p + s < n)
            a[valid_i] += (1.0 - h[p + s[valid_i]])
            # j=p: valid when 0<=p-s<n and s≠0
            valid_j = (p - s >= 0) & (p - s < n) & (s != 0)
            a[valid_j] -= h[p - s[valid_j]]
            # s=0 (k=n-1)
            a[n-1] += 1.0
            d = np.zeros(2*n-1)
            d[n-1] = -1.0
            
            c0 = c - h0 * a + h0 * h0 * d
            # Minimize f(t) = max(c0 + t*a + t^2*d) over [0,1]
            # (note: -t^2*d where d[n-1]=-1 gives +t^2 at n-1)
            
            # Grid + golden section
            grid = np.linspace(0.0, 1.0, grid_res)
            best_t = 0.5
            best_f = float('inf')
            for t in grid:
                f = float(np.max(c0 + t * a + t * t * d))
                if f < best_f:
                    best_f = f
                    best_t = t
            # Refine around best_t
            span = grid[1] - grid[0]

            hi = min(1.0, best_t + span)

                    else:
                        lo = c1; c1 = d1; fc1 = fd1
                        d1 = lo + gr * (hi - lo)

                best_t = (lo + hi) * 0.5
                best_f = float(np.max(c0 + best_t * a + best_t * best_t * d))

            f1 = float(np.max(c0 + a + d))
