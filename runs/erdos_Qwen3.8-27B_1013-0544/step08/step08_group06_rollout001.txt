            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def coord_cd(h, n, max_passes, deadline, grid_size, free_only=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        direction = 1 if p % 3 == 0 else -1
        indices = range(n) if direction == 1 else range(n - 1, -1, -1)
        for idx in indices:
            if time.time() > deadline:
                break
            h0 = h[idx]
            if free_only:
                if h0 < 0.01 or h0 > 0.99:
                    continue
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                if not free_only:
                    continue
                lo_d, hi_d = -h0, 1.0 - h0
                if hi_d - lo_d < 1e-13:
                    continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-13:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                t1 = float(c[:m_lo].max())
                if t1 > unaff_max:
                    unaff_max = t1
            if m_hi < 2 * n - 2:
                t2 = float(c[m_hi + 1:].max())
                if t2 > unaff_max:
                    unaff_max = t2
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-11 and best_val < cur_max - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < cur_max:
                    cur_max = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 400 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_phase(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() < deadline:
            new_h = project_feasible(res.x, n)
            return new_h
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c, _ = compute_c5(h, n)

    # Phase 1: SLSQP warm start
    if time.time() < deadline - 600:
        for alpha in [1000, 10000]:
            if time.time() > deadline - 600:
                break
            h_new = slsqp_phase(best_h, n, alpha, 200, deadline - 600)
            c_new, _ = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 2: Main coordinate descent
    dl2 = deadline - 300
    h2, c2 = coord_cd(best_h, n, 3000, dl2, grid_size=100)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Free-variable focus with finer grid
    dl3 = deadline - 150
    h3, c3 = coord_cd(best_h, n, 2000, dl3, grid_size=200, free_only=True)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 600]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        c_init, _ = compute_c5(h_int, n_try)
        h_try, c_try = coord_cd(h_int, n_try, 1500, deadline - 80, grid_size=80)
        if c_try < best_c:
            best_h, best_c = h_try, c_try
            x_orig = x_new

    # Phase 5: SLSQP on best
    if time.time() < deadline - 60:
        h_s = slsqp_phase(best_h, len(best_h), 20000, 150, deadline - 60)
        c_s, _ = compute_c5(h_s, len(h_s))
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # Phase 6: Final fine polish
    if time.time() < deadline - 5:
        h_f, c_f = coord_cd(best_h, len(best_h), 3000, deadline - 3, grid_size=250, free_only=True)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c, _ = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
