import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    elif diff < 0:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        idx = np.where(h < 1.0 if res > 0 else h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def _solve_coord(hp, cseg, asg, qconst, unaff):
    m = len(asg)
    if m == 0:
        f = lambda x: max(unaff, qconst + x * (1.0 - x))
    else:
        f = lambda x: max(unaff, qconst + x * (1.0 - x), float(np.max(cseg + asg * (x - hp))))
    cur = f(hp)
    if cur <= unaff + 1e-15:
        return hp
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    x = 0.5 * (lo + hi)
    cands = [x, lo, hi, 0.0, 1.0, hp, 0.5]
    best_x, best_f = hp, cur
    for cx in cands:

        if cf < best_f:

    lo2, hi2 = best_x - 0.02, best_x + 0.02

            else:
                lo2 = m1
        x2 = 0.5 * (lo2 + hi2)
        if f(x2) < best_f:
            best_x = x2
    return best_x

def cd_pass(h, n, order, deadline, proj_every=8):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    for p in order:
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg, dtype=float)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1, dtype=float)
        asg = np.empty(seg - 1, dtype=float)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi + 1:].max()))
        bx = _solve_coord(hp, cseg, asg, qconst, unaff)
        if bx != hp:
            dxv = bx - hp
            if off:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center + 1:hi + 1] += asg[off:] * dxv
            c[center] = qconst + bx * (1.0 - bx)
            h[p] = bx
        if (p + 1) % proj_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(c) * (2.0 / n))

        c = np.correlate(h, 1.0 - h, mode='full')

        if s == 0:

            g[:n - r] -= h[r:]
        h = project_feasible(h - lr * g / np.max(np.abs(g) + 1e-15), n)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c - alpha * c.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += w[m] * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += w[m] * (1.0 - x[s:])
            g[s:] -= w[m] * x[:n - s]
        else:
            r = -s
            g[r:] += w[m] * (1.0 - x[:n - r])
            g[:n - r] -= w[m] * x[r:]
    return (2.0 / n) * g
