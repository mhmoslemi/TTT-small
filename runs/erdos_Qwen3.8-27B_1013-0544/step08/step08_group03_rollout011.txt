    c = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    Tc = T * c - T * c.max()
    w = np.exp(Tc)
    w /= w.sum()
    # g[p] = sum_k w[k] * ((1-h[p-k]) - h[p+k]) with validity
    g = np.zeros(n)
    for k_idx in range(len(w)):

        #   Let me compute: j goes from j_lo to j_hi
        #   p - j goes from p - j_lo down to p - j_hi
        #   So h[p-j] = h[p-j_hi : p-j_lo+1] (in reverse order of j)
        #   Corresponding k_full = j + n - 1, goes from j_lo+n-1 to j_hi+n-1
        
        j_lo1 = max(-(n-1), p - n + 1)
        j_hi1 = min(n - 1, p)
        if j_lo1 < j_hi1:
            # j from j_lo1 to j_hi1, excluding j=0
            # k_full = j + n - 1, from j_lo1+n-1 to j_hi1+n-1
            # h[p-j]: when j=j_lo1, p-j = p-j_lo1; when j=j_hi1, p-j = p-j_hi1
            # p-j_lo1 >= 0 (by construction), p-j_hi1 <= p (since j_hi1 >= 0 or could be negative)
            h_slice = h[p - j_hi1 : p - j_lo1 + 1][::-1]  # h[p-j] for j from j_lo1 to j_hi1
            kf_slice = slice(j_lo1 + n - 1, j_hi1 + n)
            # Mask out j=0
            mask = np.ones(j_hi1 - j_lo1 + 1, dtype=bool)
            if 0 >= j_lo1 and 0 <= j_hi1:
                mask[0 - j_lo1] = False
            A[kf_slice] += (1.0 - h_slice) * mask
        # Second part
        j_lo2 = max(-(n-1), -p)
        j_hi2 = min(n - 1, n - p - 1)
        if j_lo2 < j_hi2:
            h_slice2 = h[p + j_lo2 : p + j_hi2 + 1]
            kf_slice2 = slice(j_lo2 + n - 1, j_hi2 + n)
            mask2 = np.ones(j_hi2 - j_lo2 + 1, dtype=bool)
            if 0 >= j_lo2 and 0 <= j_hi2:
                mask2[0 - j_lo2] = False
            A[kf_slice2] -= h_slice2 * mask2
        
        # Set center
        A[n-1] = 1.0
        
        # Now find optimal t in [0,1] minimizing max(c0 + t*A - t²*D)
        # where D = e_{n-1}
        D = np.zeros(2*n-1)
        D[n-1] = 1.0
        c0 = c - h0 * A + h0 * h0 * D
        
        # Minimize f(t) = max(c0 + t*A - t²*D) over t in [0,1]
        # This is a convex function of t (max of linear/quadratic functions)
        # Use golden section search
        
        # Actually, f(t) = max(c0 + t*A - t²*D)
        # = max(c0 + t*A) - t² (since D only has one nonzero at n-1)
        # = max_k (c0[k] + t*A[k]) - t²
        
        # Hmm, that's not quite right. f(t) = max_k (c0[k] + t*A[k] - t²*D[k])
        # For k ≠ n-1: c0[k] + t*A[k] (linear in t)
        # For k = n-1: c0[n-1] + t*A[n-1] - t² = c0[n-1] + t - t² (concave quadratic)
        
        # So f(t) = max(max_{k≠n-1}(c0[k] + t*A[k]), c0[n-1] + t - t²)
        
        # This is the max of lines and a concave parabola, which is convex.
        # We can find the minimum using golden section search.
        
        # For efficiency, precompute:
        # c0_no_center = c0 with center removed
        # f(t) = max(max(c0_no_center + t*A_no_center), c0[n-1] + t - t²)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, 1.0
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        
        def f_of_t(t):
            return float(np.max(c0 + t * A - t * t * D))
        
        fc1 = f_of_t(c1)
        fd1 = f_of_t(d1)
        for _ in range(40):
            if fc1 < fd1:
                b = d1; d1 = c1; fd1 = fc1
                c1 = b - gr * (b - a)
                fc1 = f_of_t(c1)
            else:
                a = c1; c1 = d1; fc1 = fd1
                d1 = a + gr * (b - a)
                fd1 = f_of_t(d1)
        t_star = (a + b) * 0.5
        
        # Also check endpoints
        f0 = f_of_t(0.0)
        f1 = f_of_t(1.0)
        ft = f_of_t(t_star)
        
        if f0 <= ft and f0 <= f1:
            t_new = 0.0
        elif f1 <= ft and f1 <= f0:
            t_new = 1.0
        else:
            t_new = t_star
        
        # Update
        current_max = float(np.max(c))
        f_new = f_of_t(t_new)
        if f_new < current_max - 1e-14:
            c = c0 + t_new * A - t_new * t_new * D
            h[p] = t_new
            improved = True
    
    return h, c, improved
