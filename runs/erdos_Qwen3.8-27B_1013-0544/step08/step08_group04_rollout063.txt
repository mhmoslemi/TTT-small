import numpy as np

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:

                h[idx[i]] += adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
    return h

        b = rng.uniform(-0.3/k, 0.3/k)
        h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)

def fourier_init(n, K, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    t = x - 1.0
    for k in range(1, K + 1):
        a = rng.uniform(-0.35 / k, 0.35 / k)
        b = rng.uniform(-0.35 / k, 0.35 / k)
        h += a * np.cos(k * np.pi * t) + b * np.sin(k * np.pi * t)
    return project_feasible(h, n)

def resample_h(old, n_new):
    n0 = len(old)
    if n0 == n_new:
        return project_feasible(old.copy(), n_new)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_feasible(np.interp(x1, x0, old), n_new)

        # Subgradient

                subgrad[i + shift] -= h_eval[i]
        else:

                subgrad[i] += (1.0 - h_eval[i - s])
                subgrad[i - s] -= h_eval[i]
        g = np.zeros(2 * K)
        for k in range(K):
            xk = (k + 1) * np.pi * (x - 1.0)
            g[2*k] = float(np.dot(subgrad, np.cos(xk))) * (2.0 / n)

        if time.time() > d3 - 50 or best_c <= 0.38073:
            break
        nt = int(min(500, max(200, 300 + (r % 3) * 100)))
        K = 5 + r % 4
        h0 = fourier_init(nt, K, rng)
        h_f, c_f = adam_fourier_opt(nt, K, h0, min(d3 - 30, t0 + (r + 1) * 0.04 * budget_s), rng, max_iter=400)
        h_f, c_f = cd_optimize(h_f, nt, 8, min(d3, t0 + (r + 1) * 0.05 * budget_s), seed=seed * 100 + r)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        # Perturbed restart from current best
        h0b = resample_h(best_h, nt) + rng.normal(0, 0.04, nt)
        h0b = project_feasible(h0b, nt)
        h_b, c_b = cd_optimize(h0b, nt, 6, d3, seed=seed * 200 + r)
        if c_b < best_c:
            best_h, best_c = h_b, c_b
    # Phase 4: Final polish
    h4, c4 = cd_optimize(best_h, len(best_h), 25, deadline, seed=seed + 7777, tol=1e-13)
    if c4 < best_c:
        best_h, best_c = h4, c4
    # Final re-resample to original n if beneficial
    if len(best_h) != n and time.time() < deadline - 15:
        h5 = resample_h(best_h, n)
        h5, c5v = cd_optimize(h5, n, 10, deadline, seed=seed + 8888, tol=1e-13)
        if c5v < best_c:
            best_h, best_c = h5, c5v
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
