**Idea 4: Direct minimax with active set**
At the optimum, suppose lags m1, m2, ..., mk are active (at the maximum). Then we want:
f_m1(h) = f_m2(h) = ... = f_mk(h) = C*
and f_m(h) ≤ C* for all other m.

This gives a system of k equations in n unknowns (plus the sum constraint). We can solve this with a Newton-like method:
- Start with an active set
- Solve the equality constraints to equalize the active lags
- Check if any inactive lag becomes active (add it) or any active lag becomes inactive (remove it)
- Iterate

This is the "active set method" for minimax problems. It's what the previous code's coordinate descent is essentially doing (implicitly), but an explicit version might be more efficient.

Let me implement a simplified version:

```python
def active_set_balance(h, n, max_iter=100):
    h = h.copy()
    for it in range(max_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        m_max = c.max()
        # Active set: lags within 1% of max
        active = np.where(c >= m_max * 0.999)[0]
        
        # Compute gradients of active lags w.r.t. h
        # Gradient of c[m] w.r.t. h[i] = (1-h[j]) if m = n-1+i-j, -h[j] if m = n-1-j+i
        # (similar to the subgradient computation)
        
        # We want to find a direction d such that:
        # For each active lag m: dc_m/dh * d = 0 (keep active lags equal)
        # And dc_m/dh * d < 0 for the most active lag (reduce the max)
        # Subject to: sum(d) = 0 (maintain sum constraint), and h + t*d stays in [0,1]
        
        # Simplified: use the average subgradient of active lags
        sg = np.zeros(n)
        for m in active:
            for i in range(n):
                j = i + m - (n - 1)
                if 0 <= j < n:
                    sg[i] += (1.0 - h[j])
                j2 = i - m + (n - 1)
                if 0 <= j2 < n:
                    sg[i] -= h[j2]
        sg /= len(active)
        
        # Project gradient to satisfy sum(d) = 0
        sg_mean = sg.mean()
        d = sg - sg_mean
        
        # Line search
        best_t = 0
        best_c = m_max
        for t in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]:
            h_new = project_feasible(h + t * d, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_t = t
        if best_t == 0:
            break
        h = project_feasible(h + best_t * d, n)
    
    return h, compute_c5(h, n)
```

This is essentially a subgradient method with the active set averaged. It's similar to what the previous code's `subgradient_polish` does, but with the active set averaging.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def obj_1d_sig(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for iteration in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += (1.0 - h[jf_arr[valid_f]])
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:

            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            nv = c_aff[:, None] + d_grid[None, :] * a_coeff[:, None]
            if diag_local is not None:
                nv[diag_local, :] -= d_grid**2
            vals = nv.max(axis=0)
            vals = np.maximum(vals, unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d_sig, gl, gh,
                                args=(c_aff, a_coeff, diag_local, unaff_max),
                                maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, best_c

def active_set_subgrad(h, n, max_iter=200, deadline=None, tol_frac=0.005):

                if 0 <= j2 < n:
                    sg[i] -= h[j2]

        sg -= sg.mean()

                best_c = c_new
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h, best_c

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def obj_1d_sig(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgrad_lag(m, h, n):
    L = m - (n - 1)
    sg = np.zeros(n)
    if L >= 0:
        sg[:n - L] += (1.0 - h[L:])
        sg[L:] -= h[:n - L]
    else:
        sg[-L:] += (1.0 - h[:n + L])
        sg[:n + L] -= h[-L:]
    return sg

def active_set_subgrad(h, n, max_iter=300, deadline=None, tol_frac=0.003):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_max = c.max()
        thresh = m_max * (1.0 - tol_frac)
        active = np.where(c >= thresh)[0]
        sg = np.zeros(n)
        for m in active:
            sg += subgrad_lag(m, h, n)
        sg /= len(active)
        sg -= sg.mean()
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_feasible(h - lr * scale * sg, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                improved = True
                stall = 0
                break
        if not improved:
            lr *= 0.6
            stall += 1
            if stall > 40 or lr < 1e-10:
                break
    return h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=55):
    h = h.copy()
    best_c = compute_c5(h, n)
    for iteration in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += (1.0 - h[jf_arr[valid_f]])
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = float(c[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            nv = c_aff[:, None] + d_grid[None, :] * a_coeff[:, None]
            if diag_local is not None:
                nv[diag_local, :] -= d_grid ** 2
            vals = nv.max(axis=0)
            vals = np.maximum(vals, unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d_sig, gl, gh,
                                 args=(c_aff, a_coeff, diag_local, unaff_max),
                                 maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, best_c

def multi_res_search(h_best, c_best, n_best, deadline, rng):
    x_orig = np.linspace(0, 2, n_best)
    for n_try in [500, 700, 800]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, h_best), n_try)
        h_o, c_o = slsqp_optimize(h_int, n_try, [200, 2000, 20000], 300, deadline)
        h_o, c_o = active_set_subgrad(h_o, n_try, 150, deadline)
        h_o, c_o = coord_cd(h_o, n_try, 80, deadline, grid_size=50)
        if c_o < c_best:
            h_best, c_best = h_o, c_o
            n_best = n_try
            x_orig = x_new
    return h_best, c_best, n_best

def perturbation_search(h_best, c_best, n, deadline, rng):
    for r in range(8):
        if time.time() > deadline - 60:
            break
        s = 0.003 + 0.03 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = slsqp_optimize(hp, n, [1000, 10000], 250, deadline)
        h_o, c_o = coord_cd(h_o, n, 60, deadline, grid_size=45)
        if c_o < c_best:
            h_best, c_best = h_o, c_o
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP smooth optimization
    if time.time() < deadline - 250:
        h1, c1 = slsqp_optimize(best_h, n, [100, 1000, 8000, 40000], 600, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Active-set subgradient
    if time.time() < deadline - 200:
        h2, c2 = active_set_subgrad(best_h, n, 300, deadline, tol_frac=0.002)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Coordinate descent
    if time.time() < deadline - 120:
        h3, c3 = coord_cd(best_h, n, 250, deadline, grid_size=60)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    if time.time() < deadline - 60:
        best_h, best_c, n = multi_res_search(best_h, best_c, n, deadline, rng)
    # Phase 5: Perturbation restarts
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        best_h, best_c = perturbation_search(best_h, best_c, n_cur, deadline, rng)
    # Phase 6: Final SLSQP + CD polish
    if time.time() < deadline - 15:
        hf, cf = slsqp_optimize(best_h, n_cur, [50000, 200000, 500000], 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n_cur, 400, deadline, grid_size=80)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
