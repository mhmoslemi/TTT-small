def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-15:

            tot = room.sum()

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

                else:
                    if a[i] > 1e-15:
                        d_hi2 = min(d_hi2, (L - c_base[i]) / a[i])
                    elif a[i] < -1e-15:
                        d_lo = max(d_lo, (L - c_base[i]) / a[i])
            d_lo = max(d_lo, lo_d)
            d_hi2 = min(d_hi2, hi_d)
            if d_lo > d_hi2:
                continue
            d_opt = (d_lo + d_hi2) / 2.0
            if abs(d_opt) > 1e-12:
                h[p] = float(np.clip(h0 + d_opt, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_step(h, n, step, max_iter=100, deadline=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        active = c >= c_max - 1e-10
        na = active.sum()
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if not active[m]:
                continue
            d = m - (n - 1)
            for i in range(n):
                j = i + d
                if 0 <= j < n:
                    grad[i] += (1.0 - h[i + d]) / na
                j2 = i - d
                if 0 <= j2 < n:
                    grad[i] -= h[i - d] / na
        h = project_feasible(h - step * grad, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_h, best_c = h.copy(), c_new
        step *= 0.95
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP polish
    if time.time() < deadline - 400:
        h1, c1 = slsqp_run(best_h, n, [5000, 50000, 500000, 2000000], 400)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: KKT linear system with varying tolerances
    if time.time() < deadline - 300:
        for tol_try in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            if time.time() > deadline - 150:
                break
            h2, c2 = kkt_solve(best_h, n, tol=tol_try, max_iter=30)
            if c2 < best_c:
                best_h, best_c = h2, c2
                # Follow up with CD after KKT improvement
                h2b, c2b = coord_cd(best_h, n, 20, deadline)
                if c2b < best_c:
                    best_h, best_c = h2b, c2b

    # Phase 3: Exact coordinate descent
    if time.time() < deadline - 60:
        rem = deadline - time.time() - 30
        passes = max(5, int(rem / 3.0))
        h3, c3 = coord_cd(best_h, n, passes, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Another KKT round
    if time.time() < deadline - 50:
        for tol_try in [0.003, 0.008, 0.03]:
            if time.time() > deadline - 30:
                break
            h4, c4 = kkt_solve(best_h, n, tol=tol_try, max_iter=20)
            if c4 < best_c:
                best_h, best_c = h4, c4
                h4b, c4b = coord_cd(best_h, n, 10, deadline)
                if c4b < best_c:
                    best_h, best_c = h4b, c4b

    # Phase 5: Subgradient for escape + SLSQP
    if time.time() < deadline - 80:
        hs, cs = subgrad_step(best_h, n, step=0.02, max_iter=30, deadline=deadline)
        if cs < best_c:
            best_h, best_c = hs, cs
        if time.time() < deadline - 40:
            h5, c5 = slsqp_run(best_h, n, [100000, 1000000], 300)
            if c5 < best_c:
                best_h, best_c = h5, c5

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h6, c6 = coord_cd(best_h, n, 10, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
