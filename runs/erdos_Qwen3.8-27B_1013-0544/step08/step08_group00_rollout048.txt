        mx = shifted.max()

        return (m + lse / alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if rnd_order:
            idx = rng.permutation(n)
        else:
            idx = np.arange(n)
        imp = 0

def obj_fn(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0

    return np.ones(n)

    h6, c6 = exact_cd(best_h, n, 200, deadline, rng, rnd_order=True)
    if c6 < best_c:
        best_h, best_c = h6, c6

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
