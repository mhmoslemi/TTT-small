def two_interval_h(a, l1, n, c_offset):
    """Build h from two intervals: [a, a+l1] and [a+c_offset, a+c_offset+1-l1] mod 2"""
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    # First interval
    mask1 = (x >= a) & (x < a + l1)
    # Second interval (shifted)
    c = (a + c_offset) % 2
    d = c + (1 - l1)
    if d <= 2:
        mask2 = (x >= c) & (x < d)
    else:
        mask2 = ((x >= c) | (x < d - 2))
    h[mask1] = 1.0
    h[mask2] = 1.0
    return h
