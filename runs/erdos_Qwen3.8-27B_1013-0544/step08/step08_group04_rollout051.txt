def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

import numpy as np

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0.0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def coord_eval(x, hp, cseg, a, qconst, unaff):
    if a.size:
        vmax = float(np.max(cseg + a * (x - hp)))
    else:
        vmax = 0.0
    return max(unaff, qconst + x * (1.0 - x), vmax)

def optimize_coord(hp, cseg, a, qconst, unaff, iters=20):

    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst)
    if zmax > low:

    high = max(low + 1e-11, unaff, qconst, qcur, cm) + 1e-8
