def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    return (m + float(np.log(s)) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    p = e / float(e.sum())
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e2 = -d
            g[:n - e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n - e2]
    return (2.0 / n) * g

    best_h = h.copy()

    work_h = best_h.copy()

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in alpha_schedule:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})

        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def compute_c5_eval(c, B, off, i, j, u):
    cn = c + u * B
    u2 = u * u
    cn[off] -= 2.0 * u2
    cn[off + i - j] += u2
    cn[off + j - i] += u2
    return float(cn.max())

def interp_h(h, m):

    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)
