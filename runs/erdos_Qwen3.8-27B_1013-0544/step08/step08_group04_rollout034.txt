def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if diff > 1e-14:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 0:
            h += (diff / tot) * cap
    elif diff < -1e-14:
        tot = h.sum()
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if len(idx): h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if len(idx): h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):

    return float(np.max(c) * (2.0 / n))

def resample_h(h, n):
    n0 = len(h)
    if n0 == n:
        return project_feasible(h, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, h), n)

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

    viol = np.maximum(0, h - 1.0) ** 2 + np.maximum(0, -h) ** 2
    penalty_val = penalty * viol.sum()
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / n) + penalty_val

            return fourier_evaluate(c, A, n, penalty)

                best_c = c_new
                best_h = h_new.copy()
            coeffs = res.x

            pass

Let me now think about the coordinate descent more carefully. The previous code's coordinate descent is quite sophisticated (exact piecewise optimization). Let me include a simplified but effective version.

The key idea of coordinate descent:
- For each coordinate p, fix all other coordinates
- The objective as a function of h[p] is: max(unaffected, max of linear functions, h[p](1-h[p]))
- Find the minimizer of this piecewise function

The "unaffected" part is the max of correlation values that don't involve h[p].
The linear part comes from correlations c_m that DO involve h[p]: c_m = const_m + a_m * h[p]
The quadratic part is from c_{center} = const + h[p](1-h[p])

So the objective is: max(unaff, max_m (const_m + a_m * x), qconst + x(1-x))

The minimum of this can be found by binary search on the threshold value, similar to what the previous code does.

def minimize_1d(hp, cseg, a, qconst, unaff):

    # Binary search on the value t: find smallest t such that there exists x with all constraints ≤ t
    tol = 1e-12
    lo = max(0.0, unaff, qconst)  # lower bound on answer
    hi = max(lo + 1e-6, unaff, qconst, float(np.max(cseg + a * hp)), 
             qconst + hp*(1-hp)) + 1e-6
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        # Check feasibility: exists x in [0,1] such that:
        #   cseg[k] + a[k]*x <= mid for all k
        #   qconst + x*(1-x) <= mid
        #   x in [0,1]
        # From linear constraints:
        #   a[k] > 0: x <= (mid - cseg[k]) / a[k]
        #   a[k] < 0: x >= (mid - cseg[k]) / a[k]
        xlo = 0.0
        xhi = 1.0
        for k in range(len(a)):
            if a[k] > tol:
                xhi = min(xhi, (mid - cseg[k]) / a[k])
            elif a[k] < -tol:
                xlo = max(xlo, (mid - cseg[k]) / a[k])
        
        # From quadratic: qconst + x - x^2 <= mid => x^2 - x + (mid - qconst) >= 0
        # Roots of x^2 - x + (mid-qconst) = 0: x = (1 ± sqrt(1 - 4(mid-qconst)))/2
        g = mid - qconst
        if g > 0.25:
            pass  # always satisfied (discriminant < 0, parabola always positive)
        elif g >= 0:
            disc = 1.0 - 4.0 * g
            if disc >= 0:
                r1 = 0.5 - 0.5 * np.sqrt(disc)
                r2 = 0.5 + 0.5 * np.sqrt(disc)
                # x^2 - x + g >= 0 means x <= r1 or x >= r2
                # So feasible: [0, r1] ∪ [r2, 1]
                # Combine with [xlo, xhi]
                # Check if [xlo, xhi] intersects [0, r1] or [r2, 1]
                if not (xlo <= r1 + tol and xhi >= -tol):
                    if not (xlo <= 1.0 + tol and xhi >= r2 - tol):
                        lo = mid
                        continue
                # Also need to check the intersection properly
                # [xlo, xhi] ∩ ([0, r1] ∪ [r2, 1]) must be non-empty
                overlap1 = xlo <= r1 + tol and xhi >= 0.0 - tol
                overlap2 = xlo <= 1.0 + tol and xhi >= r2 - tol
                if not (overlap1 or overlap2):
                    lo = mid
                    continue
            else:
                lo = mid
                continue
        else:
            # g < 0: x^2 - x + g < 0 for x in (r1, r2) where r1 < r2
            # So we need x NOT in (r1, r2), i.e., x <= r1 or x >= r2
            # Wait, x^2 - x + g >= 0 means OUTSIDE the roots
            # But g < 0 means the quadratic is negative between roots
            # So x^2 - x + g >= 0 is satisfied outside [r1, r2]

            r2 = 0.5 + 0.5 * np.sqrt(disc)  # > 1

            # Since r1 < 0 and r2 > 1, this is always true for x in [0,1]!
            pass
        
        if lo >= hi - 1e-14:
            break
        hi = mid

    for k in range(len(a)):

                qconst + x * (1 - x))
        if f < best_f:
            best_f = f
            best_x = x
    return best_x

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if diff > 1e-14:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 0:
            h += (diff / tot) * cap
    elif diff < -1e-14:
        tot = h.sum()
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if len(idx): h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if len(idx): h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def resample_h(h, n):
    if len(h) == n:
        return project_feasible(h, n)
    n0 = len(h)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, h), n)

def cd_pass(h, n, deadline):
    """One pass of coordinate descent with exact 1D minimization."""
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    center = n - 1
    L = 2 * n - 1
    for p in range(n):
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if vj.any(): a[vj] += 1.0 - h[j[vj]]
        if vi.any(): a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1)
        asg = np.empty(seg - 1)
        if off: cseg[:off] = c[lo:center]; asg[:off] = a[:off]
        if hi > center: cseg[off:] = c[center+1:hi+1]; asg[off:] = a[off+1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0: unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L: unaff = max(unaff, float(c[hi+1:].max()))
        # Minimize max(unaff, max(cseg + asg*x), qconst + x*(1-x))
        x_new = min1d(hp, cseg, asg, qconst, unaff)
        if abs(x_new - hp) > 1e-12:
            dx = x_new - hp
            if off: c[lo:center] += asg[:off] * dx
            if hi > center: c[center+1:hi+1] += asg[off:] * dx
            c[center] = qconst + x_new * (1.0 - x_new)
            h[p] = x_new
            obj = max(unaff, float(cseg.max() + asg * x_new) if len(cseg) else 0.0,
                     qconst + x_new*(1-x_new))
            if obj < cur: cur = float(c.max())
    return h, cur * (2.0 / n)

    if len(cseg) == 0:
        # Only quadratic: minimize qconst + x(1-x) subject to x >= unaff threshold
        # qconst + x - x^2 <= t => x in [0,1] works if qconst + 0.25 <= t
        # Just return hp
        return hp

        xlo, xhi = 0.0, 1.0
        pos = asg > tol
        neg = asg < -tol

            xhi = min(xhi, float(v.min()))
        if neg.any():
            v = (mid - cseg[neg]) / asg[neg]

        g = mid - qconst
        # x^2 - x + g >= 0 needed
        if g <= 0.25:

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if diff > 1e-14:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 0:
            h += (diff / tot) * cap
    elif diff < -1e-14:
        tot = h.sum()
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if res > 1e-10:
        idx = np.where(h < 1.0)[0]
        if len(idx):
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-10:
        idx = np.where(h > 0.0)[0]
        if len(idx):
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))

def resample_h(h, n):
    if len(h) == n:
        return project_feasible(h, n)
    n0 = len(h)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, h), n)

def min1d(hp, cseg, asg, qconst, unaff):
    tol = 1e-12
    if len(cseg) == 0:
        return hp
    lo = max(0.0, unaff, qconst)
    hi = max(lo + 1e-10, unaff, float(np.max(cseg + asg * hp)), qconst + hp * (1 - hp)) + 1e-8
    pos = asg > tol
    neg = asg < -tol
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        xlo, xhi = 0.0, 1.0
        if pos.any():
            xhi = min(xhi, float(((mid - cseg[pos]) / asg[pos]).min()))
        if neg.any():
            xlo = max(xlo, float(((mid - cseg[neg]) / asg[neg]).max()))
        if xlo > xhi + tol:
            lo = mid
            continue
        g = mid - qconst
        if g <= 0.25:
            disc = 1.0 - 4.0 * max(0.0, g)
            if disc < 0:
                lo = mid
                continue
            r1 = 0.5 - 0.5 * np.sqrt(disc)
            r2 = 0.5 + 0.5 * np.sqrt(disc)
            if not (xlo <= r1 + tol or xhi >= r2 - tol):
                lo = mid
                continue
        hi = mid
    t = hi
    xlo, xhi = 0.0, 1.0
    if pos.any():
        xhi = min(xhi, float(((t - cseg[pos]) / asg[pos]).min()))
    if neg.any():
        xlo = max(xlo, float(((t - cseg[neg]) / asg[neg]).max()))
    g = t - qconst
    cands = [max(xlo, 0.0), min(xhi, 1.0), 0.5]
    if g <= 0.25:
        disc = max(0.0, 1.0 - 4.0 * max(0.0, g))
        r1 = 0.5 - 0.5 * np.sqrt(disc)
        r2 = 0.5 + 0.5 * np.sqrt(disc)
        cands.extend([r1, r2])
    best_x, best_f = hp, float('inf')
    for x in cands:
        if x < 0 or x > 1:
            continue
        f = max(unaff, float(np.max(cseg + asg * x)), qconst + x * (1 - x))
        if f < best_f:
            best_f, best_x = f, x
    return best_x

def cd_pass(h, n, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    L = 2 * n - 1
    for p in range(n):
        if time.time() > deadline:
            break
        hp = float(h[p])
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        seg = hi - lo + 1
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        a = np.zeros(seg)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if vj.any():
            a[vj] += 1.0 - h[j[vj]]
        if vi.any():
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1)
        asg = np.empty(seg - 1)
        if off:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center + 1:hi + 1]
            asg[off:] = a[off + 1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:

                c[center + 1:hi + 1] += asg[off:] * dx
            c[center] = qconst + x_new * (1.0 - x_new)
            h[p] = x_new
    return h

    pen = lam * (np.maximum(0, h - 1.0)**2 + np.maximum(0, -h)**2).sum()
    hc = project_feasible(h, n)
    return compute_c(hc, n) + float(pen)

            h_new = project_feasible(0.5 + A @ res.x, n)
            c_new = compute_c(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            coeffs = res.x
        except Exception:
            pass
    return best_h, best_c

def cd_multi(h, n, passes, deadline):

    best_c = compute_c(best_h, n)
