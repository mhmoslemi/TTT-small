def project_feasible(z, n):
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    target = 0.5 * n
    if abs(z.sum() - target) < 1e-12:
        return z
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        mask = y < 1.0 if res > 0 else y > 0.0
        if mask.any():
            room = (1.0 - y) if res > 0 else y
            t = room.sum()
            if t > 1e-14:
                y[mask] += (res / t) * room[mask]
    return np.clip(y, 0.0, 1.0)

def _grad_w(h, n, p):
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - h[:n - d])
            g[:n - d] -= w * h[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - h[e:])
            g[e:] -= w * h[:n - e]
    return (2.0 / n) * g

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    z = alpha * (c - m)
    z -= z.max()
    e = np.exp(z)
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)

def smooth_jac(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    z = alpha * (c - m)
    z -= z.max()
    e = np.exp(z)
    return _grad_w(h, n, e / e.sum())

def _c_fun(x, n):
    return x.sum() - 0.5 * n

def _c_jac(x, n):
    return np.ones_like(x)

def slsqp_run(h, n, alphas, iters, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    work = best_h.copy()
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (n,)}]
    for a, mx in zip(alphas, iters):
        if time.time() > deadline:
            break
        try:
            r = minimize(smooth_obj, work, args=(n, a), method='SLSQP',
                         jac=smooth_jac, bounds=bnds, constraints=cons,
                         options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = project_feasible(np.asarray(r.x, dtype=float), n)
            c = compute_c5(x, n)
            if c < best_c:
                best_h, best_c = x, c
            work = x if c < best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def subgrad_polish(h, n, steps, eta0, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    work = best_h.copy()
    for i in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(work, 1.0 - work, mode='full')
        m = float(c.max())
        tol = max(0.005 * m, 1e-6)
        act = c >= m - tol
        if act.sum() == 0:
            act = np.zeros(2 * n - 1)
            act[int(np.argmax(c))] = 1.0
        p = act.astype(float)
        p /= p.sum()
        g = _grad_w(work, n, p)
        eta = eta0 / (1.0 + 0.02 * i)
        work = project_feasible(work - eta * g, n)
        c = compute_c5(work, n)
        if c < best_c:
            best_h, best_c = work.copy(), c
    return best_h, best_c

def pair_search(h, n, max_tries, deadline, rng):
    best_h, best_c = h.copy(), compute_c5(h, n)
    c = np.correlate(best_h, 1.0 - best_h, mode='full')
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - best_h[:n - d]
            A[:n - d, k] -= best_h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - best_h[e:]
            A[e:, k] -= best_h[:n - e]
    no_imp = 0
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.01, 0.002 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m))
                i, j = b - e, b
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        S = hi + hj
        lo_b = max(0.0, S - 1.0)
        hi_b = min(1.0, S)
        if hi_b - lo_b <= 1e-14:
            no_imp += 1
            continue
        grid = 40
        us = np.linspace(lo_b - hi, hi_b - hi, grid)
        B = A[i] - A[j]
        lines = c + us[:, None] * B[None, :]
        u2 = us * us
        lines[:, off] -= 2.0 * u2
        lines[:, off + i - j] += u2
        lines[:, off + j - i] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        if vals[idx] < cur_max - 1e-10:
            u = float(us[idx])
            t = float(np.clip(hi + u, 0.0, 1.0))
            u = t - hi
            B2 = A[i] - A[j]
            cn = c + u * B2
            u2 = u * u
            cn[off] -= 2.0 * u2
            cn[off + i - j] += u2
            cn[off + j - i] += u2
            nm = float(cn.max())
            if nm < cur_max - 1e-10:
                best_h[i] = t
                best_h[j] = S - t
                c = cn
                no_imp = 0
                A = np.zeros((n, 2 * n - 1), dtype=float)
                for k in range(2 * n - 1):
                    d = k - off
                    if d >= 0:
                        A[d:, k] += 1.0 - best_h[:n - d]
                        A[:n - d, k] -= best_h[d:]
                    else:
                        e2 = -d
                        A[:n - e2, k] += 1.0 - best_h[e2:]
                        A[e2:, k] -= best_h[:n - e2]
                if trial % 50 == 0:
                    c = np.correlate(best_h, 1.0 - best_h, mode='full')
        else:
            no_imp += 1
        if no_imp > max(300, max_tries // 3):
            break
    return best_h, compute_c5(best_h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def diverse_starts(h0, n0, rng):
    starts = [h0.copy()]
    starts.append(1.0 - h0)
    starts.append(project_feasible(h0[::-1].copy(), n0))
    # Fourier limited
    x = np.linspace(0.0, 2.0, n0, endpoint=False)
    for M in [3, 5, 8]:
        coeffs = np.zeros(M + 1)
        for m in range(1, M + 1):
            coeffs[m] = np.sum(h0 * np.cos(np.pi * m * x)) / (0.5 * n0)
        h_f = np.zeros(n0)
        h_f += coeffs[0] * 0.5
        for m in range(1, M + 1):
            h_f += coeffs[m] * np.cos(np.pi * m * x)
        starts.append(project_feasible(np.clip(h_f, 0.0, 1.0), n0))
    # Random perturbation
    starts.append(project_feasible(h0 + rng.normal(0, 0.1, n0), n0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        x = np.linspace(0.0, 2.0, 400, endpoint=False)
        h0 = project_feasible(np.clip(0.5 + 0.3 * np.sin(2.5 * np.pi * x), 0.0, 1.0), 400)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Symmetry: try complement
    hc = 1.0 - best_h
    cc = compute_c5(hc, n0)
    if cc < best_c:
        best_h, best_c = hc, cc
    # Phase 1: Continuation SLSQP on best starting point
    alphas1 = [5, 20, 80, 300, 1200, 5000, 20000, 80000]
    iters1 = [80, 60, 50, 45, 40, 35, 30, 25]
    dl1 = t0 + 220
    best_h, best_c = slsqp_run(best_h, n0, alphas1, iters1, dl1)
    # Phase 2: Subgradient polish
    dl2 = min(hard - 350, t0 + 400)
    best_h, best_c = subgrad_polish(best_h, n0, 5000, 0.04, dl2)
    # Phase 3: Pair search
    dl3 = min(hard - 250, time.time() + 150)
    best_h, best_c = pair_search(best_h, n0, 8000, dl3, rng)
    # Phase 4: Multi-resolution
    cands = [200, 300, 400, 500, 600, 750, 800, 1000]
    for m in cands:
        if time.time() > hard - 200:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        if rem < 80:
            break
        if m >= 800 and rem < 150:
            break
        dl = time.time() + min(60, 0.18 * rem)
        hm = interp_h(best_h, m)
        alphas_m = [10, 50, 200, 1000, 5000, 20000]
        iters_m = [50, 40, 35, 30, 25, 20]
        hm, cm = slsqp_run(hm, m, alphas_m, iters_m, dl)
        hm, cm = subgrad_polish(hm, m, 1000, 0.03, min(hard - 100, time.time() + 25))
        if cm < best_c:
            best_h, best_c = hm.copy(), cm
    # Phase 5: Diverse restarts on best resolution
    if time.time() < hard - 150:
        n = len(best_h)
        starts = diverse_starts(best_h, n, rng)
        for s in starts[1:]:
            if time.time() > hard - 80:
                break
            dl = time.time() + min(40, 0.1 * (hard - time.time()))
            sh = project_feasible(s, n)
            sc = compute_c5(sh, n)
            am = [50, 300, 2000, 10000, 50000]
            im = [40, 35, 30, 25, 20]
            sh, sc = slsqp_run(sh, n, am, im, dl)
            sh, sc = subgrad_polish(sh, n, 800, 0.02, min(hard - 50, time.time() + 15))
            if sc < best_c:
                best_h, best_c = sh, sc
    # Phase 6: Final aggressive polishing
    if time.time() < hard - 20:
        n = len(best_h)
        dl6 = hard - 5
        best_h, best_c = pair_search(best_h, n, 6000, dl6, rng)
        best_h, best_c = subgrad_polish(best_h, n, 3000, 0.01, dl6)
        if time.time() < dl6 - 5:
            alphas_f = [50000, 200000, 500000]
            iters_f = [25, 20, 15]
            best_h, best_c = slsqp_run(best_h, n, alphas_f, iters_f, dl6)
            best_h, best_c = subgrad_polish(best_h, n, 2000, 0.005, dl6)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
