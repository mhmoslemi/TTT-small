            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d**2
                current_c5 = min(current_c5, max(unaff_max, float(c[ms].max())))
                improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        if (p+1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(np.max(c))
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb_restart(h, n, rng, n_pert=15, mag=0.15):
    hp = h.copy()
    idxs = rng.choice(n, size=min(n_pert, n), replace=False)
    for i in idxs:
        hp[i] += rng.uniform(-mag, mag)
    return project_feasible(hp, n)

def slsqp_opt(h, n, alpha, maxiter=800):
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return (m + (mx + np.log(np.sum(np.exp(sh-mx))))/alpha) * (2.0/n)
    def obj_jac(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            wk = w[n-1+k]
            if wk < 1e-14:
                continue
            lo1, hi1 = max(0, k), min(n, k+n)
            lo2, hi2 = max(0, -k), min(n, n-k)
            if lo1 < hi1:
                grad[lo1:hi1] += wk * (1.0 - x[lo1-k:hi1-k])
            if lo2 < hi2:
                grad[lo2:hi2] += wk * (-x[lo2+k:hi2+k])
        return grad * (2.0/n)
    def sc(x):
        return x.sum() - n/2.0
    def scj(x):
        return np.ones(n)
    try:
        res = minimize(obj, opt_h, args=(n, alpha), jac=lambda x: obj_jac(x),
                       method='SLSQP', bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':sc,'jac':scj,'args':(n,)}],
                       options={'maxiter':maxiter, 'ftol':1e-20})
        nh = project_feasible(res.x, n)
        return nh, compute_c5(nh, n)
    except Exception:
        return opt_h, compute_c5(opt_h, n)

opt_h = None

def run(seed=42, budget_s=1100, **kwargs):
    global opt_h
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    
    # Phase 1: SLSQP with various alphas
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > deadline - 400:
            break
        opt_h = best_h.copy()
        oh, oc = slsqp_opt(best_h, n, alpha, maxiter=600)
        if oc < best_c:
            best_h, best_c = oh, oc
    
    # Phase 2: Coordinate descent
    if time.time() < deadline - 300:
        h2, c2 = coord_cd(best_h, n, max_passes=4000, deadline=deadline-300, grid_size=120)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase 3: Perturbation restarts
    n_pert = 8
    for i in range(n_pert):
        if time.time() > deadline - 200:
            break
        hp = perturb_restart(best_h, n, rng, n_pert=10+i*3, mag=0.05+0.03*i)
        ho, co = coord_cd(hp, n, max_passes=2000, deadline=deadline-150, grid_size=100)
        if co < best_c:
            best_h, best_c = ho, co
    
    # Phase 4: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 500, 600, 800]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        opt_h = hi.copy()
        ho, co = slsqp_opt(hi, n_try, 20000, maxiter=400)
        ho, co = coord_cd(ho, n_try, max_passes=1500, deadline=deadline-80, grid_size=100)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = np.linspace(0, 2, n_try)
    
    # Phase 5: Final coordinate descent
    if time.time() < deadline - 10:
        h5, c5 = coord_cd(best_h, len(best_h), max_passes=5000, deadline=deadline-5, grid_size=150)
        if c5 < best_c:
            best_h, best_c = h5, c5
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
