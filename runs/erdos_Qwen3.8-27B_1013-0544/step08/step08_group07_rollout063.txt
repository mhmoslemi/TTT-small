def exact_coord_step(c, h, p, n, top_k=50):
    """Find optimal d to add to h[p] that minimizes max lag value."""
    h0 = h[p]
    # Compute affected lags
    # For lag j (0-indexed from 0 to 2n-2), the shift is j - (n-1)
    # h[p] affects lag j if:
    #   - p is in range for h: 0 <= p < n (always true)
    #   - p + j - (n-1) is in range for (1-h): 0 <= p + j - (n-1) < n
    #   OR
    #   - p - j + (n-1) is in range for h: 0 <= p - j + (n-1) < n
    
    # Lags where p contributes to h term: j such that 0 <= p+j-(n-1) < n → (n-1-p) <= j <= (n-1-p)+n-1
    j_lo1 = max(0, n-1-p)
    j_hi1 = min(2*n-2, n-1-p+n-1)
    
    # Lags where p contributes to (1-h) term: j such that 0 <= p-j+(n-1) < n → p <= j <= p+n-1
    j_lo2 = max(0, p)
    j_hi2 = min(2*n-2, p+n-1)
    
    j_lo = min(j_lo1, j_lo2)
    j_hi = max(j_hi1, j_hi2)
    
    # For each affected lag j, compute A[j] (coefficient of d) and B[j] (coefficient of d²)
    # A[j] = (1-h[p+j-(n-1)]) if valid_h_term - h[p-j+(n-1)] if valid_1h_term
    # B[j] = -1 if j == n-1 else 0
    
    affected = np.arange(j_lo, j_hi+1)
    A = np.zeros(len(affected))
    B = np.zeros(len(affected))
    
    for idx, j in enumerate(affected):
        shift = j - (n-1)
        # h term: h[p] * (1-h[p+shift]), valid if 0 <= p+shift < n
        if 0 <= p + shift < n:
            A[idx] += (1.0 - h[p+shift])
        # (1-h) term: h[p-shift] * (1-h[p]), valid if 0 <= p-shift < n
        if 0 <= p - shift < n:
            A[idx] -= h[p-shift]
        # Check if both are the same term (diagonal): shift = 0, i.e., j = n-1
        if j == n-1:
            B[idx] = -1.0
    
    # Unaffected max
    unaff_idx = np.setdiff1d(np.arange(2*n-1), affected)
    unaff_max = float(c[unaff_idx].max()) if len(unaff_idx) > 0 else 0.0
    
    # Current values of affected lags
    c_aff = c[affected]
    
    # Find the top_k affected lags (those closest to the max)
    # We want to find d that minimizes max(max(c_aff + d*A - d²*B, unaff_max), ...)
    
    # Since B is 0 for all except possibly j=n-1, the function max(c_aff + d*A, unaff_max)
    # is piecewise linear, and subtracting d² at one point makes it slightly different.
    
    # Strategy: find candidate d values where the max changes
    # The max of {c_aff[i] + d*A[i]} for i in range(len) is a piecewise linear convex function
    # Its minimum occurs where the subgradient contains 0
    
    # For now, use a fine grid on the affected lags
    # The optimal d is in [-h0, 1-h0]
    
    # Find crossing points of the top lags
    # Sort by current value
    c_aff_sorted = c_aff.copy()
    
    # Take top K lags
    k = min(top_k, len(affected))
    top_indices = np.argsort(-c_aff_sorted)[:k]
    
    # Candidate d values: 0, -h0, 1-h0, and crossing points
    candidates = [0.0, -h0, 1.0-h0]
    
    # Crossing points between top lags
    for i in range(k):
        for j2 in range(k):
            if i == j2:
                continue
            # c_aff[top_indices[i]] + d*A[top_indices[i]] - d²*B[top_indices[i]]
            # = c_aff[top_indices[j2]] + d*A[top_indices[j2]] - d²*B[top_indices[j2]]
            # (c_i - c_j2) + d*(A_i - A_j2) - d²*(B_i - B_j2) = 0
            ci = c_aff[top_indices[i]]
            cj = c_aff[top_indices[j2]]
            Ai = A[top_indices[i]]
            Aj = A[top_indices[j2]]
            Bi = B[top_indices[i]]
            Bj = B[top_indices[j2]]
            
            # (Bj-Bi)*d² + (Ai-Aj)*d + (ci-cj) = 0
            a2 = Bj - Bi
            a1 = Ai - Aj
            a0 = ci - cj
            
            if abs(a2) < 1e-15:
                # Linear: a1*d + a0 = 0
                if abs(a1) > 1e-15:
                    d = -a0/a1
                    if -h0 <= d <= 1-h0:
                        candidates.append(d)
            else:
                disc = a1*a1 - 4*a2*a0
                if disc >= 0:
                    d1 = (-a1 + np.sqrt(disc)) / (2*a2)
                    d2 = (-a1 - np.sqrt(disc)) / (2*a2)
                    if -h0 <= d1 <= 1-h0:
                        candidates.append(d1)
                    if -h0 <= d2 <= 1-h0:
                        candidates.append(d2)
    
    # Also add some points around 0 and around the unconstrained minimizer
    # Evaluate at all candidates
    best_d = 0.0
    best_val = max(unaff_max, float(c_aff.max()))
    
    for d in candidates:
        d = float(np.clip(d, -h0, 1.0-h0))
        vals = c_aff + d*A - d*d*B
        val = max(unaff_max, float(vals.max()))
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d, best_val
