    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def optimize_coordinate(hp, cseg, a, qconst, unaff, iters=14):
    tol = 1e-10
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst)
    if zmax > low:
        low = zmax
    high = max(low + 1e-11, unaff, qconst, qcur, cm)
    if zmax > high:
        high = zmax
    high += 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    pc, nc = a_pos.size, a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh:
                    lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    g = max(0.0, g)
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    if max(0.0, ll) <= min(r1, lh) + tol:
                        feas = True
                    elif max(r2, ll) <= min(1.0, lh) + tol:
                        feas = True
        if feas:
            high = mid
        else:
            low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll, lh = 0.0, 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            ll = max(ll, v)
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            lh = min(lh, v)
        if ll <= 1.0 + tol and lh >= -tol:
            g = t - qconst
            if g > 0.25:
                cands.extend([max(0.0, ll), min(1.0, lh)])
            elif g >= -1e-12:
                g = max(0.0, g)
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 - 0.1) + 4.0 * (1.0 - 4.0 * (g - g))) 
                # Simplified: just use the two roots
                disc = max(0.0, 1.0 - 4.0 * g)
                if disc >= 0:
                    r1 = 0.5 - 0.5 * np.sqrt(disc)
                    r2 = 0.5 + 0.5 * np.sqrt(disc)
                    cands.extend([r1, r2, 0.0, 1.0])
    best_x, best_f = hp, qconst + hp * (1.0 - hp)
    for x in cands:
        if x < 0.0 or x > 1.0:
            continue
        val = max(unaff, qconst + x * (1.0 - x))
        if m:
            val = max(val, float(np.max(cseg + a * (x - hp))))
        if val < best_f - 1e-13:
            best_f, best_x = val, x
    return best_x, best_f

def cd_pass(h, n, c, cur, deadline):
    center = n - 1
    L = 2 * n - 1
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        hp = h[p]
        lo = min(p, center - p)
        hi = max(center + p, 2 * center - p)
        s = center - np.arange(lo, hi + 1, dtype=int)
        j = p + s
        i = p - s
        seg = hi - lo + 1
        a = np.zeros(seg)
        vj = (j >= 0) & (j < n)
        vi = (i >= 0) & (i < n)
        if np.any(vj):
            a[vj] += 1.0 - h[j[vj]]
        if np.any(vi):
            a[vi] -= h[i[vi]]
        off = center - lo
        cseg = np.empty(seg - 1)
        asg = np.empty(seg - 1)
        if off > 0:
            cseg[:off] = c[lo:center]
            asg[:off] = a[:off]
        if hi > center:
            cseg[off:] = c[center+1:hi+1]
            asg[off:] = a[off+1:]
        qconst = c[center] - hp * (1.0 - hp)
        unaff = 0.0
        if lo > 0:
            unaff = max(unaff, float(c[:lo].max()))
        if hi + 1 < L:
            unaff = max(unaff, float(c[hi+1:].max()))
        bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff)
        if bf < cur - 1e-11:
            dxv = bx - hp
            if off > 0:
                c[lo:center] += asg[:off] * dxv
            if hi > center:
                c[center+1:hi+1] += asg[off:] * dxv
            c[center] = qconst + bx * (1.0 - bx)
            h[p] = bx
            cur = bf
            improved = True
    return h, cur, improved

def cd_optimize(h, n, max_passes, deadline, project_every=5):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_c_val = (2.0 / n) * cur
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        _, cur, improved = cd_pass(h, n, c, cur, deadline)
        if (p + 1) % project_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c_val - 1e-14:
            best_c_val = cval
            best_h = h.copy()
        if not improved:
            break
    h = project_feasible(h, n)
    final_c = compute_c(h, n)
    if final_c < best_c_val:
        best_c_val, best_h = final_c, h
    return best_h, best_c_val

def basis_h(n, rng, modes=6):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = 0.5 * np.ones(n)
    for k in range(1, modes + 1):
        amp = rng.uniform(-0.3 / k, 0.3 / k)
        h += amp * np.cos(k * np.pi * t)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 15.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 500
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h_init = project_feasible(1.0 - np.abs(x - 1.0), n)
    n_init = len(h_init)
    if n_init % 2:
        h_init = resample_h(h_init, n_init + 1)
        n_init += 1
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c(best_h, n_init)
    # Phase 1: SLSQP at native resolution
    if time.time() < deadline - 200:
        h1, c1 = slsqp_opt(best_h, min(n_init, 600), deadline, alphas=(100, 1000, 10000, 100000), maxiter=150)
        if c1 < best_c:
            best_h, best_c = resample_h(h1, n_init) if len(h1) != n_init else h1, c1
        else:
            best_h, best_c = resample_h(h1, n_init) if len(h1) != n_init else h1, compute_c(best_h, n_init)
    # Phase 2: CD at lower resolution for speed
    for nt in [n_init, 400, 320, 256]:
        if time.time() > deadline - 150:
            break
        if nt > n_init:
            continue
        h0 = resample_h(best_h, nt)
        n_passes = 40 if nt <= 300 else 25
        h1, c1 = cd_optimize(h0, nt, n_passes, deadline, project_every=5)
        if c1 < best_c:
            best_h, best_c = resample_h(h1, n_init), c1
    # Phase 3: Interleave SLSQP and CD at multiple resolutions
    for r in range(20):
        if time.time() > deadline - 120:
            break
        if best_c <= 0.38075:
            break
        if r % 3 == 0:
            nt = n_init
        elif r % 3 == 1:
            nt = 320
        else:
            nt = 400
        h0 = resample_h(best_h, nt)
        # SLSQP
        if nt <= 450 and time.time() < deadline - 180:
            h1, c1 = slsqp_opt(h0, nt, deadline, alphas=(200, 2000, 20000), maxiter=100)
        else:
            h1, c1 = cd_optimize(h0, nt, 15, deadline, project_every=6)
        # CD polish
        h2, c2 = cd_optimize(h1, nt, 30, deadline, project_every=5)
        if c2 < best_c:
            best_h, best_c = resample_h(h2, n_init), c2
    # Phase 4: Random restarts
    for r in range(15):
        if time.time() > deadline - 100:
            break
        if best_c <= 0.38075:
            break
        nt = 320 if r % 2 == 0 else n_init
        base = resample_h(best_h, nt)
        typ = r % 4
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0.0, 0.03, nt), nt)
        elif typ == 1:
            h0 = project_feasible(base + rng.normal(0.0, 0.08, nt), nt)
        elif typ == 2:
            h0 = basis_h(nt, rng, modes=8)
        else:
            h0 = project_feasible(np.clip(base * rng.uniform(0.7, 1.3, nt), 0, 1), nt)
        h1, c1 = cd_optimize(h0, nt, 25, deadline, project_every=5)
        if c1 < best_c:
            best_h, best_c = resample_h(h1, n_init), c1
    # Phase 5: Final heavy CD polish at native resolution
    if time.time() < deadline - 30:
        h_final = project_feasible(best_h, n_init)
        h_final, c_final = cd_optimize(h_final, n_init, 60, deadline, project_every=4)
        if c_final < best_c:
            best_h, best_c = h_final, c_final
    # Final SLSQP polish
    if time.time() < deadline - 20 and len(best_h) <= 600:
        h_f, c_f = slsqp_opt(best_h, len(best_h), deadline, alphas=(50000, 200000), maxiter=80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
