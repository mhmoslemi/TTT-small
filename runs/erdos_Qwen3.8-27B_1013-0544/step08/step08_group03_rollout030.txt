```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h_new = np.clip(h, 0.0, 1.0)
    # Final correction: adjust non-bound elements
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        interior = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if interior.sum() > 0:
            h_new[interior] += diff / interior.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def subgradient_step(h, n, smoothing=0.01, step_size=0.01):
    """One subgradient step for minimizing max of correlation values."""
    corr = compute_corr(h, n)
    # Softmax weights for active set
    logits = corr / smoothing
    logits -= logits.max()
    weights = np.exp(logits)
    weights /= weights.sum()
    
    # Compute subgradient
    # grad[p] = sum_m weights[m] * [(1-h[p+m])*valid(p+m) - h[p-m]*valid(p-m)]
    # where m = idx - (n-1)
    grad = np.zeros(n)
    for m_idx in range(2*n-1):
        m = m_idx - (n-1)
        w = weights[m_idx]
        if w < 1e-10:
            continue
        # (1-h[p+m]) for valid p+m
        if m >= 0:
            p_range = range(n - m)
            for p in p_range:
                grad[p] += w * (1.0 - h[p + m])
        else:
            p_range = range(-m, n)
            for p in p_range:
                grad[p] += w * (1.0 - h[p + m])
        # -h[p-m] for valid p-m
        if m <= 0:
            p_range = range(-m, n)
            for p in p_range:
                grad[p] -= w * h[p - m]
        else:
            p_range = range(n - m)
            for p in p_range:
                grad[p] -= w * h[p - m]
    
    # Projected gradient descent
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(x, n, t):
    """Log-sum-exp objective and gradient for the C5 problem."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    obj = (1.0/t) * (t*np.max(corr) + np.log(np.sum(np.exp(t*corr - t*np.max(corr))))) * dx
    # Softmax weights
    tc = t * corr
    tc -= tc.max()
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    # Gradient via correlation/convolution
    T1 = np.correlate(1.0 - x, w, mode='full')[3*n-3:2*n-3:-1]
    T2 = np.convolve(x, w, mode='full')[n-1:n-1+n]
    grad = dx * (T1 - T2)
    return obj, grad

def slsqp_lse(h, n, t, deadline, maxiter=500):
    """SLSQP with log-sum-exp objective."""
    def obj(x):
        val, _ = lse_objective_and_grad(x, n, t)
        return val
    def obj_grad(x):
        _, g = lse_objective_and_grad(x, n, t)
        return g
    def con_fun(x):
        return x.sum() - n/2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=30, gs_iters=30):
    h = h.copy()
    current_max = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            # Compute A and D vectors for this coordinate
            A = np.zeros(2*n-1)
            D = np.zeros(2*n-1)
            D[n-1] = 1.0
            # Diagonal: contributes to corr[n-1] with coefficient (1-2h0) when h0 changes
            # Actually let me recompute properly
            # corr[k] = sum_i h[i]*(1-h[i+k-(n-1)])
            # d(corr[k])/dh[p] = (1-h[p+k-(n-1)])*valid - h[p-k+(n-1)]*valid (for k≠n-1)
            # for k=n-1: d(corr[k])/dh[p] = (1-h[p]) - h[p] = 1-2h[p]
            # So if we change h[p] by delta:
            # corr[k] += delta * [(1-h[p+m])*valid(p+m) - h[p-m]*valid(p-m)] for m≠0
            # corr[n-1] += delta * (1-2h[p]) for k=n-1
            # But we can write uniformly: corr[k] += delta * [(1-h[p+m])*I - h[p-m]*I]
            # where m = k-(n-1), and for m=0: (1-h[p])*1 - h[p]*1 = 1-2h[p] ✓
            
            m = np.arange(2*n-1) - (n-1)  # shifts
            # First term: (1-h[p+m]) where valid
            pm = p + m
            valid_pm = (pm >= 0) & (pm < n)
            # Second term: h[p-m] where valid
            pmm = p - m
            valid_pmm = (pmm >= 0) & (pmm < n)
            
            A = np.zeros(2*n-1)
            A[valid_pm] = 1.0 - h[pm[valid_pm]]
            A[valid_pmm] -= h[pmm[valid_pmm]]
            # Note: for m=0, both are valid and we get (1-h[p]) - h[p] = 1-2h[p] ✓
            
            # c0 = corr - h0*A + h0^2 * D
            # where D[n-1] = 1 (the quadratic term only for k=n-1)
            D = np.zeros(2*n-1)
            D[n-1] = 1.0
            c0 = corr - h0 * A + h0*h0 * D
            
            # Minimize max(c0 + t*A - t^2*D) over t in [0,1]
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.array([float(np.max(c0 + t*A - t*t*D)) for t in grid])
            best_idx = int(np.argmin(fvals))
            lo = grid[max(0, best_idx-1)]
            hi = grid[min(grid_pts-1, best_idx+1)]
            if hi - lo < 1e-14:
                t_star = grid[best_idx]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr*(b-a)
                d1 = a + gr*(b-a)
                fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        b = d1; d1 = c1; fd1 = fc1
                        c1 = b - gr*(b-a); fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                    else:
                        a = c1; c1 = d1; fc1 = fd1
                        d1 = a + gr*(b-a); fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                t_star = (a+b)*0.5
            f_star = float(np.max(c0 + t_star*A - t_star*t_star*D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            if min(f_star, f0, f1) < current_max - 1e-13:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                corr = c0 + t_new*A - t_new*t_new*D
                h[p] = t_new
                current_max = float(np.max(corr))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, n, t):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    scaled = t * corr * dx
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    obj = (scaled_max + np.log(exp_scaled.sum())) / t
    w = exp_scaled / exp_scaled.sum()
    T1 = np.correlate(1.0 - x, w, mode='full')[3*n-3:2*n-3:-1]
    T2 = np.convolve(x, w, mode='full')[n-1:n-1+n]
    grad = dx * (T1 - T2)
    return obj, grad

def slsqp_lse(h, n, t, deadline, maxiter=500):
    def obj(x):
        v, _ = lse_obj_grad(x, n, t)
        return v
    def og(x):
        _, g = lse_obj_grad(x, n, t)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h.copy(), jac=og, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)

def subgradient_descent(h, n, max_iter=100, deadline=None):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = 50.0 + 200.0 * it / max_iter
        tc = t_val * corr
        tc -= tc.max()
        exp_tc = np.exp(tc)
        w = exp_tc / exp_tc.sum()
        T1 = np.correlate(1.0 - h, w, mode='full')[3*n-3:2*n-3:-1]
        T2 = np.convolve(h, w, mode='full')[n-1:n-1+n]
        grad = (2.0 / n) * (T1 - T2)
        step = 0.02 * (1.0 - it / max_iter) + 0.001
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

            if deadline is not None and time.time() > deadline:

            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue

            pmm = p - m_arr

            valid_pmm = (pmm >= 0) & (pmm < n)
            A = np.zeros(2 * n - 1)

            A[valid_pmm] -= h[pmm[valid_pmm]]
```
