            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                t = float(c[:m_lo].max())
                if t > unaff_max:
                    unaff_max = t
            if m_hi < 2 * n - 2:
                t = float(c[m_hi + 1:].max())
                if t > unaff_max:
                    unaff_max = t
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2]
                        * (x[0] -
                         x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0])
                               + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B smooth optimization with multiple alphas
    if time.time() < deadline - 500:
        h1, c1 = l_bfgsb_opt(h, n, [30, 100, 500, 2000, 10000, 50000],
                             maxiter=200, deadline=deadline - 500)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Main coordinate descent with fine grid
    h2, c2 = coord_cd(best_h, n, max_passes=30000, deadline=deadline - 300,
                      grid_size=150)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Random shake + re-optimize to escape local optima
    if time.time() < deadline - 200:
        for trial in range(8):
            if time.time() > deadline - 180:
                break
            mag = 0.005 + 0.005 * trial
            h_sh = best_h + rng.normal(0, mag, n)
            h_sh = project_feasible(h_sh, n)
            h3, c3 = coord_cd(h_sh, n, max_passes=5000,
                              deadline=deadline - 150, grid_size=120)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 4: Try different grid sizes via interpolation
    for n_try in [300, 400, 600]:
        if time.time() > deadline - 80:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h4, c4 = l_bfgsb_opt(h_int, n_try, [2000, 20000], maxiter=150,
                             deadline=deadline - 60)
        h4, c4 = coord_cd(h4, n_try, max_passes=5000, deadline=deadline - 40,
                          grid_size=120)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Final fine-grid polish
    h5, c5 = coord_cd(best_h, len(best_h), max_passes=10000, deadline=deadline,
                      grid_size=250)
    if c5 < best_c:
        best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
