Actually wait - the max of correlations is NOT necessarily convex in h[j]. The individual correlations are linear in h[j] (except the diagonal which is quadratic), so the max is convex. Yes, max of convex functions is convex. So ternary search works.

But actually, the diagonal term h[j] * (1-h[j]) = h[j] - h[j]^2 appears in exactly one correlation (the zero-lag), and it's concave. So the zero-lag correlation as a function of h[j] is:
c[n-1] = Σ_i h[i](1-h[i]) which includes the term h[j](1-h[j]) = h[j] - h[j]^2

So the efficient coordinate descent is:
1. Precompute the "coefficient" for each (j, k) pair: a[j,k] = (1-h[j+s_k]) - h[j-s_k] (or the diagonal version)
2. For changing h[j], the new max = max(unaffected_max, max over affected k of (c[k] + δ*a[j,k]))
3. Find δ that minimizes this max

The "unaffected max" is the max of c[k] for k NOT in the affected range.

            if tot < 1e-15: break

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0/n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def fourier_perturb(h, n, n_modes, rng, deadline=None):
    """Perturb in Fourier domain, then optimize."""
    x = np.arange(n)
    h_c = h - h.mean()
    coef = np.zeros(2 * n_modes)
    for k in range(n_modes):
        coef[2*k] = (2.0/n) * np.sum(h_c * np.cos(2*np.pi*(k+1)*x/n))
        coef[2*k+1] = (2.0/n) * np.sum(h_c * np.sin(2*np.pi*(k+1)*x/n))

    h_new = project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

def fourier_perturb(h, n, n_modes, rng):
    x = np.arange(n)
    hm = h.mean()
    hc = h - hm
    h_new = np.full(n, hm)
    for k in range(n_modes):
        fk = k + 1
        ck = (2.0 / n) * np.sum(hc * np.cos(2 * np.pi * fk * x / n))
        dk = (2.0 / n) * np.sum(hc * np.sin(2 * np.pi * fk * x / n))
        cp = ck + rng.normal(0, 0.03)
        dp = dk + rng.normal(0, 0.03)
        h_new += cp * np.cos(2 * np.pi * fk * x / n) + dp * np.sin(2 * np.pi * fk * x / n)
    return project_feasible(h_new, n)
