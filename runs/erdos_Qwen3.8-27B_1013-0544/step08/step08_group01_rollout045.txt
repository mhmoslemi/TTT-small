import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sg_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += 1.0 - h[:n - lag]
            grad[:n - lag] -= h[lag:]
    else:
        kk = -lag
        if kk < n:
            grad[:n - kk] += 1.0 - h[kk:]
            grad[kk:] -= h[:n - kk]
    return grad

def nesterov_subgrad(h, n, max_iter=20000, deadline=None, multi_lag=5):
    h = h.copy()
    dx = 2.0 / n
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(c)[-multi_lag:]
        grad = np.zeros(n)
        for ki in top_idx:
            lag = ki - (n - 1)
            grad += sg_at_lag(h, n, lag)
        grad /= multi_lag
        y_new = project_feasible(y - step * grad * dx, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_new = project_feasible(y_new + ((t - 1.0) / t_new) * (y_new - y), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                step *= 0.8
            if stall > 2000:
                break
        y = y_new
        h = h_new
        t = t_new
    return best_h, best_c

def smooth_obj_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_obj_power(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_pos = np.maximum(c, 1e-30)
    val = (np.sum(c_pos ** p) ** (1.0 / p)) * (2.0 / n)
    return val

def sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_opt(h, n, alphas, method='lse', maxiter=600):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
        try:
            if method == 'lse':
                res = minimize(smooth_obj_lse, best_h, args=(n, alpha), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-16})
            else:
                res = minimize(smooth_obj_power, best_h, args=(n, int(alpha)), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=60):
    from collections import defaultdict
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            vf = (jf_arr >= 0) & (jf_arr < n)
            vb = (jb_arr >= 0) & (jb_arr < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf_arr[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb_arr[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.empty(grid_size)
            for di, d in enumerate(d_grid):
                nv = c_aff + d * a_coeff
                if diag_local is not None:
                    nv[diag_local] -= d * d
                vals[di] = max(float(nv.max()), unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(lambda d: max(float((c_aff + d * (a_coeff - (np.arange(len(ms)) == diag_local) * d)).max()), unaff_max) if diag_local is not None else max(float((c_aff + d * a_coeff).max()), unaff_max), gl, gh, maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def swap_refine(h, n, max_sweeps=50, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        improved = False
        trial = h.copy()
        for i in range(0, n, max(1, n // 200)):
            if deadline is not None and time.time() > deadline:
                break
            for j in range(i + 1, n, max(1, n // 100)):
                trial[i], trial[j] = trial[j], trial[i]
                c_trial = float(np.correlate(trial, 1.0 - trial, mode='full')[k_star])
                if c_trial < c[k_star] - 1e-10:
                    h = trial.copy()
                    c = np.correlate(h, 1.0 - h, mode='full')
                    k_star = int(np.argmax(c))
                    improved = True
                    if compute_c5(h, n) < best_c:
                        best_c = compute_c5(h, n)
                else:
                    trial[i], trial[j] = trial[j], trial[i]
        if not improved:
            break
    return h, best_c
