import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_eval_grid(c0, A, n, grid):
    """Evaluate g(t) = max(c0 + t*A - t^2*D) for t in grid. D has 1 at n-1."""
    Nm = 2 * n - 1
    lin = c0[None, :] + grid[:, None] * A[None, :]
    lin[:, n-1] -= grid**2
    return lin.max(axis=1)

def cd_golden(c0, A, n, a, b, iters=30):
    """Golden section search for min of g(t) on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    vc1 = c0 + c1 * A
    vc1[n-1] -= c1 * c1
    fc1 = float(vc1.max())
    vd1 = c0 + d1 * A
    vd1[n-1] -= d1 * d1
    fd1 = float(vd1.max())
    for _ in range(iters):
        if fc1 < fd1:
            b = d1
            d1 = c1
            fd1 = fc1
            c1 = b - gr * (b - a)
            vc1 = c0 + c1 * A
            vc1[n-1] -= c1 * c1
            fc1 = float(vc1.max())
        else:
            a = c1
            c1 = d1
            fc1 = fd1
            d1 = a + gr * (b - a)
            vd1 = c0 + d1 * A
            vd1[n-1] -= d1 * d1
            fd1 = float(vd1.max())
    t = (a + b) * 0.5
    vt = c0 + t * A
    vt[n-1] -= t * t
    return t, float(vt.max())

def exact_cd(h, n, max_passes, deadline, grid_pts=25, gs_iters=30):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    Nm = 2 * n - 1
    grid = np.linspace(0.0, 1.0, grid_pts)
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = False
        
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            
            # Compute A
            A = np.zeros(Nm)
            half = 2 * p + 1
            A[n-1-p : n+p] += (1.0 - h[:half])
            A[p : n+p] += -h[::-1]
            A[n-1] = 1.0
            
            # c0
            c0 = c - h0 * A
            c0[n-1] += h0 * h0
            
            # Grid evaluation
            gvals = cd_eval_grid(c0, A, n, grid)
            best_idx = int(gvals.argmin())
            best_g = float(gvals[best_idx])
            
            if best_g < current_max - 1e-13:
                # Refine with golden section
                lo = grid[max(0, best_idx - 1)]
                hi = grid[min(grid_pts - 1, best_idx + 1)]
                if hi - lo > 1e-10:
                    t_star, g_star = cd_golden(c0, A, n, lo, hi, gs_iters)
                    if g_star < best_g:
                        best_t, best_g = t_star, g_star
                    else:
                        best_t = grid[best_idx]
                else:
                    best_t = grid[best_idx]
                
                # Check boundaries
                g0 = float(c0.max())
                g1_c = c0 + A
                g1_c[n-1] = c0[n-1]  # at t=1, the n-1 term is c0[n-1]
                g1 = float(g1_c.max())
                
                if g0 < best_g:
                    best_t, best_g = 0.0, g0
                if g1 < best_g:
                    best_t, best_g = 1.0, g1
                
                # Apply
                c = c0 + best_t * A
                c[n-1] -= best_t * best_t
                h[p] = best_t
                current_max = float(c.max())
                improved = True
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        
        if not improved:
            break
    
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pnorm_slsqp(h, n, p_list, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.sum(c**p)) ** (1.0/p) * (2.0/n)
        
        def sum_con(x):
            return x.sum() - n/2.0
        
        def sum_con_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)]*n
        cons = [{'type':'eq','fun':sum_con,'jac':sum_con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter':maxiter_each,'ftol':1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
