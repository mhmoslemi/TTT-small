import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c * dx
    a_shifted = a - a.max()
    lse = a.max() + np.log(np.sum(np.exp(a_shifted)))
    s = x.sum() - n / 2.0
    return lse / alpha + lam * s * s

def smooth_max_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c * dx
    a_shifted = a - a.max()
    w = np.exp(a_shifted) / np.sum(np.exp(a_shifted))

    N = n

    corr2 = np.correlate(w, x, mode='full')
    grad = alpha * dx * (corr1[N-1::-1] - corr2[N-1::-1])
    s = x.sum() - N / 2.0

So the gradient should push h[i] to decrease where the increase would make c_k larger, and push h[i] to increase where the increase would make c_k smaller. This makes sense intuitively.

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c * dx
    a_max = a.max()
    lse = a_max + np.log(np.sum(np.exp(a - a_max)))
    s = x.sum() - n / 2.0
    return lse + lam * s * s

def lse_grad(x, n, alpha, lam):
    dx = 2.0 / n
    N = n
    c = np.correlate(x, 1.0 - x, mode='full')
    a = alpha * c * dx
    a_max = a.max()
    w = np.exp(a - a_max)
    w_sum = w.sum()
    w = w / w_sum
    corr1 = np.correlate(w[::-1], 1.0 - x, mode='full')
    corr2 = np.correlate(w, x, mode='full')
    grad = alpha * dx * (corr1[N-1::-1] - corr2[N-1::-1])
    s = x.sum() - N / 2.0
    grad += 2.0 * lam * s * np.ones(N)
    return grad

def lbfgs_optimize(h, n, alphas, lam=50.0, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline - 30: break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha, lam), method='L-BFGS-B',
                           jac=lse_grad, bounds=bounds,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except:
            pass
    return best_h, best_c

    h = h.copy()
    N = n

        if deadline and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')

        improved = False
        for p in range(N):
            if deadline and time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue

            # c[k] = sum_j h[j+k-(N-1)] * (1-h[j])
            # h[p] appears in:
            #   Forward: j+k-(N-1) = p => k = p - j + N - 1, for j in [0,N-1]
            #   => k in [p, p+N-1] (j from N-1 to 0)
            #   Backward: j = p => k = j - (N-1) + ... wait
            # Let me use: c[k_idx] = sum_j h[j + k_idx - (N-1)] * (1-h[j])
            # h[p] appears as h[j + k_idx - (N-1)] = h[p] when j = p - k_idx + (N-1)
            # Or h[p] appears as (1-h[j]) = 1-h[p] when j = p.

            #   => k_idx = p - j + (N-1), j in [0,N-1] => k_idx in [p, p+N-1]
            #   Contribution to c[k_idx]: (1-h[p - k_idx + (N-1)]) = (1-h[j])
            # 
            # Term 2 (h[p] in second factor): j = p, so k_idx can be anything
            #   => c[k_idx] includes h[p+k_idx-(N-1)] * (1-h[p]) for valid p+k_idx-(N-1)
            #   => k_idx in [N-1-p, 2N-2-p]
            
            k1_lo = max(0, p)
            k1_hi = min(2*N-2, p + N - 1)
            k2_lo = max(0, N - 1 - p)
            k2_hi = min(2*N-2, 2*N - 2 - p)
            
            k_lo = min(k1_lo, k2_lo)
            k_hi = max(k1_hi, k2_hi)

            # Actually, let's just compute the objective for a trial h[p] = h0 + d
            # c_new[k] = c[k] + delta_c[k] * d where delta_c[k] = dc[k]/dh[p]
            
            # dc[k_idx]/dh[p]:
            # From term 1: (1-h[p]) if k_idx in [p, p+N-1] (i.e., j = p-k_idx+N-1 in [0,N-1])
            # From term 2: -h[p+k_idx-(N-1)] if k_idx in [N-1-p, 2N-2-p]
            # So dc/dh[p] = (1-h[p]) * [p <= k_idx <= p+N-1] - h[p+k_idx-(N-1)] * [N-1-p <= k_idx <= 2N-2-p]
            
            k_range = np.arange(k_lo, k_hi + 1)
            dc = np.zeros(len(k_range))
            valid1 = (k_range >= p) & (k_range <= p + N - 1)
            dc[valid1] += (1.0 - h0)
            valid2 = (k_range >= N - 1 - p) & (k_range <= 2*N - 2 - p)
            j2 = p + k_range[valid2] - (N - 1)
            dc[valid2] -= h[j2]
            
            c_aff = c[k_range]
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(c[:k_lo].max()))
            if k_hi < 2*N - 2:
                unaff_max = max(unaff_max, float(c[k_hi+1:].max()))
            
            # c_new = c_aff + dc * d
            # Want to minimize max(c_new, unaff_max)
            # c_new[k] = c_aff[k] + dc[k] * d
            # This is a convex piecewise-linear function of d (max of linear functions)
            # Minimum is where the max is minimized
            
            # Grid search + golden section
            d_lo = -h0
            d_hi = 1.0 - h0
            grid = np.linspace(d_lo, d_hi, 50)
            best_d = 0.0
            best_val = current_max
            for d in grid:
                c_new = c_aff + dc * d
                val = max(float(c_new.max()), unaff_max)
                if val < best_val:
                    best_val = val
                    best_d = d

                improved = True
                current_max = min(current_max, best_val)

            break
        h = project_feasible(h, N)
    return h, compute_c5(h, N)
