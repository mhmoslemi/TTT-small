    # c[m] = sum_i h[i] * (1 - h[i + m - (n-1)])
    # Let s = m - (n-1) = -(n-1-m)
    # c[m] = sum_i h[i] * (1 - h[i+s]) where i+s must be in [0,n)

    # dc[m]/dh[i] = (1 - h[i+s]) if 0 <= i+s < n, else 0
    #              - h[i-s] if 0 <= i-s < n, else 0
    # (The second term comes from the -h[i+s] part when we vary h[i] in the second factor)
    # Wait no. c[m] = sum_j h[j] * (1 - h[j+s])
    # dc[m]/dh[i] = (1-h[i+s]) * 1[i+s valid] + (-h[i-s]) * 1[i-s valid]
    
    center = n - 1
    g = np.zeros(n)
    # Vectorized approach:
    # For each s in range(-(n-1), n):
    #   weight w[center + s] (since m = center + s... wait)
    #   Actually m = 0, 1, ..., 2n-2, and s = center - m = (n-1) - m
    #   So m = center - s, and s ranges from center down to -(n-1)
    #   i.e., s ranges from -(n-1) to center = n-1
    
    # Let me just do it with the loop but make it fast
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n-r])
            g[:n-r] -= wm * h[r:]
    return (2.0 / n) * g

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0.0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0.0:
            h += (diff / tot) * cap
    elif diff < 0.0:
        tot = float(h.sum())
        if tot > 0.0:
            h *= (target / tot)
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        idx = np.where(h < 1.0 if res > 0 else h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * c
    zm = z.max()
    w = np.exp(z - zm)
    w /= w.sum()
    center = n - 1
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n-r])
            g[:n-r] -= wm * h[r:]
    return (2.0 / n) * g

def fourier_basis(n, K):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(np.pi * k * x)
    return B

def h_from_coeffs(coeffs, B, n):
    return project_feasible(0.5 + B @ coeffs, n)

def adam_fourier(n, K, coeffs_init, deadline, lr=0.008, max_steps=3000):
    B = fourier_basis(n, K)
    coeffs = np.array(coeffs_init, dtype=float)
    nc = 2 * K
    mv = np.zeros(nc)
    vv = np.zeros(nc)
    t = 0
    best_h = h_from_coeffs(coeffs, B, n)
    best_c = compute_c(best_h, n)
    best_coeffs = coeffs.copy()
    stuck = 0
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        h = h_from_coeffs(coeffs, B, n)
        alpha = 80 + 40 * min(t // 80, 20)
        g_h = smooth_grad(h, n, alpha)
        g_c = B.T @ g_h
        b1, b2, eps = 0.85, 0.999, 1e-8
        mv = b1 * mv + (1 - b1) * g_c
        vv = b2 * vv + (1 - b2) * g_c ** 2
        coeffs = coeffs - lr * (mv / (1 - b1**t)) / (np.sqrt(vv / (1 - b2**t)) + eps)
        coeffs = np.clip(coeffs, -0.42, 0.42)
        h_new = h_from_coeffs(coeffs, B, n)
        c_new = compute_c(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
            best_coeffs = coeffs.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 80:
                lr *= 0.85
                stuck = 0
            if lr < 2e-6:
                break
    return best_h, best_c

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            def obj(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                z = alpha * c
                zm = z.max()
                return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)
            def jac(x):
                return smooth_grad(x, n, alpha)
            res = minimize(obj, best_h, method='SLSQP', jac=jac, bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def binary_from_h(h):
    n = len(h)
    idx = np.argpartition(h, n - n // 2)[n - n // 2:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def binary_local_search(h, n, deadline, max_rounds=200):
    h = binary_from_h(h)
    c = compute_c(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(0)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n // 4):
            if time.time() > deadline:
                break
            i = int(rng.choice(ones.size))
            j = int(rng.choice(zeros.size))
            oi, oj = ones[i], zeros[j]
            h[oi] = 0.0
            h[oj] = 1.0
            nc = compute_c(h, n)
            if nc < c - 1e-14:
                c = nc
                ones[i] = oj
                zeros[j] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, c

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def coord_descent(h, n, max_passes, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s_arr
            i = p - s_arr
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off > 0:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi+1:].max()))
            # 1D minimization via golden section + bisection on max
            best_x = hp
            best_f = float(np.max(cseg + a * (hp - hp)))
            best_f = max(best_f, unaff, qconst)
            # Scan a grid then refine
            xg = np.linspace(0.0, 1.0, 21)
            for xv in xg:
                f = max(float(np.max(cseg + a * (xv - hp))), unaff, qconst + xv*(1.0-xv))
                if f < best_f:
                    best_f = f
                    best_x = xv
            # Refine around best
            l = max(0.0, best_x - 0.05)
            r = min(1.0, best_x + 0.05)
            xg2 = np.linspace(l, r, 21)
            for xv in xg2:
                f = max(float(np.max(cseg + a[off:] * (xv-hp) if hi>center else 0.0 if seg==1 else np.max(cseg) if False else (cseg + a*(xv-hp)).max())), unaff, qconst + xv*(1.0-xv))
                if f < best_f:
                    best_f = f
                    best_x = xv
            if best_f < cur - 1e-11:
                dxv = best_x - hp
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center+1:hi+1] += asg[off:] * dxv
                c[center] = qconst + best_x * (1.0 - best_x)
                h[p] = best_x
                cur = best_f
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-13:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
