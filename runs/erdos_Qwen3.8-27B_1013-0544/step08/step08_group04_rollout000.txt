            grad[r:] -= h[:n - r]
    return grad

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def coord_minimize(hp, cseg, a, qconst, unaff):
    tol = 1e-10
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, unaff, qconst, qcur, cm) + 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    pc, nc = a_pos.size, a_neg.size
    for _ in range(22):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh:
                    lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    if g < 0.0:
                        g = 0.0
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    if max(0.0, ll) <= min(r1, lh) + tol:
                        feas = True
                    else:
                        feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas:
            high = mid
        else:
            low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll, lh = 0.0, 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll:
                ll = v
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh:
                lh = v
        if ll <= 1.0 + tol and lh >= -tol:
            g = t - qconst
            if g > 0.25:
                branches = [(max(0.0, ll), min(1.0, lh))]
            elif g >= -1e-12:
                if g < 0.0:
                    g = 0.0
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                branches = [(0.0, r1), (r2, 1.0)]
            else:
                branches = []
            for Lq, Uq in branches:
                L = max(Lq, ll, 0.0)
                U = min(Uq, lh, 1.0)
                if L <= U + tol:
                    cands.extend([L, U, 0.5 * (L + U)])
    best_x = hp
    best_f = max(unaff, qconst + hp * (1.0 - hp), float(np.max(cseg)) if m else 0.0)
    for x in cands:
        if x < 0.0 or x > 1.0:
            continue
        val = max(unaff, qconst + x * (1.0 - x))
        if m:
            val2 = float(np.max(cseg + a * (x - hp)))
            val = max(val, val2)
        if val < best_f - 1e-13:
            best_f = val
            best_x = x
    return best_x, best_f

def cd_pass(h, n, c, max_rounds, deadline, project_every=8):
    center = n - 1
    L = 2 * n - 1
    cur = float(c.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s_arr
            i = p - s_arr
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = coord_minimize(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = compute_corr(h, n)
                cur = float(c.max())
        h = project_feasible(h, n)
        c = compute_corr(h, n)
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def sg_step(h, n, c, lr, n_top=5):
    center = n - 1
    c_max = float(c.max())
    threshold = c_max * 0.92
    top_lags = np.where(c >= threshold)[0]
    if top_lags.size == 0:
        top_lags = np.array([int(np.argmax(c))])
    weights = c[top_lags]
    weights = weights / max(weights.sum(), 1e-30)
    grad = np.zeros(n, dtype=float)
    for w, m in zip(weights, top_lags):
        s = int(m) - center
        grad += w * sg_grad(h, n, s)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h
    return project_feasible(h - lr * grad / gnorm, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        x = (np.arange(500) + 0.5) * 0.004
        h_init = project_feasible(1.0 - np.abs(x - 1.0), 500)
    n_init = len(h_init)
    if n_init % 2:
        n_init += 2
        h_init = resample_h(h_init, n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = float(compute_corr(best_h, n_init).max() * 2.0 / n_init)

    # Phase 1: Aggressive coordinate descent at initial resolution
    c = compute_corr(best_h, n_init)
    best_h, best_c = cd_pass(best_h, n_init, c, max_rounds=12, deadline=deadline, project_every=6)

    # Phase 2: Subgradient + CD交替 to break plateaus
    for it in range(60):
        if time.time() > deadline - 180:
            break
        if best_c <= 0.38075:
            break
        c = compute_corr(best_h, n_init)
        # Subgradient step
        lr = 0.008 * (1.0 + 0.3 * np.sin(0.1 * it))
        h_sg = sg_step(best_h, n_init, c, lr, n_top=7)
        c_sg = float(compute_corr(h_sg, n_init).max() * 2.0 / n_init)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg
        # CD pass on the better of the two
        c_cur = compute_corr(best_h, n_init)
        h_cd, c_cd = cd_pass(best_h, n_init, c_cur, max_rounds=3, deadline=deadline, project_every=8)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 3: Multi-resolution search
    for nt in [n_init, 400, 550, 350, 450, 600]:
        if time.time() > deadline - 150:
            break
        if best_c <= 0.38075:
            break
        h0 = resample_h(best_h, nt)
        c0 = compute_corr(h0, nt)
        h1, c1 = cd_pass(h0, nt, c0, max_rounds=6, deadline=deadline, project_every=6)
        # Add a few SG steps
        for _ in range(15):
            if time.time() > deadline - 140:
                break
            c_t = compute_corr(h1, nt)
            h_t = sg_step(h1, nt, c_t, 0.006, n_top=5)
            c_t2 = compute_corr(h_t, nt)
            h1b, c1b = cd_pass(h_t, nt, c_t2, max_rounds=2, deadline=deadline, project_every=10)
            if c1b < c1:
                h1, c1 = h1b, c1b
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 4: Perturbation restarts
    for r in range(20):
        if time.time() > deadline - 80:
            break
        if best_c <= 0.38072:
            break
        nt = len(best_h)
        base = resample_h(best_h, nt)
        typ = r % 4
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0.0, 0.015, nt), nt)
        elif typ == 1:
            h0 = project_feasible(base + 0.05 * rng.randn(nt), nt)
        elif typ == 2:
            # Sinusoidal perturbation
            x = (np.arange(nt) + 0.5) * 2.0 / nt
            h0 = project_feasible(base + 0.04 * np.sin(3.7 * np.pi * x + rng.uniform(0, 6.28)), nt)
        else:
            # Local perturbation on worst lag region
            h0 = base.copy()
            idx = rng.choice(nt, nt // 5, replace=False)
            h0[idx] += rng.normal(0.0, 0.03, idx.size)
            h0 = project_feasible(h0, nt)
        c0 = compute_corr(h0, nt)
        h1, c1 = cd_pass(h0, nt, c0, max_rounds=4, deadline=deadline, project_every=8)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 5: Final polish
    if time.time() < deadline - 20:
        c = compute_corr(best_h, len(best_h))
        best_h, best_c = cd_pass(best_h, len(best_h), c, max_rounds=10, deadline=deadline, project_every=4)

    best_h = project_feasible(best_h, len(best_h))
    best_c = float(compute_corr(best_h, len(best_h)).max() * 2.0 / len(best_h))
    return best_h, best_c, len(best_h)
