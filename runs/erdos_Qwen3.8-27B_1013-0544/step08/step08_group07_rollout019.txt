        idxs = np.arange(n - 1 + 1, n - 1 + m_f + 1)
        a[idxs] += (1.0 - h[p + 1 : p + 1 + m_f])
    m_b = min(p, n - 1)
    if m_b > 0:
        idxs = np.arange(n - 1 - m_b, n - 1)
        a[idxs] -= h[p - 1 : p - 1 - m_b : -1]
    a[n - 1] += (1.0 - 2.0 * h[p])
    return a

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            a = get_coefficients(p, h, n)
            affected = np.nonzero(a)[0]
            if len(affected) == 0:
                continue
            unaff = np.ones(2*n-1, dtype=bool)
            unaff[affected] = False
            unaff_max = float(c_prev[unaff].max()) if unaff.any() else 0.0
            c_aff = c_prev[affected]
            a_aff = a[affected]
            self_in = (affected == n - 1)
            best_d = 0.0
            best_val = current_max
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            for d in d_grid:
                if abs(d) < 1e-15:
                    continue
                nv = c_aff + d * a_aff
                if self_in.any():
                    nv[self_in] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max - 1e-16:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[affected] = c_aff + best_d * a_aff
                c_prev[n-1] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def fourier_eval(coeffs, x):
    h = np.full(len(x), coeffs[0] * 0.5)
    max_k = (len(coeffs) - 1) // 2
    for k in range(1, max_k + 1):
        if 2*k < len(coeffs):
            h += coeffs[2*k] * np.cos(np.pi * k * x)
        if 2*k+1 < len(coeffs):
            h += coeffs[2*k+1] * np.sin(np.pi * k * x)
    return h

def fourier_fit(h, x, n_modes):
    n = len(h)
    coeffs = np.zeros(2 * n_modes + 1)
    coeffs[0] = 2.0 * h.mean()
    for k in range(1, n_modes + 1):
        coeffs[2*k] = (2.0 / n) * np.dot(h, np.cos(np.pi * k * x))
        coeffs[2*k+1] = (2.0 / n) * np.dot(h, np.sin(np.pi * k * x))
    return coeffs

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_fun(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(len(x))

def lbfgs_run(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 15:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def fourier_phase(h, n, x, n_modes, alphas, deadline):
    coeffs = fourier_fit(h, x, n_modes)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def f(params):
            h_eval = np.clip(fourier_eval(params, x), 0, 1)
            h_eval = project_feasible(h_eval, n)
            return lse_obj(h_eval, n, alpha)
        res = minimize(f, coeffs, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
        new_h = project_feasible(np.clip(fourier_eval(res.x, x), 0, 1), n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h
            coeffs = res.x
    return best_h, best_c

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.35 * np.sin(np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x)
    elif kind == 2:
        h = (np.sin(np.pi * x) > 0).astype(float)
    elif kind == 3:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
    elif kind == 4:
        h = np.where(np.sin(np.pi * x) > 0.2, 0.7, 0.3)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        x0 = np.linspace(0, 2, n_init, endpoint=False)
        h_init = project_feasible(0.5 + 0.3 * np.sin(np.pi * x0), n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS on initial
    if time.time() < deadline - 60:
        h1, c1 = lbfgs_run(best_h, best_n, [1000, 5000, 50000], 500, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Fourier optimization
    if time.time() < deadline - 60:
        x = np.linspace(0, 2, best_n, endpoint=False)
        for nm in [5, 8, 12, 16]:
            if time.time() > deadline - 30:
                break
            hf, cf = fourier_phase(best_h, best_n, x, nm, [5000, 20000, 100000], deadline)
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 3: Coordinate descent
    if time.time() < deadline - 30:
        h2, c2 = coord_cd(best_h, best_n, 300, deadline, grid_size=200)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 4: Try smaller n values (faster to converge)
    for n_try in [200, 250, 300, 400]:
        if time.time() > deadline - 80:
            break
        if n_try == best_n:
            continue
        x_s = np.linspace(0, 2, n_try, endpoint=False)
        h_s = project_feasible(np.interp(x_s, np.linspace(0, 2, best_n, endpoint=False), best_h)
                               + 0.02 * rng.randn(n_try), n_try)
        h_s, c_s = lbfgs_run(h_s, n_try, [5000, 50000], 400, deadline)
        h_s, c_s = coord_cd(h_s, n_try, 200, deadline, grid_size=200)
        if c_s < best_c:
            best_h, best_c, best_n = h_s, c_s, n_try

    # Phase 5: Interpolate best back if we switched n
    if best_n != len(best_h):
        pass  # already consistent

    # Phase 6: Fourier + coord on best
    if time.time() < deadline - 40:
        x = np.linspace(0, 2, best_n, endpoint=False)
        for nm in [8, 12, 20]:
            if time.time() > deadline - 20:
                break
            hf, cf = fourier_phase(best_h, best_n, x, nm, [10000, 50000], deadline)
            if cf < best_c:
                best_h, best_c = hf, cf
        h2, c2 = coord_cd(best_h, best_n, 200, deadline, grid_size=300)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 7: Random restarts
    if time.time() < deadline - 60:
        n_r = best_n
        x_r = np.linspace(0, 2, n_r, endpoint=False)
        n_restarts = max(3, int((deadline - time.time() - 30) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r % 3 == 0:
                hp = structured_start(n_r, r % 5, rng)
            elif r % 3 == 1:
                coeffs = fourier_fit(best_h, x_r, 10)
                coeffs += rng.normal(0, 0.05, len(coeffs))
                hp = project_feasible(np.clip(fourier_eval(coeffs, x_r), 0, 1), n_r)
            else:
                hp = project_feasible(best_h + 0.03 * rng.randn(n_r), n_r)
            ho, co = lbfgs_run(hp, n_r, [10000, 50000], 300, deadline)
            ho, co = coord_cd(ho, n_r, 150, deadline, grid_size=200)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 8: Final polish with high-alpha L-BFGS + fine coord
    if time.time() < deadline - 15:
        hf, cf = lbfgs_run(best_h, best_n, [100000, 500000], 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 3:
        hf, cf = coord_cd(best_h, best_n, 200, deadline, grid_size=400)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
