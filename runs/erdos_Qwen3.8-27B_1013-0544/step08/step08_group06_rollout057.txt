def binary_local_search(h, n, deadline, max_swaps=10000):
    """For near-binary solutions, try swapping pairs."""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    c = np.correlate(h, 1.0-h, mode='full')
    current_c5 = float(c.max()) * dx
    
    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        # Find the worst lag
        k_star = int(np.argmax(c))
        
        # Try to find a swap that improves
        improved = False
        for _ in range(100):
            # Random pair swap: pick a 1-position and a 0-position
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            
            # Try swapping: h[i] -= delta, h[j] += delta (keeps sum constant)
            # For binary: h[i] was ~1, h[j] was ~0. After swap: h[i] ~0, h[j] ~1.
            delta = h[i] - h[j]  # amount to transfer
            h[i] -= delta
            h[j] += delta
            
            c_new = np.correlate(h, 1.0-h, mode='full')
            c5_new = float(c_new.max()) * dx
            
            if c5_new < current_c5 - 1e-12:
                current_c5 = c5_new
                c = c_new
                improved = True
                if current_c5 < best_c:
                    best_c = current_c5
                    best_h = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
        
        swaps += 1
        if not improved and swaps > 10:
            break
    
    h = project_feasible(h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)

    if budget_s > 400:
        h, c = slsqp_opt(h, n, [2000, 20000, 200000], t0 + 35, maxiter=300)
        if c < best_c:
            best_h, best_c = h, c

    h, c = coord_descent(best_h, n, 3000, t0 + 500, 90)
    if c < best_c:
        best_h, best_c = h, c

    if time.time() < deadline - 300
        h, c = active_set_step(best_h, n, time.time() + 80, steps=3000, lr0=0.003)
        if c < best_c:
            best_h, best_c = h, c

    for _ in range(5):
        if time.time() > deadline - 250:
            break
        hp = best_h + 0.003 * np.random.randn(n)
        hp = project_feasible(hp, n)
        h, c = coord_descent(hp, n, 400, time.time() + 60, 90)
        if c < best_c:
            best_h, best_c = h, c

    h, c = coord_descent(best_h, n, 3000, deadline - 80, 150)
    if c < best_c:
        best_h, best_c = h, c

    for nt in [350, 400, 550]:
        if time.time() > deadline - 25:
            break
        if nt == n:
            continue
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(xn, xo, best_h), nt)
        ho, co = coord_descent(hi, nt, 600, min(time.time() + 40, deadline - 5), 120)
        if co < best_c:
            best_h, best_c = ho, co

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
