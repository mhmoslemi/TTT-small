            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(c.max())
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            # Compute A and D
            A = np.zeros(2*n-1)
            D = np.zeros(2*n-1)
            # First role: k in [p, p+n-1] ∩ [0, 2n-2]
            klo = max(0, p)
            khi = min(2*n-2, p+n-1)
            if klo <= khi:
                k_arr = np.arange(klo, khi+1)
                idx = p + n - 1 - k_arr
                # Non-diagonal
                nd = k_arr != (n-1)
                if np.any(nd):
                    A[k_arr[nd]] += (1.0 - h[idx[nd]])
                # Diagonal
                if np.any(~nd):
                    A[n-1] += 1.0
            # Second role: k in [n-1-p, 2n-2-p] ∩ [0, 2n-2]
            klo = max(0, n-1-p)
            khi = min(2*n-2, 2*n-2-p)
            if klo <= khi:
                k_arr = np.arange(klo, khi+1)
                idx = p + k_arr - (n-1)
                nd = k_arr != (n-1)
                if np.any(nd):
                    A[k_arr[nd]] += (-h[idx[nd]])
            # D
            D[n-1] = 1.0
            
            # c_base for the optimization
            # c_new[k] = c[k] + (t-h0)*A[k] - (t²-h0²)*D[k]
            # = (c[k] - h0*A[k] + h0²*D[k]) + t*A[k] - t²*D[k]
            c_base = c - h0*A + h0*h0*D
            
            def g(t):
                v = c_base + t*A
                v[n-1] -= t*t  # subtract t²*D[n-1] = t²
                return float(v.max())
            
            # Golden section
            a, b = 0.0, 1.0
            c1 = b - gr*(b-a)
            d1 = a + gr*(b-a)
            fc1 = g(c1)
            fd1 = g(d1)
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr*(b-a); fc1 = g(c1)
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr*(b-a); fd1 = g(d1)
            t_star = (a+b)*0.5
            f_star = g(t_star)
            f_0 = float(c_base.max())
            f_1 = g(1.0)
            
            if f_0 <= f_star - 1e-13 and f_0 <= f_1:
                t_new = 0.0
                f_new = f_0
            elif f_1 <= f_star - 1e-13 and f_1 <= f_0:
                t_new = 1.0
                f_new = f_1
            else:
                t_new = t_star
                f_new = f_star
            
            if f_new < current_max - 1e-12:
                h[p] = t_new
                # Update c
                c = c_base + t_new*A
                c[n-1] -= t_new*t_new
                current_max = f_new
                improved = True
        
        h = project_feasible(h, n)
        # Recompute c after projection
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
        if not improved:
            break
    return h, compute_c5(h, n)

def lse_slsqp(h, n, tau, maxiter=500, deadline=None):
    """Log-sum-exp smoothed SLSQP."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        m = c.max()
        scaled = tau * (c - m)
        return m + np.log(np.sum(np.exp(scaled))) / tau
    
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        m = c.max()
        weights = np.exp(tau * (c - m))
        weights /= weights.sum()
        # grad of c[k] w.r.t. x[i]
        # This is complex; use numerical gradient or skip
        # For simplicity, use finite difference with a small step
        # Actually, let's just use SLSQP without gradient
        return None
    
    def sum_con(x):
        return x.sum() - n/2.0
    
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter,'ftol':1e-16})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except:
        pass
    return h
