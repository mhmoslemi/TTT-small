    c = np.correlate(h, 1.0 - h, mode='full')

    if len(active) <= 3:
        m = active[rng.randint(len(active))]
    else:
        # Weighted average of a few random active constraints
        m = active[rng.choice(len(active), size=min(5, len(active)), replace=False)]

    dx = 2.0 / n
    grad = np.zeros(n)

        for mi in m:
            for p in range(n):
                g = 0.0
                jf = p + mi - (n - 1)
                if 0 <= jf < n:
                    g += (1.0 - h[jf])
                jb = p - mi + (n - 1)
                if 0 <= jb < n:

                # For diagonal (mi == n-1), the term h[p]*(1-h[p]) has derivative (1-2h[p])

                grad[p] += g / len(m)

            if 0 <= jb < n:
                g -= h[jb]

def compute_gradient_at_shift(h, n, mi):
    """Compute ∇_h [correlate(h, 1-h)[mi]]."""
    dx = 2.0 / n
    grad = np.zeros(n)
    jf = np.arange(n) + mi - (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    if np.any(valid_f):
        grad[valid_f] += (1.0 - h[jf[valid_f]])
    jb = np.arange(n) - mi + (n - 1)
    valid_b = (jb >= 0) & (jb < n)
    if np.any(valid_b):
        grad[valid_b] -= h[jb[valid_b]]
    return grad
