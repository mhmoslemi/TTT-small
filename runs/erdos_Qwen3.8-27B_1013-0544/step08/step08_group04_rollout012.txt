```python
import numpy as np
import time

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add: distribute to entries with room
        room = 1.0 - h
        total_room = room.sum()
        if total_room > 0:
            h += diff / total_room * room
    else:
        # Need to subtract: distribute from entries with mass
        total = h.sum()
        if total > 0:
            h += diff / total * h
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0.0, 1.0)
    return h
```

    fv = np.maximum(np.maximum(vm,vq),unaff)
    i = int(np.argmin(fv))
    if fv[i] < best_f: best_f, best_x = float(fv[i]), float(xg[i])
    return best_x, best_f

def cd_optimize(h, n, deadline, max_passes=200):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0-h, mode='full')
    center = n-1; L = 2*n-1
    cur = c.max()
    best_c = cur*(2.0/n); best_h = h.copy()
    for pp in range(max_passes):
        if time.time() > deadline: break
        imp = 0.0
        order = range(n) if pp%2==0 else range(n-1,-1,-1)
        for p in order:
            hp = h[p]
            lo = min(p, center-p); hi = max(center+p, 2*center-p)
            off = center - lo
            s = center - np.arange(lo, hi+1)
            j = p+s; i = p-s
            a = np.zeros(hi-lo+1)
            vj = (j>=0)&(j<n); vi = (i>=0)&(i<n)
            if vj.any(): a[vj] += 1.0-h[j[vj]]
            if vi.any(): a[vi] -= h[i[vi]]
            cseg = c[lo:hi+1].copy()
            qconst = c[center] - hp*(1.0-hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi+1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = _coord_opt(hp, cseg, a, qconst, unaff)
            if bf < cur - 1e-12:
                dx = bx - hp
                if off > 0: c[lo:center] += a[:off]*dx
                if off+1 < hi-lo+1: c[center+1:hi+1] += a[off+1:]*dx
                c[center] = qconst + bx*(1.0-bx)
                h[p] = bx; imp += cur-bf; cur = bf
        if imp < 1e-13 and pp > 3: break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full')
        cur = float(c.max())
        cv = cur*(2.0/n)
        if cv < best_c: best_c = cv; best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
