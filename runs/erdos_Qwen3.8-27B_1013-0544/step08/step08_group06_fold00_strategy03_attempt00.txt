import numpy as np
from scipy.optimize import minimize

dx = 2.0 / 800

def build_scale_vector(n, γ):
    t = np.linspace(0, 2, n)
    u = np.sin(np.pi * t) * (-1.5 if γ<0.3 else 1.2)
    return 0.5 + γ + u, 0.5 - γ + u

def enforce_constraints(y, n):
    y = np.clip(y, 0, 1)
    if np.abs(sum(y) * dx - 1.0) > 1e-5:
        y = y * (1.0 / (sum(y) * dx))
    return y

def compute_C₅(h):
    return (np.max(np.correlate(h, 1 - h, mode='full')) * dx).item()

def main():
    n = 800
    best_C = None
    best_h = None
    
    for γ in np.arange(0.3, 0.5, 0.02):
        h_init, _ = build_scale_vector(n, γ)
        h_init = enforce_constraints(h_init, n)
        res = minimize(compute_C₅, h_init, method='L-BFGS-B', bounds=[(0,1)]*n)
        if best_C is None or res.fun < best_C:
            best_C = res.fun
            best_h = res.x
            
    best_h = enforce_constraints(best_h, n)
    np.savetxt('erdos_soln.txt', best_h)

if __name__ == '__main__':
    main()
