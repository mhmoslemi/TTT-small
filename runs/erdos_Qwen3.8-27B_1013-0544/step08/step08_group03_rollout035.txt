def project_feasible(h, n):

    grad = np.zeros(n)

        g = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=500, deadline=None):
    h = h.copy()
    n_corr = 2 * n - 1
    center = n - 1
    D = np.zeros(n_corr)
    D[center] = 1.0
    gold_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        current_max = float(c.max())
        improved = False
        n_improved = 0
        
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            
            # Compute A
            A = np.zeros(n_corr)
            m_lo = max(0, center - p)
            m_hi = min(n_corr - 1, center - p + n - 1)
            if m_lo <= m_hi:
                A[m_lo:m_hi+1] += (1.0 - h[p + np.arange(m_lo, m_hi+1) - center])
            m_lo2 = max(0, p + center - n + 1)
            m_hi2 = min(n_corr - 1, p + center)
            if m_lo2 <= m_hi2:
                A[m_lo2:m_hi2+1] += (-h[p + center - np.arange(m_lo2, m_hi2+1)])
            
            # Minimize f(s) = max(c + s*A - s^2*D) over s in [-h0, 1-h0]
            s_lo, s_hi = -h0, 1.0 - h0
            if s_hi - s_lo < 1e-14:
                continue
            
            # Coarse grid
            grid_s = np.linspace(s_lo, s_hi, 9)
            best_s = grid_s[0]
            best_f = np.inf
            for s in grid_s:
                f = np.max(c + s * A - s * s * D)
                if f < best_f:
                    best_f = f
                    best_s = s
            
            # Golden section around best
            idx = int(np.argmin([np.max(c + s*A - s*s*D) for s in grid_s]))
            lo = grid_s[max(0, idx-1)]
            hi = grid_s[min(8, idx+1)]
            
            a, b = lo, hi
            c1 = b - gold_ratio*(b-a)
            d1 = a + gold_ratio*(b-a)
            fc1 = np.max(c + c1*A - c1*c1*D)
            fd1 = np.max(c + d1*A - d1*d1*D)
            for _ in range(15):
                if fc1 < fd1:
                    b, d1, fd1 = d1, c1, fc1
                    c1 = b - gold_ratio*(b-a)
                    fc1 = np.max(c + c1*A - c1*c1*D)
                else:
                    a, c1, fc1 = c1, d1, fd1
                    d1 = a + gold_ratio*(b-a)
                    fd1 = np.max(c + d1*A - d1*d1*D)
            s_gs = (a+b)*0.5
            f_gs = np.max(c + s_gs*A - s_gs*s_gs*D)
            
            # Compare with boundaries and grid best
            f_bound_lo = float(current_max)  # s=0
            f_bound_hi = np.max(c + s_hi*A - s_hi*s_hi*D)
            
            candidates = [(f_bound_lo, 0.0), (f_bound_hi, s_hi), (float(f_gs), s_gs)]
            best_f, best_s = min(candidates, key=lambda x: x[0])
            
            if best_f < current_max - 1e-14:
                h[p] = h0 + best_s
                c += best_s * A - best_s * best_s * D
                current_max = float(c.max())
                improved = True
                n_improved += 1
        
        h = project_feasible(h, n)
        if not improved:
            break
        if n_improved < 3:
            # Near convergence, do a few more passes then stop
            if _pass > 50:
                break
    
    return h, compute_c5(h, n)

def softmax_obj_grad(x, tau, n):
    """Compute soft-max objective and its gradient."""
    dx = 2.0 / n
    n_corr = 2*n - 1
    center = n - 1
    c = np.correlate(x, 1.0 - x, mode='full')
    val = c * dx
    scaled = tau * val
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    Z = np.sum(exps)
    w = exps / Z  # softmax weights
    obj = (mx + np.log(Z)) / tau
    
    # Gradient
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        # First factor
        m_lo = max(0, center - p)
        m_hi = min(n_corr - 1, center - p + n - 1)
        if m_lo <= m_hi:
            g += np.dot(w[m_lo:m_hi+1], 1.0 - x[p + np.arange(m_lo, m_hi+1) - center])
        # Second factor
        m_lo2 = max(0, p + center - n + 1)
        m_hi2 = min(n_corr - 1, p + center)
        if m_lo2 <= m_hi2:
            g -= np.dot(w[m_lo2:m_hi2+1], x[p + center - np.arange(m_lo2, m_hi2+1)])
        grad[p] = dx * g
    return obj, grad

def softmax_slsqp(h, n, tau, maxiter=400, deadline=None):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    def obj(x):
        o, _ = softmax_obj_grad(x, tau, n)
        return o
    
    def grad(x):
        _, g = softmax_obj_grad(x, tau, n)
        return g
    
    def sum_con(x):
        return x.sum() - n/2.0
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_con_jac}]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', jac=grad, bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter,'ftol':1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def spectral_restart(h, n, rng, freq_list, amplitude=0.1):
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for f in freq_list:
        pert += amplitude * (np.cos(2*np.pi*f*x/2.0) + np.sin(2*np.pi*f*x/2.0)) * rng.uniform(0.5, 1.5)
    return project_feasible(h + pert, n)

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=500, deadline=None):
    h = h.copy()
    n_corr = 2 * n - 1
    center = n - 1
    D = np.zeros(n_corr)
    D[center] = 1.0

    for _pass in range(max_passes):

        current_max = float(c.max())
        improved = False
        n_imp = 0

            if deadline is not None and time.time() > deadline:

            m_hi = min(n_corr - 1, center - p + n - 1)

            m_lo2 = max(0, p + center - n + 1)

def softmax_slsqp(h, n, tau, maxiter=300, deadline=None):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    center = n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        val = c * dx
        sc = tau * val
        mx = np.max(sc)
        ex = np.exp(sc - mx)
        Z = np.sum(ex)
        w = ex / Z
        o = (mx + np.log(Z)) / tau
        grad = np.zeros(n)
        for p in range(n):
            g = 0.0
            m_lo = max(0, center - p)
            m_hi = min(n_corr - 1, center - p + n - 1)
            if m_lo <= m_hi:
                g += np.dot(w[m_lo:m_hi+1], 1.0 - x[p + np.arange(m_lo, m_hi+1) - center])
            m_lo2 = max(0, p + center - n + 1)
            m_hi2 = min(n_corr - 1, p + center)
            if m_lo2 <= m_hi2:
                g -= np.dot(w[m_lo2:m_hi2+1], x[p + center - np.arange(m_lo2, m_hi2+1)])
            grad[p] = dx * g
        return o, grad
    def obj(x):
        return obj_grad(x)[0]
    def jac(x):
        return obj_grad(x)[1]
    def sc_(x):
        return x.sum() - n / 2.0
    def scj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_, 'jac': scj}]
    try:
        res = minimize(obj, best_h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

        pert += amp * rng.uniform(0.3, 1.0) * np.sin(np.pi * k * x + rng.uniform(0, 2*np.pi))
    return project_feasible(h + pert, n)
