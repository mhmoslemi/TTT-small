def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * max(1, n):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, min((n - 1) - p, p))
    m_hi = min(2 * n - 2, max((n - 1) + p, p + n - 1))
    if m_lo > m_hi:
        return h0
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx[0]] = 0.0
        c_diag = c_base[diag_idx[0]]
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    cur_max = float(c.max())
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-15:
            break
        L = 0.5 * (L_lo + L_hi)
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if ok and np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-15:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) * 0.5
                r2 = (1.0 + np.sqrt(disc)) * 0.5
                if x_lo > r1 and x_hi < r2:
                    ok = False
        L_hi = L if ok else L_lo
        if not ok:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if x_lo > r1 and x_hi < r2:
                if x_lo - r1 < r2 - x_hi:
                    x_hi = r1
                else:
                    x_lo = r2
            elif x_lo < r1 and x_hi > r2:
                mid1 = 0.5 * (x_lo + min(x_hi, r1))
                mid2 = 0.5 * (max(x_lo, r2) + x_hi)
                if abs(mid1 - h0) <= abs(mid2 - h0):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip(0.5 * (x_lo + x_hi), 0.0, 1.0))

def cd_exact(h, n, max_passes, deadline, rand_order=False):
    h = h.copy()
    rng = np.random.RandomState(9999 + n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        idx = rng.permutation(n) if rand_order else range(n)
        for p in idx:
            if time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_resample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    # Phase 1: SLSQP warm-up
    if time.time() < deadline - 300:
        h1, c1 = slsqp_optimize(h, n, [2000, 10000, 50000, 200000, 500000], 400)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Sequential CD
    h2, c2 = cd_exact(best_h, n, 50, deadline, rand_order=False)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Random-order CD (main workhorse)
    h3, c3 = cd_exact(best_h, n, 5000, deadline, rand_order=True)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    for n_try in [500, 400, 600, 800, 1000]:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        hr = interp_resample(best_h, len(best_h), n_try)
        hr, cr = slsqp_optimize(hr, n_try, [5000, 50000], 300)
        hr, cr = cd_exact(hr, n_try, 200, deadline, rand_order=True)
        if cr < best_c:
            best_h, best_c = hr, cr
    # Phase 5: Multi-restart with perturbation
    if time.time() < deadline - 100:
        n_best = len(best_h)
        for r in range(5):
            if time.time() > deadline - 60:
                break
            s = 0.003 + 0.02 * (r / 4)
            hp = project_feasible(best_h + rng.normal(0, s, n_best), n_best)
            hp, cp = slsqp_optimize(hp, n_best, [20000, 100000], 250)
            hp, cp = cd_exact(hp, n_best, 100, deadline, rand_order=True)
            if cp < best_c:
                best_h, best_c = hp, cp
    # Phase 6: Final precise CD
    h6, c6 = cd_exact(best_h, len(best_h), 2000, deadline, rand_order=False)
    if c6 < best_c:
        best_h, best_c = h6, c6
    h6, c6 = cd_exact(best_h, len(best_h), 500, deadline, rand_order=True)
    if c6 < best_c:
        best_h, best_c = h6, c6
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
