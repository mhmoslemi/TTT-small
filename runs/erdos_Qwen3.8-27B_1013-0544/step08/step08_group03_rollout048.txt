        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: thorough CD on initial grid
    h1, c1 = exact_cd(best_h, n_init, deadline, 200, 16, 35)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: multi-n exploration
    n_candidates = [300, 400, 500, 600, 700, 800, 1000, 1200, 1500]
    n_candidates = [n for n in n_candidates if n != n_init]
    n_candidates.sort(key=lambda x: abs(x - n_init))
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        rem = deadline - time.time()
        if rem < 40:
            break
        h_int = interpolate_h(best_h, len(best_h), n_try)
        cd_deadline = t0 + (time.time() - t0) + min(rem * 0.45, 150)
        h_opt, c_opt = exact_cd(h_int, n_try, cd_deadline, 120, 16, 30)
        if time.time() < deadline - 20:
            h_s, c_s = pnorm_slsqp(h_opt, n_try, 400, 300, deadline)
            if c_s < c_opt:
                h_opt, c_opt = h_s, c_s
        if time.time() < deadline - 10:
            h_f, c_f = exact_cd(h_opt, n_try, deadline, 80, 20, 35)
            if c_f < c_opt:
                h_opt, c_opt = h_f, c_f
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    # Phase 3: perturbation loop on best n
    n_best = len(best_h)
    noise_schedule = [0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
    attempts = 0
    for scale in noise_schedule:
        if time.time() > deadline - 40:
            break
        for _rep in range(2):
            if time.time() > deadline - 40:
                break
            attempts += 1
            if attempts % 3 == 0:
                a = rng.randint(0, max(1, n_best - 60))
                b = min(n_best, a + rng.randint(30, 80))
                hp = best_h.copy()
                hp[a:b] = 1.0 - hp[a:b]
                hp = project_feasible(hp, n_best)
            elif attempts % 3 == 1:
                freq = rng.randint(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                wave = scale * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                hp = project_feasible(best_h + wave, n_best)
            else:
                hp = project_feasible(best_h + rng.normal(0, scale, n_best), n_best)
            h_p, c_p = exact_cd(hp, n_best, deadline, 100, 16, 30)
            if c_p < best_c:
                best_h, best_c = h_p, c_p
    # Phase 4: SLSQP polish with increasing p
    for p_val in [500, 2000, 10000]:
        if time.time() > deadline - 15:
            break
        h_s, c_s = pnorm_slsqp(best_h, n_best, p_val, 350, deadline)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
    # Phase 5: final CD polish
    if time.time() < deadline - 5:
        h_f, c_f = exact_cd(best_h, n_best, deadline, 200, 24, 45)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
