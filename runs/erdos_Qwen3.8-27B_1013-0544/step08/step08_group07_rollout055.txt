    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            room = (1.0 - h[mask]).sum()
            if room < 1e-15:
                break
            h[mask] += diff * (1.0 - h[mask]) / room
        else:
            mask = h > 0.0
            mass = h[mask].sum()
            if mass < 1e-15:
                break
            h[mask] += diff * h[mask] / mass
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    return (mx + np.log(np.sum(np.exp(c - mx))) / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-20, 'gtol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    for _ in range(max_passes):
        if time.time() > deadline - 3:
            break

        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_lo = min(p, n - 1 - p)
            m_hi = max(p + n - 1, 2 * n - 2 - p)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            e1 = (ms >= p) & (ms <= p + n - 1)
            j1 = p - ms + n - 1
            v1 = e1 & (j1 >= 0) & (j1 < n)
            a[v1] += 1.0 - h[j1[v1]]
            e2 = (ms >= n - 1 - p) & (ms <= 2 * n - 2 - p)
            j2 = p + ms - n + 1
            v2 = e2 & (j2 >= 0) & (j2 < n)
            a[v2] -= h[j2[v2]]
            c_aff = c_prev[ms]
            unaff = 0.0
            if m_lo > 0:
