            # c[k] = c_aff[k] + d*(1-h[p+k-(n-1)]) - d*h[p-k+(n-1)] - d^2 (if both terms involve h[p])

            # For other k where exactly one term involves h[p]:
            #   linear in d
            # 
            # Let me just compute this numerically for each k.
            
            # Better: compute the coefficients directly
            # c[k] = sum_j h[j]*(1-h[j+k-(n-1)])
            # Terms involving h[p]:
            #   j=p: h[p]*(1-h[p+k-(n-1)]) if 0<=p+k-(n-1)<n
            #   j+k-(n-1)=p => j=p-k+(n-1): h[p-k+(n-1)]*(1-h[p]) if 0<=p-k+(n-1)<n
            # 
            # Let a = p+k-(n-1) (if valid) and b = p-k+(n-1) (if valid)
            # If a != p: c[k] gets contribution h0*(1-h[a]) + h[b]*(1-(h0+d)) = h0*(1-h[a]) + h[b] - h0*h[b] - d*h[b]
            #   So linear in d with coefficient -h[b]
            # If b != p: c[k] gets contribution (h0+d)*(1-h[a]) + h[b]*(1-h0)
            #   So linear in d with coefficient (1-h[a])
            # If a == p (k=n-1): (h0+d)*(1-(h0+d)) = h0(1-h0) + d(1-2h0) - d^2
            # If b == p (k=n-1): same thing (it's the same term)
            # If a == b == p: only happens when k=n-1, and we count it once
            # 
            # Actually when a == p and b == p, it's the same term h[p]*(1-h[p]).
            # 
            # For each k, the change in c[k] due to d is:
            #   If a valid and a != p: contribution from j=p term is (h0+d)*(1-h[a]) instead of h0*(1-h[a]), change = d*(1-h[a])
            #   If b valid and b != p: contribution from j=b term is h[b]*(1-(h0+d)) instead of h[b]*(1-h0), change = -d*h[b]
            #   If a == p (i.e., k = n-1): the term is (h0+d)*(1-(h0+d)) instead of h0*(1-h0)
            #     change = d*(1-2h0) - d^2
            #   If both a valid (a!=p) and b valid (b!=p): total change = d*(1-h[a]) - d*h[b]
            #   If a valid (a!=p) and b==p: this means k=n-1 and a=p, which contradicts a!=p. So this can't happen.
            #   Wait, b==p means p-k+(n-1)==p, so k=n-1. And a==p also means k=n-1. So a==b==p iff k==n-1.
            # 
            # Summary: for each k:
            #   change = 0
            #   a = p+k-(n-1); if 0<=a<n and a!=p: change += d*(1-h[a])
            #   b = p-k+(n-1); if 0<=b<n and b!=p: change += -d*h[b]
            #   if k==n-1: change += d*(1-2*h0) - d^2  [this is the a==p==b case]
            #   Actually wait, when k==n-1: a = p+(n-1)-(n-1) = p, and b = p-(n-1)+(n-1) = p.
            #   So a==b==p, and the term is h[p]*(1-h[p]).
            #   The original contribution was h0*(1-h0).
            #   New contribution: (h0+d)*(1-h0-d) = h0 - h0^2 + d - d*h0 - d*h0 - d^2 = h0(1-h0) + d(1-2h0) - d^2
            #   Change: d(1-2h0) - d^2
            #   And we should NOT add the separate a and b terms (since a==p and b==p).
            #
            # So the algorithm for each p:
            #   For each k in [0, 2n-2]:
            #     delta_c_k = 0
            #     diag = (k == n-1)
            #     a = p+k-(n-1)
            #     b = p-k+(n-1)
            #     if not diag and 0<=a<n: delta_c_k += (1-h[a])  [coefficient of d]
            #     if not diag and 0<=b<n: delta_c_k += -h[b]
            #     if diag: delta_c_k = (1-2*h0)  [coefficient of d], and we have -d^2 term
            #     new_c_k = c[k] + d * delta_c_k - d^2 * diag
            #   Objective: max(new_c_k) * dx
            #
            # This is a piecewise-linear (plus quadratic at k=n-1) function of d.
            # We minimize it over d in [-h0, 1-h0].
            
            # Compute coefficients for all k
            ks = np.arange(2 * n - 1)
            a_arr = p + ks - (n - 1)
            b_arr = p - ks + (n - 1)
            valid_a = (a_arr >= 0) & (a_arr < n) & (a_arr != p)
            valid_b = (b_arr >= 0) & (b_arr < n) & (b_arr != p)
            diag_mask = (ks == n - 1)
            
            a_coeff = np.zeros(2 * n - 1)
            a_coeff[valid_a] = 1.0 - h[a_arr[valid_a]]
            a_coeff[valid_b] += (-h[b_arr[valid_b]])
            a_coeff[diag_mask] = 1.0 - 2.0 * h0
            
            # c[k] + d*a_coeff[k] - d^2*diag_mask[k]
            # We want to minimize max_k of this over d in [-h0, 1-h0]
            # This is a convex function of d (max of affine functions, plus the -d^2 makes it concave... wait)
            # Actually -d^2 is concave, so the function is NOT convex in general.
            # But for small d it's approximately convex. Let's just do a grid search + golden section.
            
            c_curr = c  # current correlation values
            
            # Grid search
            n_grid = 100
            lo = -h0
            hi = 1.0 - h0
            d_vals = np.linspace(lo, hi, n_grid)
            
            best_d = 0.0
            best_val = current_max  # d=0 gives current_max
            
            for d in d_vals:
                new_c = c_curr + d * a_coeff
                if np.any(diag_mask):
                    new_c = new_c.copy()
                    new_c[diag_mask] -= d * d
                val = float(new_c.max())
                if val < best_val:
                    best_val = val
                    best_d = d
            
            # Refine with golden section around best_d
            idx = int(np.argmin(np.abs(d_vals - best_d)))
            if 0 < idx < n_grid - 1:
                gl = d_vals[idx - 1]
                gh = d_vals[idx + 1]
                
                def obj_func(d):
                    new_c = c_curr + d * a_coeff
                    if np.any(diag_mask):
                        new_c = new_c.copy()
                        new_c[diag_mask] -= d * d
                    return float(new_c.max())
                
                try:
                    res = golden(obj_func, gl, gh, maxiter=80)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = best_val
        
        if not improved:
            break
        h = project_feasible(h, n)
    
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_v = np.exp(shifted - mx)
    w = exp_v / exp_v.sum()
    dx = 2.0 / n
    corr = np.correlate(w, 1.0 - h, mode='full')
    conv = np.convolve(w, h, mode='full')
    return dx * (corr[:n] - conv[n-1:n-1+n])

def lse_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return (m + mx / alpha + np.log(np.sum(np.exp(s - mx)))) * (2.0 / n) / 1.0

def grad_proj(h, n, alphas, max_iter, deadline=None):

                break

            break

                c = np.correlate(x, 1.0 - x, mode='full')
                m = c.max()
                s = alpha * (c - m)
                mx = s.max()
                return (m + mx / alpha + np.log(np.sum(np.exp(s - mx)))) * (2.0 / n)
            def obj_jac(x):
                return lse_grad(x, n, alpha)
            bounds = [(0.0, 1.0)] * n

                     'fun': lambda x: x.sum() - n / 2.0,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(obj, best_h, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
