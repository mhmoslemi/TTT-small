from scipy.optimize import minimize

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max()) * (2.0 / n)

def _sobj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def _sobj(x, n, a):
    return _sobj_grad(x, n, a)[0]

def _sgrad(x, n, a):
    return _sobj_grad(x, n, a)[1]

def _sumf(x, n):
    return x.sum() - 0.5 * n

def _sumj(x, n):
    return np.ones_like(x)

def binarize(h, n):
    if n % 2:
        return None

    for a in alphas:
        if time.time() > dl:
            break
        try:
            r = minimize(_sobj, h, args=(n, a), jac=_sgrad, method="SLSQP",
                         bounds=bl, constraints=cs, options={"maxiter": int(maxit), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(r.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            h, bc = nh, nc
    return h, bc

def swap_delta(om, h, p, q, n):

    d = np.zeros(2 * n - 1)
    d[p:p + n] -= om
    d[q:q + n] += om
    d[diag - p:diag - p + n] += h
    d[diag - q:diag - q + n] -= h
    d[diag] -= 2.0
    return d
