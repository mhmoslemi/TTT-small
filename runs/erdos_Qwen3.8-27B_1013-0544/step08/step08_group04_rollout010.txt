            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
        else:
            r = -s
            g[r:] += 1.0 - h[:n-r]
            g[:n-r] -= h[r:]
    # Normalize
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        g /= gn
    h_new = project_feasible(h - step_size * g, n)
    return h_new

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size: h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size: h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n: return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def coord_opt(hp, cseg, a, qconst, unaff, iters=22):
    tol = 1e-11
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-12, unaff, qconst, qcur, cm, zmax) + 1e-8
    ap, an = a[pos], a[neg]
    bp, bn = b[pos], b[neg]
    pc, nc = ap.size, an.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - bn) / an))
                if v > ll: ll = v
            if pc:
                v = float(np.min((mid - bp) / ap))
                if v < lh: lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    if g < 0: g = 0.0
                    d = max(0.0, 1.0 - 4.0 * g)
                    r1 = 0.5 - 0.5 * np.sqrt(d)
                    r2 = 0.5 + 0.5 * np.sqrt(d)
                    if max(0.0, ll) <= min(r1, lh) + tol: feas = True
                    else: feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas: high = mid
        else: low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll, lh = 0.0, 1.0
        if nc:
            v = float(np.max((t - bn) / an))
            if v > ll: ll = v
        if pc:
            v = float(np.min((t - bp) / ap))
            if v < lh: lh = v
        if ll <= 1.0 + tol and lh >= -tol:
            g = t - qconst
            if g > 0.25:
                cands.extend([max(0.0, ll), min(1.0, lh)])
            elif g >= -1e-12:
                if g < 0: g = 0.0
                d = max(0.0, 1.0 - 4.0 * g)
                r1 = 0.5 - 0.5 * np.sqrt(d)
                r2 = 0.5 + 0.5 * np.sqrt(d)
                cands.extend([max(0.0, ll), min(r1, lh), max(r2, ll), min(1.0, lh)])
    best_x, best_f = hp, max(unaff, qcur)
    if m: best_f = max(best_f, float(np.max(cseg)))
    for x in cands:
        if x < 0.0 or x > 1.0: continue
        f = max(unaff, qconst + x*(1-x))
        if m: f = max(f, float(np.max(cseg + a*(x-hp))))
        if f < best_f - 1e-13:
            best_f, best_x = f, x
    if best_f > high + 1e-5 and m:
        xg = np.linspace(0.0, 1.0, 40)
        vals = np.maximum(np.maximum(cseg[None,:]+(xg-hp)[:,None]*a[None,:],axis=1), 
                          qconst + xg*(1-xg))
        vals = np.maximum(vals, unaff)
        idx = int(np.argmin(vals))
        if vals[idx] < best_f:
            best_f, best_x = float(vals[idx]), float(xg[idx])
    return best_x, best_f

def cd_optimize(h, n, passes, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0/n)*cur
    center = n - 1
    L = 2*n - 1
    for _ in range(passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            if time.time() > deadline - 5: break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2*center - p)
            s_arr = center - np.arange(lo, hi+1, dtype=int)
            j = p + s_arr
            i = p - s_arr
            a = np.zeros(hi - lo + 1)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj): a[vj] += 1.0 - h[j[vj]]
            if np.any(vi): a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(max(hi - lo, 0))
            asg = np.empty(max(hi - lo, 0))
            if off > 0 and center > lo:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp*(1.0-hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = coord_opt(hp, cseg, asg, qconst, unaff, 22)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off > 0 and center > lo: c[lo:center] += asg[:off]*dxv
                if hi > center: c[center+1:hi+1] += asg[off:]*dxv
                c[center] = qconst + bx*(1-bx)
                h[p] = bx
                cur = bf
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cv = (2.0/n)*cur
        if cv < best_c - 1e-14:
            best_c, best_h = cv, h.copy()
        if not improved: break
    return best_h, best_c

def subgrad_phase(h, n, steps, deadline, top_frac=0.3):
    h = h.copy()
    best_h, best_c = h.copy(), compute_c(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    step = 0.002
    for it in range(steps):
        if time.time() > deadline: break
        threshold = np.percentile(c, 100.0*(1.0 - top_frac))
        active = c[threshold:]
        g = np.zeros(n)
        for m_v in np.where(c >= threshold)[0]:
            s = center - int(m_v)
            if s == 0:
                g += 1.0 - 2.0*h
            elif s > 0:
                g[:n-s] += 1.0 - h[s:]
                g[s:] -= h[:n-s]
            else:
                r = -s
                g[r:] += 1.0 - h[:n-r]
                g[:n-r] -= h[r:]
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            h_new = project_feasible(h - step*(g/gn), n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            cv = float(c_new.max())
            if cv < float(c.max()):
                h, c = h_new, c_new
                cv2 = (2.0/n)*float(c.max())
                if cv2 < best_c:
                    best_c, best_h = cv2, h.copy()
            else:
                step *= 0.9
        step = min(step * 1.02, 0.01)
    return best_h, best_c

def fourier_perturb(h, n, rng, amp=0.05):
    x = (np.arange(n) + 0.5) * (2.0/n)
    t = x - 1.0
    pert = np.zeros(n)
    for k in range(1, 12):
        amp_k = amp / k
        pert += rng.uniform(-amp_k, amp_k) * np.cos(k * np.pi * t)
        pert += rng.uniform(-amp_k, amp_k) * np.sin(k * np.pi * t)
    return project_feasible(h + pert, n)
