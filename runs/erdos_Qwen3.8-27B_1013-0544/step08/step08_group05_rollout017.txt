def exact_coord_min(c, A, old, n, top_m=20):
    """Find exact min of max_k (c[k] + (v-old)*A[k]) over v in [0,1].
    Returns the optimal v value."""
    # Get the top M lags by correlation
    m = min(top_m, len(c))
    top_idx = np.argsort(c)[-m:]
    ct = c[top_idx]
    At = A[top_idx]
    
    candidates = [0.0, 1.0, old]  # boundaries and current
    
    # Find crossing points
    for i in range(m):
        for j in range(i+1, m):
            dA = At[j] - At[i]
            if abs(dA) > 1e-14:
                v = old + (ct[i] - ct[j]) / dA
                if 0.0 <= v <= 1.0:
                    candidates.append(v)
    
    candidates = np.array(candidates)
    # Remove duplicates
    candidates = np.unique(np.round(candidates, 10))
    
    # Evaluate
    best_v = old
    best_val = np.inf
    for v in candidates:
        val = np.max(c + (v - old) * A)
        if val < best_val:
            best_val = val
            best_v = v
    
    return float(best_v), float(best_val)

                        best_v = v

            if nm < cm - 1e-12:
                c = d
                cm = nm
                h[i] = 0.0
                h[j] = 1.0
                if cm * dx < bc:
                    best = h.copy()
                    bc = cm * dx
        if cm * dx < bc:
            stagnant = 0
            best = h.copy()
            bc = cm * dx
        else:
            stagnant += 1
            if stagnant > 5:
                break
        # Perturb
        for _ in range(3):
            i = int(np.argmax(h))
            for _2 in range(5):
                i = int(rng.randint(n))
                if h[i] >= 0.5:
                    break
            j = int(np.argmin(h))
            for _2 in range(5):
                j = int(rng.randint(n))
                if h[j] < 0.5:
                    break
            if i != j:
                h[i] = 0.0
                h[j] = 1.0
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
    return best, bc
