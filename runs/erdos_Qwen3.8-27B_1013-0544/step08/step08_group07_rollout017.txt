    m_star = int(c.argmax())

    grad = np.zeros(n)
    p = np.arange(n)
    if m_star == n - 1:

    else:

        vf = (jf >= 0) & (jf < n)
        grad[vf] += (1.0 - h[jf[vf]])
        jb = p - m_star + (n - 1)
        vb = (jb >= 0) & (jb < n)
        grad[vb] -= h[jb[vb]]
    
    # Take step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

            m2_lo = max(0, p - n + 1)
            m2_hi = min(2 * n - 2, p)
            m_lo = min(m1_lo, m2_lo)
            m_hi = max(m1_hi, m2_hi)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a_coef = np.zeros(len(ms))
            if any(vf):
                a_coef[vf] += (1.0 - h[jf[vf]])
            if any(vb):
                a_coef[vb] -= h[jb[vb]]
            diag = (ms == (n - 1))
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            if abs(a_coef).max() < 1e-15:
                continue
            best_d = 0.0
            best_val = current_max
            d_lo = -h0
            d_hi = 1.0 - h0
            d_grid = np.linspace(d_lo, d_hi, 15)
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a_coef
                if any(diag):
                    nv[diag] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                span = (d_hi - d_lo) / 15.0
                lo = max(d_lo, best_d - span)
                hi = min(d_hi, best_d + span)
                d_fine = np.linspace(lo, hi, grid_size)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a_coef
                    if any(diag):
                        nv[diag] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-12 and best_val < current_max:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a_coef
                if any(diag):
                    c[ms[diag]] -= best_d * best_d
                current_max = max(unaff_max, float(c.max()))
                improved_any = True
        if not improved_any:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        nc = float(c.max()) * (2.0 / n)
        if nc < best_c:
            best_c = nc
    return h, compute_c5(h, n)

def subgradient_polish(h, n, deadline, n_steps=2000):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    lr0 = 0.01
    for s in range(n_steps):
        if time.time() > deadline:
            break
        m_star = int(c.argmax())
        p = np.arange(n)
        grad = np.zeros(n)
        if m_star == n - 1:
            grad = 1.0 - 2.0 * h
        else:
            jf = p + m_star - (n - 1)
            vf = (jf >= 0) & (jf < n)
            grad[vf] += (1.0 - h[jf[vf]])
            jb = p - m_star + (n - 1)
            vb = (jb >= 0) & (jb < n)
            grad[vb] -= h[jb[vb]]
        lr = lr0 / (1.0 + 0.01 * s)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        nc = compute_c5(h_new, n)
        if nc < best_c:
            best_c = nc
            h = h_new
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = 1.0

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 200
        x = np.linspace(0, 2, n_init, endpoint=False)
        h_init = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x), n_init)

    # Phase 1: Optimize at initial n with L-BFGS warm-up
    h = project_feasible(h_init, n_init)
    c = compute_c5(h, n_init)
    best_h, best_c = h.copy(), c

    h, c = lbfgs_phase(h, n_init, deadline - 700, [200, 1000, 5000])
    if c < best_c:
        best_h, best_c = h.copy(), c
    h, c = coord_cd(h, n_init, 150, deadline - 300, grid_size=50)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 2: Multi-n search
    for n_try in [100, 150, 200, 250, 300, 400]:
        if time.time() > deadline - 150:
            break
        x_big = np.linspace(0, 2, len(best_h), endpoint=False)
        x_s = np.linspace(0, 2, n_try, endpoint=False)
        h_s = project_feasible(np.interp(x_s, x_big, best_h), n_try)
        h_s, c_s = lbfgs_phase(h_s, n_try, deadline - 150, [500, 3000])
        h_s, c_s = coord_cd(h_s, n_try, 120, deadline - 80, grid_size=50)
        h_s, c_s = subgradient_polish(h_s, n_try, deadline - 30, 500)
        h_s, c_s = coord_cd(h_s, n_try, 80, deadline - 10, grid_size=60)
        if c_s < best_c:
            best_h, best_c = h_s.copy(), c_s

    # Phase 3: Randomized restarts on best n
    n_best = len(best_h)
    remaining = deadline - time.time() - 20
    if remaining > 60:
        n_restarts = int(remaining / 40)
        n_restarts = max(3, min(n_restarts, 20))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            if r % 3 == 0:
                hp = project_feasible(best_h + 0.03 * rng.randn(n_best), n_best)
            elif r % 3 == 1:
                x = np.linspace(0, 2, n_best, endpoint=False)
                hp = project_feasible(0.5 + 0.4 * np.sin(2 * np.pi * x * (1 + r % 3)), n_best)
            else:
                hp = project_feasible(rng.uniform(0.2, 0.8, n_best), n_best)
            h_r, c_r = coord_cd(hp, n_best, 60, deadline - 20, grid_size=40)
            h_r, c_r = randomized_cd(h_r, n_best, 40, deadline - 10, grid_size=30, rng=rng)
            if c_r < best_c:
                best_h, best_c = h_r.copy(), c_r

    # Phase 4: Final intensive polish
    if time.time() < deadline - 5:
        h_f, c_f = coord_cd(best_h, n_best, 200, deadline - 2, grid_size=80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = randomized_cd(best_h, n_best, 100, deadline - 1, grid_size=40, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
