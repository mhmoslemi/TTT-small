from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_a(p, h, n):
    a = np.zeros(2 * n - 1)
    n1 = n - 1
    s_lo = max(-p, -n1)
    s_hi = min(n1 - p, n1)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi + 1)
        a[s + n1] += 1.0 - h[p + s]
    s_lo2 = max(-(n1 - p), -n1)
    s_hi2 = min(p, n1)
    if s_lo2 <= s_hi2:
        s = np.arange(s_lo2, s_hi2 + 1)
        a[s + n1] -= h[p - s]
    return a

def solve_1d(c, a, d_lo, d_hi):
    n1 = len(c) - 1
    pos = a > 1e-14
    neg = a < -1e-14
    T_hi = float(c.max())
    T_lo = float(c.min()) - 2.0
    for _ in range(50):
        T_mid = (T_lo + T_hi) / 2.0
        lb = d_lo
        ub = d_hi
        if pos.any():
            ub = min(ub, float(((T_mid - c[pos]) / a[pos]).min()))
        if neg.any():
            lb = max(lb, float(((T_mid - c[neg]) / a[neg]).max()))
        disc = a[n1] * a[n1] - 4.0 * (T_mid - c[n1])
        if disc >= 0.0:
            sd = np.sqrt(disc)
            r1 = (a[n1] - sd) / 2.0
            r2 = (a[n1] + sd) / 2.0
            if ub > r1 and lb < r2:
                ub1 = min(ub, r1)
                lb2 = max(lb, r2)
                if ub1 >= lb and lb2 > ub:
                    ub = ub1
                elif lb2 <= ub and ub1 < lb:
                    lb = lb2
                elif ub1 >= lb:
                    ub = ub1
                else:
                    lb = lb2
        if lb > ub + 1e-14:
            T_lo = T_mid
        else:
            T_hi = T_mid
    T_opt = T_hi
    lb = d_lo
    ub = d_hi
    if pos.any():
        ub = min(ub, float(((T_opt - c[pos]) / a[pos]).min()))
    if neg.any():
        lb = max(lb, float(((T_opt - c[neg]) / a[neg]).max()))
    disc = a[n1] * a[n1] - 4.0 * (T_opt - c[n1])
    if disc >= 0.0:
        sd = np.sqrt(disc)
        r1 = (a[n1] - sd) / 2.0
        r2 = (a[n1] + sd) / 2.0

            lb2 = max(lb, r2)
            if ub1 >= lb:
                ub = ub1
            else:
                lb = lb2
    d_opt = (lb + ub) / 2.0
    return T_opt, float(np.clip(d_opt, d_lo, d_hi))

def exact_coord_descent(h, n, max_passes=2000, deadline=None, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            a = compute_a(p, h, n)
            T_opt, d_opt = solve_1d(c, a, -h0, 1.0 - h0)
            if T_opt < current_max - 1e-10:
                h[p] = h0 + d_opt
                c = c + d_opt * a
                c[n - 1] -= d_opt * d_opt

            stagnant = 0
            best_c = new_c
        else:

    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def lbfgs_opt(h0, n, alphas, maxiter, deadline=None):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n), 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

                best_h, best_c = h_back, c_back

    # Phase 4: Diverse restarts
    if time.time() < deadline - 50:
        for r in range(6):
            if time.time() > deadline - 30:
                break
            s = 0.02 + 0.08 * r / 5.0
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = exact_coord_descent(hp, n, 1500, deadline, rng)
            h_o, c_o = lbfgs_opt(h_o, n, [5000, 50000], 400, deadline)
            h_o, c_o = exact_coord_descent(h_o, n, 1000, deadline, rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Final high-precision coordinate descent
    if time.time() < deadline - 5:
        hf, cf = exact_coord_descent(best_h, n, 5000, deadline, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
