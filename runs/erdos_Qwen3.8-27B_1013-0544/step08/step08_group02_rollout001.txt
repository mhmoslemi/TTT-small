    accepts = 0

    max_no = max(300, max_iters // 3)
    recompute_every = 30
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        if tight:
            tol = 0.0005
        else:
            tol = max(0.008, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if it % 11 == 10 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            m = n - abs(d)
            if m <= 0:
                no_imp += 1
                continue
            if d >= 0:
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e2 = -d
                b = e2 + int(rng.randint(m))
                i, j = b - e2, b
        hi_v, hj_v = h[i], h[j]
        u_lo = max(-hi_v, -(1.0 - hj_v))
        u_hi = min(1.0 - hi_v, hj_v)
        if u_hi - u_lo < 1e-14:
            no_imp += 1
            continue
        nu = min(80, max(20, int((u_hi - u_lo) * 200)))
        us = np.linspace(u_lo, u_hi, nu)
        B = A[i] - A[j]
        lines = c[None, :] + us[:, None] * B[None, :]
        uu = us * us
        lines[:, off] -= 2.0 * uu
        idx1 = off + i - j
        idx2 = off + j - i
        if 0 <= idx1 < dim:
            lines[:, idx1] += uu
        if 0 <= idx2 < dim:
            lines[:, idx2] += uu
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        if vals[idx] < cur_max - 1e-11:
            u = float(us[idx])
            old_s = hi_v + hj_v
            h[i] = hi_v + u
            h[j] = old_s - h[i]
            c = c + u * B
            c[off] -= 2.0 * u * u
            if 0 <= idx1 < dim:
                c[idx1] += u * u
            if 0 <= idx2 < dim:
                c[idx2] += u * u
            if accepts % recompute_every == recompute_every - 1:
                A = compute_jac(h, n)
            accepts += 1
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > max_no:
            break
    c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(c.max()) * 2.0 / n

def slsqp_opt(h, n, alpha, max_iter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h

def lbfgs_opt(h, n, alpha, mu, max_iter, deadline):
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = float(c.max())
        z = alpha * (c - m)
        zm = float(z.max())
        e = np.exp(z - zm)
        val = (m + np.log(e.sum()) / alpha) * 2.0 / n
        dev = x.sum() - 0.5 * n
        return val + 0.5 * mu * dev * dev

    def g(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = float(c.max())
        z = alpha * (c - m)
        zm = float(z.max())
        e = np.exp(z - zm)
        p = e / e.sum()
        grad = np.zeros(n)
        off = n - 1
        for k in range(2 * n - 1):
            w = p[k]
            if w < 1e-13:
                continue
            d = k - off
            if d >= 0:
                grad[d:] += w * (1.0 - x[:n - d])
                grad[:n - d] -= w * x[d:]
            else:
                e2 = -d
                grad[:n - e2] += w * (1.0 - x[e2:])
                grad[e2:] -= w * x[:n - e2]
        grad = 2.0 / n * grad
        dev = x.sum() - 0.5 * n
        grad += mu * dev
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'maxls': 30, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return h

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def apply_symmetries(h, n, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cands = [h, 0.5*(h + h[::-1]), 1.0 - h, 0.5*(h + 1.0 - h[::-1])]
    for c in cands:
        c2 = project_feasible(c, n)
        cc = compute_c5(c2, n)
        if cc < best_c:
            best_h, best_c = c2, cc
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n = 400
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = 0.5 + 0.2*np.sin(np.pi*x) + 0.1*np.sin(2*np.pi*x)
        h0 = project_feasible(h0, n)
    n0 = len(h0)
    best_h, best_c = apply_symmetries(project_feasible(h0, n0), n0, rng)

    # Phase 1: L-BFGS to get into good basin
    if time.time() < hard - 200:
        for alpha in [30, 100, 400]:
            if time.time() > hard - 180:
                break
            hl = lbfgs_opt(best_h, n0, alpha, 1e4, 30, time.time()+40)
            cl = compute_c5(hl, n0)
            if cl < best_c:
                best_h, best_c = hl, cl

    # Phase 2: SLSQP with increasing alpha
    if time.time() < hard - 150:
        for alpha, mx in [(80, 60), (300, 40), (1500, 30), (8000, 25)]:
            if time.time() > hard - 130:
                break
            hs = slsqp_opt(best_h, n0, alpha, mx, time.time()+45)
            cs = compute_c5(hs, n0)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 3: Main pair search (loose)
    if time.time() < hard - 100:
        dl = min(hard - 80, time.time() + 250)
        best_h, best_c = pair_search(best_h, n0, 30000, dl, rng, tight=False)

    # Phase 4: Try different resolution
    if time.time() < hard - 80:
        for m in [350, 450, 600]:
            if time.time() > hard - 60:
                break
            if m == n0:
                continue
            ht = interp_h(best_h, m)
            ht, ct = apply_symmetries(ht, m, rng)
            # Quick SLSQP
            if time.time() < hard - 55:
                ht = slsqp_opt(ht, m, 200, 30, time.time()+30)
                ct = compute_c5(ht, m)
            # Pair search
            if time.time() < hard - 50:
                dl2 = min(hard - 45, time.time() + 50)
                ht, ct = pair_search(ht, m, 5000, dl2, rng, tight=False)
            if ct < best_c:
                best_h, best_c = ht, ct

    # Phase 5: Tight pair search
    if time.time() < hard - 20:
        dl = min(hard - 8, time.time() + 120)
        best_h, best_c = pair_search(best_h, n0, 50000, dl, rng, tight=True)

    # Phase 6: Final SLSQP polish
    if time.time() < hard - 5:
        for alpha in [5000, 20000]:
            if time.time() > hard - 4:
                break
            hs = slsqp_opt(best_h, n0, alpha, 30, time.time()+20)
            cs = compute_c5(hs, n0)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 7: Final tight pair search
    if time.time() < hard - 2:
        dl = hard - 1.0
        best_h, best_c = pair_search(best_h, n0, 100000, dl, rng, tight=True)

    best_h = project_feasible(best_h, n0)
    best_c = compute_c5(best_h, n0)
    return best_h, best_c, n0
