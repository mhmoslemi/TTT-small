```python
def fista_projected(h, n, t, max_iter=2000, deadline=None):
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_fista = 1.0
    lr = 1.0 / (t * dx * (2*n-1))  # Initial step size estimate
    best_f = smooth_max_value(x, n, t)
    best_x = x.copy()
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        grad = smooth_max_gradient(y, n, t)
        # Backtracking line search
        x_new = project_feasible(y - lr * grad, n)
        f_new = smooth_max_value(x_new, n, t)
        while f_new > smooth_max_value(y, n, t) + 0.1 * lr * np.dot(grad, x_new - y):
            lr *= 0.5
            x_new = project_feasible(y - lr * grad, n)
            f_new = smooth_max_value(x_new, n, t)
        
        t_new = (1 + np.sqrt(1 + 4*t_fista**2)) / 2
        y_new = x_new + (t_fista - 1) / t_new * (x_new - x)
        x, y = x_new, y_new
        t_fista = t_new
        
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
    
    return best_x, best_f
```
