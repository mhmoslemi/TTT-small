import numpy as np

def project_simplex(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.sum() > 0:
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def softmax_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def slsqp_polish(h, n, alphas, maxiter, deadline):

    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    def sfun(x, n):
        return x.sum() - 0.5 * n
    def sjac(x, n):
        return np.ones_like(x, dtype=float)
    cons = [{"type": "eq", "fun": sfun, "jac": sjac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:

                           method="SLSQP", bounds=bounds, constraints=cons,

        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc
