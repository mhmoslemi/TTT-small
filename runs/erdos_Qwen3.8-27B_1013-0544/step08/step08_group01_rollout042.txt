        if ss < target: lo=mid
        else: hi=mid
    h=np.clip(x+(lo+hi)*0.5,0.0,1.0)

import numpy as np

def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    target = 0.5 * n
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if diff != 0.0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / float(n)
            h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            h[idx] += diff if diff < room else room
        else:
            idx = int(np.argmax(h))
            h[idx] += diff if abs(diff) < h[idx] else -h[idx]
    return h


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / float(n))


def affected_range(i, n):
    a1 = n - 1 - i
    if a1 < 0:
        a1 = 0
    b1 = 2 * n - 2 - i
    if b1 > 2 * n - 2:
        b1 = 2 * n - 2
    a2 = i
    b2 = i + n - 1
    if b2 > 2 * n - 2:
        b2 = 2 * n - 2
    lo = a1 if a1 < a2 else a2
    hi = b1 if b1 > b2 else b2
    return lo, hi


def pair_interval(t, C0, B0, pos, neg, zero, Bfull, Cfull, cross_pos, U, lo, hi):
    if U > t + 1e-12:
        return None
    lb = lo
    ub = hi
    if zero.size:
        if float(np.max(C0[zero])) > t + 1e-12:
            return None
    if pos.size:
        cand = float(np.min((t - C0[pos]) / B0[pos]))
        if cand < ub:
            ub = cand
    if neg.size:
        cand = float(np.max((t - C0[neg]) / B0[neg]))
        if cand > lb:
            lb = cand
    if lb > ub + 1e-10:
        return None
    for k in cross_pos:
        cc = float(Cfull[k])
        bb = float(Bfull[k])
        D = bb * bb - 4.0 * (cc - t)
        if D < 0.0:
            if D < -1e-10:
                return None
            D = 0.0
        r = math.sqrt(D)
        r1 = 0.5 * (-bb - r)
        r2 = 0.5 * (-bb + r)
        if r1 > lb:
            lb = r1
        if r2 < ub:
            ub = r2
        if lb > ub + 1e-10:
            return None
    return lb, ub


def pair_eval(h, c, n, p, q, cmax, min_improve):
    if p == q:
        return 0.0, cmax, 0, np.empty(0), np.empty(0)
    hp = float(h[p])
    hq = float(h[q])
    lo = -hp
    tmp = hq - 1.0
    if tmp > lo:
        lo = tmp
    hi = 1.0 - hp
    if hq < hi:
        hi = hq
    if hi - lo < 1e-12:
        return 0.0, cmax, 0, np.empty(0), np.empty(0)
    pl, ph = affected_range(p, n)
    ql, qh = affected_range(q, n)
    mlo = pl if pl < ql else ql
    mhi = ph if ph > qh else qh
    U = 0.0
    if mlo > 0:
        U = float(c[:mlo].max())
    if mhi < 2 * n - 2:
        v = float(c[mhi + 1:].max())
        if v > U:
            U = v
    if U > cmax - 1e-7:
        return 0.0, cmax, mlo, np.empty(0), np.empty(0)
    L = mhi - mlo + 1
    d_arr = np.arange(mlo, mhi + 1) - (n - 1)
    B = np.zeros(L)
    idx = p + d_arr
    v = (idx >= 0) & (idx < n)
    B[v] += 1.0 - h[idx[v]]
    idx = p - d_arr
    v = (idx >= 0) & (idx < n)
    B[v] -= h[idx[v]]
    idx = q + d_arr
    v = (idx >= 0) & (idx < n)
    B[v] -= (1.0 - h[idx[v]])
    idx = q - d_arr
    v = (idx >= 0) & (idx < n)
    B[v] += h[idx[v]]
    A = np.zeros(L)
    pos1 = n - 1 + (q - p) - mlo
    if 0 <= pos1 < L:
        A[pos1] = 1.0
    pos2 = n - 1 + (p - q) - mlo
    if 0 <= pos2 < L:
        A[pos2] = 1.0
    C = c[mlo:mhi + 1]
    line_mask = (A == 0.0)
    C0 = C[line_mask]
    B0 = B[line_mask]
    pos = np.nonzero(B0 > 1e-14)[0]
    neg = np.nonzero(B0 < -1e-14)[0]
    zero = np.nonzero(np.abs(B0) <= 1e-14)[0]
    cross_pos = np.nonzero(A)[0]
    lower = 0.0
    upper = float(cmax)
    for _ in range(30):
        mid = 0.5 * (lower + upper)
        if pair_interval(mid, C0, B0, pos, neg, zero, B, C, cross_pos, U, lo, hi) is not None:
            upper = mid
        else:
            lower = mid
    tval = upper
    interval = pair_interval(tval + 1e-8, C0, B0, pos, neg, zero, B, C, cross_pos, U, lo, hi)
    if interval is not None:
        lb, ub = interval
        cands = np.array([lo, hi, 0.0, lb, ub, 0.5 * (lb + ub)], dtype=float)
        for k in cross_pos:
            cc = float(C[k])
            bb = float(B[k])
            D = bb * bb - 4.0 * (cc - tval)
            if D >= 0.0:
                r = math.sqrt(D)
                cands = np.concatenate((cands, np.array([0.5 * (-bb - r), 0.5 * (-bb + r)])))
    else:
        cands = np.linspace(lo, hi, 25)
    cands = np.clip(cands, lo, hi)
    vals = C[None, :] + B[None, :] * cands[:, None] + A[None, :] * cands[:, None] * cands[:, None]
    vals = np.maximum(vals, U)
    vals = vals.max(axis=1)
    im = int(np.argmin(vals))
    best_val = float(vals[im])
    best_d = float(cands[im])
    if best_val >= cmax - min_improve:
        return 0.0, best_val, mlo, B, A
    return best_d, best_val, mlo, B, A


def add_shift_grad(h, n, d, w, g):
    if d == 0:
        g += w * (1.0 - 2.0 * h)
    elif d > 0:
        L = n - d
        g[:L] += w * (1.0 - h[d:])
        g[d:] -= w * h[:-d]
    else:
        nd = -d
        L = n - nd
        g[nd:] += w * (1.0 - h[:-nd])
        g[:L] -= w * h[nd:]


def compute_top_gradient(c, h, n, K, beta):
    cmax = float(c.max())
    K = max(1, min(K, c.size))
    order = np.argsort(c)[::-1][:K]
    w = np.exp(beta * (c[order] - cmax))
    s = float(w.sum())
    if (not np.isfinite(s)) or s == 0.0:
        w = np.full(K, 1.0 / K)
    else:
        w = w / s
    g = np.zeros(n)
    for idx, wi in zip(order, w):
        add_shift_grad(h, n, int(idx - (n - 1)), float(wi), g)
    return g, cmax


def gradient_pair_descent(h, c, n, end_time, K=10, beta=12.0, C=5, max_pairs=8, min_improve=1e-7):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best = float(c.max())
    stall = 0
    while time.time() < end_time:
        cmax = float(c.max())
        if cmax < best:
            best = cmax
        if cmax <= 0.0:
            break
        g, _ = compute_top_gradient(c, h, n, K, beta)
        small = np.argsort(g)[:C]
        large = np.argsort(g)[::-1][:C]
        candidates = []
        for a in small:
            if h[a] >= 1.0 - 1e-9:
                continue
            for b in large:
                if a == b or h[b] <= 1e-9:
                    continue
                candidates.append((int(a), int(b)))
                if len(candidates) >= max_pairs:
                    break
            if len(candidates) >= max_pairs:
                break
        if not candidates:
            ga = np.where(h < 1.0 - 1e-9, g, np.inf)
            gb = np.where(h > 1e-9, g, -np.inf)
            a = int(np.argmin(ga))
            b = int(np.argmax(gb))
            if a != b and np.isfinite(ga[a]) and np.isfinite(gb[b]):
                candidates = [(a, b)]
        improved = False
        for a, b in candidates:
            delta, val, mlo, B, A = pair_eval(h, c, n, a, b, cmax, min_improve)
            if delta != 0.0 and val < cmax - 1e-9 and B.size > 0:
                h[a] = min(1.0, max(0.0, float(h[a] + delta)))
                h[b] = min(1.0, max(0.0, float(h[b] - delta)))
                c[mlo:mlo + B.size] += B * delta + A * delta * delta
                cmax = val
                best = min(best, cmax)
                improved = True
                stall = 0
                break
        if not improved and stall < 30:
            gn = float(np.sqrt(np.mean(g * g)))
            if gn > 1e-14:
                g2 = g / gn
                for lr in (0.02, 0.01, 0.005):
                    hnew = project_feasible(h - lr * g2, n)
                    cnew = np.correlate(hnew, 1.0 - hnew, mode="full")
                    cn = float(cnew.max())
                    if cn < cmax - 1e-9:
                        h = hnew
                        c = cnew
                        cmax = cn
                        best = min(best, cmax)
                        improved = True
                        break
                if improved:
                    stall = 0
        if not improved:
            stall += 1
            if stall > 40:
                break
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, c, float(c.max())


def reservoir_pair_pass(h, c, q, n, end_time, max_passes=30, min_improve=1e-7):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best = float(c.max())
    noimp = 0
    for pas in range(max_passes):
        if time.time() >= end_time:
            break
        improved = False
        rng = np.random.RandomState(9973 * pas + 101 * int(q) + 7)
        order = rng.permutation(n)
        for p in order:
            if p == q or time.time() >= end_time:
                continue
            p = int(p)
            cmax = float(c.max())
            delta, val, mlo, B, A = pair_eval(h, c, n, p, q, cmax, min_improve)
            if delta != 0.0 and val < cmax - 1e-9 and B.size > 0:
                h[p] = min(1.0, max(0.0, float(h[p] + delta)))
                h[q] = min(1.0, max(0.0, float(h[q] - delta)))
                c[mlo:mlo + B.size] += B * delta + A * delta * delta
                improved = True
                if val < best:
                    best = val
        c = np.correlate(h, 1.0 - h, mode="full")
        cur = float(c.max())
        if cur < best:
            best = cur
        if not improved:
            noimp += 1
            if noimp > 2:
                break
    return h, c, best


def projected_gradient_descent(h, c, n, end_time, K=8, beta=8.0):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best = float(c.max())
    stall = 0
    while time.time() < end_time:
        cmax = float(c.max())
        if cmax < best:
            best = cmax
        g, _ = compute_top_gradient(c, h, n, K, beta)
        norm = float(np.sqrt(np.mean(g * g)))
        if norm < 1e-14:
            break
        g = g / norm
        accepted = False
        for lr in (0.08, 0.04, 0.02, 0.01, 0.005):
            hnew = project_feasible(h - lr * g, n)
            cnew = np.correlate(hnew, 1.0 - hnew, mode="full")
            cn = float(cnew.max())
            if cn < cmax - 1e-9:
                h = hnew
                c = cnew
                cmax = cn
                best = min(best, cmax)
                accepted = True
                break
        if not accepted:
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, c, float(c.max())


def make_perturbed(base_h, sigma, seed):
    n = len(base_h)
    rng = np.random.RandomState(int(seed))
    z = rng.normal(size=n)
    if n > 16:
        zf = np.fft.rfft(z)
        keep = max(2, int(0.18 * n) + 1)
        zf[keep:] = 0.0
        z = np.fft.irfft(zf, n=n)
    std = float(z.std())
    if std > 1e-12:
        z *= sigma / std
    else:
        z = rng.normal(size=n) * sigma
    return project_feasible(base_h + z, n)


def interpolate_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    hi = np.interp(x_new, x_old, h)
    return project_feasible(hi, n_new)


def polish_candidate(h, c, n, end_time):
    rem = max(0.0, end_time - time.time())
    if rem <= 2.0:
        c = np.correlate(h, 1.0 - h, mode="full")
        return h, c, float(c.max())
    e = time.time() + 0.38 * rem
    h, c, _ = gradient_pair_descent(h, c, n, e)
    if time.time() < end_time - 8.0:
        e = time.time() + 0.28 * (end_time - time.time())
        q = max(0, n // 2)
        h, c, _ = reservoir_pair_pass(h, c, q, n, e, max_passes=20)
    if time.time() < end_time - 4.0:
        h, c, _ = projected_gradient_descent(h, c, n, end_time - 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, c, float(c.max())


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 600
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    end1 = min(deadline - 30.0, t0 + 70.0)
    if time.time() < end1:
        h1, _, s1 = gradient_pair_descent(h, c, n, end1)
        v1 = s1 * (2.0 / n)
        if v1 < best_c:
            best_h = h1.copy()
            best_c = v1

    end2 = min(deadline - 580.0, t0 + 250.0)
    q_choices = [0, n // 4, n // 3, n // 2, (2 * n) // 3, (3 * n) // 4, n - 1]
    qi = 0
    res_stall = 0
    while time.time() < end2:
        q = q_choices[qi % len(q_choices)]
        qi += 1
        h2 = best_h.copy()
        c2 = np.correlate(h2, 1.0 - h2, mode="full")
        e = min(end2, time.time() + 30.0)
        h3, _, s3 = reservoir_pair_pass(h2, c2, q, n, e, max_passes=25)
        v3 = s3 * (2.0 / n)
        if v3 < best_c:
            best_h = h3.copy()
            best_c = v3
            res_stall = 0
        else:
            res_stall += 1
            if res_stall >= 3:
                break

    for scale in (0.03, 0.08, 0.15, 0.30):
        if time.time() > deadline - 210.0:
            break
        h2 = make_perturbed(best_h, scale, seed + 1000 * int(scale * 100) + 17)
        c2 = np.correlate(h2, 1.0 - h2, mode="full")
        h3, _, s3 = polish_candidate(h2, c2, n, min(deadline - 180.0, time.time() + 55.0))
        v3 = s3 * (2.0 / n)
        if v3 < best_c:
            best_h = h3.copy()
            best_c = v3

    for n_try in (500, 800, 700):
        if time.time() > deadline - 110.0:
            break
        h2 = interpolate_h(best_h, n_try)
        c2 = np.correlate(h2, 1.0 - h2, mode="full")
        h3, _, s3 = polish_candidate(h2, c2, n_try, min(deadline - 80.0, time.time() + 38.0))
        v3 = s3 * (2.0 / n_try)
        if v3 < best_c:
            best_h = h3.copy()
            best_c = v3

    n_cur = len(best_h)
    h2 = best_h.copy()
    c2 = np.correlate(h2, 1.0 - h2, mode="full")
    h3, _, s3 = polish_candidate(h2, c2, n_cur, deadline - 4.0)
    v3 = s3 * (2.0 / n_cur)
    if v3 < best_c:
        best_h = h3.copy()
        best_c = v3
        n_cur = len(best_h)

    cand_list = [1.0 - best_h, best_h[::-1].copy()]
    for cand in cand_list:
        if time.time() > deadline - 3.0:
            break
        h2 = project_feasible(cand, n_cur)
        c2 = np.correlate(h2, 1.0 - h2, mode="full")
        v2 = float(c2.max()) * (2.0 / n_cur)
        if v2 < best_c:
            best_h = h2.copy()
            best_c = v2
            n_cur = len(best_h)
        if time.time() < deadline - 2.0:
            h3, _, s3 = reservoir_pair_pass(h2, c2, n_cur // 2, n_cur, deadline - 1.0, max_passes=4)
            v3 = s3 * (2.0 / n_cur)
            if v3 < best_c:
                best_h = h3.copy()
                best_c = v3
                n_cur = len(best_h)

    best_h = project_feasible(best_h, n_cur)
    cfin = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_c = float(cfin.max()) * (2.0 / n_cur)
    return best_h, final_c, n_cur
