import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def minimize_1d_pl(b, a, u, lo, hi):
    """Minimize max(max_i(b_i + a_i*d), u) for d in [lo, hi] via bisection on subgradient."""
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = b + a * mid
        mx = vals.max()
        if mx <= u + 1e-15:
            return mid
        active = vals >= mx - 1e-14
        sg_lo = float(a[active].min())
        sg_hi = float(a[active].max())
        if sg_lo <= 1e-14 and sg_hi >= -1e-14:
            break
        elif sg_lo > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        current_max = float(c.max())
        improved = False

            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:

            m_f_lo = max(0, n - 1 - p)

            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)

    # Phase 6: Try nearby n values
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [550, 580, 620, 650]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd_fast(h_int, n_try, 100, deadline)
        h_opt, c_opt = targeted_cd(h_opt, n_try, 50, deadline, top_k=20)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try, endpoint=False)
    # Phase 7: Perturbation restarts
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        for r in range(8):
            if time.time() > deadline - 50:
                break
            s = 0.003 + 0.008 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = coord_cd_fast(hp, n_cur, 80, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 8: Final fine polishing
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = coord_cd_fast(best_h, n_cur, 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
