def block_coordinate_descent(h, n, block_size=10, max_passes=100, deadline=None):
    """Optimize blocks of coordinates using SLSQP on each block."""
    h = h.copy()
    dx = 2.0 / n
    
    for block_start in range(0, n, block_size):
        if deadline is not None and time.time() > deadline:
            break
        block_end = min(block_start + block_size, n)
        # Fix all other coordinates, optimize the block
        h_block = h[block_start:block_end]
        h_fixed = h.copy()
        
        def obj_block(x_block):
            h_trial = h_fixed.copy()
            h_trial[block_start:block_end] = x_block
            c = np.correlate(h_trial, 1.0 - h_trial, mode='full')

def make_binary_beatty(n, seed=42):

        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):

            # Use grid + golden section

                vals[gi] = max(float(nv.max()), unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                # Golden section on the 1D problem
                def obj_d(d):
                    nv = c_aff + d * a_coeff
                    if diag_local is not None:
                        nv = nv.copy()
                        nv[diag_local] -= d * d
                    return max(float(nv.max()), unaff_max)
                try:
                    res = golden(obj_d, gl, gh, maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_run(h, n, alphas, maxiter_each=300, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def smooth_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda: np.ones(n)}]
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
