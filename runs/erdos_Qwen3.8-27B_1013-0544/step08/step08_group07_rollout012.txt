    """h(x) = 0.5 + sum_k coeffs[k] * (cos(2πkx) + sin(2πkx))"""

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=500, top_k=30, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    best_c = current_max
    best_h = h.copy()
    
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        current_max = float(c.max())
        improved = False
        
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            # Compute affected range
            idx_lo = min(p, n - 1 - p)
            idx_hi = max(n - 1 + p, 2 * n - 2 - p)
            idxs = np.arange(idx_lo, idx_hi + 1)
            k_vals = idxs - (n - 1)
            
            # Linear coefficients
            a = np.zeros(len(idxs))
            valid_a = (idxs >= p) & (idxs <= n - 1 + p)
            if np.any(valid_a):
                ja = p - k_vals[valid_a]
                a[valid_a] += (1.0 - h[ja])
            valid_b = (idxs >= n - 1 - p) & (idxs <= 2 * n - 2 - p)
            if np.any(valid_b):
                jb = p + k_vals[valid_b]
                a[valid_b] -= h[jb]
            
            center_in_range = (n - 1 >= idx_lo) and (n - 1 <= idx_hi)
            center_local = (n - 1) - idx_lo if center_in_range else -1
            
            # Unaffected max

            affected[idx_lo:idx_hi + 1] = True
            unaff_max = float(c[~affected].max()) if np.any(~affected) else 0.0
            
            c_aff = c[idx_lo:idx_hi + 1]
            
            # Candidate d values
            d_cands = [-h0, 1.0 - h0, 0.0]
            if center_in_range:
                dv = a[center_local] / 2.0
                if -h0 <= dv <= 1.0 - h0:
                    d_cands.append(dv)
            
            # Top-k intersections
            order = np.argsort(-c_aff)
            tk = min(top_k, len(c_aff))
            top_idx = order[:tk]
            
            for i in range(tk):
                for j in range(i + 1, tk):
                    denom = a[top_idx[i]] - a[top_idx[j]]
                    if abs(denom) < 1e-14: continue
                    di = (c_aff[top_idx[j]] - c_aff[top_idx[i]]) / denom
                    if -h0 <= di <= 1.0 - h0:
                        d_cands.append(di)

                        for s in [1, -1]:

                            if -h0 <= di <= 1.0 - h0:
                                d_cands.append(di)

            best_d = 0.0

                nv = c_aff + d * a
                if center_in_range:

                val = max(unaff_max, float(nv.max()))
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d

        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
