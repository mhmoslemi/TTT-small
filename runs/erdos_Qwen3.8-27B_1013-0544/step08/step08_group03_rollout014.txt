import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_bounded_sum(h, S):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def pnorm_obj_grad(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if p == 1:
        w = np.ones(2 * n - 1)
        scale = dx
        obj = dx * c.sum()
    else:
        cs = np.maximum(c, 1e-30)
        cp = cs ** p
        sp = cp.sum()
        w = cs ** (p - 1)
        scale = dx * sp ** (1.0 / p - 1.0)
        obj = dx * sp ** (1.0 / p)
    m = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-22:
            continue
        mk = int(m[k])
        if mk >= 0:
            g[:n - mk] += wk * (1.0 - h[mk:])
            g[mk:] -= wk * h[:n - mk]
        else:
            mp = -mk
            g[mp:] += wk * (1.0 - h[:n - mp])
            g[:n - mp] -= wk * h[mp:]
    return obj, scale * g, c

def fista(h, n, p, max_iters, deadline, init_lr=None):
    h = h.copy()
    h_prev = h.copy()
    t = 1.0
    if init_lr is None:
        lr = 2.0 / (n * n * 0.1)
    else:
        lr = init_lr
    obj_y, g, _ = pnorm_obj_grad(h, n, p)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj_y, g, _ = pnorm_obj_grad(y, n, p)
        accepted = False
        for _ in range(40):
            hp = project_bounded_sum(y - lr * g, n * 0.5)
            obj_hp, _, _ = pnorm_obj_grad(hp, n, p)
            diff = hp - y
            model = obj_y + np.dot(g, diff) + 0.5 / lr * np.dot(diff, diff)
            if obj_hp <= model + 1e-14:
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.5
            hp = project_bounded_sum(y - lr * g, n * 0.5)
        h_prev = h
        h = hp
        t = t_new
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return h, float(dx * c.max())

def cd_pass(h, n, c, deadline):
    improved = False
    target = n * 0.5
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
            continue
        A = np.zeros(2 * n - 1)
        A[n - 1] = 1.0 - 2.0 * h0
        D = np.zeros(2 * n - 1)
        D[n - 1] = 1.0
        for m in range(1, n - p):
            A[n - 1 + m] += 1.0 - h[p + m]
        for m in range(1, p + 1):
            A[n - 1 - m] -= h[p - m]
            A[n - 1 - m] += 1.0 - h[p - m]
        for m in range(1, n - p):
            A[n - 1 + m] -= h[p + m]
        c0 = c - h0 * A + h0 * h0 * D
        f0 = float(c0.max())
        f1 = float((c0 + A - D).max())
        gr = (np.sqrt(5.0) - 1.0) * 0.5
        a, b = 0.0, 1.0
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        fc1 = float((c0 + c1 * A - c1 * c1 * D).max())
        fd1 = float((c0 + d1 * A - d1 * d1 * D).max())
        for _ in range(35):
            if fc1 < fd1:
                b, d1, fd1 = d1, c1, fc1
                c1 = b - gr * (b - a)
                fc1 = float((c0 + c1 * A - c1 * c1 * D).max())
            else:
                a, c1, fc1 = c1, d1, fd1
                d1 = a + gr * (b - a)
                fd1 = float((c0 + d1 * A - d1 * d1 * D).max())
        ts = (a + b) * 0.5
        fs = float((c0 + ts * A - ts * ts * D).max())
        best_f = min(fs, f0, f1)
        curr_max = float(c.max())
        if best_f < curr_max - 1e-13:
            if f0 <= fs and f0 <= f1:
                tn = 0.0
            elif f1 <= fs and f1 <= f0:
                tn = 1.0
            else:
                tn = float(np.clip(ts, 0.0, 1.0))
            c += (tn - h0) * A - (tn * tn - h0 * h0) * D
            h[p] = tn
            improved = True
    if improved:
        h = project_bounded_sum(h, target)
        c2 = np.correlate(h, 1.0 - h, mode='full')
        return h, c2, True
    return h, c, False

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_bounded_sum(h_new, n_new * 0.5)
