```python
                t = float(c[:m_lo].max())
                if t > unaff_max: unaff_max = t
            if m_hi < 2*n-2:
                t = float(c[m_hi+1:].max())
                if t > unaff_max: unaff_max = t
            diag_idx = np.where(ms == (n-1))[0]
            has_diag = len(diag_idx) > 0
            lo_d, hi_d = -h0, 1.0 - h0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None]**2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, xv
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-15:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] += best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d**2
                current_c5 = max(unaff_max, float(c[ms].max()))
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def two_interval_start(n, seed):
    """Generate a two-interval step function."""
    rng = np.random.RandomState(seed)
    half = n // 2
    # Pick w1 for first interval
    w1 = rng.randint(half//4, 3*half//4)
    w2 = half - w1
    # Pick positions
    max_a = n - w1
    max_b = n - w2
    a = rng.randint(0, max(max_a, 1))
    b = rng.randint(0, max(max_b, 1))
    # Ensure no overlap
    attempts = 0
    while attempts < 100:
        overlap = 0
        if a < b + w2 and b < a + w1:
            overlap = min(a + w1, b + w2) - max(a, b)
        if overlap <= 0:
            break
        # Adjust
        if a + w1 > b:
            b = a + w1
            if b >= n - w2:
                b = n - w2
        attempts += 1
    h = np.zeros(n)
    h[a:a+w1] = 1.0
    h[b:b+w2] = 1.0
    if h.sum() != half:
        # Fix
        h = project_feasible(h, n)
    return h
```

```python
def penalty_obj(x, n, penalty):
    # Sum constraint penalty
    s = x.sum() - n / 2.0
    # Smooth max of correlation
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    # Log-sum-exp for smooth max
    alpha = 1000.0
    shifted = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    return lse * (2.0/n) + penalty * s * s

def lbfgsb_optimize(h, n, penalty=1000, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    res = minimize(penalty_obj, h, args=(n, penalty), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
    new_h = project_feasible(res.x, n)
    return new_h, compute_c5(new_h, n)
```

At the optimum, the derivative of the objective with respect to each free coordinate should be zero (or the coordinate should be at a bound). If the grid is fine enough and the quadratic refinement is accurate, we should get very close to the true optimum.

Let me try a "bisection" approach for the final phase: instead of a grid, use binary search to find the optimal value of each coordinate. This gives O(log(precision)) evaluations per coordinate instead of O(grid_size).

Actually, the 1D subproblem for each coordinate is:
min_d max_k (c[k] + d * a_k) - d^2 (if k hits the diagonal)

subject to -h0 ≤ d ≤ 1-h0

The function f(d) = max_k (c[k] + d*a_k) - d^2*has_diag is piecewise linear (or quadratic if has_diag). The minimum of a piecewise linear function on an interval can be found by checking the breakpoints.

The breakpoints occur where two lines c[k1] + d*a_k1 = c[k2] + d*a_k2, i.e., d = (c[k2]-c[k1])/(a_k1-a_k2).

For n=500, there are O(n^2) breakpoints. That's 250,000 breakpoints. Checking all of them is O(n^2) per coordinate, which is O(n^3) per pass. Too slow.

But in practice, only a few lags are near the maximum. If I track the top-M lags (say M=10), then I only need to check O(M^2) breakpoints. That's 100 breakpoints. Much faster!

```python
            # f(d) = max(unaff_max, max_k (c_aff[k] + d*a[k]) - d^2*has_diag_at_k)

            # Find the top-M affected lags
            # At d=0, the values are c_aff[k]
            # Sort by c_aff to find top-M

            # The minimum is at a breakpoint or at a boundary

            candidates = [-h0, 1.0-h0]

            for i in range(len(top_idx)):

                        # c_i + d*a_i = c_j + d*a_j - d^2 (if j has diag)
                        # d^2 + d*(a_i-a_j) + (c_i-c_j) = 0 (if j has diag, i doesn't)
                        # or d^2 + d*(a_j-a_i) + (c_j-c_i) = 0 (if i has diag, j doesn't)
                        if top_has_diag[j]:

                            B = top_a[i] - top_a[j]
                            C = top_c[i] - top_c[j]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(s - s.max()))) / alpha + s.max() / alpha
    pen = (x.sum() - n / 2.0) ** 2
    return (lse) * (2.0 / n) + 1000.0 * pen

def lbfgsb_opt(h, n, alpha=500, maxiter=200):
    h0 = h.copy().flatten()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h0, args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
        new_h = project_feasible(res.x.reshape(-1), n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def cd_h(h, n, max_passes=10000, deadline=None, grid_size=100, seed=0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi+1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                t = float(c[:m_lo].max())
                if t > unaff_max:
                    unaff_max = t
            if m_hi < 2*n-2:
                t = float(c[m_hi+1:].max())
                if t > unaff_max:
                    unaff_max = t
            diag_pos = n-1 - m_lo
            has_diag = 0 <= diag_pos < len(ms)
            lo_d, hi_d = -h0, 1.0 - h0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_pos] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_pos] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, xv
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-15:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[n-1] -= best_d * best_d
                current_c5 = max(unaff_max, float(c[ms].max()))
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def ti_sweep(n, ng=15):
    half = n // 2
    best = []
    w1s = np.linspace(max(1, half//6), half-1, ng).astype(int)
    for w1 in w1s:
        w2 = half - w1
        for a in range(0, max(1, n - w1), max(1, (n - w1) // ng)):
            for b in range(0, max(1, n - w2), max(1, (n - w2) // ng)):
                if a < b + w2 and b < a + w1:
                    continue
                h = np.zeros(n)
                h[a:a+w1] = 1.0
                h[b:b+w2] = 1.0
                if h.sum() != half:
                    continue
                cv = compute_c5(h, n)
                best.append((cv, a, b, w1))
    best.sort(key=lambda x: x[0])
    return best[:10]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:

    h = project_feasible(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B (fast)
    if time.time() < deadline - 400:

        h2, c2 = lbfgsb_opt(best_h, n, alpha=2000, maxiter=200)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 2: CD convergence
    h3, c3 = cd_h(best_h, n, max_passes=3000, deadline=t0 + 150, grid_size=100, seed=seed)
    if c3 < best_c:
        best_h, best_c = h3, c1 if c3 < c1 else c3
    best_c = compute_c5(best_h, n)

    # Phase 3: Perturbation + L-BFGS + CD
    for trial in range(12):
        if time.time() > deadline - 350:
            break
        h_p = best_h + rng.randn(n) * 0.08
        h_p = project_feasible(h_p, n)
        hp, cp = lbfgsb_opt(h_p, n, alpha=1000, maxiter=150)
        t_sub = min(time.time() + 25, deadline - 330)
        hp2, cp2 = cd_h(hp, n, max_passes=500, deadline=t_sub, grid_size=80, seed=seed+trial)
        if cp2 < best_c:
            best_h, best_c = hp2, cp2

    # Phase 4: Two-interval sweep

        if time.time() > deadline - 200:
            break
        h_p = best_h.copy()
        nf = rng.randint(3, max(4, n // 30))
        fi = rng.choice(n, size=nf, replace=False)
        h_p[fi] = 1.0 - h_p[fi]
        h_p = project_feasible(h_p, n)
        t_sub = min(time.time() + 30, deadline - 180)
        ht, ct = cd_h(h_p, n, max_passes=800, deadline=t_sub, grid_size=80, seed=seed+1000+trial)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 6: Multi-n
    n_list = [250, 300, 400, 600, 750, 1000]
    for nt in n_list:
        if time.time() > deadline - 150:
            break
        if abs(nt - len(best_h)) < 40:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, nt)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), nt)
        t_sub = min(time.time() + 50, deadline - 130)
        ht, ct = cd_h(h_int, nt, max_passes=1500, deadline=t_sub, grid_size=80, seed=seed+nt)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
