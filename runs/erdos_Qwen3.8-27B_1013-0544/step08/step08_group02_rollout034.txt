  - i in [d, n-d-1]: 1-h[i+d]-h[i-d]
- i in [n-d, n-1]: A[i,k] = -h[i-d]

- i in [0, d-1]: code gives -h[i+d], correct is 1-h[i+d]. Difference: -1
- i in [d, n-d-1]: code gives 1-h[i+d]-h[i-d], correct is 1-h[i+d]-h[i-d]. Same! ✓
- i in [n-d, n-1]: code gives 1-h[i-d], correct is -h[i-d]. Difference: +1

def reflines(c, B, u, off, i, j):
    cn = c + u * B
    cn[off] -= 2.0 * u * u
    cn[off + i - j] += u * u
    cn[off + j - i] += u * u
    return cn

            best_h, best_c = hm, cm

    # Phase 4: Final polish at best resolution
    if time.time() < hard - 10.0:
        n_best = len(best_h)
        best_h, best_c = pair_search(best_h, n_best, hard - 5.0, rng, grid=200, max_iter=100000)
        best_h, best_c = try_symmetries(best_h, n_best)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
