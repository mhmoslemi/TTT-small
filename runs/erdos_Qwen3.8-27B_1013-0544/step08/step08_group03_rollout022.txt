def project_box_sum(v, target):
    # Project v onto {0 ≤ x ≤ 1, sum(x) = target}
    # Binary search on lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

def fw_step(h, n, gamma=None):
    """One Frank-Wolfe step."""
    c = np.correlate(h, 1.0-h, mode='full')
    k_star = int(c.argmax())
    s_k = k_star - (n-1)
    g = np.zeros(n)
    i1 = np.arange(n) + s_k
    i2 = np.arange(n) - s_k
    v1 = (i1>=0) & (i1<n)
    v2 = (i2>=0) & (i2<n)
    g[v1] += (1.0 - h[i1[v1]])
    g[v2] -= h[i2[v2]]
    # Linear oracle: set h=1 where g is smallest
    order = np.argsort(g)
    s_vec = np.zeros(n)
    s_vec[order[:n//2]] = 1.0
    if gamma is None: gamma = 0.5
    h_new = (1-gamma)*h + gamma*s_vec
    return h_new, compute_c5(h_new, n)
