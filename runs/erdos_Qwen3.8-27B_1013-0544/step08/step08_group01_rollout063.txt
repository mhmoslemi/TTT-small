def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    w = np.exp(alpha * (c - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10: continue
        if m >= n - 1:
            k = m - (n - 1)
            if n - k > 0:
                g[k:n] += wm * (1.0 - x[0:n-k])
                g[0:n-k] -= wm * x[k:n]
        else:
            j = n - 1 - m
            if n - j > 0:
                g[0:n-j] += wm * (1.0 - x[j:n])
                g[j:n] -= wm * x[0:n-j]
    return g * (2.0 / n)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-16: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    w = np.exp(alpha * (c - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10: continue
        if m >= n - 1:
            k = m - (n - 1)
            if k < n:
                g[k:n] += wm * (1.0 - x[0:n-k])
                g[0:n-k] -= wm * x[k:n]
        else:
            j = n - 1 - m
            if j < n:
                g[0:n-j] += wm * (1.0 - x[j:n])
                g[j:n] -= wm * x[0:n-j]
    return g * (2.0 / n)

def penalty_obj(x, n, alpha, mu):
    return lse_obj(x, n, alpha) + mu * (x.sum() - n/2.0)**2

def penalty_grad(x, n, alpha, mu):
    return lse_grad(x, n, alpha) + 2*mu*(x.sum() - n/2.0)

def lbfgs_opt(h, n, alphas, mus, maxiter=2000, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for i, alpha in enumerate(alphas):
        if deadline and time.time() > deadline: break
        mu = mus[min(i, len(mus)-1)]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(penalty_obj, best_h,
                          args=(n, alpha, mu),
                          jac=penalty_grad,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-25,
                                   'maxfun': maxiter*3})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception: pass
    return best_h, best_c

def sum_cf(x, n): return x.sum() - n/2.0
def sum_cj(n): return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline: break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type':'eq','fun':sum_cf,'jac':sum_cj,'args':(n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha),
                          method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter':maxiter,'ftol':1e-22})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception: pass
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_cd(h, n, max_passes=300, deadline=None, grid_size=100):
    h = h.copy()
    for _ in range(max_passes):
        if deadline and time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline: break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14: continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2*n-2, 2*n-2-p)
            m_b_lo = max(0, p)
            m_b_hi = min(2*n-2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi+1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf): a_coeff[vf] += 1.0 - h[jf[vf]]
            if np.any(vb): a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0: unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff = max(unaff, float(c[m_hi+1:].max()))
            diag_local = (n-1) - m_lo if m_lo <= n-1 <= m_hi else None
            d_grid = np.linspace(-h0, 1.0-h0, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff) for d in d_grid])
            bi = int(np.argmin(vals))
            bd, bv = d_grid[bi], float(vals[bi])
            if 0 < bi < grid_size-1:
                try:
                    r = golden(obj_1d, d_grid[bi-1], d_grid[bi+1],
                              args=(c_aff, a_coeff, diag_local, unaff), maxiter=80)
                    if r.fun < bv: bd, bv = float(r.x), float(r.fun)
                except Exception: pass
            if bv < current_max - 1e-14:
                h[p] = float(np.clip(h0+bd, 0.0, 1.0))
                improved = True
                current_max = min(current_max, bv)
        if not improved: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def momentum_descent(h, n, max_iter=500, deadline=None):
    h = h.copy()
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    beta = 0.9
    eta0 = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline and time.time() > deadline: break
        eta = eta0 / (1 + it * 0.01)
        g = lse_grad(h, n, 500)  # smoothed gradient
        v = beta * v - eta * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            v = 0.5 * v
            stall += 1
            if stall > 50: break
    return h, best_c
