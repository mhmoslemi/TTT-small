def cd_fast(h, n, max_passes=100, deadline=None, use_random_order=True):

    n_corr = 2 * n - 1
    center = n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)

            if h0 < 1e-10 or h0 > 1.0 - 1e-10:

            span = min(2 * p + 1, n)
            j_range = np.arange(span)
            k_first = center - p + j_range
            i_range = np.arange(span)
            k_second = center + p - i_range

            A[k_first] += 1.0 - h[j_range]

            # D is 1 at center, 0 elsewhere
            # So f(t) = max(max_{k≠c}(c[k]+(t-h0)*A[k]), c[center]+(t-h0)*A[center]-(t^2-h0^2))
            
            # Quick evaluation at candidate points
            delta = np.array([0.0, 1.0 - h0, -h0])  # t-h0 for t=0, t=1, and...
            # Actually let me just evaluate at t=0, t=1, and a few points in between
            
            # The key insight: f as a function of t has at most O(n) linear pieces
            # (from the non-center terms) and one concave quadratic piece.
            # The minimum can be found by checking:
            # 1. t=0, t=1 (endpoints)
            # 2. Points where the quadratic crosses a linear piece
            
            # For speed, let's use a coarse grid + golden section
            
            # Evaluate at key points
            cands = np.array([0.0, 1.0, h0])
            # Add points where the parabola might be at its minimum relative to lines
            # Parabola: c[center] + (t-h0)*A[center] - (t^2-h0^2)
            #         = c[center] + (t-h0)*(1) - t^2 + h0^2  [since A[center]=1]
            #         = c[center] + t - h0 - t^2 + h0^2
            # Vertex at t = 0.5 (derivative = 1 - 2t = 0)
            if 0.0 < 0.5 < 1.0:
                cands = np.append(cands, 0.5)
            
            fvals = np.array([_eval_cd(c, A, center, h0, t) for t in cands])
            best_i = np.argmin(fvals)
            t_best = cands[best_i]
            f_best = fvals[best_i]
            
            # Golden section around the best
            if len(cands) > 1:
                sorted_c = np.sort(cands)
                # Find bracket
                if best_i == np.argmin(fvals) and 0 < best_i < len(fvals)-1:
                    gl = cands[np.argsort(fvals)[1]]  # second best
                    gh = cands[np.argsort(fvals)[-1]]  # worst
                    # Actually let's just bracket around t_best
                    span = max(0.1, (1.0 - 0.0) * 0.5)
                    gl = max(0.0, t_best - span)
                    gh = min(1.0, t_best + span)
                else:
                    gl, gh = 0.0, 1.0
            else:
                gl, gh = 0.0, 1.0
            
            # Golden section
            gr = (1.0 + np.sqrt(5.0)) / 2.0
            a, b = gl, gh
            c1 = b - (b - a) / gr
            d1 = a + (b - a) / gr
            fc1 = _eval_cd(c, A, center, h0, c1)
            fd1 = _eval_cd(c, A, center, h0, d1)
            for _ in range(25):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - (b - a) / gr; fc1 = _eval_cd(c, A, center, h0, c1)
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + (b - a) / gr; fd1 = _eval_cd(c, A, center, h0, d1)
            t_gs = (a + b) / 2.0
            f_gs = _eval_cd(c, A, center, h0, t_gs)
            
            if f_gs < f_best:
                t_new = t_gs
                f_best = f_gs
            else:
                t_new = t_best
            
            if abs(t_new - h0) < 1e-12:
                continue
            
            # Update c and h
            c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * (np.arange(n_corr) == center).astype(float)
            h[p] = t_new
            improved = True
        
        h = project_feasible(h, n)
        if not improved:
            break
    
    return h, compute_c5(h, n)

def _eval_cd(c, A, center, h0, t):
    """Evaluate f(t) for the coordinate descent subproblem."""
    dt = t - h0
    val = c + dt * A
    # Center term: subtract (t^2 - h0^2)
    val[center] -= (t * t - h0 * h0)
    return float(np.max(val))

def project_feasible(h, n):

def eval_subproblem(c, A, center, h0, t):

def cd_vec(h, n, max_passes=150, deadline=None):

    center = n - 1

def binary_local_search(h, n, max_iters=5000, deadline=None):
    """Local search on binary projection: flip pairs to reduce max correlation."""
    n_corr = 2 * n - 1
    center = n - 1
    h_bin = (h > 0.5).astype(float)
    # Fix count
    count = int(h_bin.sum())
    if count > n // 2:
        idx_ones = np.where(h_bin == 1)[0]
        excess = count - n // 2
        to_flip = idx_ones[np.random.choice(len(idx_ones), excess, replace=False)]
        h_bin[to_flip] = 0
    elif count < n // 2:
        idx_zeros = np.where(h_bin == 0)[0]
        deficit = n // 2 - count
        to_flip = idx_zeros[np.random.choice(len(idx_zeros), deficit, replace=False)]
        h_bin[to_flip] = 1

    c = np.correlate(h_bin, 1.0 - h_bin, mode='full').astype(float)
    best_c = float(np.max(c))
    D_vec = np.zeros(n_corr)
    D_vec[center] = 1.0
    temp = 0.3
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        temp = max(0.05, 0.3 * (1.0 - it / max_iters))
        # Pick a 1 to flip off and a 0 to flip on
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        if p == q:
            continue
        # Compute effect: flip h[p] from 1->0, h[q] from 0->1
        # Compute A for p (value going from 1 to 0, so delta = -1)
        span_p = min(2 * p + 1, n)
        A_p = np.zeros(n_corr)
        A_p[center - p: center - p + span_p] += 1.0 - h_bin[:span_p]
        A_p[center + p - span_p: center + p + 1] += h_bin[:span_p][::-1]
        span_q = min(2 * q + 1, n)
        A_q = np.zeros(n_corr)
        A_q[center - q: center - q + span_q] += 1.0 - h_bin[:span_q]
        A_q[center + q - span_q: center + q + 1] += h_bin[:span_q][::-1]
        # New c: c - A_p (since delta=-1 for p) + A_q (since delta=+1 for q)
        # But we need to be careful: A_p was computed with h_bin[p]=1, A_q with h_bin[q]=0
        c_new = c - A_p + A_q
        f_new = float(np.max(c_new))
        if f_new < best_c or np.random.random() < np.exp(-(f_new - best_c) / temp):
            h_bin[p] = 0
            h_bin[q] = 1
            c = c_new
            if f_new < best_c:
                best_c = f_new
    return h_bin, best_c * (2.0 / n)

def pnorm_slsqp(h, n, p, maxiter=400, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum(c ** p)) ** (1.0 / p) * dx
    def sum_con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return h, compute_c5(h, n)
