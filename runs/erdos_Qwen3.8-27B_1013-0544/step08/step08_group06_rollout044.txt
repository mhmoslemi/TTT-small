def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def swap_optimize(h, n, max_passes=200, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k_arr = np.arange(2 * n - 1)
    tau = k_arr - (n - 1)
    for p_path in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_improve = 1e-14
        best_p, best_q = -1, -1
        for p in ones_idx:
            Ap = np.where((p + tau >= 0) & (p + tau < n), h[p + tau[(p+tau>=0)&(p+tau<n)]], 0.0)
            Am = np.where((p - tau >= 0) & (p - tau < n), h[p - tau[(p-tau>=0)&(p-tau<n)]], 0.0)
            A = Ap + Am
            for q in zeros_idx:
                Bp = np.where((q + tau >= 0) & (q + tau < n), h[q + tau[(q+tau>=0)&(q+tau<n)]], 0.0)
                Bm = np.where((q - tau >= 0) & (q - tau < n), h[q - tau[(q-tau>=0)&(q-tau<n)]], 0.0)
                B = Bp + Bm
                delta = A - B
                new_max = float(np.max(c + delta))
                if new_max < current_max - best_improve:
                    best_improve = current_max - new_max
                    best_p, best_q = p, q
        if best_p >= 0:
            h[best_p] = 0.0
            h[best_q] = 1.0
            c = np.correlate(h, 1.0 - h, mode='full')
            current_max = float(c.max())
        else:
            break
    return h, current_max * (2.0 / n)

def swap_optimize_fast(h, n, max_passes=500, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    k_arr = np.arange(2 * n - 1)
    tau = k_arr - (n - 1)
    hp = np.zeros(4 * n)
    hp[n:n + n] = h
    for p_path in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_improve = 1e-14
        best_p, best_q = -1, -1
        for p in ones_idx:
            Ap = np.zeros(2 * n - 1)
            pp_tau = p + tau
            valid_p = (pp_tau >= 0) & (pp_tau < n)
            Ap[valid_p] = h[pp_tau[valid_p]]
            pm_tau = p - tau
            valid_m = (pm_tau >= 0) & (pm_tau < n)
            Ap[valid_m] += h[pm_tau[valid_m]]
            q_plus_arr = zeros_idx[:, None] + tau[None, :]
            q_minus_arr = zeros_idx[:, None] - tau[None, :]
            vp = (q_plus_arr >= 0) & (q_plus_arr < n)
            vm = (q_minus_arr >= 0) & (q_minus_arr < n)
            Bp = np.where(vp, h[np.clip(q_plus_arr, 0, n - 1)], 0.0)
            Bm = np.where(vm, h[np.clip(q_minus_arr, 0, n - 1)], 0.0)
            B = Bp + Bm
            delta = A[None, :] - B
            new_maxes = np.max(c[None, :] + delta, axis=1)
            min_new = new_maxes.min()
            if min_new < current_max - best_improve:
                best_improve = current_max - min_new
                best_p = p
                best_q = zeros_idx[int(np.argmin(new_maxes))]
        if best_p >= 0:
            h[best_p] = 0.0
            h[best_q] = 1.0
            c = np.correlate(h, 1.0 - h, mode='full')
            current_max = float(c.max())
        else:
            break
    return h, current_max * (2.0 / n)

def bimodal_push(h, n):
    h2 = (h > 0.5).astype(float)
    diff = n / 2.0 - h2.sum()
    if diff > 0:
        frac = np.sort(h[h < 0.5])[::-1]
        for i in range(int(diff)):
            if i < len(frac):
                h2[h < 0.5] = 1.0
                h2[h < 0.5] = 0.0
        h2 = project_feasible(h2, n)
    elif diff < 0:
        for i in range(int(-diff)):
            h2[h > 0.5] = 0.0
        h2 = project_feasible(h2, n)
    return h2
