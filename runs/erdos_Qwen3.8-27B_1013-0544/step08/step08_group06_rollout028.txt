import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    c = np.correlate(h, 1.0 - h, mode='full')

        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)

            vh = (jh >= 0) & (jh < n)

def subgradient_step(h, n, step_size, deadline):
    c = np.correlate(h, 1-h, mode='full')
    max_val = c.max()
    active = c >= max_val - 1e-6
    # Gradient of c[k] w.r.t. h[j] for each active k
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        m = k - (n-1)
        for j in range(n):
            # c[k] = sum_i h[i]*(1-h[i+m])
            # d/dh[j] = (1-h[j+m]) - h[j-m] (where valid)
            jm = j + m
            jg = j - m
            g = 0.0
            if 0 <= jm < n:
                g += (1 - h[jm])
            if 0 <= jg < n:
                g -= h[jg]
            grad[j] += g
    grad *= (2.0/n)
    # Projected step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha),
                method='SLSQP', bounds=[(0.0,1.0)]*n,
                constraints=[{'type':'eq','fun':sum_con,'jac':sum_con_jac,'args':(n,)}],
                options={'maxiter':maxiter,'ftol':1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c: best_h, best_c = nh, nc
        except: pass
    return best_h, best_c
