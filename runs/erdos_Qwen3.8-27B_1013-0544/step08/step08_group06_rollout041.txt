def deriv_matrix(h,n):

  h1=1-h

    t=k-mh

def project_box_sum(z, target):
    z = np.array(z, dtype=float, copy=True)
    out = np.clip(z, 0.0, 1.0)
    if abs(float(out.sum()) - target) < 1e-12:
        return out
    lo = float(out.min()) - 1.0
    hi = float(out.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(out - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(out - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - out
        else:
            cap = out
        tot = float(cap.sum())
        if tot < 1e-15:
            break
        out += (diff / tot) * cap
        np.clip(out, 0.0, 1.0, out=out)
        diff = target - float(out.sum())
    return out

def lp_step(h,n,radius):
    target=0.5*n
    h=project_box_sum(h,target)
    c=corr_counts(h)
    cmax=float(c.max())
    m=2*n-1
    J=deriv_matrix(h,n)
    A=np.empty((m,n+1),float)
    A[:,:n]=J; A[:,n]=-1.0
    b=-c
    Aeq=np.ones(n+1,float); Aeq[-1]=0.0
    lo=np.maximum(-h, -radius); hi=np.minimum(1.0-h, radius)
    bounds=[(lo[i],hi[i]) for i in range(n)]
    bounds.append((0.0, cmax+1.0))
    obj=np.zeros(n+1,float); obj[-1]=1.0
    try:
       res=linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds, method='highs')
    except Exception:
       return h,cmax
    if res.status != 0:
       return h,cmax
    y=res.x[:n]
    if float(np.abs(y).max()) < 1e-10:
       return h,cmax
    for lam in (1.0,0.75,0.5,0.35,0.25,0.15,0.08,0.04,0.02):
       cand=project_box_sum(h+lam*y,target)
       cc=corr_counts(cand)
       cm=float(cc.max())
       if cm < cmax - 1e-10:
          return cand, cm
    return h,cmax

            if 0 <= cr2 < L:
                qlocal[cr2] += 1.0

import numpy as np

def project_box_sum(z, target):
    z = np.array(z, dtype=float, copy=True)
    out = np.clip(z, 0.0, 1.0)
    if abs(float(out.sum()) - target) < 1e-12:
        return out
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    for _ in range(25):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            cap = 1.0 - out
        else:
            cap = out
        tot = float(cap.sum())
        if tot < 1e-15:
            break
        out += (diff / tot) * cap
        np.clip(out, 0.0, 1.0, out=out)
        diff = target - float(out.sum())
    return out


def corr_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h, n):
    c = corr_counts(h)
    return float(c.max()) * (2.0 / n)


def affected_lo_hi(i, n):
    mh = n - 1
    a0 = mh - i
    a1 = 2 * mh - i
    b0 = i
    b1 = mh + i
    return min(a0, b0), max(a1, b1)


def coeffs_for(h, i, ms, n):
    a = np.zeros(ms.size, dtype=float)
    s = i + ms - (n - 1)
    vs = (s >= 0) & (s < n)
    if np.any(vs):
        a[vs] += 1.0 - h[s[vs]]
    p = i - ms + (n - 1)
    vp = (p >= 0) & (p < n)
    if np.any(vp):
        a[vp] -= h[p[vp]]
    return a


def deriv_matrix(h, n):
    m = 2 * n - 1
    mh = n - 1
    J = np.zeros((m, n), dtype=float)
    h1 = 1.0 - h
    for k in range(m):
        t = k - mh
        if t >= 0:
            J[k, :n - t] = h1[t:]
            J[k, t:] -= h[:n - t]
        else:
            u = -t
            J[k, u:] = h1[:n - u]
            J[k, :n - u] -= h[u:]
    return J


def lp_step(h, n, radius):
    target = 0.5 * n
    h = project_box_sum(h, target)
    c = corr_counts(h)
    cmax = float(c.max())
    m = 2 * n - 1
    J = deriv_matrix(h, n)
    A = np.empty((m, n + 1), dtype=float)
    A[:, :n] = J
    A[:, n] = -1.0
    b = -c
    Aeq = np.ones(n + 1, dtype=float)
    Aeq[-1] = 0.0
    lo = np.maximum(-h, -radius)
    hi = np.minimum(1.0 - h, radius)
    bounds = [(float(lo[i]), float(hi[i])) for i in range(n)]
    bounds.append((0.0, float(cmax) + 1.0))
    obj = np.zeros(n + 1, dtype=float)
    obj[-1] = 1.0
    try:
        res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds,
                      options={"presolve": True})
    except Exception:
        return h, cmax
    if res.status != 0:
        return h, cmax
    y = res.x[:n]
    if float(np.abs(y).max()) < 1e-11:
        return h, cmax
    for lam in (1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02):

        cc = corr_counts(cand)

            uniq = []
            for d in cand:
                if d < lo - 1e-10 or d > hi + 1e-10:
                    continue
                if abs(d) < 1e-12:
                    d = 0.0
                dup = False
                for u in uniq:
                    if abs(d - u) < 1e-10:
                        dup = True
                        break
                if not dup:
                    uniq.append(d)
            best_d = 0.0
            best_nm = cmax
            best_seg = None
            for d in uniq:
                new_seg = c[ms] + d * slope + d * d * qlocal
                if um_out > -1e90:
                    nm = max(um_out, float(new_seg.max()))
                else:
                    nm = float(new_seg.max())
                if nm < best_nm - 1e-11:
                    best_nm = nm
                    best_d = d
                    best_seg = new_seg
            if best_seg is not None:
                c[ms] = best_seg
                h[idx] += best_d
                h[comp] -= best_d
                cmax = best_nm
                improved = True
        if not improved:
            break
    h = project_box_sum(h, target)
    c = corr_counts(h)
    return h, float(c.max())


def refine(h, n, budget):
    deadline = time.time() + max(1.0, float(budget))
    h = project_box_sum(h, 0.5 * n)
    c = corr_counts(h)
    cmax = float(c.max())
    if budget > 70:
        lp_deadline = min(deadline, time.time() + min(120.0, 0.18 * budget))
        for r in (0.12, 0.06, 0.03, 0.015, 0.008):
            if time.time() > lp_deadline - 4.0:
                break
            h2, cm2 = lp_step(h, n, r)
            if cm2 < cmax - 1e-10:
                h, cmax = h2, cm2
    steps = (0.8, 0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006)
    alloc = (0.10, 0.18, 0.20, 0.15, 0.12, 0.10, 0.08, 0.07)
    orig = max(0.0, deadline - time.time())
    for i, s in enumerate(steps):
        if orig <= 25.0:
            break
        dur = orig * alloc[i]
        if dur < 8.0:
            dur = min(orig, 8.0)
        stage_deadline = min(deadline - 1.0, time.time() + dur)
        h2, cm2 = coord_paired(h, n, stage_deadline, step=s,
                               margin=max(0.18, 2.0 * s + 0.05), grid=140)
        if cm2 < cmax - 1e-10:
            h, cmax = h2, cm2
    if time.time() < deadline - 18 and budget > 160:
        h2, cm2 = lp_step(h, n, 0.012)
        if cm2 < cmax - 1e-10:
            h, cmax = h2, cm2
        if time.time() < deadline - 8:
            h2, cm2 = coord_paired(h, n, deadline - 2.0, step=0.003,
                                   margin=0.12, grid=170)
            if cm2 < cmax - 1e-10:
                h, cmax = h2, cm2
    h = project_box_sum(h, 0.5 * n)
    c = corr_counts(h)
    return h, float(c.max())


def resample_h(h, n):
    ns = len(h)
    if ns == n:
        return project_box_sum(h, 0.5 * n)
    x0 = np.linspace(0.0, 2.0, ns)
    x1 = np.linspace(0.0, 2.0, n)
    return project_box_sum(np.interp(x1, x0, h), 0.5 * n)


def make_start(n):
    h = np.full(n, 0.5)
    return project_box_sum(h, 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s)) - 5.0
    try:
        best_h = np.array(initial_h_values, dtype=float, copy=True)
        best_n = len(best_h)
    except Exception:
        best_n = 500
        best_h = make_start(best_n)
    best_h = project_box_sum(best_h, 0.5 * best_n)
    c0 = corr_counts(best_h)
    best_cmax = float(c0.max())
    best_c = best_cmax * (2.0 / best_n)

    rem = deadline - time.time()
    if rem > 70:
        b = min(300.0, max(70.0, 0.28 * rem))
        h2, cm2 = refine(best_h, best_n, b)
        v2 = cm2 * (2.0 / best_n)
        if v2 < best_c:
            best_h, best_cmax, best_c, best_n = h2, cm2, v2, best_n

    attempts = [100, 120, 150, 200, 250, 300, 350, 400, 500, 600]
    fracs = {100: 0.13, 120: 0.11, 150: 0.10, 200: 0.10,
             250: 0.08, 300: 0.07, 350: 0.06, 400: 0.06,
             500: 0.06, 600: 0.03}
    for na in attempts:
        if time.time() > deadline - 60:
            break
        if na == best_n:
            continue
        h1 = resample_h(best_h, na)
        h2 = make_start(na)
        c1 = corr_counts(h1)
        c2 = corr_counts(h2)
        v1 = float(c1.max()) * (2.0 / na)
        v2 = float(c2.max()) * (2.0 / na)
        if v1 <= v2:
            hcur, cv = h1, v1
        else:
            hcur, cv = h2, v2
        if cv < best_c:
            best_h, best_c, best_n = hcur, cv, na
            best_cmax = float(corr_counts(best_h).max())
        rem = deadline - time.time()
        if rem < 55:
            break
        b = min(140.0, max(25.0, fracs.get(na, 0.08) * (deadline - t0)))
        if b > rem - 30:
            b = max(20.0, rem - 30)
        if b < 20:
            continue
        h3, cm3 = refine(hcur, na, b)
        v3 = cm3 * (2.0 / na)
        if v3 < best_c:
            best_h, best_cmax, best_c, best_n = h3, cm3, v3, na

    rem = deadline - time.time()
    if rem > 25:
        b = min(90.0, max(15.0, rem - 5.0))
        h2, cm2 = refine(best_h, best_n, b)
        v2 = cm2 * (2.0 / best_n)
        if v2 < best_c:
            best_h, best_cmax, best_c, best_n = h2, cm2, v2, best_n

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = c5_value(best_h, best_n)
    return best_h, best_c, best_n
