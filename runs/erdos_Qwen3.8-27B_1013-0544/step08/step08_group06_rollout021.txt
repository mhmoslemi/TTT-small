            # d >= -h0, d <= 1-h0, d >= -(1-h0) = h0-1, d <= 1-h0
            lo = max(-h0, h0 - 1.0)
            hi = min(1.0 - h0, 1.0 - h0)
            
            if hi - lo < 1e-12:
                continue
            
            # Two indices change: idx by +d, n-1-idx by -d
            idx2 = n - 1 - idx
            h0_2 = h[idx2]  # should be 1 - h0
            
            # Affected correlation indices for idx
            m_lo1 = max(0, min(n - 1 - idx, idx))
            m_hi1 = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            # Affected correlation indices for idx2
            m_lo2 = max(0, min(n - 1 - idx2, idx2))
            m_hi2 = min(2 * n - 2, max(2 * n - 2 - idx2, idx2 + n - 1))
            
            m_lo = min(m_lo1, m_lo2)
            m_hi = max(m_hi1, m_hi2)
            ms = np.arange(m_lo, m_hi + 1)
            
            # Coefficient for idx change (+d)
            a = np.zeros(len(ms))
            j1 = idx + ms - (n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            a[v1] += (1.0 - h[j1[v1]])
            j2 = idx - ms + (n - 1)
            v2 = (j2 >= 0) & (j2 < n)
            a[v2] -= h[j2[v2]]
            
            # Coefficient for idx2 change (-d)
            b = np.zeros(len(ms))
            j1b = idx2 + ms - (n - 1)
            v1b = (j1b >= 0) & (j1b < n)
            b[v1b] += (1.0 - h[j1b[v1b]])
            j2b = idx2 - ms + (n - 1)
            v2b = (j2b >= 0) & (j2b < n)
            b[v2b] -= h[j2b[v2b]]
            
            # Total coefficient
            total_coeff = a - b  # change is +d for idx, -d for idx2
            
            c_aff = c[ms].copy()
            
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            if m_hi < 2 * n - 2:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            
            # Diagonal: h[idx] and h[idx2] each have a diagonal term
            # For idx: diagonal at ms = n-1, coefficient is -d² (from h*(1-h) term)
            # For idx2: diagonal at ms = n-1, coefficient is -(-d)² = -d²
            # But wait, the diagonal term only appears when the SAME index is in both h and 1-h
            # c[n-1] = sum_i h[i]*(1-h[i]), so each h[i] contributes h[i]*(1-h[i])
            # If h[idx] changes by d: contribution changes by 2*d*(1-2*h0) - d²... 
            # Actually the change in h[idx]*(1-h[idx]) when h[idx] → h[idx]+d is:
            # (h0+d)(1-h0-d) - h0(1-h0) = h0 - h0² + d - 2*h0*d - d² - h0 + h0² = d(1-2*h0) - d²
            # Similarly for h[idx2] → h[idx2]-d: change = -d*(1-2*h[idx2]) - d²
            # Note h[idx2] = 1-h0, so 1-2*h[idx2] = 1-2+2*h0 = 2*h0-1
            # Change for idx2 = -d*(2*h0-1) - d² = d*(1-2*h0) - d²
            # Total quadratic change = 2*d*(1-2*h0) - 2*d²
            
            # But in our linear coefficient calculation, the diagonal term is already partially included.
            # Let me check: for ms = n-1 (shift 0):
            # a[n-1-m_lo]: from j1 = idx + (n-1) - (n-1) = idx → (1-h[idx]) = 1-h0
            #             from j2 = idx - (n-1) + (n-1) = idx → -h[idx] = -h0
            # So a at diagonal = (1-h0) + (-h0) = 1-2*h0 ✓
            # Similarly b at diagonal = (1-h[idx2]) + (-h[idx2]) = 1-(1-h0)-(1-h0) = 2*h0-1
            # total_coeff at diagonal = (1-2*h0) - (2*h0-1) = 2-4*h0
            
            # The quadratic terms: -d² from idx change and -(-d)² = -d² from idx2 change
            # Total: -2d²
            # But we also need to account for the cross term if idx and idx2 both appear in the same correlation element
            # For c[n-1] = sum_i h[i]*(1-h[i]): h[idx] and h[idx2] are separate terms, so no cross term.
            # For c[s] where s≠n-1: could h[idx] and h[idx2] both appear?
            # c[s] = sum_i h[i]*(1-h[i+s-(n-1)])
            # h[idx] appears as h[i] (i=idx) or as h[i+s-(n-1)] (i=idx-s+(n-1))
            # h[idx2] appears as h[i] (i=idx2) or as h[i+s-(n-1)] (i=idx2-s+(n-1))
            # For both to appear in the same c[s], we need one of idx or idx2 to be i and the other to be i+s-(n-1)
            # idx2 = idx + s - (n-1) → s = idx2 - idx + (n-1) = (n-1-idx) - idx + (n-1) = 2(n-1-idx)
            # or idx = idx2 + s - (n-1) → s = idx - idx2 + (n-1) = -(n-1-2*idx) + (n-1) = 2*idx
            # So at specific shifts, both h[idx] and h[idx2] appear in the same correlation element,
            # creating a cross term d*(-d) = -d² between them.
            
            # This is getting complicated. Let me just handle it numerically:
            # The total change in c[ms] is: d*total_coeff[ms] + quadratic_terms
            # The quadratic terms are -d² * (count of diagonal appearances)
            
            # For simplicity, let me just include the -2d² at the diagonal and check
            # if there are cross terms.
            
            diag_idx1 = ms == (n - 1)
            has_diag = np.any(diag_idx1)
            # Each index contributes -d² at the diagonal
            # So total quadratic = -2*d² at ms = n-1
            # Plus cross terms if applicable (neglecting for now, they're small)
            
            d_grid = np.linspace(lo, hi, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * total_coeff[None, :]
            if has_diag:
                nv[:, diag_idx1] -= 2.0 * d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            
            # Parabolic refinement
            if 0 < best_i < grid_size - 1:
                y0, y1, y2 = vals[best_i-1], vals[best_i], vals[best_i+1]
                x0, x1, x2 = d_grid[best_i-1], d_grid[best_i], d_grid[best_i+1]
                denom = y0*(x1-x2) + y1*(x2-x0) + y2*(x0-x1)
                if abs(denom) > 1e-15:
                    xv = 0.5*(y0*(x1-x2) + y1*(x2-x0) + y2*(x0-x1))/denom
                    if x0 < xv < x2:
                        nv1 = c_aff + xv * total_coeff
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx1] -= 2.0*xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                h[idx2] = float(np.clip(h0_2 - best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * total_coeff
                if has_diag:
                    c[ms[diag_idx1]] -= 2.0 * best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = antisymmetric_project(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    
    h = antisymmetric_project(h, n)
    return h, compute_c5(h, n)
