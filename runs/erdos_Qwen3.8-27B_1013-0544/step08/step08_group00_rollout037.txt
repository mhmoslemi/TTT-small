One idea: **alternating between "reduce max" and "equalize" modes**. In the first mode, we focus on reducing the maximum correlation. In the second mode, we try to equalize the top few correlations (bring them all to the same level). This might help because at the optimum, multiple correlations should be at the maximum.

Another idea: **use the derivative of the max to determine the best direction to move**. If the max is at a unique k*, then we want to decrease c[k*]. The direction to move h is the negative gradient of c[k*] projected onto the feasible set. This is essentially a subgradient method.

Let me implement a subgradient method as an alternative:

```python
def subgrad_step(h, n, lr):
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = c.max()
    # Find all k where c[k] is close to max
    threshold = max_val - 0.01 * max_val
    active = c > threshold
    # Gradient of c[k] w.r.t. h[p]:
    # c[k] = sum_i h[i+k-n+1]*(1-h[i])
    # dc[k]/dh[p] = (1-h[p-k+n-1]) - h[p+k-n+1] (for valid indices)
    #             (from first factor)    (from second factor)
    # For all active k, the subgradient of max is the average gradient
    grad = np.zeros(n)
    for k in range(len(c)):
        if not active[k]:
            continue
        offset = k - (n - 1)
        # h[p] appears in c[k] when:
        # first factor: p = i + offset, i.e., i = p - offset, need 0<=i<n
        # second factor: i = p, need 0<=p+offset<n
        for p in range(n):
            i1 = p - offset  # index in second factor (1-h[i1])
            i2 = p + offset  # index in first factor h[i2]
            g = 0.0
            if 0 <= i1 < n:
                g += (1.0 - h[i1])
            if 0 <= i2 < n:
                g -= h[i2]
            grad[p] += g
    grad /= active.sum()
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

                best_h, best_c = h_s, c_s
                cur_h = h_s.copy()

    # Phase 2: Thorough CD on best n
    n_best = len(best_h)
    h2, c2 = cd_optimize(best_h, n_best, max_passes=500, deadline=deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Random restarts with perturbation
    if time.time() < deadline - 150:
        n_best = len(best_h)
        for trial in range(8):
            if time.time() > deadline - 120:
                break
            if trial == 0:
                hp = best_h.copy()
            elif trial == 1:
                hp = project_feasible(best_h + rng.normal(0, 0.005, n_best), n_best)
            elif trial == 2:
                hp = project_feasible(best_h + rng.normal(0, 0.02, n_best), n_best)
            elif trial == 3:
                hp = project_feasible(best_h + rng.normal(0, 0.05, n_best), n_best)
            elif trial == 4:
                c_arr = np.correlate(best_h, 1.0 - best_h, mode='full')
                km = c_arr.argmax()
                off = km - (n_best - 1)
                grad = np.zeros(n_best)
                for i in range(n_best):
                    j1, j2 = i - off, i + off
                    if 0 <= j1 < n_best:
                        grad[i] += 1.0 - best_h[j1]
                    if 0 <= j2 < n_best:
                        grad[i] -= best_h[j2]
                mg = np.abs(grad).max()
                if mg > 1e-15:
                    hp = project_feasible(best_h - 0.03 * grad / mg, n_best)
                else:
                    hp = project_feasible(best_h + rng.normal(0, 0.01, n_best), n_best)
            elif trial == 5:
                hp = project_feasible(best_h * 0.8 + rng.uniform(0, 1, n_best) * 0.2, n_best)
            elif trial == 6:
                hp = project_feasible(rng.uniform(0, 1, n_best), n_best)
            else:
                hp = project_feasible(best_h + rng.normal(0, 0.01, n_best), n_best)
            orders = [rng.permutation(n_best) for _ in range(4)]
            h_r, c_r = cd_optimize(hp, n_best, max_passes=200, deadline=deadline, order_list=orders)
            if c_r < best_c:
                best_h, best_c = h_r, c2 if False else c_r

    # Phase 4: Try a couple of alternative n values
    if time.time() < deadline - 100:
        for n_alt in [int(n_best * 0.6), int(n_best * 1.4)]:
            if time.time() > deadline - 80:
                break
            if n_alt < 50 or n_alt == n_best or n_alt > 2000:
                continue
            x_o = np.linspace(0, 2, len(best_h))
            x_n = np.linspace(0, 2, n_alt)
            h_a = project_feasible(np.interp(x_n, x_o, best_h), n_alt)
            h_a, c_a = cd_optimize(h_a, n_alt, max_passes=200, deadline=deadline)
            if c_a < best_c:
                best_h, best_c = h_a, c_a

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        h_f, c_f = cd_optimize(best_h, n_f, max_passes=100, deadline=deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
