    if total is None:
        total = 0.5 * n

    cur = float(h.sum())
    if abs(cur - total) < 1e-13:

        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = total - float(h.sum())
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.any(free) and abs(diff) > 1e-14:
        h[free] += diff / max(1, int(np.count_nonzero(free)))
        h = np.clip(h, 0.0, 1.0)
    diff = total - float(h.sum())
    if np.any(free) and abs(diff) > 1e-16:
        idx = int(np.argmax(np.where(free, 1.0 - h, 0.0)))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def resample(h, n_new):
    n_old = len(h)
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return np.asarray(h, dtype=float).copy()
    x_old = (2.0 / n_old) * np.arange(n_old, dtype=float)
    x_new = (2.0 / n_new) * np.arange(n_new, dtype=float)
    return np.interp(x_new, x_old, np.asarray(h, dtype=float))

def corr_lse(h, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = beta * (c - m)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        w = np.zeros(c.shape, dtype=float)
        hit = (c == m)
        w[hit] = 1.0 / max(1, int(hit.sum()))
        logs = 0.0
    else:
        w = ez / s
        logs = float(np.log(s))
    return c, m, w, logs

def weighted_cross_grad(h, w):
    n = len(h)
    g = np.zeros(n, dtype=float)
    L = 2 * n - 1
    for idx in range(L):
        wi = float(w[idx])
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            l = n - k
            if l > 0:
                g[k:] += wi * (1.0 - h[:l])
                g[:l] -= wi * h[k:]
        else:
            kk = -k
            l = n - kk
            if l > 0:
                g[:l] += wi * (1.0 - h[kk:])
                g[kk:] -= wi * h[:l]
    return g

def lift_anti(v):
    m = len(v)
    h = np.empty(2 * m, dtype=float)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h

def antisym_params_from(h):
    n = len(h)
    if n % 2:
        n = max(4, n - 1)
        h = resample(h, n)
    h = np.asarray(h, dtype=float)
    m = n // 2
    v = 0.5 * (h[:m] + (1.0 - h[n - 1::-1][:m]))
    return np.clip(v, 0.0, 1.0)

def unit_bounds(n):
    return [(0.0, 1.0)] * int(n)

def anti_value(v, n, beta):
    m = len(v)
    c, mm, w, logs = corr_lse(lift_anti(v), beta)
    dx = 1.0 / m
    return float(dx * (mm + logs / beta))

def anti_grad(v, n, beta):
    m = len(v)
    h = lift_anti(v)
    c, mm, w, logs = corr_lse(h, beta)
    dx = 1.0 / m
    gh = dx * weighted_cross_grad(h, w)
    return gh[:m] - gh[::-1][:m]

def full_pen_value(x, n, beta, mu):
    dx = 2.0 / n
    c, mm, w, logs = corr_lse(x, beta)
    val = dx * (mm + logs / beta)
    diff = float(x.sum() - 0.5 * n)
    return float(val + mu * diff * diff)

def full_pen_grad(x, n, beta, mu):
    dx = 2.0 / n
    c, mm, w, logs = corr_lse(x, beta)
    g = dx * weighted_cross_grad(x, w)
    diff = float(x.sum() - 0.5 * n)
    return g + (2.0 * mu * diff) * np.ones(n, dtype=float)

def eval_anti_1d(d, c_aff, L, Q, unaff):
    val = c_aff + d * L + d * d * Q
    return max(float(val.max()), float(unaff))

def refine_anti_d(c_aff, L, Q, unaff, lo, hi, grid):
    if len(c_aff) == 0:
        return 0.0, 0.0
    dg = np.linspace(lo, hi, grid)
    d2 = dg * dg
    vals = c_aff[:, None] + L[:, None] * dg[None, :] + Q[:, None] * d2[None, :]
    vv = np.maximum(vals.max(axis=0), unaff)
    bi = int(np.argmin(vv))
    best = float(vv[bi])
    bd = float(dg[bi])
    if 0 < bi < grid - 1 and hi - lo > 1e-10:
        gl = float(dg[bi - 1])
        gr = float(dg[bi + 1])
        dg2 = np.linspace(gl, gr, 14)
        d22 = dg2 * dg2
        vals2 = c_aff[:, None] + L[:, None] * dg2[None, :] + Q[:, None] * d22[None, :]
        vv2 = np.maximum(vals2.max(axis=0), unaff)
        b2 = int(np.argmin(vv2))
        v2 = float(vv2[b2])
        if v2 < best:
            best = v2
            bd = float(dg2[b2])
            if 0 < b2 < len(dg2) - 1:
                try:
                    res = golden(eval_anti_1d, float(dg2[b2 - 1]), float(dg2[b2 + 1]),
                                 args=(c_aff, L, Q, unaff), maxiter=24)
                    if float(res.fun) < best:
                        best = float(res.fun)
                        bd = float(res.x)
                except Exception:
                    pass
    return bd, best

def coord_cd_anti(v, stop, grid=18, rng=None):
    m = len(v)
    n = 2 * m
    v = np.clip(np.asarray(v, dtype=float).copy(), 0.0, 1.0)
    if time.time() >= stop:
        return v, compute_c5(lift_anti(v))
    h = lift_anti(v)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    last_check = time.time()
    for _ in range(600):
        if time.time() >= stop:
            break
        order = np.arange(m)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for i in order:
            if time.time() > stop:
                break
            vi = float(v[i])
            if vi <= 1e-11 or vi >= 1.0 - 1e-11:
                continue
            p = i
            q = n - 1 - i
            lo = max(-(n - 1), min(p - (n - 1), -p, q - (n - 1), -q))
            hi = min(n - 1, max(p, n - 1 - p, q, n - 1 - q))
            if lo > hi:
                continue
            k_arr = np.arange(lo, hi + 1, dtype=int)
            L = np.zeros(hi - lo + 1, dtype=float)
            Q = np.zeros(hi - lo + 1, dtype=float)
            b = p - k_arr
            valid = (b >= 0) & (b < n)
            if np.any(valid):
                L[valid] += 1.0 - h[b[valid]]
                selfv = valid & (k_arr == 0)
                if np.any(selfv):
                    Q[selfv] -= 1.0
            a = p + k_arr
            valid2 = (a >= 0) & (a < n)
            if np.any(valid2):
                L[valid2] -= h[a[valid2]]
            b = q - k_arr
            valid = (b >= 0) & (b < n)
            if np.any(valid):
                L[valid] -= 1.0 - h[b[valid]]
                selfv = valid & (k_arr == 0)
                if np.any(selfv):
                    Q[selfv] -= 1.0
            a = q + k_arr
            valid2 = (a >= 0) & (a < n)
            if np.any(valid2):
                L[valid2] += h[a[valid2]]
            kpq = p - q
            if lo <= kpq <= hi:
                Q[kpq - lo] += 1.0
            kqp = q - p
            if lo <= kqp <= hi:
                Q[kqp - lo] += 1.0
            s = lo + (n - 1)
            e = hi + (n - 1)
            c_aff = c[s:e + 1]
            unaff = -1e100
            if s > 0:
                unaff = max(unaff, float(c[:s].max()))
            if e < 2 * n - 2:
                unaff = max(unaff, float(c[e + 1:].max()))
            l_d = -vi
            h_d = 1.0 - vi
            if h_d - l_d < 1e-12:
                continue
            bd, bv = refine_anti_d(c_aff, L, Q, unaff, l_d, h_d, grid)
            if bv < cur - 1e-12:
                nvi = float(np.clip(vi + bd, 0.0, 1.0))
                d = nvi - vi
                if abs(d) > 1e-15:
                    new_vals = c_aff + d * L + d * d * Q
                    v[i] = nvi
                    h[p] += d
                    h[q] -= d
                    c[s:e + 1] = new_vals
                    cur = max(float(new_vals.max()), float(unaff))
                    improved = True
        if time.time() - last_check > 0.5:
            h = lift_anti(v)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur = float(c.max())
            last_check = time.time()
        if not improved:
            break
    return v, compute_c5(lift_anti(v))

def optimize_anti(v, stop, fast=False, rng=None):
    v = np.clip(np.asarray(v, dtype=float).copy(), 0.0, 1.0)
    if time.time() >= stop:
        return v, compute_c5(lift_anti(v))
    m = len(v)
    n = 2 * m
    best_v = v.copy()
    best_c = compute_c5(lift_anti(v))
    cur_v = v.copy()
    cur_c = best_c
    bounds = unit_bounds(m)
    if fast:
        sched = [(8, 50), (60, 70), (400, 55), (2500, 35)]
    else:
        sched = [(8, 90), (40, 90), (200, 75), (1200, 55), (7000, 35)]
    for beta, it in sched:
        if time.time() > stop - 15:
            break
        try:
            res = minimize(anti_value, cur_v, args=(n, beta), jac=anti_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': it, 'ftol': 1e-13, 'gtol': 1e-6, 'maxls': 50})
            nv = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
            if not np.all(np.isfinite(nv)):
                continue
            nc = compute_c5(lift_anti(nv))
            if not np.isfinite(nc):
                continue
            cur_v = nv
            cur_c = nc
            if nc < best_c - 1e-12:
                best_c = nc
                best_v = nv.copy()
            elif nc > best_c + 0.003:
                cur_v = best_v.copy()
                cur_c = best_c
        except Exception:
            break
    g1 = 16 if fast else 20
    g2 = 24 if fast else 36
    if time.time() < stop - 30:
        vv, cc = coord_cd_anti(best_v, stop - 10, g1, rng)
        if cc < best_c - 1e-12:
            best_c = cc
            best_v = vv.copy()
        if time.time() < stop - 25:
            vv, cc = coord_cd_anti(best_v, stop - 5, g2, rng)
            if cc < best_c - 1e-12:
                best_c = cc
                best_v = vv.copy()
    if not np.array_equal(cur_v, best_v) and time.time() < stop - 10:
        vv, cc = coord_cd_anti(cur_v, stop - 5, 20, rng)
        if cc < best_c - 1e-12:
            best_c = cc
            best_v = vv.copy()
    return best_v, compute_c5(lift_anti(best_v))

def optimize_full_penalty(h, stop, fast=False):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    if time.time() >= stop or n < 2:
        return h, compute_c5(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    bounds = unit_bounds(n)
    mu = 1.0
    if fast:
        sched = [(30, 45), (600, 35)]
    else:
        sched = [(30, 70), (200, 55), (2000, 35)]
    for beta, it in sched:
        if time.time() > stop - 15:
            break
        try:
            res = minimize(full_pen_value, cur_h, args=(n, beta, mu), jac=full_pen_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': it, 'ftol': 1e-13, 'gtol': 1e-6, 'maxls': 50})
            nx = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(nx)):
                continue
            nh = project_feasible(nx)
            nc = compute_c5(nh)
            if not np.isfinite(nc):
                continue
            cur_h = nh
            cur_c = nc
            if nc < best_c - 1e-12:
                best_c = nc
                best_h = nh.copy()
            elif nc > best_c + 0.003:
                cur_h = best_h.copy()
                cur_c = best_c
        except Exception:
            break
    return best_h, compute_c5(best_h)

def random_anti(n, rng):
    n = int(max(4, n))
    if n % 2:
        n -= 1
    m = n // 2
    x = np.linspace(0.0, 1.0, m, endpoint=False)
    v = 0.5 * np.ones(m, dtype=float)
    for f in range(1, 6):
        v += (0.22 / float(f + 0.5)) * np.sin(2.0 * np.pi * f * x + rng.uniform(0.0, 2.0 * np.pi))
    v += rng.normal(0.0, 0.03, m)
    return np.clip(v, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    hard = t0 + max(60.0, budget) - 10.0
    rng = np.random.RandomState(seed)

    try:
        ph = np.asarray(initial_h_values, dtype=float)
        if ph.ndim != 1 or ph.size < 2:
            raise ValueError
    except Exception:
        ph = np.full(400, 0.5, dtype=float)
    ph = project_feasible(ph)
    if ph.size > 900:
        ph = project_feasible(resample(ph, 898))

    best_h = ph.copy()
    best_c = compute_c5(best_h)

    try:
        n_base = len(ph) if (len(ph) % 2 == 0) else max(4, len(ph) - 1)
        n_base = min(n_base, 900)
        if n_base % 2:
            n_base -= 1
        ph_base = project_feasible(resample(ph, n_base))
        cb = compute_c5(ph_base)
        if cb < best_c:
            best_h = ph_base.copy()
            best_c = cb

        v0 = antisym_params_from(ph_base)
        c0 = compute_c5(lift_anti(v0))
        if c0 < best_c:
            best_h = lift_anti(v0).copy()
            best_c = c0

        stop1 = hard - 430 if hard - time.time() > 430 else max(time.time() + 5.0, hard - 80)
        if time.time() < stop1:
            v1, c1 = optimize_anti(v0, stop1, fast=False, rng=rng)
            if c1 < best_c:
                best_h = lift_anti(v1).copy()
                best_c = c1

        if time.time() < hard - 250:
            vp_stop = hard - 210 if hard - time.time() > 210 else max(time.time() + 5.0, hard - 45)
            if time.time() < vp_stop:
                vb = antisym_params_from(resample(best_h, n_base))
                vp = np.clip(vb + 0.03 * rng.normal(size=vb.size), 0.0, 1.0)
                vp, cp = optimize_anti(vp, vp_stop, fast=True, rng=rng)
                if cp < best_c:
                    best_h = lift_anti(vp).copy()
                    best_c = cp

        for n_try in (320, 440, 560):
            if time.time() > hard - 190:
                break
            try_stop = hard - 150 if hard - time.time() > 150 else max(time.time() + 5.0, hard - 35)
            if time.time() < try_stop:
                vtry = antisym_params_from(resample(best_h, n_try))
                vtry, ctry = optimize_anti(vtry, try_stop, fast=True, rng=rng)
                if ctry < best_c:
                    best_h = lift_anti(vtry).copy()
                    best_c = ctry

        if time.time() < hard - 140:
            rz_stop = hard - 110 if hard - time.time() > 110 else max(time.time() + 5.0, hard - 30)
            if time.time() < rz_stop:
                vr = random_anti(360, rng)
                vr, cr = optimize_anti(vr, rz_stop, fast=True, rng=rng)
                if cr < best_c:
                    best_h = lift_anti(vr).copy()
                    best_c = cr

        if time.time() < hard - 120:
            fp_stop = hard - 90 if hard - time.time() > 90 else max(time.time() + 5.0, hard - 25)
            if time.time() < fp_stop:
                hf = project_feasible(resample(best_h, n_base))
                hf, cf = optimize_full_penalty(hf, fp_stop, fast=True)
                if cf < best_c:
                    best_h = hf.copy()
                    best_c = cf

        if time.time() < hard - 45:
            fin_stop = hard - 20 if hard - time.time() > 45 else max(time.time() + 2.0, hard - 10)
            if time.time() < fin_stop:
                hf = project_feasible(resample(best_h, n_base))
                vf = antisym_params_from(hf)
                vf, cfv = optimize_anti(vf, fin_stop, fast=True, rng=rng)
                if cfv < best_c:
                    best_h = lift_anti(vf).copy()
                    best_c = cfv

        if time.time() < hard - 25:
            f2_stop = max(time.time() + 2.0, hard - 10)
            if time.time() < f2_stop:
                hf = project_feasible(resample(best_h, n_base))
                hf, cf = optimize_full_penalty(hf, f2_stop, fast=True)
                if cf < best_c:
                    best_h = hf.copy()
                    best_c = cf

    except Exception:
        pass

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
