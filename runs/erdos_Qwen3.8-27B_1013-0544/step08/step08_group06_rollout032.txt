            # j+k-(n-1) = j-k+(n-1), i.e., k = n-1.

            #   Grand total at k=n-1: (1-2h[idx]) + (1-2h[idx]) = 2(1-2h[idx])
            #   This is still linear in da (no da² term), which is correct because the two
            #   positions are different (idx ≠ idx+nh).

            # This is the max of linear functions of da, hence convex in da.
            
            # Find the minimum using grid + refinement
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            vals = np.maximum(unaff_max, (c_aff[None,:] + d_grid[:,None]*a[None,:]).max(axis=1))

            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            
            # Parabolic refinement
            if 0 < best_i < grid_size-1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])

                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom

                    current_c5 = new_max

            break

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-10
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def make_two_interval(n, L1, gap1):
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    m1 = (x >= gap1) & (x < gap1 + L1)
    m2 = (x >= 1.0 + L1) & (x <= 2.0)
    h[m1 | m2] = 1.0
    return project_feasible(h, n)
