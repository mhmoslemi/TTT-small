    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

def gps_flat(h, n, n_iter=1000, betas=None):
    """Gerchberg-Saxton for flat power spectrum"""
    if betas is None:
        betas = [5.0, 3.0, 2.0, 1.5, 1.0, 0.8]
    h = h.copy()

            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

                c[ms] = ca + best_d * a
                if diag.any():
                    c[ms[diag]] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.4 * np.sin(4 * np.pi * x)
    elif kind == 2:
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif kind == 3:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(6 * np.pi * x)
    elif kind == 4:
        h = 0.5 + 0.2 * np.sin(2 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x) + 0.1 * np.sin(10 * np.pi * x)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = structured_start(n_init, 0, rng)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: GPS flat spectrum to escape local optimum
    if time.time() < deadline - 400:
        h1 = gps_flat(best_h, n_init, n_iter=3000, betas=[10.0, 5.0, 3.0, 2.0, 1.5])
        h1, c1 = cd_golden(h1, n_init, 200, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: L-BFGS with high alpha
    if time.time() < deadline - 300:
        h2, c2 = lbfgs_opt(best_h, n_init, [50000, 200000], 600, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = cd_golden(best_h, n_init, 200, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Multi-scale optimization
    if time.time() < deadline - 250:
        for ns in [150, 250, 350]:
            if time.time() > deadline - 120:
                break
            xs = np.linspace(0, 2, ns, endpoint=False)
            xb = np.linspace(0, 2, len(best_h), endpoint=False)
            hs = project_feasible(np.interp(xs, xb, best_h), ns)
            hs = gps_flat(hs, ns, n_iter=1000, betas=[5.0, 2.0, 1.0])
            hs, cs = cd_golden(hs, ns, 300, deadline)
            hb = project_feasible(np.interp(xb, xs, hs), len(best_h))
            hb, cb = cd_golden(hb, len(best_h), 200, deadline)
            if cb < best_c:
                best_h, best_c = hb, cb

    # Phase 4: Random-order CD
    if time.time() < deadline - 80:
        h4, c4 = cd_golden(best_h, n_init, 400, deadline, random_order=True, seed=777)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Diverse restarts with GPS
    if time.time() < deadline - 100:
        for r in range(8):
            if time.time() > deadline - 40:
                break
            if r < 3:
                hp = structured_start(n_init, r, rng)
                hp = gps_flat(hp, n_init, n_iter=1500, betas=[5.0, 2.0, 1.0])
            else:
                s = 0.02 + 0.06 * (r / 8.0)
                hp = project_feasible(best_h + rng.normal(0, s, n_init), n_init)
            ho, co = cd_golden(hp, n_init, 200, deadline)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 6: Final polish - GPS + CD
    if time.time() < deadline - 15:
        h6 = gps_flat(best_h, n_init, n_iter=800, betas=[2.0, 1.5, 1.0])
        h6, c6 = cd_golden(h6, n_init, 200, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Final sequential CD polish
    if time.time() < deadline - 3:
        hf, cf = cd_golden(best_h, n_init, 150, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
