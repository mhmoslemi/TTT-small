import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + (target - s) / tot * room
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n, deadline, alpha=1.0):
    """One pass of exact coordinate descent."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    d = n - 1
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue

        # A[k] for k in 0..2n-2
        A = np.zeros(2*n-1)

        # First term: (1-h[p+d]) for d in [-p, n-1-p]
        d_lo1 = max(-(n-1), -p)
        d_hi1 = min(n-1, n-1-p)
        if d_lo1 <= d_hi1:
            k1 = (n-1) + np.arange(d_lo1, d_hi1+1)
            j1 = p + np.arange(d_lo1, d_hi1+1)
            A[k1] += (1.0 - h[j1])
        # Second term: -h[p-d] for d in [p-(n-1), p]
        d_lo2 = max(-(n-1), p-(n-1))
        d_hi2 = min(n-1, p)
        if d_lo2 <= d_hi2:
            k2 = (n-1) + np.arange(d_lo2, d_hi2+1)
            j2 = p - np.arange(d_lo2, d_hi2+1)
            A[k2] -= h[j2]
        # D: quadratic term only at k=n-1 (d=0)
        D = np.zeros(2*n-1)
        D[d] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:
            h = h + (target - s) / tot * room
    elif s > target:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    d = n - 1
    D = np.zeros(2 * n - 1)
    D[d] = 1.0
    j = np.arange(n)
    improved = False
    for p in range(n):
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue
        A = np.zeros(2 * n - 1)
        k1 = j + (d - p)
        k2 = (p + d) - j
        A[k1] += (1.0 - h[j])
        A[k2] -= h[j]
        c0 = c - h0 * A + h0 * h0 * D
        gr = 0.618033988749895
        a, b = 0.0, 1.0
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
        fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
        for _ in range(28):
            if fc1 < fd1:
                b, d1, fd1 = d1, c1, fc1
                c1 = b - gr * (b - a)
                fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
            else:
                a, c1, fc1 = c1, d1, fd1
                d1 = a + gr * (b - a)
                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
        t_star = (a + b) * 0.5
        f_star = float(np.max(c0 + t_star * A - t_star * t_star * D))
        f0 = float(np.max(c0))
        f1 = float(np.max(c0 + A - D))
        cur = float(np.max(c))
        if f0 < cur - 1e-13:
            t_new = 0.0
        elif f1 < cur - 1e-13:
            t_new = 1.0
        elif f_star < cur - 1e-13:
            t_new = float(np.clip(t_star, 0.0, 1.0))
        else:
            continue
        delta = t_new - h0
        c += delta * A - (t_new * t_new - h0 * h0) * D
        h[p] = t_new
        improved = True
    h = project_feasible(h, n)
    return h, improved
