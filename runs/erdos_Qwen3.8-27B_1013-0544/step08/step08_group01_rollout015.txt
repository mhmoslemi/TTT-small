```python
def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    # Find min of max(c_aff[k] + d*a_coeff[k] - d^2*(k==diag), unaff_max) for d in [lo, hi]
    ms = len(c_aff)
    candidates = [lo, hi]
    
    # Pairwise intersections of affine terms
    for i in range(ms):
        for j in range(i+1, ms):
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-15:
                d = (c_aff[j] - c_aff[i]) / da
                if lo <= d <= hi:
                    candidates.append(d)
    
    # Intersection of affine with quadratic (if diag exists)
    if diag_local is not None:
        cd = c_aff[diag_local]
        for i in range(ms):
            if i == diag_local:
                continue
            # c_aff[i] + d*a_coeff[i] = cd - d^2
            # d^2 + a_coeff[i]*d + (c_aff[i] - cd) = 0
            A = 1.0
            B = a_coeff[i]
            C = c_aff[i] - cd
            disc = B*B - 4*A*C
            if disc >= 0:
                sqrt_disc = np.sqrt(disc)
                d1 = (-B + sqrt_disc) / (2*A)
                d2 = (-B - sqrt_disc) / (2*A)
                if lo <= d1 <= hi:
                    candidates.append(d1)
                if lo <= d2 <= hi:
                    candidates.append(d2)
    
    best_d = lo
    best_val = float('inf')
    for d in candidates:
        d = float(np.clip(d, lo, hi))
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        val = float(nv.max())
        val = max(val, unaff_max)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d, best_val
```

```python
def project_water_filling(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Need to reduce: find t such that sum(clip(h - t, 0, 1)) = target
        # Since h is already in [0,1], clip(h-t, 0, 1) = max(0, min(1, h-t))
        # For t in [0,1]: max(0, h-t) for h>=t, and 0 for h<t
        # sum = sum_{h[i]>=t} (h[i]-t) = sum_{h[i]>=t} h[i] - t * count(h[i]>=t)
        # This is a piecewise linear function of t, decreasing.
        sorted_h = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_h)
        # At threshold t = sorted_h[j], the sum is cumsum[j] - t*(j+1)
        # We need cumsum[j] - t*(j+1) = target
        # t = (cumsum[j] - target) / (j+1)
        # We need t to be in (sorted_h[j+1], sorted_h[j]] (0-indexed: sorted_h[j] >= t > sorted_h[j+1])
        for j in range(len(sorted_h)):
            t = (cumsum[j] - target) / (j + 1)
            if j == len(sorted_h) - 1 or t > sorted_h[j+1]:
                break
        h = np.maximum(0, h - t)
    else:
        # Need to increase: find t such that sum(clip(h + t, 0, 1)) = target
        sorted_h = np.sort(h)
        cumsum_from_top = np.cumsum(1.0 - sorted_h[::-1])[::-1]
```

    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_exact(c_aff, a_coeff, diag_local, unaff_max, lo, hi, top_k=40):
    ms = len(c_aff)
    if ms == 0:
        return 0.0, unaff_max
    cur_vals = c_aff.copy()
    if ms > top_k:
        top_idx = np.argsort(-cur_vals)[:top_k]
    else:
        top_idx = np.arange(ms)
    cands = [lo, hi]
    ntop = len(top_idx)
    for ii in range(ntop):
        for jj in range(ii + 1, ntop):
            i, j = int(top_idx[ii]), int(top_idx[jj])
            da = a_coeff[i] - a_coeff[j]
            if abs(da) > 1e-14:
                d = (c_aff[j] - c_aff[i]) / da
                if lo - 1e-10 <= d <= hi + 1e-10:
                    cands.append(float(np.clip(d, lo, hi)))
    if diag_local is not None:
        cd = c_aff[diag_local]
        for i in top_idx:
            i = int(i)
            if i == diag_local:
                continue
            B = a_coeff[i]
            C = c_aff[i] - cd
            disc = B * B - 4.0 * C
            if disc >= 0:
                sd = np.sqrt(disc)
                d1 = (-B + sd) * 0.5
                d2 = (-B - sd) * 0.5
                if lo - 1e-10 <= d1 <= hi + 1e-10:
                    cands.append(float(np.clip(d1, lo, hi)))
                if lo - 1e-10 <= d2 <= hi + 1e-10:
                    cands.append(float(np.clip(d2, lo, hi)))
    best_d = 0.5 * (lo + hi)
    best_val = float('inf')
    for d in cands:
        d = float(np.clip(d, lo, hi))
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        val = float(nv.max())
        if val < unaff_max:
            val = unaff_max
        if val < best_val:
            best_val = val
            best_d = d
    return best_d, best_val

def func_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)
