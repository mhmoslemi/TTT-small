    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def exact_cd(h, n, max_passes, deadline, bisect_iters=40, shuffle_order=False, rng=None):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if shuffle_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_base = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(np.max(c[:m_lo])))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_hi + 1:])))
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            c_diag = c_base[diag_idx[0]] if has_diag else 0.0
            if has_diag:
                a[diag_idx] = 0.0
            neg = a < -1e-15
            pos = a > 1e-15
            zero = np.abs(a) <= 1e-15
            L_lo = unaff
            L_hi = float(np.max(c))
            for _b in range(bisect_iters):
                L = (L_lo + L_hi) / 2.0
                x_lo = 0.0
                x_hi = 1.0
                feasible = True
                if np.any(zero) and np.any(c_base[zero] > L + 1e-15):
                    feasible = False
                if feasible and np.any(neg):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
                if feasible and np.any(pos):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
                if feasible and has_diag:
                    disc = (1.0 - 2.0 * h0) ** 2 - 4.0 * (c_diag - L)
                    if disc < 0:
                        feasible = False
                    else:
                        sq = np.sqrt(disc)
                        r1 = ((1.0 - 2.0 * h0) - sq) / 2.0
                        r2 = ((1.0 - 2.0 * h0) + sq) / 2.0
                        x_lo = max(x_lo, r1)
                        x_hi = min(x_hi, r2)
                if x_lo > x_hi + 1e-13:
                    feasible = False
                if feasible:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo = 0.0
            x_hi = 1.0
            if np.any(neg):
                x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
            if np.any(pos):
                x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
            if has_diag:
                disc = (1.0 - 2.0 * h0) ** 2 - 4.0 * (c_diag - L)
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = ((1.0 - 2.0 * h0) - sq) / 2.0
                    r2 = ((1.0 - 2.0 * h0) + sq) / 2.0
                    x_lo = max(x_lo, r1)
                    x_hi = min(x_hi, r2)
            x_new = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alphas, maxiter):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def block_cd(h, n, block_size, max_passes, deadline, rng):
    """Block coordinate descent: optimize blocks of consecutive variables."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        c = np.correlate(h, 1.0 - h, mode='full')
        for start in range(0, n, block_size):
            if time.time() > deadline:
                break
            end = min(start + block_size, n)
            idx = np.arange(start, end)
            h0 = h[idx].copy()
            fixed_h = h.copy()
            def block_obj(d):
                ht = fixed_h.copy()
                ht[idx] = np.clip(h0 + d, 0.0, 1.0)
                ct = np.correlate(ht, 1.0 - ht, mode='full')
                return ct.max()
            try:
                bounds = [(-h0[i], 1.0 - h0[i])] for i in range(len(idx))
                res = minimize(block_obj, np.zeros(len(idx)), method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
                new_vals = np.clip(h0 + res.x, 0.0, 1.0)
                if res.fun < c.max() * (2.0 / n) - 1e-10:
                    h[idx] = new_vals
                    improved = True
            except Exception:
                pass
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(200, 0.5)
        n = 200
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Exact CD convergence
    if time.time() < deadline - 80:
        h1, c1 = exact_cd(h, n, 100, deadline - 80, bisect_iters=40)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP polish
    if time.time() < deadline - 50:
        h2, c2 = slsqp_opt(best_h, n, [50000, 200000, 1000000], 800)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: More exact CD with shuffled order
    if time.time() < deadline - 30:
        h3, c3 = exact_cd(best_h, n, 60, deadline - 30, bisect_iters=45,
                          shuffle_order=True, rng=rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 800, 400, 900, 1000, 600]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_int, c_int = exact_cd(h_int, n_try, 50, deadline - 120, bisect_iters=35)
        if c_int < best_c:
            best_h, best_c = h_int, c_int
            x_orig = np.linspace(0, 2, n_try)

    # Phase 5: Multi-restart with diverse perturbations
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 60
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r % 4 == 0:
                s = 0.005 + 0.03 * (r / max(1, n_restarts))
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            elif r % 4 == 1:
                hp = best_h.copy()
                for _ in range(5):
                    st = rng.randint(0, max(1, n_cur - 30))
                    ln = rng.randint(5, 30)
                    hp[st:st + ln] = rng.uniform(0, 1, ln)
                hp = project_feasible(hp, n_cur)
            elif r % 4 == 2:
                hp = project_feasible(rng.uniform(0, 1, n_cur), n_cur)
            else:
                # Mirror perturbation
                hp = best_h.copy()
                mid = n_cur // 2
                hp[mid:] = 1.0 -best_h[:mid + (n_cur % 2)][::-1]
                hp[:n_cur - mid] = 1.0 - best_h[n_cur - mid + (n_cur % 2):][::-1]
                hp = project_feasible(hp, n_cur)
            hp, cp = exact_cd(hp, n_cur, 40, deadline - 60, bisect_iters=35,
                              shuffle_order=(r % 2 == 0), rng=rng)
            if cp < best_c:
                best_h, best_c = hp, cp

    # Phase 6: Final high-precision CD
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        remaining = deadline - time.time() - 4
        max_p = max(5, int(remaining / 1.5))
        hf, cf = exact_cd(best_h, n_fin, max_p, deadline - 3, bisect_iters=50)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
