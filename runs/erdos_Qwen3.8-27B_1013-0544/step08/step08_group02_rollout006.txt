But h is defined on [0,2], so h(x+k) is only defined for x+k ∈ [0,2]. In the discretization, the "full" correlation handles this by zero-padding.

So C₅ = max_k c(k) where c(k) are the full correlation values scaled by dx.

The key observation is: C₅ = 1 - min_k R(k) where R(k) is the autocorrelation (with appropriate normalization).

c_full[k] = Σ_i h[i] * (1-h[i+k]) for appropriate ranges, times dx.

For k=0: c_full[off] = Σ_i h[i](1-h[i]) * dx = (1 - Σ_i h[i]²) * dx = (1 - Σ_i h[i]²) * 2/n

For k≠0 (say k corresponds to shift d>0): c_full[off+d] = Σ_{i=0}^{n-1-d} h[i](1-h[i+d]) * dx

So C₅ = max(1 - Σh[i]², max_{d≠0} Σ_{i} h[i](1-h[i+d])) * 2/n

To minimize C₅, we want to:
1. Maximize Σh[i]² (make h as binary as possible)
2. Minimize the cross-correlations Σ h[i](1-h[i+d]) for d≠0

These two goals are in tension. The optimal solution balances them.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    low = float(z.min()) - 1.0
    high = float(z.max()) + 1.0
    for _ in range(58):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-12:
        mask = y < 1.0 if res > 0 else y > 0.0
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-14:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def smooth_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    return (m + np.log(s) / alpha) * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    p = np.exp(z - zm)
    p /= p.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-13:
            continue
        d = k - off
        if d >= 0:
            ml = n - d
            g[:ml] += w * (1.0 - h[d:ml])
            g[d:] -= w * h[:ml]
        else:
            e_ = -d
            ml = n - e_
            g[:ml] += w * (1.0 - h[e_:])
            g[e_:] -= w * h[:ml]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x)

def pen_val(h, n, alpha, mu):
    v = smooth_val(h, n, alpha)
    dev = h.sum() - 0.5 * n
    return v + 0.5 * mu * dev * dev

def pen_grad(h, n, alpha, mu):
    g = smooth_grad(h, n, alpha)
    dev = h.sum() - 0.5 * n
    g += mu * dev
    return g

def slsqp_opt(h, n, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
                work = cur
        except Exception:
            pass
    return best_h, best_c

def lbfgs_opt(h, n, plans, mu, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(pen_val, work, args=(n, alpha, mu), method='L-BFGS-B',
                           jac=pen_grad, bounds=bounds,

            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:

                work = cur
        except Exception:
            pass
    return best_h, best_c

def pair_search(h, n, max_tries, deadline, rng, allow_neutral=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    cur_max = float(c.max())
    best_c = cur_max * (2.0 / n)
    accepts = 0
    no_improve = 0
    
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.015, 0.001 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e_ = -d
                m = n - e_
                if m <= 0:
                    continue
                b = e_ + int(rng.randint(m))
                i, j = b - e_, b
        
        # Find optimal step for pair (i,j)
        old_sum = h[i] + h[j]
        lo = max(0.0, old_sum - 1.0)
        hi = min(1.0, old_sum)
        if hi - lo <= 1e-14:
            continue
        
        # Compute B = gradient direction for this pair
        # c_new = c + u*(A[i]-A[j]) where A is the "influence matrix"
        # For pair (i,j): changing h[i] by +u and h[j] by -u
        # Delta c_k = u * (dc_k/dh_i - dc_k/dh_j)
        B = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            d = k - off
            di = 0.0
            dj = 0.0
            if d >= 0:
                ml = n - d
                if 0 <= i < ml:
                    di = 1.0 - h[i + d]
                if i >= d:
                    di += -h[i - d]
                if 0 <= j < ml:
                    dj = 1.0 - h[j + d]
                if j >= d:
                    dj += -h[j - d]
            else:
                e_ = -d
                ml = n - e_
                if 0 <= i - e_ < ml:
                    di = 1.0 - h[i]
                if i >= e_:
                    di += -h[i - e_]
                if 0 <= j - e_ < ml:
                    dj = 1.0 - h[j]
                if j >= e_:
                    dj += -h[j - e_]
            B[k] = di - dj
        
        # c_new[k] = c[k] + u*B[k]
        # Also the k=off term gets a quadratic correction: -2u^2 (from h[i]^2 term)
        # Actually for the k=off-i+j and k=off+j-i terms there are also quadratic terms
        # Let's just do a 1D golden section search
        
        best_u = 0.0
        best_cn = cur_max
        a = lo - h[i]
        b = hi - h[i]
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        x1 = a + gr * (b - a)
        x2 = a + (1.0 - gr) * (b - a)
        
        def f_at(u):
            cn = c + u * B
            cn[off] -= 2.0 * u * u
            if abs(off + i - j) < 2 * n:
                cn[off + i - j] += u * u
            if abs(off + j - i) < 2 * n:
                cn[off + j - i] += u * u
            return float(cn.max())
        
        f1 = f_at(x1)
        f2 = f_at(x2)
        for _ in range(40):
            if f1 < f2:
                b = x2
                x2 = x1
                f2 = f1
                x1 = a + gr * (b - a)
                f1 = f_at(x1)
            else:
                a = x1
                x1 = x2
                f1 = f2
                x2 = a + (1.0 - gr) * (b - a)
                f2 = f_at(x2)
        
        best_u = 0.5 * (a + b)
        best_cn = f_at(best_u)
        
        # Check constraints
        if best_cn < cur_max - 1e-10 or (allow_neutral and best_cn <= cur_max + 1e-8):
            new_hi = h[i] + best_u
            new_hi = float(np.clip(new_hi, 0.0, 1.0))
            new_hj = old_sum - new_hi
            if new_hj < -1e-9 or new_hj > 1.0 + 1e-9:
                continue
            new_hj = float(np.clip(new_hj, 0.0, 1.0))
            # Recompute actual
            h[i] = new_hi
            h[j] = new_hj
            c = np.correlate(h, 1.0 - h, mode='full')
            actual_max = float(c.max())
            if actual_max < cur_max - 1e-10:
                accepts += 1
                no_improve = 0
                if actual_max * (2.0 / n) < best_c:
                    best_c = actual_max * (2.0 / n)
                if accepts % 50 == 0:
                    c = np.correlate(h, 1.0 - h, mode='full')
            elif actual_max > cur_max + 1e-6:
                # Revert
                h[i] = new_hi  # already set, revert
                # We need to restore
                h[i] = float(np.clip(h[i] - best_u, 0.0, 1.0))
                h[j] = float(np.clip(new_hj + best_u, 0.0, 1.0))
                c = np.correlate(h, 1.0 - h, mode='full')
                no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if accepts > 200 and no_improve > 500:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.5 * a * np.cos(np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_full(h, n, deadline, rng, aggressive=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    # Symmetries
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    if compute_c5(hs, n) < c:
        h, c = hs, compute_c5(hs, n)
    hc = 1.0 - h
    if compute_c5(hc, n) < c:
        h, c = hc, compute_c5(hc, n)
    
    # L-BFGS-B phase
