    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Polyak step size: need an upper bound on optimal value
    # Lower bound: C5 >= 0 (trivial)
    # Upper bound: our current value
    c_opt_lower = 0.0
    c_opt_upper = best_c
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c5_val, grad = c5_and_subgrad(h, n)
        if c5_val < best_c - 1e-12:
            best_c = c5_val
            best_h = h.copy()

    """Smooth max of correlation values + penalty for sum constraint."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0-h, mode='full') * dx

    c_max = c.max()

    dx = 2.0 / n

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
