def exact_1d_min(c_affected, a_vec, unaff_max, lo_d, hi_d, diag_pos):
    """Find d in [lo_d, hi_d] minimizing max(unaff_max, max_i(c_affected[i] + d*a_vec[i] - d^2*(i==diag_pos)))
    
    Returns (d_opt, opt_value)
    """
    M = len(c_affected)
    
    if unaff_max == 0 and M == 0:
        return 0.0, 0.0
    
    # Binary search on T
    T_hi = max(unaff_max, float(np.max(c_affected)))  # value at d=0
    T_lo = max(0.0, unaff_max - abs(lo_d)*np.max(np.abs(a_vec)) - 1)  # rough lower bound
    T_lo = max(0.0, min(float(np.min(c_affected + lo_d*a_vec - (lo_d**2 if diag_pos is not None and ...)))))
    
    # Simpler: just use T_lo = 0
    T_lo = 0.0
    
    def check(T):
        if T < unaff_max:
            return False, 0.0
        L = lo_d
        U = hi_d
        for i in range(M):
            if i == diag_pos:
                continue
            ck = c_affected[i]
            ai = a_vec[i]
            if ck > T + 1e-15:
                return False, 0.0
            if ai > 1e-15:
                U = min(U, (T - ck) / ai)
            elif ai < -1e-15:
                L = max(L, (T - ck) / ai)
            # else: ck ≤ T already checked
        if L > U + 1e-15:
            return False, 0.0
        
        # Diagonal constraint
        if diag_pos is not None:
            cd = c_affected[diag_pos]
            ad = a_vec[diag_pos]
            # d^2 - ad*d + (T - cd) >= 0
            disc = ad*ad - 4.0*(T - cd)
            if disc >= 0:
                sqrt_disc = np.sqrt(disc)
                r1 = (ad - sqrt_disc) / 2.0
                r2 = (ad + sqrt_disc) / 2.0
                # Need d <= r1 or d >= r2
                # Current: [L, U]
                # Option 1: [L, min(U, r1)]
                # Option 2: [max(L, r2), U]
                opt1_u = min(U, r1)
                opt2_l = max(L, r2)
                if opt1_u < L and opt2_l > U:
                    return False, 0.0
                elif opt1_u < L:
                    L = opt2_l
                elif opt2_l > U:
                    U = opt1_u
                else:
                    # Two intervals, pick the one containing the "center" or the shorter one
                    # For minimization, we want the tightest T, so either works
                    # Pick the one with smaller width for better conditioning
                    w1 = opt1_u - L
                    w2 = U - opt2_l
                    if w1 <= w2:
                        U = opt1_u
                    else:
                        L = opt2_l
        
        if L > U + 1e-15:
            return False, 0.0
        return True, (L + U) / 2.0
    
    for _ in range(80):
        T_mid = (T_lo + T_hi) / 2.0
        ok, d_candidate = check(T_mid)
        if ok:
            T_hi = T_mid
        else:
            T_lo = T_mid
    
    ok, d_opt = check(T_hi + 1e-12)
    if not ok:
        d_opt = 0.0
    
    # Compute actual value at d_opt
    vals = c_affected + d_opt * a_vec
    if diag_pos is not None:
        vals[diag_pos] -= d_opt * d_opt
    opt_val = max(unaff_max, float(np.max(vals)))
    
    return d_opt, opt_val

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
