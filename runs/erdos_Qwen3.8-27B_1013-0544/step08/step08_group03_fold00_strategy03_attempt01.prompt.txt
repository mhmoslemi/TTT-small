def perturb_direction(h):
"""Generate signal-level rather than pixel-level modifications"""
    # Strategy: Add sinusoidal components and binary flips
    new_h = h.copy()
    # Additive sinusoids with random amplitude and frequency
    n = len(h)
    for _ in range(2):
        f1 = 3 + np.random.randint(5)
        phase1 = np.random.random() * 2*np.pi
        amp1 = 0.1 * (np.random.random() + 0.5)
        new_h += amp1 * np.sin(2*np.pi*f1*np.arange(n)/n + phase1)
        # Random bit flips in significant regions
        if np.random.random() < 0.3:
            mask = (np.random.random(n) < 0.4).astype(bool)
            flips_count = np.sum(mask)
            rand_vals = np.random.uniform(0.8,1.2,size=flips_count)
            np.random.shuffle(rand_vals)
            new_h[mask] *= rand_vals
            new_h[new_h>1] = 1.0
            new_h[new_h<0] = 0.0
    return new_h
