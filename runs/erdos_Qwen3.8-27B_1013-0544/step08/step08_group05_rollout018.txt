    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    return (m + np.log(z) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (A - B)[s:s + n] * (2.0 / n)

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def guided_binary_search(h, n, rng, deadline):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h > 0.5)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h < 0.5)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        h[to_flip] = 1.0
    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = float(c.max())
    best_h = h.copy()
    dx = 2.0 / n
    stagnant = 0
    iters = 0
    while time.time() < deadline:
        iters += 1
        if iters % 100000 == 0 and stagnant > 50000:
            break
        worst_idx = int(np.argmax(c))
        use_guided = (stagnant > 200 or rng.rand() < 0.3)
        improved = False
        if use_guided:
            L = worst_idx - (n - 1)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                if L > 0:
                    scores_i = np.zeros_like(ones, dtype=float)
                    for idx_i in range(len(ones)):
                        i = ones[idx_i]
                        s = 0.0
                        if L <= i <= n - 1:
                            s -= (1.0 - h[i - L])
                        if 0 <= i <= n - 1 - L:
                            s += h[i + L]
                        scores_i[idx_i] = s
                    scores_j = np.zeros_like(zeros, dtype=float)
                    for idx_j in range(len(zeros)):
                        j = zeros[idx_j]
                        s = 0.0
                        if L <= j <= n - 1:
                            s += (1.0 - h[j - L])
                        if 0 <= j <= n - 1 - L:
                            s -= h[j + L]
                        scores_j[idx_j] = s
                    i = ones[int(np.argmin(scores_i))]
                    j = zeros[int(np.argmin(scores_j))]
                else:
                    M = -L
                    scores_i = np.zeros_like(ones, dtype=float)
                    for idx_i in range(len(ones)):
                        i = ones[idx_i]
                        s = 0.0
                        if M <= i <= n - 1:
                            s -= (1.0 - h[i - M])
                        if 0 <= i <= n - 1 - M:
                            s += h[i + M]
                        scores_i[idx_i] = s
                    scores_j = np.zeros_like(zeros, dtype=float)
                    for idx_j in range(len(zeros)):
                        j = zeros[idx_j]
                        s = 0.0
                        if M <= j <= n - 1:
                            s += (1.0 - h[j - M])
                        if 0 <= j <= n - 1 - M:
                            s -= h[j + M]
                        scores_j[idx_j] = s
                    i = ones[int(np.argmin(scores_i))]
                    j = zeros[int(np.argmin(scores_j))]
                if i != j:
                    trial = h.copy()
                    trial[i] = 0.0
                    trial[j] = 1.0
                    tc = float(np.correlate(trial, 1.0 - trial, mode="full").max())
                    if tc < best_c - 1e-12:
                        h = trial
                        c = np.correlate(h, 1.0 - h, mode="full")
                        best_c = tc
                        best_h = h.copy()
                        stagnant = 0
                        improved = True
        if not improved and rng.rand() < 0.5:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                if len(ones) >= 2 and len(zeros) >= 2 and rng.rand() < 0.35:
                    i1, i2 = rng.choice(ones, 2, replace=False)
                    j1, j2 = rng.choice(zeros, 2, replace=False)
                    trial = h.copy()
                    trial[i1] = 0.0; trial[i2] = 0.0
                    trial[j1] = 1.0; trial[j2] = 1.0
                    tc = float(np.correlate(trial, 1.0 - trial, mode="full").max())
                    if tc < best_c - 1e-12:
                        h = trial; c = np.correlate(h, 1.0 - h, mode="full")
                        best_c = tc; best_h = h.copy(); stagnant = 0; improved = True
                if not improved:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    trial = h.copy()
                    trial[i] = 0.0; trial[j] = 1.0
                    tc = float(np.correlate(trial, 1.0 - trial, mode="full").max())
                    if tc < best_c - 1e-12:
                        h = trial; c = np.correlate(h, 1.0 - h, mode="full")
                        best_c = tc; best_h = h.copy(); stagnant = 0; improved = True
        if not improved:
            stagnant += 1
            if stagnant > 3000:
                for _ in range(rng.randint(2, 7)):
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h[i] = 0.0; h[j] = 1.0
                c = np.correlate(h, 1.0 - h, mode="full")
                stagnant = 0
    return best_h, best_c * dx

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), method="SLSQP",
                           bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-12})
            nh = project_simplex(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-13:
                best = nh; bc = nc
        except Exception:
            pass
    return best, bc

def grad_descent(h, n, lr, steps, deadline):
    x = project_simplex(h, n).astype(float)
    best = x.copy()
    bc = compute_c5(best, n)
    alpha_start, alpha_end = 500.0, 200000.0
    for s in range(steps):
        if time.time() > deadline:
            break
        frac = s / max(steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        g = smooth_grad(x, n, alpha)
        g -= g.mean()
        x = x - lr * g / (np.max(np.abs(g)) + 1e-15)
        x = project_simplex(x, n)
        cv = compute_c5(x, n)
        if cv < bc - 1e-13:
            best = x.copy(); bc = cv
    return best, bc

def transfer_h(h_src, n_src, n_tgt):
    if n_tgt == n_src:
        return project_simplex(h_src, n_tgt).copy()
    xs = np.linspace(0.0, 2.0, n_src)
    ys = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(ys, xs, h_src), n_tgt)

def coord_cd(h, n, max_passes=4, deadline=None, grid=32, refine=10, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        for p in range(n):
            old = float(x[p])
            om = 1.0 - x[::-1]
            A = np.zeros(2 * n - 1)
            A[p:p + n] += om
            st = (n - 1) - p
            A[st:st + n] -= x
            cur = float(c.max())
            cand = c[None, :] + (xs0 - old)[:, None] * A[None, :]
            cand[:, n - 1] -= (xs0 - old) ** 2
            vals = cand.max(axis=1)
            i = int(np.argmin(vals))
            bx = float(xs0[i])
            bv = float(vals[i])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp); hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                cand2 = c[None, :] + (xx - old)[:, None] * A[None, :]
                cand2[:, n - 1] -= (xx - old) ** 2
                vals2 = cand2.max(axis=1)
                j = int(np.argmin(vals2))
                if vals2[j] < bv - tol:
                    bx = float(xx[j]); bv = float(vals2[j])
            if bv < cur - tol:
                d = bx - old
                c[p:p + n] += d * om
                c[st:st + n] -= d * x
                c[n - 1] -= d * d
                x[p] = bx
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp; bc = cp
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 10.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    n = n0
    h = project_simplex(h0, n)
    best_h = h.copy()
    best_c = float(compute_c5(h, n))
    # Symmetrize
    rev = h[::-1]
    for v in (0.5 * (h + rev), 0.5 * (h + (1.0 - rev))):
        vv = project_simplex(v, n)
        cv = compute_c5(vv, n)
        if cv < best_c:
            best_h = vv; best_c = cv
    # Phase 1: Binary local search
    t_bin_end = min(deadline - 120.0, t0 + 0.55 * budget_s)
    if n % 2 == 0 and time.time() < t_bin_end:
        b = binarize(best_h, n)
        if b is not None:
            cb_raw = compute_c5(b, n)
            if cb_raw < best_c:
                best_h = b; best_c = cb_raw
            hb, cb = guided_binary_search(b, n, rng, t_bin_end)
            if cb < best_c:
                best_h = hb; best_c = cb
    # Phase 2: Smooth refinement
    t_sm_end = min(deadline - 60.0, time.time() + 0.25 * budget_s)
    if time.time() < t_sm_end:
        h1, c1 = slsqp_polish(best_h, n, [2000, 10000, 50000, 200000], 80, t_sm_end)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n, max_passes=5, deadline=t_sm_end, grid=32, refine=12)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 3: Try another resolution
    if time.time() < deadline - 30.0:
        n2 = n // 2 if n > 200 else n + 50
        if n2 % 2 == 0 and 100 <= n2 <= 1000:
            ht = transfer_h(best_h, n, n2)
            cv = compute_c5(ht, n2)
            t_res_end = min(deadline - 20.0, time.time() + 60.0)
            h1, c1 = guided_binary_search(binarize(ht, n2) if n2 % 2 == 0 else ht, n2, rng, t_res_end) if n2 % 2 == 0 else (ht, cv)
            if c1 < best_c:
                best_h, best_c, n = h1, c1, n2
                h2, c2 = slsqp_polish(best_h, n, [10000, 50000], 60, deadline - 10.0)
                if c2 < best_c:
                    best_h, best_c = h2, c2
    # Phase 4: Final polish
    t_final = deadline - 5.0
    if time.time() < t_final:
        h1, c1 = slsqp_polish(best_h, n, [50000, 200000, 500000], 100, t_final)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n, max_passes=3, deadline=t_final, grid=40, refine=14)
        if c1 < best_c:
            best_h, best_c = h1, c1
    final_h = project_simplex(best_h, n)
    final_c = float(compute_c5(final_h, n))
    return final_h, final_c, int(n)
