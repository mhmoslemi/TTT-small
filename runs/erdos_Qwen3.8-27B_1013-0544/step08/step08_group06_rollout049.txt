import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes=2000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            lo_d, hi_d = -h0, 1.0 - h0
            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi+1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag = np.where(ms == (n-1))[0]
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if len(diag) > 0:
                nv[:, diag] -= d_grid[:, None]**2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            # Refine with parabolic interpolation
            if 0 < best_i < grid_size - 1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = (y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv*a
                        if len(diag) > 0:
                            nv1 = nv1.copy()
                            nv1[diag] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, float(xv)
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if len(diag) > 0:
                    c[ms[diag]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

where c = np.correlate(h, 1-h, mode='full').

The constraint is sum(h) = n/2, 0 ≤ h[i] ≤ 1.

This is a non-convex optimization problem (the max makes it non-convex). However, we can reformulate it as:
minimize T
subject to: c[k] ≤ T/dx for all k
           sum(h) = n/2
           0 ≤ h[i] ≤ 1

But c[k] is quadratic in h (it's a correlation, i.e., h·h type terms), so this is a non-convex QP.

Alternatively, we can use the bisection approach:
- Binary search on T
- For a given T, check feasibility: does there exist h with sum(h)=n/2, 0≤h≤1, and c[k] ≤ T/dx for all k?
- This feasibility check is still non-convex due to the quadratic constraints

So direct convex optimization won't work easily.

Another improvement: **warm-starting the grid**. If we know the previous best d, start the grid around that value.

def smooth_slsqp(h, n, alpha, maxiter=500):
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        m = cc.max()
        sh = alpha * (cc - m)
        mx = sh.max()
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
        return (m + lse / alpha) * (2.0 / n)
    def obj_grad(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        m = cc.max()
        sh = alpha * (cc - m)
        mx = sh.max()
        weights = np.exp(sh - mx)
        weights /= weights.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k] * (2.0 / n)
            lag = k - (n - 1)
            if lag >= 0:
                j1 = slice(0, n - lag)
                j2 = slice(lag, n)
                grad[j1] += w * (1.0 - x[j2])
                grad[j2] -= w * x[j1]
            else:
                lag2 = -lag
                j1 = slice(lag2, n)
                j2 = slice(0, n - lag2)
                grad[j1] += w * (1.0 - x[j2])
                grad[j2] -= w * x[j1]
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
