    A_ub, b_ub, A_eq, b_eq, c_obj, bounds = build_slp_lp(h, n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return project_feasible(res.x[:n], n)
    return h

def slp_optimize(h, n, max_iter=60, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        new_h = slp_step(best_h, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-14:
            best_h, best_c = new_h, new_c
        else:
            improved = False
            for alpha in [0.5, 0.25, 0.1]:
                dh = project_feasible(best_h + alpha*(new_h - best_h), n)
                dc = compute_c5(dh, n)
                if dc < best_c - 1e-14:
                    best_h, best_c = dh, dc
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c

def cd_eval(d, c_aff, coeffs, unaff):
    return max(float(np.max(c_aff + d * coeffs)), unaff)

def coord_descent(h, n, max_passes=80, deadline=None, grid_size=70):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            k_lo = max(0, p - (n-1))
            k_hi = min(2*n-2, p + (n-1))
            ks = np.arange(k_lo, k_hi+1)
            ds = ks - (n-1)
            L = len(ks)
            coeffs = np.zeros(L)
            for i, d in enumerate(ds):
                fp = p + d
                bp = p - d
                if 0 <= fp < n:
                    coeffs[i] += 1.0 - h[fp]
                if 0 <= bp < n:
                    coeffs[i] -= h[bp]
            c_aff = c[ks]
            unaff = 0.0
            if k_lo > 0:
                unaff = max(unaff, float(c[:k_lo].max()))
            if k_hi < 2*n-2:
                unaff = max(unaff, float(c[k_hi+1:].max()))
            lo, hi = -h0, 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.maximum(np.max(c_aff[None,:] + d_grid[:,None]*coeffs[None,:], axis=1), unaff)
            bi = int(np.argmin(vals))
            bd, bv = d_grid[bi], float(vals[bi])
            if 0 < bi < grid_size - 1:
                try:
                    res = golden(cd_eval, d_grid[bi-1], d_grid[bi+1],
                                args=(c_aff, coeffs, unaff), maxiter=50)
                    if res.fun < bv:
                        bd, bv = float(res.x), float(res.fun)
                except Exception:
                    pass
            if bv < current_max - 1e-13:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                improved = True
                current_max = min(current_max, bv)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def diverse_init(n, rng):
    x = np.linspace(0, 2, n)
    patterns = []
    patterns.append(np.full(n, 0.5))
    mid = n//2
    p1 = np.zeros(n)
    p1[mid//2:mid+mid//2] = 1.0
    patterns.append(p1)
    p2 = np.where(x < 0.5, 1.0, np.where(x > 1.5, 1.0, 0.0))
    patterns.append(p2)
    p3 = np.sin(x*np.pi)**2
    p3 = p3 / max(p3.max(), 1e-10)
    patterns.append(np.clip(p3, 0, 1))
    p4 = np.where(np.abs(x-1.0) < 0.4, 1.0, 0.3)
    patterns.append(p4)
    for _ in range(3):
        patterns.append(rng.uniform(0, 1, n))
    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLP from initial
    if time.time() < deadline - 400:
        h1, c1 = slp_optimize(best_h, n, 80, deadline-400)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent
    if time.time() < deadline - 300:
        h2, c2 = coord_descent(best_h, n, 80, deadline-300, 70)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: SLP again
    if time.time() < deadline - 200:
        h3, c3 = slp_optimize(best_h, n, 50, deadline-200)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Coordinate descent again
    if time.time() < deadline - 150:
        h4, c4 = coord_descent(best_h, n, 60, deadline-150, 80)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Multi-restart with perturbations
    n_cur = len(best_h)
    if time.time() < deadline - 120:
        for r in range(12):
            if time.time() > deadline - 80:
                break
            s = 0.005 + 0.02 * (r % 6)
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = slp_optimize(hp, n_cur, 30, deadline-100)
            h_o, c_o = coord_descent(h_o, n_cur, 40, deadline-70, 50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 6: Try diverse initializations
    if time.time() < deadline - 120:
        starts = diverse_init(n_cur, rng)
        for si in range(len(starts)):
            if time.time() > deadline - 80:
                break
            hs = project_feasible(starts[si], n_cur)
            cs = compute_c5(hs, n_cur)
            if cs < best_c + 0.05:
                h_o, c_o = slp_optimize(hs, n_cur, 40, deadline-100)
                h_o, c_o = coord_descent(h_o, n_cur, 40, deadline-70, 50)
                if c_o < best_c:
                    best_h, best_c = h_o, c_o
    # Phase 7: Different n values
    for n_try in [400, 500, 550, 650, 700, 800]:
        if time.time() > deadline - 60:
            break
        if n_try == n_cur:
            continue
        x_o = np.linspace(0, 2, n_cur)
        x_n = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_n, x_o, best_h), n_try)
        h_o, c_o = slp_optimize(h_int, n_try, 30, deadline-80)
        h_o, c_o = coord_descent(h_o, n_try, 30, deadline-50, 50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            n_cur = n_try
    # Phase 8: Final SLP + CD
    if time.time() < deadline - 20:
        hf, cf = slp_optimize(best_h, n_cur, 40, deadline-20)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = coord_descent(best_h, n_cur, 150, deadline-3, 100)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Symmetry check
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
