```python
def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    # grad[j] = sum_s W[s] * (1-h[j+s]) * 1[0<=j+s<n] - sum_s W[s] * h[j-s] * 1[0<=j-s<n]
    # s = m - (n-1), m = 0..2n-2, so s = -(n-1)..(n-1)
    # W[s] = w[s + n - 1]
    # Let's make W an array indexed by s from 0 to 2n-2 where W[k] = w[k] (k = s + n - 1)
    # So W[k] = w[k], and s = k - (n-1)
    
    # For each j, we need:
    # term1[j] = sum_k w[k] * (1-h[j+k-n+1]) * 1[0<=j+k-n+1<n]
    # term2[j] = sum_k w[k] * h[j-k+n-1] * 1[0<=j-k+n-1<n]
    
    # Let's use broadcasting
    j = np.arange(n)[:, None]  # (n, 1)
    k = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    # For term1: index into h is j + k - (n-1)
    idx1 = j + k - (n - 1)  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
    term1 = (w[None, :] * (1.0 - h1) * valid1).sum(axis=1)
    
    # For term2: index into h is j - k + (n-1)
    idx2 = j - k + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    term2 = (w[None, :] * h2 * valid2).sum(axis=1)
    
    return term1 - term2
```

This is O(n * (2n-1)) = O(n^2) memory and compute, which for n=500 is about 500K operations. Very fast.

Let me also think about the step size for Nesterov. For the smooth max with parameter alpha, the function is alpha-smooth (Lipschitz gradient with constant alpha * ||J||^2 where J is the "Jacobian" of the correlation). A safe step size is 1/L.

Actually, for the log-sum-exp smoothing: f_alpha(h) = (1/alpha) * log(sum(exp(alpha * c[m])))

The Hessian of f_alpha is:
H = (1/alpha) * (alpha * J^T J - alpha^2 * (J^T p)(J^T p)^T) * ... 

where p is the softmax probability and J is the Jacobian. This is bounded by (1/alpha) * alpha^2 * lambda_max(J^T J) = alpha * lambda_max(J^T J).

So L ≈ alpha * lambda_max(J^T J). For the correlation structure, lambda_max(J^T J) is roughly O(n) (each row of J has O(1) nonzeros of magnitude O(1), and there are O(n) columns). So L ≈ alpha * n.

A safe step size is 1/(alpha * n * C) for some safety factor C.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    lse = c.max() + np.log(np.sum(np.exp(shifted - shifted.max()))) / alpha
    return lse * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    j_arr = np.arange(n)[:, None]
    k_arr = np.arange(2 * n - 1)[None, :]
    idx1 = j_arr + k_arr - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    term1 = (w[None, :] * (1.0 - h1) * valid1).sum(axis=1)
    idx2 = j_arr - k_arr + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    term2 = (w[None, :] * h2 * valid2).sum(axis=1)
    dx = 2.0 / n
    return (term1 - term2) * (alpha * (2.0 / n))

def nesterov_opt(h, n, alphas, maxiter_each=3000, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        x = best_h.copy()
        y = x.copy()
        t = 1.0
        step = 1.0 / (alpha * n * 2)
        for it in range(maxiter_each):
            if deadline is not None and time.time() > deadline - 10:
                break
            g = smooth_grad(y, n, alpha)
            step = min(step * 1.001, 500.0 / (alpha * np.dot(g, g) + 1e-30))
            y_new = project_feasible(x - step * g, n)
            f_new = smooth_obj_val(y_new, n, alpha)
            f_old = smooth_obj_val(x, n, alpha)
            if f_new > f_old + 0.5 * step * np.dot(g, g):
                step *= 0.5
                y_new = project_feasible(x - step * g, n)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            mom = (t - 1.0) / (t_new + 1.0)
            x = y_new
            y = project_feasible(x + mom * (x - (x - mom * (x - x + g * step))), n)
            y = project_feasible(x + mom * (x - (x - step * g * (1 + mom))), n)
            y = x + mom * (x - project_feasible(x - step * (1 + mom) * smooth_grad(x, n, alpha), n))
            y = project_feasible(y, n)
            t = t_new
            c_cur = compute_c5(x, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = x.copy()
    return best_h, best_c

def nesterov_simple(h, n, alphas, maxiter_each=5000, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        x = best_h.copy()
        v = np.zeros(n)
        t = 1.0
        lr = 1.0 / (alpha * n)
        for it in range(maxiter_each):
            if deadline is not None and time.time() > deadline - 10:
                break
            g = smooth_grad(x, n, alpha)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            v = ((t - 1) / t_new) * v - lr * g
            x_new = project_feasible(x + v, n)
            f_new = smooth_obj_val(x_new, n, alpha)
            f_old = smooth_obj_val(x, n, alpha)
            if f_new > f_old * 1.001 + 1e-10:
                v *= 0.1
                x_new = project_feasible(x + v, n)
            x = x_new
            t = t_new
            if (it + 1) % 200 == 0:
                c_cur = compute_c5(x, n)
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = x.copy()
        c_cur = compute_c5(x, n)
        if c_cur < best_c:
            best_c = c_cur
            best_h = x.copy()
    return best_h, best_c

def coord_cd_fast(h, n, max_passes=200, deadline=None, grid_size=60, eps=0.0):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            s = np.arange(2 * n - 1)
            jf = p + s - (n - 1)
            jb = p - s + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(2 * n - 1)
            if np.any(valid_f):
                a[valid_f] = 1.0 - h[jf[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            c_aff = c_prev.copy()
            m_off = n - 1
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                nv[m_off] -= d * d
                val = float(nv.max())
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max - eps:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev = c_prev + best_d * a
                c_prev[m_off] -= best_d * best_d
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, compute_c5(h, n)

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.4 * np.sin(4 * np.pi * x)
    elif kind == 2:
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif kind == 3:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(6 * np.pi * x)
    elif kind == 4:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x + 0.5) + 0.2 * np.sin(4 * np.pi * x)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:

            h_s = project_feasible(np.interp(x_s, x_orig, best_h) + 0.03 * rng.randn(n_small), n_small)
            h_s, c_s = nesterov_simple(h_s, n_small, [200, 1000, 5000], 4000, deadline)

            h_o, c_o = nesterov_simple(hp, n, [500, 5000], 3000, deadline)

    return best_h, best_c, len(best_h)
