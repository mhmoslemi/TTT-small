def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, c, n, order, gs_iters=40):
    """One pass of CD. Returns updated h, c, current_max."""
    current_max = float(c.max())
    for p in order:
        h0 = h[p]
        if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
            continue
        d = np.zeros(2*n-1)
        j_lo = max(0, p - (2*n-2))
        j_hi = min(n-1, p)
        if j_lo <= j_hi:
            d[p - np.arange(j_lo, j_hi+1)] += 1.0 - h[j_lo:j_hi+1]
        j_lo2 = p
        j_hi2 = min(n-1, p + 2*n - 2)
        if j_lo2 <= j_hi2:
            d[np.arange(j_lo2, j_hi2+1) - p] -= h[j_lo2:j_hi2+1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, 1.0
        c1 = b - gr*(b-a)
        d1 = a + gr*(b-a)
        # Inline evaluation for speed
        v0_0 = c[0] + (c1-h0) - (c1*c1-h0*h0)
        vk_0 = np.max(c[1:] + (c1-h0)*d[1:])
        fc1 = v0_0 if v0_0 > vk_0 else vk_0
        v0_1 = c[0] + (d1-h0) - (d1*d1-h0*h0)
        vk_1 = np.max(c[1:] + (d1-h0)*d[1:])
        fd1 = v0_1 if v0_1 > vk_1 else vk_1
        for _ in range(gs_iters):
            if fc1 < fd1:
                b = d1; d1 = c1; fd1 = fc1
                c1 = b - gr*(b-a)
                v0 = c[0] + (c1-h0) - (c1*c1-h0*h0)
                vk = np.max(c[1:] + (c1-h0)*d[1:])
                fc1 = v0 if v0 > vk else vk
            else:
                a = c1; c1 = d1; fc1 = fd1
                d1 = a + gr*(b-a)
                v0 = c[0] + (d1-h0) - (d1*d1-h0*h0)
                vk = np.max(c[1:] + (d1-h0)*d[1:])
                fd1 = v0 if v0 > vk else vk
        t_star = (a+b)*0.5
        v0s = c[0] + (t_star-h0) - (t_star*t_star-h0*h0)
        vks = np.max(c[1:] + (t_star-h0)*d[1:])
        f_star = v0s if v0s > vks else vks
        # Check boundaries
        v0b = c[0] - h0 + h0*h0  # c_0(0) = c_0(1) = c[0] - h0 + h0^2
        vkb0 = np.max(c[1:] - h0*d[1:])
        f0 = v0b if v0b > vkb0 else vkb0
        vkb1 = np.max(c[1:] + (1-h0)*d[1:])
        f1 = v0b if v0b > vkb1 else vkb1
        # Pick best
        if f0 <= f_star and f0 <= f1:
            t_new = 0.0
        elif f1 <= f_star and f1 <= f0:
            t_new = 1.0
        else:
            t_new = t_star
        if t_new != h0:
            delta = t_new - h0
            h[p] = t_new
            c[1:] += delta * d[1:]
            c[0] += delta - (t_new*t_new - h0*h0)
            new_max = float(c.max())
            if new_max < current_max:
                current_max = new_max
    return h, c, current_max

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_d(p, h, n):
    d = np.zeros(2 * n - 1)
    j_max = min(n, 2 * n - 1 - p)
    if j_max > 0:
        js = np.arange(j_max)
        d[p + js] += 1.0 - h[n - 1 - js]
    j_max2 = min(n, 2 * n - 1 - (n - 1 - p), n - 1 - p + 1)
    j_max2 = min(n, 2 * n - 1 - max(0, n - 1 - p))
    start2 = max(0, n - 1 - p)
    j_max2 = min(n, 2 * n - 1 - start2)
    if j_max2 > 0:
        js = np.arange(j_max2)
        d[start2 + js] -= h[js]
    return d
