```python
            if time.time() > deadline:
                break

            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)

            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off > 0:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))

            best_x = hp
            best_f = max(unaff, qconst + hp*(1-hp), float(cseg.max() + a * (hp - hp)))  # placeholder
            # More careful: evaluate at hp
            if seg - 1 > 0:
                best_f = max(unaff, qconst + hp*(1-hp), float(np.max(cseg)))
            else:

            tol = 1e-12
            lo_t = max(0.0, unaff, qconst)
            hi_t = max(lo_t + 1e-10, unaff, qconst + hp*(1-hp))
            if seg - 1 > 0:

            hi_t += 1e-8

                # and unaff <= mid_t

                # Let rhs = mid_t - cseg[j] + a[j]*hp

                    if zero.any() and np.max(cseg[zero]) > mid_t + tol:

                g = mid_t - qconst

                    xh = min(xh, 0.5 - sqrt_val)  # might be infeasible

                    # Branch 2: x in [max(xl, 0.5+sqrt_val), min(1, xh)]
                    # Actually this is more complex. Let me handle it differently.
                    # The feasible set for the quadratic is [0, 0.5-sqrt_val] ∪ [0.5+sqrt_val, 1]
                    # So combined with linear: check if either branch is feasible
                    b1_xh = min(xh, 0.5 - sqrt_val)
                    b2_xl = max(xl, 0.5 + sqrt_val)
                    if max(xl, 0.0) <= b1_xh + tol or b2_xl <= min(xh, 1.0) + tol:
                        hi_t = mid_t
                    else:
                        lo_t = mid_t
                    continue
                else:

                    lo_t = mid_t
                    continue
                
                if xl <= xh + tol:
                    hi_t = mid_t
                else:
                    lo_t = mid_t
            
            # After binary search, hi_t is the minimum feasible max value
            t = hi_t
            # Find best x achieving t
            best_x = hp
            best_f = best_f  # current value at hp
            
            if t >= unaff - tol:
                xl = 0.0
                xh = 1.0
                if seg - 1 > 0:
                    pos = a[:seg-1] > 1e-14
                    neg = a[:seg-1] < -1e-14
                    if neg.any():
                        rhs = (t + a[neg]*hp - cseg[neg]) / a[neg]
                        xl = max(xl, float(np.max(rhs)))
                    if pos.any():
                        rhs = (t + a[pos]*hp - cseg[pos]) / a[pos]
                        xh = min(xh, float(np.min(rhs)))
                g = t - qconst
                if g > 0.25:
                    branches = [(max(xl, 0.0), min(xh, 1.0))]
                elif g >= 0:
                    sv = np.sqrt(max(0.0, 0.25 - g))
                    branches = [(0.0, min(xh, 0.5 - sv)), (max(xl, 0.5 + sv), 1.0)]
                else:
                    branches = []
                for Lq, Uq in branches:
                    L = max(Lq, xl, 0.0)
                    U = min(Uq, xh, 1.0)
                    if L <= U + tol:
                        for x in [L, U, 0.5*(L+U)]:
                            # Evaluate
                            f = max(unaff, qconst + x*(1-x))
                            if seg - 1 > 0:
                                f = max(f, float(np.max(cseg + a[:seg-1]*(x-hp))))
                            if f < best_f:
                                best_f = f
                                best_x = x
            
            if best_f < cur - 1e-12:
                dxv = best_x - hp
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center+1:hi+1] += asg[off:] * dxv
                c[center] = qconst + best_x * (1.0 - best_x)
                h[p] = best_x
                cur = best_f
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
```

```python
def flow_local_search(h, n, deadline, max_iters=10000):

        # Compute derivative of max correlation w.r.t. this flow

        # where a_i[m] is the coefficient of h[i] in c[m]
```

This is O(n²) per iteration (due to the correlate call), so for n=500, each iteration is about 250K operations. With 10000 iterations, that's 2.5 billion operations - too slow.

Better: for a pairwise flow, compute the change in each correlation analytically.

For a flow of amount d from j to i:
Δc[m] = d * (coef_i[m] - coef_j[m])

where coef_k[m] is the partial derivative of c[m] with respect to h[k].

The derivative of c[m] = sum_r h[r+m'](1-h[r]) with respect to h[k]:
- If k = r+m' for some valid r: derivative = (1-h[r]) = (1-h[k-m'])
- If k = r for some valid r: derivative = -h[r+m'] = -h[k+m']

So coef_k[m] = (1-h[k-m']) - h[k+m'] (with appropriate boundary handling).

For all m simultaneously, coef_k is a vector of length 2n-1.

Computing coef_i and coef_j is O(n) each. Then Δc = d * (coef_i - coef_j), and we need max(c + Δc).

This is O(n) per flow attempt! Much better.

But we still need to find the optimal d for each (i,j) pair. Since the objective is max of linear functions in d, the optimal d is at a "knee point" - where the maximum changes identity.

For simplicity, let's just try d = 0.5 * min(h[i], 1-h[j]) and d = -0.5 * min(1-h[i], h[j]) and see if either improves.

Actually, let me think about this more carefully. The function f(d) = max_m (c[m] + d * (coef_i[m] - coef_j[m])) is a convex piecewise-linear function of d. We want to minimize it over d ∈ [d_min, d_max] where d_min = -min(h[i], 1-h[j]) and d_max = min(1-h[i], h[j]).

The minimum of a convex piecewise-linear function over an interval is at a breakpoint or an endpoint. The breakpoints are where two adjacent "active" correlations cross.

For simplicity, let's evaluate at a few candidate d values:
1. d = 0 (current)
2. d = d_min
3. d = d_max
4. The d that makes the two worst correlations equal (if they bracket the current max)

This gives a fast 2-coordinate optimization.

```python
def flow_delta(h, i, j, n):
    """Compute Δc/d for a unit flow from j to i."""
    center = n - 1
    m = np.arange(2 * n - 1)
    k = m - center  # the shift for each correlation index
    
    # For coordinate i:
    # Term 1: (1 - h[i - k]) if 0 <= i-k < n
    # Term 2: -h[i + k] if 0 <= i+k < n
    t1_i = np.zeros(2*n-1)
    valid1_i = (i - k >= 0) & (i - k < n)
    t1_i[valid1_i] = 1.0 - h[i - k[valid1_i]]
    
    t2_i = np.zeros(2*n-1)
    valid2_i = (i + k >= 0) & (i + k < n)
    t2_i[valid2_i] = -h[i + k[valid2_i]]
    
    # Same for j:
    t1_j = np.zeros(2*n-1)
    valid1_j = (j - k >= 0) & (j - k < n)
    t1_j[valid1_j] = 1.0 - h[j - k[valid1_j]]
    
    t2_j = np.zeros(2*n-1)
    valid2_j = (j + k >= 0) & (j + k < n)
    t2_j[valid2_j] = -h[j + k[valid2_j]]
    
    return (t1_i + t2_i) - (t1_j + t2_j)
```

Each call is O(n) (vectorized numpy operations on arrays of length 2n-1). For n=500, that's very fast.

```python
import numpy as np

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def binary_from_h(h):

    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]

def flow_delta(h, i, j, n):
    center = n - 1
    m = np.arange(2 * n - 1)
    k = m - center
    t1i = np.zeros(2 * n - 1)
    v1i = (i - k >= 0) & (i - k < n)
    t1i[v1i] = 1.0 - h[i - k[v1i]]
    t2i = np.zeros(2 * n - 1)
    v2i = (i + k >= 0) & (i + k < n)
    t2i[v2i] = -h[i + k[v2i]]
    t1j = np.zeros(2 * n - 1)
    v1j = (j - k >= 0) & (j - k < n)

    t2j = np.zeros(2 * n - 1)
    v2j = (j + k >= 0) & (j + k < n)
    t2j[v2j] = -h[j + k[v2j]]
    return (t1i + t2i) - (t1j + t2j)

    cur = float(c.max())
    best_h = h.copy()

    rng = np.random.RandomState(seed)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = flow_delta(h, i, j, n)
        d_lo = -min(h[i], 1.0 - h[j])
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-12:
            continue
        # Find best d: minimize max(c + d*delta)
        # Evaluate at boundaries and at intersection points of top active set
        active = np.argsort(-c)[:min(20, 2 * n - 1)]
        cands = [0.0, d_lo, d_hi]
        for a1 in active[:6]:

                    if d_lo <= d* <= d_hi:
                        cands.append(float(d*))
        best_d = 0.0
        best_val = cur
        for d in cands:
            d = max(d_lo, min(d_hi, float(d)))
            val = float(np.max(c + d * delta))
            if val < best_val - 1e-13:
                best_val = val
                best_d = float(d)
        if abs(best_d) > 1e-12:
            h[i] += best_d
            h[j] -= best_d
            c += best_d * delta
            cur = best_val
        if (2.0 / n) * cur < best_c - 1e-14:
            best_c = (2.0 / n) * cur
            best_h = h.copy()
    return best_h, best_c
```
