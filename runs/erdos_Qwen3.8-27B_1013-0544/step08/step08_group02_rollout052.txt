def sv(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m) - (alpha * (c - m)).max())
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)

def sg(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    off = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - h[:n-d])
            g[:n-d] -= w * h[d:]
        else:
            e2 = -d
            g[:n-e2] += w * (1.0 - h[e2:])
            g[e2:] -= w * h[:n-e2]
    return (2.0 / n) * g

def sumc(x, n):
    return x.sum() - 0.5 * n

def sumj(x, n):
    return np.ones_like(x)

def slsqp(h, n, alphas, maxits, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sumc, 'jac': sumj, 'args': (n,)}]
    for alpha, mi in zip(alphas, maxits):
        if time.time() > deadline - 1:
            break
        try:
            r = minimize(sv, h, args=(n, alpha), method='SLSQP', jac=sg,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': mi, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(r.x, dtype=float)
            if np.all(np.isfinite(x)):
                h = project_feasible(x, n)
        except Exception:
            pass
    return h

def interp(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = np.linspace(0, 2, n, endpoint=False)
    x1 = np.linspace(0, 2, m, endpoint=False)
    return project_feasible(np.interp(x1, x0, h), m)

def pair_search(h, n, deadline, max_passes, tol):
    """Systematic pair search: sweep all pairs in active set."""
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    # Precompute A matrix: A[i,k] = d(c_k)/d(h[i])
    A = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += (1.0 - h[:n-d])
            A[:n-d, k] -= h[d:]
        else:
            e = -d
            A[:n-e, k] += (1.0 - h[e:])
            A[e:, k] -= h[:n-e]
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        cur_max = c.max()
        active = np.flatnonzero(c >= cur_max - tol)
        improved = False
        for kk in active:
            d = kk - off
            if d >= 0:
                m = n - d
                for t in range(m):
                    i, j = t + d, t
                    _try_pair(h, c, A, i, j, n, off)
                    improved = True
            else:
                e = -d
                m = n - e
                for t in range(m):
                    i, j = t + e, t
                    _try_pair(h, c, A, i, j, n, off)
                    improved = True
        if not improved:
            break
        # Recompute c and A periodically
        c = np.correlate(h, 1.0 - h, mode='full')
        # Don't recompute A every time (expensive); every few shifts is enough
    return h, compute_c5(h, n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n

    for _ in range(60):

        else: hi = tau
