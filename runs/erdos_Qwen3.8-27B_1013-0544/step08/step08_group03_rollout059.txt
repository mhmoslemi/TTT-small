        best_pair = None
        # Sample a few random pairs
        for _ in range(50):
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]

            h_bin[j] = 0

            i, j = best_pair
            h_bin[i] = 1
            h_bin[j] = 0
            c -= best_improvement
        else:
            break
    return h_bin, c

For a binary pattern of length n with n/2 ones, the overlap at lag τ is:

We want to minimize max_τ c(τ).

A good pattern would be one where the 1s are arranged so that for every shift, many of them "hit" their shifted positions.

For a periodic pattern with period p (where n is a multiple of p), the autocorrelation is periodic. If p=2 (alternating 1010...), then odd shifts give 0 overlap and even shifts give maximum overlap. So that's bad.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max of c = correlate(h, 1-h) and its gradient."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    # Shift for numerical stability
    a = np.exp(alpha * (c - m))
    s = a.sum()
    val = m + np.log(s) / alpha
    w = a / s  # weights for the max
    
    # Gradient: d/dh_j of c_raw[k] for each k
    # c_raw[k] = sum_i h[i] * (1 - h[i+k-n+1]) for valid i
    # Let shift s = k - n + 1
    # c_raw[k] = sum_i h[i] * (1 - h[i+s]) for i where 0<=i<n and 0<=i+s<n
    # 
    # d(c_raw[k])/d(h[j]):
    #   If j appears as i: (1 - h[j+s]) for valid j+s
    #   If j appears as i+s (i.e., i = j-s): -h[j-s] for valid j-s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            # i ranges over [0, n-1-s], j = i (first role): j in [0, n-1-s]
            # j = i+s (second role): j in [s, n-1], i = j-s
            for j in range(n - s):
                grad[j] += wk * (1.0 - h[j + s])
            for j in range(s, n):
                grad[j] -= wk * h[j - s]
        else:
            # s < 0, let r = -s > 0
            # i ranges over [r, n-1], j = i (first role): j in [r, n-1]
            # j = i+s = i-r (second role): j in [0, n-1-r], i = j+r
            r = -s
            for j in range(r, n):
                grad[j] += wk * (1.0 - h[j - r])
            for j in range(n - r):
                grad[j] -= wk * h[j + r]
    return val * 1.0, grad * dx

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, p, maxiter, deadline):
    """Minimize p-norm of correlation using SLSQP."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return (np.sum(np.abs(c)**p) ** (1.0/p)) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        # Gradient of sum(|c|^p)^(1/p) * dx
        # = (1/p) * (sum(|c|^p))^(1/p - 1) * sum(p * |c|^(p-1) * sign(c) * dc/dx) * dx
        # This is complex, let SLSQP use numerical gradient
        return None
    def sum_con(x):
        return x.sum() - n/2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':sum_con,'jac':sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter,'ftol':1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except:
        return h, compute_c5(h, n)

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=30, gs_iters=30):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(c.max())
    D = (np.arange(2*n-1) == (n-1)).astype(float)
    
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            # Compute A: change in c when h[p] goes from 0 to 1
            # c[k] = sum_i h[i] * (1-h[i+k-n+1])
            # A[k] = contribution of h[p] to c[k]
            # h[p] appears as h[i] when i=p: contributes (1-h[p+k-n+1])
            # h[p] appears as h[i+k-n+1] when i+k-n+1=p, i=p-(k-n+1)=p+n-1-k: contributes -h[p+n-1-k]
            A = np.zeros(2*n-1)
            s_vals = np.arange(2*n-1) - (n-1)
            # First role: h[p] as h[i], i=p, j+s = p+s for s in s_vals
            valid1 = (p + s_vals >= 0) & (p + s_vals < n)
            A[valid1] += (1.0 - h[p + s_vals[valid1]])
            # Second role: h[p] as h[i+s], i+s=p, i=p-s
            valid2 = (p - s_vals >= 0) & (p - s_vals < n)
            A[valid2] += (-h[p - s_vals[valid2]])
            
            # c_new[k] = c[k] + (t-h0)*A[k] - (t^2-h0^2)*D[k]
            # Minimize max_k (c[k] + (t-h0)*A[k] - (t^2-h0^2)*D[k]) over t in [0,1]
            # = max_k (c[k] - h0*A[k] + h0^2*D[k] + t*A[k] - t^2*D[k])
            c0 = c - h0 * A + h0*h0 * D
            # f(t) = max_k (c0[k] + t*A[k] - t^2*D[k])
            
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.array([float(np.max(c0 + t*A - t*t*D)) for t in grid])
            bi = int(np.argmin(fvals))
            lo = grid[max(0, bi-1)]
            hi = grid[min(grid_pts-1, bi+1)]
            if hi - lo < 1e-14:
                t_star = grid[bi]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr*(b-a)
                d1 = a + gr*(b-a)
                fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        b = d1; d1 = c1; fd1 = fc1
                        c1 = b - gr*(b-a); fc1 = float(np.max(c0 + c1*A - c1*c1*D))
                    else:
                        a = c1; c1 = d1; fc1 = fd1
                        d1 = a + gr*(b-a); fd1 = float(np.max(c0 + d1*A - d1*d1*D))
                t_star = (a+b)*0.5
            
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            fs = float(np.max(c0 + t_star*A - t_star*t_star*D))
            
            if min(f0, f1, fs) < current_max - 1e-12:
                if f0 <= f1 and f0 <= fs:
                    t_new = 0.0
                elif f1 <= f0 and f1 <= fs:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new*t_new - h0*h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                improved = True
        
        h = project_feasible(h, n)
        # Need to recompute c after projection
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
        if not improved:
            break
    
    return h, compute_c5(h, n)

def binary_hill_climb(h_bin, n, deadline, max_rounds=50):
    """Hill climbing on binary pattern using pair flips."""
    dx = 2.0 / n
    h = h_bin.copy().astype(float)
    c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        best_improve = 0
        best_flip = None
        n_samples = min(200, len(ones) * len(zeros))
        rng = np.random.RandomState(round_i * 1000 + 42)
        for _ in range(n_samples):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i] = 1.0
            h[j] = 0.0
            new_c = compute_c5(h, n)
            imp = c - new_c
            h[i] = 0.0
            h[j] = 1.0
            if imp > best_improve:
                best_improve = imp
                best_flip = (i, j)
        if best_flip is not None:
            i, j = best_flip
            h[i] = 1.0
            h[j] = 0.0
            c -= best_improve
            # Update sets
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            break
    return h, c

    elif cur < target:
        zeros = np.where(h_bin < 0.5)[0]
        order = np.argsort(-h_bin[zeros])[:target - cur]
        h_bin[zeros[order]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(400, 0.5)
        n = 400
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP p-norm continuation
    for p in [10, 30, 100, 300, 1000]:
        if time.time() > deadline - 250:
            break
        h, c = slsqp_optimize(best_h, n, p, 400, deadline)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 2: Exact CD
    if time.time() < deadline - 60:
        h, c = exact_cd(best_h, n, 250, deadline, 35, 35)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 3: Binary landscape search + re-optimize
    if time.time() < deadline - 150:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = adjust_binary(h_bin, n)
        h_bin, _ = binary_hill_climb(h_bin, n, deadline - 80, 35)
        h_bin, _ = block_search(h_bin, n, deadline - 30, 12)
        h2 = h_bin.copy()
        h2, c2 = slsqp_optimize(h2, n, 50, 200, deadline)
        h2, c2 = exact_cd(h2, n, 120, deadline, 35, 35)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 4: Multi-resolution
    n_list = [350, 400, 450, 550, 600, 650]
    for n_try in n_list:
        if time.time() > deadline - 100:
            break
        if n_try == n:
            continue
        h_int = interpolate_h(best_h, len(best_h), n_try)
        h_int, c_int = slsqp_optimize(h_int, n_try, 30, 200, deadline)
        h_int, c_int = exact_cd(h_int, n_try, 100, deadline, 30, 30)
        if c_int < best_c:
            best_h, best_c = h_int, c_int

    # Phase 5: Perturbation restarts on best solution
    nb = len(best_h)
    if time.time() < deadline - 60:
        rng = np.random.RandomState(seed)
        for r in range(3):
            if time.time() > deadline - 30:
                break
            scale = 0.02 + 0.04 * r
            hp = project_feasible(best_h + rng.normal(0, scale, nb), nb)
            hp, cp = slsqp_optimize(hp, nb, 100, 200, deadline)
            hp, cp = exact_cd(hp, nb, 80, deadline, 30, 30)
            if cp < best_c:
                best_h, best_c = hp, cp

    # Phase 6: Final fine CD
    if time.time() < deadline - 10:
        h, c = exact_cd(best_h, len(best_h), 200, deadline, 50, 45)
        if c < best_c:
            best_h, best_c = h, c

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
