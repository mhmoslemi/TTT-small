   ```python
           perturbed = project_feasible(h_final + rng.normal(0, 0.01, n_val), n_val)
           _, cand = cvxpy_relaxation(perturbed.astype(float))
           new_val = compute_c5(cand, n_val)
           if new_val < best_c5:
               h_final = cand
               best_c5 = new_val

       best_c5 = float(np.min(compute_c5(h_final, n_val), best_c5)
       return h_final, best_c5, n_val
   ```
