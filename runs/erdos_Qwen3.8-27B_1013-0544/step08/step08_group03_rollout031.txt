        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            ks, iv = struct[p]
            if len(ks) == 0:
                continue
            A = np.zeros(2 * n - 1)
            A[ks] = 1.0 - 2.0 * h[iv]
            A[n - 1] = 1.0
            c0 = c.copy()
            c0[ks] -= h0 * A[ks]
            c0[n - 1] = c[n - 1] - h0 + h0 * h0
            t_star = cd_1d(c0, A, D, h0)
            if abs(t_star - h0) > 1e-12:
                f_new = float(np.max(c0 + t_star * A - t_star * t_star * D))
                if f_new < current_max - 1e-13:
                    h[p] = t_star
                    c = c0 + t_star * A - t_star * t_star * D
                    current_max = f_new
                    improved += 1
        h = project_feasible(h, n)
        if improved == 0:
            break
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, p_val=500, maxiter=400, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return float(np.sum(c ** p_val)) ** (1.0 / p_val)
    def sc(x):
        return x.sum() - n / 2.0
    def scj(x):
        return np.ones_like(x)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: CD on initial
    h1, c1 = exact_cd(best_h, n_init, 400, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: SLSQP then CD
    if time.time() < deadline - 150:
        h2, c2 = slsqp_smooth(best_h, n_init, 300, 250, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
            h2b, c2b = exact_cd(best_h, n_init, 300, deadline)
            if c2b < best_c:
                best_h, best_c = h2b, c2b

    # Phase 3: Multi-n exploration
    n_list = [400, 450, 500, 550, 600, 700, 800, 1000]
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        if n_try == len(best_h):
            continue
        h_int = interp_h(best_h, len(best_h), n_try)
        h_opt, c_opt = exact_cd(h_int, n_try, 300, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        if time.time() < deadline - 120:
            h_s, c_s = slsqp_smooth(h_opt, n_try, 200, 150, deadline)
            if c_s < c_opt:
                h_s, c_s = exact_cd(h_s, n_try, 200, deadline)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s

    # Phase 4: Multi-restart
    n_best = len(best_h)
    if time.time() < deadline - 100:
        sizes = [0.02, 0.04, 0.08, 0.15, 0.25]
        for s in sizes:
            if time.time() > deadline - 60:
                break
            hp = project_feasible(best_h + rng.normal(0, s, n_best), n_best)
            h_o, c_o = exact_cd(hp, n_best, 250, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
            if time.time() < deadline - 80:
                h_o2, c_o2 = slsqp_smooth(h_o, n_best, 500, 200, deadline)
                if c_o2 < best_c:
                    h_o2, c_o2 = exact_cd(h_o2, n_best, 200, deadline)
                    if c_o2 < best_c:
                        best_h, best_c = h_o2, c_o2

    # Phase 5: Final high-pass CD
    if time.time() < deadline - 10:
        hf, cf = exact_cd(best_h, len(best_h), 2000, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
