import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n)), c

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def sa_binary(h, n, t0, tmin, cooldown, max_iters, deadline, rng):
    """Simulated annealing on binary sequences."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_c = float(c.max())
    best_h = h.copy()
    best_c = cur_c * (2.0 / n)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    it = 0
    t = t0
    while it < max_iters:
        if time.time() > deadline:
            break
        # Pick random 1 and random 0
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[p], h[q] = h[q], h[p]
        c_new = np.correlate(h, 1.0 - h, mode='full')
        new_c = float(c_new.max())
        
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(t, tmin)):
            cur_c = new_c
            c = c_new
            if cur_c < float(best_c * n / 2.0):
                best_c = cur_c * (2.0 / n)
                best_h = h.copy()
        else:
            # Revert
            h[p], h[q] = h[q], h[p]
        
        it += 1
        t = max(tmin, t * (1.0 - 1.0 / cooldown))
    
    return best_h, best_c

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0:
            h += (diff / tot) * cap
    else:
        tot = float(h.sum())
        if tot > 0:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def sa_phase(h, n, t0, tmin, n_iter, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h = h.copy()
    best_raw = cur
    t = t0
    decay = max(0.999, 1.0 - 1.0 / max(n_iter, 100))
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        po = np.where(h == 1.0)[0]
        pz = np.where(h == 0.0)[0]
        if not po.size or not pz.size:
            break
        p = po[rng.randint(po.size)]
        q = pz[rng.randint(pz.size)]
        h[p], h[q] = 0.0, 1.0
        c = np.correlate(h, 1.0 - h, mode='full')
        new_raw = float(c.max())
        d = new_raw - cur
        if d <= 0 or rng.random() < np.exp(-d / max(t, 1e-10)):
            cur = new_raw
            if cur < best_raw:
                best_raw = cur
                best_h = h.copy()
        else:
            h[p], h[q] = 1.0, 0.0
        t = max(tmin, t * decay)
    
    return best_h, best_raw * (2.0 / n)

def greedy_swaps(h, n, max_rounds, deadline, rng):
    """First-improvement greedy binary swaps."""
    h = h.copy()
    c5 = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for p in ones:
            if time.time() > deadline:
                break
            for q in zeros[:50]:  # Only check first 50 zeros for speed
                h[p], h[q] = 0.0, 1.0
                new_c5 = compute_c5(h, n)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    break
                else:
                    h[p], h[q] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, c5
