    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dxv = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dxv, (A - B)[s:s+n] * dxv

def obj_fn(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def grad_fn(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def eq_fun(x, n):
    return x.sum() - 0.5 * n

def eq_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = c5_from_corr(corr_full(best, n), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_fn, best, args=(n, a), jac=grad_fn,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = c5_from_corr(corr_full(nh, n), n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = c5_from_corr(corr_full(h0, n0), n0)
    n_best = n0

    rev = h0[::-1]
    for v in (0.5*(h0 + rev), 0.5*(h0 + 1.0 - rev)):
        vv = project_simplex(v, n_best)
        cv = c5_from_corr(corr_full(vv, n_best), n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    # Phase 1: Coordinate descent (major time)
    t1 = min(deadline - 300.0, t0 + 400.0)
    h1, c1, _ = coord_cd(best_h, n_best, t1, max_passes=10000, grid_n=256)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Binary SA (if even n)
    if n_best % 2 == 0:
        t2 = min(deadline - 200.0, t0 + 650.0)
        if time.time() < t2:
            hb, cb = sa_binary(best_h, n_best, rng, t2)
            if cb < best_c:
                best_h, best_c = hb, cb
                # Polish after binary
                t2b = min(deadline - 150.0, time.time() + 80.0)
                h2b, c2b = coord_cd(best_h, n_best, t2b, max_passes=2000, grid_n=256)
                if c2b < best_c:
                    best_h, best_c = h2b, c2b

    # Phase 3: Try other resolutions
    cands = [400, 600, 450, 550, 350, 700, 500]
    tried = set([n_best])
    t3 = min(deadline - 100.0, t0 + 950.0)
    for nt in cands:
        if time.time() > t3:
            break
        if nt in tried or nt < 100 or nt > 1000 or nt % 2:
            continue
        tried.add(nt)
        ht = transfer_h(best_h, n_best, nt)
        # Quick CD at new resolution
        chunk = min(t3, time.time() + 40.0)
        ht2, ct2, _ = coord_cd(ht, nt, chunk, max_passes=500, grid_n=200)
        if ct2 < best_c:
            best_h, best_c, n_best = ht2, ct2, nt
        # If promising, do more CD
        if ct2 < best_c + 0.002 and time.time() < t3:
            chunk2 = min(t3, time.time() + 30.0)
            ht3, ct3, _ = coord_cd(ht2, nt, chunk2, max_passes=2000, grid_n=256)
            if ct3 < best_c:
                best_h, best_c, n_best = ht3, ct3, nt

    # Phase 4: SLSQP polish
    t4 = min(deadline - 30.0, time.time() + 80.0)
    if time.time() < t4:
        h4, c4 = slsqp_polish(best_h, n_best, [5000, 50000, 500000], 80, t4)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Final CD
    t5 = min(deadline - 5.0, time.time() + 60.0)
    if time.time() < t5:
        h5, c5, _ = coord_cd(best_h, n_best, t5, max_passes=5000, grid_n=512, n_perturb=12)
        if c5 < best_c:
            best_h, best_c = h5, c5

    final_h = project_simplex(best_h, n_best)
    final_c = float(c5_from_corr(corr_full(final_h, n_best), n_best))
    return final_h, final_c, int(n_best)
