    return float(c.max()) * (2.0 / n)

        g[k2:] -= h[:max(1, n-k2)]

    h = project_simplex(h.copy(), n)

        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-8) & (y < 1.0 - 1e-8)
        if mask.any():
            y[mask] += d / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def c5_grad_k(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        k2 = -lag
        g[:n-k2] += (1.0 - h[k2:])
        g[k2:] -= h[:n-k2]
    return g

def subgradient_opt(h, n, deadline, n_it=80000, n_top=4):
    h = project_simplex(h.copy(), n)
    best = h.copy()
    bc = compute_c5(best, n)
    L = 2 * n - 1
    eta = 2.0 / np.sqrt(n)
    for it in range(n_it):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        if cm < bc:
            bc = cm
            best = h.copy()
        top = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for kk in top:
            g += c5_grad_k(h, n, kk - (n - 1))
        g /= n_top
        h_new = project_simplex(h - eta * g, n)
        nc = compute_c5(h_new, n)
        if nc < cm:
            h = h_new
            if it % 200 < 50:
                eta = min(eta * 1.03, 20.0 / np.sqrt(n))
        else:
            eta = max(eta * 0.92, 0.1 / np.sqrt(n))
        if it % 10000 == 5000:
            hp = project_simplex(h + 0.04 * np.random.randn(n), n)
            if compute_c5(hp, n) < cm * 1.02:
                h = hp
    return best, bc

def smooth_opt(h, n, alphas, maxiter, deadline):
    dx = 2.0 / n
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n

    def sobj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = float(c.max())
        r = alphas[0] * (c - m)
        r -= r.max()
        e = np.exp(r)
        z = float(e.sum())
        return (m + np.log(z) / alphas[0]) * dx

    def sgrad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = float(c.max())
        r = alphas[0] * (c - m)
        r -= r.max()
        e = np.exp(r)
        z = float(e.sum())
        w = e / z
        A = np.convolve(1.0 - x, w, mode="full")
        B = np.convolve(x, w[::-1], mode="full")
        s = n - 1
        return (A - B)[s:s + n] * dx

    cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
             "jac": lambda x: np.ones_like(x)}]
    for a in alphas:
        if time.time() > deadline:
            break
        alphas[0] = a
        try:
            res = minimize(sobj, best, jac=sgrad, method="SLSQP",
                           bounds=bounds, constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-14})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best, bc = nh, nc
    return best, bc

def binary_local(h_init, n, rng, deadline):
    if n % 2:
        return None, np.inf
    target = n // 2
    idx = np.argpartition(h_init, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    dx = 2.0 / n
    best = b.copy()
    bc = float(np.correlate(best, 1.0 - best, mode="full").max())
    one = np.where(b == 1.0)[0]
    zero = np.where(b == 0.0)[0]
    stagnant = 0
    while time.time() < deadline and stagnant < 5:
        c = np.correlate(best, 1.0 - best, mode="full")
        cm = float(c.max())
        improved = False
        tabu = set()
        for _ in range(50000):
            if time.time() > deadline:
                break
            if len(one) == 0 or len(zero) == 0:
                break
            i = int(one[rng.randint(len(one))])
            j = int(zero[rng.randint(len(zero))])
            if (i, j) in tabu:
                continue
            d = c.copy()
            d[i:i+n] -= (1.0 - best[i+1:i+1+n]) if i + n <= 2*n-1 else 0
            st = n - 1 + i
            d[st:st+n] += best
            d[j:j+n] += (1.0 - best[j+1:j+1+n]) if j + n <= 2*n-1 else 0
            st2 = n - 1 + j
            d[st2:st2+n] -= best
            d[n-1] -= 2.0
            d[n-1+i-j] += 1.0
            d[n-1+j-i] += 1.0
            nm = float(d.max())
            if nm < cm - 1e-10:
                c = d
                cm = nm
                best[i] = 0.0
                best[j] = 1.0
                one = one[one != i]
                zero = zero[zero != j]
                zero = np.append(zero, i)
                one = np.append(one, j)
                improved = True
                if cm < bc:
                    bc = cm
                    stall
                tabu = set()
            else:
                tabu.add((i, j))
                if len(tabu) > 5000:
                    tabu = set(list(tabu)[-2000:])
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            for _ in range(10):
                i = int(one[rng.randint(len(one))])
                j = int(zero[rng.randint(len(zero))])
                best[i] = 0.0
                best[j] = 1.0
                one = one[one != i]
                zero = zero[zero != j]
                zero = np.append(zero, i)
                one = np.append(one, j)
    return best, bc * dx

def transfer_h(h, n, nt):
    if nt == n:
        return project_simplex(h.copy(), nt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nt)
    return project_simplex(np.interp(y, x, h), nt)

def coord_cd(h, n, deadline, grid=40, refine=12, max_passes=4):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A = np.zeros(L)
            A[p:p+n] += om
            st = diag - p
            A[st:st+n] -= x
            cur = float(c.max())
            d_arr = xs0 - old
            cand = c[None, :] + d_arr[:, None] * A[None, :]
            cand[:, n-1] -= d_arr * d_arr
            i = int(np.argmin(cand.max(axis=1)))
            bx = float(xs0[i])
            if bx != old:
                sp = 1.0 / (grid - 1)
                lo2, hi2 = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo2, hi2, refine + 1)
                dd = xx - old
                cc = c[None, :] + dd[:, None] * A[None, :]
                cc[:, n-1] -= dd * dd
                j = int(np.argmin(cc.max(axis=1)))
                bx = float(xx[j])
                if bx != old:
                    d = bx - old
                    c[p:p+n] += d * om
                    c[st:st+n] -= x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-13:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    n_best = n0
    rev = h0[::-1]
    for v in (0.5*(h0 + rev), 0.5*(h0 + 1.0 - rev)):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv
    if time.time() > deadline - 60:
        return project_simplex(best_h, n_best), best_c, n_best
    t1 = min(deadline - 500, t0 + 250)
    h1, c1 = subgradient_opt(best_h, n_best, t1, n_it=60000, n_top=3)
    if c1 < best_c:
        best_h, best_c = h1, c1
    h1, c1 = smooth_opt(best_h, n_best, [500, 5000, 50000], 100, t1)
    if c1 < best_c:
        best_h, best_c = h1, c1
    t2 = min(deadline - 350, t0 + 450)
    h1, c1 = coord_cd(best_h, n_best, t2, grid=40, refine=14, max_passes=5)
    if c1 < best_c:
        best_h, best_c = h1, c1
    t3 = min(deadline - 200, t0 + 700)
    hb, cb = binary_local(best_h, n_best, rng, t3)
    if hb is not None and cb < best_c:
        best_h, best_c = hb, cb
        t4 = min(deadline - 150, time.time() + 120)
        h1, c1 = smooth_opt(best_h, n_best, [10000, 100000], 80, t4)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, t4, grid=50, refine=16, max_passes=3)
        if c1 < best_c:
            best_h, best_c = h1, c1
    cands = [500, 498, 502, 400, 600, 450, 550, 350, 700, 496, 504]
    tried = [n_best]
    for nt in cands:
        if len(tried) >= 6:
            break
        if time.time() > deadline - 80:
            break
        if nt in tried or nt < 200 or nt > 1000 or nt % 2:
            continue
        tried.append(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = project_simplex(ht, nt), cv, nt
        chunk = min(deadline - 60, time.time() + 60)
        if time.time() < chunk:
            h1, c1 = smooth_opt(ht, nt, [5000], 50, chunk)
            h1, c1 = coord_cd(h1, nt, chunk, grid=30, refine=10, max_passes=2)
            hb2, cb2 = binary_local(h1, nt, rng, chunk)
            if hb2 is not None and cb2 < best_c:
                best_h, best_c, n_best = hb2, cb2, nt
    tf1 = min(deadline - 40, time.time() + 80)
    if time.time() < tf1:
        h1, c1 = subgradient_opt(best_h, n_best, tf1, n_it=40000, n_top=5)
        if c1 < best_c:
            best_h, best_c = h1, c1
    tf2 = min(deadline - 10, time.time() + 60)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, tf2, grid=60, refine=20, max_passes=3)
        if c1 < best_c:
            best_h, best_c = h1, c1
    final_h = project_simplex(best_h, n_best)
    final_c = compute_c5(final_h, n_best)
    return final_h, final_c, int(n_best)
