import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.clip(np.nan_to_num(z), 0.0, 1.0)
    target = 0.5 * n
    if abs(z.sum() - target) < 1e-12:
        return z.copy()
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = y < 1.0
            room = (1.0 - y)[mask]
            if room.sum() > 1e-12:
                y[mask] += (res / room.sum()) * room
        else:
            mask = y > 0.0
            room = y[mask]
            if room.sum() > 1e-12:
                y[mask] += (res / room.sum()) * room
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    return (m + np.log(s) / alpha) * (2.0 / n)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    p = e / e.sum()
    off = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e2 = -d
            g[:n - e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n - e2]
    return (2.0 / n) * g

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_max_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        x = project_feasible(res.x, n)
        return x, compute_c5(x, n)
    except Exception:
        return h, compute_c5(h, n)

def pair_search(h, n, max_tries, grid_pts, deadline, rng):
    h = h.copy()
    off = n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    accepts = 0
    no_imp = 0
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.01, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        # Select pair linked to an active offset
        if trial % 3 == 2:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m))
                i, j = b + d, b
            else:
                e2 = -d
                m = n - e2
                if m <= 0:
                    continue
                b = e2 + int(rng.randint(m))
                i, j = b - e2, b
            if i == j:
                continue
        # Evaluate: change h[i] by u, h[j] by -u (preserves sum)
        hi, hj = h[i], h[j]
        S = hi + hj
        lo_t = max(0.0, S - 1.0)
        hi_t = min(1.0, S)
        if hi_t - lo_t < 1e-14:
            no_imp += 1
            continue
        us = np.linspace(lo_t - hi, hi_t - hi, grid_pts)
        # Compute effect on c
        # B[k] = dc[k]/d(h[i]) - dc[k]/d(h[j])
        B = np.zeros(2 * n - 1)
        B[off] = (1 - 2*hi) - (1 - 2*hj)
        for d in range(1, n):
            bi = 0.0
            if i >= d:
                bi += 1.0 - h[i-d]
            if i + d < n:
                bi -= h[i+d]
            bj = 0.0
            if j >= d:
                bj += 1.0 - h[j-d]
            if j + d < n:
                bj -= h[j+d]
            B[off+d] = bi - bj
            B[off-d] = bi - bj
        # c_new = c + u*B, with quadratic corrections
        # c[off]: h[i](1-h[i]) changes by (hi+u)(1-hi-u) - hi(1-hi) = u - u^2
        #         h[j](1-h[j]) changes by (hj-u)(1-hj+u) - hj(1-hj) = -u - u^2
        # Total c[off] change: -2u^2
        # c[off | i-j|]: h[i] and h[j] interact
        # For offset d=|i-j|: the term h[max(i,j)]*(1-h[min(i,j)]) or similar
        # Quadratic corrections:
        di = i - j
        # c[off+di] or c[off-di] gets +u^2, c[off] gets -2u^2
        # Actually from the previous code: cn[off] -= 2*u^2, cn[off+i-j] += u^2, cn[off+j-i] += u^2
        u2 = us * us
        c_arr = c[None, :] + us[:, None] * B[None, :]
        c_arr[:, off] -= 2.0 * u2[:, None]
        c_arr[:, off + di] += u2[:, None]
        c_arr[:, off - di] += u2[:, None]
        f = c_arr.max(axis=1)
        idx = int(np.argmin(f))
        if f[idx] < cur_max - 1e-10:
            u = us[idx]
            t = np.clip(hi + u, 0.0, 1.0)
            u = t - hi
            h[i] = t
            h[j] = S - t
            c = c + u * B
            c[off] -= 2.0 * u * u
            c[off + di] += u * u
            c[off - di] += u * u
            accepts += 1
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > max(500, max_tries // 3):
            break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_full(h, n, deadline, rng):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    # Symmetries
    for sym in [0.5*(h + h[::-1]), 1.0-h, 0.5*(h + 1.0 - h[::-1])]:
        sym = project_feasible(sym, n)
        cs = compute_c5(sym, n)
        if cs < c:
            h, c = sym, cs
    # SLSQP with increasing alpha
    for alpha, mx in [(50, 60), (200, 50), (1000, 40), (5000, 35), (20000, 30), (100000, 25)]:
        if time.time() > deadline - 10.0:
            break
        h2, c2 = slsqp_opt(h, n, alpha, mx, deadline)
        if c2 < c:
            h, c = h2, c2
    # Pair search (intensive)
    rem = deadline - time.time()
    mt = int(min(8000, max(500, rem * 15)))
    h, c = pair_search(h, n, mt, 48, deadline, rng)
    return h, compute_c5(h, n)

    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n = 300
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = np.full(n, 0.5)
        for m in range(1, 6):
            h0 += rng.uniform(-0.4, 0.4) * np.sin(np.pi * m * x + rng.uniform(0, 6.28))
        h0 = project_feasible(h0, n)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    # Phase 1: Optimize at native resolution
    main_dl = hard - max(60.0, 0.3 * budget_s)
    if main_dl < t0 + 20.0:
        main_dl = hard
    best_h, best_c = optimize_full(best_h, n0, main_dl, rng)
    # Phase 2: Try other resolutions
    cands = [n0]
    for m in [200, 300, 400, 500, 600]:
        if m != n0:
            cands.append(m)
    cands.sort(key=lambda x: abs(x - n0))
    for m in cands:
        if time.time() > hard - 40.0:
            break
        rem = hard - time.time()
        if m > 500 and rem < 60.0:
            continue
        if m < 300 and rem < 30.0:
            continue
        dl = time.time() + min(45.0, 0.15 * rem)
        if dl < time.time() + 10.0:
            continue
        ht = interp_h(best_h, m)
        ht, ct = optimize_full(ht, m, dl, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Phase 3: Perturbation restarts
    n = len(best_h)
    for s in [0.05, 0.1, 0.2]:
        if time.time() > hard - 40.0:
            break
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        dl = min(hard - 10.0, time.time() + 35.0)
        ht, ct = optimize_full(hp, n, dl, rng)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Phase 4: Final pair search polish
    if time.time() < hard - 10.0:
        rem = hard - time.time()
        n = len(best_h)
        mt = int(min(5000, max(300, rem * 20)))
        best_h, best_c = pair_search(best_h, n, mt, 64, hard - 2.0, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
