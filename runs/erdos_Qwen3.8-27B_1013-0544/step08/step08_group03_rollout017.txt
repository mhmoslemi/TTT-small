def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2} using bisection on lambda."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Find lambda such that sum(clamp(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, n, beta):
    """Compute value and gradient of smooth max approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2n-1,)
    # Softmax weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(beta * c_shifted)
    w_sum = w.sum()
    w = w / w_sum  # shape (2n-1,)
    val = float(c.max() + np.log(w_sum) / beta)
    
    # Gradient: grad_j = dx * sum_k w_k * [(1-h[j-s_k]) * 1_{0<=j-s_k<n} - h[j+s_k] * 1_{0<=j+s_k<n}]

    grad = np.zeros(n)
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)

        sk = s[k]
        wk = w[k] * dx
        # j such that j - sk is in [0, n-1]: j in [sk, sk+n-1] ∩ [0, n-1]
        j_lo = max(0, sk)
        j_hi = min(n-1, n-1+sk)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wk * (1.0 - h[j_lo-sk:j_hi-sk+1])
        # j such that j + sk is in [0, n-1]: j in [-sk, n-1-sk] ∩ [0, n-1]
        j_lo2 = max(0, -sk)
        j_hi2 = min(n-1, n-1-sk)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wk * h[j_lo2+sk:j_hi2+sk+1]
    
    return val, grad

s_4=2: 0-2=-2, not valid.

So first = conv[n-1 : 2n-1] works.

Now for second:
second[0] = ∑_k w[k] * h[0+s_k] * 1_{0≤0+s_k<n}
s_0=-2: 0+(-2)=-2, not valid
s_1=-1: 0+(-1)=-1, not valid
s_2=0: 0+0=0, valid. w[2]*h[0]
s_3=1: 0+1=1, valid. w[3]*h[1]
s_4=2: 0+2=2, valid. w[4]*h[2]
So second[0] = w[2]*h[0] + w[3]*h[1] + w[4]*h[2]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_value_grad(h, n, beta, fft_cache):
    """Compute smooth max value and gradient using FFT for speed."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx  # (2n-1,)
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    val = float(c_max + np.log(exp_c.sum()) / beta)
    
    # FFT-based gradient
    L = 1
    while L < 3*n:  # need length >= 3n-1 for full convolution of length-n and length-(2n-1)
        L *= 2
    
    H1 = np.fft.rfft(1.0 - h, n=L)
    H2 = np.fft.rfft(h, n=L)
    W = np.fft.rfft(w, n=L)
    
    first = np.fft.irfft(H1 * W, n=L)[n-1:2*n-1]
    
    W_rev = np.fft.rfft(w[::-1], n=L)
    second = np.fft.irfft(H2 * W_rev, n=L)[n-1:2*n-1]
    
    grad = dx * (first - second)
    return val, grad

def adam_smooth_max(h, n, beta_list, steps_per_beta, deadline, lr=0.01):
    """Optimize using smoothed max with Adam."""
    h = h.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Precompute FFT size
    L = 1
    while L < 3 * n:
        L *= 2
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_list:
        for step in range(steps_per_beta):
            if time.time() > deadline:
                return best_h, best_c
            t += 1
            val, grad = smooth_max_value_grad(h, n, beta, L)
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad**2
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, target)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        # Evaluate at end of each beta
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_val_grad(h, n, beta, L):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    w = e / e.sum()
    val = c_max + np.log(e.sum()) / beta
    H1 = np.fft.rfft(1.0 - h, n=L)
    H2 = np.fft.rfft(h, n=L)
    W = np.fft.rfft(w, n=L)
    Wr = np.fft.rfft(w[::-1], n=L)
    first = np.fft.irfft(H1 * W, n=L)[n-1:2*n-1]
    second = np.fft.irfft(H2 * Wr, n=L)[n-1:2*n-1]
    grad = dx * (first - second)
    return val, grad

def adam_smooth(h, n, beta_list, steps_per_beta, deadline, lr=0.005):
    h = h.copy()
    target = n * 0.5
    best_h, best_c = h.copy(), compute_c5(h, n)
    L = 1
    while L < 3 * n:
        L *= 2
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for beta in beta_list:
        for step in range(steps_per_beta):
            if time.time() > deadline:
                return best_h, best_c
            t += 1
            _, grad = smooth_max_val_grad(h, n, beta, L)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            h = project_feasible(h - lr * mh / (np.sqrt(vh) + eps), target)
            if step % 10 == 0:
                c_now = compute_c5(h, n)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    return best_h, best_c

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=30, gs_iters=30):
    h = h.copy()
    target = n * 0.5
    s_arr = np.arange(2 * n - 1) - (n - 1)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            j1 = p - s_arr
            valid1 = (j1 >= 0) & (j1 < n)
            j2 = p + s_arr
            valid2 = (j2 >= 0) & (j2 < n)
            A = np.zeros(2 * n - 1)
            A[valid1] += (1.0 - h[j1[valid1]])
            A[valid2] -= h[j2[valid2]]
            D = (np.arange(2 * n - 1) == (n - 1)).astype(float)
            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.array([np.max(c0 + t * A - t * t * D) for t in grid])
            bi = int(np.argmin(fvals))
            lo = grid[max(0, bi - 1)]
            hi = grid[min(len(grid) - 1, bi + 1)]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
            fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
            for _ in range(gs_iters):
                if fc1 < fd1:
                    b = d1; d1 = c1; fd1 = fc1
                    c1 = b - gr * (b - a)
                    fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                else:
                    a = c1; c1 = d1; fc1 = fd1
                    d1 = a + gr * (b - a)
                    fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
            t_star = (a + b) * 0.5
            f_star = float(np.max(c0 + t_star * A - t_star * t_star * D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            if min(f_star, f0, f1) < cur_max - 1e-13:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                cur_max = float(np.max(c))
                improved = True
        h = project_feasible(h, target)
        if not improved:
            break
    return h, compute_c5(h, n)

def interp_h(h_old, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(xn, xo, h_old), n_new * 0.5)

    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:

        n0 = len(h0)
    except Exception:

    best_h = project_feasible(h0, n0 * 0.5)

    # Phase 1: Smoothed max Adam with increasing beta
    h1, c1 = adam_smooth(best_h, n0,
                         beta_list=[5, 15, 40, 100, 250, 600, 1500],
                         steps_per_beta=1000, deadline=deadline, lr=0.006)
    if c1 < best_c:

    # Phase 2: Coordinate descent
    h2, c2 = exact_cd(best_h, n0, max_passes=150, deadline=deadline, grid_pts=30, gs_iters=30)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Try different n
    n_opts = [400, 600, 800]
    for n_try in n_opts:
        if time.time() > deadline - 200:
            break
        ht = interp_h(best_h, len(best_h), n_try)
        ht, ct = adam_smooth(ht, n_try,
                            beta_list=[20, 100, 500],
                            steps_per_beta=800, deadline=deadline, lr=0.005)
        ht, ct = exact_cd(ht, n_try, max_passes=100, deadline=deadline)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 4: Restart with noise / random
    n_best = len(best_h)
    time_left = deadline - time.time()
    n_restarts = max(1, int(time_left / 150))
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        if r % 3 == 0:
            hr = project_feasible(rng.uniform(0, 1, n_best), n_best * 0.5)
        else:

            hr = project_feasible(best_h + rng.normal(0, noise_scale, n_best), n_best * 0.5)
        hr, cr = adam_smooth(hr, n_best,
                            beta_list=[30, 100, 500, 2000],
                            steps_per_beta=600, deadline=deadline, lr=0.005)
        hr, cr = exact_cd(hr, n_best, max_passes=100, deadline=deadline)
        if cr < best_c:
            best_h, best_c = hr, cr

    # Phase 5: Final high-beta polish
    if time.time() < deadline - 25:
        hf, cf = adam_smooth(best_h, len(best_h),
                            beta_list=[3000, 8000, 20000],
                            steps_per_beta=400, deadline=deadline, lr=0.002)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 6: Final CD with fine grid
    if time.time() < deadline - 5:

                         grid_pts=50, gs_iters=45)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h) * 0.5)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
