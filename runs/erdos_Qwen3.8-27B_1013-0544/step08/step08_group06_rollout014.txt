            a[vh] += (1.0 - h[jh[vh]])

                            best_d = xv

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes, deadline, grid_size):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if p % 2 == 0:
            idx_order = range(n)
        else:
            idx_order = range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            if m_hi < 2 * n - 2:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            di = diag_idx[0] if has_diag else -1
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, di] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y0, y1, y2 = vals[best_i - 1], vals[best_i], vals[best_i + 1]
                x0, x1, x2 = d_grid[best_i - 1], d_grid[best_i], d_grid[best_i + 1]
                denom = y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)) / denom
                    if x0 < xv < x2:
                        nv1 = c_aff + xv * a
                        if has_diag:

                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-15:

                            best_d = xv
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-15:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[di] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))

            current_c5 = float(c.max())
    h = project_feasible(h, n)
