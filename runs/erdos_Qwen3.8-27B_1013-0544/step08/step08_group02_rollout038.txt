    m = float(c.max())

    e = np.exp(z - z.max())
    p = e / e.sum()
    # Grad of c_k w.r.t. h[i]

            break
        cur_max = float(c.max())
        
        # Pair selection
        tol = max(0.003, 0.001 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        
        if trial % 11 == 10 or active.size <= 1:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (i + 1) % n
        else:
            k = int(active[int(rng.randint(active.size))])
            d = k - off
            if abs(d) >= n:
                continue
            ae = abs(d)
            m = n - ae
            if m <= 1:
                continue
            b = int(rng.randint(m))
            if d >= 0:
                i, j = b + d, b
            else:
                i, j = b + ae, b + ae + d  # i = b-ae, j = b... 
        
        # Hmm let me reconsider the pair selection from active constraint k
        # c[off+d] involves pairs (i, i+d) for i=0..n-1-d (if d>0)
        # The "within the same product" pairs are (i, i+d)
        # But changing h[i] and h[j] affects all c_k, not just c[off+d]
        # So the pair selection based on active k is: pick two variables that
        # both appear in c[off+d]
        
        # Actually, for the purpose of reducing c[off+d], we should pick
        # variables that have large |A[., off+d]|
        
        # Let me use a simpler strategy: pick a random k from active,
        # then pick i and j to be two random indices that appear in c[off+k-off]
        # i.e., i and i+d (or i and i-d)
        
        S = h[i] + h[j]
        lo = max(0.0, S - 1.0)
        hiT = min(1.0, S)
        if hiT - lo <= 1e-14:
            no_imp += 1
            if no_imp > stall:
                no_imp = 0
            continue
        
        B = A[i] - A[j]
        u_lo = lo - h[i]
        u_hi = hiT - h[i]
        
        # Vectorized 1D search
        us = np.linspace(u_lo, u_hi, grid)
        u2 = us * us
        lines = c[None, :] + us[:, None] * B[None, :]
        lines[:, off] -= 2.0 * u2
        lines[:, off + i - j] += u2
        lines[:, off + j - i] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        
        if vals[idx] < cur_max - 5e-10:
            u = float(us[idx])
            h[i] += u
            h[j] -= u
            c += u * B
            c[off] -= 2.0 * u * u
            c[off + i - j] += u * u
            c[off + j - i] += u * u
            accepts += 1
            no_imp = 0
            if accepts % 50 == 0:
                A = compute_A(h, n)
                c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        else:
            no_imp += 1
            if no_imp > stall:
                no_imp = 0
    
    c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(c.max()) * (2.0 / n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)
