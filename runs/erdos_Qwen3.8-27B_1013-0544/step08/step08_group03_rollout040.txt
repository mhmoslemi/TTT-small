def fast_c(h,n):
    N=_fft_len(n)
    Fh=np.fft.rfft(h,N)
    aut=np.fft.irfft(Fh*np.conj(Fh),N)
    Gpos=aut[:n]
    if Gpos.size:
        Gpos=np.maximum(Gpos,0.0)
    pref=np.empty(n+1,dtype=float); pref[0]=0.0; pref[1:]=np.cumsum(h)

    if n>1:

import numpy as np
import time
from scipy.optimize import minimize

_SM_N = 0
_SM_P = 1.0


def _fft_len(n):
    return 1 << (2 * n - 1).bit_length()


def _scaled_mit(base, n):
    return max(40, int(base * 500.0 / max(50, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if n == 1:
        return np.array([0.5])
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    if abs(float(h.sum()) - target) <= 1e-14:
        return h
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min())
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-13:
        mask = (out > 1e-15) & (out < 1.0 - 1e-15)
        if mask.any():
            out[mask] += diff / float(mask.sum())
            out = np.clip(out, 0.0, 1.0)
    return out


def fast_c(h, n):
    N = _fft_len(n)
    Fh = np.fft.rfft(h, N)
    aut = np.fft.irfft(Fh * np.conj(Fh), N)
    Gpos = np.maximum(aut[:n], 0.0)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    L = 2 * n - 1
    c = np.empty(L, dtype=float)
    r = np.arange(n)
    c[n - 1:n - 1 + n] = pref[n - r] - Gpos
    if n > 1:
        c[:n - 1] = pref[n] - pref[1:n] - Gpos[1:n]
    np.clip(c, 0.0, None, out=c)
    return c


def c5_fast(h):
    n = len(h)
    return float(np.max(fast_c(h, n)) * (2.0 / n))


def c5_true(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _grad_from_weights(h, n, w):
    L = 2 * n - 1
    N = _fft_len(n)
    Fh = np.fft.rfft(h, N)
    if n == 1:
        return float(w[0]) * (1.0 - 2.0 * h)
    a1 = np.zeros(N)
    a1[1:n] = w[n:]
    a1[N - (n - 1):] = w[:n - 1]
    a2 = np.zeros(N)
    a2[1:n] = w[:n - 1][::-1]
    a2[N - (n - 1):] = w[n:2 * n - 1][::-1]
    B1 = np.fft.irfft(np.fft.rfft(a1, N) * np.conj(Fh), N)
    B2 = np.fft.irfft(np.fft.rfft(a2, N) * np.conj(Fh), N)
    idx = np.arange(n)
    B1 = np.maximum(B1[(-idx) % N], 0.0)
    B2 = np.maximum(B2[(-idx) % N], 0.0)
    w0 = float(w[n - 1])
    wno = w.copy()
    wno[n - 1] = 0.0
    cw = np.empty(L + 1)
    cw[0] = 0.0
    cw[1:] = np.cumsum(wno)
    K1 = cw[2 * n - 1 - idx] - cw[n - 1 - idx]
    return w0 * (1.0 - 2.0 * h) + K1 - B1 - B2


def _smooth_eval(x):
    n = _SM_N
    p = float(_SM_P)
    x = np.asarray(x, dtype=float)
    c = fast_c(x, n)
    maxc = float(c.max())
    if maxc <= 1e-15:
        return 0.0, np.zeros(n, dtype=float)
    z = p * (c - maxc)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = e / s
    val = float(maxc + (2.0 / n) * (np.log(s) / p))
    grad = (2.0 / n) * _grad_from_weights(x, n, w)
    return val, grad


def _sm_objective(x):
    return _smooth_eval(x)[0]


def _sm_jacobian(x):
    return _smooth_eval(x)[1]


def _sum_fun(x, target):
    return float(np.sum(x) - target)


def _sum_jac(x, target):
    return np.ones_like(x, dtype=float)


def slsqp_smooth(h, n, p, maxiter, deadline=None):
    global _SM_N, _SM_P
    h = project_feasible(h, n)
    _SM_N = int(n)
    _SM_P = float(p)
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_fun, "jac": _sum_jac, "args": (target,)}]
    x = h
    try:
        res = minimize(
            _sm_objective,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": 0},
        )
        if res.x is not None:
            x = res.x
    except Exception:
        x = h
    x = project_feasible(np.asarray(x, dtype=float), n)
    return x, c5_true(x)


def interpolate_h(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.42 * np.cos(np.pi * x) + 0.04 * rng.normal(0.0, 1.0, n)
    return project_feasible(h, n)


def _grad_topk_uniform(h, n, idx):
    g = np.zeros(n, dtype=float)
    if idx.size == 0:
        return g
    w = 1.0 / float(idx.size)
    center = n - 1
    for s in idx:
        lag = int(s) - center
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            g[:m] += w * (1.0 - h[lag:lag + m])
            g[lag:] += -w * h[:m]
        else:
            r = -lag
            m = n - r
            g[r:] += w * (1.0 - h[:m])
            g[:m] += -w * h[r:r + m]
    return g


def active_polish(h, n, max_iters=100, deadline=None):
    h = project_feasible(h, n)
    cur = c5_fast(h)
    topks = (1, 2, 4, 8, 16, 32)
    it = 0
    prog = 0
    stale = 0
    L = 2 * n - 1
    while it < max_iters:
        if deadline is not None and time.time() > deadline:
            break
        c = fast_c(h, n)
        k = topks[prog % len(topks)]
        k = min(k, L)
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, -k)[-k:]
        g = _grad_topk_uniform(h, n, idx)
        gn = float(np.linalg.norm(g))
        it += 1
        prog += 1
        if gn <= 1e-14:
            stale += 1
        else:
            alpha = 0.05 / gn
            improved = False
            for _ in range(12):
                y = project_feasible(h - alpha * g, n)
                ny = c5_fast(y)
                if ny < cur - 1e-15:
                    h = y
                    cur = ny
                    stale = 0
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-10:
                    break
            if not improved:
                stale += 1
        if stale >= 3 * len(topks):
            break
    h = project_feasible(h, n)
    return h, c5_true(h)


def _cd_f(base, A, t, tmp, n):
    np.multiply(A, t, out=tmp)
    tmp += base
    tmp[n - 1] -= t * t
    return float(tmp.max())


def cd_polish(h, n, max_passes=3, deadline=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_unscaled = float(np.max(fast_c(best_h, n)))
    dx = 2.0 / n
    L = 2 * n - 1
    ts = np.linspace(0.0, 1.0, 15)
    gr = 0.3819660112501051
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        hw = best_h.copy()
        c = fast_c(hw, n)
        cur = float(c.max())
        tmp = np.empty(L, dtype=float)
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            hp = hw[p]
            A = np.zeros(L, dtype=float)
            A[n - 1 - p:n - 1 - p + n] += 1.0 - hw
            A[p:p + n] -= hw[::-1]
            A[n - 1] += 2.0 * hp
            base = c - hp * A
            base[n - 1] += hp * hp
            fvals = np.empty(ts.size, dtype=float)
            for i, t in enumerate(ts):
                fvals[i] = _cd_f(base, A, t, tmp, n)
            bi = int(np.argmin(fvals))
            lo = ts[max(0, bi - 1)]
            hi = ts[min(ts.size - 1, bi + 1)]
            if hi - lo > 1e-5:
                a, b = lo, hi
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc1 = _cd_f(base, A, c1, tmp, n)
                fd1 = _cd_f(base, A, d1, tmp, n)
                for __ in range(22):
                    if fc1 < fd1:
                        b = d1
                        d1 = c1
                        fd1 = fc1
                        c1 = b - gr * (b - a)
                        fc1 = _cd_f(base, A, c1, tmp, n)
                    else:
                        a = c1
                        c1 = d1
                        fc1 = fd1
                        d1 = a + gr * (b - a)
                        fd1 = _cd_f(base, A, d1, tmp, n)
                tb = 0.5 * (a + b)
            else:
                tb = float(ts[bi])
            f0 = float(fvals[0])
            f1 = float(fvals[-1])
            ftb = _cd_f(base, A, tb, tmp, n)
            fbest = min(f0, f1, ftb)
            if fbest < cur - 1e-13:
                if f0 <= ftb and f0 <= f1:
                    tb = 0.0
                    fbest = f0
                elif f1 <= ftb and f1 <= f0:
                    tb = 1.0
                    fbest = f1
                else:
                    fbest = ftb
                hw[p] = tb
                c += (tb - hp) * A
                c[n - 1] -= tb * tb - hp * hp
                cur = max(cur, fbest)
                cur = float(c.max())
        hw = project_feasible(hw, n)
        cm = float(np.max(fast_c(hw, n)))
        if cm < best_unscaled - 1e-12:
            best_unscaled = cm
            best_h = hw
    return best_h, best_unscaled * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 500
        h = make_initial(n, seed)
    if n < 16 or n > 1400:
        target_n = 500 if n < 1000 else 800
        h = interpolate_h(project_feasible(h, n), n, target_n)
        n = target_n
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c = c5_true(best_h)
    cur_n = int(len(best_h))
    rng = np.random.RandomState(seed)

    def _try_update(h_new, c_new):
        nonlocal best_h, best_c, cur_n
        if c_new < best_c - 1e-16:
            best_h = h_new
            best_c = c_new
            cur_n = int(len(best_h))

    if time.time() < deadline - 90.0:
        for p, base in ((50, 220), (150, 260), (500, 300), (2000, 340)):
            if time.time() > deadline - 80.0:
                break
            h_new, c_new = slsqp_smooth(best_h, cur_n, p, _scaled_mit(base, cur_n))
            _try_update(h_new, c_new)
            if time.time() < deadline - 60.0:
                h_new, c_new = active_polish(best_h, cur_n, 25, deadline - 5.0)
                _try_update(h_new, c_new)

    for nt in (400, 550, 600, 700, 800, 350):
        if time.time() > deadline - 180.0:
            break
        if nt == cur_n or nt > 1000:
            continue
        h_try = interpolate_h(best_h, cur_n, nt)
        for p, base in ((100, 140), (500, 180)):
            if time.time() > deadline - 120.0:
                break
            h_new, c_new = slsqp_smooth(h_try, nt, p, _scaled_mit(base, nt))
            if c_new < best_c - 1e-16:
                best_h = h_new
                best_c = c_new
                cur_n = nt
                h_try = best_h
        if time.time() < deadline - 100.0:
            h_new, c_new = active_polish(h_try, nt, 30, deadline - 5.0)
            _try_update(h_new, c_new)

    if time.time() < deadline - 160.0:
        for sig in (0.025, 0.07):
            if time.time() > deadline - 100.0:
                break
            h_r = project_feasible(best_h + sig * rng.normal(0.0, 1.0, cur_n), cur_n)
            h_new, c_new = slsqp_smooth(h_r, cur_n, 200, _scaled_mit(180, cur_n))
            _try_update(h_new, c_new)

    if time.time() < deadline - 110.0:
        for p, base in ((1000, 330), (5000, 380)):
            if time.time() > deadline - 50.0:
                break
            h_new, c_new = slsqp_smooth(best_h, cur_n, p, _scaled_mit(base, cur_n))
            _try_update(h_new, c_new)

    if time.time() < deadline - 70.0:
        h_new, c_new = cd_polish(best_h, cur_n, 3, deadline - 5.0)
        _try_update(h_new, c_new)

    if time.time() < deadline - 25.0:
        h_new, c_new = active_polish(best_h, cur_n, 130, deadline - 8.0)
        _try_update(h_new, c_new)

    if time.time() < deadline - 10.0:
        h_new, c_new = cd_polish(best_h, cur_n, 2, deadline - 4.0)
        _try_update(h_new, c_new)

    safe_h = project_feasible(best_h, len(best_h))
    safe_c = c5_true(safe_h)

    out_h = project_feasible(best_h, len(best_h))
    out_c = c5_true(out_h)
    if out_c > safe_c:
        out_h = safe_h
        out_c = safe_c

    return out_h, out_c, int(len(out_h))
