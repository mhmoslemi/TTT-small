The key insight is: f(x) is the pointwise maximum of a finite set of functions. The minimum of f over [0,1] is the smallest T such that the set {x ∈ [0,1] : g_i(x) ≤ T for all i} is non-empty.

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-14:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size:
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size:
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def optimize_coord(hp, cseg, a, qconst, unaff):
    m = a.size
    def fval(x):
        val = unaff
        q = qconst + x * (1.0 - x)
        if q > val:
            val = q
        if m:
            v = float(np.max(cseg + a * (x - hp)))
            if v > val:
                val = v
        return val
    if m == 0:
        best_x, best_f = hp, fval(hp)
        for x in [0.0, 1.0, 0.5]:
            fx = fval(x)
            if fx < best_f:
                best_f, best_x = fx, x
        return best_x, best_f
    b = cseg - a * hp
    zero = np.abs(a) < 1e-15
    T_lo = unaff
    if zero.any():
        T_lo = max(T_lo, float(b[zero].max()))
    T_hi = min(fval(0.0), fval(1.0), fval(hp))
    pos = a > 1e-15
    neg = a < -1e-15
    for _ in range(30):
        T = 0.5 * (T_lo + T_hi)
        if T < unaff - 1e-13 or T < T_lo - 1e-13:
            T_lo = T
            continue
        x_lo, x_hi = 0.0, 1.0
        if pos.any():
            v = (T - b[pos]) / a[pos]
            x_hi = min(x_hi, float(v.min()))
        if neg.any():
            v = (T - b[neg]) / a[neg]
            x_lo = max(x_lo, float(v.max()))
        if x_lo > x_hi + 1e-12:
            T_lo = T
            continue
        disc = 1.0 - 4.0 * (T - qconst)
        feasible = False
        if disc <= 0:
            feasible = True
        else:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if x_hi <= r1 + 1e-13 or x_lo >= r2 - 1e-13:
                feasible = True
            elif x_lo <= r1 + 1e-13 and x_hi >= r2 - 1e-13:
                feasible = True
        if feasible:
            T_hi = T
        else:
            T_lo = T
    T_opt = T_hi
    x_lo, x_hi = 0.0, 1.0
    if pos.any():
        v = (T_opt - b[pos]) / a[pos]
        x_hi = min(x_hi, float(v.min()))
    if neg.any():
        v = (T_opt - b[neg]) / a[neg]
        x_lo = max(x_lo, float(v.max()))
    disc = 1.0 - 4.0 * (T_opt - qconst)
    if disc > 0:
        r1 = (1.0 - np.sqrt(disc)) * 0.5
        r2 = (1.0 + np.sqrt(disc)) * 0.5
        if x_lo <= min(x_hi, r1):
            x_opt = 0.5 * (x_lo + min(x_hi, r1))
        else:
            x_opt = 0.5 * (max(x_lo, r2) + x_hi)
    else:
        x_opt = 0.5 * (max(0.0, x_lo) + min(1.0, x_hi))
    x_opt = max(0.0, min(1.0, x_opt))
    best_x, best_f = x_opt, fval(x_opt)
    for x in [0.0, 1.0, hp, x_opt]:
        x = max(0.0, min(1.0, x))
        fx = fval(x)
        if fx < best_f:
            best_f, best_x = fx, x
    return best_x, best_f

def cd_full(h, n, deadline, max_passes=200, project_every=15):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cmax
    center = n - 1
    L = 2 * n - 1
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                return best_h, best_c
            hp = h[p]
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=np.int64)
            j = p + s_arr
            i = p - s_arr
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if vj.any():
                a[vj] += 1.0 - h[j[vj]]
            if vi.any():
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1)
            asg = np.empty(seg - 1)
            if off > 0:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff)
            if bf < cmax - 1e-13:
                dxv = bx - hp
                if off > 0:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cmax = bf
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(c.max())
        cv = (2.0 / n) * cmax
        if cv < best_c - 1e-14:
            best_c = cv
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def thue_morse_init(n):
    t = np.zeros(n, dtype=float)
    for k in range(1, n):
        t[k] = t[k >> 1] ^ (k & 1)
    ones = int(t.sum())
    target = n // 2
    if ones > target:
        idx = np.where(t == 1)[0]
        t[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        t[idx[:target - ones]] = 1
    return t

def fourier_init(n, rng, nmodes=16):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    t = x - 1.0
    h = np.full(n, 0.5)
    for k in range(1, nmodes + 1):
        h += rng.uniform(-0.35, 0.35) / np.sqrt(k) * np.cos(k * np.pi * t + rng.uniform(0, 6.28))
    return project_feasible(h, n)

def block_init(n, rng):
    h = np.zeros(n)
    bl = rng.randint(3, 12)
    v = 0.0
    for i in range(0, n, bl):
        v = 1.0 - v
        h[i:i+bl] = v
    return project_feasible(h, n)

def perturb(h, n, rng, mode):
    if mode == 0:
        return project_feasible(h + rng.normal(0, 0.03, n), n)
    elif mode == 1:
        k = n // 2
        idx = np.argpartition(h, n - k)[n - k:]
        h2 = np.zeros(n)
        h2[idx] = 1.0
        return h2
    elif mode == 2:
        h2 = np.zeros(n)
        h2[rng.choice(n, n // 2, replace=False)] = 1.0
        return h2
    elif mode == 3:
        return fourier_init(n, rng)
    elif mode == 4:
        return thue_morse_init(n)
    elif mode == 5:
        return block_init(n, rng)
    elif mode == 6:
        s = rng.uniform(0.5, 1.5)
        return project_feasible(h * s, n)
    else:
        p = rng.randint(n // 4, 3 * n // 4)
        h2 = h.copy()
        h2[p:] = h[:n-p]
        h2[:n-p] = h[p:]
        return project_feasible(h2, n)
