def project_simplex(h, n):
    """Project h onto {x : 0<=x_i<=1, sum(x)=n/2}."""
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_max_and_grad(c, alpha):
    """Compute smooth max of c and its gradient w.r.t. c."""
    m = float(c.max())
    r = alpha * (c - m)
    r = r - r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    return m + np.log(z) / alpha, w

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    val, _ = smooth_max_and_grad(c, alpha)
    return val * (2.0 / n)

    _, w = smooth_max_and_grad(c, alpha)
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (A - B)[s:s + n] * (2.0 / n)

    idx = np.argpartition(h, target - 1)[:target]

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def _smooth_valgrad(c, alpha):
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    return m + np.log(z) / alpha, w

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    v, _ = _smooth_valgrad(c, alpha)
    return v * (2.0 / n)

    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (A - B)[s:s + n] * (2.0 / n)

def _sc(x, n):
    return x.sum() - 0.5 * n

def _scj(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy(); bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sc, "jac": _scj, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc
