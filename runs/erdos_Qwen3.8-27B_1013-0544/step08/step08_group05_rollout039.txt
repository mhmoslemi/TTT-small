def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)

def swap_delta_vec(h, i, j, n):
    """Vectorized computation of correlation change when h[i]:1->0, h[j]:0->1"""
    s = np.arange(1, n)
    # h[i+s]
    h_is = np.zeros(n - 1)
    v = (i + s) < n
    h_is[v] = h[i + s[v]]
    # h[j+s]
    h_js = np.zeros(n - 1)
    v = (j + s) < n
    h_js[v] = h[j + s[v]]
    # h[i-s]
    h_ims = np.zeros(n - 1)
    v = (i - s) >= 0
    h_ims[v] = h[i - s[v]]
    # h[j-s]
    h_jms = np.zeros(n - 1)
    v = (j - s) >= 0
    h_jms[v] = h[j - s[v]]
    # Positive shift s: delta at index n-1+s
    d_pos = h_is - h_js + h_ims - h_jms
    # Negative shift -s: delta at index n-1-s
    d_neg = h_is - h_js - h_ims + h_jms
    delta = np.zeros(2 * n - 1)
    delta[n:2*n-1] = d_pos  # indices n to 2n-2 correspond to s=1 to n-1
    delta[0:n-1] = d_neg    # indices 0 to n-2 correspond to s=-1 to -(n-1)
    # Wait, let me check the indexing
    # Correlation array has indices 0 to 2n-2
    # Index n-1 corresponds to s=0
    # Index n-1+s (s>0) corresponds to positive shift s
    # Index n-1-s (s>0) corresponds to negative shift -s
    # So for s=1: positive is at index n, negative is at index n-2
    # d_pos[s-1] goes to index n-1+s = n+s-1
    # d_neg[s-1] goes to index n-1-s = n-s-1
    delta[n-1+1:n-1+n] = d_pos   # indices n to 2n-2
    delta[0:n-1] = d_neg          # indices 0 to n-2
    # But d_neg[s-1] should go to index n-1-s = n-1-(s) 
    # For s=1: index n-2, d_neg[0]
    # For s=2: index n-3, d_neg[1]
    # So d_neg is in reverse order!
    delta[0:n-1] = d_neg[::-1]
    return delta

def smooth_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    # Gradient: d/dx of (m + log(z)/alpha) * dx
    # = (1/alpha) * (1/z) * sum_k exp(alpha*(c_k-m)) * dc_k/dx * dx
    # But c_k is bilinear in x, so dc_k/dx depends on x
    # A = conv(1-x, w), B = conv(x, w[::-1])
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    grad = (A - B)[s:s + n] * dx
    obj = (m + np.log(z) / alpha) * dx
    return obj, grad

    bounds = [(0.0, 1.0)] * n
    for a in alphas:

        try:

                smooth_obj_grad, best, args=(n, a), jac=True,

        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:

                (b[:n-s_star] == 0) & (b[s_star:] == 1)

                (b[t:] == 0) & (b[:-t] == 1)

            i = int(ones[rng.randint(len(ones))])

            d = swap_delta_vec(b, i, j, n)

            if new_max < cm:

                b[i] = 0
                b[j] = 1
                cm = new_max
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50000:
                    break
            continue
        
        # Try a subset of (offender, target) pairs
        n_try = min(20, len(offenders) * len(targets))
        found = False

            if no_improve > 100000:
                break
    
    return best_b, best_cm

For the subgradient method, the step is:
h_{k+1} = project(h_k - α_k * g_k)
where g_k is a subgradient of the max at h_k, and project is the Euclidean projection onto the constraint set.

The subgradient of max(c) at h is the average of the gradients of the active components. For the max to decrease, we need to reduce all active components simultaneously.

Let me implement this:

```python
def subgradient_step(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = c.max()
    active = c >= mx - 1e-6
    L = 2 * n - 1
    # Gradient of c[m] w.r.t h is:
    # For s = m - (n-1):
    # grad[p] = (1-h[p+s])*[0<=p+s<n] - h[p-s]*[0<=p-s<n]
    g = np.zeros(n)
    for m in np.where(active)[0]:
        s = int(m) - (n - 1)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            t = -s
            g[t:] += (1.0 - h[:t])
            g[:n-t] -= h[t:]
    g /= active.sum()
    # Projected gradient step
    h_new = project_simplex(h - alpha * g, n)
    return h_new
```

This is O(n * |active|) per step, which is O(n²) in the worst case. With n=500, that's 250,000 operations per step. We can do maybe 10,000 steps in a few seconds.

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d2 = t - y.sum()
    if abs(d2) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d2 / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def swap_delta_vec(h, i, j, n):
    s = np.arange(1, n)
    h_is = np.zeros(n - 1)
    v = (i + s) < n
    h_is[v] = h[i + s[v]]
    h_js = np.zeros(n - 1)
    v = (j + s) < n
    h_js[v] = h[j + s[v]]
    h_ims = np.zeros(n - 1)
    v = (i - s) >= 0
    h_ims[v] = h[i - s[v]]
    h_jms = np.zeros(n - 1)
    v = (j - s) >= 0
    h_jms[v] = h[j - s[v]]
    d_pos = h_is - h_js + h_ims - h_jms
    d_neg = h_is - h_js - h_ims + h_jms
    delta = np.zeros(2 * n - 1)
    delta[n:2 * n - 1] = d_pos
    delta[0:n - 1] = d_neg[::-1]
    return delta

def binary_search(b, n, deadline, rng, max_rounds=50000):
    b = b.astype(float).copy()
    c = np.correlate(b, 1.0 - b, mode="full").astype(float)
    cm = float(c.max())
    best_b = b.copy()
    best_cm = cm
    stall = 0
    T = 2.0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        T *= 0.9997
        max_val = c.max()
        if max_val < best_cm:
            best_cm = max_val
            best_b = b.copy()
            stall = 0
            continue
        max_idx = int(np.argmax(c))
        s_star = max_idx - (n - 1)
        found = False
        ntry = 80
        if s_star > 0:
            off = np.where((b[:n - s_star] == 1) & (b[s_star:] == 0))[0]
            tgt = np.where((b[:n - s_star] == 0) & (b[s_star:] == 1))[0]
        elif s_star < 0:
            t = -s_star
            off = np.where((b[t:] == 1) & (b[:-t] == 0))[0] + t
            tgt = np.where((b[t:] == 0) & (b[:-t] == 1))[0] + t
        else:
            off, tgt = np.where(b == 1)[0], np.where(b == 0)[0]
        if len(off) == 0:
            off = np.where(b == 1)[0]
        if len(tgt) == 0:
            tgt = np.where(b == 0)[0]
        for _ in range(ntry):
            i = int(off[rng.randint(len(off))])
            j = int(tgt[rng.randint(len(tgt))])
            if i == j:
                continue
            d = swap_delta_vec(b, i, j, n)
            nm = float((c + d).max())
            delta_e = nm - cm
            if delta_e < -1e-9:
                accept = True
            elif T > 0.01:
                accept = rng.rand() < np.exp(-delta_e / T)
            else:
                accept = False
            if accept:
                c += d
                b[i] = 0.0
                b[j] = 1.0
                cm = nm
                found = True
                break
        if not found:
            stall += 1
            if stall > 5000:
                break
    return best_b, best_cm

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    return float((m + np.log(np.exp(r).sum()) / alpha) * (2.0 / n))

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    w = ee / ee.sum()
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (A - B)[s:s + n] * (2.0 / n)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n).copy()
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                           method="SLSQP", bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x, n=n: x.sum() - 0.5 * n,
                                         'jac': lambda x, n=n: np.ones(n)}],
                           options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc

def coord_cd(h, n, max_passes, deadline, grid=32, refine=10):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A = np.zeros(L)
            A[p:p + n] += om
            st = diag - p
            if st >= 0:
                A[st:st + n] -= x
            else:
                A[-st:] -= x[-st:]
            cur = float(c.max())
            darr = xs0 - old
            cand = c[None, :] + darr[:, None] * A[None, :]
            cand[:, diag] -= darr * darr
            i = int(np.argmin(cand.max(axis=1)))
            bx, bv = float(xs0[i]), float(cand[i].max())
            if bv < cur - 1e-12:
                sp = 1.0 / (grid - 1)
                lo, hi2 = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo, hi2, refine + 1)
                darr2 = xx - old
                cand2 = c[None, :] + darr2[:, None] * A[None, :]
                cand2[:, diag] -= darr2 * darr2
                j = int(np.argmin(cand2.max(axis=1)))
                if cand2[j].max() < bv - 1e-12:
                    bx = float(xx[j])
                dd = bx - old
                c[p:p + n] += dd * om
                if st >= 0:
                    c[st:st + n] -= dd * x
                else:
                    c[-st:] -= dd * x[-st:]
                c[diag] -= dd * dd
                x[p] = bx
                om[diag - p] = 1.0 - bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-13:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc
