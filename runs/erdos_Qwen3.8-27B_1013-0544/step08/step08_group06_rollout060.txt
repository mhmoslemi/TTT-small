        base_lr = 4.0 / (alpha * dx)
        for it in range(15000):
            if time.time() > deadline:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            shifted = alpha * (c - c.max())
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300:
                break
            w /= ws
            g = np.zeros(n)
            active = np.where(w > 1e-7)[0]
            for m in active:
                k = m - (n - 1)
                wk = w[m]
                if k > 0:
                    g[:n-k] += wk * (1.0 - h[k:n])
                    g[k:] -= wk * h[:n-k]
                elif k < 0:
                    kp = -k
                    g[kp:] += wk * (1.0 - h[:n-kp])
                    g[:n-kp] -= wk * h[kp:]
                else:
                    g += wk * (1.0 - 2.0 * h)
            v = beta * v + (1.0 - beta) * g
            step = base_lr / (1.0 + 0.0005 * it)
            h = project_box_sum(h - step * v, n / 2.0)
            cur = compute_c5(h, n)
            if cur < best_c - 1e-10:
                best_c = cur
                best_h = h.copy()
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                tm = float(c[:m_lo].max())
                if tm > unaff_max:
                    unaff_max = tm
            if m_hi < 2 * n - 2:
                tm = float(c[m_hi + 1:].max())
                if tm > unaff_max:
                    unaff_max = tm
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = xv
            if abs(best_d) > 1e-12 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_mx = max(unaff_max, float(c[ms].max()))
                if new_mx < current_c5:
                    current_c5 = new_mx
                improved = True
        if not improved:
            break
    h = project_box_sum(h, n / 2.0)
    return h, compute_c5(h, n)

def topk_objective(h, n, alpha, k_frac=0.1):
    c = np.correlate(h, 1.0 - h, mode='full')
    k = max(3, int(k_frac * len(c)))
    topk = np.sort(c)[-k:]
    m = topk.max()
    shifted = alpha * (topk - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / k
    return lse * (2.0 / n)

def topk_gradient(h, n, alpha, k_frac=0.1):
    c = np.correlate(h, 1.0 - h, mode='full')
    k = max(3, int(k_frac * len(c)))
    topk_idx = np.argsort(c)[-k:]
    topk_vals = c[topk_idx]
    m = topk_vals.max()
    shifted = alpha * (topk_vals - m)
    w = np.exp(shifted)
    w /= (w.sum() * k)
    g = np.zeros(n)
    for i, m in enumerate(topk_idx):
        kk = m - (n - 1)
        wi = w[i]
        if kk > 0:
            g[:n-kk] += wi * (1.0 - h[kk:n])
            g[kk:] -= wi * h[:n-kk]
        elif kk < 0:
            kp = -kk
            g[kp:] += wi * (1.0 - h[:n-kp])
            g[:n-kp] -= wi * h[kp:]
        else:
            g += wi * (1.0 - 2.0 * h)
    return g * (2.0 / n)

def topk_sgd(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for alpha in [20.0, 100.0, 500.0]:
        if time.time() > deadline:
            break
        h = best_h.copy()
        v = np.zeros(n)
        beta = 0.9
        base_lr = 6.0 / (alpha * dx)
        for it in range(10000):
            if time.time() > deadline:
                break
            g = topk_gradient(h, n, alpha, 0.15)
            v = beta * v + (1.0 - beta) * g
            step = base_lr / (1.0 + 0.001 * it)
            h = project_box_sum(h - step * v, n / 2.0)
            cur = compute_c5(h, n)
            if cur < best_c - 1e-10:
                best_c = cur
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Top-k smoothed gradient (focus on top lags)
    phase1_deadline = min(deadline - 500, t0 + 80)
    h1, c1 = topk_sgd(h, n, phase1_deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Full smooth max gradient
    phase2_deadline = min(deadline - 400, t0 + 150)
    h2, c2 = sgd_phase(best_h, n, phase2_deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: CD polish
    h3, c3 = coord_cd(best_h, len(best_h), 3000, deadline - 350, grid_size=120)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: ILS with diverse perturbations
    for r in range(15):
        if time.time() > deadline - 100:
            break
        h_p = best_h.copy()
        n_pert = max(10, len(h_p) // 15)
        idxs = rng.choice(len(h_p), n_pert, replace=False)
        noise = rng.uniform(-0.15, 0.15, n_pert)
        h_p[idxs] = np.clip(h_p[idxs] + noise, 0.0, 1.0)
        h_p = project_box_sum(h_p, len(h_p) / 2.0)
        # Also try a "swap" perturbation: flip a few near-binary values
        n_flip = max(3, len(h_p) // 50)
        fx = rng.choice(len(h_p), n_flip, replace=False)
        h_p[fx] = 1.0 - h_p[fx]
        h_p = project_box_sum(h_p, len(h_p) / 2.0)
        h_r, c_r = coord_cd(h_p, len(h_p), 2000, deadline - 100, grid_size=90)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 5: Multi-n exploration
    n_list = [350, 400, 450, 600]
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in n_list:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_box_sum(np.interp(x_new, x_orig, best_h), n_try / 2.0)
        h_opt, c_opt = coord_cd(h_int, n_try, 2500, deadline - 70, grid_size=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Final CD with very fine grid
    if time.time() < deadline - 5:
        h_f, c_f = coord_cd(best_h, len(best_h), 5000, deadline, grid_size=200)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
