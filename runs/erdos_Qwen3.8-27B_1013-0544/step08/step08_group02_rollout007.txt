def compute_A(h, n):
    """Compute the matrix A where A[i,k] = dc_k/dh_i for each i,k"""
    A = np.zeros((n, 2*n-1))
    off = n - 1
    for k in range(2*n-1):
        d = k - off
        if d >= 0:
            A[d:, k] = 1.0 - h[:n-d]
            A[:n-d, k] -= h[d:]
        else:
            ei = -d
            A[:n-ei, k] = 1.0 - h[ei:]
            A[ei:, k] -= h[:n-ei]

def pair_search(h, n, max_tries, deadline, rng, grid=50):

    off = n - 1
    
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        # Active set: lags within 0.002 of max
        tol = 0.002
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        
        # Choose which lag to focus on
        kk = active[int(rng.randint(active.size))]
        d = kk - off
        if d >= 0:
            m = n - d
            if m <= 0: continue
            b = int(rng.randint(m))
            i, j = b + d, b
        else:
            ei = -d
            m = n - ei
            if m <= 0: continue
            b = ei + int(rng.randint(m))
            i, j = b - ei, b
        if i == j:
            continue
        
        # Try to find best u: h[i] += u, h[j] -= u
        # c_k changes by u * (A[i,k] - A[j,k]) for all k
        # Plus quadratic terms for k=off, off+i-j, off+j-i
        B = A[i] - A[j]
        
        # Binary search / grid search for best u
        hi_val = h[i] + (1.0 - h[j])  # max u such that h[i]+u <= 1 and h[j]-u >= 0
        lo_val = -(h[i])  # min u such that h[i]+u >= 0
        hi_val = min(hi_val, 1.0 - h[j])
        
        # Evaluate on a grid
        us = np.linspace(lo_val, hi_val, grid)
        best_u = 0
        best_nm = cur_max
        for u in us:
            if abs(u) < 1e-15:
                continue
            cn = c + u * B
            u2 = u * u
            cn[off] -= 2.0 * u2
            if 0 <= off + i - j < len(cn):
                cn[off + i - j] += u2
            if 0 <= off + j - i < len(cn):
                cn[off + j - i] += u2
            nm = float(cn.max())
            if nm < best_nm:
                best_nm = nm
                best_u = u
        
        if best_nm < cur_max - 1e-10:
            h[i] += best_u
            h[j] -= best_u
            h = np.clip(h, 0.0, 1.0)
            c = np.correlate(h, 1.0 - h, mode='full')
            A = compute_A(h, n)
    
    return h, compute_c5(h, n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def obj_smooth(x, n, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    if topk is not None and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        thresh = c[idx].min()
        c_eff = np.where(c >= thresh, c, c.max() - 10.0 / alpha)
    else:
        c_eff = c
    m = float(c_eff.max())
    z = alpha * (c_eff - m)
    e = np.exp(z - z.max())
    return (2.0 / n) * (m + np.log(e.sum()) / alpha)

def grad_smooth(x, n, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    if topk is not None and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        thresh = c[idx].min()
        c_eff = np.where(c >= thresh, c, c.max() - 10.0 / alpha)
    else:
        c_eff = c
    m = float(c_eff.max())
    z = alpha * (c_eff - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            ei = -d
            g[:n - ei] += w * (1.0 - x[ei:])
            g[ei:] -= w * x[:n - ei]
    return (2.0 / n) * g

def pen_obj(x, n, alpha, mu, topk):
    v = obj_smooth(x, n, alpha, topk)
    dev = x.sum() - 0.5 * n
    return v + 0.5 * mu * dev * dev

def pen_grad(x, n, alpha, mu, topk):
    g = grad_smooth(x, n, alpha, topk)
    dev = x.sum() - 0.5 * n
    return g + mu * dev

def slsqp_phase(h, n, bounds, cons, plans, deadline):
