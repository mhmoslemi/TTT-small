def multiplicative_weights(h, n, iterations, deadline):
    """Multiplicative weights approach to the zero-sum game."""
    n_full = 2 * n - 1
    q = np.ones(n_full) / n_full  # uniform distribution over lags
    T = 0.1  # temperature
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    dx = 2.0 / n
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        
        # Step a: Minimize sum_m q[m] * corr[m] over h
        # Use projected gradient descent with a few iterations
        for inner in range(20):
            grad = game_gradient_fast(h, q, n)
            c = max_corr(h, n)
            # Scale gradient: the actual objective is sum q[m] * corr[m] * dx
            # but dx is constant so gradient direction is the same
            h_new = project_feasible(h - 0.01 * grad, n)
            # Check if improved
            obj_new = np.dot(q, max_corr(h_new, n))
            obj_old = np.dot(q, max_corr(h, n))
            if obj_new < obj_old:
                h = h_new
            else:
                # Try smaller step
                h_new = project_feasible(h - 0.005 * grad, n)
                obj_new = np.dot(q, max_corr(h_new, n))
                if obj_new < obj_old:
                    h = h_new
                else:
                    break
        
        # Evaluate
        c = max_corr(h, n)
        c_val = float(c.max()) * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Step b: Update q

        c_max = c.max()
        log_q = (c - c_max) / T
        log_q -= log_q.max()
        q = np.exp(log_q)
        q /= q.sum()
        
        # Decrease temperature
        T = max(0.01, T * 0.99)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def max_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, q, n):
    n_full = 2 * n - 1
    grad = np.ones(n)
    for j in range(n):
        d_min = max(-j, -(n - 1))
        d_max = min(n - 1 - j, n - 1)
        if d_min <= d_max:
            ds = np.arange(d_min, d_max + 1)
            grad[j] -= np.dot(q[ds + n - 1], h[j + ds])
        d_min2 = max(j - (n - 1), -(n - 1))
        d_max2 = min(j, n - 1)
        if d_min2 <= d_max2:
            ds2 = np.arange(d_min2, d_max2 + 1)
            grad[j] -= np.dot(q[ds2 + n - 1], h[j - ds2])
    return grad

def fista_optimize(h, n, beta_init, beta_final, max_outer, deadline):
    n_full = 2 * n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    x_prev = h.copy()
    t_k = 1.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        beta = beta_init * (beta_final / beta_init) ** (outer / max(1, max_outer - 1))
        c = max_corr(x, n)
        c_max = c.max()
        log_q = (c - c_max) / beta
        log_q -= log_q.max()
        q = np.exp(log_q)
        q /= q.sum()
        n_inner = 8
        for inner in range(n_inner):
            if time.time() > deadline:
                break
            y = x + ((t_k - 1) / (t_k + 1)) * (x - x_prev)
            grad = compute_grad(y, q, n)
            step = 0.3 / (beta + 1.0)
            z = y - step * grad
            x_new = project_feasible(z, n)
            x_prev = x
            x = x_new
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            t_k = t_new
        c_val = compute_c5(x, n)
        if c_val < best_c:
            best_c = c_val
            best_h = x.copy()
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.empty(grid_size)
            for gi in range(grid_size):
                d = d_grid[gi]
                nv = c_aff + d * a_coeff
                if diag_local is not None:
                    nv[diag_local] -= d * d
                vals[gi] = max(float(nv.max()), unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                c_a = c_aff.copy()
                a_c = a_coeff.copy()
                try:
                    res = golden(lambda d: _obj_eval(d, c_a, a_c, diag_local, unaff_max), gl, gh, maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _obj_eval(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def resample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 600
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: FISTA with smooth approximation
    if time.time() < deadline - 300:
        h1, c1 = fista_optimize(best_h, n, 20.0, 200.0, 40, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        h2, c2 = coord_cd(best_h, n, 150, deadline, grid_size=80)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Resample to finer grid and optimize
    for n_try in [800, 1000, 1200]:
        if time.time() > deadline - 150:
            break
        h_int = resample_h(best_h, len(best_h), n_try)
        h_o, c_o = fista_optimize(h_int, n_try, 50.0, 500.0, 30, deadline)
        if time.time() < deadline - 100:
            h_o, c_o = coord_cd(h_o, n_try, 80, deadline, grid_size=60)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 4: Multi-restart perturbations
    n_cur = len(best_h)
    if time.time() < deadline - 100:
        for r in range(8):
            if time.time() > deadline - 60:
                break
            s = 0.01 + 0.03 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = fista_optimize(hp, n_cur, 100.0, 500.0, 20, deadline)
            if time.time() < deadline - 40:
                h_o, c_o = coord_cd(h_o, n_cur, 50, deadline, grid_size=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final refinement with very high precision CD
    if time.time() < deadline - 20:
        hf, cf = coord_cd(best_h, len(best_h), 300, deadline, grid_size=120)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Try 1-h symmetry
    n_cur = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
