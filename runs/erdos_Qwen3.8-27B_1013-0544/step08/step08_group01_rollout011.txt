def slsqp_refine(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        def smooth_obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            m = cc.max()
            shifted = alpha * (cc - m)
            mx = shifted.max()
            lse = mx + np.log(np.sum(np.exp(shifted - mx)))
            return (m + lse / alpha) * (2.0 / n)
        def sum_con(x):
            return x.sum() - n / 2.0
        def sum_con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-17})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        worst = np.argsort(c)[-topk:]
        grad = np.zeros(n)
        for k in worst:
            off = int(k) - (n - 1)
            for i in range(n):
                j = i + off
                if 0 <= j < n:
                    grad[i] += (1.0 - h[j]) - grad[i] * 0
                    grad[j] -= h[i]
        grad /= topk
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h, best_c, stall = h_new, c_new, 0
            lr = min(lr * 1.05, 0.15)
        else:
            lr *= 0.7
            stall += 1
            if stall > 40 or lr < 1e-11:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Pairwise descent on worst shifts
    if time.time() < deadline - 200:
        h1, c1 = pairwise_descent(best_h, n, max_passes=400, deadline=deadline - 200, topk=3)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Expand to more worst shifts
    if time.time() < deadline - 150:
        h2, c2 = pairwise_descent(best_h, n, max_passes=200, deadline=deadline - 150, topk=5)
        if c2 < best_c:
            best_h, best_c = h2, c1 if c1 < best_c else c2

    # Phase 3: Random pair exploration to escape local minima
    if time.time() < deadline - 120:
        h3, c3 = random_pair_descent(best_h, n, max_pairs=3000, deadline=deadline - 120)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Another pairwise round after randomization
    if time.time() < deadline - 80:
        h4, c4 = pairwise_descent(best_h, n, max_passes=200, deadline=deadline - 80, topk=4)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Targeted gradient
    if time.time() < deadline - 60:
        h5, c5v = targeted_gradient(best_h, n, max_iter=300, deadline=deadline - 60, topk=5)
        if c5v < best_c:
            best_h, best_c = h5, c5v

    # Phase 6: SLSQP with high alpha
    if time.time() < deadline - 20:
        h6, c6 = slsqp_refine(best_h, n, [2000, 10000, 50000], maxiter=400)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final pairwise polish
    if time.time() < deadline:
        h7, c7 = pairwise_descent(best_h, n, max_passes=200, deadline=deadline, topk=2)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
