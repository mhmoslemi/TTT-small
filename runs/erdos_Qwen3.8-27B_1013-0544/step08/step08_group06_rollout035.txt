def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-16: break

            tot = h.sum()
            if tot < 1e-16: break

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp(h, n, alphas, maxiter=400):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, a), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff: unaff = tmp
            if m_hi < 2 * n - 2:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff: unaff = tmp
            diag = np.where(ms == (n - 1))[0]
            has_diag = len(diag) > 0
            dg = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + dg[:, None] * a[None, :]
            if has_diag:
                nv[:, diag] -= dg[:, None] ** 2
            vals = np.maximum(unaff, nv.max(axis=1))
            bi = int(np.argmin(vals))
            bd = float(dg[bi])
            bv = float(vals[bi])
            if 0 < bi < grid_size - 1:
                y = vals[bi - 1:bi + 2]
                x = dg[bi - 1:bi + 2]
                den = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(den) > 1e-18:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])) / den
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag] -= xv ** 2
                        v1 = max(unaff, float(nv1.max()))
                        if v1 < bv - 1e-15:
                            bv, bd = v1, xv
            if abs(bd) > 1e-10 and bv < c_max - 1e-13:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c_aff + bd * a
                if has_diag:
                    c[ms[diag]] -= bd ** 2
                nm = max(unaff, float(c[ms].max()))
                if nm < c_max: c_max = nm
                improved = True
        if not improved: break
        if (p + 1) % 300 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(c.max())
        if (p + 1) % 100 == 0:
            if abs(prev_c_max - c_max) < 1e-12:
                stall += 1
                if stall > 3: break
            else:
                stall = 0
            prev_c_max = c_max
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def joint_cd(h, n, top_k=20, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    scores = np.zeros(n)
    for idx in range(n):
        h0 = h[idx]
        if h0 < 1e-8 or h0 > 1 - 1e-8: continue
        m_lo = max(0, min(n-1-idx, idx))
        m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
        ms = np.arange(m_lo, m_hi+1)
        a = np.zeros(len(ms))
        jh = idx + ms - (n-1)
        vh = (jh>=0)&(jh<n)
        a[vh] += (1.0 - h[jh[vh]])
        jg = idx - ms + (n-1)
        vg = (jg>=0)&(jg<n)
        a[vg] -= h[jg[vg]]
        for step in [0.005, -0.005]:
            if h0+step < 0 or h0+step > 1: continue
            cn = c[ms].copy() + step * a
            dm = ms == (n-1)
            if np.any(dm): cn[dm] -= step*step
            unaff = 0.0
            if m_lo > 0: unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2: unaff = max(unaff, float(c[m_hi+1:].max()))
            scores[idx] = max(scores[idx], cmax - max(unaff, float(cn.max())))
    top_idx = np.argsort(-scores)[:top_k]
    if len(top_idx) < 3: return h, compute_c5(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    fixed_sum = h.sum() - h[top_idx].sum()
    def sub_obj(sv):
        ht = h.copy()
        ht[top_idx] = np.clip(sv, 0, 1)
        return lse_obj(ht, n, 20000)
    def sub_con(sv):
        return sv.sum() + fixed_sum - n/2.0
    def sub_jac(sv):
        return np.ones(len(sv))
    x0 = h[top_idx].copy()
    bounds_s = [(max(0, h[i]-0.1), min(1, h[i]+0.1)) for i in top_idx]
    try:
        res = minimize(sub_obj, x0, method='SLSQP', bounds=bounds_s,
                      constraints=[{'type':'eq','fun':sub_con,'jac':sub_jac}],
                      options={'maxiter':300,'ftol':1e-14})
        h_new = h.copy()
        h_new[top_idx] = np.clip(res.x, 0, 1)
        h_new = project_feasible(h_new, n)
        cn = compute_c5(h_new, n)
        if cn < best_c: best_h, best_c = h_new, cn
    except Exception: pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    if time.time() < deadline - 350:
        h1, c1 = slsqp(h, n, [5000, 50000], maxiter=300)
        if c1 < best_c: best_h, best_c = h1, c1
    h2, c2 = coord_cd(best_h, best_h__n if hasattr(best_h,'__n__') else n, max_passes=2500, deadline=deadline-300, grid_size=200)
    if c2 < best_c: best_h, best_c = h2, c2
    if time.time() < deadline - 250:
        h3, c3 = joint_cd(best_h, len(best_h), top_k=20, deadline=deadline-250)
        if c3 < best_c: best_h, best_c = h3, c3
        h3, c3 = coord_cd(best_h, len(best_h), max_passes=1500, deadline=deadline-250, grid_size=200)
        if c3 < best_c: best_h, best_c = h3, c3
    for attempt in range(4):
        if time.time() > deadline - 200: break
        noise_scale = 0.03 * (1 - attempt * 0.2)
        hp = best_h + rng.randn(len(best_h)) * noise_scale
        hp = project_feasible(hp, len(best_h))
        hp, cp = slsqp(hp, len(best_h), [50000], maxiter=200)
        hp, cp = coord_cd(hp, len(best_h), max_passes=1000, deadline=deadline-200, grid_size=180)
        if cp < best_c:
            best_h, best_c = hp, cp
    for n_try in [600, 400]:
        if time.time() > deadline - 100: break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp(h_int, n_try, [10000], maxiter=200)
        h_opt, c_opt = coord_cd(h_opt, n_try, max_passes=1000, deadline=deadline-100, grid_size=180)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    if time.time() < deadline - 5:
        h5, c5 = coord_cd(best_h, len(best_h), max_passes=8000, deadline=deadline, grid_size=250, skip_eps=1e-5)
        if c5 < best_c: best_h, best_c = h5, c5
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
