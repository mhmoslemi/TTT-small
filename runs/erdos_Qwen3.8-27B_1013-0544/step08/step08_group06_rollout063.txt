            if tot < 1e-15:

            if m_hi < nc - 1:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            # Adaptive grid: coarse then fine
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:

def slsqp_round(h, n, alphas, maxiter=800, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / alpha) * (2.0 / n)
        def constraint(x):
            return x.sum() - n / 2.0
        def jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint, 'jac': jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_step(h, n, step_size, deadline=None):
    """Move h in the direction that reduces the worst correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    # Find active set (correlations near the max)
    threshold = c_max - 0.05 * c_max  # within 5% of max
    active = np.where(c >= threshold)[0]
    # Compute "subgradient": for each active c_k, compute dc_k/dh
    # dc_k/dh_j = (1-h[j-k]) * I[valid] - h[j+k] * I[valid] where k is the shift
    n_pts = n
    grad = np.zeros(n_pts)
    for k_idx in active:
        shift = k_idx - (n_pts - 1)  # physical shift in grid units
        # c[k_idx] = sum_i h[i] * (1-h[i-shift]) for valid i
        # dc/dh[j] = (1-h[j-shift])*I[0<=j-shift<n] - h[j+shift]*I[0<=j+shift<n]
        g = np.zeros(n_pts)
        for j in range(n_pts):
            js = j - shift
            if 0 <= js < n_pts:
                g[j] += (1.0 - h[js])
            jp = j + shift
            if 0 <= jp < n_pts:
                g[j] -= h[jp]
        grad += g
    # Normalize and take step
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return h
    h_new = h - step_size * grad / grad_norm
    h_new = project_feasible(h_new, n_pts)
    return h_new

- If k* = argmax c_k, then ∇f = c_k*'s gradient = ∂c[k*]/∂h

But ∇f points in the direction that INCREASES c[k*]. So -∇f decreases c[k*].

h_new = h - step * (1/|S|) Σ_{k∈S} ∂c[k]/∂h

where S is the active set (k values where c[k] is close to the max).

This pushes h in a direction that decreases all active correlations simultaneously.

Let me integrate this into the algorithm. The step size is important: too large and we might increase non-active correlations; too small and we make no progress.

A good strategy: use backtracking line search. Try step = s, check if the max correlation decreased. If not, halve the step.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, c, step_size, active_frac=0.1):
    """Take a subgradient step to decrease the max correlation."""
    c_max = c.max()
    threshold = c_max - active_frac * c_max
    active_k = np.where(c >= threshold)[0]
    if len(active_k) == 0:
        return h
    shifts = active_k - (n - 1)
    # Compute subgradient
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            sp = -s
            grad[:n - sp] += (1.0 - h[sp:])
            grad[sp:] -= h[:n - sp]
    # Backtracking line search
    h_new = h
    step = step_size
    current_max = c_max
    for _ in range(20):
        h_try = project_feasible(h - step * grad, n)
        c_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        if c_try.max() < current_max - 1e-12:
            h_new = h_try
            break
        step *= 0.5
        if step < 1e-10:
            break
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descend(h, n, max_passes=5000, deadline=None, grid_size=80):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    nc = 2 * n - 1
    no_imp = 0
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-14:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(nc - 1, max(nc - 1 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            um = 0.0
            if m_lo > 0:
                um = max(um, float(c[:m_lo].max()))
            if m_hi < nc - 1:
                um = max(um, float(c[m_hi + 1:].max()))
            dp = np.where(ms == (n - 1))[0]
            hd = len(dp) > 0
            dg = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + dg[:, None] * a[None, :]
            if hd:
                nv[:, dp] -= dg[:, None] ** 2
            vals = np.maximum(um, nv.max(axis=1))
            bi = int(np.argmin(vals))
            bd = float(dg[bi])
            bv = float(vals[bi])
            if 0 < bi < grid_size - 1:
                df = np.linspace(dg[bi - 1], dg[bi + 1], 30)
                nv2 = c_aff[None, :] + df[:, None] * a[None, :]
                if hd:
                    nv2[:, dp] -= df[:, None] ** 2
                vals2 = np.maximum(um, nv2.max(axis=1))
                bi2 = int(np.argmin(vals2))
                if vals2[bi2] < bv:
                    bd = float(df[bi2])
                    bv = float(vals2[bi2])
            if abs(bd) > 1e-11 and bv < cur_max - 1e-13:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c_aff + bd * a
                if hd:
                    c[ms[dp]] -= bd ** 2
                nm = max(um, float(c[ms].max()))
                if nm < cur_max:
                    cur_max = nm
                improved = True
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_step(h, n, step_frac=0.02, active_frac=0.08):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    if c_max < 1e-15:
        return h
    thr = c_max - active_frac * c_max
    active_k = np.where(c >= thr)[0]
    if len(active_k) == 0:
        return h
    shifts = active_k - (n - 1)
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            if n - s > 0:
                grad[:n - s] -= h[s:]
        else:
            sp = -s
            grad[:n - sp] += (1.0 - h[sp:])
            if sp < n:
                grad[sp:] -= h[:n - sp]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h
    step = step_frac / gnorm
    for _ in range(15):
        h_try = project_feasible(h - step * grad, n)
        c_try_max = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
        if c_try_max < c_max - 1e-12:
            return h_try
        step *= 0.5
        if step < 1e-14:
            break
    return h

def slsqp_round(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            return (m + mx + np.log(np.sum(np.exp(sh - mx)))) / alpha * (2.0 / n)
        def ceq(x):
            return x.sum() - n / 2.0
        def ceqj(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': ceq, 'jac': ceqj}],
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c
