import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def fast_coord_cd(h, n, deadline, max_passes=100, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            # Compute a_coeff vectorized
            a = np.zeros(2*n-1)
            k_lo1 = max(0, n-1-p)
            k_hi1 = min(2*n-2, 2*n-2-p)
            j1 = np.arange(k_lo1, k_hi1+1)
            idx1 = p + j1 - (n-1)
            a[j1] -= h[idx1]
            
            k_lo2 = max(0, n-1-p)
            k_hi2 = min(2*n-2, n-1+p)
            j2 = np.arange(k_lo2, k_hi2+1)
            idx2 = p - j2 + (n-1)
            a[j2] += (1.0 - h[idx2])
            
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            
            # Vectorized evaluation
            nv = c[None, :] + d_grid[:, None] * a[None, :]
            nv[:, n-1] -= d_grid**2
            vals = nv.max(axis=1)
            
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            
            # Refine with golden section if interior
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                # Define 1D objective
                def _obj1d(d):
                    v = (c + d * a).max()
                    v_center = c[n-1] + d * a[n-1] - d*d
                    return max(v, v_center)
                
                res = minimize_scalar(_obj1d, bounds=(gl, gh), method='bounded', options={'maxiter': 40})
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                # Incremental update
                c += best_d * a
                c[n-1] -= best_d * best_d
                # Slight numerical drift correction
                if _pass == 0:  # track
                    pass
                current_max = min(current_max, best_val)
        
        if not improved:
            break
        # Periodic recompute to correct drift
        if (_pass + 1) % 5 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_max = float(c.max())
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
