    c = np.correlate(h, 1.0 - h, mode='full')

            if tot < 1e-15:

def active_gradients(h, n, active_lags):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cutting_plane_step(h, n, n_active, deadline):
    """One cutting-plane LP step. Returns (h_new, f_values, active_lags, t_LP)."""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    f = c * dx
    T = float(f.max())
    
    # Active lags
    threshold = T * 0.85
    active_idx = np.where(f >= threshold)[0]
    if len(active_idx) > n_active:
        active_idx = np.argsort(f)[-n_active:]
    active_idx = np.sort(active_idx)
    
    # Gradients
    n_act = len(active_idx)
    grad = np.zeros((n_act, n))
    for idx, k in enumerate(active_idx):
        j = int(k) - (n - 1)
        if j == 0:
            grad[idx] = (1.0 - 2.0 * h) * dx
        else:
            p_lo1 = max(0, -j)
            p_hi1 = min(n, n - j)
            if p_hi1 > p_lo1:
                grad[idx, p_lo1:p_hi1] += (1.0 - h[p_lo1+j:p_hi1+j])
            p_lo2 = max(0, j)
            p_hi2 = min(n, n + j)
            if p_hi2 > p_lo2:
                grad[idx, p_lo2:p_hi2] += -h[p_lo2-j:p_hi2-j]
            grad[idx] *= dx
    
    f_active = f[active_idx]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    cons.append(f_active + grad @ (h_var - h) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-9)
    except Exception:
        return None, f, active_idx, T
    
    if h_var.value is None:
        return None, f, active_idx, T
    
    h_new = project_feasible(np.clip(h_var.value, 0, 1), n)
    return h_new, f, active_idx, T

def cutting_plane_opt(h0, n, max_iter=300, n_active=150, deadline=None):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    h = h0.copy()
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        f = c * (2.0 / n)
        T = float(f.max())
        
        if T < best_c:
            best_c = T
            best_h = h.copy()
        
        h_new, f, active_idx, T = cutting_plane_step(h, n, n_active, deadline)
        if h_new is None:
            break
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        if c_new <= T + 1e-6:
            h = h_new
        else:
            h = project_feasible(0.5 * (h + h_new), n)
    
    return best_h, best_c
