def coord_one(c,h,p,n):
    h0=h[p]
    dlo = -h0; dhi = 1.0-h0
    if dhi - dlo < 1e-15: return h0
    M = p if p > n-1-p else n-1-p

    start = mid - M
    end = mid + M
    idxs = np.arange(start,end+1)
    c_aff = c[idxs]

    a = np.zeros(2*M+1, dtype=float)

    if np.any(validA):
        ia = p - ells[validA]
        a[validA] += 1.0 - h[ia]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(u, n):
    S = 0.5 * float(n)
    a = np.asarray(u, dtype=float)
    y = a + (S - float(a.sum())) / float(n)
    if float(y.min()) >= -1e-10 and float(y.max()) <= 1.0 + 1e-10:
        y = np.clip(y, 0.0, 1.0)
        for _ in range(4):
            d = S - float(y.sum())
            if abs(d) < 1e-12:
                break
            f = (y > 1e-12) & (y < 1.0 - 1e-12)
            if not f.any():
                break
            y[f] += d / int(f.sum())
            y = np.clip(y, 0.0, 1.0)
        if abs(float(y.sum()) - S) <= 1e-8:
            return y
    lo = float(a.min()) - 1.0

def smooth_val(c, n, alpha):
    dx = 2.0 / n
    m = float(c.max())
    z = alpha * (c - m)
    z = np.clip(z, -700.0, 0.0)
    s = float(np.exp(z).sum())
    return (m + np.log(s) / alpha) * dx

def top_weights(c, alpha, K):
    L = int(len(c))
    K = int(min(max(1, int(K)), L))
    cm = float(c.max())
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    z = alpha * (c[idx] - cm)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros(K)
        w[0] = 1.0
    else:
        w = w / s
    return idx, w

def grad_idx(h, idx, w, n):
    g = np.zeros(n)
    mid = n - 1
    for i, wt in zip(idx, w):
        if wt <= 1e-15:
            continue
        ell = int(i) - mid
        if ell > 0:
            g[ell:] += wt * (1.0 - h[:-ell])
            g[:n - ell] -= wt * h[ell:]
        elif ell < 0:
            L = -ell
            g[:-L] += wt * (1.0 - h[L:])
            g[L:] -= wt * h[:-L]
        else:
            g += wt * (1.0 - 2.0 * h)
    return g
