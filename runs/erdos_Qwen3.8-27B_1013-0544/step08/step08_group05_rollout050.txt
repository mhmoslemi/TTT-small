Actually, for the upper envelope of lines y = m_i * x + b_i:
- Sort by slope m_i
- Use a gift-wrapping / convex hull approach
- The upper envelope is the sequence of lines that form the "top" of the arrangement

The standard algorithm:
1. Sort lines by slope (and by intercept for ties)
2. Maintain a stack of lines that form the upper envelope
3. For each new line, while the stack top is "dominated" (intersection with new line is to the left of intersection with stack second-top), pop the stack top
4. The resulting stack is the upper envelope from left to right

Then the minimum of the upper envelope over [0,1] is:
- Check the value at each "vertex" (intersection of consecutive lines on the envelope)
- Also check the values at x=0 and x=1
- Pick the minimum

This is O(n log n) per coordinate.

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    # Final correction
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def exact_coord_min(c_vals, A_coeff, old_val, n):
    """
    Find min_{a in [0,1]} max_s (c_vals[s] + (a - old_val) * A_coeff[s])
    using the upper envelope of lines.
    Returns the optimal a and the optimal value.
    """
    m = len(A_coeff)
    B = c_vals - old_val * A_coeff
    # f(a) = max_s (A_coeff[s] * a + B[s])
    # Find the upper envelope and its minimum over [0,1]
    
    # Sort by slope
    order = np.argsort(A_coeff)
    A_sorted = A_coeff[order]
    B_sorted = B[order]
    
    # Build upper envelope using stack
    # Each entry: (slope, intercept, x_start) where x_start is the x-coordinate
    # where this line becomes the maximum
    stack = []  # list of [slope, intercept, x_start]
    
    for i in range(m):
        s_i = A_sorted[i]
        b_i = B_sorted[i]
        new_start = -np.inf
        while stack:
            s_top, b_top, x_top = stack[-1]
            if s_i == s_top:
                # Keep the one with larger intercept
                if b_i > b_top:
                    new_start = x_top
                    stack.pop()
                else:
                    break
            else:
                # Intersection: s_i * x + b_i = s_top * x + b_top
                # x = (b_top - b_i) / (s_i - s_top)
                x_int = (b_top - b_i) / (s_i - s_top)
                if x_int <= x_top:
                    new_start = x_top
                    stack.pop()
                else:
                    new_start = x_int
                    break
        stack.append([s_i, b_i, new_start])
    
    # Find minimum of the upper envelope over [0,1]
    best_a = 0.0
    best_f = np.inf
    # Check x=0 and x=1
    for test_x in [0.0, 1.0]:
        # Find which line is active at test_x
        f_val = max(A_sorted[i] * test_x + B_sorted[i] for i in range(m))
        # Actually, use the stack
        active_f = -np.inf
        for s_i, b_i, x_start in stack:
            if test_x >= x_start:
                active_f = s_i * test_x + b_i
        if active_f < best_f:
            best_f = active_f
            best_a = test_x
    
    # Check vertices (intersections of consecutive lines on the envelope)
    for i in range(len(stack)):
        x_v = stack[i][2]  # x_start of this line
        if 0.0 <= x_v <= 1.0:
            f_v = stack[i][0] * x_v + stack[i][1]
            if f_v < best_f:
                best_f = f_v
                best_a = x_v
    
    return best_a, best_f
