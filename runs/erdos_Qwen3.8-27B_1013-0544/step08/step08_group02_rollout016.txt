def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def pair_search(h, n, max_tries, grid, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    accepts, no_improve, max_accept = 0, 0, min(300, max(80, max_tries // 3))
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.008, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 9 == 8 or active.size <= 1:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 1:
                    continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 1:
                    continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        hi, hj = h[i], h[j]
        S = hi + hj
        lo_b, hi_b = max(0.0, S - 1.0), min(1.0, S)
        if hi_b - lo_b <= 1e-14:
            no_improve += 1
            continue
        B = A[i] - A[j]
        ts = np.linspace(lo_b, hi_b, grid)
        u = ts - hi
        base = c[None, :] + u[:, None] * B[None, :]
        base[:, off] -= 2.0 * (u ** 2)
        if 0 <= off + i - j < 2 * n - 1:
            base[:, off + i - j] += u ** 2
        if 0 <= off + j - i < 2 * n - 1:
            base[:, off + j - i] += u ** 2
        maxes = base.max(axis=1)
        idx = int(np.argmin(maxes))
        if maxes[idx] < cur_max - 1e-10:
            u_best = u[idx]
            h[i] += u_best; h[j] -= u_best
            u2 = u_best * u_best
            c = c + u_best * B
            c[off] -= 2.0 * u2
            if 0 <= off + i - j < 2 * n - 1:
                c[off + i - j] += u2
            if 0 <= off + j - i < 2 * n - 1:
                c[off + j - i] += u2
            accepts += 1
            no_improve = 0
            if accepts >= max_accept:
                break
            A = compute_A(h, n)
            if accepts % 40 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
        else:
            no_improve += 1
            if no_improve > max(400, max_tries // 2):
                break
    return h, compute_c5(h, n)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    for t in ['rev', 'inv', 'ri']:
        if t == 'rev':
            hs = project_feasible(0.5 * (h + h[::-1]), n)
        elif t == 'inv':
            hs = project_feasible(1.0 - h, n)
        else:
            hs = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
        cs = compute_c5(hs, n)
        if cs < c - 1e-12:
            h, c = hs, cs
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if not quick and time.time() < deadline - 15.0:
        plans = [(50, 40), (300, 35), (1500, 30), (8000, 25)]
    elif quick:
        plans = [(30, 25), (500, 20)]
    else:
        plans = []
    if plans:
        h, c = slsqp_optimize(h, n, bounds, cons, plans, deadline)
    if n <= 1000 and time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mt = int(min(12000, max(800, rem * 35)))
        h, c = pair_search(h, n, mt, 80, deadline, rng)
    return h, compute_c5(project_feasible(h, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n0 = 500
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        h0 = project_feasible(0.5 + 0.1 * np.sin(np.pi * x) + 0.05 * np.cos(2 * np.pi * x), n0)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    main_dl = hard - max(120.0, 0.22 * float(budget_s))
    if main_dl < t0 + 30.0:
        main_dl = hard
    best_h, best_c = optimize_instance(best_h, n0, main_dl, rng, False)
    n = len(best_h)
    for att in range(8):
        if time.time() > hard - 30.0:
            break
        s = 0.015 * (att + 1) + 0.01
        hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
        dl = min(hard - 10.0, time.time() + 55.0)
        if dl < time.time() + 12.0:
            break
        ht, ct = optimize_instance(hp, n, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    if time.time() < hard - 10.0:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        if time.time() < hard - 20.0:
            ht, ct = slsqp_optimize(best_h, n, bounds, cons, [(10000, 20), (30000, 15)], hard - 5.0)
            if ct < best_c:
                best_h, best_c = ht, ct
        if n <= 1000 and time.time() < hard - 3.0:
            rem = hard - time.time()
            mt = int(min(15000, max(2000, rem * 30)))
            best_h, best_c = pair_search(best_h, n, mt, 128, hard, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
