            h = project_01_sum(h, n / 2.0)
        else:
            break
    return h, compute_c5(h, n)

def grad_polish(h, n, deadline, max_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    mu = 0.9
    dx = 2.0 / n
    k = 8
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(c, -k)[-k:]
        grad = np.zeros(n)
        for m in top_idx:
            d = m - (n - 1)
            if d >= 0:
                grad[:n - d] += (1.0 - h[d:n])
                grad[d:] -= h[:n - d]
            else:
                dd = -d
                grad[dd:] += (1.0 - h[:n - dd])
                grad[:n - dd] -= h[dd:]
        grad *= dx / k
        lr = 0.03 / (1.0 + 0.05 * np.sqrt(it))
        h_new = project_01_sum(h + mu * v - lr * grad, n / 2.0)
        v = h_new - h
        h = h_new
        if it % 50 == 0:
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    c_new = compute_c5(h, n)
    if c_new < best_c:
        best_c = c_new
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(500, 0.5)
    n = len(h_init)
    best_h = project_01_sum(h_init, n / 2.0)
    best_c = compute_c5(best_h, n)
    # Phase 1: Heavy coordinate descent
    h1, c1 = coord_descent(best_h, n, deadline - 200)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Gradient polishing to escape flat regions
    if time.time() < deadline - 80:
        h2, c2 = grad_polish(best_h, n, deadline - 80)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: More CD after gradient steps
    if time.time() < deadline - 30:
        h3, c3 = coord_descent(best_h, n, deadline - 10)
        if c3 < best_c:
            best_h, best_c = h3, c5 if False else c3
            best_h, best_c = h3, c3
    # Phase 4: Try different resolutions
    if time.time() < deadline - 40:
        for n_try in [n + 100, n - 100, n + 200, n - 200]:
            if time.time() > deadline - 15:
                break
            if n_try < 100 or n_try == n:
                continue
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new_pt = np.linspace(0, 2, n_try, endpoint=False)
            h_int = project_01_sum(np.interp(x_new_pt, x_old, best_h), n_try / 2.0)
            h_int = project_01_sum(h_int + rng.normal(0, 0.005, n_try), n_try / 2.0)
            h_opt, c_opt = coord_descent(h_int, n_try, deadline - 15)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
    best_h = project_01_sum(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
