import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=600):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=10000, deadline=None, grid_size=200):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    flen = 2 * n - 1
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if p % 4 == 0:
            idx_order = range(n)
        elif p % 4 == 1:
            idx_order = range(n - 1, -1, -1)
        else:
            impacts = np.zeros(n)
            for idx in range(n):
                m_lo = max(0, min(n - 1 - idx, idx))
                m_hi = min(flen - 1, max(flen - 1 - idx, idx + n - 1))
                impacts[idx] = c[m_lo:m_hi + 1].max()
            idx_order = np.argsort(-impacts)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-12:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(flen - 1, max(flen - 1 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                tmp = float(c[:m_lo].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            if m_hi < flen - 1:
                tmp = float(c[m_hi + 1:].max())
                if tmp > unaff_max:
                    unaff_max = tmp
            dm = (ms == (n - 1))
            has_diag = np.any(dm)
            dpos = np.where(dm)[0]
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, dpos] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = float(d_grid[best_i])
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y0, y1, y2 = vals[best_i - 1], vals[best_i], vals[best_i + 1]
                x0, x1, x2 = d_grid[best_i - 1], d_grid[best_i], d_grid[best_i + 1]
                denom = (y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1))
                if abs(denom) > 1e-15:
                    xv = 0.5 * (y0 * (x1 - x2) + y1 * (x2 - x0) + y2 * (x0 - x1)) / denom
                    if x0 < xv < x2:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[dpos] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val = v1
                            best_d = float(xv)
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-13:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[dpos]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 150 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb_optimize(h, n, rng, scale, deadline, grid_size=150, max_passes=5000):
    h2 = h + rng.randn(n) * scale
    h2 = project_feasible(h2, n)
    return coord_cd(h2, n, max_passes=max_passes, deadline=deadline, grid_size=grid_size)
