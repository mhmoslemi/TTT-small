def project_feasible(h, n):

            r = -s
            grad[r:] += wk * (1.0 - h[:n-r])
            grad[:n-r] -= wk * h[r:]
    return grad, w, c

def adam_optimize(h0, n, alpha, steps, lr, deadline, restart_every=30):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.85, 0.999
    stall = 0
    for step in range(steps):
        if time.time() > deadline - 5:
            break
        t += 1
        if t % restart_every == 0:
            m *= 0.5
            v *= 0.5
        grad, w, c = compute_grad_smooth(h, n, alpha)
        grad -= grad.mean()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h_new = project_feasible(h_new, n)
        c_new = compute_c(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > restart_every * 2:
                h = best_h.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                break
        h = h_new
    return best_h, best_c

def coord_descent(h, n, passes, deadline, pe=10):
    h = project_feasible(h, n).copy()
    cvals = np.correlate(h, 1.0 - h, mode='full')
    cur = float(cvals.max())
    best_h = h.copy()
    best_c = (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(passes):
        if time.time() > deadline - 5:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline - 5:
                break
            hp = h[p]
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo - 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n) & (np.arange(lo, hi+1) != center)
            vi = (i >= 0) & (i < n) & (np.arange(lo, hi+1) != center)
            for mi in range(seg):
                k = lo + mi
                if k == center:
                    continue
                sm = center - k
                if sm > 0:
                    jj = p + sm
                    ii = p - sm
                    if 0 <= jj < n:
                        a[mi] += 1.0 - h[jj]
                    if 0 <= ii < n:
                        a[mi] -= h[ii]
                elif sm < 0:
                    rr = -sm
                    jj = p - rr
                    ii = p + rr
                    if 0 <= jj < n:
                        a[mi] += 1.0 - h[jj]
                    if 0 <= ii < n:
                        a[mi] -= h[ii]
            off = center - lo
            cseg = np.empty(seg, dtype=float)
            for mi in range(seg):
                k = lo + mi
                if k != center:
                    cseg[mi] = cvals[k]
            qconst = cvals[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(cvals[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(cvals[hi+1:].max()))
            # Binary search for optimal x
            lo_x = 0.0
            hi_x = 1.0
            for _bis in range(40):
                mid_x = 0.5 * (lo_x + hi_x)
                feasible = mid_x >= unaff
                if feasible:
                    feasible = True
                    for mi in range(seg):
                        val = cseg[mi] + a[mi] * (mid_x - hp)
                        if val > mid_x + 1e-15:
                            feasible = False
                            break
                qval = qconst + mid_x * (1.0 - mid_x)
                if qval > mid_x + 1e-15:
                    feasible = False
                if feasible:
                    hi_x = mid_x
                else:
                    lo_x = mid_x
            x_opt = hi_x
            f_new = 0.0
            for mi in range(seg):
                f_new = max(f_new, cseg[mi] + a[mi] * (x_opt - hp))
            f_new = max(f_new, qconst + x_opt * (1.0 - x_opt), unaff)
            if f_new < cur - 1e-12:
                dxv = x_opt - hp
                for mi in range(seg):
                    k = lo + mi
                    if k != center:
                        cvals[k] += a[mi] * dxv
                cvals[center] = qconst + x_opt * (1.0 - x_opt)
                h[p] = x_opt
                cur = f_new
                improved = True
            if (p + 1) % pe == 0:
                h = project_feasible(h, n)
                cvals = np.correlate(h, 1.0 - h, mode='full')
                cur = float(cvals.max())
        h = project_feasible(h, n)
        cvals = np.correlate(h, 1.0 - h, mode='full')
        cur = float(cvals.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-13:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def make_initial(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.3 * np.cos(np.pi * (x - 1.0)) + rng.uniform(-0.1, 0.1, n)
    return project_feasible(h, n)

def active_set_refine(h, n, deadline, eps=0.02, max_iters=50):
    h = project_feasible(h, n).copy()
    best_c = compute_c(h, n)
    best_h = h.copy()
    for _ in range(max_iters):
        if time.time() > deadline - 10:
            break
        c = correlations(h, n)
        c_max = c.max()
        active = np.where(c >= c_max - eps)[0]
        center = n - 1
        grad_sum = np.zeros(n)
        for k in active:
            s = center - k
            if s == 0:
                grad_sum += (1.0 - 2.0 * h)
            elif s > 0:
                grad_sum[:n-s] += (1.0 - h[s:])
                grad_sum[s:] -= h[:n-s]
            else:
                r = -s
                grad_sum[r:] += (1.0 - h[:n-r])
                grad_sum[:n-r] -= h[r:]
        grad_sum /= len(active)
        grad_sum -= grad_sum.mean()
        # Line search
        for lam in [0.5, 0.3, 0.1, 0.05, 0.02]:
            h_try = project_feasible(h + lam * grad_sum, n)
            c_try = compute_c(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                break
        if best_c < c_max - 1e-6:
            eps = max(0.005, eps * 0.85)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = make_initial(500, rng)
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c(h, n)

    # Phase 1: Adam smooth-max with increasing alpha
    for alpha in [20, 50, 100, 300, 1000, 3000]:
        if time.time() > deadline - 200:
            break
        steps = 80 if alpha < 500 else 50
        lr = 0.04 if alpha < 100 else 0.02
        h, c = adam_optimize(h, n, alpha, steps, lr, deadline)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Coordinate descent
    pe = 8 if n <= 600 else 12
    h, c = coord_descent(h, n, 8, deadline, pe)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Active set refine
    h2, c2 = active_set_refine(best_h, n, deadline, eps=0.03, max_iters=40)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 4: Multi-resolution
    for n_try in [400, 640, 800, 320, 960]:
        if time.time() > deadline - 180:
            break
        if best_c <= 0.38072:
            break
        h_r = resample_h(best_h, n_try)
        for alpha_r in [50, 500]:
            if time.time() > deadline - 40:
                break
            h_r, c_r = adam_optimize(h_r, n_try, alpha_r, 60, 0.03, deadline)
        h_r, c_r = coord_descent(h_r, n_try, 5, deadline, 10)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()

    # Phase 5: Random restarts with different structures
    for r in range(30):
        if time.time() > deadline - 60:
            break
        if best_c <= 0.38070:
            break
        n_r = [400, 500, 600, 320][r % 4]
        typ = r % 5
        if typ == 0:
            h_r = project_feasible(best_h[:n_r].copy() + rng.normal(0, 0.03, n_r) if n_r == len(best_h) else resample_h(best_h, n_r) + rng.normal(0, 0.03, n_r), n_r)
        elif typ == 1:
            base = resample_h(best_h, n_r)
            h_r = project_feasible(base + 0.05 * rng.randn(n_r), n_r)
        elif typ == 2:
            base = resample_h(best_h, n_r)
            k = n_r // 2
            idx = np.argpartition(base, n_r - k)[n_r - k:]
            h_r = np.zeros(n_r)
            h_r[idx] = 1.0
        elif typ == 3:
            h_r = make_initial(n_r, rng)
        else:
            base = resample_h(best_h, n_r)
            h_r = project_feasible(np.clip(base + 0.15 * np.sin(np.pi * np.arange(n_r) / n_r), 0, 1), n_r)
        h_r, c_r = adam_optimize(h_r, n_r, 200, 60, 0.03, deadline)
        h_r, c_r = coord_descent(h_r, n_r, 4, deadline, 12)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()

    # Final polish
    if time.time() < deadline - 20:
        h_f, c_f = coord_descent(best_h, len(best_h), 6, deadline, 5)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
        h_f, c_f = active_set_refine(best_h, len(best_h), deadline, eps=0.01, max_iters=30)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
