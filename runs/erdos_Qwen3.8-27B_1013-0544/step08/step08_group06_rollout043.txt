def precompute_info(n):
  starts=np.empty(n,dtype=int); lens=np.empty(n,dtype=int)
  left_pos=[]; left_idx=[]; right_pos=[]; right_idx=[]
  for p in range(n):
    M=max(p,n-1-p); l0=n-1-M; L=2*M+1
    starts[p]=l0; lens[p]=L
    s=np.arange(-M,M+1)
    left=p-s; right=p+s
    lv=(left>=0)&(left<n); rv=(right>=0)&(right<n)
    posl=np.nonzero(lv)[0]; lind=left[lv]
    posr=np.nonzero(rv)[0]; rind=right[rv]
    left_pos.append(posl.astype(np.intp)); left_idx.append(lind.astype(np.intp))
    right_pos.append(posr.astype(np.intp)); right_idx.append(rind.astype(np.intp))
  return starts,lens,left_pos,left_idx,right_pos,right_idx

def compute_c_sum(h):

def project_feasible(h, n):
    s_target = 0.5 * float(n)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(h.sum() - s_target) < 1e-14:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(72):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if val < s_target:
            lo = mid
        else:
            hi = mid
    ht = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = s_target - ht.sum()
    if diff > 1e-15:
        idxs = np.where(ht < 1.0)[0]
        for i in idxs:
            if diff <= 1e-15:
                break
            room = 1.0 - ht[i]
            add = room if room < diff else diff
            if add > 0.0:
                ht[i] += add
                diff -= add
    elif diff < -1e-15:
        idxs = np.where(ht > 0.0)[0]
        for i in idxs:
            if diff >= -1e-15:
                break
            room = ht[i]
            add = -room if room < -diff else -diff
            if add < 0.0:
                ht[i] += add
                diff -= add
    return ht


def detect_flip(n=10):
    rng = np.random.RandomState(12345)
    h = rng.rand(n)
    c_corr = np.correlate(h, 1.0 - h, mode="full")
    c_conv = np.convolve(h, 1.0 - h[::-1], mode="full")
    if np.allclose(c_corr, c_conv, rtol=1e-11, atol=1e-11):
        return False
    if np.allclose(c_corr, c_conv[::-1], rtol=1e-11, atol=1e-11):
        return True
    return False

def sum_con(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(len(x), dtype=float)


def active_fun(x, top_idx, w, flip=False):
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(np.dot(c[top_idx], w))


def active_jac(x, top_idx, w, flip=False):
    n = len(x)
    center = n - 1
    c1 = np.zeros(n, dtype=float)
    d1 = np.zeros(n, dtype=float)
    d2 = np.zeros(n, dtype=float)
    for s, wt in zip(top_idx - center, w):
        s = int(s)
        if flip:
            s = -s
        if abs(wt) <= 1e-30:
            continue
        js = -s if -s > 0 else 0
        je = n - s
        if je > n:
            je = n
        if js < je:
            c1[js + s:je + s] += wt
            d1[js + s:je + s] += wt * x[js:je]
        js = s if s > 0 else 0
        je = n + s
        if je > n:
            je = n
        if js < je:
            d2[js - s:je - s] += wt * x[js:je]
    return c1 - d1 - d2

            w = e / e.sum()
        else:
            w = np.full(K, 1.0 / float(K))

        try:
            res = minimize(
                active_fun,
                h,
                args=(top_idx, w, flip),
                jac=active_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": mi, "ftol": 1e-15, "disp": False},
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                nh = project_feasible(res.x, n)
                nc = compute_c_sum(nh)
                if nc < best_C - 1e-13:
                    best_h = nh.copy()
                    best_C = nc
                    h = nh.copy()
        except Exception:
            pass
    return best_h, best_C


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + max(50.0, budget) - 20.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        x0 = np.linspace(0.0, 2.0, n0)
        h0 = np.full(n0, 0.5)
        best_c = compute_c_sum(h0)
        for amp in (0.20, 0.30, 0.40):
            for freq in (1, 2, 3):
                for ph in (0.0, 0.5):
                    hc = np.clip(0.5 + amp * np.cos(math.pi * freq * x0 + ph), 0.0, 1.0)
                    hc = project_feasible(hc, n0)
                    cc = compute_c_sum(hc)
                    if cc < best_c:
                        h0 = hc.copy()
                        best_c = cc

    h = project_feasible(h0, n0)
    best_n = int(n0)
    best_Csum = compute_c_sum(h)
    best_C5 = best_Csum * (2.0 / float(best_n))
    best_h = h.copy()

    flip = detect_flip()

    hs = project_feasible(0.5 * (h + h[::-1]), best_n)
    cs = compute_c_sum(hs)
    c5s = cs * (2.0 / float(best_n))
    if c5s < best_C5:
        best_h = hs.copy()
        best_Csum = cs
        best_C5 = c5s

    d1 = t0 + 0.10 * budget
    if time.time() < d1:
        hh, cc = active_slsqp(
            best_h, best_n, flip, d1,
            [(12, 70, False), (24, 70, False), (60, 55, True)],
        )
        c5 = cc * (2.0 / float(best_n))
        if c5 < best_C5 - 1e-12:
            best_h, best_Csum, best_C5 = hh, cc, c5

    d2 = t0 + 0.36 * budget
    if time.time() < d2:
        info = precompute_info(best_n)
        hh, cc = coord_cd_binary(
            best_h, best_n, info, flip,
            max_passes=90, deadline=d2, max_step=0.05,
        )
        c5 = cc * (2.0 / float(best_n))
        if c5 < best_C5 - 1e-12:
            best_h, best_Csum, best_C5 = hh, cc, c5

    d3 = t0 + 0.58 * budget
    if time.time() < d3:
        hh, cc = active_slsqp(
            best_h, best_n, flip, d3,
            [(24, 60, False), (60, 50, True), (100, 40, True)],
        )
        c5 = cc * (2.0 / float(best_n))
        if c5 < best_C5 - 1e-12:
            best_h, best_Csum, best_C5 = hh, cc, c5

    d4 = t0 + 0.82 * budget
    if time.time() < d4:
        info = precompute_info(best_n)
        hh, cc = coord_cd_binary(
            best_h, best_n, info, flip,
            max_passes=140, deadline=d4, max_step=0.01,
        )
        c5 = cc * (2.0 / float(best_n))
        if c5 < best_C5 - 1e-12:
            best_h, best_Csum, best_C5 = hh, cc, c5

    n_list = [499, 450, 400, 350, 300]
    if time.time() < d4 - 20.0:
        for nt in n_list:
            if time.time() > d4 - 30.0:
                break
            if int(nt) == best_n:
                continue
            nt = int(nt)
            sub_dead = min(d4 - 10.0, time.time() + 130.0)
            x_old = np.linspace(0.0, 2.0, best_n)
            x_new = np.linspace(0.0, 2.0, nt)
            ht = np.interp(x_new, x_old, best_h)
            ht = project_feasible(ht, nt)
            info_t = precompute_info(nt)
            hh, cc = coord_cd_binary(
                ht, nt, info_t, flip,
                max_passes=70, deadline=sub_dead, max_step=0.006,
            )
            c5 = cc * (2.0 / float(nt))
            if c5 < best_C5 - 1e-12:
                best_h, best_n, best_Csum, best_C5 = hh, nt, cc, c5

    final_dead = deadline - 5.0
    if time.time() < final_dead:
        info = precompute_info(best_n)
        hh, cc = coord_cd_binary(
            best_h, best_n, info, flip,
            max_passes=800, deadline=final_dead, max_step=0.0025,
        )
        c5 = cc * (2.0 / float(best_n))
        if c5 < best_C5 - 1e-12:
            best_h, best_Csum, best_C5 = hh, cc, c5

    best_h = project_feasible(best_h, best_n)
    best_Csum = compute_c_sum(best_h)
    best_C5 = best_Csum * (2.0 / float(best_n))
    return best_h, float(best_C5), int(best_n)
