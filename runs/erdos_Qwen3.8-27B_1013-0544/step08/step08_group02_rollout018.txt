```python
def active_set_lp_polish(h, n, deadline, rng, M=5, max_rounds=50, eps=0.05):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)  # A[i,k] = ∂c_k/∂h[i]
    
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.005, 0.001 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size > M:
            # Keep top M
            active = active[np.argsort(c[active])[-M:]]
        
        # Build LP: minimize t
        # Variables: d[0..n-1], t (total n+1 vars)
        # Constraints:
        #   A[:, k] · d - t ≤ -c[k]  for k in active  (i.e., c[k] + A[:,k]·d - t ≤ 0)
        #   ∑ d = 0
        #   0 ≤ d + h ≤ 1  =>  -h ≤ d ≤ 1-h
        #   -eps ≤ d ≤ eps (limit step size)
        
        from scipy.optimize import linprog
        
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # t is the last variable
        
        # Inequality constraints: A_active · d - t ≤ -c[active]
        # i.e., [A_active, -1] · [d, t] ≤ -c[active]
        A_ineq = np.zeros((len(active), n + 1))
        for idx, k in enumerate(active):
            A_ineq[idx, :n] = A[:, k]
            A_ineq[idx, n] = -1.0
        b_ineq = -c[active]
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        
        # Bounds on d: max(-h, -eps) ≤ d ≤ min(1-h, eps)
        lower = np.maximum(-h, -eps)
        upper = np.minimum(1.0 - h, eps)
        bounds_lp = list(zip(lower, upper)) + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.status != 0:
            break
        
        d = res.x[:n]
        if np.abs(d).max() < 1e-12:
            break
        
        # Take the step
        h_new = np.clip(h + d, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if float(c_new.max()) < cur_max - 1e-10:
            h = h_new
            c = c_new
            A = compute_A(h, n)
        else:
            # Try half step
            h_half = project_feasible(h + 0.5 * d, n)
            c_half = np.correlate(h_half, 1.0 - h_half, mode='full')
            if float(c_half.max()) < cur_max - 1e-10:
                h = h_half
                c = c_half
                A = compute_A(h, n)
            else:
                break
    
    return h, compute_c5(h, n)
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def pair_line_search(h, c, A, i, j, n, cur_max):
    hi, hj = h[i], h[j]
    S = hi + hj
    lo = max(0.0, S - 1.0)
    hiT = min(1.0, S)
    if hiT - lo <= 1e-14:
        return None
    n_grid = 50
    us = np.linspace(lo - hi, hiT - hi, n_grid)
    B = A[i] - A[j]
    off = n - 1
    best_val = cur_max
    best_u = None
    for u in us:
        if hi + u < 0 or hi + u > 1:
            continue
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        v = float(cn.max())
        if v < best_val:
            best_val = v
            best_u = u
    if best_u is None or best_val >= cur_max - 1e-10:
        return None
    # Refine around best_u
    u_range = (us[1] - us[0]) * 2 if n_grid > 1 else 0.01
    us2 = np.linspace(best_u - u_range, best_u + u_range, 30)
    for u in us2:
        if hi + u < 0 or hi + u > 1:
            continue
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        v = float(cn.max())
        if v < best_val:
            best_val = v
            best_u = u
    return best_u

def targeted_pair_search(h, n, max_tries, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    accepts = 0
    no_improve = 0
    
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.01, 0.0005 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            while i == j:
                j = int(rng.randint(n))
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 1: continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 1: continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        
        u = pair_line_search(h, c, A, i, j, n, cur_max)
        if u is None:
            no_improve += 1
            continue
        old = float(h[i] + h[j])
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        u = t - float(h[i])
        if abs(u) < 1e-15:
            no_improve += 1
            continue
        B = A[i] - A[j]
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        nm = float(cn.max())
        if nm < cur_max - 1e-10:
            h[i] = t; h[j] = old - t; c = cn
            accepts += 1; no_improve = 0
            if accepts % 10 == 0:
                A = compute_A(h, n)
            if accepts % 50 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
            if accepts >= min(300, max_tries):
                break
        else:
            no_improve += 1
        if no_improve > max(200, max_tries // 3):
            break
    return h, compute_c5(h, n)

def lp_polish(h, n, deadline, eps=0.04, max_rounds=200):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    n_lags = 2 * n - 1
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.column_stack([A.T, -np.ones(n_lags)])
        b_ub = -c
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        lower = np.maximum(-h, -eps)
        upper = np.minimum(1.0 - h, eps)
        bounds_lp = list(zip(lower, upper)) + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        d = res.x[:n]
        if np.abs(d).max() < 1e-10:
            break
        
        improved = False
        for frac in [1.0, 0.5, 0.25]:
            h_new = project_feasible(h + frac * d, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            if float(c_new.max()) < cur_max - 1e-10:
                h = h_new; c = c_new
                A = compute_A(h, n)
                improved = True
                break
        if not improved:
            break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(np.pi * m * x + ph)
        h += 0.3 * a * np.cos(2.0 * np.pi * m * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c: h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c: h, c = hc, cs
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    
    if quick:
        plans = [(50, 30), (500, 30), (5000, 25)]
    else:
        plans = [(20, 50), (100, 50), (500, 40), (2500, 30), (10000, 25), (50000, 20)]
        if time.time() < deadline - 30.0:
            plans.append((200000, 15))
    h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    
    if n <= 1000 and time.time() < deadline - 5.0:
        rem = deadline - time.time()
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(60):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_components(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    return (m + float(np.log(s)) / alpha) * (2.0 / n), p, c

def grad_weights(x, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - x[e:])
            g[e:] -= w * x[:n - e]
    return (2.0 / n) * g

def smooth_val(x, n, alpha):
    return smooth_components(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = smooth_components(x, n, alpha)
    return grad_weights(x, n, p)

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def pair_line_search(h, c, A, i, j, n, cur_max):
    hi = h[i]
    S = hi + h[j]
    lo = max(0.0, S - 1.0)
    hiT = min(1.0, S)
    if hiT - lo <= 1e-14:
        return None
    us = np.linspace(lo - hi, hiT - hi, 50)
    B = A[i] - A[j]
    off = n - 1
    best_val = cur_max
    best_u = None
    for u in us:
        if hi + u < 0 or hi + u > 1:
            continue
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        v = float(cn.max())
        if v < best_val:
            best_val = v
            best_u = u
    if best_u is None or best_val >= cur_max - 1e-10:
        return None
    ur = (us[1] - us[0]) * 2
    us2 = np.linspace(best_u - ur, best_u + ur, 25)
    for u in us2:
        if hi + u < 0 or hi + u > 1:
            continue
        u2 = u * u
        cn = c + u * B
        cn[off] -= 2.0 * u2
        cn[off + i - j] += u2
        cn[off + j - i] += u2
        v = float(cn.max())
        if v < best_val:
            best_val = v
            best_u = u
    return best_u

def targeted_pair_search(h, n, max_tries, deadline, rng):
    h = h.copy()

    A = compute_A(h, n)
    off = n - 1
    accepts = 0
    no_improve = 0

        if time.time() > deadline:
            break

            i = int(rng.randint(n)); j = int(rng.randint(n))

                j = int(rng.randint(n))
        else:

            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 1: continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d; m = n - e
                if m <= 1: continue
                b = e + int(rng.randint(m)); i = b - e; j = b

        if u is None:
            no_improve += 1
            continue
        old = float(h[i] + h[j])
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        u = t - float(h[i])
        if abs(u) < 1e-15:

            continue

        cn = c + u * B

def lp_polish(h, n, deadline, eps=0.03, max_rounds=150):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    n_lags = 2 * n - 1
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = A.T
        A_ub[:, n] = -1.0
        b_ub = -c
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        lower = np.maximum(-h, -eps)
        upper = np.minimum(1.0 - h, eps)
        bounds_lp = list(zip(lower, upper)) + [(0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        d = res.x[:n]
        if np.abs(d).max() < 1e-12:
            break
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.125]:
            h_new = project_feasible(h + frac * d, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            if float(c_new.max()) < cur_max - 1e-11:
                h = h_new; c = c_new
                A = compute_A(h, n)
                improved = True
                break
        if not improved:
            break
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)
```
