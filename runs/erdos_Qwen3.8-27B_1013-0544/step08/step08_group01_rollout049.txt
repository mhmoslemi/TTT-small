        t_hi = best_c * n / 2.0 + 5.0
        for _ in range(8):
            if time.time() > deadline:
                break
            t_mid = (t_lo + t_hi) / 2.0
            h_try = penalty_solve(best_h, n, t_mid, 1500)
            c_try = compute_c5(h_try, n)
            if c_try <= t_mid * 2.0 / n + 1e-6:
                t_lo = t_mid
                if c_try < best_c:
                    best_h, best_c = h_try, c_try
            else:
                t_hi = t_mid
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: refine parent with aggressive SLSQP
    if time.time() < deadline - 300:
        alphas = [200, 2000, 20000, 100000]
        best_h, best_c = refine_candidate(best_h, n, min(time.time() + 200, deadline - 200), alphas, 60, 50)
    # Phase 2: Binary SA for diverse starts
    n_sa_pool = []
    sa_deadline = min(time.time() + 120, deadline - 250)
    for sa_seed in [seed, seed + 1, seed + 7]:
        if time.time() > sa_deadline:
            break
        b_sol, b_c = binary_sa(best_h, n, 150000, deadline=sa_deadline, seed=sa_seed)
        h_from_b = project_feasible(b_sol, n)
        n_sa_pool.append((h_from_b, compute_c5(h_from_b, n)))
    # Refine best binary solutions
    n_sa_pool.sort(key=lambda x: x[1])
    for h_b, c_b in n_sa_pool[:2]:
        if time.time() > deadline - 200:
            break
        h_r, c_r = refine_candidate(h_b, n, min(time.time() + 100, deadline - 100),
                                    [1000, 10000, 80000], 50, 40)
        if c_r < best_c:
            best_h, best_c = h_r, c_r
    # Phase 3: Penalty bisection on best
    if time.time() < deadline - 80:
        t_lo = best_c * n / 2.0 * 0.95
        t_hi = best_c * n / 2.0 * 1.05
        for _ in range(10):
            if time.time() > deadline - 80:
                break
            t_mid = (t_lo + t_hi) / 2.0
            h_try = penalty_solve(best_h, n, t_mid, 2000)
            c_try = compute_c5(h_try, n)
            if c_try <= t_mid * 2.0 / n + 1e-5:
                t_lo = t_mid
                if c_try < best_c:
                    best_h, best_c = h_try, c_try
            else:
                t_hi = t_mid
    # Phase 4: Final aggressive refinement
    if time.time() < deadline - 20:
        h_f, c_f = refine_candidate(best_h, n, deadline, [500000, 2000000, 5000000], 200, 80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Phase 5: Try flip symmetry
    h_flip = project_feasible(1.0 - best_h, n)
    c_flip = compute_c5(h_flip, n)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    # Final coordinate descent polish
    if time.time() < deadline:
        h_f, c_f = coord_cd(best_h, n, 300, deadline, 100)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
