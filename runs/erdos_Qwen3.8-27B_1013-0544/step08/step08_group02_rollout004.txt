    cn[off + j - i] += u * u
    return cn

def find_best_u(h, c, A, i, j, n, grid, off):
    hi, hj = h[i], h[j]
    S = hi + hj
    lo = max(0.0, S - 1.0)
    hiT = min(1.0, S)
    if hiT - lo <= 1e-14:
        return None
    us = np.linspace(lo - hi, hiT - hi, grid)
    B = A[i] - A[j]
    lines = c + us[:, None] * B[None, :]
    u2 = us * us
    lines[:, off] -= 2.0 * u2
    lines[:, off + i - j] += u2
    lines[:, off + j - i] += u2
    vals = lines.max(axis=1)
    idx = int(np.argmin(vals))
    if vals[idx] < float(c.max()) - 1e-12:
        return float(us[idx])
    return None

def pair_search(h, n, max_tries, grid, deadline, rng, tol_frac=0.001):
    if n < 2 or max_tries <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    accepts = 0
    no_improve = 0
    max_accept = min(200, max(60, max_tries // 4))
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.01, tol_frac * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e = -d
                m = n - e
                if m <= 0:
                    continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        u = find_best_u(h, c, A, i, j, n, grid, off)
        if u is None:
            no_improve += 1
            continue
        old = float(h[i] + h[j])
        t = float(np.clip(h[i] + u, 0.0, 1.0))
        u = t - float(h[i])
        if abs(u) < 1e-15:
            no_improve += 1
            continue
        cn = apply_move(c, A, i, j, u, n, off)
        nm = float(cn.max())
        if nm < cur_max - 1e-10:
            h[i] = t
            h[j] = old - t
            c = cn
            accepts += 1
            no_improve = 0
            if accepts >= max_accept:
                break
            if accepts % 10 == 0:
                A = compute_A(h, n)
            if accepts % 50 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
        else:
            no_improve += 1
        if no_improve > max(500, max_tries // 2):
            break
    return h, compute_c5(h, n)

def exhaustive_active_pairs(h, n, max_rounds, grid, deadline, max_lags=15):
    """Try all pairs on each active lag systematically."""
    if n < 2 or max_rounds <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = compute_A(h, n)
    off = n - 1
    improved = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.005, 0.0005 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        active = active[:max_lags]
        best_move = None
        best_val = cur_max
        for kk in active:
            d = kk - off
            if d >= 0:
                idxs = range(d, n)
                pairs = [(ii + d, ii) for ii in idxs]
            else:
                e = -d
                pairs = [(ii - e, ii) for ii in range(e, n)]
            for i, j in pairs:
                u = find_best_u(h, c, A, i, j, n, grid, off)
                if u is None:
                    continue
                t = float(np.clip(h[i] + u, 0.0, 1.0))
                u2 = t - float(h[i])
                if abs(u2) < 1e-15:
                    continue
                cn = apply_move(c, A, i, j, u2, n, off)
                nm = float(cn.max())
                if nm < best_val - 1e-12:
                    best_val = nm
                    best_move = (i, j, u2, cn)
        if best_move is None:
            break
        i, j, u, cn = best_move
        old = float(h[i] + h[j])
        h[i] = float(np.clip(h[i] + u, 0.0, 1.0))
        h[j] = old - h[i]
        c = cn
        improved += 1
        if improved % 5 == 0:
            A = compute_A(h, n)
    return h, compute_c5(h, n)

def slsqp_stage(h, n, bounds, cons, plans, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def lbfgs_stage(h, n, bounds, plans, mu, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    for alpha, mx in plans:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(pen_val, work_h, args=(n, alpha, mu), method='L-BFGS-B',
                           jac=pen_jac, bounds=bounds,
                           options={'maxiter': mx, 'ftol': 1e-14, 'maxls': 20, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-7 else best_h
        except Exception:
            pass
    return best_h, best_c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def optimize_instance(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    if compute_c5(hs, n) < c:
        h, c = hs, compute_c5(hs, n)
    hc = 1.0 - h
    if compute_c5(hc, n) < c:
        h, c = hc, compute_c5(hc, n)
    hsc = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    if compute_c5(hsc, n) < c:
        h, c = hsc, compute_c5(hsc, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    if (not quick) and time.time() < deadline - 30.0:
        h, c = lbfgs_stage(h, n, bounds, [(50, 40), (300, 30), (1500, 20)], 1e5, deadline)
    if quick:
        plans = [(100, 30), (1000, 25), (10000, 20)]
    else:
        plans = [(50, 40), (500, 40), (5000, 30), (50000, 20), (500000, 15)]
    h, c = slsqp_stage(h, n, bounds, cons, plans, deadline)
    if n <= 800 and time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mt = int(min(8000, max(500, rem * 30)))
        h, c = pair_search(h, n, mt, 64, deadline, rng, tol_frac=0.0008)
    if n <= 600 and time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mr = int(min(50, max(10, rem * 5)))
        h, c = exhaustive_active_pairs(h, n, mr, 32, deadline, max_lags=10)
    return h, compute_c5(project_feasible(h, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        x = np.linspace(0, 2, 200, endpoint=False)
        h0 = project_feasible(0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x), 200)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    for transform in [lambda v: 0.5*(v + v[::-1]), lambda v: 1.0 - v, lambda v: 0.5*(v + 1.0 - v[::-1])]:
        ht = project_feasible(transform(best_h), n0)
        ct = compute_c5(ht, n0)
        if ct < best_c:
            best_h, best_c = ht, ct
    if hard - t0 < 40.0:
        return best_h, compute_c5(best_h, n0), n0
    main_dl = hard - max(100.0, 0.2 * float(budget_s))
    if main_dl < t0 + 20.0:
        main_dl = hard
    best_h, best_c = optimize_instance(best_h, n0, main_dl, rng, False)
    cands = [300, 400, 500, 600, 800]
    for m in cands:
        if time.time() > hard - 30.0:
            break
        if m == len(best_h):
            continue
        rem = hard - time.time()
        if rem < 50.0:
            break
        dl = time.time() + min(40.0, 0.18 * rem)
        if dl < time.time() + 6.0:
            continue
        ht, ct = optimize_instance(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    n = len(best_h)
    if time.time() < hard - 40.0:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        hd = hard - 3.0
        ht, ct = slsqp_stage(best_h, n, bounds, cons, [(100000, 20), (1000000, 10)], hd)
        if ct < best_c:
            best_h, best_c = ht, ct
    if n <= 800 and time.time() < hard - 3.0:
        rem = hard - time.time()
        mt = int(min(15000, max(2000, rem * 40)))
        ht, ct = pair_search(best_h, n, mt, 128, hard - 2.0, rng, tol_frac=0.0005)
        if ct < best_c:
            best_h, best_c = ht, ct
    if n <= 600 and time.time() < hard - 3.0:
        rem = hard - time.time()
        mr = int(min(80, max(20, rem * 8)))
        ht, ct = exhaustive_active_pairs(best_h, n, mr, 48, hard - 2.0, max_lags=20)
        if ct < best_c:
            best_h, best_c = ht, ct
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
