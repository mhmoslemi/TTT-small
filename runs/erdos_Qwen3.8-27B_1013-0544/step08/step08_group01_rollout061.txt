```python
        except Exception:
            break
```

```python
import numpy as np
from scipy.optimize import linprog, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp(hk, n):
    n_full = 2 * n - 1
    A = np.zeros((n_full, n))
    B = np.zeros(n_full)
    for j in range(n_full):
        i_lo = max(0, n - 1 - j)
        i_hi = min(n - 1, 2 * n - 2 - j)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        idx2 = idx + j - (n - 1)
        A[j, idx] += 1.0 - hk[idx2]
        A[j, idx2] -= hk[idx]
        B[j] = float(np.dot(hk[idx], hk[idx2]))
    return A, B

def slp_solve(h, n, max_iter=60, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    damp = 1.0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        A, B = build_lp(best_h, n)
        n_full = 2 * n - 1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A, -np.eye(n_full)])
        b_ub = -B
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                damp = max(0.1, damp * 0.5)
                if damp < 0.15:
                    break
                continue
            h_lp = res.x[:n]
            h_new = (1.0 - damp) * best_h + damp * h_lp
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new
                damp = min(1.0, damp * 1.15)
            else:
                damp = max(0.1, damp * 0.6)
                if damp < 0.12:
                    break
        except Exception:
            break
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def coord_cd(h, n, max_passes=150, deadline=None, grid_size=70):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh,
                                 args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_polish(h, n, max_iter=100, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        sg = np.zeros(n)
        for i in range(n):
            j = i + m_star - (n - 1)
            if 0 <= j < n:
                sg[i] += (1.0 - h[j])
            ip = i - m_star + (n - 1)
            if 0 <= ip < n:
                sg[i] -= h[ip]
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 25 or lr < 1e-10:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLP from initial
    if time.time() < deadline - 200:
        h1, c1 = slp_solve(best_h, n, 80, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent polish
    if time.time() < deadline - 150:
        h2, c2 = coord_cd(best_h, n, 200, deadline, 80)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Subgradient
    if time.time() < deadline - 100:
        h3, c3 = subgradient_polish(best_h, n, 150, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c2 if False else c3
    # Phase 4: SLP again (from improved point)
    if time.time() < deadline - 120:
        h4, c4 = slp_solve(best_h, n, 60, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h5, c5 = coord_cd(best_h, n, 150, deadline, 80)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 5: Multi-scale
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 500, 800]:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slp_solve(h_int, n_try, 50, deadline)
        h_opt, c_opt = coord_cd(h_opt, n_try, 100, deadline, 60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try)
    # Phase 6: Multi-restart perturbations
    n_cur = len(best_h)
    for r in range(8):
        if time.time() > deadline - 40:
            break
        s = 0.003 + 0.015 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = slp_solve(hp, n_cur, 40, deadline)
        h_o, c_o = coord_cd(h_o, n_cur, 80, deadline, 50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 7: Final high-precision SLP + CD
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = slp_solve(best_h, n_cur, 80, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 10:
        hf, cf = coord_cd(best_h, n_cur, 400, deadline, 100)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
```
