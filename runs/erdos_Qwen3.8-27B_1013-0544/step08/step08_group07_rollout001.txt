            if tot < 1e-15:

        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_try, c_try = lbfgs_opt(h_try, n_try, [5000, 50000, 200000], 300, deadline)
        h_try, c_try = coord_cd(h_try, n_try, 150, deadline, grid_size=100)
        if c_try < best_c:
            best_h, best_c = h_try, c_try

    # Phase 5: Small perturbations + CD from best
    if time.time() < deadline - 80:
        n = len(best_h)
        for s in [0.003, 0.007, 0.012]:
            if time.time() > deadline - 40:
                break
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = coord_cd(hp, n, 200, deadline, grid_size=100)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: L-BFGS with very high alpha on current best
    if time.time() < deadline - 40:
        n = len(best_h)
        hf, cf = lbfgs_opt(best_h, n, [500000, 1000000, 2000000], 500, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 7: Final ultra-fine CD
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, len(best_h), 250, deadline, grid_size=200)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Final repair and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
