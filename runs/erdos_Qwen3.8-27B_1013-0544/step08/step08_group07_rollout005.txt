        c_new = compute_c5(hp, n)
        if c_new < best_c:
            h = hp
            best_c = c_new
    return h, best_c

def structured_start(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.35 * np.sin(4 * np.pi * x) + 0.1 * np.sin(8 * np.pi * x)
    elif kind == 2:
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif kind == 3:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x)
    elif kind == 4:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * x) * np.cos(4 * np.pi * x)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = structured_start(n_init, 0, rng)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B active set on initial
    t_rem = deadline - time.time()
    if t_rem > 60:
        h1, c1 = lbfgs_opt(best_h, n_init, [500, 3000, 15000, 80000], 60, 400, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Active CD
    if time.time() < deadline - 30:
        h2, c2 = active_cd(best_h, n_init, 200, 40, 50, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Block perturb + re-optimize
    if time.time() < deadline - 60:
        h3, c3 = block_perturb(best_h, n_init, rng, [3, 5, 8, 12, 20], 60, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = active_cd(best_h, n_init, 150, 35, 50, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-n exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [200, 250, 300, 400, 500, 600, 800]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_t = np.linspace(0, 2, n_try, endpoint=False)
        h_t = project_feasible(np.interp(x_t, x_best, best_h) + 0.01 * rng.randn(n_try), n_try)
        h_t, c_t = lbfgs_opt(h_t, n_try, [2000, 10000, 50000], 50, 200, deadline)
        h_t, c_t = active_cd(h_t, n_try, 100, 35, 45, deadline)
        if c_t < best_c:
            best_h, best_c, n_final = h_t, c_t, n_try
            x_best = np.linspace(0, 2, n_final, endpoint=False)
    # Phase 5: Diverse restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        kinds = [0, 1, 2, 3, 4, 5]
        for r in range(min(8, kinds.__len__())):
            if time.time() > deadline - 30:
                break
            hd = structured_start(n_cur, kinds[r] % 6, rng)
            hd, cd = lbfgs_opt(hd, n_cur, [3000, 20000], 50, 250, deadline)
            hd, cd = active_cd(hd, n_cur, 100, 35, 45, deadline)
            if cd < best_c:
                best_h, best_c = hd, cd
    # Phase 6: Final high-alpha polish
    if time.time() < deadline - 10:
        n_f = len(best_h)
        hf, cf = lbfgs_opt(best_h, n_f, [50000, 200000, 500000], 80, 500, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = active_cd(best_h, n_f, 300, 50, 70, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
