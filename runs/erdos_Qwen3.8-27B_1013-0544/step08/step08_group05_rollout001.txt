```python
    # Second swap
    d[i2:i2+n] -= om2
    st = n - 1 - i2
    d[st:st+n] += h2
    d[j2:j2+n] += om2
    st = n - 1 - j2
    d[st:st+n] -= h2
    d[n-1] -= 2.0
    d[n-1+i2-j2] += 1.0
    d[n-1+j2-i2] += 1.0
    return d, float(d.max())
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                method="SLSQP", bounds=bounds, constraints=cons,
                options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def apply_swap(c, h, om, i, j, n):
    d = c.copy()
    d[i:i + n] -= om
    st = n - 1 - i
    d[st:st + n] += h
    d[j:j + n] += om
    st = n - 1 - j
    d[st:st + n] -= h
    d[n - 1] -= 2.0
    d[n - 1 + i - j] += 1.0
    d[n - 1 + j - i] += 1.0
    return d, float(d.max())

def binary_ils(h_init, n, rng, deadline):
    L = 2 * n - 1
    best = h_init.copy()
    bc = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(bc.max())
    ones = np.where(best > 0.5)[0]
    zeros = np.where(best < 0.5)[0]
    stagnant = 0
    tabu = []
    tabu_len = 200
    while time.time() < deadline:
        h = best.copy()
        c = bc.copy()
        om = 1.0 - h[::-1]
        cm = float(c.max())
        improved_phase = False
        plateau = 0
        max_plateau = 5000 + 2000 * stagnant
        trials = 0
        while time.time() < deadline and plateau < max_plateau and trials < 60000:
            if rng.rand() < 0.15 and len(ones) > 2 and len(zeros) > 2:
                i1 = int(ones[rng.randint(len(ones))])
                i2 = int(ones[rng.randint(len(ones))])
                while i2 == i1:
                    i2 = int(ones[rng.randint(len(ones))])
                j1 = int(zeros[rng.randint(len(zeros))])
                j2 = int(zeros[rng.randint(len(zeros))])
                while j2 == j1:
                    j2 = int(zeros[rng.randint(len(zeros))])
                d1, _ = apply_swap(c, h, om, i1, j1, n)
                h2 = h.copy()
                h2[i1] = 0.0
                h2[j1] = 1.0
                om2 = 1.0 - h2[::-1]
                d2, nm = apply_swap(d1, h2, om2, i2, j2, n)
                trials += 1
                key = (min(i1,i2), max(i1,i2), min(j1,j2), max(j1,j2))
                if nm < cm - 1e-9 and key not in tabu:
                    h = h2
                    h[i2] = 0.0
                    h[j2] = 1.0
                    c = d2
                    cm = nm
                    om = 1.0 - h[::-1]
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    plateau = 0
                    improved_phase = True
                    if cm < bcm:
                        best = h.copy()
                        bc = c.copy()
                        bcm = cm
                        stagnant = 0
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                c2, nm = apply_swap(c, h, om, i, j, n)
                trials += 1
                key = (i, j)
                if nm < cm - 1e-9 and key not in tabu:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = c2
                    cm = nm
                    om[n - 1 - i] = 1.0
                    om[n - 1 - j] = 0.0
                    ones[ones == i] = j
                    zeros[zeros == j] = i
                    plateau = 0
                    improved_phase = True
                    if cm < bcm:
                        best = h.copy()
                        bc = c.copy()
                        bcm = cm
                        stagnant = 0
                else:
                    plateau += 1
            if len(tabu) > tabu_len:
                tabu.pop(0)
            tabu.append((i, j)) if isinstance(key, tuple) and len(key) == 2 else None
        if not improved_phase:
            stagnant += 1
            if stagnant > 8:
                break
        if time.time() > deadline - 5:
            break
        n_swap = min(3 + stagnant, 8)
        h = best.copy()
        for _ in range(n_swap):
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            if ii != jj:
                h[ii] = 0.0
                h[jj] = 1.0
        bc = np.correlate(h, 1.0 - h, mode="full")
        bc_m = float(bc.max())
        if bc_m > bcm:
            best = h.copy()
            bc = bc.copy()
            bcm = bc_m
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
    return best, bcm * (2.0 / n)

def coord_cd(h, n, deadline, grid=32, refine=12, max_passes=4):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            old = float(x[p])
            A = np.zeros(L)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            d = xs0 - old
            cand = c[None, :] + d[:, None] * A[None, :]
            cand[:, n - 1] -= d * d
            i = int(np.argmin(cand.max(axis=1)))
            bx = float(xs0[i])
            bv = float(cand.max(axis=1)[i])
            if bv < cur - 1e-12:
                sp = 1.0 / (grid - 1)
                lo = max(0.0, bx - sp)
                hi = min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                dd = xx - old
                cc = c[None, :] + dd[:, None] * A[None, :]
                cc[:, n - 1] -= dd * dd
                mm = cc.max(axis=1)
                j = int(np.argmin(mm))
                if mm[j] < bv - 1e-12:
                    bx = float(xx[j])
                    bv = float(mm[j])
            if bv < float(c.max()) - 1e-12:
                dd = bx - old
                c[p:p + n] += dd * om
                c[st:st + n] -= dd * x
                c[diag] -= dd * dd
                x[p] = bx
                om[diag - p] = 1.0 - bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-12:
            best = xp
            bc = cp
        if not improved:
            break
    return best, bc

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)
```
