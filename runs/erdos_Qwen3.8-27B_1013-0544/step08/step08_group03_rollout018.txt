            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        b_nd = b_valid & ~b_diag
        idx_cache[p] = (ks, j1, a_nd, a_diag, j2, b_nd, b_diag)
    return idx_cache
