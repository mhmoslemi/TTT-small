def subgradient_step(h, n, c, lr):
    """One subgradient step toward reducing the max correlation."""
    dx = 2.0 / n
    target = c.max()
    # Find lags at or near the max
    threshold = target - 0.01 * dx
    active = np.where(c >= threshold)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # dc[k]/dh[i] = (1-h[i+k-(n-1)]) - h[i-k+(n-1)] (for valid indices)
        # Plus the diagonal term if i = i+k-(n-1), i.e., k = n-1
        jf = np.arange(n) + k - (n - 1)
        jb = np.arange(n) - k + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        g = np.zeros(n)
        if np.any(vf):
            g[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            g[vb] -= h[jb[vb]]
        if k == n - 1:
            g -= 2.0 * h  # diagonal: d(h*(1-h))/dh = 1-2h, but we already have (1-h) - h = 1-2h
            # Wait, for k=n-1: jf = i, jb = i, so g[i] += (1-h[i]) - h[i] = 1-2h[i]
            # That's already correct! No extra term needed.
        grad += g
    
    grad /= len(active)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d(h, p, n, c):
    h0 = h[p]
    m_lo = max(0, n - 1 - p, p)
    m_hi = min(2 * n - 2, n - 1 + p)
    if m_lo > m_hi:
        return h0
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_aff = c[ms]
    diag = np.where(ms == (n - 1))[0]
    has_diag = len(diag) > 0
    c_d = c_aff[diag[0]] if has_diag else 0.0
    a_d = a[diag[0]] if has_diag else 0.0
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    L_lo = max(unaff, 0.0)
    L_hi = float(c.max()) + 1e-10
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo = -h0
        x_hi = 1.0 - h0
        ok = True
        if np.any(neg_m):
            x_lo = max(x_lo, float(((L - c_aff[neg_m]) / a[neg_m]).max()))
        if np.any(pos_m):
            x_hi = min(x_hi, float(((L - c_aff[pos_m]) / a[pos_m]).min()))
        if x_lo > x_hi + 1e-13:
            ok = False
        if ok and has_diag:
            disc = a_d * a_d - 4.0 * (L - c_d)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if max(x_lo, -h0) >= r1 - 1e-13 and min(x_hi, 1.0 - h0) <= r2 + 1e-13:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L_opt = L_hi
    x_lo = -h0
    x_hi = 1.0 - h0
    if np.any(neg_m):
        x_lo = max(x_lo, float(((L_opt - c_aff[neg_m]) / a[neg_m]).max()))
    if np.any(pos_m):
        x_hi = min(x_hi, float(((L_opt - c_aff[pos_m]) / a[pos_m]).min()))
    x_lo = max(x_lo, -h0)
    x_hi = min(x_hi, 1.0 - h0)
    if has_diag:
        disc = a_d * a_d - 4.0 * (L_opt - c_d)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if max(x_lo, -h0) >= r1 - 1e-13 and min(x_hi, 1.0 - h0) <= r2 + 1e-13:
                mid = (max(x_lo, -h0) + min(x_hi, 1.0 - h0)) / 2.0
                if mid <= (r1 + r2) / 2.0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    x_lo = max(x_lo, -h0)
    x_hi = min(x_hi, 1.0 - h0)
    if x_lo > x_hi:
        return h0
    return float(np.clip(h0 + (x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_cd(h, n, max_passes, deadline, reverse=False):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = range(n - 1, -1, -1) if reverse else range(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xv = solve_1d(h, p, n, c)
            if abs(xv - h0) > 1e-11:
                h[p] = xv
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def block_cd_2(h, n, max_passes, deadline):
    """Optimize pairs of adjacent coordinates via grid search on 2D."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(0, n - 1, 2):
            if time.time() > deadline:
                break
            h0a, h0b = h[p], h[p + 1]
            if (h0a < 1e-12 or h0a > 1 - 1e-12) and (h0b < 1e-12 or h0b > 1 - 1e-12):
                continue
            best_c = float(c.max())
            best_da, best_db = 0.0, 0.0
            for da in np.linspace(-h0a, 1 - h0a, 12):
                if abs(da) < 1e-13:
                    continue
                h2 = h.copy()
                h2[p] = h0a + da
                row = c.copy()
                if True:
                    for dd in np.linspace(-h0b, 1 - h0b, 12):
                        if abs(dd) < 1e-13 and abs(da) < 1e-13:
                            continue
                        h2[p + 1] = h0b + dd
                        if abs(dd) > 1e-13:
                            c2 = np.correlate(h2, 1.0 - h2, mode='full')
                            v = float(c2.max())
                            if v < best_c:
                                best_c = v
                                best_da, best_db = da, dd
            if abs(best_da) > 1e-12 or abs(best_db) > 1e-12:
                h[p] = float(np.clip(h0a + best_da, 0, 1))
                h[p + 1] = float(np.clip(h0b + best_db, 0, 1))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP warm-up
    h1, c1 = slsqp_run(best_h, n_init, [200, 1000, 5000, 20000], maxiter_each=350)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Exact CD (forward)
    h2, c2 = coord_cd(best_h, n_init, 30, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Exact CD (backward) for different ordering
    h3, c3 = coord_cd(best_h, n_init, 20, deadline, reverse=True)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Multi-n exploration
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [300, 400, 500, 600, 800, 1000, 1200, 200, 250]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [500, 3000, 10000, 50000], maxiter_each=250)
        h_opt, c_opt = coord_cd(h_opt, n_try, 25, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 5: Perturbation restarts
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(2, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            s = 0.008 + 0.05 * (r / max(1, n_restarts - 1))
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = slsqp_run(hp, n, [2000, 10000, 50000], maxiter_each=200)
            h_o, c_o = coord_cd(h_o, n, 25, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final alternating CD
    if time.time() < deadline - 8:
        n = len(best_h)
        passes = max(3, int((deadline - time.time() - 4) / 3.0))
        for i in range(passes):
            if time.time() > deadline - 4:
                break
            hf, cf = coord_cd(best_h, n, max(1, passes // 3), deadline, reverse=(i % 2 == 1))
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = slsqp_run(best_h, n, [200000, 1000000], maxiter_each=300)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
