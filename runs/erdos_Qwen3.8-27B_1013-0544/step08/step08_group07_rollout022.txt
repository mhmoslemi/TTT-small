def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h

            return lse * (2.0 / n)

            b_hi = min(2 * n - 2, p + n - 1)
            lo = min(f_lo, b_lo)
            hi = max(f_hi, b_hi)
            
            if lo >= hi:
                continue
            
            ms = np.arange(lo, hi + 1)
            q = p + ms - (n - 1)
            r = p - ms + (n - 1)
            a = np.zeros(len(ms))
            vq = (q >= 0) & (q < n)
            vr = (r >= 0) & (r < n)
            a[vq] += 1.0 - h[q[vq]]
            a[vr] -= h[r[vr]]
            
            diag_pos = (n - 1) - lo
            has_diag = 0 <= diag_pos < len(ms)
            
            unaff_max = 0.0
            if lo > 0:
                unaff_max = max(unaff_max, float(c[:lo].max()))
            if hi < n_corr - 1:
                unaff_max = max(unaff_max, float(c[hi+1:].max()))
            
            c_aff = c[lo:hi+1]
            d_lo = -v + 1e-10
            d_hi = 1.0 - v - 1e-10
            if d_lo >= d_hi:
                continue
            
            # Vectorized grid search
            deltas = np.linspace(d_lo, d_hi, n_grid)
            nv = c_aff[None, :] + deltas[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_pos] -= deltas[:, None]**2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_idx = int(vals.argmin())
            best_d = float(deltas[best_idx])
            best_val = float(vals[best_idx])
            
            if abs(best_d) > 1e-10 and best_val < current_max - 1e-14:
                h[p] = float(np.clip(v + best_d, 0.0, 1.0))
                any_improved = True
                c[lo:hi+1] = c_aff + best_d * a
                if has_diag:
                    c[lo + diag_pos] -= best_d * best_d
        
        if not any_improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    
    return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

            break

            c = np.correlate(x, 1.0 - x, mode='full')
            mx = c.max()
            sh = a * (c - mx)
            lse = mx + np.log(np.sum(np.exp(sh - sh.max()))) / a
            return lse * (2.0 / n)
        def sc(x):
            return x.sum() - n / 2.0

            return np.ones(n)

        try:
