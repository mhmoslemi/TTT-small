            room = 1.0 - h

            # Compute new c: c + sens_p - sens_q
            # sens[p][m] = (1-h[p-ell_m])*1[valid] - h[p+ell_m]*1[valid]
            ell = np.arange(2*n-1) - center
            sens_p = np.zeros(2*n-1)
            sens_q = np.zeros(2*n-1)
            v1p = (p - ell >= 0) & (p - ell < n)
            v2p = (p + ell >= 0) & (p + ell < n)
            sens_p[v1p] = 1.0 - h[p - ell[v1p]]
            sens_p[v2p] -= h[p + ell[v2p]]
            v1q = (q - ell >= 0) & (q - ell < n)
            v2q = (q + ell >= 0) & (q + ell < n)
            sens_q[v1q] = 1.0 - h[q - ell[v1q]]
            sens_q[v2q] -= h[q + ell[v2q]]
            new_cmax = float(np.max(c + sens_p - sens_q))
            if new_cmax < cmax - 1e-10:
                # Accept
                c = c + sens_p - sens_q
                h[p] = 1.0
                h[q] = 0.0
                cmax = new_cmax
                ones.remove(p)
                zeros.append(p)
                zeros.remove(q)
                ones.append(q)
                improved = True
                if cmax * (2.0/n) < best_c:
                    best_c = cmax * (2.0/n)
                    best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c
