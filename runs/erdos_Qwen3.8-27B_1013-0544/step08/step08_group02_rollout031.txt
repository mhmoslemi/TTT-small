import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    lo, hi = float(z.min()) - 2.0, float(z.max()) + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-12:
        mask = (y < 1.0) if res > 0 else (y > 0.0)
        if mask.any():
            room = (1.0 - y) if res > 0 else y
            tot = room[mask].sum()
            if tot > 1e-14:
                y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def smooth_max(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    val = (m + float(np.log(s)) / alpha) * (2.0 / n)
    return val, p, c

def grad_from_p(h, n, p):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-13:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - h[:n - d])
            g[:n - d] -= w * h[d:]
        else:
            e = -d
            g[:n - e] += w * (1.0 - h[e:])
            g[e:] -= w * h[:n - e]
    return (2.0 / n) * g

def fista_opt(h, n, alpha, max_iter, deadline):
    h = h.copy()
    y = h.copy()
    t = 1.0
    step = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        fv, p, _ = smooth_max(y, n, alpha)
        g = grad_from_p(y, n, p)
        L = step
        x = y - g / L
        ok = False
        for _ in range(30):
            x = project_feasible(y - g / L, n)
            ft = smooth_max(x, n, alpha)[0]
            rhs = fv + float(np.dot(g, x - y)) + 0.5 * L * float(np.dot(x - y, x - y))
            if ft <= rhs + 1e-12:
                ok = True
                break
            L *= 2.0
        if not ok:
            break
        step = L
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - h)
        h = x
        t = t_new
        if it % 50 == 0 or it >= max_iter - 5:
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def slsqp_refine(h, n, alpha, max_iter, deadline):
    from scipy.optimize import minimize

    def f_obj(x):
        return smooth_max(x, n, alpha)[0]

    def f_grad(x):
        _, p, _ = smooth_max(x, n, alpha)
        return grad_from_p(x, n, p)

    def c_fun(x):
        return x.sum() - 0.5 * n

    def c_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(f_obj, h, jac=f_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        x = project_feasible(np.asarray(res.x, float), n)
        c = compute_c5(x, n)
        if c < compute_c5(h, n):
            return x, c
    except Exception:
        pass
    return h, compute_c5(h, n)

def pair_search(h, n, max_tries, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    accepts = 0
    no_imp = 0
    max_acc = min(200, max(50, max_tries // 4))
    for trial in range(max_tries):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.03, 0.002 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        if trial % 7 == 6 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 1:
                    continue
                b = int(rng.randint(m)); j = b; i = b + d
            else:
                e = -d
                m = n - e
                if m <= 1:
                    continue
                b = e + int(rng.randint(m)); i = b - e; j = b
        hi, hj = h[i], h[j]
        S = hi + hj
        lo = max(0.0, S - 1.0)
        hiT = min(1.0, S)
        if hiT - lo < 1e-14:
            no_imp += 1
            continue
        us = np.linspace(lo - hi, hiT - hi, 41)
        B = A[i] - A[j]
        lines = c + us[:, None] * B[None, :]
        u2 = us * us
        lines[:, off] -= 2.0 * u2
        if i != j:
            lines[:, off + i - j] += u2
            lines[:, off + j - i] += u2
        vals = lines.max(axis=1)
        idx = int(np.argmin(vals))
        if vals[idx] < cur_max - 5e-10:
            u = float(us[idx])
            t = float(np.clip(hi + u, 0.0, 1.0))
            h[i], h[j] = t, S - t
            cn = c + u * B
            u2v = u * u
            cn[off] -= 2.0 * u2v
            if i != j:
                cn[off + i - j] += u2v
                cn[off + j - i] += u2v
            c = np.clip(cn, 0.0, None)
            accepts += 1
            no_imp = 0
            if accepts >= max_acc:
                break
            A = compute_A(h, n)
        else:
            no_imp += 1
        if no_imp > max(200, max_tries // 3):
            break
    return h, compute_c5(h, n)

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def apply_sym(h, n):
    best = h.copy()
    best_c = compute_c5(h, n)
    for transform in [lambda x: 1.0 - x, lambda x: x[::-1],
                      lambda x: 0.5*(x + x[::-1]), lambda x: 0.5*(x + 1.0 - x[::-1])]:
        ht = project_feasible(transform(h), n)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best, best_c = ht, ct
    return best, best_c

def optimize_at_n(h, n, deadline, rng):
    h, c = apply_sym(project_feasible(h, n), n)
    alphas = [50, 200, 800, 3000, 12000]
    for a in alphas:
        if time.time() > deadline:
            break
        rem = max(1.0, deadline - time.time())
        mx = int(min(20000, max(500, rem * 80)))
        h, c = fista_opt(h, n, a, mx, deadline)

    if n <= 600 and time.time() < deadline:

    if n <= 800 and time.time() < deadline - 3.0:
        rem = deadline - time.time()
        mt = int(min(4000, max(300, rem * 25)))
        h, c = pair_search(h, n, mt, deadline, rng)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_to(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 10):
        a = rng.uniform(-0.5, 0.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += a * np.sin(0.5 * np.pi * m * x + ph)
        h += 0.3 * a * np.cos(np.pi * m * x)
    h += 0.1 * (x - 1.0)
    return project_feasible(np.clip(h, 0.0, 1.0), n)
