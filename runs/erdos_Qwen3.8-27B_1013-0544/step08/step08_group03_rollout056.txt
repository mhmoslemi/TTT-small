def cccp_solve(h, n, max_iters=50, deadline=None):
    """Concave-Convex Procedure with LP subproblems."""
    h = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        # Compute current correlations
        c = np.correlate(h, 1.0 - h, mode='full')
        # For each shift k (0 to 2n-2), the correlation is c[k]
        # We want to linearize: c[k] ≈ c[k]_current + dc[k]/dh * (h_new - h)
        # dc[k]/dh[p] = (1-h[p+dm]) - h[p-dm] where dm = k-(n-1)
        
        # Build the Jacobian J where J[k,p] = dc[k]/dh[p]
        # This is a (2n-1) x n matrix
        # J[k,p] = (1-h[p+dm]) - h[p-dm] where dm = k-(n-1)
        # = (1-h[p+k-n+1]) - h[p-k+n-1]
        
        # For efficiency, only solve with the active constraint(s)
        # The active constraint is k* = argmax c
        k_star = int(np.argmax(c))
        dm = k_star - (n-1)
        
        # Gradient of c[k_star] with respect to h
        grad = np.zeros(n)
        # term1: (1-h[p+dm]) for p+dm in [0,n)
        idx1 = np.arange(n) + dm
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += (1.0 - h[idx1[mask1]])
        # term2: -h[p-dm] for p-dm in [0,n)  
        idx2 = np.arange(n) - dm
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += -h[idx2[mask2]]
        
        # LP: min t
        # s.t. c[k_star] + grad^T (h_new - h) ≤ t  (linearized active constraint)
        #      Also include all other constraints linearized (for safety)
        #      0 ≤ h_new ≤ 1
        #      sum(h_new) = n/2
        
        # Actually, for CCCP to work properly, I should include ALL constraints
        # (not just the active one), because other constraints might become active.
        # But that's 2n-1 constraints, which makes the LP larger.
        # Let me include just the top few active constraints.
        
        # For now, use just the single active constraint for speed:
        # min t
        # s.t. c[k_star] + grad^T h_new - grad^T h ≤ t
        #      0 ≤ h_new ≤ 1
        #      sum(h_new) = n/2
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n/2.0,

                break

def subgradient_avg(h, n, max_iters=500, deadline=None, eta_init=0.05):

    h = h.copy()
    h_avg = h.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        eta = eta_init / (1.0 + 0.01 * it)
        
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        dm = k_star - (n-1)
        
        # Subgradient
        grad = np.zeros(n)
        idx1 = np.arange(n) + dm
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += (1.0 - h[idx1[mask1]])
        idx2 = np.arange(n) - dm
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += -h[idx2[mask2]]
        
        # Update
        h = project_feasible(h - eta * grad * dx, n)
        h_avg = ((it * h_avg + h) / (it + 1))
        h_avg = project_feasible(h_avg, n)
        
        # Track best
        c_curr = compute_c5(h, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        c_avg = compute_c5(h_avg, n)
        if c_avg < best_c:
            best_c = c_avg
            best_h = h_avg.copy()
    
    return best_h, best_c
