```python
def smooth_c5_grad(h, n, alpha):
    """Compute gradient of smoothed C5 w.r.t. h."""
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = ee.sum()
    w = ee / z
    val = (m + np.log(z) / alpha) * (2.0/n)
    
    # Only keep significant weights
    thresh = 1e-10
    active = w > thresh
    if not active.any():
        return val, np.zeros(n)
    
    ka = np.where(active)[0] - (n-1)  # actual k values (can be negative)
    wa = w[active]
    
    grad = np.zeros(n)
    idx = np.arange(n)
    for ki, wki in zip(ka, wa):
        if ki >= 0:
            v1 = idx + ki < n
            grad[v1] += wki * (1.0 - h[idx[v1] + ki])
            v2 = idx >= ki
            grad[v2] -= wki * h[idx[v2] - ki]
        else:
            kp = -ki
            v1 = idx >= kp
            grad[v1] += wki * (1.0 - h[idx[v1] - kp])
            v2 = idx + kp < n
            grad[v2] -= wki * h[idx[v2] + kp]
    
    return val, grad * (2.0/n)
```

```python
import numpy as np

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max()) * (2.0 / n)

def smooth_c5_og(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = ee.sum()
    w = ee / z
    val = (m + np.log(z) / alpha) * (2.0 / n)
    active = w > 1e-10
    if not active.any():
        return val, np.zeros(n)
    ka = np.where(active)[0] - (n - 1)
    wa = w[active]
    grad = np.zeros(n)
    idx = np.arange(n)
    for ki, wki in zip(ka, wa):
        if ki >= 0:
            v1 = idx + ki < n
            if v1.any():
                grad[v1] += wki * (1.0 - h[idx[v1] + ki])
            v2 = idx >= ki
            if v2.any():
                grad[v2] -= wki * h[idx[v2] - ki]
        else:
            kp = -ki
            v1 = idx >= kp
            if v1.any():
                grad[v1] += wki * (1.0 - h[idx[v1] - kp])
            v2 = idx + kp < n
            if v2.any():
                grad[v2] -= wki * h[idx[v2] + kp]
    return val, grad * (2.0 / n)

def subgradient_phase(h0, n, deadline, n_iter=400):
    h = h0.copy()
    best = h.copy()
    bc = compute_c5(best, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = c.max()
    alpha = 50.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        k = k_star - (n - 1)
        idx = np.arange(n)
        g = np.zeros(n)
        if k >= 0:
            v1 = idx + k < n
            g[v1] += (1.0 - h[idx[v1] + k])
            v2 = idx >= k
            g[v2] -= h[idx[v2] - k]
        elif k < 0:
            kp = -k
            v1 = idx >= kp
            g[v1] += (1.0 - h[idx[v1] - kp])
            v2 = idx + kp < n
            g[v2] -= h[idx[v2] + kp]
        else:
            g[:] = 1.0 - 2.0 * h
        step = (cm - (cm - 0.5)) / max(np.dot(g, g), 1e-12)
        step = min(step, 0.5)
        h_new = project_simplex(h + step * g, n)
        c = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_cm = c.max()
        if new_cm < cm - 1e-10:
            h = h_new
            cm = new_cm
            nc = cm * dx
            if nc < bc:
                best = h.copy()
                bc = nc
        else:
            step *= 0.5
            h_new = project_simplex(h + step * g, n)
            c = np.correlate(h_new, 1.0 - h_new, mode="full")
            if c.max() < cm - 1e-10:
                h = h_new
                cm = c.max()
                nc = cm * dx
                if nc < bc:
                    best = h.copy()
                    bc = nc
    return best, bc

def lbfgs_phase(h0, n, alphas, deadline, maxiter=80):
    h = project_simplex(h0, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_c5_og(x, n, a)[0]
        def grad(x):
            return smooth_c5_og(x, n, a)[1]
        try:
            res = minimize(obj, best, jac=grad, method="L-BFGS-B",
                           bounds=bounds, options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-8})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc

def cd_slope_1d(h, n, max_passes=8, deadline=None):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    dx = 2.0 / n
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        cm = c.max()
        improved = False
        nm1 = n - 1
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            def eval_1d(xp):
                c2 = c.copy()
                d = xp - x[p]
                if abs(d) < 1e-15:
                    return cm
                for k in range(-(n-1), n):
                    kk = n - 1 + k
                    wk = 1.0
                    if k >= 0:

                    elif k < 0:

            lo, hi = 0.0, 1.0

def cd_grid(h, n, max_passes=6, deadline=None, grid=24, refine=12):

        if deadline is not None and time.time() > deadline:

        x = best.copy()

            if deadline is not None and time.time() > deadline:

def binary_search(h_init, n, rng, deadline):
    if n % 2 != 0:
        return None, np.inf
    target = n // 2
    idx = np.argpartition(h_init, target - 1)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    best = h.copy()
    bc = cm
    dx = 2.0 / n
    stagnant = 0
    while time.time() < deadline and stagnant < 5:
        h = best.copy()
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max())
        om = 1.0 - h[::-1]
        plateau = 0
        for _ in range(30000):
            if time.time() > deadline:
                break
            if plateau > 5000:
                break
            i = int(np.argmax(h))
            j = int(np.argmin(h))
            if h[i] < h[j]:
                i, j = j, i
            if i == j:
                break
            i = int(rng.randint(n))
            while h[i] < 0.5:
                i = int(rng.randint(n))
            j = int(rng.randint(n))
            while h[j] >= 0.5:
                j = int(rng.randint(n))
            if i == j:
                continue
            d = c.copy()
            d[i:i + n] -= om
            st = n - 1 - i
            d[st:st + n] += h
            d[j:j + n] += om
            st = n - 1 - j
            d[st:st + n] -= h
            d[n - 1] -= 2.0
            d[n - 1 + i - j] += 1.0
            d[n - 1 + j - i] += 1.0
            nm = float(d.max())
            if nm < cm - 1e-9:
                c, cm = d, nm
                h[i] = 0.0
                h[j] = 1.0
                om[n - 1 - i] = 1.0
                om[n - 1 - j] = 0.0
                plateau = 0
                if cm < bc:
                    best = h.copy()
                    bc = cm
            else:
                plateau += 1
        if bc < cm * 0.999:
            stagnant = 0
        else:
            stagnant += 1
    return best, bc * dx

def transfer_h(h, n, n_t):
    if n_t == n:
        return project_simplex(h.copy(), n_t)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.linspace(0.0, 2.0, n_t, endpoint=False)
    return project_simplex(np.interp(y, x, h), n_t)
```
