from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0: return z
    if z.min() >= -1e-12 and z.max() <= 1.0+1e-12 and abs(z.sum()-target) <= 1e-9*max(1,n):
        return np.clip(z, 0.0, 1.0)
    low, high = float(z.min())-2.0, float(z.max())+2.0
    for _ in range(60):
        tau = 0.5*(low+high)
        s = np.clip(z-tau, 0.0, 1.0).sum()
        if s > target: low = tau
        else: high = tau
    y = np.clip(z - 0.5*(low+high), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        mask = (y < 1.0) if res > 0 else (y > 0.0)
        room = (1.0-y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res/tot)*room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n): return float(x.sum() - 0.5*n)
def sum_jac(x, n): return np.ones_like(x)

def smooth_c5_and_weights(h, n, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    z = alpha*(c - m)
    zm = z.max()
    e = np.exp(z - zm)
    p = e / e.sum()
    val = (m + np.log(e.sum())/alpha) * (2.0/n)
    return val, p, c

def grad_from_weights(x, n, p):
    g = np.zeros(n)
    off = n-1
    for k in range(2*n-1):
        w = p[k]
        if w <= 1e-14: continue
        d = k - off
        if d >= 0:
            g[d:] += w*(1.0-x[:n-d])
            g[:n-d] -= w*x[d:]
        else:
            e = -d
            g[:n-e] += w*(1.0-x[e:])
            g[e:] -= w*x[:n-e]
    return (2.0/n)*g

def smooth_grad(x, n, alpha): return grad_from_weights(x, n, smooth_c5_and_weights(x, n, alpha)[1])

def slsqp_optimize(h, n, alpha, maxiter, deadline):

    cons = [{'type':'eq','fun':sum_con,'jac':sum_jac,'args':(n,)}]

                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':1e-14,'disp':False})

        x = project_feasible(x, n)

    except: pass
    return h, compute_c5(h, n)

def compute_A(h, n):
    A = np.zeros((n, 2*n-1), dtype=float)
    off = n-1
    for k in range(2*n-1):
        d = k-off
        if d >= 0:
            A[d:, k] += 1.0-h[:n-d]
            A[:n-d, k] -= h[d:]
        else:
            e = -d
            A[:n-e, k] += 1.0-h[e:]
            A[e:, k] -= h[:n-e]
    return A

def eval_pair(h, c, A, i, j, n, ng):
    hi, hj = h[i], h[j]
    S = hi+hj
    lo = max(0.0, S-1.0)
    hiT = min(1.0, S)
    if hiT-lo <= 1e-14: return None
    B = A[i]-A[j]
    us = np.linspace(lo-hi+1e-7, hiT-hi-1e-7, max(5,ng))
    off = n-1
    lines = c + us[:,None]*B[None,:]
    u2 = us*us
    lines[:, off] -= 2.0*u2
    if abs(i-j) < 2*n-1:
        lines[:, off+i-j] += u2
        lines[:, off+j-i] += u2
    vals = lines.max(axis=1)
    idx = int(np.argmin(vals))
    if vals[idx] < float(c.max())-1e-10:

    return None

        dl = min(hard - 10.0, time.time() + 40.0)
        if dl < time.time() + 10.0: break
        ht, ct = optimize_at_resolution(hp, n_best, dl, rng, aggressive=False)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Phase 4: Final refinement with maximum effort
    if time.time() < hard - 15.0:
        n = len(best_h)
        # Final SLSQP with high alpha
        for alpha, mx in [(10000, 30), (50000, 25)]:
            if time.time() > hard - 8.0: break
            h_new = slsqp_opt(best_h, n, alpha, mx, hard)
            if np.any(h_new != best_h):
                best_h = h_new
        # Final aggressive pair search
        if n <= 700 and time.time() < hard - 3.0:
            rem = hard - time.time()
            mt = int(min(12000, max(1000, rem*40)))
            best_h, best_c = annealed_pair_search(best_h, n, mt, hard, rng, T0=0.002, decay=0.999)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
