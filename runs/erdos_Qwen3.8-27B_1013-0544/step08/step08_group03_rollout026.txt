    # Stabilize: shift so max(d) is at 0 to avoid overflow
    d_shifted = d - d.max()
    w = np.exp(beta * d_shifted)
    S = w.sum()
    
    # Gradient: for each j, grad[j] = -(dx/S) * sum_k w[k] * (h[j-m_k] + h[j+m_k])
    # where m_k = k - (n-1)
    # This equals -(dx/S) * 2 * (w * h)[j] (by symmetry of w and ac)
    # Actually let's just compute it as convolution
    
    # (w * h)[j] = sum_k w[k] * h[j-k] (full convolution)
    # We want sum_k w[k] * h[j - (k-(n-1))] = sum_k w[k] * h[j - k + n - 1]
    # Let l = k - (n-1), so k = l + (n-1), l ranges from -(n-1) to (n-1)
    # sum_l w[l+(n-1)] * h[j-l]
    # This is the correlation of w_full with h, where w_full is indexed by l
    
    # Actually, let's use the fact that ac is symmetric, so w is symmetric.
    # sum_l w[l+(n-1)] * h[j-l] = convolution of w_full (reversed) with h
    # Since w_full is symmetric, reversed w_full = w_full
    # So it's just the convolution!
    
    conv = np.convolve(w, h, mode='full')  # length (2n-1) + n - 1 = 3n-2
    # conv[i] = sum_k w[k] * h[i-k]
    # We want sum_k w[k] * h[j-k+n-1] = conv[j+n-1]
    # Wait: conv[j+n-1] = sum_k w[k]*h[j+n-1-k] = sum_k w[k]*h[(j+n-1)-k]
    # Let m = k-(n-1), so k = m+(n-1):
    # = sum_m w[m+n-1]*h[j-m]
    # Yes! That's exactly what we want.
    
    grad = -(dx / S) * conv[n-1:n-1+n]  # length n
    
    return grad

        best_h, best_c = h1, c1

    # Phase 2: Coordinate descent polishing
    h2, c2 = coord_descend(best_h, n, 300, t0 + 450)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: More gradient descent with higher beta
    h3, c3 = smooth_gradient_descent(best_h, n, t0 + 600, n_iters=3000, beta_lo=50, beta_hi=2000)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Coordinate descent again
    h4, c4 = coord_descend(best_h, n, 400, t0 + 800)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Try different n values
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 80:
            break
        if n_try == n:
            continue
        h_int = interpolate_h(best_h, len(best_h), n_try)
        h_t, c_t = coord_descend(h_int, n_try, 200, deadline - 40)
        if c_t < best_c:
            best_h, best_c = h_t, c_t

    # Phase 6: Random restarts on best n
    n_best = len(best_h)
    attempts = 3
    for r in range(attempts):
        if time.time() > deadline - 30:
            break
        if r == 0:
            hp = project_feasible(best_h + rng.normal(0, 0.02, n_best), n_best)
        else:
            hp = make_param_init(n_best, r)
        hp, cp = smooth_gradient_descent(hp, n_best, deadline - 20, n_iters=1500, beta_lo=10, beta_hi=500)
        hp, cp = coord_descend(hp, n_best, 150, deadline - 5)
        if cp < best_c:
            best_h, best_c = hp, cp

    # Phase 7: Final ultra-fine coordinate descent
    if time.time() < deadline - 3:
        hf, cf = coord_descend(best_h, len(best_h), 500, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
