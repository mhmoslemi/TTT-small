            zero_m = ~(pos_m | neg_m)
            zero_max = float(np.max(base_lin[zero_m])) if zero_m.any() else -1e100
            for _ in range(3):
                ok, _, _ = is_feasible_pre(Lhi, pos_inv, pos_a, neg_inv, neg_a, zero_max, c0c)
                if ok:
                    break
                Lhi = max(Lhi + 1e-6, float(np.max(c)) + 1e-6)
                if Llo > Lhi:
                    Lhi = Llo
            ok, _, _ = is_feasible_pre(Llo, pos_inv, pos_a, neg_inv, neg_a, zero_max, c0c)
            if ok:
                Lmin = Llo
            else:
                lo, hi = Llo, Lhi
                for _ in range(BIS):
                    mid = 0.5 * (lo + hi)
                    ok, _, _ = is_feasible_pre(mid, pos_inv, pos_a, neg_inv, neg_a, zero_max, c0c)
                    if ok:
                        hi = mid
                    else:
                        lo = mid
                Lmin = hi
            rem = max(1, n - processed)
            desired = h0 + (target - S) / rem
            desired = max(0.0, min(1.0, desired))
            x = choose_x_pre(Lmin, pos_inv, pos_a, neg_inv, neg_a, zero_max, c0c, desired, h0)
            diff = x - h0
            if abs(diff) > 1e-13:
                d[R] = 0.0
                c[start:end + 1] += diff * d
                c[n - 1] += x * (1.0 - x) - h0 * (1.0 - h0)
                h[p] = x
                S += diff
                aff_max = float(np.max(c[start:end + 1]))
                cur_max = max(unaff, aff_max)
                improved = True
            processed += 1
        c = corr_rev(h)
        cur_max = float(c.max())
        if abs(float(h.sum()) - target) > 1e-9:
            h = project_feasible(h, n)
            c = corr_rev(h)
            cur_max = float(c.max())
        cur_c5 = cur_max * dx
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
        if not improved:
            break
    curr_h = project_feasible(h, n)
    curr_c5 = compute_c5(curr_h)
    if curr_c5 < best_c5:
        best_h, best_c5 = curr_h, curr_c5
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def grad_from_weights(w, h, n):
    mid = n - 1
    s = np.zeros(n)
    if n > 1:
        s[1:] = w[mid + 1:] + w[:mid][::-1]
    left = np.convolve(h, s, mode='full')[:n]
    right = np.correlate(h, s, mode='full')[mid:mid + n]
    lc = np.zeros(n)
    rc = np.zeros(n)
    if n > 1:
        lc[1:] = np.cumsum(w[:mid][::-1])
        rc[:n - 1] = np.cumsum(w[mid + 1:])[::-1]
    return w[mid] * (1.0 - 2.0 * h) + lc + rc - left - right


def gradient_descent(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.08
    for act_tol in (15.0, 5.0, 1.0, 0.3):
        fail = 0
        for _ in range(90):
            if time.time() >= deadline:
                break
            c = corr_rev(h)
            M = float(c.max())
            act = c >= M - act_tol
            if not act.any():
                act = (c >= M - 1e-6)
            if not act.any():
                act = (c == M)
            w = act.astype(float)
            ws = float(w.sum())
            if ws <= 0.0:
                break
            w /= ws
            g = grad_from_weights(w, h, n)
            gmax = float(np.max(np.abs(g)))
            if gmax < 1e-14:
                break
            g /= gmax
            accepted = False
            s = step
            M2 = M
            for _ in range(6):
                cand = project_feasible(h - s * g, n)
                c2 = corr_rev(cand)
                M2 = float(c2.max())
                if M2 < M - 1e-9:
                    h = cand
                    step = min(0.25, max(step * 1.05, s * 1.1))
                    accepted = True
                    break
                s *= 0.5
            if accepted:
                fail = 0
                cur = M2 * 2.0 / n
                if cur < best_c5 - 1e-13:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                fail += 1
                step *= 0.8
                if fail > 8:
                    break
        if time.time() >= deadline:
            break
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    if n % 2:
        h[idx[n // 2]] = 0.5
    return project_feasible(h, n)


def random_uniform(n, rng):
    return project_feasible(rng.uniform(0.0, 1.0, n), n)


def noisy_copy(h, rng, sigma):
    return project_feasible(np.asarray(h, dtype=float) + rng.normal(0.0, sigma, len(h)), len(h))


def interp_h(hold, n_new):
    x_old = np.linspace(0.0, 2.0, len(hold))
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(x_new, x_old, hold), n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        n_parent = 700
        h_parent = np.full(n_parent, 0.5)
    best_h = project_feasible(h_parent, n_parent)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    d1 = min(deadline, t0 + min(450.0, 0.55 * float(budget_s)))
    h1, c1 = coordinate_descent(best_h, best_n, d1, rng)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, len(h1)

    t = time.time()
    if t < deadline - 80:
        h2 = noisy_copy(best_h, rng, 0.04)
        c2 = compute_c5(h2)
        if c2 <= best_c5 + 0.05:
            lim = min(deadline, t + 90.0)
            h2, c2 = coordinate_descent(h2, len(h2), lim, rng)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2, c2, len(h2)

    t = time.time()
    if t < deadline - 150:
        h2 = random_binary(best_n, rng)
        c2 = compute_c5(h2)
        if c2 <= best_c5 + 0.25:
            lim1 = min(deadline, t + 35.0)
            h2, c2g = gradient_descent(h2, best_n, lim1, rng)
            if c2g < c2:
                h2, c2 = h2, c2g
            lim2 = min(deadline, t + 140.0)
            h2, c2 = coordinate_descent(h2, best_n, lim2, rng)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2, c2, len(h2)

    t = time.time()
    if t < deadline - 150:
        h2 = random_uniform(best_n, rng)
        c2 = compute_c5(h2)
        if c2 <= best_c5 + 0.25:
            lim1 = min(deadline, t + 25.0)
            h2, c2g = gradient_descent(h2, best_n, lim1, rng)
            if c2g < c2:
                h2, c2 = h2, c2g
            lim2 = min(deadline, t + 120.0)
            h2, c2 = coordinate_descent(h2, best_n, lim2, rng)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2, c2, len(h2)

    t = time.time()
    if t < deadline - 220:
        n2 = best_n + 200
        n2 = min(n2, 1200)
        if n2 % 2:
            n2 += 1
        h2 = interp_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 <= best_c5 + 0.03:
            lim = min(deadline, t + 160.0)
            h2, c2 = coordinate_descent(h2, n2, lim, rng)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2, c2, len(h2)

    if time.time() < deadline:
        h2, c2 = coordinate_descent(best_h, len(best_h), deadline, rng)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, len(h2)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
