import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        if tot > 0: h += (diff / tot) * cap
    elif diff < 0:
        tot = float(h.sum())
        if tot > 0: h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size: h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size: h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lsp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = z.max()
    return float((zm + np.log(np.sum(np.exp(z - zm)))) / alpha * (2.0 / n))

def lsp_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * (c - c.max())
    w = np.exp(z); w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15: continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n-s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n-r])
            g[:n-r] -= wm * x[r:]
    return (2.0 / n) * g

def _sum_eq(x, n): return float(x.sum()) - 0.5 * n
def _sum_jac(x, n): return np.ones(n)

def slsqp_opt(h, n, deadline, alphas=(30, 300, 2000, 15000), maxiter=100):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline: break
        try:
            r = minimize(lsp_obj, best_h, args=(n, a), method='SLSQP',
                         jac=lsp_jac, bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(r.x, n)
            c = compute_c(x, n)
            if c < best_c: best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def _cd_eval(x, hp, cseg, a, qconst, unaff):
    if a.size:
        vmax = float(np.max(cseg + a * (x - hp)))
    else:
        vmax = 0.0
    return max(unaff, qconst + x * (1.0 - x), vmax)

def _cd_vec_eval(xg, hp, cseg, a, qconst, unaff):
    d = xg - hp
    vmax = np.max(cseg[None, :] + d[:, None] * a[None, :], axis=1) if a.size else np.zeros_like(xg)
    return np.maximum(np.maximum(vmax, qconst + xg * (1.0 - xg)), unaff)

def optimize_coord(hp, cseg, a, qconst, unaff, iters=20):
    tol = 1e-10
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12; neg = a < -1e-12; zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    cm = float(np.max(cseg)) if m else 0.0
    qcur = qconst + hp * (1.0 - hp)
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-11, unaff, qconst, qcur, cm) + 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    pc, nc = a_pos.size, a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= qconst - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll: ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh: lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * max(0,g)))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * max(0,g)))
                    if max(0.0, ll) <= min(r1, lh) + tol: feas = True
                    else: feas = max(r2, ll) <= min(1.0, lh) + tol
        if feas: high = mid
        else: low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - tol and t >= qconst - tol and t >= zmax - tol:
        ll, lh = 0.0, 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll: ll = v
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh: lh = v
        if ll <= 1.0 + tol and lh >= -tol:
            g = max(0.0, t - qconst)
            if g > 0.25:
                branches = [(max(0.0, ll), min(1.0, lh))]
            else:
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                branches = [(0.0, r1), (r2, 1.0)]
            for Lq, Uq in branches:
                L, U = max(Lq, ll, 0.0), min(Uq, lh, 1.0)
                if L <= U + tol: cands.extend([L, U, 0.5*(L+U)])
    best_x, best_f = hp, _cd_eval(hp, hp, cseg, a, qconst, unaff)
    for x in cands:
        if 0.0 <= x <= 1.0:
            f = _cd_eval(x, hp, cseg, a, qconst, unaff)
            if f < best_f - 1e-12: best_f, best_x = f, x
    if best_f > high + 1e-4 and m:
        xg = np.linspace(0.0, 1.0, 40)
        vals = _cd_vec_eval(xg, hp, cseg, a, qconst, unaff)
        idx = int(np.argmin(vals))
        x = float(xg[idx])
        if vals[idx] < best_f:
            l, r = max(0.0, x-0.04), min(1.0, x+0.04)
            if r <= l: l, r = 0.0, 1.0
            xg2 = np.linspace(l, r, 40)
            vals2 = _cd_vec_eval(xg2, hp, cseg, a, qconst, unaff)
            idx2 = int(np.argmin(vals2))
            best_x, best_f = float(xg2[idx2]), float(vals2[idx2])
        else:
            best_x, best_f = x, float(vals[idx])
    return best_x, best_f

def cd_descent(h, n, max_passes, deadline, project_every=8):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0/n)*cur
    center = n - 1
    L = 2*n - 1
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            if time.time() > deadline: break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2*center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi+1, dtype=int)
            j, i = p + s, p - s
            a = np.zeros(seg, dtype=float)
            vj = (j >= 0) & (j < n); vi = (i >= 0) & (i < n)
            if np.any(vj): a[vj] += 1.0 - h[j[vj]]
            if np.any(vi): a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg-1, dtype=float)
            asg = np.empty(seg-1, dtype=float)
            if off: cseg[:off], asg[:off] = c[lo:center], a[:off]
            if hi > center: cseg[off:], asg[off:] = c[center+1:hi+1], a[off+1:]
            qconst = c[center] - hp*(1.0-hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi+1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff, 20)
            if bf < cur - 1e-10:
                dxv = bx - hp
                if off: c[lo:center] += asg[:off]*dxv
                if hi > center: c[center+1:hi+1] += asg[off:]*dxv
                c[center] = qconst + bx*(1.0-bx)
                h[p] = bx; cur = bf; improved = True
            if (p+1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0-h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full')
        cur = float(c.max())
        cval = (2.0/n)*cur
        if cval < best_c - 1e-13: best_c, best_h = cval, h.copy()
        if not improved: break
    return best_h, best_c

def binary_search(h, n, deadline, max_rounds=80, rng=None):
    if rng is None: rng = np.random.RandomState(77)
    hb = np.where(h >= 0.5, 1.0, 0.0)
    project_feasible(hb, n)
    c = np.correlate(hb, 1.0-hb, mode='full')
    cur = float(c.max())
    best = cur * 2.0/n
    for rnd in range(max_rounds):
        if time.time() > deadline: break
        improved = False
        worst = int(np.argmax(c))
        s = (n-1) - worst
        if s >= 0:
            bad_ones = np.where(hb[:n-s] == 1.0)[0]
            bad_zeros = np.where(hb[s:] == 0.0)[0]
        else:
            bad_ones = np.where(hb[-s:] == 1.0)[0]
            bad_zeros = np.where(hb[:n+s] == 0.0)[0]
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        n_trials = min(2000, max(200, 300 // (rnd+1)))
        for _ in range(n_trials):
            if time.time() > deadline: break
            if bad_ones.size and bad_zeros.size and rng.rand() < 0.6:
                i = int(bad_ones[rng.randint(bad_ones.size)])
                j = int(bad_zeros[rng.randint(bad_zeros.size)])
                if s >= 0: j += s
                else: j = j
            else:
                i = int(ones[rng.randint(ones.size)])
                j = int(zeros[rng.randint(zeros.size)])
            hb[i], hb[j] = 0.0, 1.0
            c_new = np.correlate(hb, 1.0-hb, mode='full')
            new_max = float(c_new.max())
            hb[i], hb[j] = 1.0, 0.0
            if new_max < cur - 1e-12:
                hb[i], hb[j] = 0.0, 1.0
                cur = new_max; improved = True
                ones = np.where(hb == 1.0)[0]
                zeros = np.where(hb == 0.0)[0]
                bad_ones = np.where(hb[:n-s] == 1.0)[0] if s >= 0 else np.where(hb[-s:] == 1.0)[0]
                bad_zeros = np.where(hb[s:] == 0.0)[0] if s >= 0 else np.where(hb[:n+s] == 0.0)[0]
                if cur*2.0/n < best: best = cur*2.0/n
        if not improved: break
    return hb, best

def resample_h(old, n):
    n0 = len(old)
    if n0 == n: return project_feasible(old, n)
    x0 = (np.arange(n0)+0.5)*(2.0/n0)
    x1 = (np.arange(n)+0.5)*(2.0/n)
    return project_feasible(np.interp(x1, x0, old), n)
