def binary_local_search(h, n, max_iters, deadline):
    """Local search on binary h via swaps."""
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Fix integral
    diff = n // 2 - int(h.sum())
    if diff > 0:
        idx0 = np.where(h == 0)[0]
        for i in range(diff):
            h[idx0[i]] = 1.0
    elif diff < 0:
        idx1 = np.where(h == 1)[0]
        for i in range(-diff):
            h[idx1[i]] = 0.0
    
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_c_val = cur_max * (2.0 / n)
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        # Try random swaps
        for trial in range(200):
            if time.time() > deadline:
                break
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            # Compute new correlation
            h[p] = 0.0
            h[q] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(c_new.max())
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                c = c_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
            else:
                h[p] = 1.0
                h[q] = 0.0
        if cur_max * (2.0/n) < best_c_val - 1e-13:
            best_c_val = cur_max * (2.0/n)
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c_val
