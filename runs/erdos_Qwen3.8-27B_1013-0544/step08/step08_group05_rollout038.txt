import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += d / mask.sum()
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    fh = 1.0 - x
    xrev = x[::-1]
    grad = np.zeros(n)
    for p in range(n):
        s = n - 1 - p
        if s >= 0:
            grad[p] += np.dot(fh[s:], w[:n - s])
        else:
            grad[p] += np.dot(fh, w[-s:2 * n - 1 - s])
        grad[p] -= np.dot(xrev, w[p:p + n])
    return (m + np.log(z) / alpha) * dx, grad * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-14})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc

def coord_cd(h, n, max_passes=8, deadline=None, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        improved = False
        for p in range(n):
            old = float(x[p])
            A = np.zeros(L)
            A[n - 1 - p: n - 1 - p + n] += (1.0 - x)
            A[p: p + n] -= x[::-1]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-14:
                continue
            for _ in range(35):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = max((c + m1 * A).max(), c[diag] + m1 * A[diag] - m1 * m1)
                f2 = max((c + m2 * A).max(), c[diag] + m2 * A[diag] - m2 * m2)
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            d_opt = 0.5 * (lo_d + hi_d)
            bx = max(0.0, min(1.0, old + d_opt))
            f_opt = max((c + d_opt * A).max(), c[diag] + d_opt * A[diag] - d_opt * d_opt)
            cur = float(c.max())
            if f_opt < cur - tol:
                c += d_opt * A
                c[diag] -= d_opt * d_opt
                x[p] = bx
                improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best = xp
            bc = cp

        return b

    if time.time() < tf1:
        h1, c1 = slsqp_polish(best_h, n_best, [100000, 1000000, 5000000], 150, tf1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    tf2 = min(deadline - 5.0, time.time() + 80.0)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=10, deadline=tf2)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline, t0=0.01, t1=0.0001, max_trials=100000)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
