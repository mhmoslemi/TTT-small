    h = project_simplex(h, n).copy()
    v = np.zeros(n)  # momentum
    lr = 0.1  # initial learning rate
    bc = compute_c5(h, n)
    stagnant = 0
    
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode="full")
        cm = float(c.max()) * (2.0 / n)
        if cm < bc - 1e-10:
            bc = cm
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:

                if lr < 1e-6:
                    break
        
        # Get sub-gradient
        grad = active_subgrad(c, h, n, top_k)
        
        # Momentum step
        v = momentum * v - lr * grad

        # Check if it improved

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-14:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def smooth_grad(h, n, alpha):

    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)

        else:
            dd = -d
            gk[dd:] -= h[:n-dd]
            gk[:n-dd] += 1.0 - h[dd:]
        grad += w[k] * gk
    return grad * dx

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        r = alpha * (c - m)
        r -= r.max()
        return (m + np.log(r.exp().sum()) / alpha) * (2.0 / n)
    def cons_fun(x):
        return x.sum() - 0.5 * n
    def cons_jac(x):
        return np.ones_like(x)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=smooth_grad, args=(n, alpha),
                       method="SLSQP", bounds=bounds,
                       constraints=[{"type": "eq", "fun": cons_fun, "jac": cons_jac}],
                       options={"maxiter": maxiter, "ftol": 1e-14})
        return project_simplex(res.x, n), compute_c5(project_simplex(res.x, n), n)
    except Exception:
        return h, compute_c5(h, n)

            if 0 <= b_idx < n:
                delta += b[b_idx]
            # Add contribution of j (was 0, now 1)
            a_idx2 = j + k - diag
            b_idx2 = k - j + diag
            if 0 <= a_idx2 < n:
                delta += (1.0 - b[a_idx2])
            if 0 <= b_idx2 < n:
                delta -= b[b_idx2]
            # Self-interaction at diag
            if k == diag:
                delta -= 1.0  # (1-0) was 0 for i, now (1-1)=0... 
                # Actually: i was 1, (1-b[i])=0. Now b[i]=0, (1-b[i])=1.
                # The self-term at k=diag: h[i]*(1-h[i]) was 1*0=0, now 0*1=0. No change.
                # But j: was 0, h[j]*(1-h[j])=0. Now 1*(1-1)=0. No change.
                # Wait, the -1 is wrong. Let me remove it.
                delta += 1.0  # cancel the -1 above, net 0
            if i == j:
                pass
            c_new[k] += delta
        # Handle interaction: if a_idx == j or b_idx == j, the b value changed
        # This is complex; for simplicity, just recompute if the change is small
        nm = float(c_new.max())
        accept = nm < bc - 1e-9 or (nm < bc + 1e-9 and rng.rand() < np.exp(-(nm - bc) / max(T, 1e-4)))
        if accept and nm < bc + T * 2:
            b[i] = 0.0
            b[j] = 1.0
            c = c_new
            if nm < bc:
                bc = nm
                best = b.copy()
                stagnant = 0
            else:
                stagnant += 1
            T *= 0.999
        else:
            stagnant += 1
        if stagnant > 20000:
            break
    return best, bc * dx

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    n_best = n0

    # Symmetry improvements
    rev = h0[::-1]
    for v in [0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))]:
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    # Phase 1: Quick SLSQP polish on initial
    t1 = min(deadline - 300.0, t0 + 80.0)
    if time.time() < t1:
        h1, c1 = slsqp_step(best_h, n_best, 5000, 60, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=8, deadline=t1, grid=64)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Multi-resolution exploration
    n_candidates = [n_best]
    for dn in [-2, 2, -4, 4, -10, 10, -20, 20, -30, 30, 50, -50]:
        nt = n_best + dn
        if 200 <= nt <= 800 and nt % 2 == 0:
            n_candidates.append(nt)

    for nt in n_candidates:
        if time.time() > deadline - 150.0:
            break
        if nt == n_best:
            continue
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        t_chunk = min(deadline - 120.0, time.time() + 60.0)
        if time.time() < t_chunk:
            h1, c1 = slsqp_step(ht, nt, 5000, 40, t_chunk)
            h1, c1 = coord_cd(h1, nt, max_passes=3, deadline=t_chunk, grid=48)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, nt

    # Phase 3: Binary SA optimization
    if n_best % 2 == 0:
        t_bin = min(deadline - 100.0, time.time() + 200.0)
        hb, cb = binary_sa(best_h, n_best, rng, t_bin)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
            # Polish after binary
            t_pol = min(deadline - 60.0, time.time() + 60.0)
            h1, c1 = slsqp_step(best_h, n_best, 10000, 50, t_pol)
            h1, c1 = coord_cd(h1, n_best, max_passes=4, deadline=t_pol, grid=64)
            if c1 < best_c:
                best_h, best_c = h1, c1

    # Phase 4: More perturbation attempts
    for attempt in range(5):
        if time.time() > deadline - 80.0:
            break
        # Perturb: random sinusoidal
        freq = rng.randint(3, 20)
        phase = rng.rand() * 2 * np.pi
        amp = rng.uniform(0.05, 0.15)
        t_arr = np.linspace(0, 2, n_best)
        pert = best_h + amp * np.sin(freq * np.pi * t_arr + phase)
        pert = project_simplex(pert, n_best)
        t_ram = min(deadline - 50.0, time.time() + 40.0)
        h1, c1 = slsqp_step(pert, n_best, 5000, 30, t_ram)
        h1, c1 = coord_cd(h1, n_best, max_passes=3, deadline=t_ram, grid=48)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 5: Final deep polish
    t_final = deadline - 10.0
    if time.time() < t_final:
        h1, c1 = slsqp_step(best_h, n_best, 20000, 100, t_final)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=6, deadline=t_final, grid=80)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Final binary check
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline - 2.0)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = compute_c5(final_h, n_best)
    return final_h, final_c, int(n_best)
