```python
import numpy as np
from scipy.optimize import minimize, linear_sum_assignment
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def bottleneck_lags(h, n, top_k=10):
    """Find the lags that give the highest overlap."""
    c = np.correlate(h, 1.0 - h, mode='full')
    return c  # return full correlation, caller can pick top

def coord_cd_focused(h, n, max_passes=5000, deadline=None, grid_size=80):
    """Coordinate descent focusing on reducing the maximum overlap."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            lo_d = -h0 if h0 > 0 else 0.0
            hi_d = 1.0 - h0 if h0 < 1 else 1.0
            if hi_d - lo_d < 1e-12:
                continue
            
            # Compute affected correlation indices
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            
            # Linear coefficient for the change
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            
            # Grid search for best delta
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            
            # Parabolic refinement
            if 0 < best_i < grid_size - 1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom

                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, xv
            
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True

            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())

def two_interval_h(n, a1, L1, a2):
    """Create h as two intervals [a1, a1+L1] and [a2, a2+L2] where L2=1-L1."""
    L2 = 1.0 - L1
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # cell centers
    h[(x >= a1) & (x < a1 + L1)] = 1.0
    h[(x >= a2) & (x < a2 + L2)] = 1.0
    return project_feasible(h, n)

    best_h = None
    best_c = 1.0
    rng = np.random.RandomState(123)

            for a2 in np.linspace(a2_min, a2_max, 9):

            improved = False
            for scale in [0.05, 0.02, 0.01]:
```

First sum for position j: Σ_i A[i] * W[n-1+i-j] 
This is the cross-correlation of A and W at offset (n-1-j):

```python
def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    ac = alpha * c
    ac_shift = ac - ac.max()
    w = np.exp(ac_shift)
    w /= w.sum()
    
    # grad[j] = sum_k w[n-1+k] * (1-x[j+k]) * I(0<=j+k<n) - sum_k w[n-1+k] * x[j-k] * I(0<=j-k<n)
    # = sum_k w[n-1+k] * (1 - 2*x[j+k] + x[j+k] - x[j-k])... no that's wrong
    
    # Let me compute each term separately using vectorized operations
    # For each j, k ranges from max(0, -(j)) to min(n-1-j, ...) 
    # Actually k ranges such that both j+k and j-k are valid, or one of them is invalid
    
    # Simpler: for each valid k (0 to 2n-2), the contribution to grad[j] is:
    # w[k] * [I(j + (k-n+1) valid) * (1 - x[j + (k-n+1)]) - I(j - (k-n+1) valid) * x[j - (k-n+1)]]
    # where "valid" means 0 <= index < n
    
    # Let offset = k - (n-1) = lag
    # grad[j] = sum_offset w[n-1+offset] * [(1-x[j+offset])*I(0<=j+offset<n) - x[j-offset]*I(0<=j-offset<n)]
    
    grad = np.zeros(n)
    for offset in range(-(n-1), n):
        ko = n - 1 + offset  # index into w
        wo = w[ko]
        if wo < 1e-15:
            continue
        # First part: (1-x[j+offset]) for j+offset in [0,n)
        # j + offset in [0, n) → j in [-offset, n-offset)
        if offset >= 0:
            j_start, j_end = 0, n - offset
        else:
            j_start, j_end = -offset, n
        if j_end > j_start:
            grad[j_start:j_end] += wo * (1.0 - x[j_start+offset:j_end+offset])
        
        # Second part: -x[j-offset] for j-offset in [0,n)
        # j - offset in [0, n) → j in [offset, n+offset)
        if offset >= 0:
            j_start2, j_end2 = offset, n
        else:
            j_start2, j_end2 = 0, n + offset
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= wo * x[j_start2-offset:j_end2-offset]
    
    return grad * (2.0 / n)
```

This is O(n²) but with small constants. For n=500, it's about 500*1000 = 500,000 operations per gradient call, which should take about 1-5ms in Python (slower than numpy vectorized but still OK).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, max_passes=5000, deadline=None, grid_size=80, 
             anneal=False, T0=0.001, T_end=1e-6, swap_prob=0.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(42)
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (p / max(1, max_passes)) if anneal else 0.0
        
        idx_order = range(n) if p % 2 == 0 else range(n-1, -1, -1)
        improved = False
        
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            
            # Occasionally do a swap instead of a single-coordinate update
            if swap_prob > 0 and rng.random() < swap_prob:
                # Swap h[idx] with h[rng.randint(n)]
                j = rng.randint(n)
                if j == idx:
                    continue
                h[idx], h[j] = h[j], h[idx]
                c = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(c.max())
                accept = new_c5 < current_c5 - 1e-14
                if not accept and T > 1e-8:
                    delta = new_c5 - current_c5
                    if delta < 0.005 and rng.random() < np.exp(-delta / T):
                        accept = True
                if accept:
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
                else:
                    h[idx], h[j] = h[j], h[idx]  # undo
                continue
            
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            lo_d = max(0.0, -h0)
            hi_d = min(1.0, 1.0 - h0)
            if hi_d - lo_d < 1e-12:
                continue
            
            m_lo = max(0, min(n-1-idx, idx))
            m_hi = min(2*n-2, max(2*n-2-idx, idx+n-1))
            ms = np.arange(m_lo, m_hi+1)
            M = len(ms)
            
            a = np.zeros(M)
            jh = idx + ms - (n-1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n-1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            diag_idx = np.where(ms == (n-1))[0]
            has_diag = len(diag_idx) > 0
            
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None,:] + d_grid[:,None] * a[None,:]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:,None]**2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            
            if 0 < best_i < grid_size-1:
                y = vals[best_i-1:best_i+2]
                x = d_grid[best_i-1:best_i+2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv*a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv**2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, xv
            
            accept = best_val < current_c5 - 1e-14
            if not accept and T > 1e-8:
                delta = best_val - current_c5
                if delta < 0.01 and rng.random() < np.exp(-delta/T):
                    accept = True
            
            if accept and abs(best_d) > 1e-10:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d**2
                new_max = max(unaff_max, float(c[ms].max()))
                current_c5 = new_max
                if new_max < best_c5:
                    best_c5 = new_max
                    best_h = h.copy()
                improved = True
        
        if not improved and not anneal:
            break
        if (p+1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    
    h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)
