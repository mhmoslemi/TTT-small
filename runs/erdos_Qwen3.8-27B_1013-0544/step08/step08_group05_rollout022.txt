import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best = nh
            bc = nc
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=int)
    b[idx] = 1
    return b

def sa_binary(b_init, n, rng, deadline, t_init=2.0, t_final=0.05, max_iters=800000):
    """Simulated annealing on binary sequences with one-zero swaps."""
    b = b_init.copy()
    c = np.correlate(b.astype(float), 1.0 - b.astype(float), mode="full")
    cm = float(c.max())
    best_b = b.copy()
    best_cm = cm
    best_c = c.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    n_ones = len(ones)
    L = 2 * n - 1
    it = 0
    n_total = max_iters
    while it < n_total and time.time() < deadline:
        T = t_init * (t_final / t_init) ** (it / float(n_total))
        i = ones[rng.randint(n_ones)]
        j = zeros[rng.randint(n_ones)]
        if i == j:
            it += 1
            continue
        d = i - j
        if d > 0:
            c[n-1-i : n-1-j] -= 1
            c[2*n-1-i : 2*n-1-j] += 1
        else:
            c[2*n-1-j : 2*n-1-i] -= 1
            c[n-1-j : n-1-i] += 1
        new_cm = float(c.max())
        delta = new_cm - cm
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
            cm = new_cm
            if d > 0:
                k1, k2 = i, j
            else:
                k1, k2 = j, i
            pos_in_ones = np.where(ones == i)[0]
            pos_in_zeros = np.where(zeros == j)[0]
            if len(pos_in_ones) == 1:
                ones[pos_in_ones[0]] = j
            if len(pos_in_zeros) == 1:
                zeros[pos_in_zeros[0]] = i
            b[i] = 0
            b[j] = 1
            if cm < best_cm:
                best_cm = cm
                best_b = b.copy()
                best_c = c.copy()
        else:
            if d > 0:
                c[n-1-i : n-1-j] += 1
                c[2*n-1-i : 2*n-1-j] -= 1
            else:
                c[2*n-1-j : 2*n-1-i] += 1
                c[n-1-j : n-1-i] -= 1
        it += 1
    return best_b, best_cm * (2.0 / n), best_c

def hill_climb_binary(b_init, n, max_passes=50, deadline=None):
    """Greedy local search with one-zero swaps."""
    b = b_init.copy().astype(int)
    c = np.correlate(b.astype(float), 1.0 - b.astype(float), mode="full")
    cm = float(c.max())
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _pass in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        rng_local = np.random.RandomState(12345)
        order = list(range(len(ones)))
        rng_local.shuffle(order)
        for idx in order:
            if deadline and time.time() > deadline:
                break
            i = ones[idx]
            for j in zeros[rng_local.randint(0, len(zeros), 20)]:
                if i == j:
                    continue
                d = i - j
                if d > 0:
                    c[n-1-i : n-1-j] -= 1
                    c[2*n-1-i : 2*n-1-j] += 1
                else:
                    c[2*n-1-j : 2*n-1-i] -= 1
                    c[n-1-j : n-1-i] += 1
                new_cm = float(c.max())
                if new_cm < cm:
                    cm = new_cm
                    b[i] = 0
                    b[j] = 1
                    ones[idx] = j
                    pos_j = np.where(zeros == j)[0]
                    if len(pos_j) == 1:
                        zeros[pos_j[0]] = i
                    improved = True
                    break
                else:
                    if d > 0:
                        c[n-1-i : n-1-j] += 1
                        c[2*n-1-i : 2*n-1-j] -= 1
                    else:
                        c[2*n-1-j : 2*n-1-i] += 1
                        c[n-1-j : n-1-i] -= 1
        if not improved:
            break
    return b, cm * (2.0 / n)

def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h.copy(), n_tgt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 3.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = float(compute_c5(h0, n0))
    n_best = int(n0)
    # Symmetrize
    rev = h0[::-1]
    for v in (0.5*(h0+rev), 0.5*(h0+(1.0-rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv
    # SA on binary at n0
    if n_best % 2 == 0 and time.time() < deadline - 200.0:
        tb = binarize(best_h, n_best)
        if tb is not None:
            t_sa1 = min(deadline - 200.0, t0 + 250.0)
            bb, bc_sa, _ = sa_binary(tb, n_best, rng, t_sa1, t_init=3.0, t_final=0.02, max_iters=1000000)
            bb2, bc_hc = hill_climb_binary(bb, n_best, max_passes=30, deadline=min(deadline-190.0, time.time()+30.0))
            if bc_hc < best_c:
                best_h = bb2.astype(float).copy()
                best_c = bc_hc
                n_best = int(n_best)
    # Try other n values
    n_list = [500, 498, 502, 496, 504, 494, 506]
    tried_ns = [n_best]
    for nt in n_list:
        if time.time() > deadline - 150.0:
            break
        if nt in tried_ns or nt % 2 or nt < 200 or nt > 600:
            continue
        tried_ns.append(nt)
        ht = transfer_h(best_h, n_best, nt)
        cb = compute_c5(ht, nt)
        tb = binarize(ht, nt)
        if tb is None:
            continue
        t_dead = min(deadline - 120.0, time.time() + 100.0)
        bb, bc_sa, _ = sa_binary(tb, nt, rng, t_dead, t_init=2.5, t_final=0.03, max_iters=600000)
        bb2, bc_hc = hill_climb_binary(bb, nt, max_passes=20, deadline=min(deadline-110.0, time.time()+20.0))
        if bc_hc < best_c:
            best_h = bb2.astype(float).copy()
            best_c = bc_hc
            n_best = int(nt)
        elif cb < best_c:
            best_h, best_c, n_best = ht, cb, int(nt)
    # Continuous refinement of best
    if time.time() < deadline - 60.0:
        t_ref = min(deadline - 50.0, time.time() + 120.0)
        h1, c1 = slsqp_polish(best_h, n_best, [5000, 50000, 500000], 100, t_ref)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Final binary attempt at best n
    if n_best % 2 == 0 and time.time() < deadline - 20.0:
        tb = binarize(best_h, n_best)
        if tb is not None:
            bb, bc_sa, _ = sa_binary(tb, n_best, rng, deadline - 5.0, t_init=1.5, t_final=0.01, max_iters=400000)
            bb2, bc_hc = hill_climb_binary(bb, n_best, max_passes=15, deadline=deadline - 3.0)
            if bc_hc < best_c:
                best_h = bb2.astype(float).copy()
                best_c = bc_hc
    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
