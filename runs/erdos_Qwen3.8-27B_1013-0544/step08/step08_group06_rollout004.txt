def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, step_size, deadline=None):
    """Take a projected subgradient descent step on C5."""
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    tol = 1e-6 * mx
    active = np.where(c >= mx - tol)[0]
    if len(active) == 0:
        return h
    grad = np.zeros(n)
    for m in active:
        for idx in range(n):
            k = m - (n - 1)
            j1 = idx + k
            j2 = idx - k
            g = 0.0
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            if 0 <= j2 < n:
                g -= h[j2]
            if k == 0:
                g -= 1.0
            grad[idx] += g
    grad /= len(active)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    if c_new < compute_c5(h, n):
        return h_new
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    # Phase 5: Try different resolutions
    for n_try in [300, 400, 600, 700]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = np.interp(x_new, x_orig, best_h)
        h_int = project_feasible(h_int, n_try)
        h_opt, c_opt = coord_descent(h_int, n_try, max_passes=2000, deadline=deadline - 100, grid_size=80)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h5, c5 = coord_descent(best_h, len(best_h), max_passes=5000, deadline=deadline, grid_size=120)
        if c5 < best_c:
            best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
