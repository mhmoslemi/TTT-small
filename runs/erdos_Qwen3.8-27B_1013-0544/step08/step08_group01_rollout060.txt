def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def weighted_gradient(h, lam, n):

```python
import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_from_float(h):
    """Convert to binary with exactly n/2 ones"""
    n = len(h)
    target = n // 2
    # Use the top n/2 values as 1s
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def solve_lp_for_b(b, n):
    """Solve the LP: min t s.t. A @ dh + c(b) <= t, sum(dh)=0, bounds on dh"""
    n_shifts = 2 * n - 1
    # Build A matrix: A[m, i] = (1 - 2*b[i+k]) if valid
    k_vals = np.arange(n_shifts) - (n - 1)
    A = np.zeros((n_shifts, n))
    for m, k in enumerate(k_vals):
        if k >= 0:
            valid_i = np.arange(0, n - k)
            A[m, valid_i] = 1.0 - 2.0 * b[valid_i + k]
        else:
            valid_i = np.arange(-k, n)
            A[m, valid_i] = 1.0 - 2.0 * b[valid_i + k]
    
    # c(b): correlation of b with 1-b
    c_b = np.correlate(b, 1.0 - b, mode='full')
    
    # LP: min t
    # subject to: A @ dh - t <= -c_b  (i.e., A @ dh + c_b <= t)
    #            sum(dh) = 0
    #            -b[i] <= dh[i] <= (1-b[i])
    
    # Variables: [dh (n), t (1)] total n+1
    # Objective: [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A @ dh - t <= -c_b
    # In standard form: [A, -1] @ [dh, t] <= -c_b
    A_ineq = np.hstack([A, -np.ones((n_shifts, 1))])
    b_ineq = -c_b
    
    # Equality: sum(dh) = 0
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = [0.0]
    
    # Bounds: -b[i] <= dh[i] <= (1-b[i]), t >= 0 (or t >= -inf)
    bounds = []
    for i in range(n):
        bounds.append((-b[i], 1.0 - b[i]))
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    dh = res.x[:n]
    t_val = res.x[-1]
    h = b + dh
    h = project_feasible(h, n)
    return h, t_val
```

import numpy as np
import time
from scipy.optimize import linprog, minimize
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_from_float(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def build_A(b, n):
    """Build matrix A where A[m,i] = (1-2*b[i+k]) if valid"""
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    k_vals = np.arange(n_shifts) - (n - 1)
    for m, k in enumerate(k_vals):
        if k >= 0:
            A[m, :n-k] = 1.0 - 2.0 * b[k:]
        else:
            A[m, -k:] = 1.0 - 2.0 * b[:n+k]
    return A

def solve_lp(b, n):
    """LP: min t s.t. A@dh + c_b <= t, sum(dh)=0, bounds"""
    n_shifts = 2 * n - 1
    A = build_A(b, n)
    c_b = np.correlate(b, 1.0 - b, mode='full')
    
    # Variables: [dh (n), t (1)]
    # min t
    # s.t. A@dh - t <= -c_b
    #      sum(dh) = 0
    #      -b[i] <= dh[i] <= 1-b[i], t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((n_shifts, 1))])
    b_ub = -c_b
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-b[i], 1.0 - b[i]) for i in range(n)] + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    h = b + res.x[:n]
    return project_feasible(h, n)

def binary_local_search(b, n, rng, max_swaps=200):
    """Try random swaps to improve binary pattern's C5"""
    b = b.copy()
    current_c = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for _ in range(max_swaps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = b[j], b[i]
        new_c = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
        if new_c > current_c + 1e-10:
            b[i], b[j] = b[j], b[i]
        else:
            current_c = new_c
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
    return b, current_c

            if best_v < cur_max - 1e-13:
                h[p] = float(np.clip(h0 + d_grid[best_i], 0.0, 1.0))
                improved = True
                cur_max = min(cur_max, best_v)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, nn, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / nn)

def sum_constraint(x, nn):
    return x.sum() - nn / 2.0

def sum_constraint_jac(x, nn):
    return np.ones(nn)

def slsqp_polish(h, n, alphas, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 600
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP
    if time.time() < deadline - 400:
        h1, c1 = slsqp_polish(best_h, n, [200, 2000, 20000], 300, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Binary + LP iteration
    b = binary_from_float(best_h)
    for it in range(400):
        if time.time() > deadline - 200:
            break
        h_lp = solve_lp_for_b(b, n)
        if h_lp is not None:
            c_lp = compute_c5(h_lp, n)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
        # Binary local search
        cur_bin_c = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        for _ in range(150):
            if time.time() > deadline - 195:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = b[j], b[i]
            new_bin_c = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
            if new_bin_c < cur_bin_c - 1e-12:
                cur_bin_c = new_bin_c
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
            else:
                b[i], b[j] = b[j], b[i]

    # Phase 3: Projected gradient focus
    if time.time() < deadline - 150:
        h3, c3 = projected_grad_focus(best_h, n, 300, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Coordinate descent
    if time.time() < deadline - 100:
        h4, c4 = coord_descent(best_h, n, 80, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Try other n
    for n_try in [450, 700, 800]:
        if time.time() > deadline - 80:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        b_new = binary_from_float(h_int)
        for it2 in range(80):
            if time.time() > deadline - 60:
                break
            h_lp2 = solve_lp_for_b(b_new, n_try)
            if h_lp2 is not None:
                c_lp2 = compute_c5(h_lp2, n_try)
                if c_lp2 < best_c:
                    best_h, best_c = h_lp2, c_lp2
            cur_c2 = float(np.max(np.correlate(b_new, 1.0 - b_new, mode='full')))
            ones2 = np.where(b_new == 1)[0]
            zeros2 = np.where(b_new == 0)[0]
            for _ in range(40):
                i2 = ones2[rng.randint(len(ones2))]
                j2 = zeros2[rng.randint(len(zeros2))]
                b_new[i2], b_new[j2] = b_new[j2], b_new[i2]
                new_c2 = float(np.max(np.correlate(b_new, 1.0 - b_new, mode='full')))
                if new_c2 < cur_c2 - 1e-12:
                    cur_c2 = new_c2
                    ones2 = np.where(b_new == 1)[0]
                    zeros2 = np.where(b_new == 0)[0]
                else:
                    b_new[i2], b_new[j2] = b_new[j2], b_new[i2]

    # Phase 6: Final SLSQP
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = slsqp_polish(best_h, n_cur, [50000, 200000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 7: Final coord descent
    if time.time() < deadline - 10:
        hf2, cf2 = coord_descent(best_h, n_cur, 200, deadline)
        if cf2 < best_c:
            best_h, best_c = hf2, cf2

    # Symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
