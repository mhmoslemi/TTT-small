            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full').copy()
    return h, compute_c5(h, n)

def obj_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_phase(h, n, alphas, maxiter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(len(x))}]
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def random_start(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def structured_start(n, rng):
    x = np.linspace(0, 2, n)
    freq = rng.uniform(1.0, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.3, 0.8)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP
    h1, c1 = slsqp_phase(h, n, [500, 2000, 10000, 50000, 200000], 500, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Long exact CD with incremental updates
    h2, c2 = cd_exact_fast(best_h, n, 2000, deadline - 150)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 900, 1000, 600]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_phase(h_int, n_try, [2000, 20000, 100000], 400, deadline)
        h_o, c_o = cd_exact_fast(h_o, n_try, 500, deadline)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)

    # Phase 4: Multi-restart
    n = len(best_h)
    if time.time() < deadline - 80:
        n_restart = max(2, int((deadline - time.time() - 60) / 60))
        for r in range(n_restart):
            if time.time() > deadline - 50:
                break
            if r % 4 == 1:
                hp = random_start(n, rng)
            elif r % 4 == 2:
                hp = structured_start(n, rng)
            else:
                s = 0.003 + 0.03 * r / max(1, n_restart)
                hp = project_feasible(best_h + rng.normal(0, s, n), n)
            h_o, c_o = slsqp_phase(hp, n, [5000, 30000], 300, deadline)
            h_o, c_o = cd_exact_fast(h_o, n, 300, deadline)
            if c_o < best_c:
                best_h, best_c = h_o, c2 if False else c_o
                best_h, best_c = h_o, c_o

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        h_f, c_f = cd_exact_fast(best_h, len(best_h), 5000, deadline)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
