        best_nm = cm
        best_i, best_j = -1, -1
        best_c = None
        for _ in range(batch_size):
            i = ones[int(rng.randint(len(ones)))]
            j = zeros[int(rng.randint(len(zeros)))]
            if i == j:
                continue
            c2, nm = binary_flip_trial(c, h, om, i, j, n)
            if nm < best_nm:
                best_nm, best_i, best_j, best_c = nm, i, j, c2
        if best_i >= 0:
            c = best_c
            cm = best_nm
            h[best_i], h[best_j] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            om = 1.0 - h[::-1]
            if cm < bcm - 1e-12:
                best = h.copy()
                bcm = cm
                stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                for _ in range(10):
                    i = ones[int(rng.randint(len(ones)))]
                    j = zeros[int(rng.randint(len(zeros)))]
                    if i != j:
                        h[i], h[j] = 0.0, 1.0
                        c = np.correlate(h, 1.0 - h, mode="full")
                        cm = float(c.max())
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        om = 1.0 - h[::-1]
                        break
        if stagnant > 200:
            break
    return best, bcm * dx


def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    n_best = n0

    # Symmetrization attempts
    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + 1.0 - rev)):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    # Binarize and quick check
    if n_best % 2 == 0:
        hb = binarize(h0, n_best)
        if hb is not None:
            cb = compute_c5(hb, n_best)
            if cb < best_c:
                best_h, best_c = hb, cb

    # Phase 1: SLSQP polish
    t1 = min(deadline - 400.0, t0 + 120.0)
    if time.time() < t1:
        h1, c1 = slsqp_polish(best_h, n_best, [500, 2000, 10000, 50000], 200, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_cd(best_h, n_best, max_passes=8, deadline=t1)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Binary search (main workhorse)
    if n_best % 2 == 0:
        t2 = min(deadline - 200.0, time.time() + 250.0)
        hb, cb = batch_binary(best_h, n_best, rng, t2, batch_size=300)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
        t2b = min(deadline - 150.0, time.time() + 200.0)
        if time.time() < t2b:
            hb2, cb2 = binary_search(best_h, n_best, rng, t2b)
            if hb2 is not None and cb2 < best_c:
                best_h, best_c = hb2, cb2

    # Phase 3: Try other resolutions
    cands = [400, 600, 550, 450, 700, 350]
    tried = {n_best}
    for nt in cands:
        if time.time() > deadline - 120.0:
            break
        if nt in tried or nt < 100 or nt > 1000 or nt % 2:
            continue
        tried.add(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt
        tc = min(deadline - 100.0, time.time() + 60.0)
        if time.time() < tc:
            h1, c1 = slsqp_polish(ht, nt, [5000, 20000], 100, tc)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, nt
            h1, c1 = coord_cd(best_h, n_best, max_passes=3, deadline=tc, grid=24, refine=8)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, int(nt)
            if nt % 2 == 0 and time.time() < deadline - 80.0:
                hb, cb = batch_binary(best_h, n_best, rng, min(deadline - 60.0, time.time() + 80.0), batch_size=200)
                if hb is not None and cb < best_c:
                    best_h, best_c, n_best = hb, cb, int(nt)

    # Phase 4: Final polish on best
    tf1 = min(deadline - 40.0, time.time() + 100.0)
    if time.time() < tf1:
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 200000], 150, tf1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    tf2 = min(deadline - 10.0, time.time() + 80.0)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=tf2, grid=40, refine=16)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best % 2 == 0 and time.time() < deadline - 15.0:
        hb, cb = batch_binary(best_h, n_best, rng, deadline - 5.0, batch_size=400)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = compute_c5(final_h, n_best)
    return final_h, final_c, int(n_best)
