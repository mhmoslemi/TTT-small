def compute_c5(h):

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5*n
    if z.size == 0: return z
    if z.min() >= -1e-12 and z.max() <= 1.0+1e-12 and abs(z.sum()-target) < 1e-8*max(1,n):
        return np.clip(z, 0.0, 1.0)
    lo, hi = float(z.min())-2.0, float(z.max())+2.0
    for _ in range(60):
        t = 0.5*(lo+hi)
        if np.clip(z-t, 0, 1).sum() > target: lo = t
        else: hi = t
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-13:
        m = (y < 1.0) if r > 0 else (y > 0.0)
        if m.any():
            rm = (1.0-y) if r > 0 else y
            s = rm[m].sum()
            if s > 1e-12: y[m] += (r/s)*rm[m]
    return np.clip(y, 0, 1)

def build_A(h):
    n = len(h)
    nc = 2*n-1
    off = n-1
    A = np.zeros((n, nc))
    A[:, off] = 1.0 - 2.0*h
    for d in range(1, n):
        col = off + d
        r1 = np.arange(n-d)
        A[r1, col] += 1.0 - h[d:]
        if d < n:
            r2 = np.arange(d, n)
            A[r2, col] -= h[:n-d]
        col2 = off - d
        r3 = np.arange(d, n)
        A[r3, col2] += 1.0 - h[:n-d]
        r4 = np.arange(n-d)
        A[r4, col2] -= h[d:]
    return A

def smooth_fn(h, n, alpha, mu):
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    z = alpha*(c-m) - alpha*(c-m).max()
    e = np.exp(z)
    val = (m + np.log(e.sum())/alpha) * (2.0/n)
    dev = h.sum() - 0.5*n
    return val + 0.5*mu*dev*dev

def smooth_jac(h, n, alpha, mu):
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    z = alpha*(c-m) - alpha*(c-m).max()
    p = np.exp(z); p /= p.sum()
    off = n-1
    g = np.zeros(n)
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-14: continue
        d = k - off
        if d > 0:
            g[:n-d] += w*(1.0-h[d:])
            g[d:] -= w*h[:n-d]
        elif d < 0:
            e = -d
            g[e:] += w*(1.0-h[:n-e])
            g[:n-e] -= w*h[e:]
        else:
            g += w*(1.0-2.0*h)
    dev = h.sum() - 0.5*n
    return (2.0/n)*g + mu*dev

def pair_search(h, n, deadline, rng, max_iter=200000):
    h = h.copy()
    off = n-1
    nc = 2*n-1
    c = np.correlate(h, 1-h, mode='full')
    A = build_A(h)
    
    accepts = 0
    stale = 0
    rebuild_counter = 0
    gr = (np.sqrt(5)-1)/2
    
    for trial in range(max_iter):
        if time.time() > deadline:
            break
        cur_max = c.max()
        
        # Select pair based on active lags
        tol = max(0.05, 0.003*cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(c))])
        
        if trial % 13 == 12 or active.size <= 1:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n-1)) % n
        else:
            kk = int(active[rng.randint(len(active))])
            d = kk - off
            if 0 <= d < n:
                b = rng.randint(n-d)
                i = b+d; j = b
            elif -n < d <= 0 and d != 0:
                e = -d
                b = e + rng.randint(n-e)
                i = b; j = b+e
            elif d == 0:
                i = rng.randint(n-1)
                j = i+1
            else:
                continue
        
        if i >= n or j >= n or i == j:
            continue

        B = A[i] - A[j]

        d1 = 0 <= k1 < nc
        d2 = 0 <= k2 < nc

            stale += 1

        a, b_g = lo_u, hi_u

            if d1: cn[k1] += u*u
            if d2: cn[k2] += u*u
