```python
def pair_grad_diff(h, i, j, n):
    """Compute ∂c/∂h[i] - ∂c/∂h[j] as a full vector."""
    off = n - 1
    B = np.zeros(2*n - 1)
    # For each k, B[k] = ∂c[k]/∂h[i] - ∂c[k]/∂h[j]
    # Using the formula:
    # For r = k - off >= 0: ∂c[k]/∂h[x] = (1-h[x+r])*[x+r<n] - h[x-r]*[x>=r]
    # For r < 0, e = -r: ∂c[k]/∂h[x] = (1-h[x-e])*[x>=e] - h[x+e]*[x+e<n]
    
    # Vectorized:
    r = np.arange(1, n)  # positive lags
    
    # For k = off + r (positive lag):
    # B[off+r] = (1-h[i+r])*[i+r<n] - h[i-r]*[i>=r] - (1-h[j+r])*[j+r<n] + h[j-r]*[j>=r]
    ip = i + r < n
    jm = r <= i
    jp = j + r < n
    jmn = r <= j
    
    B[off + r] = np.where(ip, 1.0 - h[i + r], 0.0) - np.where(jm, h[i - r], 0.0) \
               - np.where(jp, 1.0 - h[j + r], 0.0) + np.where(jmn, h[j - r], 0.0)
    
    # For k = off - r (negative lag, e = r):
    # B[off-r] = (1-h[i-r])*[i>=r] - h[i+r]*[i+r<n] - (1-h[j-r])*[j>=r] + h[j+r]*[j+r<n]
    B[off - r] = np.where(jm, 1.0 - h[i - r], 0.0) - np.where(ip, h[i + r], 0.0) \
               - np.where(jmn, 1.0 - h[j - r], 0.0) + np.where(jp, h[j + r], 0.0)
    
    # For k = off (zero lag):
    B[off] = (1.0 - 2.0*h[i]) - (1.0 - 2.0*h[j])
    
    return B
```

This is fully vectorized and O(n) time.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(50):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    p = e / e.sum()
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e2 = -d
            g[:n - e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n - e2]
    return (2.0 / n) * g

def slsqp_opt(h, n, deadline, scales, iters):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in zip(scales, iters):
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def pair_grad_diff(h, i, j, n):
    off = n - 1
    B = np.zeros(2 * n - 1, dtype=float)
    r = np.arange(1, n)
    ip = i + r < n
    jm = r <= i
    jp = j + r < n
    jmn = r <= j
    B[off + r] = np.where(ip, 1.0 - h[i + r], 0.0) - np.where(jm, h[i - r], 0.0) \
               - np.where(jp, 1.0 - h[j + r], 0.0) + np.where(jmn, h[j - r], 0.0)

    B[off] = 2.0 * (h[j] - h[i])
    return B
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(50):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    p = e / e.sum()
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w <= 1e-14:
            continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - x[:n - d])
            g[:n - d] -= w * x[d:]
        else:
            e2 = -d
            g[:n - e2] += w * (1.0 - x[e2:])
            g[e2:] -= w * x[:n - e2]
    return (2.0 / n) * g

def slsqp_opt(h, n, deadline, scales, iters):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    work_h = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha, mx in zip(scales, iters):
        if time.time() > deadline - 2.0:
            break
        try:
            res = minimize(smooth_val, work_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': mx, 'ftol': 1e-15, 'disp': False})
            x = np.asarray(res.x, dtype=float)
            if not np.all(np.isfinite(x)):
                continue
            cur = project_feasible(x, n)
            cc = compute_c5(cur, n)
            if cc < best_c:
                best_h, best_c = cur, cc
            work_h = cur if cc <= best_c + 1e-8 else best_h
        except Exception:
            pass
    return best_h, best_c

def pair_grad_diff(h, i, j, n):
    off = n - 1
    B = np.zeros(2 * n - 1, dtype=float)
    r = np.arange(1, n)
    ip = i + r < n
    jm = r <= i
    jp = j + r < n
    jmn = r <= j
    B[off + r] = np.where(ip, 1.0 - h[i + r], 0.0) - np.where(jm, h[i - r], 0.0) \
               - np.where(jp, 1.0 - h[j + r], 0.0) + np.where(jmn, h[j - r], 0.0)
    B[off - r] = np.where(jm, 1.0 - h[i - r], 0.0) - np.where(ip, h[i + r], 0.0) \
               - np.where(jmn, 1.0 - h[j - r], 0.0) + np.where(jp, h[j + r], 0.0)
    B[off] = 2.0 * (h[j] - h[i])
    return B

def pair_search(h, n, max_tries, deadline, rng, temp0=0.0, grid=25):
    if n < 2 or max_tries <= 0:
        return h, compute_c5(h, n)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    cur_max = float(c.max())
    best_h = h.copy()
    best_c = cur_max * 2.0 / n
    stagnation = 0
    max_stag = max(200, max_tries // 10)

    for trial in range(max_tries):
        if time.time() > deadline:
            break
        T = temp0 * (1.0 - float(trial) / max(1, max_tries)) ** 2 if temp0 > 0 else 0.0

        tol = max(0.005, 0.0008 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if active.size == 0:
            active = np.array([off])

        if trial % 9 == 8 or active.size <= 2:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j + 1) % n
                if i == j:
                    continue
        else:
            kk = int(active[int(rng.randint(active.size))])
            d = kk - off
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                b = int(rng.randint(m)); i = b + d; j = b
            else:
                e2 = -d
                m = n - e2
                if m <= 0:
                    continue
                b = e2 + int(rng.randint(m)); i = b - e2; j = b
        if i == j:
            continue

        hi = h[i] + h[j]
        lo = max(0.0, hi - 1.0)
        hiT = min(1.0, hi)
        if hiT - lo <= 1e-14:
            continue

        us = np.linspace(lo - h[i], hiT - h[i], grid)
        B = pair_grad_diff(h, i, j, n)
        u2 = us * us
        cn = c + us[:, None] * B[None, :]
        cn[:, off] -= 2.0 * u2
        cn[:, off + i - j] += u2
        cn[:, off + j - i] += u2
        maxes = cn.max(axis=1)
        idx = int(np.argmin(maxes))
        u_best = float(us[idx])
        new_max = float(maxes[idx])
        delta = new_max - cur_max

        accept = delta < -1e-12
        if not accept and T > 1e-12 and delta > 0:
            if np.log(rng.random()) > delta / max(T * cur_max, 1e-15):
                accept = True

        if accept:
            h[i] += u_best
            h[j] -= u_best
            c += u_best * B
            c[off] -= 2.0 * u_best * u_best
            c[off + i - j] += u_best * u_best
            c[off + j - i] += u_best * u_best
            cur_max = float(c.max())
            if new_max * 2.0 / n < best_c - 1e-12:
                best_h = h.copy()
                best_c = new_max * 2.0 / n
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        if stagnation > max_stag and T <= 1e-12:
            break

    best_c2 = compute_c5(best_h, n)
    if best_c2 < best_c:
        best_c = best_c2
    return best_h, best_c

def apply_symmetries(h, n):
    c = compute_c5(h, n)
    hs = project_feasible(0.5 * (h + h[::-1]), n)
    cs = compute_c5(hs, n)
    if cs < c:
        h, c = hs, cs
    hc = 1.0 - h
    cs = compute_c5(hc, n)
    if cs < c:
        h, c = hc, cs
    hsc = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
    cs = compute_c5(hsc, n)
    if cs < c:
        h, c = hsc, cs
    return h, c

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def run(seed=42, budget_s=1100, **kwargs):
```
