def binary_local_search(h, n, max_iter, rng, deadline):
    h = h.copy()
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    cur_max = c.max()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a random one and zero to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        # h[i]: 1->0, h[j]: 0->1
        # c[k] changes by:
        # For each k, the positions where h[i] or h[j] contributes
        # c[k] = sum_p h[p] * (1-h[p-d_k])
        # Change: -h[i]*(1-h[i-d_k]) + (1-h[j])*(-h[j-d_k]) ... 
        # Actually let's just recompute the affected lags
        # h[i] appears in c[k] for lags where i is in the valid range
        # This is complex; let's just recompute c from scratch for simplicity
        # (n=500, correlate is fast)
        
        h[i] = 0; h[j] = 1
        c_new = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
        new_max = c_new.max()

            c = c_new
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
