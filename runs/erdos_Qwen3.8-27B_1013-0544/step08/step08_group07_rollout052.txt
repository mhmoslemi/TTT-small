def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    # Log-sum-exp for numerical stability
    shifted = alpha * (c - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    # Gradient
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    # dc[m']/dx[p] for each m' weighted by w[m']
    # = sum_{m'} w[m'] * [(1-x[p-m'+n-1])*I - x[p+m'-n+1]*I]
    g = np.zeros(n)
    nm1 = n - 1
    for m_idx in range(len(c)):
        wv = w[m_idx]
        if wv < 1e-10:
            continue
        shift = m_idx - nm1
        if shift == 0:
            g += wv * (1.0 - 2.0 * x)
        else:
            if shift > 0:
                g[shift:] += wv * (1.0 - x[:n-shift])
                g[:n-shift] -= wv * x[shift:]
            else:
                s = -shift
                g[:n-s] += wv * (1.0 - x[s:])
                g[s:] -= wv * x[:n-s]
    dx = 2.0 / n
    return lse * dx, g * dx

def lbfgs_run(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda n=n: np.ones(n)}]
        try:
            res = minimize(obj_and_grad, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def obj_and_grad(x, n, alpha, wthresh=1e-10):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    lse = m + np.log(exp_shifted.sum()) / alpha
    g = np.zeros(n)
    nm1 = n - 1
    for m_idx in range(2 * n - 1):
        wv = w[m_idx]
        if wv < wthresh:
            continue
        shift = m_idx - nm1
        if shift == 0:
            g += wv * (1.0 - 2.0 * x)
        elif shift > 0:
            g[shift:] += wv * (1.0 - x[:n - shift])
            g[:n - shift] -= wv * x[shift:]
        else:
            s = -shift
            g[:n - s] += wv * (1.0 - x[s:])
            g[s:] -= wv * x[:n - s]
    dx = 2.0 / n
    return lse * dx, g * dx

def lbfgs_run(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(obj_and_grad, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, max_passes=100, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    nm1 = n - 1
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        order = np.random.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            f_lo = max(0, nm1 - p)
            f_hi = min(2 * n - 2, 2 * n - 2 - p)
            b_lo = max(0, p)
            b_hi = min(2 * n - 2, p + nm1)
            m_lo = min(f_lo, b_lo)
            m_hi = max(f_hi, b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jb = p - ms + nm1
            vb = (jb >= 0) & (jb < n)
            if np.any(vb):
                a[vb] += 1.0 - h[jb[vb]]
            jf = p + ms - nm1
            vf = (jf >= 0) & (jf < n)
            if np.any(vf):
                a[vf] -= h[jf[vf]]
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            lo = -h0
            hi = 1.0 - h0
            best_d = 0.0
            best_val = cur_max
            n_grid = 64
            deltas = np.linspace(lo, hi, n_grid)
            vals = np.maximum((c_aff[:, None] + deltas[None, :] * a[:, None]).max(axis=0), unaff)
            idx = int(np.argmin(vals))
            if vals[idx] < best_val:
                best_val = vals[idx]
                best_d = deltas[idx]
                if 0 < idx < n_grid - 1:
                    lo2 = deltas[idx - 1]
                    hi2 = deltas[idx + 1]
                    d2 = np.linspace(lo2, hi2, 32)
                    v2 = np.maximum((c_aff[:, None] + d2[None, :] * a[:, None]).max(axis=0), unaff)
                    idx2 = int(np.argmin(v2))
                    best_d = d2[idx2]
                    best_val = v2[idx2]
            if abs(best_d) > 1e-12 and best_val < cur_max * (1.0 - 1e-12):
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c[ms] = c_aff + best_d * a
                cur_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
        new_c = compute_c5(h, n)
        if new_c < best_c:
            best_c = new_c
    return h, best_c

def make_structured(n, kind, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        return 0.5 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 6.28))
    elif kind == 1:
        return 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x + rng.uniform(0, 6.28))
    elif kind == 2:
        return 0.5 + 0.35 * np.sin(3 * np.pi * x + rng.uniform(0, 6.28))
    elif kind == 3:
        return (np.sin(2 * np.pi * x + rng.uniform(0, 6.28)) > 0).astype(float)
    elif kind == 4:
        return 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.15 * np.sin(5 * np.pi * x + rng.uniform(0, 6.28))
    else:
        return rng.uniform(0, 1, n)

    t0 = time.time()

    try:
