def coord_descent_exact(h, n, max_passes=200, deadline=None):

    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            lo = -h0
            hi = 1.0 - h0
            if hi - lo < 1e-12:
                continue
            
            # Get affected correlation indices
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2*n-2, n - 1 + p)
            m_b_lo = max(0, p - (n-1))
            m_b_hi = min(2*n-2, p + (n-1))
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            
            diag_mask = (ms == (n-1))
            c_aff = c[ms].copy()
            
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            
            # Ternary search for d in [lo, hi]
            # f(d) = max(unaff_max, max(c_aff + d*a - d²*diag))
            # This is convex in d
            for _ in range(60):
                t1 = lo + (hi - lo) / 3.0
                t2 = hi - (hi - lo) / 3.0
                v1 = unaff_max
                nv1 = c_aff + t1 * a
                if np.any(diag_mask):
                    nv1[diag_mask] -= t1 * t1
                v1 = max(v1, float(nv1.max()))
                
                v2 = unaff_max
                nv2 = c_aff + t2 * a
                if np.any(diag_mask):
                    nv2[diag_mask] -= t2 * t2
                v2 = max(v2, float(nv2.max()))
                
                if v1 < v2:
                    hi = t2
                else:
                    lo = t1
            
            best_d = (lo + hi) / 2.0
            if abs(best_d) < 1e-10:
                continue
            
            # Evaluate at best_d
            nv = c_aff + best_d * a
            if np.any(diag_mask):
                nv[diag_mask] -= best_d * best_d
            new_max = max(unaff_max, float(nv.max()))
            
            if new_max < best_c - 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = nv
                best_c = compute_c5(h, n)  # recompute to be safe (or just update)
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        if deadline is not None and time.time() > deadline:

        improved = False

            h0 = h[p]

            if hi - lo < 1e-12:

            m_f_hi = min(2*n-2, n - 1 + p)
            m_b_lo = max(0, p - (n-1))
            m_b_hi = min(2*n-2, p + (n-1))
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_descent_ternary(h, n, max_passes=200, deadline=None, ters=50):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            lo = -h0
            hi = 1.0 - h0
            if hi - lo < 1e-12:
                continue
            ml = max(0, min(n - 1 - p, p - (n - 1)))
            mh = min(2 * n - 2, max(n - 1 + p, p + (n - 1)))
            ms = np.arange(ml, mh + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag = (ms == (n - 1))
            ca = c[ms]
            umax = 0.0
            if ml > 0:
                umax = max(umax, float(c[:ml].max()))
            if mh < 2 * n - 2:
                umax = max(umax, float(c[mh + 1:].max()))
            tlo, thi = lo, hi
            for _ in range(ters):
                t1 = tlo + (thi - tlo) * 0.381966
                t2 = thi - (thi - tlo) * 0.381966
                nv1 = ca + t1 * a
                if np.any(diag):
                    nv1 = nv1.copy()
                    nv1[diag] -= t1 * t1
                v1 = max(umax, float(nv1.max()))
                nv2 = ca + t2 * a
                if np.any(diag):
                    nv2 = nv2.copy()
                    nv2[diag] -= t2 * t2
                v2 = max(umax, float(nv2.max()))
                if v1 < v2:
                    thi = t2
                else:
                    tlo = t1
            bd = (tlo + thi) * 0.5
            if abs(bd) < 1e-10:
                continue
            nv = ca + bd * a
            if np.any(diag):
                nv = nv.copy()
                nv[diag] -= bd * bd
            nm = max(umax, float(nv.max()))
            if nm < best_c - 1e-10:
                h[p] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = ca + bd * a
                if np.any(diag):
                    c[ms[diag]] -= bd * bd
                if nm < best_c:
                    best_c = nm
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        if cm < best_c:
            best_c = cm
    return h, compute_c5(h, n)
