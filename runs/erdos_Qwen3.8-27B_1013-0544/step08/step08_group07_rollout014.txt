import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fourier_h(params, n, K):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[2*(k-1)] * np.cos(np.pi * k * x)
        h += params[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def fourier_obj(params, n, K):
    h = fourier_h(params, n, K)
    pen = 0.0
    if h.min() < 0:
        pen += 100.0 * h.min()**2
    if h.max() > 1.0:
        pen += 100.0 * (h.max() - 1.0)**2
    return compute_c5(np.clip(h, 0, 1), n) + pen

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pidx in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_idx = (ms == (n - 1))
            any_diag = np.any(diag_idx)
            best_d = 0.0
            best_val = current_max
            d_coarse = np.linspace(-h0, 1.0 - h0, grid_size)
            for d in d_coarse:
                nv = c_aff + d * a
                if any_diag:
                    nv = nv.copy()
                    nv[diag_idx] -= d * d
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                sp = (1.0) / grid_size
                lo = max(-h0, best_d - sp * 3)
                hi = min(1.0 - h0, best_d + sp * 3)
                d_fine = np.linspace(lo, hi, 15)
                for d in d_fine:
                    nv = c_aff + d * a
                    if any_diag:
                        nv = nv.copy()
                        nv[diag_idx] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-10 and best_val < current_max - 1e-13:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c = c.copy()
                c[ms] = c_aff + best_d * a
                if any_diag:
                    c[ms[diag_idx]] -= best_d * best_d
        if not improved:
            break
    return h, compute_c5(h, n)

def lbfgs_polish(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-11})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
