            if time.time() > hard - 25: break
            hp = project_feasible(h + rng.normal(0, s, n), n)
            dl = time.time() + min(30, 0.2 * (hard - time.time()))
            ht, ct = nesterov_stage(hp, n, 200, 1000, min(dl, hard-20))
            if ct < c: h, c = ht, ct
            if time.time() < hard - 30:
                ht, ct = targeted_search(h, n, int(min(3000, max(500, (hard-time.time())*10))), hard-20, rng)
                if ct < c: h, c = ht, ct; n = len(h)
    # Phase 6: Final polish
    if time.time() < hard - 15:
        h, c = slsqp_refine(h, n, hard - 8)
    if time.time() < hard - 8:
        h, c = targeted_search(h, n, int(min(5000, max(500, (hard - time.time()) * 30))), hard - 4, rng)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c, n
