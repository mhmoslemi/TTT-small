Let g[i] = sum_k μ_k h[i+k'] (the "blurred" version of h). Then:
sum_k μ_k c_k = sum_i h[i] * (1 - g[i]) = sum_i h[i] - sum_i h[i]*g[i]

The objective becomes: sum_i h[i] - sum_i h[i]*g[i] + λ sum_i h[i] - λ n/2
= (1+λ) sum_i h[i] - sum_i h[i]*g[i] - λ n/2

But g depends on h! So this is still a coupled problem. However, we can use a **fixed-point iteration**:
1. Given current h, compute g = blur(h, μ)
2. Solve: min_{0≤h≤1, sum(h)=n/2} (1+λ) sum(h) - sum h[i]*g[i]
   This is: min sum_i [(1+λ) - g[i]] * h[i] subject to sum(h) = n/2, 0≤h≤1
   Solution: set h[i] = 1 where coefficient is most negative, h[i] = 0 where most positive, and one fractional value to satisfy the sum constraint.
   Coefficient for h[i] is (1+λ) - g[i]. So we want h[i] = 1 where g[i] > 1+λ (most), h[i] = 0 where g[i] < 1+λ.
   Wait, we're minimizing, so we want h[i] = 1 where the coefficient is most negative, i.e., where g[i] - (1+λ) is most positive, i.e., g[i] is largest.
   And h[i] = 0 where g[i] is smallest.

3. Update μ using the subgradient: μ_k ← μ_k * exp(η * c_k(h)) (normalized)
4. Update λ using: λ ← λ + η * (sum(h) - n/2) (but this is always 0 by construction)

This is essentially an **alternating direction method** or **dual ascent**.

Let me formalize:

**Algorithm: Dual Ascent for min-max**
- Initialize μ = uniform, λ = 0
- Repeat:
  a. Compute g[i] = sum_k μ[k] * h[i + k'] (convoled/blurred h)
  b. Solve inner problem: min (1+λ)sum(h) - h·g s.t. sum(h)=n/2, 0≤h≤1
     → h[i] = 1 for top n/2 values of g[i], 0 for rest (possibly one fractional)
     Actually: coefficient is (1+λ) - g[i]. Minimize sum_i ((1+λ)-g[i]) h[i] = (1+λ)n/2 - sum_i g[i]h[i]
     So maximize sum_i g[i]h[i] subject to sum(h)=n/2, 0≤h≤1
     → h[i] = 1 where g[i] is largest (n/2 of them), 0 elsewhere
  c. Compute c = correlate(h, 1-h)
  d. Update μ: μ[k] ∝ μ[k] * exp(η * c[k]) (or softer: μ[k] ∝ μ[k]^(1+η*c[k]))
  e. Adjust λ: not needed since we enforce sum constraint exactly

This is clean and efficient! Each iteration is O(n log n) (for the sort in step b) or O(n) if we use a selection algorithm.

The multiplicative weight update in step d is the key - it focuses the "attention" on the worst lags.

So output = [h0*m2, h0*m1+h1*m2, h0*m0+h1*m1, h1*m0]
Indices: 0,1,2,3.

g[0] = output[n-1] = output[1] = h0*m1 + h1*m2 ✓
g[1] = output[n] = output[2] = h0*m0 + h1*m1 ✓

So the formula is: g = np.correlate(h, mu, 'full')[n-1 : n-1+n]

Now for the full dual ascent algorithm:

```python
def dual_ascent(h_init, n, deadline, eta0=0.5, n_iter=100000):
    dx = 2.0 / n
    L = 2 * n - 1
    h = project_simplex(h_init, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    best = h.copy()
    best_c = cm
    mu = np.ones(L) / L
    eta = eta0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute g = blur of h using mu
        g = np.correlate(h, mu, 'full')[n-1 : 2*n-1]
        # Inner problem: max sum_i g[i]*h[i] s.t. sum(h)=n/2, 0<=h<=1
        # Solution: h[i]=1 for top n/2 values of g, 0 for rest
        # (with possible fractional value)
        if n % 2 == 0:
            target = n // 2
            # h[i] = 1 where g[i] is in top 'target' positions
            sort_idx = np.argsort(g)[::-1]
            h_new = np.zeros(n)
            h_new[sort_idx[:target]] = 1.0
        else:
            # For odd n, we need sum = n/2 which is not integer
            # Use softmax or linear interpolation
            target = (n - 1) // 2
            sort_idx = np.argsort(g)[::-1]
            h_new = np.zeros(n)
            h_new[sort_idx[:target]] = 1.0
            # One fractional value
            frac = n / 2.0 - target
            h_new[sort_idx[target]] = frac
        
        h = h_new
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        if cm < best_c:
            best_c = cm
            best = h.copy()
        
        # Update mu: multiplicative weights
        # mu[k] ∝ mu[k] * exp(eta * c[k])
        # Use log-sum-exp for numerical stability
        log_mu = np.log(mu)
        log_mu += eta * c
        log_mu -= log_mu.max()
        mu = np.exp(log_mu)
        mu /= mu.sum()
        
        # Decay eta
        if it > 1000:
            eta = max(eta0 * 0.999**it, 0.01)
    
    return best, best_c * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return smooth_val_grad(x, n, alpha)[1]

def sum_con(x, n):
    return x.sum() - 0.5 * n

def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best = h.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a), jac=smooth_grad,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc

def coord_eval(c, A, xs, old, n):
    d = xs - old
    cand = c[None, :] + d[:, None] * A[None, :]
    cand[:, n - 1] -= d * d
    return cand.max(axis=1)

def coord_cd(h, n, max_passes=4, deadline=None, grid=32, refine=10, tol=1e-12):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    L = 2 * n - 1
    diag = n - 1
    A = np.zeros(L)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            vals = coord_eval(c, A, xs0, old, n)
            i = int(np.argmin(vals))
            bx = float(xs0[i])
            bv = float(vals[i])
            if bv < cur - tol:
                sp = 1.0 / (grid - 1) if grid > 1 else 1.0
                lo, hi = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                mm = coord_eval(c, A, xx, old, n)
                j = int(np.argmin(mm))
                if mm[j] < bv - tol:
                    bx, bv = float(xx[j]), float(mm[j])
                if lo < bx < hi:
                    sp2 = (hi - lo) / float(refine)
                    lo2, hi2 = max(0.0, bx - sp2), min(1.0, bx + sp2)
                    yy = np.linspace(lo2, hi2, 9)
                    ww = coord_eval(c, A, yy, old, n)
                    k = int(np.argmin(ww))
                    if ww[k] < bv - tol:
                        bx, bv = float(yy[k]), float(ww[k])
                if bv < cur - tol:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - tol:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

def dual_ascent(h_init, n, deadline, eta0=1.0, n_iter=50000, soft=False):
    """Dual ascent with multiplicative weights."""
    dx = 2.0 / n
    L = 2 * n - 1
    h = project_simplex(h_init, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(c.max())
    best = h.copy()
    best_c = cm
    mu = np.ones(L) / L
    eta = eta0
    target = 0.5 * n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute g[i] = sum_k mu[k] * h[i+k-n+1]
        g = np.correlate(h, mu, 'full')[n-1:2*n-1]
        # Inner solution
        if soft:
            h_new = project_simplex(g, n)
        else:
            if n % 2 == 0:
                t_int = n // 2
                sort_idx = np.argsort(g)[::-1]
                h_new = np.zeros(n)
                h_new[sort_idx[:t_int]] = 1.0
            else:
                t_int = (n - 1) // 2
                sort_idx = np.argsort(g)[::-1]
                h_new = np.zeros(n)
                h_new[sort_idx[:t_int]] = 1.0
                h_new[sort_idx[t_int]] = target - t_int
        h = h_new
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        if cm < best_c:
            best_c = cm
            best = h.copy()
        # Update mu (multiplicative weights)
        # Use a temperature-scaled version for stability
        log_mu = np.log(mu + 1e-300) + eta * c
        log_mu -= log_mu.max()
        mu = np.exp(log_mu)
        s = mu.sum()
        if s < 1e-30:
            mu = np.ones(L) / L
        else:
            mu /= s
        # Decay eta
        if it > 500:
            eta = max(eta0 * (0.9995 ** it), 0.05)
    return best, best_c * dx
