def solve_1d(c_aff, a_coeff, diag_local, unaff_max, lo, hi):
    """Solve min_d max(c_aff[i] + d*a_coeff[i] - d^2*delta) for d in [lo, hi]
    where the max also includes unaff_max."""
    k = len(a_coeff)
    if k == 0:
        return 0.0  # no change helps
    
    # The objective at d is max(max_i(c_aff[i] + d*a_coeff[i] - d^2*delta_i), unaff_max)
    # We want to minimize this.
    # Strategy: find candidate d values and pick the best.
    
    candidates = [lo, hi]
    
    # Add pairwise intersection points
    for i in range(k):
        for j in range(i+1, k):
            # c_aff[i] + d*a_coeff[i] - d^2*delta_i = c_aff[j] + d*a_coeff[j] - d^2*delta_j
            # d*(a_coeff[i]-a_coeff[j]) = c_aff[j]-c_aff[i] + d^2*(delta_i - delta_j)
            da = a_coeff[i] - a_coeff[j]
            dc = c_aff[j] - c_aff[i]
            dd = (1 if diag_local == i else 0) - (1 if diag_local == j else 0)
            # dd * d^2 + da * d - dc = 0
            if dd == 0:
                # Linear: da * d = dc
                if abs(da) > 1e-15:
                    d = dc / da
                    if lo <= d <= hi:
                        candidates.append(d)
            else:
                # Quadratic: dd*d^2 + da*d - dc = 0
                disc = da*da + 4*dd*dc
                if disc >= 0:
                    d1 = (-da + np.sqrt(disc)) / (2*dd)
                    d2 = (-da - np.sqrt(disc)) / (2*dd)
                    if lo <= d1 <= hi:
                        candidates.append(d1)
                    if lo <= d2 <= hi:
                        candidates.append(d2)
    
    # Also add the point where the max equals unaff_max
    # This is where f(d) = unaff_max, which is hard to find directly
    # Skip for now, golden section will handle it
    
    # Evaluate at all candidates
    best_d = lo
    best_val = obj_1d(lo, c_aff, a_coeff, diag_local, unaff_max)
    for d in candidates:
        d = np.clip(d, lo, hi)
        val = obj_1d(d, c_aff, a_coeff, diag_local, unaff_max)
        if val < best_val:
            best_val = val
            best_d = d
    
    # Refine with golden section around the best candidate
    if 0 < best_d - lo and hi - best_d > 1e-10:
        # Use golden section in a narrow bracket around best_d
        margin = (hi - lo) * 0.1
        gl = max(lo, best_d - margin)
        gh = min(hi, best_d + margin)
        if gh > gl:
            try:
                res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except:
                pass
    
    return best_d

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

    if k == 0:
        return 0.0
    
    candidates = [lo, hi]

    for i in range(k):

            dc = c_aff[j] - c_aff[i]
            dd = 0.0
            if diag_local == i:
                dd -= 1.0
            if diag_local == j:
                dd += 1.0

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            ho, co = coord_cd_fast(hp, n_cur, max_passes=80, deadline=deadline - 30, top_k=30)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 8: High-precision SLSQP
    if time.time() < deadline - 30:
        h7, c7 = slsqp_polish(best_h, n_cur, [10000, 50000, 200000], 500, deadline - 15)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 10:
        h8, c8 = coord_cd_fast(best_h, n_cur, max_passes=800, deadline=deadline - 5, top_k=40)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Try 1-h symmetry
    n_fin = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_fin)
    c_flip = compute_c5(h_flip, n_fin)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)
    return best_h, best_c, n_fin
