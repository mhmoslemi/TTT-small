    return (m + np.log(z) / alpha) * dx, (A - B)[s:s+n] * dx

def slsqp_polish(h, n, alphas, maxiter, deadline):
    from scipy.optimize import minimize
    best = project_simplex(h.copy(), n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        cons = [{"type": "eq", "fun": lambda x, n=n: x.sum() - 0.5*n,
                 "jac": lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(lambda x, n=n, a=a: smooth_val_grad(x, n, a)[0],
                          best, jac=lambda x, n=n, a=a: smooth_val_grad(x, n, a)[1],
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": int(maxiter), "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-13:
            best, bc = nh, nc
    return best, bc

def transfer_h(h, n, n_tgt):

        return project_simplex(h.copy(), n_tgt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

import numpy as np
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def binarize(h, n):
    target = n // 2
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def swap_trial(c, h, om, i, j, n):
    d = c.copy()
    d[i:i+n] -= om
    st = n - 1 - i
    d[st:st+n] += h
    d[j:j+n] += om
    st2 = n - 1 - j
    d[st2:st2+n] -= h
    d[n-1] -= 2.0
    d[n-1+i-j] += 1.0
    d[n-1+j-i] += 1.0
    return d
