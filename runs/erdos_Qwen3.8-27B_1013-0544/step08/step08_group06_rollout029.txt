```python
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    if alpha > 1e6:
        return m * (2.0 / n)
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

                    c[ms[diag_mask]] -= best_d * best_d
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved += 1
        
        if current_c5 < best_c - 1e-15:
            best_c = current_c5
            best_h = h.copy()

            break
    
    best_h = project_feasible(best_h, n)
    final_c = compute_c5(best_h, n)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c

def swap_local_search(h, n, max_iterations, deadline, n_swaps_per_iter=50, rng=None):
    """Try swapping values at pairs of positions to reduce the objective."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    best_c = current_c5
    best_h = h.copy()
    
    for it in range(max_iterations):
        if time.time() > deadline:
            break
        n_tried = 0
        improved = False
        while n_tried < n_swaps_per_iter:

            j = rng.randint(n)
            if i == j or abs(h[i] - h[j]) < 1e-10:
                n_tried += 1
                continue

            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)  # should maintain feasibility
            c_new = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(c_new.max())
            
            if new_c5 < current_c5 - 1e-14:
                current_c5 = new_c5
                c = c_new
                improved = True
                if new_c5 < best_c:
                    best_c = new_c5
                    best_h = h.copy()
            else:
                # Undo
                h[i], h[j] = h[j], h[i]
            
            n_tried += 1
        
        if not improved:
            break
    
    return h, current_c5
```

For the problem min max_k ∫ h(x)(1-h(x+k)) dx with ∫h = 1 on [0,2]:

If h is the indicator of a set A with |A| = 1, then ∫ h(x)(1-h(x+k)) dx = |A \ (A-k)| (the part of A that, when shifted by -k, falls outside A).

By the properties of sets in [0,2], for small k, |A \ (A-k)| ≈ k * (number of "edges" of A). For the optimal set, we want to minimize the maximum over k.

A set that is "evenly distributed" would have |A \ (A-k)| ≈ k/2 for all small k (since the density is 1/2). The maximum over k would then be approximately 1/2 * k_max where k_max is the largest shift we consider.

But we're maximizing over ALL k, including large ones. For large k (near 2), the overlap is small anyway.

The critical shifts are probably in the range [0, 1] or so.

I think the optimal structure for this problem (Erdős 5-fold) is related to a "balanced" set. Let me not overthink the theory and focus on the computational approach.

```python
def project_feasible(h, n):

            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.exp(alpha * (c - m))
    lse = m + np.log(s.sum()) / alpha
    return lse * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_phase(h, n, alphas, maxiter, deadline):

            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)

            pass

        for idx in order:
            if time.time() > deadline:
                break

            if h0 < 1e-10 or h0 > 1.0 - 1e-10:

            lo, hi = -h0, 1.0 - h0
            if hi - lo < 1e-12:

            m_lo = max(0, min(n - 1 - idx, idx))
```

```python
def block_rotate_search(h, n, deadline, rng, block_sizes=(3, 5, 10), n_rounds=5000):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_c, best_h = cur, h.copy()
    
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        bs = block_sizes[r % len(block_sizes)]
        start = rng.randint(0, n - bs)
        # Cyclic shift by 1
        seg = h[start:start+bs].copy()
        h[start:start+bs] = np.roll(seg, -1)
        c_new = np.correlate(h, 1.0 - h, mode='full')
        nc = float(c_new.max())
        if nc < cur - 1e-14:
            cur, c = nc, c_new
            if cur < best_c:
                best_c, best_h = cur, h.copy()
        else:
            h[start:start+bs] = seg
    
    return best_h, best_c
```

    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.exp(alpha * (c - m))
    lse = m + np.log(s.sum()) / alpha
    return lse * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_phase(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size=100, random=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_c, best_h = cur, h.copy()
    stall_count = 0

    for p in range(max_passes):
        if time.time() > deadline:
            break
        if random and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)

        improved = False
        for idx in order:
            if time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            lo, hi = -h0, 1.0 - h0
            if hi - lo < 1e-12:
                continue

            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)

            a = np.zeros(len(ms))
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]

            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))

            dm = (ms == n - 1)
            hd = np.any(dm)

            dg = np.linspace(lo, hi, grid_size)
            nv = c_aff[None, :] + dg[:, None] * a[None, :]
            if hd:
                nv[:, dm] -= dg[:, None] ** 2
            vals = np.maximum(unaff, nv.max(axis=1))

            bi = int(np.argmin(vals))
            bd, bv = float(dg[bi]), float(vals[bi])

            if 0 < bi < grid_size - 1:
                y = vals[bi - 1:bi + 2]
                x = dg[bi - 1:bi + 2]
                den = y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])
                if abs(den) > 1e-18:
                    xv = 0.5 * (y[0] * (x[1] - x[2]) + y[1] * (x[2] - x[0]) + y[2] * (x[0] - x[1])) / den
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if hd:
                            nv1 = nv1.copy()
                            nv1[dm] -= xv * xv
                        v1 = max(unaff, float(nv1.max()))
                        if v1 < bv:
                            bv, bd = v1, xv

            if abs(bd) > 1e-10 and bv < cur - 1e-15:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c
