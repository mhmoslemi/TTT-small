import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_exact(h, n):
    target = n / 2.0
    s0 = np.sum(np.clip(h, 0.0, 1.0))
    if abs(s0 - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    logit = alpha * (c - m)
    logit -= logit.max()
    p = np.exp(logit)
    p /= p.sum()
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(n):
        j_lo_a = max(0, k - n + 1)
        js_a = np.arange(j_lo_a, n)
        ms_a = k - js_a + (n - 1)
        ms_a = ms_a[(ms_a >= 0) & (ms_a < 2*n-1)]
        js_a = js_a[(ms_a >= 0) & (ms_a < 2*n-1)]
        if len(js_a) > 0:
            A[k] = np.dot(p[ms_a], h[js_a])
        j_lo_b = max(0, k - n + 1)
        j_hi_b = min(n - 1, k + n - 2)
        js_b = np.arange(j_lo_b, j_hi_b + 1)
        ms_b = js_b - k + (n - 1)
        if len(js_b) > 0:
            B[k] = np.dot(p[ms_b], h[js_b])
    return (1.0 - A - B) * (2.0 / n)

def nesterov_pg(h, n, alphas, iters_per_alpha, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        h_curr = h.copy()
        h_prev = h.copy()
        for it in range(iters_per_alpha):
            if time.time() > deadline - 20:
                break
            y = h_curr + ((np.sqrt(1.0 + 4.0 * (it+1)**2) - 1.0) / (2.0 * (it+1))) * (h_curr - h_prev)
            g = smooth_gradient(y, n, alpha)
            step = 0.05 / (1.0 + 0.01 * it)
            h_new = project_exact(y - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_prev = h_curr
            h_curr = h_new
        h = h_curr
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng_local = np.random.RandomState(0)
    for pidx in range(max_passes):
        if time.time() > deadline:
            break
        c_prev = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c_prev.max())
        improved = False
        order = rng_local.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            m_lo = max(0, min(p, n - 1 - p) * 0 + max(0, p - (n-1)))
            # Forward: h[p] appears in c[m] where j = p + m - (n-1) valid -> m in [n-1-p, 2n-2-p]
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            # Backward: h[p] appears in c[m] where j = p - m + (n-1) valid -> m in [p, p+n-1]
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a[vf] += (1.0 - h[jf[vf]])
            a[vb] -= h[jb[vb]]
            c_aff = c_prev[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c_prev[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_prev[m_hi + 1:].max()))
            diag = (ms == (n - 1))
            best_d = 0.0
            best_val = float(c_aff.max())
            if unaff_max > best_val:
                best_val = unaff_max
            d_coarse = np.linspace(-h0, 1.0 - h0, 24)
            for d in d_coarse:
                if abs(d) < 1e-14:
                    val = best_val if best_val == float(c_prev.max()) else float(c_aff.max())
                    if val < best_val:
                        best_val = val
                        best_d = d
                    continue
                nv = c_aff + d * a
                if np.any(diag):
                    nv[diag] -= d * d
                val = max(unaff_max, float(nv.max()))

                    best_val = val

                d_fine = np.linspace(lo_d, hi_d, grid_size)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a
                    if np.any(diag):
                        nv[diag] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            elif grid_size > 30:
                d_fine = np.linspace(-min(h0, 0.005), min(1.0-h0, 0.005), grid_size)
                for d in d_fine:
                    if abs(d) < 1e-14:
                        continue
                    nv = c_aff + d * a
                    if np.any(diag):
                        nv[diag] -= d * d
                    val = max(unaff_max, float(nv.max()))
                    if val < best_val:
                        best_val = val
                        best_d = d
            if abs(best_d) > 1e-14 and best_val < current_max - 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                c_prev[ms] = c_aff + best_d * a

                    current_max = best_val
        if not improved:

        h = project_exact(h, n)
        new_c = compute_c5(h, n)

        best_h, best_c = h1, c1

    # Phase 2: Nesterov projected gradient with alpha annealing
    h2, c2 = nesterov_pg(best_h, n, [200, 1000, 5000, 20000], 300, min(deadline, t0 + 350))
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Coordinate descent refine
    h3, c3 = coord_cd(best_h, n, 200, min(deadline, t0 + 550), grid_size=100)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Multi-resolution tries
    x_big = np.linspace(0, 2, n, endpoint=False)
    for n_s in [200, 300, 400]:
        if time.time() > deadline - 120:
            break
        x_s = np.linspace(0, 2, n_s, endpoint=False)
        h_s = project_exact(np.interp(x_s, x_big, best_h) + 0.01 * rng.randn(n_s), n_s)
        h_s, c_s = nesterov_pg(h_s, n_s, [500, 5000, 50000], 200, min(deadline, t0 + 750))
        h_s, c_s = coord_cd(h_s, n_s, 150, min(deadline, t0 + 800), grid_size=80)
        h_back = project_exact(np.interp(x_big, x_s, h_s), n)
        h_back, c_back = coord_cd(h_back, n, 60, deadline - 60, grid_size=80)
        if c_back < best_c:
            best_h, best_c = h_back, c_back

    # Phase 5: Diverse restarts
    if time.time() < deadline - 80:
        n_restarts = 3
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r == 0:
                hp = project_exact(best_h + 0.03 * rng.randn(n), n)
            elif r == 1:
                hp = structured_start(n, 1 + r, rng)
            else:
                hp = project_exact(best_h + 0.08 * rng.randn(n), n)
            h_o, c_o = nesterov_pg(hp, n, [1000, 10000], 200, deadline - 40)
            h_o, c_o = coord_cd(h_o, n, 80, deadline - 40, grid_size=80)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final fine coordinate descent
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n, 200, deadline, grid_size=150)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_exact(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
