import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n

                           bounds=bounds, constraints=cons,

            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] -= pk * h[idx2 + k]
    return grad

def game_optimize(h, n, max_outer=80, inner_steps=15, tau_init=3.0, tau_decay=0.93, deadline=None):
    """Minimax game: inner = PG descent on h, outer = exp weights on p"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    p = np.ones(2*n-1) / (2*n-1)
    tau = tau_init
    lr = 0.05

            break

        for it in range(inner_steps):

            h = project_feasible(h - lr * grad, n)

            if c < best_c:

                lr = min(lr * 1.1, 0.2)

            best_c = c_max
            best_h = h.copy()

        c_flip = compute_c5(h_flip, n)

            best_h = h_flip.copy()

        c_shifted = c - c.max()
        p = np.exp(c_shifted / tau)
        p /= p.sum()
        tau *= tau_decay
    
    return best_h, best_c

    d is the lag shift (m = n-1+d)."""

    ts.add(i)
    ts.add(i+d)
    ts.add(j)
    ts.add(j+d)
    
    delta = 0.0

        if t < 0 or t >= n:
            continue
        td = t - d
        if td < 0 or td >= n:
            continue
        # Old: b[t]*(1-b[td])
        bt_old = 1.0 if t == i else (0.0 if t == j else b[t])

        delta += bt_new * (1 - btd_new) - bt_old * (1 - btd_old)
    return delta

from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def compute_grad_p(h, p, n):
    """Gradient of f_p(h) = sum_m p[m] * corr_m(h) w.r.t. h"""
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[n-1+k]
        if pk < 1e-10:
            continue
        i_lo = max(0, k)
        i_hi = min(n, n + k)
        if i_hi > i_lo:
            idx = np.arange(i_lo, i_hi)
            grad[idx] += pk * (1.0 - h[idx - k])
        i_lo2 = max(0, -k)
        i_hi2 = min(n, n - k)
        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] -= pk * h[idx2 + k]
    return grad

    h = h.copy()

    p = np.ones(2*n-1) / (2*n-1)
    tau = tau_init
    lr = 0.03

            h = project_feasible(h - lr * grad, n)
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
                lr = min(lr * 1.05, 0.15)
            else:
                lr *= 0.95
        c = np.correlate(h, 1-h, mode='full')
        c_now = float(c.max()) * (2.0/n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        c_shifted = c - c.max()
        p = np.exp(c_shifted / tau)
        p /= p.sum()
        tau *= tau_decay
    return best_h, best_c

def topk_subgradient(h, n, k=5, max_iter=200, deadline=None):
    """Subgradient method focusing on top-k lags"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full')
        top_idx = np.argsort(-c)[:k]
        p = np.zeros(2*n-1)
        p[top_idx] = 1.0 / k
        grad = compute_grad_p(h, p, n)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.8
            stall += 1
            if stall > 50 or lr < 1e-8:
                break
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)
