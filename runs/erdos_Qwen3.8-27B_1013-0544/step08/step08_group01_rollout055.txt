```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate, convolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    val = (m + np.log(np.sum(np.exp(shifted - sm))) / alpha) * (2.0 / n)
    val += lam * (x.sum() - n/2.0)**2
    return val

def smooth_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    p = exps / exps.sum()
    
    # g1[j] = correlate(1-x, p, 'full')[j+n-1]
    # g2[j] = convolve(x, p, 'full')[j+n-1]
    g1_full = correlate(1.0 - x, p, mode='full')
    g2_full = convolve(x, p, mode='full')
    grad = dx * (g1_full[n-1:n-1+n] - g2_full[n-1:n-1+n])
    grad += 2*lam*(x.sum() - n/2.0)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate as scipy_correlate, convolve as scipy_convolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    val = (m + np.log(np.sum(np.exp(shifted - sm))) / alpha) * (2.0 / n)
    val += lam * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    p = exps / exps.sum()
    g1_full = scipy_correlate(1.0 - x, p, mode='full')
    g2_full = scipy_convolve(x, p, mode='full')
    grad = dx * (g1_full[n - 1:n - 1 + n] - g2_full[n - 1:n - 1 + n])
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def optimize_lbfgs(h, n, alphas, lam, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha, lam),
                           method='L-BFGS-B', jac=smooth_grad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    val = float(nv.max())
    return max(val, unaff_max)

def coord_cd(h, n, max_passes, deadline, grid_size=60):
    from scipy.optimize import golden
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            valid_f = (jf_arr >= 0) & (jf_arr < n)
            valid_b = (jb_arr >= 0) & (jb_arr < n)
            if np.any(valid_f):
                a_coeff[valid_f] += 1.0 - h[jf_arr[valid_f]]
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb_arr[valid_b]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    r = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if r.fun < best_val:
                        best_d, best_val = float(r.x), float(r.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_local_search(h, n, max_iters, deadline):
    hb = (h > 0.5).astype(float)
    n_ones = int(hb.sum())
    n_target = n // 2
    if n_ones > n_target:
        ones = np.where(hb == 1)[0]
        rng = np.random.RandomState(0)
        for i in range(n_ones - n_target):
            idx = rng.randint(len(ones))
            hb[ones[idx]] = 0.0
    elif n_ones < n_target:
        zeros = np.where(hb == 0)[0]
        rng = np.random.RandomState(1)
        for i in range(n_target - n_ones):
            idx = rng.randint(len(zeros))
            hb[zeros[idx]] = 1.0
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    temp = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        hb[i], hb[j] = 0.0, 1.0
        c_new = compute_c5(hb, n)
        delta = c_new - best_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            if c_new < best_c:
                best_c = c_new
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
        if it % 100 == 99:
            temp *= 0.9
    return best_hb, best_c
```
