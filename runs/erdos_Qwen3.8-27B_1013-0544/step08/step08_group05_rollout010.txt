    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    d = h.sum() - t
    if abs(d) < 1e-13:
        return h

        # Need to decrease sum, reduce from non-zero entries
        mask = h > 1e-9
        if mask.sum() > 0:
            h[mask] -= d / mask.sum()

    return np.clip(h, 0.0, 1.0)


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)


def smooth_max_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    r = alpha * (c - c.max())
    exp_r = np.exp(r)
    z = exp_r.sum()
    p = exp_r / z
    obj = (c.max() + np.log(z) / alpha) * (2.0 / n)
    p_rev = p[::-1]
    first = np.convolve(h, p, mode='full')[n - 1:2 * n - 1]
    second = np.convolve(h, p_rev, mode='full')[n - 1:2 * n - 1]
    grad = (1.0 - first - second) * (2.0 / n)
    return obj, grad


def ltb_optimize(h, n, alpha_list, max_iter, deadline, lambda_pen):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n

    def obj_pen(x):
        sm, g = smooth_max_obj_grad(x, n, alpha)
        pen = lambda_pen * (x.sum() - 0.5 * n) ** 2
        grad_pen = g + 2.0 * lambda_pen * (x.sum() - 0.5 * n) * np.ones(n)
        return pen, grad_pen

    for alpha in alpha_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_pen, best, jac=True, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best, bc = nh, nc
    return best, bc


def coord_cd(h, n, max_passes=5, deadline=None, grid=24, refine=8):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    diag = n - 1
    A = np.zeros(2 * n - 1)
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        for p in range(n):
            old = float(x[p])
            A.fill(0.0)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            d = xs0[None, :] - old
            cand = c[None, :] + d * A[None, :]
            cand[:, n - 1] -= d * d
            vals = cand.max(axis=1)
            i = int(np.argmin(vals))
            bx, bv = float(xs0[i]), float(vals[i])
            if bv < cur - 1e-12:
                sp = 1.0 / (grid - 1)
                lo, hi = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo, hi, refine + 1)
                d2 = xx[None, :] - old
                cand2 = c[None, :] + d2 * A[None, :]
                cand2[:, n - 1] -= d2 * d2
                mm = cand2.max(axis=1)
                j = int(np.argmin(mm))
                if mm[j] < bv:
                    bx, bv = float(xx[j]), float(mm[j])
                if bv < cur - 1e-12:
                    dd = bx - old
                    c[p:p + n] += dd * om
                    c[st:st + n] -= dd * x
                    c[diag] -= dd * dd
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-13:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc


def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        return None
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def binary_sa(h_init, n, rng, deadline, max_rounds=200000):
    b = binarize(h_init, n)
    if b is None:
        return None, np.inf
    dx = 2.0 / n
    best = b.copy()
    bc = float(np.correlate(best, 1.0 - best, mode="full").max())
    h = best.copy()
    c = np.correlate(h, 1.0 - h, mode="full").copy()
    cm = float(c.max())
    T = 0.01
    cool = 0.9995
    best_T = T
    stagnant = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if rnd % 5000 == 0 and rnd > 0:
            T = max(0.001, best_T * 0.7)
        for _ in range(50):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j or h[i] == h[j]:
                continue
            om = 1.0 - h[::-1]
            d = c.copy()
            si, sj = i, j
            if h[i] > h[j]:
                d[si:si + n] -= om
                d[n - 1 - si:n - 1 - si + n] += h
                d[sj:sj + n] += om
                d[n - 1 - sj:n - 1 - sj + n] -= h
            else:
                d[si:si + n] -= om
                d[n - 1 - si:n - 1 - si + n] += h
                d[sj:sj + n] += om
                d[n - 1 - sj:n - 1 - sj + n] -= h
            d[n - 1] -= 2.0
            d[n - 1 + i - j] += 1.0
            d[n - 1 + j - i] += 1.0
            nm = float(d.max())
            delta = nm - cm
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                c = d
                cm = nm
                h[i] = 0.0 if h[i] > h[j] else 1.0
                h[j] = 1.0 if h[i] > h[j] else 0.0
                if cm < bc:
                    best = h.copy()
                    bc = cm
                    stagnant = 0
        T *= cool
        stagnant += 1
        if stagnant > 50:
            break
    return best, bc * dx


def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 8.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = float(compute_c5(h0, n0))
    n_best = n0

    # Quick symmetry improvements
    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    # Phase 1: L-BFGS-B optimization at n_best
    t1 = min(deadline - 200.0, t0 + 0.25 * budget_s)
    if time.time() < t1:
        h1, c1 = ltb_optimize(best_h, n_best, [500, 2000, 10000, 50000], 200, t1, 50000.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coordinate descent
    t2 = min(deadline - 150.0, time.time() + 0.15 * budget_s)
    if time.time() < t2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=8, deadline=t2, grid=32, refine=10)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 3: Binary search with simulated annealing
    t3 = min(deadline - 100.0, time.time() + 0.30 * budget_s)
    if n_best % 2 == 0 and time.time() < t3:
        hb, cb = binary_sa(best_h, n_best, rng, t3, max_rounds=300000)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb
        # Polish the binary result
        t3b = min(deadline - 80.0, time.time() + 60.0)
        if time.time() < t3b:
            h1, c1 = ltb_optimize(best_h, n_best, [5000, 50000, 200000], 100, t3b, 100000.0)
            if c1 < best_c:
                best_h, best_c = h1, c1
            h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=t3b, grid=24, refine=8)
            if c1 < best_c:
                best_h, best_c = h1, c1

    # Phase 4: Try other n values
    n_cands = sorted(set([n0 - 20, n0 - 10, n0, n0 + 10, n0 + 20, n0 + 40, n0 - 40, n0 + 60]),
                     key=lambda x: abs(x - n0))
    tried = {n_best}
    for nt in n_cands:
        if nt % 2 != 0 or nt < 200 or nt > 1000:
            continue
        if nt in tried:
            continue
        if time.time() > deadline - 80.0:
            break
        tried.add(nt)
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt
        chunk = min(deadline - 60.0, time.time() + 40.0)
        if time.time() < chunk:
            h1, c1 = ltb_optimize(ht, nt, [2000, 20000], 80, chunk, 50000.0)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, nt
            h1, c1 = coord_cd(best_h, n_best, max_passes=3, deadline=chunk, grid=20, refine=6)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, int(nt)

    # Phase 5: Final polish
    tf = min(deadline - 5.0, time.time() + 80.0)
    if time.time() < tf:
        h1, c1 = ltb_optimize(best_h, n_best, [20000, 200000, 1000000], 150, tf, 200000.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    tf2 = min(deadline - 3.0, time.time() + 60.0)
    if time.time() < tf2:
        h1, c1 = coord_cd(best_h, n_best, max_passes=4, deadline=tf2, grid=40, refine=12)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Final binary attempt if time permits
    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb, cb = binary_sa(best_h, n_best, rng, deadline - 2.0, max_rounds=100000)
        if hb is not None and cb < best_c:
            best_h, best_c = hb, cb

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
