        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: CD after SLSQP
    h6, c6 = coord_cd(best_h, n_cur, max_passes=500, deadline=deadline,
                      shuffle=True, rng=rng)
    if c6 < best_c:
        best_h, best_c = h6, c6

    # Phase 7: Random perturbations + CD
    n_cur = len(best_h)
    for r in range(15):
        if time.time() > deadline - 40:
            break
        s = 0.002 + 0.004 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_r, c_r = coord_cd(hp, n_cur, max_passes=80, deadline=deadline)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 8: Final SLSQP + CD
    if time.time() < deadline - 30:
        hf, cf = slsqp_polish(best_h, len(best_h), [200000, 800000], 800, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        hf, cf = coord_cd(best_h, len(best_h), max_passes=500, deadline=deadline)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
