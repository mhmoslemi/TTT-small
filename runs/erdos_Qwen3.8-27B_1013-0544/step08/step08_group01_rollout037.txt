    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize, brent
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return (m + mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha * (2.0 / n)

def coord_cd(h, n, max_passes=500, deadline=None, grid_size=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Compute affected shifts
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            diag_idx = None
            if m_lo <= (n-1) <= m_hi:
                diag_idx = (n-1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            # Use brent on a 1D function
            d_grid = np.linspace(lo, hi, min(grid_size, 200))
            best_d = 0.0
            best_val = current_max
            for d in d_grid:
                nv = c_aff + d * a_coeff
                if diag_idx is not None:
                    nv[diag_idx] -= d * d
                val = float(nv.max())
                val = max(val, unaff_max)
                if val < best_val:
                    best_val = val
                    best_d = d
            # Refine with brent
            # Find the best grid point, then use brent in its neighborhood

    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP warmup
    if time.time() < deadline - 400:
        h1, c1 = slsqp_run(best_h, n, [100, 1000, 10000], 300)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Main coordinate descent
    if time.time() < deadline - 250:
        h2, c2 = coord_cd(best_h, n, 200, deadline - 250, grid_size=120)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Try different n values
    x_orig = np.linspace(0, 2, len(best_h))
    n_candidates = [400, 500, 700, 800, 1000]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [500, 5000], 200)
        h_opt, c_opt = coord_cd(h_opt, n_try, 100, deadline - 120, grid_size=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 4: High-precision CD on best solution
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        h4, c4 = coord_cd(best_h, n_cur, 400, deadline - 40, grid_size=200)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Annealing polish
    if time.time() < deadline - 30:
        h5, c5 = anneal_polish(best_h, n_cur, 0.001, 1e-7, 50000, deadline - 15)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 6: Final ultra-fine CD
    if time.time() < deadline - 10:
        h6, c6 = coord_cd(best_h, n_cur, 300, deadline - 5, grid_size=300)
        if c6 < best_c:
            best_h, best_c = h6, c6
    # Phase 7: Small perturbation restarts
    if time.time() < deadline - 10:
        for r in range(3):
            if time.time() > deadline - 5:
                break
            s = 0.005 + 0.01 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = coord_cd(hp, n_cur, 80, deadline - 3, grid_size=150)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
