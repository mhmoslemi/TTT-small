def transfer_h(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h, n_tgt).copy()
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(5.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_simplex(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    n_best = n0

    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n_best)
        cv = compute_c5(vv, n_best)
        if cv < best_c:
            best_h, best_c = vv, cv

    hb = binarize(h0, n_best)
    if hb is not None:
        cb = compute_c5(hb, n_best)
        if cb < best_c:
            best_h, best_c = hb, cb

    t1 = min(deadline - 400.0, t0 + 60.0)
    if time.time() < t1:
        h1, c1 = slsqp_polish(best_h, n_best, [500, 5000, 50000], 150, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1
        hb = binarize(h1, n_best)
        if hb is not None:
            cb = compute_c5(hb, n_best)
            if cb < best_c:
                best_h, best_c = hb, cb

    t2 = min(deadline - 200.0, t0 + max(200.0, 0.55 * budget_s))
    if n_best % 2 == 0 and time.time() < t2:
        h_bin = binarize(best_h, n_best)
        if h_bin is None:
            h_bin = best_h.copy()
        T_roof = min(t2 - 30.0, time.time() + 80.0)
        h1, c1 = binary_sa_single(h_bin, n_best, rng, T_roof, T0=0.008, cool=0.99985)
        if c1 < best_c:
            best_h, best_c = h1, c1
        T_roof2 = min(t2, time.time() + 120.0)
        h1, c1 = binary_pair_swaps(best_h, n_best, rng, T_roof2, max_trials=60000)
        if c1 < best_c:
            best_h, best_c = h1, c1
        T_roof3 = min(t2, time.time() + 100.0)
        h1, c1 = binary_sa_single(best_h, n_best, rng, T_roof3, T0=0.004, cool=0.9999)
        if c1 < best_c:
            best_h, best_c = h1, c1

    t3 = min(deadline - 80.0, time.time() + 150.0)
    for nt in [400, 600, 450, 550, 350, 700, 500]:
        if time.time() > t3:
            break
        if nt == n_best or nt % 2 or nt < 100 or nt > 800:
            continue
        ht = transfer_h(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt
            hb2 = binarize(ht, nt)
            if hb2 is not None:
                cb2 = compute_c5(hb2, nt)
                if cb2 < best_c:
                    best_h, best_c = hb2, cb2
        chunk = min(t3, time.time() + 50.0)
        if time.time() < chunk and n_best % 2 == 0:
            hb3 = binarize(best_h, n_best)
            h1, c1 = binary_pair_swaps(hb3, n_best, rng, chunk, max_trials=20000)
            if c1 < best_c:
                best_h, best_c = h1, c1

    t4 = min(deadline - 20.0, time.time() + 80.0)
    if time.time() < t4:
        h1, c1 = slsqp_polish(best_h, n_best, [50000, 500000], 100, t4)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if n_best % 2 == 0 and time.time() < deadline - 5.0:
        hb4 = binarize(best_h, n_best)
        h1, c1 = binary_pair_swaps(hb4, n_best, rng, deadline, max_trials=30000)
        if c1 < best_c:
            best_h, best_c = h1, c1

    final_h = project_simplex(best_h, n_best)
    final_c = float(compute_c5(final_h, n_best))
    return final_h, final_c, int(n_best)
