        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _make_indices(n):
    cache = {}
    for p in range(n):
        kA_lo = max(0, p)
        kA_hi = min(2 * n - 2, p + n - 1)
        a_ks = np.arange(kA_lo, kA_hi + 1) if kA_lo <= kA_hi else np.array([], dtype=int)
        a_iv = p + (n - 1) - a_ks
        a_valid = (a_iv >= 0) & (a_iv < n)
        a_diag = a_valid & (a_iv == p)
        a_valid_nd = a_valid & ~a_diag
        kB_lo = max(0, n - 1 - p)
        kB_hi = min(2 * n - 2, 2 * n - 2 - p)
        b_ks = np.arange(kB_lo, kB_hi + 1) if kB_lo <= kB_hi else np.array([], dtype=int)
        b_jv = b_ks - (n - 1) + p
        b_valid = (b_jv >= 0) & (b_jv < n)
        b_diag = b_valid & (b_jv == p)
        b_valid_nd = b_valid & ~b_diag
        cache[p] = (a_ks, a_iv, a_valid_nd, a_diag, b_ks, b_jv, b_valid_nd, b_diag)
    return cache

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=30, gs_iters=35):
    h = h.copy()
    idx_cache = _make_indices(n)
    D = (np.arange(2 * n - 1) == (n - 1)).astype(float)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            a_ks, a_iv, a_vn, a_d, b_ks, b_jv, b_vn, b_d = idx_cache[p]
            A = np.zeros(2 * n - 1)
            if np.any(a_vn):
                A[a_ks[a_vn]] += (1.0 - h[a_iv[a_vn]])
            if np.any(a_d):
                A[n - 1] += 1.0
            if np.any(b_vn):
                A[b_ks[b_vn]] += (-h[b_jv[b_vn]])
            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.array([float(np.max(c0 + t * A - t * t * D)) for t in grid])
            bi = int(np.argmin(fvals))
            lo = grid[max(0, bi - 1)]
            hi = grid[min(grid_pts - 1, bi + 1)]
            if hi - lo < 1e-14:
                t_star = grid[bi]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                d1 = a + gr * (b - a)
                fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        b = d1; d1 = c1; fd1 = fc1
                        c1 = b - gr * (b - a)
                        fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                    else:
                        a = c1; c1 = d1; fc1 = fd1
                        d1 = a + gr * (b - a)
                        fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                t_star = (a + b) * 0.5
            fs = float(np.max(c0 + t_star * A - t_star * t_star * D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            if min(fs, f0, f1) < current_max - 1e-14:
                if f0 <= fs and f0 <= f1:
                    t_new = 0.0
                elif f1 <= fs and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def pnorm_slsqp(h, n, p_list, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return (np.sum(c ** p) ** (1.0 / p)) * (2.0 / n)
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def binary_search(h, n, max_iters, deadline, rng):
    h = (h > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target = n // 2
    if len(ones) > target:
        h[rng.choice(ones, len(ones) - target, replace=False)] = 0
    elif len(ones) < target:
        h[rng.choice(zeros, target - len(ones), replace=False)] = 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    M = float(c.max())
    best_M = M
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        p = zeros[rng.randint(len(zeros))]
        q = ones[rng.randint(len(ones))]
        dc = np.zeros(2 * n - 1)
        k_lo = max(0, p)
        k_hi = min(2 * n - 2, p + n - 1)
        ks = np.arange(k_lo, k_hi + 1)
        jv = p + (n - 1) - ks
        v = (jv >= 0) & (jv < n) & (jv != p)
        if v.any():
            dc[ks[v]] += (1.0 - h[jv[v]])
        k_lo2 = max(0, n - 1 - p)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - p)
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        jv2 = ks2 - (n - 1) + p
        v2 = (jv2 >= 0) & (jv2 < n) & (jv2 != p)
        if v2.any():
            dc[ks2[v2]] -= h[jv2[v2]]
        k_lo = max(0, q)
        k_hi = min(2 * n - 2, q + n - 1)
        ks = np.arange(k_lo, k_hi + 1)
        jv = q + (n - 1) - ks
        v = (jv >= 0) & (jv < n) & (jv != q)
        if v.any():
            dc[ks[v]] -= (1.0 - h[jv[v]])
        k_lo2 = max(0, n - 1 - q)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - q)
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        jv2 = ks2 - (n - 1) + q
        v2 = (jv2 >= 0) & (jv2 < n) & (jv2 != q)
        if v2.any():
            dc[ks2[v2]] += h[jv2[v2]]
        new_M = float((c + dc).max())
        if new_M < M - 1e-10:
            h[p], h[q] = 1.0, 0.0
            c += dc
            M = new_M
            if M < best_M:
                best_M = M
                best_h = h.copy()
        elif new_M <= M + 1e-10 and rng.rand() < 0.3:
            h[p], h[q] = 1.0, 0.0
            c += dc
            M = new_M
    return best_h, best_M * (2.0 / n)

def interp_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Binary local search
    h_bin = (best_h > 0.5).astype(float)
    h_b, c_b = binary_search(h_bin, n_init, 300000, min(deadline, t0 + 200), rng)
    if c_b < best_c:
        best_h, best_c = h_b, c_b
    for tr in range(4):
        if time.time() > min(deadline, t0 + 350):
            break
        hr = rng.randint(0, 2, n_init).astype(float)
        hr = project_feasible(hr, n_init)
        h_b2, c_b2 = binary_search(hr, n_init, 150000, min(deadline, t0 + 400), rng)
        if c_b2 < best_c:
            best_h, best_c = h_b2, c_b2
    # Phase 2: Continuous optimization from best
    h_cont = project_feasible(best_h + rng.normal(0, 0.03, n_init), n_init)
    h2, c2 = pnorm_slsqp(h_cont, n_init, [8, 30, 100, 500], 350, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Exact CD
    if time.time() < deadline - 80:
        h3, c3 = exact_cd(best_h, n_init, 300, deadline, grid_pts=35, gs_iters=35)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-resolution
    n_list = [400, 450, 480, 520, 550, 600, 650, 700, 800, 900, 1000]
    for nt in n_list:
        if time.time() > deadline - 120:
            break
        if nt == n_init:
            continue
        hi = interp_h(best_h, len(best_h), nt)
        ho, co = pnorm_slsqp(hi, nt, [30, 200, 1000], 300, deadline)
        ho, co = exact_cd(ho, nt, 150, deadline, grid_pts=30, gs_iters=30)
        if co < best_c:
            best_h, best_c = ho, co
    # Phase 5: Final high-p SLSQP + CD
    if time.time() < deadline - 40:
        n = len(best_h)
        hf, cf = pnorm_slsqp(best_h, n, [2000, 10000, 50000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 15:
        hf, cf = exact_cd(best_h, len(best_h), 300, deadline, grid_pts=50, gs_iters=45)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
