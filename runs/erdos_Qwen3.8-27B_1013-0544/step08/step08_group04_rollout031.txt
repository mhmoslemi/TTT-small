    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def lgb_opt(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c5(x)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def optimize_coord(hp, cseg, asg, qconst, unaff):
    tol = 1e-10
    m = len(asg)
    b = cseg - asg * hp
    pos = asg > 1e-12
    neg = asg < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    low = max(0.0, unaff, qconst, zmax)
    high = max(low + 1e-10, unaff, qconst)
    if m:
        high = max(high, float(np.max(cseg)), float(np.max(cseg + asg)))
    high += 1e-8
    a_p, a_n = asg[pos], asg[neg]
    b_p, b_n = b[pos], b[neg]
    pc, nc = len(a_p), len(a_n)
    for _ in range(25):
        mid = 0.5 * (low + high)
        feas = False
        if mid >= unaff - tol and mid >= zmax - tol:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - b_n) / a_n))
                if v > ll: ll = v
            if pc:
                v = float(np.min((mid - b_p) / a_p))
                if v < lh: lh = v
            if ll <= 1.0 + tol and lh >= -tol:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + tol
                elif g > -1e-12:
                    if g < 0: g = 0
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 1.0 - r1
                    if max(0.0, ll) <= min(r1, lh) + tol:
                        feas = True
                    elif max(r2, ll) <= min(1.0, lh) + tol:
                        feas = True
        if feas: high = mid
        else: low = mid
    T = high
    cands = [0.0, 1.0, hp, 0.5]
    if T >= qconst + 0.25 - tol:
        ll, lh = 0.0, 1.0
        if nc: ll = max(ll, float(np.max((T - b_n) / a_n)))
        if pc: lh = min(lh, float(np.min((T - b_p) / a_p)))
        if ll <= 1.0 + tol and lh >= -tol:
            cands.extend([max(0.0, ll), min(1.0, lh), 0.5 * (max(0.0, ll) + min(1.0, lh))])
    else:
        g = max(0.0, T - qconst)
        r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
        r2 = 1.0 - r1
        cands.extend([r1, r2, max(0.0, r1 - 0.05), min(1.0, r2 + 0.05)])
    def _eval(x):
        v = max(unaff, qconst + x * (1.0 - x))
        if m: v = max(v, float(np.max(cseg + asg * x)))
        return v
    best_x, best_f = hp, _eval(hp)
    for x in cands:
        x = float(np.clip(x, 0, 1))
        f = _eval(x)
        if f < best_f - 1e-13:
            best_f, best_x = f, x
    if best_f > high + 1e-5:
        xg = np.linspace(0, 1, 40)
        vals = np.maximum(qconst + xg * (1.0 - xg), unaff)
        if m:
            vals = np.maximum(vals, np.max(cseg[None, :] + asg[None, :] * xg[:, None], axis=1))
        idx = int(np.argmin(vals))
        if vals[idx] < best_f:
            best_f, best_x = float(vals[idx]), float(xg[idx])
    return best_x, best_f

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=False):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c5 = h.copy(), cur * 2.0 / n
    center = n - 1
    L = 2 * n - 1
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for p in order:
            if time.time() > deadline:
                break
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            lag = np.arange(lo, hi + 1) - center
            j = p + lag
            i = p - lag
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj): a[vj] += 1.0 - h[j[vj]]
            if np.any(vi): a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1)
            asg = np.empty(seg - 1)
            if off:
                cseg[:off] = c[lo:center] - a[:off] * hp
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1] - a[off+1:] * hp
                asg[off:] = a[off+1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0: unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L: unaff = max(unaff, float(c[hi+1:].max()))
            bx, bf = optimize_coord(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-12:
                dx = bx - hp
                if off: c[lo:center] += asg[:off] * dx
                if hi > center: c[center+1:hi+1] += asg[off:] * dx
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cv = cur * 2.0 / n
        if cv < best_c5 - 1e-13:
            best_c5 = cv
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def full_optimize(h0, n, deadline, use_lgb=True, cd_passes=6, random_cd=True):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    rng = np.random.RandomState(int(time.time() * 1000) % 10000)
    if use_lgb and n <= 500 and time.time() < deadline - 60:
        h2, c2 = lgb_opt(h, n, (30, 300, 3000), 60, deadline)
        if c2 < c: h, c = h2, c2
    if time.time() < deadline - 30:
        h2, c2 = coord_descent(h, n, cd_passes, deadline, rng, random_cd)
        if c2 < c: h, c = h2, c2
    if use_lgb and n <= 400 and time.time() < deadline - 50:
        h3, c3 = lgb_opt(h, n, (200, 2000), 40, deadline)
        if c3 < c: h, c = h3, c3
    if time.time() < deadline - 20:
        h4, c4 = coord_descent(h, n, 4, deadline, rng, random_cd)
        if c4 < c: h, c = h4, c4
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 15.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 300
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = project_feasible(1.0 - np.abs(x - 1.0), n0)
    n_init = len(h_init)
    best_h, best_c = full_optimize(h_init, n_init, deadline, use_lgb=True, cd_passes=8, random_cd=True)

    # Try multiple smaller n values - these are likely to give better bounds
    n_targets = [n_init, 350, 300, 250, 200, 180, 150, 128, 100]
    for nt in n_targets:
        if time.time() > deadline - 180:
            break
        if nt == n_init or nt > n_init:
            continue
        h0 = resample_h(best_h, nt)
        remaining = deadline - time.time()
        cd_p = min(8, max(3, int(remaining / 60)))
        use_l = nt <= 350
        h1, c1 = full_optimize(h0, nt, deadline, use_lgb=use_l, cd_passes=cd_p, random_cd=True)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Random restarts with diverse initializations
    for r in range(40):
        if time.time() > deadline - 80:
            break
        if best_c <= 0.38070:
            break
        nt = len(best_h) if r % 3 == 0 else rng.choice([150, 200, 250, 300])
        base = resample_h(best_h, nt)
        typ = r % 5
        if typ == 0:
            h0 = project_feasible(base + rng.normal(0, 0.03, nt), nt)
        elif typ == 1:
            idx = np.argpartition(base, nt - nt // 2)[nt - nt // 2:]
            h0 = np.zeros(nt); h0[idx] = 1.0
        elif typ == 2:
            h0 = np.zeros(nt)
            idx = rng.choice(nt, nt // 2, replace=False)
            h0[idx] = 1.0
        elif typ == 3:
            # Informed perturbation: push h where correlation is highest
            c = np.correlate(base, 1.0 - base, mode='full')
            m_star = int(np.argmax(c))
            s = (nt - 1) - m_star
            h0 = base.copy()
            if s >= 0 and s < nt:
                h0[:nt - s] -= 0.05 * base[s:]
                h0[s:] += 0.05 * base[:nt - s]
            elif s < 0:
                r2 = -s
                h0[r2:] += 0.05 * base[:nt - r2]
                h0[:nt - r2] -= 0.05 * base[r2:]
            h0 = project_feasible(h0, nt)
        else:
            x = (np.arange(nt) + 0.5) * (2.0 / nt)
            t = x - 1.0
            h0 = 0.5 + 0.3 * np.cos(np.pi * t) + 0.15 * np.cos(2 * np.pi * t * rng.uniform(0.5, 1.5))
            h0 = project_feasible(h0, nt)
        use_l = nt <= 300 and time.time() < deadline - 150
        h1, c1 = full_optimize(h0, nt, deadline, use_lgb=use_l, cd_passes=3 if nt > 200 else 5, random_cd=True)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Final refinement on best solution
    if time.time() < deadline - 40:
        best_h, best_c = full_optimize(best_h, len(best_h), deadline, use_lgb=True, cd_passes=10, random_cd=False)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
