def weighted_grad(w,h,n):
  grad=np.zeros(n)
  center=n-1
  wc=w[center]
  if abs(wc)>1e-15: grad += wc*(1-2*h)
  for q in range(1,n):
     A=w[center-q]; B=w[center+q]
     if abs(A)+abs(B) < 1e-14: continue

        grad[q:] += -A*v
     if B != 0:
        grad[:L] += -B*u
        grad[q:] += B*(1.0-v)
  return grad

def project_feasible(h,n):
  h=np.clip(h,0,1)
  S=n/2
  if abs(h.sum()-S)<1e-15: return h
  lo=-1.0; hi=1.0  # shift range enough
  for _ in range(80):
    mid=(lo+hi)/2
    y=np.clip(h+mid,0,1)
    s=y.sum()
    if s < S: lo=mid
    else: hi=mid
  h=np.clip(h+(lo+hi)/2,0,1)
  # final adjust one element
  diff=S-h.sum()

    else: idx=np.argmax(h); h[idx]=max(0,h[idx]+diff)

def project_feasible(h,n):

    target=n/2.0
    s=float(h.sum())
    if abs(s-target) <= 1e-13: return h
    lo=-1.0; hi=1.0
    for _ in range(60):
       mid=(lo+hi)*0.5

       sy=float(y.sum())
       if sy < target: lo=mid
       else: hi=mid
    h=np.clip(h+0.5*(lo+hi),0,1)

       else:
          idx=int(np.argmax(h)); val=max(0.0, h[idx]+diff); h[idx]=val

    center_base = float(base[center_pos])

    unaff = 0.0
    if m0 > 0:
        unaff = max(unaff, float(np.max(c[:m0])))
    if m1 < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m1 + 1:])))

    cur_max = max(unaff, float(np.max(base)))
    if cur_max <= 0.0:
        return False

    nc = np.ones(len(ms), dtype=bool)
    nc[center_pos] = False
    ac = a[nc]
    bc = base[nc]

    eps = 1e-13
    neg = ac < -eps
    pos = ac > eps
    zero = ~(neg | pos)

    zmax = 0.0
    if np.any(zero):
        zmax = float(np.max(bc[zero]))

    has_neg = bool(np.any(neg))
    has_pos = bool(np.any(pos))
    bneg = ac[neg] if has_neg else None
    bnc = bc[neg] if has_neg else None
    bpos = ac[pos] if has_pos else None
    bpc = bc[pos] if has_pos else None

    Llo = max(unaff, zmax)
    Lhi = cur_max
    for _ in range(iters):
        L = 0.5 * (Llo + Lhi)
        feasible = L >= zmax - 1e-13
        lo = 0.0
        hi = 1.0

        if feasible and has_neg:
            val = x0 + float(np.max((L - bnc) / bneg))
            if val > lo:
                lo = val
        if feasible and has_pos:
            val = x0 + float(np.min((L - bpc) / bpos))
            if val < hi:
                hi = val

        if lo < 0.0:
            lo = 0.0
        if hi > 1.0:
            hi = 1.0
        if feasible and lo > hi + 1e-13:
            feasible = False

        if feasible:
            Cq = L - center_base + x0 - x0 * x0
            D = 1.0 - 4.0 * Cq
            if D >= -1e-13:
                if D < 0.0:
                    D = 0.0
                r1 = 0.5 * (1.0 - np.sqrt(D))
                r2 = 0.5 * (1.0 + np.sqrt(D))
                if lo > r1 + 1e-12 and hi < r2 - 1e-12:
                    feasible = False

        if feasible:
            Lhi = L
        else:
            Llo = L

    L = Lhi + 1e-12
    lo = 0.0
    hi = 1.0
    if has_neg:
        lo = max(lo, x0 + float(np.max((L - bnc) / bneg)))
    if has_pos:
        hi = min(hi, x0 + float(np.min((L - bpc) / bpos)))
    if lo < 0.0:
        lo = 0.0
    if hi > 1.0:
        hi = 1.0
    if lo > hi + 1e-13:
        return False

    Cq = L - center_base + x0 - x0 * x0
    D = 1.0 - 4.0 * Cq
    if D >= -1e-13:
        if D < 0.0:
            D = 0.0
        r1 = 0.5 * (1.0 - np.sqrt(D))
        r2 = 0.5 * (1.0 + np.sqrt(D))

        cand = []
        if hi <= r1 + 1e-13:
            cand.append(hi)
        elif lo >= r2 - 1e-13:
            cand.append(lo)
        else:
            if lo <= r1 + 1e-13:
                cand.append(0.5 * (lo + min(hi, r1)))
            if hi >= r2 - 1e-13:
                cand.append(0.5 * (max(lo, r2) + hi))

        if not cand:
            cand.append(x0)
        x = cand[0]
        best_d = abs(x - x0)
        for y in cand[1:]:
            d = abs(y - x0)
            if d < best_d:
                best_d = d
                x = y
    else:
        x = 0.5 * (lo + hi)

    x = float(np.clip(x, 0.0, 1.0))
    if abs(x - x0) <= 1e-13:
        return False

    delta = x - x0
    aff_new = base + delta * a
    center_new = center_base + x * (1.0 - x) - x0 * (1.0 - x0)
    aff_new[center_pos] = center_new
    max_aff = max(unaff, float(np.max(aff_new)))
    if max_aff >= cur_max - 1e-11:
        return False

    c[ms] += delta * a
    c[n - 1] = center_new
    h[p] = x
    return True


def exact_cd(h, n, max_passes, deadline, chunk=18, iters=32):
    h = project_feasible(h.copy(), n)
    c = make_c(h)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    chunk = max(1, int(chunk))

    for pass_idx in range(max_passes):
        if time.time() > deadline:
            break
        improved = 0
        counter = 0
        rev = (pass_idx % 2 == 1)
        iterator = range(n - 1, -1, -1) if rev else range(n)

        for p in iterator:
            if time.time() > deadline:
                break
            if update_one_coordinate(h, c, n, p, iters):
                improved += 1
            counter += 1
            if counter == n or counter % chunk == 0:
                h = project_feasible(h, n)
                c = make_c(h)
                cur = float(np.max(c) * dx)
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()

        if improved == 0:
            break

    return best_h, best_c


def weighted_grad(w, h, n):
    g = np.zeros(n, dtype=float)
    ctr = n - 1
    wc = float(w[ctr])
    if abs(wc) > 1e-16:
        g += wc * (1.0 - 2.0 * h)

    for q in range(1, n):
        A = float(w[ctr - q])
        B = float(w[ctr + q])
        if abs(A) < 1e-16 and abs(B) < 1e-16:
            continue
        L = n - q
        u = h[q:]
        v = h[:L]
        S = A + B
        g[:L] += A - S * u
        g[q:] += B - S * v
    return g


def smooth_valgrad(h, n, alpha):
    c = make_c(h)
    shift = alpha * c
    mx = float(np.max(shift))
    e = np.exp(shift - mx)
    se = float(np.sum(e))
    val = (mx + float(np.log(se))) / alpha * (2.0 / n)
    w = e / se
    grad = (2.0 / n) * weighted_grad(w, h, n)
    return val, grad


def obj_lse(x, n, alpha):
    val, _ = smooth_valgrad(x, n, alpha)
    return val


def jac_lse(x, n, alpha):
    _, grad = smooth_valgrad(x, n, alpha)
    return grad


def sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n, dtype=float)


def slsqp_opt(h, n, alphas, maxiter, stop_time):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_jac, "args": (n,)}]

    for alpha in alphas:
        if time.time() > stop_time:
            break
        try:
            res = minimize(
                obj_lse,
                best_h,
                args=(n, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max(1, int(maxiter)), "ftol": 1e-15, "iprint": 0},
            )
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h = new_h
                best_c = new_c
        except Exception:
            pass
    return best_h, best_c


def make_start(base, n, rng, mode):
    if base is not None:
        if mode == 0:
            return project_feasible(base + 0.04 * rng.uniform(-1.0, 1.0, n), n)
        if mode == 1:
            return project_feasible(base + 0.12 * rng.normal(0.0, 1.0, n), n)
        if mode == 2:
            return project_feasible(base + 0.25 * rng.normal(0.0, 1.0, n), n)

    t = 2.0 * np.pi * np.arange(n, dtype=float) / float(n)
    A = 0.35 * np.exp(-0.6 * np.arange(5, dtype=float))
    ph = rng.uniform(0.0, 2.0 * np.pi, 5)
    y = np.full(n, 0.5, dtype=float)
    for k in range(5):
        y += A[k] * np.cos((k + 1) * t + ph[k])

    if mode == 5:
        y = (rng.rand(n) < 0.5).astype(float)
    return project_feasible(y, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 700
        x = np.arange(n, dtype=float)
        h0 = 0.5 + 0.25 * np.cos(np.pi * x)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    chunk = max(8, n // 40)

    if deadline - time.time() > 60.0 and n <= 2000:
        stop = deadline - 300.0
        if stop < time.time() + 20.0:
            stop = deadline - 120.0
        maxp = max(4, int((stop - time.time()) / 8.0))
        h, c = exact_cd(best_h, n, maxp, stop, chunk, 32)
        if c < best_c:
            best_h, best_c = h, c

    now = time.time()
    if deadline - now > 160.0 and n <= 1000:
        stop = deadline - 130.0
        iters = max(8, min(22, int((stop - now) / 20.0)))
        h, c = slsqp_opt(best_h, n, [40.0, 300.0], iters, stop)
        if c < best_c:
            best_h, best_c = h, c

    now = time.time()
    if best_c > 0.38080 and deadline - now > 170.0:
        stop = deadline - 60.0
        mode = 0
        while mode < 6 and deadline - time.time() > 100.0:
            h_try = make_start(best_h, n, rng, mode)
            h_try, c_try = exact_cd(h_try, n, 4, stop, chunk, 26)
            if c_try < best_c:
                best_h, best_c = h_try, c_try
            mode += 1
            if best_c <= 0.38080:
                break

    now = time.time()
    if deadline - now > 35.0:
        h, c = exact_cd(best_h, n, 4, deadline - 2.0, chunk, 28)
        if c < best_c:
            best_h, best_c = h, c

    now = time.time()
    if deadline - now > 25.0 and n <= 1000:
        stop = deadline - 1.0
        iters = max(6, min(14, int((stop - now) / 12.0)))
        h, c = slsqp_opt(best_h, n, [1500.0], iters, stop)
        if c < best_c:
            best_h, best_c = h, c

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
