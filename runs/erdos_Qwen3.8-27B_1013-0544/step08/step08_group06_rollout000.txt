import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_two_interval(n, a, b, L1):
    h = np.zeros(n)
    dx = 2.0 / n
    x = np.arange(n) * dx
    L2 = 1.0 - L1
    h[(x >= a) & (x < a + L1)] = 1.0
    h[(x >= b) & (x < b + L2)] = 1.0
    return h

def make_three_interval(n, a, b, c, L1, L2):
    h = np.zeros(n)
    dx = 2.0 / n
    x = np.arange(n) * dx
    L3 = max(0.0, 1.0 - L1 - L2)
    h[(x >= a) & (x < a + L1)] = 1.0
    h[(x >= b) & (x < b + L2)] = 1.0
    h[(x >= c) & (x < c + L3)] = 1.0
    return project_feasible(h, n)

            break
        if adapt:
            gsize = min(200, max(20, grid_size + (p // 200) * 20))
        fwd = (p % 4 < 3)

            if deadline is not None and time.time() > deadline:

            m_lo = max(0, min(n - 1 - idx, idx))

            jh = idx + ms - (n - 1)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_jac(x, n):
    return np.ones(n)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_max_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def perturb_and_project(h, n, eps, rng):
    h2 = h + eps * rng.normal(size=n)
    return project_feasible(h2, n)
