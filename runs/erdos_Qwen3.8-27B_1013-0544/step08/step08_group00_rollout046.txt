    j = idx + m_off
    valid = (j >= 0) & (j < n)
    c_k = float(np.sum(h[idx[valid]] * (1.0 - h[j[valid]])))
    
    # Gradient
    grad = np.zeros(n)
    # dc/dx[p] for p in [0,n):
    # p appears as first factor (i=p): (1-h[p+m_off]) if 0 <= p+m_off < n
    # p appears as second factor (i+m_off=p, i=p-m_off): -h[p-m_off] if 0 <= p-m_off < n
    j1 = np.arange(n) + m_off
    v1 = (j1 >= 0) & (j1 < n)
    grad[v1] += (1.0 - h[j1[v1]])
    
    j2 = np.arange(n) - m_off
    v2 = (j2 >= 0) & (j2 < n)
    grad[v2] -= h[j2[v2]]
    
    return c_k, grad

        active = np.where(c > L_current - 0.02 * max(L_current, 1e-6))[0]

        if argmax_k not in active:

        x = cp.Variable(n)

            constraints.append(c_k + grad_k @ (x - h) <= L)

        prob = cp.Problem(obj, constraints)

        if x.value is None or L.value is None:

        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < L_current * (1 - 1e-10):
            h = new_h
        else:
            # No improvement, maybe we're at a local optimum of the cutting plane
            break

- Compute the subgradient of the max (i.e., the gradient of the correlation at the argmax shift, or a weighted average if multiple shifts are near max).
- Take a step in the negative subgradient direction.
- Project to the feasible set.
- Use a backtracking line search.

```python
def projected_subgradient(h, n, max_iters=10000, deadline=None):
    """Projected subgradient method with backtracking line search."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        L = float(c.max())
        
        # Find active set (shifts within tolerance of max)
        tol = 0.01 * max(L, 1e-6)
        active = np.where(c > L - tol)[0]
        
        # Subgradient: average of gradients at active shifts
        grad = np.zeros(n)
        for k in active:
            grad += compute_gradient_at_k(h, n, k)
        grad /= len(active)
        
        # Backtracking line search
        step = 1.0
        for _ in range(30):
            new_h = project_feasible(h - step * grad, n)
            new_c = compute_c5(new_h, n)
            if new_c < L:
                h = new_h
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                break
            step *= 0.5
        else:
            # No improvement even with tiny step
            break
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def grad_at_k(h, n, k):
    """Gradient of c[k] w.r.t. h."""
    m = k - (n - 1)
    grad = np.zeros(n)
    j1 = np.arange(n) + m
    v1 = (j1 >= 0) & (j1 < n)
    grad[v1] += (1.0 - h[j1[v1]])
    j2 = np.arange(n) - m
    v2 = (j2 >= 0) & (j2 < n)
    grad[v2] -= h[j2[v2]]
    return grad

def c_at_k(h, n, k):
    """Value of c[k]."""
    m = k - (n - 1)
    idx = np.arange(n)
    j = idx + m
    v = (j >= 0) & (j < n)
    return float(np.sum(h[idx[v]] * (1.0 - h[j[v]])))

def subgradient_method(h, n, max_iters, deadline, tol_frac=0.01):
    """Projected subgradient with backtracking."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        L = float(c.max())
        
        tol = tol_frac * max(L, 1e-6)
        active = np.where(c > L - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            grad += grad_at_k(h, n, k)
        grad /= len(active)
        
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break
        
        step = 2.0 / (norm_g * norm_g) if norm_g > 0 else 1.0
        # Actually for max of smooth functions, a fixed step with backtracking is better
        step = 0.1
        improved = False
        for _ in range(40):
            new_h = project_feasible(h - step * grad, n)
            new_c = compute_c5(new_h, n)
            if new_c < L - 1e-14:
                h = new_h
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try smaller tolerance
            tol_frac *= 0.5
            if tol_frac < 1e-6:
                break
    
    return best_h, best_c

def cutting_plane(h, n, max_iters, deadline):
    """Cutting plane / outer approximation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        L = float(c.max())
        
        # Active set
        tol = 0.01 * max(L, 1e-6)
        active = list(np.where(c > L - tol)[0])
        if not active:
            active = [int(np.argmax(c))]
        
        # Solve LP
        x = cp.Variable(n)
        Lvar = cp.Variable()
        constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2.0]
        
        for k in active:
            ck, gk = c_at_k(h, n, k), grad_at_k(h, n, k)
            constraints.append(ck + gk @ (x - h) <= Lvar)
        
        prob = cp.Problem(cp.Minimize(Lvar), constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK)
            except Exception:
                break
        
        if x.value is None:
            break
        
        new_h = project_feasible(np.clip(x.value, 0, 1), n)
        new_c = compute_c5(new_h, n)
        
        if new_c < best_c - 1e-12:
            best_c = new_c
            h = new_h
        else:
            # Try blending
            for alpha in [0.5, 0.3, 0.1]:
                blend = project_feasible(alpha * new_h + (1 - alpha) * h, n)
                bc = compute_c5(blend, n)
                if bc < best_c:
                    best_c = bc
                    h = blend
                    break
    
    return h, best_c
