            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_descend(h_int, n_try, max_passes=1500, deadline=deadline - 100, grid_size=80)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            x_orig = x_new
            # Quick refinement at this n
            if time.time() < deadline - 80:
                h_opt2, c_opt2 = multi_start_search(h_opt, n_try, rng, n_starts=5, deadline=deadline - 60)
                if c_opt2 < best_c:
                    best_h, best_c = h_opt2, c_opt2
                    x_orig = x_new

    # Phase 5: Also try random binary initializations at current best n
    n_best = len(best_h)
    if time.time() < deadline - 60:
        for _ in range(5):
            if time.time() > deadline - 40:
                break
            h_bin = random_binary_init(n_best, rng)
            h_b, c_b = coord_descend(h_bin, n_best, max_passes=800, deadline=deadline - 30, grid_size=60)
            if c_b < best_c:
                best_h, best_c = h_b, c_b

    # Phase 6: Final refinement
    if time.time() < deadline:
        h5, c5 = coord_descend(best_h, len(best_h), max_passes=5000, deadline=deadline, grid_size=120)
        if c5 < best_c:
            best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
