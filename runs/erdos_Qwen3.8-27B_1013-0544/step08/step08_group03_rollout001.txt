import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d(c0_line, c0_parab, A, n_m1):
    """Minimize max(c0_line + t*A, c0_parab + t - t^2) over [0,1]."""
    def f_of_t(t):
        vals = c0_line + t * A
        vals[n_m1] = c0_parab + t - t * t
        return float(np.max(vals))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc1 = f_of_t(c1)
    fd1 = f_of_t(d1)
    for _ in range(35):
        if fc1 < fd1:
            b = d1; d1 = c1; fd1 = fc1
            c1 = b - gr * (b - a); fc1 = f_of_t(c1)
        else:
            a = c1; c1 = d1; fc1 = fd1
            d1 = a + gr * (b - a); fd1 = f_of_t(d1)
    t_star = (a + b) * 0.5
    f_star = f_of_t(t_star)
    f0 = f_of_t(0.0)
    f1 = f_of_t(1.0)
    if f0 <= f_star and f0 <= f1:
        return 0.0, f0
    elif f1 <= f_star and f1 <= f0:
        return 1.0, f1
    else:
        return t_star, f_star

def vectorized_cd(h, n, max_passes, deadline, seed=0, sa_temp=0.0):

    h = h.copy()

    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(c.max())
    s_vals = np.arange(2 * n - 1) - n_m1
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for p in order:
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            # Compute A vector (coefficient of h[p] in each c[k])
            A = np.zeros(2 * n - 1)
            valid1 = (p + s_vals >= 0) & (p + s_vals < n)
            A[valid1] += 1.0 - h[p + s_vals[valid1]]
            valid2 = (s_vals != 0) & (p - s_vals >= 0) & (p - s_vals < n)
            A[valid2] += -h[p - s_vals[valid2]]
            A[n_m1] = 1.0 - 2.0 * h0
            
            c0_line = c - h0 * A
            c0_parab = c[n_m1] - h0 + h0 * h0
            
            t_star, f_star = solve_1d(c0_line, c0_parab, A, n_m1)
            
            if sa_temp > 0 and f_star >= current_max:
                accept_prob = np.exp(-(f_star - current_max) / (sa_temp + 1e-15))
                if rng.random() > accept_prob:
                    continue
            
            if f_star < current_max - 1e-15 or (sa_temp > 0 and t_star != h0):
                t_new = float(np.clip(t_star, 0.0, 1.0))
                for k_idx in range(2 * n - 1):
                    if k_idx == n_m1:
                        c[k_idx] = c0_parab + t_new - t_new * t_new
                    else:
                        c[k_idx] = c[k_idx] + (t_new - h0) * A[k_idx]
                h[p] = t_new
                current_max = float(np.max(c))

        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())

def subgradient_kick(h, n, eta, deadline):
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(c.max())
    n_m1 = n - 1
    s_vals = np.arange(2 * n - 1) - n_m1
    active = np.where(c >= current_max - 1e-10)[0]
    if len(active) == 0:
        return h
    grad = np.zeros(n)
    for k in active:
        s = k - n_m1
        for p in range(n):
            g = 0.0
            if 0 <= p + s < n:
                g += 1.0 - h[p + s]
            if s != 0 and 0 <= p - s < n:
                g += -h[p - s]
            if s == 0:
                g = 1.0 - 2.0 * h[p]
            grad[p] += g
    grad /= len(active)
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new

def pnorm_slsqp(h, n, p, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum(c ** p) ** (1.0 / p)) * dx
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
