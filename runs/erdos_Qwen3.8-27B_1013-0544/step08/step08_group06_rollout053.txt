**Structural insight**: At the optimum, if the max is achieved at multiple lags k_1, k_2, ..., k_m, then the "subgradient" is a convex combination of the gradients at those lags. The first-order optimality condition is that for any feasible direction d with sum(d) = 0, we have:
sum_k w_k * grad c_k(h) · d ≥ 0

where w_k are the weights on the active lags.

This means that at the optimum, the "net force" from the active lags should push h in a direction that's compatible with the constraints.

**Practical algorithm:**
1. Find active lags (those within epsilon of the max)
2. Compute the "combined gradient" G = sum_k w_k * grad c_k(h)
3. The steepest descent direction is to move h in the direction -G (projected onto the feasible set)
4. Since G is a vector, we can do: h_new = project(h - step * G)
5. The projection is: clip to [0,1], then adjust to satisfy sum = n/2

This is essentially a subgradient method! Let me implement it.

The gradient of c_k with respect to h[j]:
c_k = sum_i h[i] * (1 - h[i+k-(n-1)])

d(c_k)/d(h[j]) = (1 - h[j+k-(n-1)]) - h[j-k+(n-1)] (when the indices are valid)

So the combined gradient is:
G[j] = sum_k w_k * [(1 - h[j+k_k]) - h[j-k_k]] (with valid indices)

This is O(m * n) where m is the number of active lags.

Then we do:
h_new = project(h - eta * G)

The projection onto {0 ≤ h ≤ 1, sum(h) = n/2} can be done with the same `project_feasible` function.

```python
def subgradient_method(h, n, max_iters=5000, deadline=None, eta=0.01, active_tol=1e-4):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = c.max()
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Find active lags
        active = np.where(c >= current_max - active_tol * current_max)[0]
        m = len(active)
        if m == 0:
            active = np.array([np.argmax(c)])
            m = 1
        
        # Compute combined gradient
        G = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            # d(c_k)/d(h[j]) = (1 - h[j+offset]) - h[j-offset] (with valid indices)
            for j in range(n):
                term = 0.0
                j_pos = j + offset
                j_neg = j - offset
                if 0 <= j_pos < n:
                    term += (1.0 - h[j_pos])
                if 0 <= j_neg < n:
                    term -= h[j_neg]
                G[j] += term
        
        # Normalize gradient
        G_norm = np.sqrt(np.sum(G**2))
        if G_norm < 1e-15:
            break
        
        G /= G_norm  # Unit direction
        
        # Try a few step sizes
        best_h = h.copy()
        best_c = current_max
        for eta_trial in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]:
            h_trial = h - eta_trial * G * current_max  # Scale by current_max
            h_trial = project_feasible(h_trial, n)
            c_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            c_max_trial = c_trial.max()
            if c_max_trial < best_c:
                best_c = c_max_trial
                best_h = h_trial.copy()
        
        if best_c < current_max - 1e-10:
            h = best_h
            c = np.correlate(h, 1.0 - h, mode='full')
            current_max = c.max()
        else:
            # Decrease active tolerance or break
            if eta < 0.001:
                break
            # Try smaller step
            break
    
    return h, current_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def compute_gradient(h, n, active_lags):
    """Compute combined gradient of active lags."""
    G = np.zeros(n)
    for k in active_lags:
        offset = k - (n - 1)
        j1_lo = max(0, -offset)
        j1_hi = min(n - 1, n - 1 - offset)
        if j1_lo <= j1_hi:
            G[j1_lo:j1_hi + 1] += (1.0 - h[j1_lo + offset:j1_hi + offset + 1])
        j2_lo = max(0, offset)
        j2_hi = min(n - 1, offset + n - 1)
        if j2_lo <= j2_hi:
            G[j2_lo:j2_hi + 1] -= h[j2_lo - offset:j2_hi - offset + 1]
    return G

def subgradient_optimize(h, n, max_iters=5000, deadline=None, eta_init=0.5, beta=0.9):
    """Subgradient method with momentum for the max correlation."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    v = np.zeros(n)
    eta = eta_init
    stall = 0
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        active_tol = max(1e-6, 0.01 * current_max)
        active = np.where(c >= current_max - active_tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        G = compute_gradient(h, n, active)
        G_norm = np.sqrt(np.sum(G**2))
        if G_norm < 1e-12:
            break
        
        v = beta * v - eta * G / G_norm
        h_new = project_feasible(h + v * current_max, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_max_new = float(c_new.max())
        
        if c_max_new < current_max - 1e-10:
            h = h_new
            c = c_new
            current_max = c_max_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                eta *= 0.5
                v = np.zeros(n)
                stall = 0
                if eta < 1e-6:
                    break
    
    return h, current_max

def coord_cd(h, n, max_passes=2000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:

def pair_swap_optimize(h, n, max_iters=200, deadline=None):
    """Try to reduce the max lag by swapping mass between pairs."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        offset = k_star - (n - 1)
        if offset == 0:
            break
        # Find pairs (i, i+offset) where swapping helps
        improved = False
        j_lo = max(0, -offset)
        j_hi = min(n - 1, n - 1 - offset)
        for i in range(j_lo, j_hi + 1):
            j = i + offset
            hi_i, hi_j = h[i], h[j]
            # Moving delta from i to j changes c[k_star] by:
            # delta * (hi_i + hi_j - 1) - delta^2
            # We want this negative: delta * (hi_i + hi_j - 1) < delta^2
            # i.e., hi_i + hi_j < 1 + delta (approximately hi_i + hi_j < 1)
            if hi_i + hi_j < 0.99:
                # Moving from i to j helps c[k_star]
                delta = min(0.05, (hi_i - hi_j) * 0.5) if hi_i > hi_j else 0.02
                if delta <= 0:
                    continue
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                c_new = np.correlate(h, 1.0 - h, mode='full')
                if c_new.max() <= current_max + 1e-10:
                    c = c_new
                    current_max = float(c.max())
                    improved = True
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0, 1)
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def compute_active_gradient(h, n, k_star):
    offset = k_star - (n - 1)
    G = np.zeros(n)
    j1_lo = max(0, -offset)
    j1_hi = min(n - 1, n - 1 - offset)
    if j1_lo <= j1_hi:
        G[j1_lo:j1_hi + 1] += (1.0 - h[j1_lo + offset:j1_hi + offset + 1])
    j2_lo = max(0, offset)
    j2_hi = min(n - 1, offset + n - 1)
    if j2_lo <= j2_hi:
        G[j2_lo:j2_hi + 1] -= h[j2_lo - offset:j2_hi - offset + 1]
    return G

def subgrad_momentum(h, n, max_iters=5000, deadline=None, beta=0.85):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    v = np.zeros(n)
    eta = 0.3
    stall = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        active_tol = max(1e-7, 0.005 * current_max)
        active = np.where(c >= current_max - active_tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        G = np.zeros(n)
        for k in active:
            G += compute_active_gradient(h, n, k)
        G_norm = np.sqrt(np.sum(G * G))
        if G_norm < 1e-12:
            break
        v = beta * v - eta * G / G_norm
        h_new = project_feasible(h + v * current_max, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_max_new = float(c_new.max())
        if c_max_new < current_max - 5e-10:
            h, c = h_new, c_new
            current_max = c_max_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                eta *= 0.7
                v *= 0.5
                if eta < 1e-4:
                    break
    return h, current_max

def coord_cd(h, n, max_passes=2000, deadline=None, grid_size=100):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        idx_order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        improved = False
        for idx in idx_order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-10:
                lo_d, hi_d = 0.0, 1.0
            elif h0 > 1.0 - 1e-10:
                continue
            else:
                lo_d, hi_d = -h0, 1.0 - h0
            if hi_d - lo_d < 1e-12:
                continue
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            d_grid = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if has_diag:
                nv[:, diag_idx] -= d_grid[:, None] ** 2
            vals = np.maximum(unaff_max, nv.max(axis=1))
            best_i = int(np.argmin(vals))
            best_d = d_grid[best_i]
            best_val = float(vals[best_i])
            if 0 < best_i < grid_size - 1:
                y = vals[best_i - 1:best_i + 2]
                x = d_grid[best_i - 1:best_i + 2]
                denom = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(denom) > 1e-15:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])) / denom
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if has_diag:
                            nv1 = nv1.copy()
                            nv1[diag_idx] -= xv ** 2
                        v1 = max(unaff_max, float(nv1.max()))
                        if v1 < best_val - 1e-14:
                            best_val, best_d = v1, xv
            if abs(best_d) > 1e-10 and best_val < current_c5 - 1e-14:
                h[idx] = float(np.clip(h0 + best_d, 0.0, 1.0))
                c[ms] = c_aff + best_d * a
                if has_diag:
                    c[ms[diag_idx]] -= best_d ** 2
                new_max = max(unaff_max, float(c[ms].max()))
                if new_max < current_c5:
                    current_c5 = new_max
                improved = True
        if not improved:
            break
        if (p + 1) % 500 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
```
