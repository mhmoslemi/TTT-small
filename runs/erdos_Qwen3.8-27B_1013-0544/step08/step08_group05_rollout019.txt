    if abs(d) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.sum() > 0:
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def coord_descent(h, n, deadline, grid=40, max_passes=20, seed=0):
    """Coordinate descent with grid-based 1D optimization."""
    rng = np.random.RandomState(seed)
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    diag = n - 1
    xs0 = np.linspace(0.0, 1.0, grid)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        x = best.copy()
        c = np.correlate(x, 1.0 - x, mode="full")
        om = 1.0 - x[::-1]
        improved = False
        order = rng.permutation(n)
        for p in order:
            old = float(x[p])
            if old <= 1e-15 or old >= 1.0 - 1e-15:
                # Still try, but skip if clearly at boundary and not improving
                pass
            A = np.zeros(2 * n - 1)
            A[p:p + n] += om
            st = diag - p
            A[st:st + n] -= x
            cur = float(c.max())
            # Grid evaluation
            dd = xs0 - old
            cand = (c[None, :] + dd[:, None] * A[None, :])
            cand[:, diag] -= dd * dd
            vals = cand.max(axis=1)
            idx = int(np.argmin(vals))
            bx, bv = float(xs0[idx]), float(vals[idx])
            if bv < cur - 1e-13:
                # Refine around best
                sp = 1.0 / (grid - 1)
                lo_b, hi_b = max(0.0, bx - sp), min(1.0, bx + sp)
                xx = np.linspace(lo_b, hi_b, 16)
                dd2 = xx - old
                mm = (c[None, :] + dd2[:, None] * A[None, :])
                mm[:, diag] -= dd2 * dd2
                mm = mm.max(axis=1)
                j = int(np.argmin(mm))
                if mm[j] < bv - 1e-13:
                    bx, bv = float(xx[j]), float(mm[j])
                # Final refinement
                sp2 = (hi_b - lo_b) / 16.0
                lo2, hi2 = max(0.0, bx - sp2), min(1.0, bx + sp2)
                yy = np.linspace(lo2, hi2, 7)
                dd3 = yy - old
                ww = (c[None, :] + dd3[:, None] * A[None, :])
                ww[:, diag] -= dd3 * dd3
                ww = ww.max(axis=1)
                k = int(np.argmin(ww))
                if ww[k] < bv - 1e-13:
                    bx, bv = float(yy[k]), float(ww[k])
                if bv < cur - 1e-13:
                    d = bx - old
                    c[p:p + n] += d * om
                    c[st:st + n] -= d * x
                    c[diag] -= d * d
                    x[p] = bx
                    om[diag - p] = 1.0 - bx
                    improved = True
        xp = project_simplex(x, n)
        cp = compute_c5(xp, n)
        if cp < bc - 1e-14:
            best, bc = xp, cp
        if not improved:
            break
    return best, bc

    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0

def binary_anneal(b, n, rng, deadline, T0=0.01, T1=0.0001, max_iters=200000):
    """Simulated annealing on binary vectors."""
    dx = 2.0 / n
    best = b.copy()
    c = np.correlate(best, 1.0 - best, mode="full")
    bcm = float(c.max())
    h = best.copy()
    cm = bcm
    diag = n - 1
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        alpha = it / max(1, max_iters)
        T = T0 * (T1 / T0) ** alpha
        # Pick random one and zero
        i = ones[int(rng.randint(len(ones)))]
        j = zeros[int(rng.randint(len(zeros)))]
        # Compute new correlation
        om = 1.0 - h[::-1]
        c2 = c.copy()
        c2[i:i+n] -= om
        st_i = diag - i
        c2[st_i:st_i+n] += h
        c2[j:j+n] += om
        st_j = diag - j
        c2[st_j:st_j+n] -= h
        c2[diag] -= 2.0
        c2[diag + i - j] += 1.0
        c2[diag + j - i] += 1.0
        nm = float(c2.max())
        delta = nm - cm
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c = c2
            cm = nm
            h[i], h[j] = 0.0, 1.0
            om[diag - i], om[diag - j] = 1.0, 0.0
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            if cm < bcm:
                best, bcm = h.copy(), cm
    return best, bcm * dx

def smooth_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    w = ee / z
    dx = 2.0 / n
    A = np.convolve(1.0 - h, w, mode="full")
    B = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    val = (m + np.log(z) / alpha) * dx
    grad = (A - B)[s:s+n] * dx
    return val, grad

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj_func(x, a):
        return smooth_obj_grad(x, n, a)[0]
    def grad_func(x, a):
        return smooth_obj_grad(x, n, a)[1]
    def con_fun(x):
        return x.sum() - 0.5 * n
    def con_jac(x):
        return ones
    cons = [{"type": "eq", "fun": con_fun, "jac": con_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_func, best, args=(a,), jac=grad_func,
                          method="SLSQP", bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
            nh = project_simplex(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-14:
                best, bc = nh, nc
        except Exception:
            pass
    return best, bc

def transfer(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h.copy(), n_tgt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)

    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def binary_anneal(b, n, rng, deadline, T0=0.04, T1=0.002, max_iters=400000):
    dx = 2.0 / n
    best = b.copy()
    c = np.correlate(best, 1.0 - best, mode="full").astype(float)
    bcm = float(c.max())
    h = best.copy()
    cm = bcm
    diag = n - 1
    om = (1.0 - h[::-1]).copy()
    c_buf = np.zeros_like(c)
    ones = np.where(h == 0.0)[0].tolist()
    zero_list = np.where(h == 1.0)[0].tolist()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        alpha = min(1.0, it / max(1, max_iters))
        T = T0 * (T1 / T0) ** alpha
        i = ones[int(rng.randint(len(ones)))]
        j = zero_list[int(rng.randint(len(zero_list)))]
        c_buf[:] = c
        c[i:i + n] -= om
        st_i = diag - i
        c[st_i:st_i + n] += h
        c[j:j + n] += om
        st_j = diag - j
        c[st_j:st_j + n] -= h
        c[diag] -= 2.0
        c[diag + i - j] += 1.0
        c[diag + j - i] += 1.0
        nm = float(c.max())
        delta = nm - cm
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[i], h[j] = 0.0, 1.0
            om[diag - i], om[diag - j] = 1.0, 0.0
            ones.remove(i)
            zero_list.remove(j)
            cm = nm
            if cm < bcm:
                best, bcm = h.copy(), cm
        else:
            c[:] = c_buf
    return best, bcm * dx


def smooth_val_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx


def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
             "jac": lambda x: np.ones(n)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_val_grad, best, args=(n, a),
                           jac=True, method="SLSQP", bounds=bounds,
                           constraints=cons,
                           options={"maxiter": int(maxiter), "ftol": 1e-12})
            nh = project_simplex(res.x, n)
            nc = compute_c5(nh, n)
            if nc < bc - 1e-14:
                best, bc = nh, nc
        except Exception:
            pass
    return best, bc


def transfer(h, n, n_tgt):
    if n_tgt == n:
        return project_simplex(h.copy(), n_tgt)
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, n_tgt)
    return project_simplex(np.interp(y, x, h), n_tgt)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    h0 = project_simplex(h0, n0)
    best_h, best_c, n_best = h0.copy(), compute_c5(h0, n0), n0
    rev = h0[::-1]
    for v in (0.5 * (h0 + rev), 0.5 * (h0 + (1.0 - rev))):
        vv = project_simplex(v, n0)
        cv = compute_c5(vv, n0)
        if cv < best_c:
            best_h, best_c = vv, cv
    t1 = min(deadline - 250, t0 + 150)
    best_h, best_c = coord_descent(best_h, n_best, t1, grid=48, max_passes=12, seed=seed)
    t2 = min(deadline - 200, time.time() + 120)
    best_h, best_c = slsqp_polish(best_h, n_best, [3000, 15000, 60000, 200000], 60, t2)
    t3 = min(deadline - 150, time.time() + 150)
    best_h, best_c = coord_descent(best_h, n_best, t3, grid=64, max_passes=12, seed=seed + 1)
    if n_best % 2 == 0 and time.time() < deadline - 120:
        b0 = binarize(best_h, n_best)
        t4 = min(deadline - 100, time.time() + 250)
        b_opt, c_bin = binary_anneal(b0, n_best, np.random.RandomState(seed + 2), t4,
                                     T0=0.04, T1=0.002, max_iters=500000)
        if c_bin < best_c:
            best_h, best_c = b_opt, c_bin
            t5 = min(deadline - 60, time.time() + 80)
            best_h, best_c = coord_descent(best_h, n_best, t5, grid=56, max_passes=8, seed=seed + 3)
    for nt in (400, 600, 300, 450):
        if nt == n_best or nt < 100 or nt > 1000 or nt % 2:
            continue
        if time.time() > deadline - 80:
            break
        ht = transfer(best_h, n_best, nt)
        cv = compute_c5(ht, nt)
        chunk1 = min(deadline - 60, time.time() + 100)
        ht, cv = coord_descent(ht, nt, chunk1, grid=40, max_passes=8, seed=seed + nt)
        chunk2 = min(deadline - 40, time.time() + 80)
        b = binarize(ht, nt)
        b_opt, c_b = binary_anneal(b, nt, np.random.RandomState(seed + nt + 7), chunk2,
                                   T0=0.03, T1=0.002, max_iters=250000)
        if c_b < cv:
            ht, cv = b_opt, c_b
        if cv < best_c:
            best_h, best_c, n_best = ht, cv, nt
    if time.time() < deadline - 10:
        tf = min(deadline - 2, time.time() + 90)
        best_h, best_c = coord_descent(best_h, n_best, tf, grid=64, max_passes=12, seed=seed + 99)
    final_h = project_simplex(best_h, n_best)
    final_c = compute_c5(final_h, n_best)
    return final_h, final_c, int(n_best)
