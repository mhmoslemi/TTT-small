import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d(c, a, h0, n):
    d_lo = -h0
    d_hi = 1.0 - h0
    for _ in range(55):
        d = (d_lo + d_hi) / 2.0
        vals = c + d * a
        vals[n - 1] -= d * d
        f_val = vals.max()
        active = vals >= f_val - 1e-15
        active_idx = np.where(active)[0]
        sg = a[active_idx] - 2.0 * d * (active_idx == n - 1).astype(float)
        if sg.min() > 1e-14:
            d_hi = d
        elif sg.max() < -1e-14:
            d_lo = d
        else:
            return d
    return (d_lo + d_hi) / 2.0

def coord_descent(h, n, deadline):
    h = h.copy()
    offsets = np.arange(-(n - 1), n)
    for _ in range(500):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            pf = p + offsets
            pb = p - offsets
            valid_f = (pf >= 0) & (pf < n) & (offsets != 0)
            valid_b = (pb >= 0) & (pb < n)
            a = np.zeros(2 * n - 1)
            if np.any(valid_f):
                a[valid_f] = 1.0 - h[pf[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[pb[valid_b]]
            a[n - 1] = 1.0 - 2.0 * h0
            d = solve_1d(c, a, h0, n)
            if abs(d) > 1e-13:
                h[p] = float(np.clip(h0 + d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgrad_cd(h, n, deadline, n_iter=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    offsets = np.arange(-(n - 1), n)
    step = 0.1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        tol = max(1e-10, (c_max - c.min()) * 0.01)
        active = c >= c_max - tol
        n_act = max(1, int(active.sum()))
        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        for p in range(n):
            pf = p + offsets
            pb = p - offsets
            vf = (pf >= 0) & (pf < n)
            vb = (pb >= 0) & (pb < n)
            t1 = np.sum(active[vf] * one_minus_h[pf[vf]])
            t2 = np.sum(active[vb] * h[pb[vb]])
            grad[p] = (t1 - t2) / n_act
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        step *= 0.97
    return h, best_c
