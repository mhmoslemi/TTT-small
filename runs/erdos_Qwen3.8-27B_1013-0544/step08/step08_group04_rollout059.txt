import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        cnt = mask.sum()
        if cnt > 0:
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
    else:
        mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            step = diff / cnt
            h[mask] = np.maximum(0.0, h[mask] + step)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + res, 0, 1)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def obj_func(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * c
    zmax = z.max()
    return float((zmax + np.log(np.sum(np.exp(z - zmax)))) / alpha * (2.0 / n))

def obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * (c - c.max())
    w = np.exp(z)
    w /= w.sum()
    center = n - 1
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
        else:
            r = -s
            g[r:] += wm * (1.0 - h[:n-r])
            g[:n-r] -= wm * h[r:]
    return (2.0 / n) * g

def sum_con(h, n):
    return h.sum() - 0.5 * n

def sum_jac(h, n):
    return np.ones(n)

def lbfgs_optimize(h, n, alpha, maxiter, deadline):
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    res = minimize(obj_func, h, args=(n, alpha), method='L-BFGS-B',

def anneal_optimize(h, n, deadline, alphas=(10, 50, 200, 1000, 5000)):

        if c_new < best_c:

        maxiter = max(30, int(3000.0 / (alpha / 10.0)))

        c_new = compute_c(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    return best_h, best_c

def slsqp_round(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                   jac=smooth_grad, bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_feasible(res.x, n)

def coord_descend(h, n, max_passes, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c = float(c.max())
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            hp = h[p]
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            seg = hi - lo + 1
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(seg)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1)
            asg = np.empty(seg - 1)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center+1:hi+1]
                asg[off:] = a[off+1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi+1:].max()))
            # Bisection for optimal x
            lo_b, hi_b = max(0.0, unaff), min(1.0, max(unaff + 1e-11, qconst + hp*(1-hp), float(cseg.max()) if cseg.size else 0.0) + 1e-8)
            if hi_b <= lo_b:
                lo_b, hi_b = 0.0, 1.0
            a_pos = a[:seg-1] if off < seg - 1 else np.array([])
            # Simplified: just evaluate at a few candidate points
            cands = [hp, 0.0, 1.0, 0.5]
            if cseg.size:
                cands.append(float(np.clip(qconst, 0, 1)))
            # Find best candidate
            best_x = hp
            best_val = _eval_coord(hp, hp, cseg, asg, qconst, unaff)
            for cx in cands:
                cx = float(np.clip(cx, 0, 1))
                v = _eval_coord(cx, hp, cseg, asg, qconst, unaff)
                if v < best_val:
                    best_val = v
                    best_x = cx
            # Local refinement around best_x
            l = max(0.0, best_x - 0.1)
            r = min(1.0, best_x + 0.1)
            grid = np.linspace(l, r, 11)
            for cx in grid:
                v = _eval_coord(float(cx), hp, cseg, asg, qconst, unaff)
                if v < best_val:
                    best_val = v
                    best_x = float(cx)
            if best_val < float(c.max()) - 1e-10:
                dxv = best_x - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center+1:hi+1] += asg[off:] * dxv
                c[center] = qconst + best_x * (1.0 - best_x)
                h[p] = best_x
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        if cur < best_c - 1e-13:
            best_c = cur
            best_h = h.copy()
        if not improved:
            break
    return best_h, (2.0 / n) * best_c

def _eval_coord(x, hp, cseg, asg, qconst, unaff):
    if asg.size:
        vmax = float(np.max(cseg + asg * (x - hp)))
    else:
        vmax = 0.0
    qv = qconst + x * (1.0 - x)
    return max(vmax, qv, unaff)

def subgrad_explore(h, n, deadline, rng, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    step = 0.008
    center = n - 1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        cmax = float(c.max())
        active = np.where(c >= cmax - 1e-8)[0]
        if len(active) > 60:
            active = active[np.random.choice(len(active), 60, replace=False)]
        g = np.zeros(n)
        for m in active:
            s = center - m
            if s == 0:
                g += (1.0 - 2.0 * h)
            elif s > 0:
                g[:n-s] += (1.0 - h[s:])
                g[s:] -= h[:n-s]
            else:
                r = -s
                g[r:] += (1.0 - h[:n-r])
                g[:n-r] -= h[r:]
        g /= len(active)
        h_new = project_feasible(h - step * g, n)
        c_new_val = compute_c(h_new, n)
        if c_new_val < best_c:
            best_h, best_c = h_new.copy(), c_new_val
            h = h_new
            c = np.correlate(h, 1.0 - h, mode='full')
        else:
            if it % 30 == 0:
                step *= 0.85
            if it % 200 == 0:
                h = project_feasible(best_h + 0.02 * rng.randn(n), n)
                c = np.correlate(h, 1.0 - h, mode='full')
        if best_c <= 0.38070:
            break
    return best_h, best_c

def resample(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old, n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = triangle_init(400)
    n_init = len(h_init)
    if n_init % 2:
        n_init += 1
        h_init = resample(h_init, n_init)
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c(best_h, n_init)

    # Phase 1: L-BFGS-B anneal at initial resolution
    if time.time() < deadline - 400:
        h1, c1 = anneal(best_h, n_init, deadline - 350, alphas=(5, 20, 80, 300, 1500, 8000), pen=20.0)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP refine (exact constraint)
    if time.time() < deadline - 300 and n_init <= 700:
        h2, c2 = slsqp_round(best_h, n_init, 500, 60)
        c2 = compute_c(h2, n_init)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Coordinate descent polish
    if time.time() < deadline - 200:
        h3, c3 = coord_descend(best_h, n_init, 15, deadline - 180)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution
    for n_try in [max(100, n_init // 2), 256, 384, 512, 640, min(n_init * 2, 960)]:
        if time.time() > deadline - 150:
            break
        if n_try < 80 or n_try == n_init:
            continue
        h0 = resample(best_h, n_try)
        h_new, c_new = anneal(h0, n_try, deadline - 130, alphas=(10, 50, 200, 1000), pen=30.0)
        if c_new < best_c:
            h_cd, c_cd = coord_descend(h_new, n_try, 8, deadline - 120)
            if c_cd < c_new:
                h_new, c_new = h_cd, c_cd
            best_h, best_c = h_new, c_new
        if best_c <= 0.38075:
            break

    # Phase 5: Subgradient exploration
    if time.time() < deadline - 100:
        h4, c4 = subgrad_explore(best_h, len(best_h), deadline - 80, rng, n_iter=2000)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 6: Final coordinate descent
    if time.time() < deadline - 30:
        h5, c5 = coord_descend(best_h, len(best_h), 10, deadline - 15)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 15 and len(best_h) <= 600:
        h6 = slsqp_round(best_h, len(best_h), 2000, 40)
        c6 = compute_c(h6, len(best_h))
        if c6 < best_c:
            best_h, best_c = h6, c6

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c(best_h, len(best_h))
    return best_h, best_c, len(best_h)
