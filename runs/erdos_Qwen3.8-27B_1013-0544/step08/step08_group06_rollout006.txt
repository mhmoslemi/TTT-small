                den = y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1])
                if abs(den) > 1e-18:
                    xv = 0.5*(y[0]*(x[1]-x[2]) + y[1]*(x[2]-x[0]) + y[2]*(x[0]-x[1]))/den
                    if x[0] < xv < x[2]:
                        nv1 = c_aff + xv * a
                        if hd:
                            nv1 = nv1.copy()
                            nv1[di] -= xv**2
                        v1 = max(unaff, float(nv1.max()))
                        if v1 < bv:
                            bv = v1
                            bd = xv
            if abs(bd) > 1e-13 and bv < cur_max - 1e-15:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c_aff + bd * a
                if hd:
                    c[ms[dm]] -= bd**2
                nm = max(unaff, float(c[ms].max()))
                if nm < cur_max:
                    cur_max = nm
                improved = True
        if not improved:
            break
        if (p + 1) % 400 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(c.max())
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def randomized_restarts(best_h, best_c, n, deadline, rng, max_rounds=30):
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        h_p = best_h.copy()
        np_count = max(3, n // 40)
        idxs = rng.choice(n, np_count, replace=False)
        scale = 0.05 + 0.15 * (r / max(1, max_rounds))
        h_p[idxs] = np.clip(h_p[idxs] + rng.randn(np_count) * scale, 0, 1)
        h_p = project_feasible(h_p, n)
        rem = deadline - time.time()
        if rem < 25:
            break
        tl = min(80, rem * 0.4)
        ho, co = coord_cd(h_p, n, max_passes=3000, deadline=time.time() + tl, grid_size=120, rng=rng)
        if co < best_c:
            best_h, best_c = ho, co
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: SLSQP for quick smoothing
    if time.time() < deadline - 700:
        h1, c1 = slsqp_optimize(h, n, [500, 5000, 50000, 200000], maxiter=500)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Main coordinate descent
    h2, c2 = coord_cd(best_h, n, max_passes=8000, deadline=deadline - 350, grid_size=150, rng=rng)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Randomized restarts to escape local optima
    best_h, best_c = randomized_restarts(best_h, best_c, n, deadline - 180, rng, max_rounds=40)
    # Phase 4: Try different resolutions
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 600, 800]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_t, c_t = coord_cd(h_int, n_try, max_passes=3000, deadline=deadline - 100, grid_size=120, rng=rng)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
            x_orig = x_new
    # Phase 5: Final long refinement
    if time.time() < deadline - 10:
        h5, c5 = coord_cd(best_h, len(best_h), max_passes=8000, deadline=deadline, grid_size=200, rng=rng)
        if c5 < best_c:
            best_h, best_c = h5, c5
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
