            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            if np.any(valid_f):
                a_coeff[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a_coeff[valid_b] -= h[jb[valid_b]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo

            def obj1d(d):
                val = c_aff + d * a_coeff
                if diag_local is not None:
                    val[diag_local] -= d * d
                return max(float(val.max()), unaff_max)

            lo = -h0
            hi = 1.0 - h0
            grid = np.linspace(lo, hi, grid_pts)
            vals = np.array([obj1d(d) for d in grid])
            best_idx = int(np.argmin(vals))
            best_d = grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_pts - 1:
                gl = grid[best_idx - 1]
                gh = grid[best_idx + 1]
                try:
                    res = golden(obj1d, gl, gh, maxiter=gs_iter)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_round(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def projected_gradient_step(h, n, lr):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    sg = np.zeros(n)
    for i in range(n):
        j = i + m_star - (n - 1)
        if 0 <= j < n:
            sg[i] += (1.0 - h[j])
        ip = i - m_star + (n - 1)
        if 0 <= ip < n:
            sg[i] -= h[ip]
    return project_feasible(h - lr * sg, n)

def subgradient_polish(h, n, max_iter=500, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h_new = projected_gradient_step(h, n, lr)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.6
            stall += 1
            if stall > 50 or lr < 1e-12:
                break
    return h, best_c

def multires_optimize(h, n, deadline):
    coarses = [max(100, n // 3), max(150, n // 2)]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for nc in coarses:
        if time.time() > deadline - 300:
            break
        if nc >= n:
            continue
        x_orig = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, nc)
        h_c = project_feasible(np.interp(x_coarse, x_orig, best_h), nc)
        h_c, c_c = lbfgsb_run(h_c, nc, [100, 1000, 5000], 2000)
        if time.time() > deadline - 250:
            break
        h_c, c_c = coord_cd(h_c, nc, 200, deadline - 200, grid_pts=25, gs_iter=70)
        x_fine = np.linspace(0, 2, n)
        h_up = project_feasible(np.interp(x_fine, x_coarse, h_c), n)
        h_up, c_up = coord_cd(h_up, n, 100, deadline - 150, grid_pts=20, gs_iter=60)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing smoothing
    if time.time() < deadline - 600:
        h1, c1 = lbfgsb_run(best_h, n, [100, 500, 2000, 10000], 3000)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coordinate descent (main refinement)
    if time.time() < deadline - 200:
        h2, c2 = coord_cd(best_h, n, 300, deadline - 100, grid_pts=35, gs_iter=90)
        if c2 < best_c:
            best_h, best_c = h2, c1 if False else c2

    # Phase 3: Binary rounding trick
    if time.time() < deadline - 300:
        h_bin = binary_round(best_h, n)
        h_bin, c_bin = lbfgsb_run(h_bin, n, [50, 500, 5000], 2000)
        h_bin, c_bin = coord_cd(h_bin, n, 150, deadline - 200, grid_pts=30, gs_iter=80)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    # Phase 4: Multi-resolution
    if time.time() < deadline - 250:
        h4, c4 = multires_optimize(best_h, n, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Subgradient polish
    if time.time() < deadline - 100:
        h5, c5 = subgradient_polish(best_h, n, 400, deadline - 50)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Try different n values
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [400, 500, 700, 800, 1000]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = lbfgsb_run(h_int, n_try, [200, 2000, 10000], 2000)
        h_opt, c_opt = coord_cd(h_opt, n_try, 150, deadline - 50, grid_pts=30, gs_iter=80)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 7: Final high-precision coordinate descent
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        h7, c7 = coord_cd(best_h, n_cur, 1000, deadline - 10, grid_pts=40, gs_iter=100)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Phase 8: Final L-BFGS-B with very high alpha
    if time.time() < deadline - 20:
        h8, c8 = lbfgsb_run(best_h, n_cur, [50000, 200000], 5000)
        if c8 < best_c:
            best_h, best_c = h8, c8
            if time.time() < deadline - 10:
                h8b, c8b = coord_cd(h8, n_cur, 200, deadline - 5, grid_pts=40, gs_iter=100)
                if c8b < best_c:
                    best_h, best_c = h8b, c8b

    # Try 1-h flip
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
