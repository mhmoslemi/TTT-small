        n_init = 500
    n = n_init
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Softmax gradient descent
    if time.time() < deadline - 200:
        h_g, c_g = softmax_grad_descent(best_h, n, 20000, deadline - 100,
                                         T_init=0.5, T_end=0.1, lr_init=0.003)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

    # Phase 2: SLSQP p-norm
    for p_val in [50, 200, 1000]:
        if time.time() > deadline - 150:
            break
        h_s, c_s = slsqp_pnorm(best_h, n, p_val, 400, deadline - 100)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # Phase 3: Coordinate descent
    if time.time() < deadline - 80:
        h_cd, c_cd = exact_cd(best_h, n, 200, deadline - 30, grid_pts=35, gs_iters=35)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 4: Binary SA
    if time.time() < deadline - 120:
        h_bin = make_binary(best_h, n)
        sa_budget = min(500000, int((deadline - time.time() - 60) / 0.00002))
        h_b, c_b_raw = binary_sa(h_bin, n, sa_budget, deadline - 60)
        c_b = c_b_raw * (2.0/n)
        if c_b < best_c:
            best_h = project_feasible(h_b, n)
            best_c = compute_c5(best_h, n)
        # Refine binary result with CD
        if time.time() < deadline - 30:
            h_ref, c_ref = exact_cd(best_h, n, 100, deadline - 10, grid_pts=30, gs_iters=30)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # Phase 5: Multi-restart with different initializations
    if time.time() < deadline - 60:
        for restart in range(5):
            if time.time() > deadline - 20:
                break
            if restart == 0:
                hp = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
            elif restart == 1:
                hp = np.where(np.arange(n) % 4 < 2, 1.0, 0.0)
            elif restart == 2:
                hp = project_feasible(best_h + np.random.normal(0, 0.1, n), n)
            elif restart == 3:
                hp = project_feasible(np.random.uniform(0, 1, n), n)
            else:
                hp = project_feasible(best_h + np.random.normal(0, 0.05, n), n)
            h_r, c_r = slsqp_pnorm(hp, n, 200, 300, deadline - 30)
            h_r, c_r = exact_cd(h_r, n, 80, deadline - 15, grid_pts=25, gs_iters=25)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Phase 6: Final high-p SLSQP + CD polish
    if time.time() < deadline - 20:
        hf, cf = slsqp_pnorm(best_h, n, 5000, 400, deadline - 10)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 5:
        hf, cf = exact_cd(best_h, n, 100, deadline - 2, grid_pts=50, gs_iters=45)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
