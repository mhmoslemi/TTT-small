Actually, there's a simpler approach: since F(t) is convex and piecewise linear, its minimum is at a breakpoint or at a boundary. The breakpoints are where two pieces a_k + b_k*t = a_{k'} + b_{k'}*t, i.e., t = (a_{k'} - a_k)/(b_k - b_{k'}). There are O(L²) breakpoints, which is too many.

1. Evaluate F at t=0 and t=1
2. If F is flat, any t works
3. Otherwise, use ternary search (since F is convex)

Ternary search on [0,1] with O(log(1/ε)) iterations, each O(L) → O(L*log(1/ε)) per coordinate. Total: O(n*L*log(1/ε)) per pass.

def _sj(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best, bc = h.copy(), compute_c5(h, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sc, "jac": _sj, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            r = minimize(_sf, best, args=(n, a), jac=_sg, method="SLSQP",
                         bounds=bnds, constraints=cons,
                         options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(r.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best, bc = nh, nc
    return best, bc

def nesterov(h, n, deadline, iters=2000, top_k=8, step0=0.003):
    h = project_simplex(h, n)
    y = h.copy()
    best, bc = h.copy(), compute_c5(h, n)
    for t in range(iters):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode="full")
        idx = np.argpartition(c, -top_k)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            k = int(ki) - (n - 1)
            if k >= 0:
                g[:n - k] += (1.0 - y[k:])
                if k < n:
                    g[k:] -= y[:n - k]
            else:
                ka = -k
                g[:n - ka] -= y[ka:]
                g[ka:] += (1.0 - y[:n - ka])
        g /= top_k
        s = step0 / np.sqrt(t + 1)
        hn = project_simplex(y - s * g, n)
        at = (1.0 + np.sqrt(1.0 + 4.0 * (t + 1) ** 2)) / 2.0
        y = np.clip(hn + (at - 1.0) / (at + 1.0) * (hn - h), 0.0, 1.0)
        cn = compute_c5(hn, n)
        if cn < bc:
            bc, best = cn, hn.copy()
        h = hn
    return best, bc

import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-14:
        return h
    lo, hi = -2.0, 3.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max()) * (2.0 / n)

def _soft_vg(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = ee.sum()
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * (2.0 / n), (A - B)[s:s + n] * (2.0 / n)

def _sf(x, n, a):
    return _soft_vg(x, n, a)[0]

def _sg_x(x, n, a):
    return _soft_vg(x, n, a)[1]

def _sc(x, n):
    return x.sum() - 0.5 * n

def _sj(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter, deadline):
    h = project_simplex(h, n)
    best, bc = h.copy(), compute_c5(h, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sc, "jac": _sj, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            r = minimize(_sf, best, args=(n, a), jac=_sg_x, method="SLSQP",
                         bounds=bnds, constraints=cons,
                         options={"maxiter": maxiter, "ftol": 1e-12})
        except Exception:
            continue
        nh = project_simplex(r.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best, bc = nh, nc
    return best, bc

def nesterov(h, n, deadline, iters=3000, top_k=10, step0=0.002):
    h = project_simplex(h, n)
    y = h.copy()
    best, bc = h.copy(), compute_c5(h, n)
    for t in range(iters):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1.0 - y, mode="full")
        idx = np.argpartition(c, -top_k)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            k = int(ki) - (n - 1)
            if k >= 0:
                g[:n - k] += (1.0 - y[k:])
                if k < n:
                    g[k:] -= y[:n - k]
            else:
                ka = -k
                g[:n - ka] -= y[ka:]
                g[ka:] += (1.0 - y[:n - ka])
        g /= top_k
        s = step0 / np.sqrt(t + 1)
        hn = project_simplex(y - s * g, n)
        at = (1.0 + np.sqrt(1.0 + 4.0 * (t + 1) ** 2)) / 2.0
        y = np.clip(hn + (at - 1.0) / (at + 1.0) * (hn - h), 0.0, 1.0)
        cn = compute_c5(hn, n)
        if cn < bc:
            bc, best = cn, hn.copy()
        h = hn
    return best, bc

def binarize(h, n):
    if n % 2:
        return None
    target = n // 2
    if target <= 0 or target >= n:
        return None
    idx = np.argpartition(h, target - 1)[:target]
    b = np.zeros(n, dtype=float)

        T *= cool
    return best, bc * (2.0 / n)

def transfer_h(h, n, nt):
    if nt == n:
        return project_simplex(h, nt).copy()
    x = np.linspace(0.0, 2.0, n)
    y
    return project_simplex(np.interp(y, x, h), nt)
