def compute_c_lags(h):
    n = len(h)
    dx = 2.0/n
    m = 2*n-1
    d = np.arange(m) - (n-1)
    prefix = np.empty(n+1); prefix[0]=0; np.cumsum(h, out=prefix[1:])
    lin = np.empty(m)
    pos = d >= 0
    lin[pos] = prefix[n - d[pos]]
    lin[~pos] = prefix[n] - prefix[-d[~pos]]
    q = np.correlate(h, h, mode='full').astype(float)
    return (lin - q) * dx

def project_box_sum(v, S):
    v = np.array(v, dtype=float, copy=False)
    if S <= 0: return np.zeros_like(v)
    if S >= v.size: return np.ones_like(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
       mid = 0.5*(lo+hi)
       s = np.clip(v - mid, 0.0, 1.0).sum(dtype=np.float64)
       if s > S:
           lo = mid
       else:
           hi = mid
    tau = 0.5*(lo+hi)
    h = np.clip(v - tau, 0.0, 1.0)
    # exact residual using interior
    diff = S - float(h.sum(dtype=np.float64))
    if abs(diff) > 1e-15:
       if diff > 0:
          room = 1.0 - h
          tot = room.sum(dtype=np.float64)
          if tot > 0:
             h += (diff/tot)*room
       else:
          tot = h.sum(dtype=np.float64)
          if tot > 0:
             h += (diff/tot)*h
    return np.clip(h,0,1)

def compute_c_raw(h):
    n=len(h)
    m=2*n-1
    prefix=np.empty(n+1); prefix[0]=0.0
    np.cumsum(h,out=prefix[1:])
    d=np.arange(m)-(n-1)

    q=np.empty(m)
    q[n-1:m]=ac[0:n]
    q[0:n-1]=ac[1:n][::-1]
    return lin-q

def project_box_sum(v,S):
    v=np.asarray(v,dtype=float)
    nv=v.size
    if S<=0: return np.zeros(nv)
    if S>=nv: return np.ones(nv)
    lo=float(np.min(v))-1.0; hi=float(np.max(v))+1.0
    for _ in range(45):

       s=float(np.clip(v-mid,0.0,1.0).sum(dtype=np.float64))

import numpy as np
import time
import math


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)


def compute_c_raw(h):
    n = len(h)
    m = 2 * n - 1
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    d = np.arange(m) - (n - 1)
    lin = np.empty(m, dtype=float)
    pos = d >= 0
    lin[pos] = prefix[n - d[pos]]
    neg = ~pos
    lin[neg] = prefix[n] - prefix[-d[neg]]

    N = 2 * n
    F = np.fft.rfft(h, n=N)
    ac = np.fft.irfft(F * np.conj(F), n=N)
    q = np.empty(m, dtype=float)
    q[n - 1:m] = ac[0:n]
    q[0:n - 1] = ac[1:n][::-1]
    return lin - q


def smooth_weights_value(c, beta):
    mx = float(np.max(c))
    if beta <= 1e-3:
        w = np.full(c.size, 1.0 / c.size, dtype=float)
        return w, mx
    z = beta * (c - mx)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    w = e / s
    return w, float(mx + math.log(s) / beta)


def smooth_value(c, beta):
    mx = float(np.max(c))
    if beta <= 1e-3:
        return mx
    z = beta * (c - mx)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    return float(mx + math.log(s) / beta)


def weighted_grad(w, h):
    n = len(h)
    wrev = w[::-1]
    g1 = np.convolve(1.0 - h, wrev, mode="full")[n - 1:n - 1 + n]
    g2 = np.convolve(h, w, mode="full")[n - 1:n - 1 + n]
    return g1 - g2


def project_box_sum(v, S):
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    nv = v.size
    if S <= 0.0:
        return np.zeros(nv, dtype=float)
    if S >= float(nv):
        return np.ones(nv, dtype=float)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum(dtype=np.float64))
        if s > S:
            lo = mid
        else:
            hi = mid

    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(h.sum(dtype=np.float64))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum(dtype=np.float64))
            if tot > 0.0:
                h += (diff / tot) * room
        else:
            tot = float(h.sum(dtype=np.float64))
            if tot > 0.0:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def scalar_minimize_coordinate(A, base, idx_noncen, center_idx, current_max, bits=32):
    An = A[idx_noncen]
    Bn = base[idx_noncen]
    center_val = float(base[center_idx])

    if An.size:
        BnA = Bn + An
        f0 = max(float(np.max(Bn)), center_val)
        f1 = max(float(np.max(BnA)), center_val)
        lb = max(float(np.max(np.minimum(Bn, BnA))), center_val)
    else:
        BnA = np.array([], dtype=float)
        f0 = center_val
        f1 = center_val
        lb = center_val

    best_t = 0.0
    best_f = f0
    if f1 < best_f:
        best_t = 1.0
        best_f = f1

    hi = min(best_f, current_max)
    if hi >= current_max - 1e-12:
        return None, current_max

    eps = 1e-14
    pos = An > eps
    neg = An < -eps
    has_pos = bool(np.any(pos))
    has_neg = bool(np.any(neg))
    zero = ~(pos | neg)
    has_zero = bool(np.any(zero))

    Apos = An[pos] if has_pos else None
    Bpos = Bn[pos] if has_pos else None
    Aneg = An[neg] if has_neg else None
    Bneg = Bn[neg] if has_neg else None
    Bzero_max = float(np.max(Bn[zero])) if has_zero else -1e300

    lo = 0.0
    hi_val = hi
    t_best = None

    for _ in range(bits):
        tau = 0.5 * (lo + hi_val)
        L = 0.0
        R = 1.0
        bad = False

        if has_zero and Bzero_max > tau:
            bad = True

        if has_pos:
            up = float(np.min((tau - Bpos) / Apos))
            if up < R:
                R = up

        if has_neg:
            low = float(np.max((tau - Bneg) / Aneg))
            if low > L:
                L = low

        if L < 0.0:
            L = 0.0
        if R > 1.0:
            R = 1.0

        if not bad and L > R:
            bad = True

        if not bad:
            disc = 1.0 - 4.0 * (tau - center_val)
            if disc <= 0.0:
                t_best = 0.5 * (L + R)
            else:
                sq = math.sqrt(disc)
                r1 = 0.5 - 0.5 * sq
                r2 = 0.5 + 0.5 * sq
                R1 = r1 if r1 < R else R
                if L <= R1:
                    t_best = 0.5 * (L + R1)
                else:
                    L2 = r2 if r2 > L else L
                    if L2 <= R:
                        t_best = 0.5 * (L2 + R)
                    else:
                        bad = True

        if bad:
            lo = tau
        else:
            hi_val = tau

    cand_t = best_t
    cand_f = best_f

    if t_best is not None:
        t = float(np.clip(t_best, 0.0, 1.0))
        vals = base + t * A
        vals[center_idx] -= t * t
        f = float(np.max(vals))
        if f < cand_f - 1e-14:
            cand_t, cand_f = t, f

    if cand_f >= current_max - 1e-12:
        return None, current_max
    return cand_t, cand_f


def coordinate_polish(h, n, deadline, max_passes=20, bits=32):
    S = 0.5 * n
    scale = 0.5 * float(n)
    h = project_box_sum(np.asarray(h, dtype=float), S)
    c = compute_c_raw(h)
    curr = float(np.max(c))
    best_c = curr / scale
    best_h = h.copy()

    m = 2 * n - 1
    center = n - 1
    idx_noncen = np.concatenate((np.arange(center), np.arange(center + 1, m)))
    order = np.arange(n)
    shift = max(1, n // 13)

    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break

        order = np.roll(order, shift)
        c = compute_c_raw(h)
        curr = float(np.max(c))
        curr_c = curr / scale
        if curr_c < best_c:
            best_c = curr_c
            best_h = h.copy()

        sum_h = float(np.sum(h, dtype=np.float64))
        improved = False

        for pp in order:
            if deadline is not None and time.time() > deadline:
                break
            p = int(pp)
            h0 = float(h[p])

            A = np.zeros(m, dtype=float)
            A[center - p:center - p + n] += 1.0 - h
            A[p:p + n] -= h[::-1]
            A[center] = 1.0

            base = c - h0 * A
            base[center] = c[center] - h0 + h0 * h0

            t_new, f_new = scalar_minimize_coordinate(A, base, idx_noncen, center, curr, bits)
            if t_new is not None and f_new < curr - 1e-10:
                old = h0
                h[p] = t_new
                c = base + t_new * A
                c[center] -= t_new * t_new
                curr = f_new
                sum_h += t_new - old
                improved = True

                if abs(sum_h - S) > 0.5:
                    h = project_box_sum(h, S)
                    c = compute_c_raw(h)
                    curr = float(np.max(c))
                    sum_h = float(np.sum(h, dtype=np.float64))
                    cand_c = curr / scale
                    if cand_c < best_c:
                        best_c = cand_c
                        best_h = h.copy()
                    elif cand_c > best_c + 1e-9:
                        h = best_h.copy()
                        c = compute_c_raw(h)
                        curr = best_c * scale
                        break

        if improved:
            h = project_box_sum(h, S)
            c = compute_c_raw(h)
            curr = float(np.max(c))
            cand_c = curr / scale
            if cand_c < best_c:
                best_c = cand_c
                best_h = h.copy()
            else:
                h = best_h.copy()
        else:
            break

    best_h = project_box_sum(best_h, S)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def smooth_optimize(h, n, deadline, betas, max_iter_total=800):

            cmax = float(np.max(c))

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_box_sum(np.asarray(h_old, dtype=float), 0.5 * n_new)
    x0 = np.linspace(0.0, 2.0, n_old)
    x1 = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x1, x0, h_old)
    return project_box_sum(h_new, 0.5 * n_new)


def random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    target = 0.5 * n
    if abs(target - round(target)) < 1e-12:
        s = int(float(h.sum()))
        diff = int(round(target)) - s
        if diff > 0:
            idx = np.flatnonzero(h < 0.5)
            if idx.size >= diff:
                h[rng.permutation(idx)[:diff]] = 1.0
            else:
                return project_box_sum(h, target)
        elif diff < 0:
            idx = np.flatnonzero(h > 0.5)
            k = -diff
            if idx.size >= k:
                h[rng.permutation(idx)[:k]] = 0.0
            else:
                return project_box_sum(h, target)
        return h
    return project_box_sum(h, target)


def push_extremes(h, gain):
    return np.clip(0.5 + (np.asarray(h, dtype=float) - 0.5) * gain, 0.0, 1.0)

                max_iter_total=350
            )
            h_try, c_try = coordinate_polish(h_try, n_try, deadline, max_passes=12, bits=32)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, n_try

    for r in range(6):
        if deadline - time.time() < 160.0:
            break
        n_best = best_n
        if r == 0:
            h_try = push_extremes(best_h, 2.0)
        elif r == 1:
            h_try = push_extremes(best_h, 4.0)
        elif r == 2:
            h_try = np.asarray(best_h, dtype=float) + rng.normal(0.0, 0.08, n_best)
        elif r == 3:
            h_try = np.asarray(best_h, dtype=float) + rng.normal(0.0, 0.20, n_best)
        elif r == 4:
            h_try = random_binary(n_best, rng)
        else:
            h_try = random_binary(n_best, rng)

        h_try = project_box_sum(h_try, 0.5 * n_best)
        h_try, c_try = coordinate_polish(h_try, n_best, deadline, max_passes=6, bits=30)
        if c_try < best_c + 0.002 and deadline - time.time() > 120.0:
            h_try, c_try = smooth_optimize(
                h_try, n_best, deadline,
                betas=[80.0, 300.0],
                max_iter_total=350
            )
            h_try, c_try = coordinate_polish(h_try, n_best, deadline, max_passes=12, bits=32)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, n_best

    if deadline - time.time() > 60.0:
        passes = 30 if best_n <= 600 else 15
        h_try, c_try = coordinate_polish(best_h, best_n, deadline, max_passes=passes, bits=40)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, best_n

    best_n = len(best_h)
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
