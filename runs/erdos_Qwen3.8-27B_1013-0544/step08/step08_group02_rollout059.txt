import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0: return z
    lo, hi = z.min() - 2, z.max() + 2
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0, 1).sum()
        if s > target: lo = mid
        else: hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0, 1)
    r = target - y.sum()
    if abs(r) > 1e-13:
        if r > 0:
            m = y < 1; room = (1-y)*m
            if room.sum() > 1e-12: y += r/room.sum()*room
        else:
            m = y > 0; tot = (y*m).sum()
            if tot > 1e-12: y += r/tot*y*m
    return np.clip(y, 0, 1)

def make_grad(h, n, p):
    """Compute gradient of smooth max using probabilities p over lags."""
    g = np.zeros(n)
    off = n - 1
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-14: continue
        d = k - off
        if d >= 0:
            g[d:] += w * (1.0 - h[:n-d])
            g[:n-d] -= w * h[d:]
        else:
            e = -d
            g[:n-e] += w * (1.0 - h[e:])
            g[e:] -= w * h[:n-e]
    return (2.0/n) * g

def smooth_obj(h, n, alpha):
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    z = alpha*(c - m); zm = z.max()
    e = np.exp(z - zm); s = e.sum()
    return (m + np.log(s)/alpha) * (2.0/n), e/s

def sum_fun(h, n): return h.sum() - 0.5*n
def sum_jac(h, n): return np.ones(n)

def slsqp_opt(h, n, deadline, alphas):
    bnds = [(0,1)]*n
    cons = [{'type':'eq','fun':sum_fun,'jac':sum_jac,'args':(n,)}]
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha, mx in alphas:
        if time.time() > deadline - 1: break
        def f(x): return smooth_obj(x, n, alpha)[0]
        def fjac(x): return make_grad(x, n, smooth_obj(x, n, alpha)[1])
        try:
            res = minimize(f, best_h, args=(), method='SLSQP', jac=fjac,
                          bounds=bnds, constraints=cons,
                          options={'maxiter':mx,'ftol':1e-14,'disp':False})
            x = project_feasible(res.x, n)
            c = compute_c5(x, n)
            if c < best_c: best_h, best_c = x, c
        except: pass
    return best_h, best_c
