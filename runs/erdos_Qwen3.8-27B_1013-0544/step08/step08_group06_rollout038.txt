def smooth_c5_grad(x, n, alpha):

    s = alpha * (c - mc)

    # Vectorized gradient

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_c5_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    s = alpha * (c - mc)
    return (mc + np.log(np.sum(np.exp(s))) / alpha) * dx

def smooth_c5_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    s = alpha * (c - mc)
    e = np.exp(s)
    p = e / e.sum()
    ni = np.arange(n)[:, None]
    ki = np.arange(2 * n - 1)[None, :]
    j1 = ni + ki - (n - 1)
    j2 = ni - ki + (n - 1)
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    pv = np.where(p[None, :] < 1e-15, 0.0, p[None, :])
    s1 = np.sum(pv * (1.0 - x[j1]) * m1, axis=1)
    s2 = np.sum(pv * x[j2] * m2, axis=1)
    return (s1 - s2) * dx

def smooth_opt(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [30, 60, 120, 250, 500, 1000, 2000, 4000]
    total_iters = 800
    per_alpha = max(50, total_iters // len(alphas))
    for alpha in alphas:
        if time.time() > deadline:
            break
        x = best_h.copy()
        y = x.copy()
        t_mom = 1.0
        lr = 0.8 / (alpha + 1.0)
        for it in range(per_alpha):
            if time.time() > deadline:
                break
            g = smooth_c5_grad(y, n, alpha)
            z = y - lr * g
            x_new = project_feasible(z, n)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
            y = np.clip(y, 0.0, 1.0)
            x = x_new
            t_mom = t_new
        c = compute_c5(x, n)
        if c < best_c:
            best_h, best_c = x.copy(), c
    return best_h, best_c

def coord_cd(h, n, max_passes, deadline, grid_size):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        for idx in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[idx]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            lo_d, hi_d = -h0, 1.0 - h0
            m_lo = max(0, min(n - 1 - idx, idx))
            m_hi = min(2 * n - 2, max(2 * n - 2 - idx, idx + n - 1))
            ms = np.arange(m_lo, m_hi + 1)
            M = len(ms)
            a = np.zeros(M)
            jh = idx + ms - (n - 1)
            vh = (jh >= 0) & (jh < n)
            a[vh] += (1.0 - h[jh[vh]])
            jg = idx - ms + (n - 1)
            vg = (jg >= 0) & (jg < n)
            a[vg] -= h[jg[vg]]
            c_aff = c[ms]
            unm = 0.0
            if m_lo > 0:
                tm = float(c[:m_lo].max())
                if tm > unm:
                    unm = tm
            if m_hi < 2 * n - 2:
                tm = float(c[m_hi + 1:].max())
                if tm > unm:
                    unm = tm
            di = np.where(ms == (n - 1))[0]
            hd = len(di) > 0
            dg = np.linspace(lo_d, hi_d, grid_size)
            nv = c_aff[None, :] + dg[:, None] * a[None, :]
            if hd:
                nv[:, di] -= dg[:, None] ** 2
            vals = np.maximum(unm, nv.max(axis=1))
            bi = int(np.argmin(vals))
            bd = float(dg[bi])
            bv = float(vals[bi])
            if 0 < bi < grid_size - 1:
                yy = vals[bi - 1:bi + 2]
                xx = dg[bi - 1:bi + 2]
                dn = yy[0]*(xx[1]-xx[2]) + yy[1]*(xx[2]-xx[0]) + yy[2]*(xx[0]-xx[1])
                if abs(dn) > 1e-15:
                    xv = 0.5*(yy[0]*(xx[1]-xx[2]) + yy[1]*(xx[2]-xx[0]) + yy[2]*(xx[0]-xx[1]))/dn
                    if xx[0] < xv < xx[2]:
                        nv1 = c_aff + xv * a
                        if hd:
                            nv1 = nv1.copy()
                            nv1[di] -= xv ** 2
                        v1 = max(unm, float(nv1.max()))
                        if v1 < bv - 1e-14:
                            bv, bd = v1, float(xv)
            if abs(bd) > 1e-10 and bv < cur - 1e-14:
                h[idx] = float(np.clip(h0 + bd, 0.0, 1.0))
                c[ms] = c_aff + bd * a
                if hd:
                    c[ms[di]] -= bd ** 2
                cur = max(unm, float(c[ms].max()))
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: smooth optimization
    if time.time() < deadline - 400:
        best_h, best_c = smooth_opt(best_h, n, deadline - 400)

    # Phase 2: coordinate descent on main n
    if time.time() < deadline - 280:
        best_h, best_c = coord_cd(best_h, n, 2000, deadline - 280, 80)
    if time.time() < deadline - 200:
        best_h, best_c = coord_cd(best_h, n, 2000, deadline - 200, 120)

    # Phase 3: try many nearby n values
    n_curr = len(best_h)
    n_candidates = sorted(set([n_curr - 2, n_curr - 1, n_curr + 1, n_curr + 2,
                                n_curr - 5, n_curr + 5, n_curr - 3, n_curr + 3,
                                max(100, n_curr // 2), 300, 400]))
    x_orig = np.linspace(0, 2, n_curr)
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if n_try < 50 or n_try > 2000:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd(h_int, n_try, 800, deadline - 60, 70)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            n_curr = n_try
            x_orig = x_new

    # Phase 4: randomized restarts
    while time.time() < deadline - 25:
        scale = 0.03 + 0.04 * (1.0 - (time.time() - t0) / budget_s)
        h_p = best_h + rng.normal(0, scale, len(best_h))
        h_p = project_feasible(h_p, len(best_h))
        h_opt, c_opt = coord_cd(h_p, len(best_h), 600, deadline - 20, 60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 5: final polish
    if time.time() < deadline - 5:
        best_h, best_c = coord_cd(best_h, len(best_h), 3000, deadline, 150)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
