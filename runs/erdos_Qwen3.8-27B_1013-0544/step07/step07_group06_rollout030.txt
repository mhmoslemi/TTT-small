```python
    # B[j, k] = current contribution of position j (where h[j]=1) to c_k

    B = np.zeros((n, 2*n-1))
    for i in range(n):
        # A[i, k]: (1-h[i-k]) if 0<=i-k<n, minus h[i+k] if 0<=i+k<n
        idx1 = i - lags  # = i-k
        valid1 = (idx1 >= 0) & (idx1 < n)
        A[i, valid1] += (1.0 - h[idx1[valid1]])
        idx2 = i + lags  # = i+k
        valid2 = (idx2 >= 0) & (idx2 < n)
        A[i, valid2] -= h[idx2[valid2]]
    for j in range(n):
        if h[j] == 1:
            # B[j, k] = (1-h[j-k]) if 0<=j-k<n, minus h[j+k] if 0<=j+k<n
            idx1 = j - lags

            B[j, valid1] += (1.0 - h[idx1[valid1]])
            idx2 = j + lags
            valid2 = (idx2 >= 0) & (idx2 < n)
            B[j, valid2] -= h[idx2[valid2]]
    return A, B
```

Wait, let me redo this. c_k = ∑_n h[n+k](1-h[n]).

∂c_k/∂h[p] = ∂/∂h[p] [∑_n h[n+k](1-h[n])]

The term h[p] appears when:
- n+k = p (n = p-k): ∂/∂h[p] h[p](1-h[p-k]) = 1-h[p-k]
- n = p: ∂/∂h[p] h[p+k](1-h[p]) = -h[p+k]

So ∂c_k/∂h[p] = (1-h[p-k])*[0≤p-k<n] - h[p+k]*[0≤p+k<n]

This is exactly the A[p,k] I computed above (for the binary case, but it works for continuous too)!

So the subgradient of max_k c_k at h is A[p, k*] where k* = argmax_k c_k (if unique).

h_new = clip(h - step * A[:, k*], 0, 1)

```python
def proximal_gradient_search(h, n, deadline, step=0.1, max_iter=100000):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    F = float(c.max())
    best = h.copy()
    bestF = F
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        # Subgradient
        p = k_star - (n-1)  # the actual lag
        # ∂c_k/∂h[i] = (1-h[i-p])*[0≤i-p<n] - h[i+p]*[0≤i+p<n]
        g = np.zeros(n)
        # First term: (1-h[i-p]) for i where 0≤i-p<n
        if p >= 0:
            g[p:] = (1.0 - h[:n-p])
        else:
            g[:n+p] = (1.0 - h[-p:])
        # Second term: -h[i+p] for i where 0≤i+p<n
        if p >= 0:
            g[:n-p] -= h[p:]
        else:
            g[-p:] -= h[:n+p]
        
        h = np.clip(h - step * g, 0.0, 1.0)
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        F = float(c.max())
        if F < bestF:
            bestF = F
            best = h.copy()
    return best, bestF * (2.0/float(n))
```
