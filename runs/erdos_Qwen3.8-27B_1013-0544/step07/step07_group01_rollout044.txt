    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick L-BFGS-B warmup
    t1 = min(t0 + 50, deadline - 300)
    if time.time() < t1:
        h1 = lbfgs_optimize(best_h, n, 500, 200, t1)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c, best_h = c1, h1.copy()

    # Phase 2: Adam with increasing alpha
    t2 = min(t0 + 250, deadline - 200)
    if time.time() < t2:
        h2 = best_h.copy()
        for a in [1000, 5000, 20000]:
            if time.time() > t2 - 10:
                break
            h2, c2 = adam_projected(h2, n, a, t2 - 10, lr0=0.03)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()

    # Phase 3: L-BFGS-B polish with higher alpha
    t3 = min(t0 + 500, deadline - 150)
    if time.time() < t3:
        h3 = best_h.copy()
        for a in [10000, 50000, 200000]:
            if time.time() > t3 - 15:
                break
            h3 = lbfgs_optimize(h3, n, a, 400, t3 - 15)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c, best_h = c3, h3.copy()

    # Phase 4: Multi-grid exploration
    t4 = min(t0 + 750, deadline - 100)
    if time.time() < t4:
        best_h, best_c = multi_grid_search(best_h, best_c, n, t4, rng)

    # Phase 5: Structured restarts
    t5 = deadline - 50
    for trial in range(15):
        if time.time() > t5 - 4:
            break
        if trial % 3 == 0:
            h5 = project_feasible(best_h + 0.004 * rng.randn(n), n)
        elif trial % 3 == 1:
            h5 = project_feasible(best_h + 0.002 * (rng.rand(n) * 2 - 1), n)
        else:
            h5 = best_h.copy()
            nb = n // 40
            for bi in range(nb):
                if rng.rand() < 0.25:
                    h5[bi * 40:(bi + 1) * 40] += 0.08 * rng.randn(40)
            h5 = project_feasible(h5, n)
        # Optimize the perturbed candidate
        t_sub = min(time.time() + 25, t5 - 3)
        h5 = lbfgs_optimize(h5, n, 20000, 200, t_sub)
        h5, c5 = adam_projected(h5, n, 100000, min(time.time() + 10, t5 - 2), lr0=0.008)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 8:
        h6 = best_h.copy()
        h6 = lbfgs_optimize(h6, n, 500000, 300, deadline - 8)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()
        h6, c6 = adam_projected(h6, n, 500000, deadline - 2, lr0=0.003)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
