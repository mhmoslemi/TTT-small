    n_list = sorted(set(n_list))
    L = len(n_list)

    fr = [0.5]
    rem = (0.88 - 0.5) / max(1, L - 1)
    for _ in range(L - 1):
        fr.append(rem)

    for k, nh in enumerate(n_list):
        if time.time() > deadline - 30:
            break
        dl_phase = t0 + (deadline - t0) * (fr[:k+1].sum()) - 10
        dl_phase = max(dl_phase, time.time() + 15)
        dl_phase = min(dl_phase, deadline - 20)

        if nh == bestn:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)

        hopt, copt = optimize_single(hinit, nh, seed + 13 * k, dl_phase)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh

    if time.time() < deadline - 20:
        for restart in range(3):
            if time.time() > deadline - 20:
                break
            hh = project_box_sum(besth + 0.01 * rng.randn(bestn), bestn)
            hopt, copt = optimize_single(hh, bestn, seed + 999 + restart,
                                         min(time.time() + 60, deadline - 10))
            if copt < bestc:
                bestc = copt
                besth = hopt.copy()

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
