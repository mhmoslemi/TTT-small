        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.5 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        try:
            res = minimize(
                lbfgs_obj, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback,
                options={"maxiter": 1000000, "maxls": 40, "ftol": 1e-13, "gtol": 1e-8}
            )
            x = res.x if res is not None and hasattr(res, "x") else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

def binary_sa(h_start, n, deadline, n_restarts=5):
    """Simple simulated annealing on binary vectors."""
    h = top_n_binary(h_start, n)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(123)
    
    t_total = max(1.0, deadline - time.time())
    t_per_restart = t_total / n_restarts
    
    for restart in range(n_restarts):
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        curr_c = float(np.max(corr)) * dx
        t0 = time.time()
        t_dur = t_per_restart - 1.0
        T = 0.03
        while time.time() - t0 < t_dur:
            zeros = np.nonzero(h == 0)[0]
            ones = np.nonzero(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            # 1-swap
            i = zeros[rng.randint(zeros.size)]
            j = ones[rng.randint(ones.size)]
            h[i] = 1.0
            h[j] = 0.0
            new_corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            new_c = float(np.max(new_corr)) * dx
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                corr = new_corr
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] = 0.0
                h[j] = 1.0
            # Cool
            T *= 0.9999
            T = max(T, 0.0005)
    
    return best_h, best_c
