import numpy as np
import time
from scipy.optimize import minimize

_BOUNDS = {}
_CONS = {}

def get_bounds(n):
    if n not in _BOUNDS:
        _BOUNDS[n] = [(0.0, 1.0)] * n
    return _BOUNDS[n]

def cons_eq(x, n):
    return x.sum() - n / 2.0

def get_cons(n):
    if n not in _CONS:
        _CONS[n] = [{'type': 'eq', 'fun': cons_eq, 'args': (n,)}]
    return _CONS[n]

def corr_vals(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def compute_c5(h, n):
    return float(np.max(corr_vals(h, n)))

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.5)
    target = n / 2.0
    if abs(h.sum() - target) < 1e-13 and np.min(h) >= -1e-12 and np.max(h) <= 1 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = np.clip(h - tau, 0.0, 1.0)
    for _ in range(6):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += diff / m.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad(r, x, n):
    r = np.asarray(r, dtype=float)
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    idx = n - 1
    if r[idx] != 0.0:
        g += r[idx] * (1.0 - 2.0 * x) * dx
    for d in range(1, n):
        a = r[idx + d]
        b = r[idx - d]
        if a == 0.0 and b == 0.0:
            continue
        s = a + b
        nd = n - d
        g[d:] += dx * (a - s * x[:nd])
        g[:nd] += dx * (b - s * x[d:])
    return g

def obj_lse(x, n, alpha):
    c = corr_vals(x, n)
    m = float(np.max(c))
    if m <= 0.0:
        return 0.0
    z = alpha * (c - m)
    return m + float(np.log(np.exp(z).sum())) / alpha

def jac_lse(x, n, alpha):
    c = corr_vals(x, n)
    m = float(np.max(c))
    e = np.exp(alpha * (c - m))
    s = e.sum()
    if s == 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    return weighted_grad(e / s, x, n)

def obj_pn(x, n, p):
    c = np.maximum(corr_vals(x, n), 1e-15)
    m = float(np.max(c))
    if m <= 0.0:
        return 0.0
    u = c / m
    logu = np.log(u)
    vals = np.exp(np.clip(p * logu, -745.0, 0.0))
    S = float(vals.sum())
    if S <= 0.0:
        return 0.0
    return m * float(np.exp(np.log(S) / p))

def jac_pn(x, n, p):

    m = float(np.max(c))

    if mix > 0.0:

    return w

    x = np.asarray(x, dtype=float)

        return x
    opts = {'maxiter': int(maxiter), 'ftol': float(ftol)}

    except Exception:
        pass
    return x
