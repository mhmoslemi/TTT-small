```python
        h_new = y - step * g
        h_new = project_feasible(h_new, n)
        h_prev = h
        h = h_new
        # Check improvement

        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
            patience = 0
            # Increase lr slightly?
        else:
            patience += 1
            if patience > 1000:
                lr *= 0.9
                patience = 0
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    c = compute_correlation(h, n)
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at_lag(h, n, l):
    g = np.zeros(n)
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        g[0:n - l] += (1.0 - h[l:n])
        g[l:n] -= h[0:n - l]
    else:
        p = -l
        g[p:n] += (1.0 - h[0:n - p])
        g[0:n - p] -= h[p:n]
    return g

def nesterov_subgrad(h, n, deadline, top_k=5, lr0=0.05, max_iters=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_prev = h.copy()
    beta = 1.0
    lr = lr0
    patience = 0
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        if t > 1:
            y = h + ((beta - 1.0) / beta) * (h - h_prev)
        else:
            y = h.copy()
        y = project_feasible(y, n)
        c = compute_correlation(y, n)
        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for k in top_idx:
            l = int(k) - (n - 1)
            g += subgrad_at_lag(y, n, l)
        g /= top_k
        step = lr / (1.0 + 0.01 * t)
        h_new = y - step * g
        h_new = project_feasible(h_new, n)
        h_prev = h
        h = h_new
        c_val = compute_c5(h, n)
        if c_val < best_c - 1e-10:
            best_c = c_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                lr *= 0.85
                patience = 0
    return best_h, best_c

def pnorm_obj_and_grad(x, n, p):
    dx = 2.0 / n
    c_raw = compute_correlation(x, n)
    c = c_raw * dx
    f = np.sum(c ** p)
    if p == 1:
        return f, None
    w = p * (c_raw ** (p - 1)) * (dx ** p)
    grad = np.zeros(n)
    for l in range(1, n):
        w_pos = w[n - 1 + l]
        if abs(w_pos) > 1e-30:
            grad[0:n - l] += w_pos * (1.0 - x[l:n])
            grad[l:n] -= w_pos * x[0:n - l]
        w_neg = w[n - 1 - l]
        if abs(w_neg) > 1e-30:
            grad[l:n] += w_neg * (1.0 - x[0:n - l])
            grad[0:n - l] -= w_neg * x[l:n]
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * x)
    return f, grad

def lbfgs_pnorm(h, n, p, maxiter, deadline):
    def obj(x):
        f, g = pnorm_obj_and_grad(x, n, p)
        return f, g
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def binary_local_search_vec(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    c = compute_correlation(h_b, n).astype(float)
    cur_c = c.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h_b == 0)[0]
        ones = np.where(h_b == 1)[0]
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            flip_up = np.zeros(2 * n - 1)
            for l in range(1, n):
                if i + l < n:
                    flip_up[n - 1 + l] += (1.0 - h_b[i + l])
                    flip_up[n - 1 - l] += (1.0 - h_b[i + l]) if i + l < n else 0
                if i - l >= 0:
                    flip_up[n - 1 + l] -= h_b[i - l]
                    flip_up[n - 1 - l] -= h_b[i - l]
            best_j = -1
            best_nc = cur_c
            for j in ones:
                flip_down = np.zeros(2 * n - 1)
                for l in range(1, n):
                    if j + l < n:
                        flip_down[n - 1 + l] -= (1.0 - h_b[j + l])
                        flip_down[n - 1 - l] -= (1.0 - h_b[j + l])
                    if j - l >= 0:
                        flip_down[n - 1 + l] += h_b[j - l]
                        flip_down[n - 1 - l] += h_b[j - l]
                nc = (c + flip_up + flip_down).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                new_c = c + flip_up
                flip_down2 = np.zeros(2 * n - 1)
                j = best_j
                for l in range(1, n):
                    if j + l < n:
                        flip_down2[n - 1 + l] -= (1.0 - h_b[j + l])
                        flip_down2[n - 1 - l] -= (1.0 - h_b[j + l])
                    if j - l >= 0:
                        flip_down2[n - 1 + l] += h_b[j - l]
                        flip_down2[n - 1 - l] += h_b[j - l]
                new_c = new_c + flip_down2
                h_b[i], h_b[j] = 1.0, 0.0
                c = new_c
                cur_c = best_nc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c

def fast_binary_swap(h_b, n, deadline, max_iters=50000):
    dx = 2.0 / n
    c = compute_correlation(h_b, n).astype(float)
    cur_max = c.max()
    best_h = h_b.copy()
    best_c = cur_max * dx
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    rng = np.random.RandomState(12345)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = np.zeros(2 * n - 1)
        for l in range(1, n):
            if i + l < n:
                delta[n - 1 + l] += (1.0 - h_b[i + l])
                delta[n - 1 - l] += (1.0 - h_b[i + l])
            if i - l >= 0:
                delta[n - 1 + l] -= h_b[i - l]
                delta[n - 1 - l] -= h_b[i - l]
        for l in range(1, n):
            if j + l < n:
                delta[n - 1 + l] -= (1.0 - h_b[j + l])
                delta[n - 1 - l] -= (1.0 - h_b[j + l])
            if j - l >= 0:
                delta[n - 1 + l] += h_b[j - l]
                delta[n - 1 - l] += h_b[j - l]
        nc = (c + delta).max()
        if nc < cur_max - 1e-12:
            c = c + delta
            h_b[i], h_b[j] = 1.0, 0.0
            cur_max = nc
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            nc_val = cur_max * dx
            if nc_val < best_c:
                best_c = nc_val
                best_h = h_b.copy()
    return best_h, best_c
```
