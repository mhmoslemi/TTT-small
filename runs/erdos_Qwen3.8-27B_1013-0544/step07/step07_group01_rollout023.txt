    return float(np.max(c) * dx)

def compute_autocorr(h, n):
    """Compute autocorrelation of h at all lags."""
    # (h*h)(k) = sum_i h[i]*h[i+k]
    # Using FFT: autocorr = ifft(|fft(h, 2n)|^2)
    N = 2 * n
    H = fft(h, N)
    ac = np.real(ifft(H * np.conj(H)))
    # ac[k] for k=0..n-1 is the autocorrelation at lag k
    # ac[k] for k=n..2n-1 is the autocorrelation at lag k-n (negative lags)
    return ac[:n]  # non-negative lags 0..n-1

            if s < 1e-14:

    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_max_grad(h, n, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()

    # dc[k]/dh[i] = (1-h[i+k]) - h[i-k] (with bounds)

    # the contribution to grad[i] is p[lag_idx] * dc[lag]/dh[i]

        w = p[li] * dx
        if w < 1e-15:

        if lag >= 0:

            # d/dh[i] for i in [0, n-1-lag]: (1-h[i+lag])
            # d/dh[i] for i in [lag, n-1]: -h[i-lag]
            if lag < n:

            ll = -lag
            if ll < n:

                grad[:n-ll] += w * (-h[ll:])
    
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    p = np.exp(sh - sm)
    p /= p.sum()
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        lag = li - (n - 1)
        w = p[li] * dx
        if w < 1e-15:
            continue
        if lag >= 0 and lag < n:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n-lag])
        elif lag < 0:
            ll = -lag
            if ll < n:
                grad[ll:] += w * (1.0 - h[:n-ll])
                grad[:n-ll] += w * (-h[ll:])
    return val, grad

def opt_lbfgs(h, n, alpha, penalty, maxiter):
    def obj(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v + penalty * (x.sum() - n / 2.0) ** 2
    
    def grad(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g + 2.0 * penalty * (x.sum() - n / 2.0)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxls': 100})
        return project_feasible(res.x, n)
    except Exception:
        return h

def opt_slsqp(h, n, alpha, maxiter):
    def obj(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v
    
    def grad(x):
        _, g = smooth_obj_grad(x, n, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_steps(h, n, n_steps, lr):
    dx = 2.0 / n
    for _ in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(c)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
        elif lag < 0:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - h[:n-ll])
                grad[:n-ll] += (-h[ll:])
        grad *= dx
        h = project_feasible(h - lr * grad, n)
    return h
