        best_c, best_h = c3, h3.copy()

    # Phase 4: Second binary SA round with reheat (15%)
    p4 = t0 + budget_s * 0.85
    h4, c4 = binary_sa(best_h, n, p4, rng)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Aggressive SLSQP polish (10%)
    if time.time() < deadline - 10:
        h5 = best_h.copy()
        for a in [100000, 1000000, 10000000, 100000000]:
            if time.time() > deadline - 5:
                break
            h5 = slsqp_smooth(h5, n, a, 2000, 1e-17)
            c5 = compute_c5(h5, n)
            if c5 < best_c:
                best_c, best_h = c5, h5.copy()

    # Phase 6: Noise perturb + final polish
    if time.time() < deadline - 20:
        for trial in range(4):
            if time.time() > deadline - 8:
                break
            noise = 0.002 * (1 + trial)
            h6 = best_h + noise * rng.randn(n)
            h6 = project_feasible(h6, n)
            h6 = slsqp_smooth(h6, n, 500000, 1500, 1e-16)
            h6 = slsqp_smooth(h6, n, 5000000, 1000, 1e-16)
            c6 = compute_c5(h6, n)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
