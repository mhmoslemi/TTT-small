def smooth_obj(x,n,alpha):
    C=corr_counts(x)*(2.0/n)
    m=C.max()
    s=np.minimum(C-m,0.0)
    e=np.exp(alpha*s)
    return float(m + np.log(e.sum())/alpha)
def smooth_grad(x,n,alpha):
    C=corr_counts(x)*(2.0/n)
    m=C.max(); s=np.minimum(C-m,0); e=np.exp(alpha*s); ssum=e.sum()
    p=e/ssum
    return (2.0/n)*add_weighted_gradient(x,np.arange(C.size),p,n)
def mass_fun(x,target): return float(x.sum()-target)
def mass_jac(x,target): return np.ones_like(x)
def slsqp_smooth(h,n,alpha,maxiter,deadline=None):
    h=project_feasible(h,n)
    bounds=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':mass_fun,'jac':mass_jac,'args':(0.5*n,)}]
    try:
       res=minimize(smooth_obj,h,method='SLSQP',bounds=bounds,constraints=cons,jac=smooth_grad,args=(n,alpha),options={'maxiter':maxiter,'ftol':1e-10,'maxls':10,'disp':False})
       h2=project_feasible(res.x,n)
    except Exception:
       h2=h
    return h2,c5_value(h2,n)

def add_weighted_gradient(h, idx, weights, n):
    g=np.zeros(n)
    m=n-1
    for t in range(len(idx)):
        p=float(weights[t])
        if p <= 1e-12: continue

        if lag==0:
            g += p*(1.0-2.0*h)

            g[lag:] -= p*h[:q]
        else:
            d=-lag; q=n-d
            g[d:] += p*(1.0-h[:q])
            g[:q] -= p*h[d:]
    return g

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    np.clip(x, 0.0, 1.0, out=x)
    t = 0.5 * n
    for _ in range(80):
        d = t - x.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - x
            s = r.sum()
            if s < 1e-15:
                break
            x += d * r / s
        else:
            s = x.sum()
            if s < 1e-15:
                break
            x += d * x / s
        np.clip(x, 0.0, 1.0, out=x)
    d = t - x.sum()
    if abs(d) > 1e-14:
        if d > 0:
            free = np.where(x < 1.0)[0]
            if free.size:
                x[free] += d / free.size
        else:
            free = np.where(x > 0.0)[0]
            if free.size:
                x[free] += d / free.size
    np.clip(x, 0.0, 1.0, out=x)
    return x

def corr_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h, n):
    return float(np.max(corr_counts(h)) * (2.0 / n))

def add_weighted_gradient(h, idx, weights, n):
    g = np.zeros(n)
    m = n - 1
    for t in range(len(idx)):
        p = float(weights[t])
        if p <= 1e-12:
            continue
        lag = m - int(idx[t])
        if lag == 0:
            g += p * (1.0 - 2.0 * h)
        elif lag > 0:
            q = n - lag
            g[:q] += p * (1.0 - h[lag:])
            g[lag:] -= p * h[:q]
        else:
            d = -lag
            q = n - d
            g[d:] += p * (1.0 - h[:q])
            g[:q] -= p * h[d:]
    return g

def make_weights(C, K):
    m = C.size
    if K >= m:
        idx = np.arange(m)
        vals = C
    else:
        K = max(1, min(K, m - 1))
        idx = np.argpartition(C, -K)[-K:]
        vals = C[idx]
    vmin = vals.min()
    vmax = vals.max()
    if vmax - vmin > 1e-12:
        sigma = max((vmax - vmin) / 3.0, 1e-7)
        w = np.exp((vals - vmax) / sigma)
        s = w.sum()
        if s <= 0:
            w = np.ones_like(vals) / vals.size
        else:
            w /= s
    else:
        w = np.ones_like(vals) / vals.size
    return idx, w

def proj_descent(h, n, deadline, rng, K_init=48, K_max=256):
    h = project_feasible(h, n)
    best = h.copy()
    best_c = c5_value(h, n)
    dx = 2.0 / n
    cc = corr_counts(h)
    v = np.zeros(n)
    lr = 0.8
    mu = 0.88
    stall = 0
    K = min(max(1, K_init), 2 * n - 1)
    Kcap = min(max(2, K_max), 2 * n - 1)
    while time.time() < deadline:
        C = cc * dx
        idx, w = make_weights(C, K)
        grad = dx * add_weighted_gradient(h, idx, w, n)
        v = mu * v - lr * grad
        vabs = np.max(np.abs(v)) if v.size else 0.0
        if not np.isfinite(vabs):
            v = -lr * grad
            vabs = np.max(np.abs(v))
        if vabs > 0.5:
            v *= 0.5 / vabs
        h2 = project_feasible(h + v, n)
        cc2 = corr_counts(h2)
        c2 = float(cc2.max() * dx)
        h = h2
        cc = cc2
        if c2 < best_c - 1e-12:
            best_c = c2
            best = h.copy()
            stall = 0
            lr = min(lr * 1.03, 8.0)
        else:
            stall += 1
            if stall % 100 == 0:
                K = min(K + 8, Kcap)
                lr = max(lr * 0.9, 0.05)
            if stall > 600:
                h = best.copy()
                v = np.zeros(n)
                lr = 0.8
                K = min(K_init, Kcap)
                stall = 0
                h = project_feasible(h + 0.01 * rng.randn(n), n)
                cc = corr_counts(h)
    return best, best_c

def smooth_obj(x, n, alpha):
    C = corr_counts(x) * (2.0 / n)
    m = C.max()
    s = np.minimum(C - m, 0.0)
    e = np.exp(alpha * s)
    return float(m + np.log(e.sum()) / alpha)

def smooth_grad(x, n, alpha):
    C = corr_counts(x) * (2.0 / n)
    m = C.max()
    s = np.minimum(C - m, 0.0)
    e = np.exp(alpha * s)
    p = e
    ss = p.sum()
    if ss > 0:
        p /= ss
    return (2.0 / n) * add_weighted_gradient(x, np.arange(C.size), p, n)

def mass_fun(x, target):
    return float(x.sum() - target)

def mass_jac(x, target):
    return np.ones_like(x)

def slsqp_smooth(h, n, alpha, maxiter, deadline=None):
    h = project_feasible(h, n)
    if deadline is not None and time.time() >= deadline:
        return h, c5_value(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": mass_fun, "jac": mass_jac, "args": (0.5 * n,)}]
    try:
        res = minimize(smooth_obj, h, method="SLSQP", bounds=bounds,
                       constraints=cons, jac=smooth_grad, args=(n, alpha),
                       options={"maxiter": maxiter, "ftol": 1e-10, "maxls": 8, "disp": False})
        h2 = project_feasible(res.x, n)
    except Exception:
        h2 = h
    return h2, c5_value(h2, n)

    best_h = None
    best_c = 1e9
    best_n = 0

    if parent_h is not None and parent_n > 0:
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c = c5_value(parent_h, parent_n)
        best_n = parent_n

    small_deadline = min(deadline, t0 + min(330, max(90, int(0.30 * budget_s))))
    sh, sc, sn = small_n_hunt(small_deadline, best_h if best_h is not None else None, best_n, rng)
    if sh is not None and sc < best_c:
        best_h, best_c, best_n = sh, sc, sn

    if best_h is None:
        best_n = 16
        best_h = project_feasible(np.zeros(best_n), best_n)
        best_h[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        best_c = c5_value(best_h, best_n)

    rem = deadline - time.time()
    if best_n > 80:
        if rem > 50:
            pd = deadline - 80 if rem > 180 else deadline - 15
            best_h, best_c = proj_descent(best_h, best_n, pd, rng, 64, 300)
        if deadline - time.time() > 65:
            h2, c2 = slsqp_smooth(best_h, best_n, 20000, 22, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if deadline - time.time() > 18:
            best_h, best_c = proj_descent(best_h, best_n, deadline - 2, rng, 32, 180)
    else:
        if rem > 25:
            pd = deadline - 18 if rem > 90 else deadline - 4
            best_h, best_c = proj_descent(best_h, best_n, pd, rng,
                                          min(16, 2 * best_n - 1),
                                          min(4 * best_n - 1, 3 * best_n + 20))
        if deadline - time.time() > 25:
            h2, c2 = slsqp_smooth(best_h, best_n, 20000, 70 if best_n <= 32 else 36, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if deadline - time.time() > 12:
            h2, c2 = slsqp_smooth(best_h, best_n, 800, 60 if best_n <= 32 else 30, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if deadline - time.time() > 6:
            best_h, best_c = proj_descent(best_h, best_n, deadline - 1, rng, 8, 2 * best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = c5_value(best_h, best_n)
    return best_h, float(best_c), int(best_n)
