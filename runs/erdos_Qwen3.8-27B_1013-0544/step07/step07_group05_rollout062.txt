                    t_hi = t_mid

                    t_lo = t_mid

        cand = project_box_sum(res.x, target)

        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

def coord_descent(h, n, deadline, n_sweeps=500, seed=42):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    dx = 2.0 / n
    m_len = 2 * n - 1
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c = float(c.max())
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline - 2.0:
            break
        order = rng.permutation(n)
        max_change = 0.0
        for i in order:
            i = int(i)
            x = h[i]
            B = np.zeros(m_len)
            m_lo1 = max(0, n - 1 - i)
            m_hi1 = min(m_len - 1, n - 1 + i)
            if m_hi1 >= m_lo1:
                phi_lo = i + m_lo1 - n + 1
                ln1 = m_hi1 - m_lo1 + 1
                B[m_lo1:m_lo1 + ln1] += (1.0 - h[phi_lo:phi_lo + ln1])
            m_lo2 = max(0, i)
            m_hi2 = min(m_len - 1, i + n - 1)
            if m_hi2 >= m_lo2:
                ln2 = m_hi2 - m_lo2 + 1
                psi_lo = i - m_hi2 + n - 1
                B[m_lo2:m_lo2 + ln2] -= h[psi_lo:psi_lo + ln2][::-1]
            d_lo = -x
            d_hi = 1.0 - x
            if d_hi - d_lo < 1e-14:
                continue
            pos = B > 1e-15
            neg = B < -1e-15
            zero = ~(pos | neg)
            mins = np.where(pos, c + B * d_lo, np.where(neg, c + B * d_hi, c))
            t_lo = float(mins.max())
            maxs = np.where(pos, c + B * d_hi, np.where(neg, c + B * d_lo, c))
            t_hi = float(maxs.max())
            d_star = 0.5 * (d_lo + d_hi)
            for _ in range(40):
                t_mid = 0.5 * (t_lo + t_hi)
                if t_hi - t_lo < 1e-13:
                    break
                lb = d_lo
                ub = d_hi
                if zero.any() and float(c[zero].max()) > t_mid + 1e-14:
                    t_hi = t_mid
                    continue
                if neg.any():
                    v = (t_mid - c[neg]) / B[neg]
                    cl = float(v.max())
                    if cl > lb:
                        lb = cl
                if pos.any():
                    v = (t_mid - c[pos]) / B[pos]
                    cu = float(v.min())
                    if cu < ub:
                        ub = cu
                if lb > ub + 1e-14:
                    t_hi = t_mid
                else:
                    t_lo = t_mid
                    d_star = 0.5 * (lb + ub)
            x_new = x + d_star
            x_new = max(0.0, min(1.0, x_new))
            d_act = x_new - x
            if abs(d_act) > 1e-14:
                h[i] = x_new
                c += d_act * B
                max_change = max(max_change, abs(d_act))
        h = project_box_sum(h, target)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h.copy()
        if max_change < 1e-11:
            break
    return best_h, best_c

def resample_n(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)

def pair_polish(h, n, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    rng = np.random.RandomState(int(seed))
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size == 0:
        return h, cur
    score = np.minimum(h[frac], 1.0 - h[frac])
    order = frac[np.argsort(score)[::-1]]
    m = min(20, len(order))
    top = order[:m]
    for _ in range(4):
        improved = False
        pairs = [(int(top[a]), int(top[b])) for a in range(len(top)) for b in range(a+1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 4.0:
                break
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            best_d, best_c = 0.0, cur
            for tt in (lo, hi, 0.5*(lo+hi)):
                hh = h.copy()
                hh[i] = hi0 + tt
                hh[j] = hj0 - tt
                cc = erdos_c5(hh, n)
                if cc < best_c:
                    best_c, best_d = cc, tt
            try:
                def _obj(d):
                    a, b = hi0 + d, hj0 - d
                    if a < -1e-12 or a > 1+1e-12 or b < -1e-12 or b > 1+1e-12:
                        return 1e100
                    hh = h.copy()
                    hh[i], hh[j] = a, b
                    return erdos_c5(hh, n)
                rs = so.minimize_scalar(_obj, bounds=(lo, hi), method='bounded',
                                        options={'maxiter': 20, 'xatol': 1e-10})
                if rs.x is not None:
                    hh = h.copy()
                    hh[i] = hi0 + float(rs.x)
                    hh[j] = hj0 - float(rs.x)
                    cc = erdos_c5(hh, n)
                    if cc < best_c:
                        best_c, best_d = cc, float(rs.x)
            except Exception:
                pass
            if best_c < cur - 1e-14:
                h[i] = hi0 + best_d
                h[j] = hj0 - best_d
                cur = best_c
                improved = True
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    n = min(n, 800)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if len(h) != n:
        h = np.interp(np.linspace(0., 2., n), np.linspace(0., 2., len(h)), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    end1 = t0 + 0.40 * total
    end2 = t0 + 0.65 * total
    end3 = t0 + 0.85 * total
    end4 = deadline - 10.0

    best_h, best_c = coord_descent(best_h, best_n, end1, n_sweeps=800, seed=seed)
    best_n = best_h.size

    if time.time() < end2 - 20.0:
        for nh in (550, 450):
            if time.time() > end2 - 20.0 or nh == best_n:
                continue
            hh = resample_n(best_h, best_n, nh)
            hh, c = coord_descent(hh, nh, end2 - 15.0, n_sweeps=400, seed=seed + nh)
            if c < best_c - 1e-6:
                best_h = hh.copy()
                best_n = nh
                best_c = c

    if time.time() < end3 - 5.0:
        cand, c = refine_lbfgs(best_h, 3000.0, 200, end3 - 5.0)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
        cand, c = refine_lbfgs(best_h, 30000.0, 150, end3)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    if time.time() < end4 - 15.0 and best_n <= 800:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 10.0, seed)

    if time.time() < end4 - 5.0:
        for s in (seed + 7, seed + 13, seed + 23):
            if time.time() > end4 - 8.0:
                break
            rng = np.random.RandomState(s)
            hh = project_box_sum(best_h + 0.03 * rng.randn(best_n), 0.5 * best_n)
            hh, c = coord_descent(hh, best_n, end4 - 5.0, n_sweeps=200, seed=s)
            if c < best_c - 1e-15:
                best_h, best_c = hh, c

    if time.time() < deadline - 8.0:
        cand, c = refine_lbfgs(best_h, 100000.0, 100, deadline - 5.0)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    best_h = project_box_sum(best_h, 0.5 * best_h.size)
    best_c = erdos_c5(best_h, best_h.size)
    return best_h, best_c, int(best_h.size)
