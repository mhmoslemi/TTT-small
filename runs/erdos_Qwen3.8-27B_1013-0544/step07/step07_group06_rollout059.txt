    elif a.size > n:

    a = np.clip(a, -1.0, 2.0)

        else:

            if s < 1e-15: break

    if s < 1e-300:
        return 1e10, np.zeros(n)

    P = p
    R = P[::-1]

    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(P, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    dev = h.sum() - 0.5*n

from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * dx)

def bin_proj(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size < n:
        a = np.concatenate([a, np.zeros(n-a.size)])
    elif a.size > n:
        a = a[:n]
    h = np.zeros(n)
    h[np.argsort(-a)[:n//2]] = 1.0
    return h

def cont_proj(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size < n:
        a = np.concatenate([a, np.full(n-a.size, 0.5)])
    elif a.size > n:
        a = a[:n]
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if np.clip(a-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5*(lo+hi), 0, 1)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1-h; s = r.sum()
            if s < 1e-15: break
            h += d*r/s
        else:
            s = h.sum()
            if s < 1e-15: break
            h += d*h/s
    return np.clip(h, 0, 1)

def lse_fg(x, n, beta):
    h = np.clip(x, 0, 1)
    dx = 2.0/n
    c = np.maximum(np.correlate(h, 1-h, mode="full"), 0)*dx
    m = c.max()
    y = beta*(c-m)
    e = np.exp(np.minimum(y, 0))
    s = e.sum()
    if s < 1e-300:
        return 1e10, np.zeros(n)
    p = e/s
    F = m + np.log(s)/beta
    R = p[::-1]; v = 1-h
    A = np.convolve(R, v, "full")[n-1:2*n-1]
    B = np.convolve(p, h, "full")[n-1:2*n-1]
    g = dx*(A-B)
    d = h.sum() - 0.5*n
    F += 1e4*d*d
    g += 2e4*d
    return F, g

            r = minimize(lse_fg, h, method="L-BFGS-B", jac=True,
                        bounds=bnds, args=(n,b), callback=stop_cb,

            h = r.x
        except: pass
        h = cont_proj(h, n)
        c = compute_c5(h, n)
        if c < bc: bc, best = c, h.copy()
    return best, bc

def cbase(h, n):
    l = np.arange(-(n-1), n, dtype=np.int64)
    base = np.zeros((n, 2*n-1))
    for p in range(n):
        f = p+l; mf = (f>=0)&(f<n)
        base[p,mf] = 1.0-h[f[mf]]
        b = p-l; mb = (b>=0)&(b<n)
        base[p,mb] -= h[b[mb]]
    return base

def bin_opt(h, n, deadline, restarts=3):

    corr = np.correlate(h, 1-h, mode="full").astype(float)
    M = int(round(corr.max()))
    besth, bestM = h.copy(), M
    nl = 2*n-1

from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def bin_proj(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    elif a.size > n:
        a = a[:n]
    h = np.zeros(n)
    h[np.argsort(-a)[:n // 2]] = 1.0
    return h

def cont_proj(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size < n:
        a = np.concatenate([a, np.full(n - a.size, 0.5)])
    elif a.size > n:
        a = a[:n]
    target = 0.5 * n
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0, 1).sum() > target:

        else:

            if s < 1e-15: break
            h += d * h / s
    return np.clip(h, 0, 1)

def lse_fg(x, n, beta):

    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1 - h, mode="full"), 0) * dx
    m = c.max()

    e = np.exp(np.minimum(y, 0))
    s = e.sum()
    if s < 1e-300:
        return 1e10, np.zeros(n)
