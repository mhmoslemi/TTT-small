import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum()) - target) < 1e-14:
        return y
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def wgrad(h, w, n):
    """O(n) weighted corr gradient. w length 2n-1."""
    mid = n - 1
    dx = 2.0 / n
    if n == 1:
        return np.array([(1.0 - 2.0 * h[0]) * w[0] * dx])
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    s[0] = w[mid]
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def softmax_beta(c, beta):
    mx = float(np.max(c))
    z = beta * (c - mx)
    np.clip(z, -700.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mx, w
    return mx + np.log(ws) / beta, ew / ws

def lbfgs_smooth(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()

        return h, erdos_c5(h, n)
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        _, w = softmax_beta(c, beta)
        f, g = wgrad(x, w, n)
        res = float(x.sum() - target)
        f += 1e6 * res * res
        g = g + 2e6 * res
        return f, g

    mx = int(min(maxiter, max(10, int(rem / 2.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        r = so.minimize(obj, h, method='L-BFGS-B', jac=True, bounds=bounds,
                        options={'maxiter': mx, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 30})
        return project_box_sum(r.x, target), erdos_c5(project_box_sum(r.x, target), n)
    except Exception:

def active_subgrad(h, n_active, n_iter, deadline):

    h = np.asarray(h, dtype=float).copy()

    def eval_pair(h, i, j, d):

        hh = h.copy()

        for kidx in active:

            for j in range(n):

                if 0 <= j - k < n:
                    s += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    s -= h[j + k]
                sens[j] += s
        # Find extreme sensitivity positions
        order = np.argsort(sens)
        bottom = order[:min(20, n // 2)]

        # Generate candidate pairs
        pairs = []
        for a in top:
