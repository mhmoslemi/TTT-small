            # k=1: sum_n a_{n+1} * v_n = a_1 * v_0 = 2*3 = 6

            # Let me use a different indexing. Let output index be k (0 to 2n-2).
            # The "lag" is lag = k - (n-1), ranging from -(n-1) to (n-1).

            # h[i] appears in G[k] when:
            # (a) n+lag = i, i.e., n = i-lag. Valid if 0≤i-lag<n.
            #     The term is h[i]*(1-h[i-lag]) → v*(1-h[i-lag])
            #     Change: (v - v_old)*(1 - h[i-lag])
            # (b) n = i. Valid if 0≤i<n (always).
            #     The term is h[i+lag]*(1-h[i]) → h[i+lag]*(1-v)
            #     Change: h[i+lag]*((1-v)-(1-v_old)) = -h[i+lag]*(v-v_old)

            # dG/d(v) = (1-h[i-lag])*1_{0≤i-lag<n} - h[i+lag]*1_{0≤i+lag<n}
            
            # Let me compute this as a vector of length 2n-1:
            lags = np.arange(-(n-1), n)  # length 2n-1
            
            # i - lag: valid if 0 ≤ i-lag < n
            idx_a = i - lags  # array of same shape as lags
            mask_a = (idx_a >= 0) & (idx_a < n)
            
            # i + lag: valid if 0 ≤ i+lag < n
            idx_b = i + lags
            mask_b = (idx_b >= 0) & (idx_b < n)

            coeff = np.zeros(2*n-1)
            coeff[mask_a] += 1.0 - h[idx_a[mask_a]]
            coeff[mask_b] -= h[idx_b[mask_b]]

            # This is convex piecewise linear in v.
            # Use subgradient: f'(v) = coeff[k*] where k* is the argmax (subgradient)
            # At optimal v: 0 ∈ subgradient, i.e., there exists k* with coeff[k*] ≈ 0
            # or v is at a boundary (0 or 1)
            
            # For convex piecewise linear: find the kink where slope changes sign.
            # Approach: bisection on the subgradient
            # At v, the active set is argmax_k (G[k] + (v-v_old)*coeff[k])
            # The subgradient is the set of coeff values for active k's.
            # If min active coeff > 0, increase v. If max active coeff < 0, decrease v.
            # Otherwise, v is optimal.
            
            lo = 0.0
            hi = 1.0
            # Actually, the min should be at v_old or... no, we're free to set v to any value in [0,1].

            for _ in range(50):
                mid = 0.5 * (lo + hi)
                vals = G + (mid - v_old) * coeff
                k_max = vals.argmax()
                val_max = vals.max()
                # Check all active (within tolerance)
                active = vals >= val_max - 1e-12
                if active.sum() == 0:
                    active = np.zeros(2*n-1, dtype=bool)
                    active[k_max] = True
                min_coeff = coeff[active].min()
                max_coeff = coeff[active].max()
                if min_coeff > 1e-12:
                    # All active slopes positive → increase v
                    lo = mid
                elif max_coeff < -1e-12:
                    # All active slopes negative → decrease v
                    hi = mid
                else:
                    # 0 is in subgradient → optimal
                    lo = hi = mid
                    break
            
            v_new = 0.5 * (lo + hi)
            
            # Update h and G
            delta = v_new - v_old
            if abs(delta) > 1e-14:
                h[i] = v_new
                G += delta * coeff
            else:
                h[i] = v_new
        
        # After sweep, check improvement
        h = project_feasible(h, n)
        G = np.correlate(h, 1.0 - h, mode='full')
        c = compute_c5(h, n)
        
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    
    return best_h, best_c

def solve_1d_subproblem(G, coeff, v_old, S, n_half, mu, iters=60):
    """Solve min_v max_k (G[k] + (v-v_old)*coeff[k]) + mu*(S+v-n_half)^2 over v in [0,1]"""
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        vals = G + (mid - v_old) * coeff
        val_max = vals.max()
        active = vals >= val_max - 1e-10
        # Subgradient of max term: any value in [min(coeff[active]), max(coeff[active])]
        # Derivative of penalty: 2*mu*(S + mid - n_half)

            hi = mid
        elif sg_max < -1e-14:
            # Function decreasing → increase v
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:
            break

            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def slsqp_opt(h, n, alpha, maxiter, ftol):

    target = n / 2.0
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
