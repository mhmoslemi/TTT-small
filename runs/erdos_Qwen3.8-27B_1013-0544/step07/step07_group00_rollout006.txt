        h2, c2 = coord_cd(best_h, len(best_h), max_passes=200, deadline=deadline, seed=seed + 5)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Upsample to 2n and optimize
    if time.time() < deadline - 150:
        n_base = len(best_h)
        n2 = n_base * 2
        if n2 <= 2000:
            x_old = np.linspace(0, 2, n_base)
            x_new = np.linspace(0, 2, n2)
            h3 = np.interp(x_new, x_old, best_h)
            h3 = project_feasible(h3, n2)
            c3 = compute_c5(h3, n2)
            h3, c3 = coord_cd(h3, n2, max_passes=300, deadline=deadline, seed=seed + 20)
            if c3 < best_c:
                best_h, best_c = h3, c3
            if time.time() < deadline - 60:
                h3, c3 = slsqp_polish(best_h, len(best_h), [50000, 200000], 300, deadline)
                if c3 < best_c:
                    best_h, best_c = h3, c3
                h3, c3 = coord_cd(best_h, len(best_h), max_passes=200, deadline=deadline, seed=seed + 50)
                if c3 < best_c:
                    best_h, best_c = h3, c3

    # Phase 4: Multi-restart with perturbations
    if time.time() < deadline - 40:
        n_best = len(best_h)
        rng = np.random.RandomState(seed + 777)
        remaining = max(1, int((deadline - time.time() - 15) / 25))
        for r in range(remaining):
            if time.time() > deadline - 15:
                break
            if r % 6 == 5:
                hp = rng.uniform(0, 1, n_best)
            elif r % 3 == 0:
                sigma = 0.003 + 0.02 * (r / max(1, remaining - 1))
                hp = best_h + rng.normal(0, sigma, n_best)
            else:
                # Block perturbation
                hp = best_h.copy()
                blk = max(5, n_best // 50)
                for _ in range(5):
                    start = rng.randint(0, max(1, n_best - blk))
                    hp[start:start+blk] += rng.uniform(-0.1, 0.1, blk)
            hp = project_feasible(hp, n_best)
            ho, co = coord_cd(hp, n_best, max_passes=60, deadline=deadline, seed=seed + 777 + r)
            if co < best_c:
                best_h, best_c = ho, co

    # Phase 5: Final intensive CD
    if time.time() < deadline - 5:
        h_final, c_final = coord_cd(best_h, len(best_h), max_passes=300, deadline=deadline, seed=seed + 500)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
