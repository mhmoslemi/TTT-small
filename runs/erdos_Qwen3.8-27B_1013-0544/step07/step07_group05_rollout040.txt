    except Exception:
        pass
    if best_f < cur - 1e-14:
        h[i] = a + best_t
        h[j] = b - best_t
        h = np.clip(h, 0.0, 1.0)
    return h, best_f

def active_pairs(h, n, k_top=5, n_pairs=20):
    """Find pairs that contribute most to the top overlaps."""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mid = n - 1
    top_idx = np.argsort(c)[-k_top:]
    edge_mask = (h > 0.05) & (h < 0.95)
    edge_idx = np.where(edge_mask)[0]
    if len(edge_idx) < 2:
        edge_idx = np.arange(n)
    pairs = set()
    for ki in top_idx:
        s = ki - mid
        if s >= 0:
            for j in edge_idx[:n_pairs]:
                i = j + s
                if 0 <= i < n and i != j:
                    pairs.add((min(i, j), max(i, j)))
        else:
            ss = -s
            for j in edge_idx[:n_pairs]:
                i = j - ss
                if 0 <= i < n and i != j:
                    pairs.add((min(i, j), max(i, j)))
    return list(pairs)[:n_pairs * 2]

def pair_descent(h, n, deadline, seed, n_rounds=200, pairs_per_round=30):
    rng = np.random.RandomState(seed)
    cur = erdos_c5(h, n)
    best_h = h.copy()
    best_c = cur
    stall = 0
    for rnd in range(n_rounds):
        if time.time() > deadline - 10.0:
            break
        improved = False
        active = active_pairs(h, n, k_top=5, n_pairs=10)
        random_pairs = [(int(rng.randint(n)), int(rng.randint(n))) for _ in range(pairs_per_round)]
        random_pairs = [(a, b) for a, b in random_pairs if a != b]
        all_pairs = active + random_pairs
        rng.shuffle(all_pairs)
        for i, j in all_pairs:
            if i == j:
                continue
            if time.time() > deadline - 10.0:
                break
            h, c = opt_pair(h, n, min(i, j), max(i, j), deadline - 10.0)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                improved = True
            if c < cur - 1e-14:
                cur = c
                improved = True
        h = project_box_sum(h, 0.5 * n)
        if not improved:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0
    return best_h, best_c

def resample(h, n, nh):
    x = np.linspace(0.0, 2.0, n)
    y = np.linspace(0.0, 2.0, nh)
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * nh)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(20.0, float(budget_s) - 8.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 400
        h_init = np.full(n_init, 0.5)

    best_h = None
    best_n = 0
    best_c = 1e10

    # Phase 1: Try multiple initializations with L-BFGS
    end1 = t0 + 0.25 * total
    n_trials = [400, 300, 250, 200]
    for ni in n_trials:
        if time.time() > end1 - 30.0:
            break
        for kind in range(7):
            if time.time() > end1 - 15.0:
                break
            rng2 = np.random.RandomState(seed + kind * 100 + ni)
            if ni == n_init and kind == 0:
                h0 = project_box_sum(h_init, 0.5 * ni) if ni == n_init else resample(h_init, n_init, ni)
            else:
                h0 = make_structured(ni, kind, rng2)
            for beta in [500.0, 5000.0, 50000.0]:
                if time.time() > end1 - 5.0:
                    break
                rem = end1 - 5.0 - time.time()
                if rem < 10.0:
                    break
                mx = int(min(150, max(20, rem / 2.0)))
                hc, cc = lbfqg_refine(h0, ni, beta, mx, deadline)
                if cc < best_c:
                    best_c = cc
                    best_h = hc.copy()
                    best_n = ni
                h0 = hc.copy()

    # Phase 2: Aggressive pair coordinate descent
    end2 = t0 + 0.75 * total
    if best_h is not None:
        n_pdw = best_n
        h_pdw = best_h.copy()
        # Also try at different resolutions
        for nh_try in [200, 250, 300, 400]:
            if time.time() > end2 - 60.0:
                break
            if nh_try == best_n:
                h_td = h_pdw.copy()
            else:
                h_td = resample(best_h, best_n, nh_try)
            hc, cc = pair_descent(h_td, nh_try, end2 - 30.0, seed + nh_try,
                                  n_rounds=300, pairs_per_round=40)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()
                best_n = nh_try
                h_pdw = hc.copy()
        # Continue on best so far
        h_pdw, c_pdw = pair_descent(best_h, best_n, end2 - 5.0, seed + 9999,
                                    n_rounds=500, pairs_per_round=50)
        if c_pdw < best_c:
            best_c = c_pdw
            best_h = h_pdw.copy()

    # Phase 3: L-BFGS polish on the pair-descent result
    end3 = deadline - 20.0
    if best_h is not None:
        for beta in [10000.0, 100000.0, 1000000.0]:
            if time.time() > end3 - 15.0:
                break
            rem = end3 - 15.0 - time.time()
            if rem < 10.0:
                break
            mx = int(min(100, max(15, rem / 3.0)))
            hc, cc = lbfqg_refine(best_h, best_n, beta, mx, deadline)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

        # Final pair polish
        if time.time() > end3 - 10.0:
            end3 = deadline - 5.0
        hc, cc = pair_descent(best_h, best_n, end3, seed + 7777,
                              n_rounds=200, pairs_per_round=25)
        if cc <= best_c:
            best_c = cc
            best_h = hc.copy()

    # Final projection and evaluation
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
