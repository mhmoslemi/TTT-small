    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha

    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())

        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:

            g[s_:] -= w * x[:m_]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    return m + np.log(s) / alpha if s > 0 else m

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros(len(c)); q[int(np.argmax(c))] = 1.0
    else:
        q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            g[k:n] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            s_ = -k
            g[:n-s_] += w * (1.0 - x[s_:])
            g[s_:] -= w * x[:n-s_]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def sum_pen(x, n, mu):
    d = float(x.sum() - 0.5 * n)
    return 0.5 * mu * d * d

def full_obj(x, n, alpha, mu):
    return lse_obj(x, n, alpha) + sum_pen(x, n, mu)

def full_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def lbfgsb_opt(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(full_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=full_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 3*maxiter, "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def sum_con_fun(x, n):
    return float(x.sum() - 0.5 * n)

def slsqp_opt(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_con_fun, "args": (n,)}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:

def compute_coeff(h, i, j, n):
    """Compute the coefficient vector for pair (i,j) move."""
    off = n - 1
    L = 2 * n - 1
    coeff = np.zeros(L)
    for k in range(-(n-1), n):
        ci = 0.0
        if 0 <= i + k <= n - 1:
            ci += 1.0 - h[i + k]
        if 0 <= i - k <= n - 1:
            ci -= h[i - k]
        cj = 0.0
        if 0 <= j + k <= n - 1:
            cj += 1.0 - h[j + k]
        if 0 <= j - k <= n - 1:
            cj -= h[j - k]
        coeff[off + k] = ci - cj
    return coeff

def pair_optimize(h, c, i, j, n):
    """Find optimal move for pair (i,j) keeping sum constant. Returns (x_new, y_new, val) or None."""
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    dx = 2.0 / n
    coeff = compute_coeff(h, i, j, n)
    a = coeff * dx
    dmin = lo - x0
    dmax = hi - x0
    # Golden section search on f(d) = max(c + d*a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    pa, pb = dmin, dmax
    pc = pb - gr * (pb - pa)
    pd = pa + gr * (pb - pa)
    fc = float(np.max(c + pc * a))
    fd = float(np.max(c + pd * a))
    for _ in range(60):
        if fc < fd:
            pb = pd; pd = pc; fd = fc
            pc = pb - gr * (pb - pa)
            fc = float(np.max(c + pc * a))
        else:
            pa = pc; pc = pd; fc = fd
            pd = pa + gr * (pb - pa)
            fd = float(np.max(c + pd * a))
    d_opt = 0.5 * (pa + pb)
    d_opt = float(np.clip(d_opt, dmin, dmax))
    val = float(np.max(c + d_opt * a))
    if val < float(c.max()) - 1e-12:
        xn = float(np.clip(x0 + d_opt, 0.0, 1.0))
        yn = float(np.clip(S - xn, 0.0, 1.0))
        return xn, yn, val
    return None

def pair_search(h, deadline, seed):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    noimp = 0
    maxno = 3000
    bestc_ref = bestc
    stall = 0

    while time.time() < deadline:
        if noimp > maxno:
            break
        # Select pair: 70% weighted by worst lags, 30% random
        if rng.random() < 0.70:
            # Focus on lags where correlation is high
            topk = min(20, n - 1)
            idx_k = np.argpartition(c, -topk)[-topk:]
            k_pick = idx_k[int(rng.randint(topk))]
            d = abs(int(k_pick - off))
            if d == 0 or d >= n:
                d = int(rng.randint(1, n))
            i = int(rng.randint(n - d))
            j = i + d
        else:
            d = int(rng.randint(1, n))
            i = int(rng.randint(n - d))
            j = i + d
        if i == j:
            continue
        res = pair_optimize(h, c, i, j, n)
        if res is not None:
            x, y, val = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
                noimp = 0
                stall = 0
            else:
                noimp += 1
                # Undo if not best
                h[i] = x  # keep the move (it reduced the objective even if not the global max)
                noimp += 1
        else:
            noimp += 1

        # Periodic restart if stalled
        stall += 1
        if stall > 2000 and rng.random() < 0.1:
            h = besth.copy() + 0.005 * rng.randn(n)
            h = project_box_sum(h, n)
            c = compute_c(h, n)
            stall = 0

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

        if time.time() > deadline - 30:
            break
        h2 = lbfgsb_opt(h, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
            h = h2
        else:
            h = besth.copy()

    # Pair search
    if time.time() < deadline - 5:
        besth, bestc = pair_search(besth, deadline, seed=seed)

    return besth, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 384
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.45 * np.sin(3.0 * np.pi * x) + 0.1 * np.sin(5.0 * np.pi * x)

    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0

    # Phase 1: Optimize at initial n (most time)
    dl1 = t0 + (deadline - t0) * 0.45
    h1, c1 = multi_start_opt(besth, bestn, seed, dl1)
    if c1 < bestc:
        bestc = c1
        besth = h1.copy()
        bestn = len(h1)

    # Phase 2: Try alternative n values
    n_candidates = [n0]
    for m in [256, 288, 320, 352, 384, 420, 448, 480]:
        if m != n0 and 128 <= m <= 512:
            n_candidates.append(m)
    # Remove duplicates, keep order
    seen = set()
    n_candidates = [x for x in n_candidates if x not in seen and not seen.add(x)]

    remaining_dl = deadline - 10
    elapsed = time.time() - t0
    remaining = remaining_dl - time.time()
    per_n = remaining / max(1, len(n_candidates) - 1)

    for ni in n_candidates[1:]:
        if time.time() > remaining_dl:
            break
        dl_n = min(time.time() + per_n, remaining_dl)
        if dl_n <= time.time() + 15:
            dl_n = min(time.time() + 15, remaining_dl)
        hinit = interpolate_h(besth, bestn, ni)
        hinit = project_box_sum(hinit + 0.003 * rng.randn(ni), ni)
        hn, cn = multi_start_opt(hinit, ni, seed + 100 * ni, dl_n)
        if cn < bestc:
            bestc = cn
            besth = hn.copy()
            bestn = ni

    # Phase 3: Final polish on best
    if time.time() < deadline - 15:
        hf, cf = slsqp_opt(besth, bestn, 500.0, 150)
        hf = project_box_sum(hf, bestn)
        cf = float(compute_c(hf, bestn).max())
        if cf < bestc:
            bestc = cf
            besth = hf.copy()
        if time.time() < deadline - 10:
            besth, bestc = pair_search(besth, deadline, seed=seed + 9999)

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
