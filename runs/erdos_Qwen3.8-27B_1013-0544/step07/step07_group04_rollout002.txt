def soft_weights(c, alpha):
    m=float(c.max())
    if alpha <= 1e-8:
       q=np.full(c.shape, 1.0/c.size); return q

    s=float(e.sum())

import numpy as np
from scipy.optimize import minimize
import time


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def is_better(c, n, bestc, bestn):
    if not np.isfinite(c):
        return False
    if c < bestc - 1e-15:
        return True
    return abs(c - bestc) <= 1e-12 and n < bestn


def tighten_sum(v, n):
    if n <= 0:
        return v
    tgt = 0.5 * float(n)
    for _ in range(8):
        d = tgt - float(v.sum())
        if abs(d) < 1e-15:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        idxs = np.flatnonzero(m)
        if idxs.size:
            if d > 0:
                room = 1.0 - v
                pos = int(idxs[np.argmax(room[idxs])])
            else:
                pos = int(idxs[np.argmax(v[idxs])])
            nv = v[pos] + d
            if 0.0 <= nv <= 1.0:
                v[pos] = nv
                continue
            v[idxs] += d / float(idxs.size)
            np.clip(v, 0.0, 1.0, out=v)
        else:
            v += d / float(n)
            np.clip(v, 0.0, 1.0, out=v)
    return v


def project_box_sum(v, n):
    if n <= 0:
        return np.array([], dtype=float)
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    if v.size != n:
        v = np.resize(v, n)
    tgt = 0.5 * float(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_value(c, alpha):
    m = float(c.max())
    if alpha <= 1e-8:
        return m
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + float(np.log(s)) / alpha


def soft_weights(c, alpha):
    m = float(c.max())
    L = c.size
    if alpha <= 1e-8:
        return np.full(L, 1.0 / float(L))
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        q = np.zeros(L)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    q[q < 1e-12] = 0.0
    s2 = float(q.sum())
    if s2 > 0.0:
        q /= s2
    else:
        q[int(np.argmax(c))] = 1.0
    return q


def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return 1e100
    return smooth_value(c, alpha)


def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = soft_weights(c, alpha)
    g = np.zeros(n)
    off = n - 1
    dx = 2.0 / float(n)
    nz = np.flatnonzero(q > 1e-12)
    for idx in nz:
        k = int(idx - off)
        w = float(q[idx])
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            m = n - k
            g[k:] += w * (1.0 - x[:m])
            g[:m] -= w * x[k:]
        else:
            s = -k
            m = n - s
            g[:m] += w * (1.0 - x[s:])
            g[s:] -= w * x[:m]
    return dx * g


def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * float(n))
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d


def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * float(n))
    return lse_grad(x, n, alpha) + mu * d


def lbfgsb_phase(h, n, alpha, mu, maxiter):

    if n < 3 or maxiter < 1:
        return h
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            pen_obj, h, args=(n, float(alpha), float(mu)),
            method="L-BFGS-B", jac=pen_grad, bounds=bounds,
            options={"maxiter": int(maxiter), "maxfun": 4 * int(maxiter),
                     "ftol": 1e-16, "gtol": 1e-9}
        )
        if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return h


def sum_fun(x, n):
    return float(x.sum() - 0.5 * float(n))


def sum_jac(x, n):
    return np.ones(x.size)


def slsqp_phase(h, n, alpha, maxiter):

    if n < 3 or maxiter < 1:
        return h
    try:

        )
        if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return h

        f = 4.5 + 1.0 * rng.rand()
        h = 0.5 + 0.30 * np.sin(np.pi * f * x + 0.2) + 0.15 * rng.randn() * np.cos(2.0 * np.pi * f * x)
    return project_box_sum(h, n)


def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 24)))
    a = max(lo, x0 - 0.04)
    b = min(hi, x0 + 0.04)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 96)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)


def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / float(n)
    omh = 1.0 - h
    omhrev = omh[::-1]
    dx1 = xs - x0
    dx2 = x0 - xs
    yv = S - xs

    delta = np.zeros((len(xs), L), dtype=float)
    delta[:, i:i + n] += dx1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += dx2[:, None] * h[None, :]
    delta[:, j:j + n] += dx2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += dx1[:, None] * h[None, :]

    idx0 = off
    base_self = dx1 * (1.0 - x0) + x0 * dx2 + dx2 * (1.0 - y0) + y0 * dx1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self

    d = j - i
    idxp = off - d
    base_p = dx1 * (1.0 - y0) + x0 * dx1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p

    idxm = off + d
    base_m = dx2 * (1.0 - x0) + y0 * dx2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m

    return np.max(c_base[None, :] + dx * delta, axis=1)


def pair_search(h, c_base, i, j, n, target):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None

    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)

    for _ in range(2):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-12:
            break
        xs2 = np.linspace(a, b, 64)
        vals = pair_eval(h, c_base, i, j, xs2, n, x0, y0, S)
        xs = xs2

    k = int(np.argmin(vals))
    bv = float(vals[k])
    limit = min(float(target), float(c_base.max()))
    if bv < limit - 1e-14:
        xb = float(np.clip(xs[k], lo, hi))
        yb = S - xb
        if yb < 0.0:
            yb = 0.0
            xb = S - yb
        elif yb > 1.0:
            yb = 1.0
            xb = S - yb
        return xb, yb, bv
    return None


def pair_local_search(h, deadline, seed=1):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    curc = float(c.max())
    bestc = curc
    besth = h.copy()
    if n < 2 or time.time() > deadline or not np.isfinite(curc):
        besth = project_box_sum(besth, n)
        return besth, float(compute_c(besth, n).max())

    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 1000 if n > 300 else 2200
    off = n - 1
    pokes = 0
    maxpokes = 8

    while time.time() < deadline:
        if noimp > maxno:
            if pokes < maxpokes and time.time() < deadline - 30:
                amp = 0.003 * float(pokes + 1)
                if amp > 0.02:
                    amp = 0.02
                h = project_box_sum(besth + amp * rng.randn(n), n)
                c = compute_c(h, n)
                curc = float(c.max())
                pokes += 1
                noimp = 0
                maxno = int(max(450, 0.65 * maxno))
            else:
                break

        if rng.random() < 0.85:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-10
            ws = w / float(w.sum())
            d = int(rng.choice(n - 1, p=ws)) + 1
            m = n - d
            if m > 1 and rng.random() < 0.85:
                a = h[:m]
                b = h[d:d + m]
                loc = a * (1.0 - b) + b * (1.0 - a)
                lw = loc + 1e-10
                ls = float(lw.sum())
                if ls > 0.0:
                    i = int(rng.choice(m, p=lw / ls))
                else:
                    i = int(rng.randint(m))
            else:
                i = int(rng.randint(m))
        else:
            d = int(rng.randint(1, n))
            m = n - d
            i = int(rng.randint(m))

        j = i + d
        res = pair_search(h, c, i, j, n, curc)
        if res is not None:
            oldi, oldj = h[i], h[j]
            h[i] = res[0]
            h[j] = res[1]
            cnew = compute_c(h, n)
            vn = float(cnew.max())
            if vn < curc - 1e-15:
                c = cnew
                curc = vn
                noimp = 0
                if vn < bestc - 1e-15:
                    bestc = vn
                    besth = h.copy()
            else:
                h[i] = oldi
                h[j] = oldj
                noimp += 1
        else:
            noimp += 1

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return h, 1e100
    bestc = float(c.max())
    besth = h.copy()
    work = h.copy()
    workc = bestc
    if time.time() > deadline - 2:
        return project_box_sum(besth, n), bestc

    avail = max(1.0, deadline - time.time())
    reserve = max(18.0, min(110.0, 0.22 * avail))

    phases = (
        (6.0, 8.0, 140),
        (20.0, 20.0, 110),
        (80.0, 60.0, 90),
        (300.0, 200.0, 70),
        (1200.0, 1200.0, 55),
    )
    for alpha, mu, maxit in phases:
        if time.time() > deadline - reserve:
            break
        h2 = lbfgsb_phase(work, n, alpha, mu, maxit)
        h2 = project_box_sum(h2, n)
        c2 = compute_c(h2, n)
        if c2.size == 0 or not np.all(np.isfinite(c2)):
            continue
        v2 = float(c2.max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        if v2 <= workc + 1e-7:
            work = h2
            workc = v2
        else:
            work = besth.copy()
            workc = bestc

    if time.time() < deadline - max(25.0, 0.45 * reserve) and n <= 520:
        h2 = slsqp_phase(work, n, 800.0, 45)
        h2 = project_box_sum(h2, n)
        c2 = compute_c(h2, n)
        if c2.size and np.all(np.isfinite(c2)):
            v2 = float(c2.max())
            if v2 < bestc:
                bestc = v2
                besth = h2.copy()
                work = h2.copy()
                workc = v2

    if time.time() < deadline - 1:
        besth, bestc = pair_local_search(besth, deadline, seed)

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def final_polish(h, n, seed, deadline):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return h, 1e100
    besth = h.copy()
    bestc = float(c.max())

    if time.time() < deadline - 55 and n <= 520:
        h2 = lbfgsb_phase(h, n, 5000.0, 30000.0, 40)
        h2 = project_box_sum(h2, n)
        c2 = compute_c(h2, n)
        if c2.size and np.all(np.isfinite(c2)):
            v2 = float(c2.max())
            if v2 < bestc:
                bestc = v2
                besth = h2.copy()
        if time.time() < deadline - 30:
            h3 = slsqp_phase(besth, n, 5000.0, 30)
            h3 = project_box_sum(h3, n)
            c3 = compute_c(h3, n)
            if c3.size and np.all(np.isfinite(c3)):
                v3 = float(c3.max())
                if v3 < bestc:
                    bestc = v3
                    besth = h3.copy()

    if time.time() < deadline - 1:
        besth, bestc = pair_local_search(besth, deadline, seed + 17)

    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    bud = max(1.0, float(budget_s))
    deadline = t0 + max(1.0, bud - 8.0)
    rng = np.random.RandomState(seed)

    try:
        hp = np.asarray(initial_h_values, dtype=float).reshape(-1).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        hp = 0.5 + 0.35 * np.sin(3.0 * np.pi * x) + 0.10 * np.sin(7.0 * np.pi * x)

    if n0 < 16 or n0 > 600:
        nn = 400
        if n0 > 0:
            hp = interpolate_h(hp, n0, nn)
        else:
            x = (np.arange(nn) + 0.5) * (2.0 / float(nn))
            hp = 0.5 + 0.35 * np.sin(3.0 * np.pi * x)
        n0 = nn

    hp = project_box_sum(hp, n0)
    c0 = compute_c(hp, n0)
    besth = hp.copy()
    bestn = n0
    bestc = float(c0.max()) if c0.size else 1e100

    for var in (1, 2, 3, 4):
        hs = quick_start(hp, n0, rng, var)
        cs = float(compute_c(hs, n0).max())
        if is_better(cs, n0, bestc, bestn):
            besth = hs
            bestc = cs

    total = max(1.0, deadline - t0)
    core_t = total * 0.82
    if total > 30.0:
        core_t = min(core_t, total - 15.0)
    core_end = t0 + max(1.0, core_t)
    end1 = t0 + max(1.0, min(total * 0.48, max(1.0, total - 20.0)))
    end2 = t0 + max(end1 + 5.0, min(core_t - 10.0, total * 0.66))

    if time.time() < end1 - 3:
        h1, c1 = optimize_n(besth, bestn, seed + 1, end1)
        if is_better(c1, bestn, bestc, bestn):
            besth = h1
            bestc = c1

    alt = []
    for m in (400, 350, 300, 256):
        if m != bestn and 16 <= m <= 700 and m not in alt:
            alt.append(m)
        if len(alt) >= 2:
            break

    if alt and time.time() < end2 - 3:
        n2 = alt[0]
        h2s = quick_start(besth, n2, rng, 2)
        h2, c2 = optimize_n(h2s, n2, seed + 11, end2)
        if is_better(c2, n2, bestc, bestn):
            besth = h2
            bestc = c2
            bestn = n2

    if len(alt) > 1 and time.time() < core_end - 3:
        n3 = alt[1]
        h3s = quick_start(besth, n3, rng, 3)
        h3, c3 = optimize_n(h3s, n3, seed + 21, core_end)
        if is_better(c3, n3, bestc, bestn):
            besth = h3
            bestc = c3
            bestn = n3

    if time.time() < deadline - 3:
        hf, cf = final_polish(besth, bestn, seed + 99, deadline)
        if is_better(cf, bestn, bestc, bestn):
            besth = hf
            bestc = cf

    besth = project_box_sum(besth, bestn)
    cf = float(compute_c(besth, bestn).max())
    return besth, cf, int(bestn)
