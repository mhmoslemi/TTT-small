def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    return m + np.log(s) / alpha if s > 0 else m

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    q = e / s if s > 0 else np.zeros_like(c)
    if s <= 0:
        q[int(np.argmax(c))] = 1.0
    off = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            g[k:] += w * (1.0 - x[:n-k])
            g[:n-k] -= w * x[k:]
        elif k < 0:
            s = -k
            g[:n-s] += w * (1.0 - x[s:])
            g[s:] -= w * x[:n-s]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def lbfgsb_opt(h, n, alpha, mu, maxiter):
    def obj(x):
        return lse_obj(x, n, alpha) + 0.5 * mu * (x.sum() - 0.5*n)**2
    def grad(x):
        return lse_grad(x, n, alpha) + mu * (x.sum() - 0.5*n)
    try:
        res = minimize(obj, h, method="L-BFGS-B", jac=grad,
                       bounds=[(0,1)]*n,
                       options={"maxiter": maxiter, "maxfun": 4*maxiter, "ftol": 1e-15})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
