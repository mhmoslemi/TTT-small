import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps, dl):
    """Coordinate descent minimizing max correlation."""

def project_feasible(h, n):

            if s < 1e-14:

            break
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sm = c.max()
        shifted = alpha * (c - sm)
        return sm + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha

    def cons(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_search(h_init, n, dl):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float) * dx
    cur_c = float(corr.max())
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()

    for _ in range(300):
        if time.time() > dl:
            break
        best_gain = 1e-14
        best_pair = None
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                nc = float(np.correlate(h, 1 - h, mode='full').max() * dx)
                h[i] = 0.0
                h[j] = 1.0
                gain = cur_c - nc
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float) * dx
        ones.remove(j)
        zeros.remove(i)
        ones.append(i)
        zeros.append(j)
        cur_c = float(corr.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        excess = np.maximum(c - t, 0)
        return 0.5 * np.sum(excess ** 2)

    def cons(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 20
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Coordinate descent (fast, O(n^2) per sweep)
    p1 = t0 + 60
    h, c = coord_descent(h, n, p1, max_sweeps=20000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: LSE + L-BFGS-B with increasing alpha
    p2 = t0 + 320
    h = best_h.copy()
    log_alphas = np.linspace(np.log(50), np.log(50000000), 12)
    for la in log_alphas:
        if time.time() > p2 - 15:
            break
        alpha = int(np.exp(la))
        mi = 300 if alpha < 1000 else (500 if alpha < 1e5 else 800)
        h = lse_lbfgsb(h, n, alpha, mi, p2 - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Binary local search
    p3 = t0 + 470
    hb, cb = binary_search(best_h, n, p3)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: Coordinate descent on best solution
    p4 = t0 + 620
    h, c = coord_descent(best_h, n, p4, max_sweeps=20000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: LSE + SLSQP fine polish
    p5 = t0 + 850
    h = best_h.copy()
    for alpha in [1000000, 5000000, 20000000, 80000000, 200000000]:
        if time.time() > p5 - 20:
            break
        h = lse_slsqp(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Barrier push
    p6 = dl - 30
    h = best_h.copy()
    c = best_c
    for i in range(8):
        if time.time() > p6 - 3:
            break
        t_target = c * (1.0 - 0.002 * (i + 1))
        h = barrier_slsqp(h, n, t_target, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Final coordinate polish
    h, c = coord_descent(best_h, n, dl - 5, max_sweeps=5000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
