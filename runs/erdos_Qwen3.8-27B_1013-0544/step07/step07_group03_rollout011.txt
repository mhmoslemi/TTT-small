def subgrad_step(h, n, alpha, lr):
    """One subgradient step with log-sum-exp smoothing."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    # Log-sum-exp for smooth max
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    obj_val = m + logsum / alpha
    # Subgradient: weighted average of subgradients of top lags
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    grad = np.zeros(n)
    # For each lag k, the subgradient of c_k w.r.t. h is:
    # dc_k/dh[i] = (1-h[i+k]) - h[i-k] (with boundary handling)
    # We compute this for all lags weighted by weights
    # This is expensive to do for all lags, so we use top-k
    top_k = max(1, min(n, int(50 / (weights.max() + 1e-10))))
    # Actually, let's just use the top few lags
    k_vals = np.argsort(c)[::-1][:top_k]
    for p in k_vals:
        k = p - (n - 1)
        w = weights[p] / (sum(weights[k_vals]))  # renormalize
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            kk = -k

                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n-kk] += w * (1.0 - h[kk:])
                grad[kk:] -= w * h[:n-kk]
    return h - lr * grad, obj_val

    best_c = compute_c5(h, n)

    alpha = 1000.0
    t = 0
    stall = 0
    stall_limit = 5000
    for it in range(max_iters):

        t += 1
        alpha = min(alpha * 1.01, 1e8)
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)

            stall = 0

            stall += 1
        if stall > stall_limit:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        # Compute subgradient
        shifted = alpha * (c - cur_c)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        w_sum = weights[top_idx].sum()
        for p in top_idx:
            k = p - (n - 1)
            w = weights[p] / w_sum
            if k >= 0:
                if k == 0:
                    grad += w * (1.0 - 2.0 * h)
                else:
                    grad[k:] += w * (1.0 - h[:n-k])
                    grad[:n-k] -= w * h[k:]
            else:

        G += momentum ** 2

        h = project_feasible(h - lr * momentum, n)
    return best_h, best_c
