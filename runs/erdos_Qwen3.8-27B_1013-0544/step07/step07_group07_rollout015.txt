        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def adam_lse(h, n, alpha, lr_schedule, n_iter, dl):
    """Adam on LSE objective with projection onto feasible set."""
    dx = 2.0 / n
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for t in range(1, n_iter + 1):
        if time.time() > dl:
            break
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        mx = c.max()
        shifted = alpha * (c - mx)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        wts = exps / exps.sum()
        
        # Gradient approximation using top weights
        grad = np.zeros(n)
        k0 = n - 1
        # Use all weights with significant contribution
        thresh = wts.max() * 0.01
        active = np.where(wts > thresh)[0]
        for idx in active:
            w = wts[idx]
            lag = idx - k0
            jp = np.arange(n) + lag
            jm = np.arange(n) - lag
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            t1 = np.zeros(n)
            t1[vp] = 1.0 - h[jp[vp]]
            t2 = np.zeros(n)
            t2[vm] = h[jm[vm]]
            grad += w * dx * (t1 - t2)
        
        lr = lr_schedule(t)
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        m_hat = m_buf / (1.0 - b1 ** t)
        v_hat = v_buf / (1.0 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    
    return best_h, best_c

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0.0)
        return 0.5 * np.sum(excess * excess)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=100):

                k1 = n - 1 + (j - i)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def adam_lse(h, n, alpha, lr0, decay, n_iter, dl):
    dx = 2.0 / n
    mb = np.zeros(n)
    vb = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, n_iter + 1):
        if time.time() > dl:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        mx = c.max()
        sh = alpha * (c - mx)
        sm = sh.max()
        ex = np.exp(sh - sm)
        w = ex / ex.sum()
        # Gradient via active lags
        g = np.zeros(n)
        k0 = n - 1
        thr = w.max() * 0.005
        for idx in np.where(w > thr)[0]:
            lag = idx - k0
            wv = w[idx]
            jp = np.arange(n) + lag
            jm = np.arange(n) - lag
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            g[vp] += wv * (1.0 - h[jp[vp]])
            g[vm] -= wv * h[jm[vm]]
        g *= dx
        lr = lr0 / (1.0 + decay * t)
        mb = b1 * mb + (1 - b1) * g
        vb = b2 * vb + (1 - b2) * g * g
        mh = mb / (1 - b1 ** t)
        vh = vb / (1 - b2 ** t)
        h = project_feasible(h - lr * mh / (np.sqrt(vh) + eps), n)
        cv = compute_c5(h, n)
        if cv < best_c:
            best_c, best = cv, h.copy()
    return best, best_c

def barrier_slsqp(h, n, t, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        e = np.maximum(c - t, 0.0)
        return 0.5 * np.dot(e, e)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def adam_lse(h, n, alpha, lr0, decay, n_iter, dl):
    dx = 2.0 / n
    mb = np.zeros(n)
    vb = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, n_iter + 1):
        if time.time() > dl:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        mx = c.max()
        sh = alpha * (c - mx)
        sm = sh.max()
        ex = np.exp(sh - sm)
        w = ex / ex.sum()
        g = np.zeros(n)
        k0 = n - 1
        thr = w.max() * 0.003
        active = np.where(w > thr)[0]
        for idx in active:
            lag = idx - k0
            wv = w[idx]
            jp = np.arange(n) + lag
            jm = np.arange(n) - lag
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            g[vp] += wv * (1.0 - h[jp[vp]])
            g[vm] -= wv * h[jm[vm]]
        g *= dx
        lr = lr0 / (1.0 + decay * t)
        mb = b1 * mb + (1 - b1) * g
        vb = b2 * vb + (1 - b2) * g * g
        mh = mb / (1 - b1 ** t)
        vh = vb / (1 - b2 ** t)
        h = project_feasible(h - lr * mh / (np.sqrt(vh) + eps), n)
        cv = compute_c5(h, n)
        if cv < best_c:

    return best, best_c

def barrier_slsqp(h, n, t, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        e = np.maximum(c - t, 0.0)
        return 0.5 * np.dot(e, e)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binary_swap(h_init, n, deadline, max_sweeps=150):
    dx = 2.0 / n
    mrange = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        found = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in ones:
                d = np.zeros(2 * n - 1)
                im = i + mrange
                msk = (im >= 0) & (im < n)
                d[msk] += 1.0 - h[im[msk]]
                imn = i - mrange
                msk = (imn >= 0) & (imn < n)
                d[msk] -= h[imn[msk]]
                jm = j + mrange
                msk = (jm >= 0) & (jm < n)
                d[msk] -= 1.0 - h[jm[msk]]
                jmn = j - mrange
                msk = (jmn >= 0) & (jmn < n)
                d[msk] += h[jmn[msk]]
                d[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(d):
                    d[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(d):
                    d[k2] += 1.0
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    found = True
                    break
            if found:
                break
        if not found:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    h0 = project_feasible(h0, n0)
    best_h, best_c, best_n = h0.copy(), compute_c5(h0, n0), n0

    # Phase 1: Higher resolution + Adam (main improvement vector)
    for tn in [500, 600]:
        if time.time() > dl - 300:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, tn)
        hi = project_feasible(np.interp(xn, xo, best_h), tn)
        ci = compute_c5(hi, tn)
        if ci > best_c + 0.008:
            continue
        dlp = min(time.time() + 200, dl - 150)
        h = hi.copy()
        c = ci
        for alpha in [3000, 10000, 30000, 100000, 300000, 1000000]:
            if time.time() > dlp - 15:
                break
            lr0 = 0.8 * tn / alpha
            h, c = adam_lse(h, tn, alpha, lr0, 0.0003, 1200, dlp - 15)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), tn
        if time.time() < dlp:
            for alpha in [500000, 5000000, 50000000]:
                if time.time() > dlp - 5:
                    break
                h = lse_slsqp(h, tn, alpha, 1200)
                c = compute_c5(h, tn)
                if c < best_c:
                    best_c, best_h, best_n = c, h.copy(), tn

    # Phase 2: SLSQP refinement at best resolution
    n = best_n
    dl2 = t0 + total * 0.52
    h = best_h.copy()
    for alpha in [50000, 200000, 1000000, 5000000, 20000000, 100000000]:
        if time.time() > dl2 - 10:
            break
        mi = 1200 if n > 450 else 2000
        h = lse_slsqp(h, n, alpha, mi)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Barrier push
    dl3 = t0 + total * 0.65
    h, c = best_h.copy(), best_c
    for i in range(18):
        if time.time() > dl3:
            break
        t = c * (1.0 - 0.004 * (i + 1))
        h = barrier_slsqp(h, n, t, 1200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Binary local search + re-optimize
    dl4 = t0 + total * 0.78
    hb, cb = binary_swap(best_h, n, dl4, max_sweeps=100)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    if time.time() < dl4:
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > dl4 - 5:
                break
            h = lse_slsqp(hb, n, alpha, 1000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 5: Diverse restarts with Adam
    dl5 = t0 + total * 0.91
    x = np.linspace(0, 2, n)
    perts = [
        0.01 * rng.randn(n),
        0.015 * np.sin(2 * np.pi * 3.5 * x + 0.5),
        0.008 * np.sin(2 * np.pi * 9.0 * x + 1.2),
        0.02 * rng.randn(n),
        0.006 * np.cos(2 * np.pi * 14.0 * x + 2.0),
        0.012 * np.sin(2 * np.pi * 6.0 * x + 3.0),
    ]
    for p in perts:
        if time.time() > dl5 - 20:
            break
        hp = project_feasible(best_h + p, n)
        lr0 = 0.5 * n / 50000.0
        hp, cp = adam_lse(hp, n, 50000, lr0, 0.002, 600, dl5 - 20)
        if time.time() < dl5 - 12:
            hp = lse_slsqp(hp, n, 2000000, 800)
            cp = compute_c5(hp, n)
        if cp < best_c:
            best_c, best_h = cp, hp.copy()

    # Phase 6: Final barrier + LSE polish
    if time.time() < dl - 10:
        h, c = best_h.copy(), best_c
        for i in range(8):
            if time.time() > dl - 5:
                break
            h = barrier_slsqp(h, n, c * (1 - 0.003 * (i + 1)), 1000)
            c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        for a in [100000000, 500000000]:
            if time.time() > dl - 3:
                break
            h = lse_slsqp(best_h, n, a, 1200)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
