            vf = (idx_f >= 0) & (idx_f < n)

            a[vf] -= h[idx_f[vf]]

            best_f = np.inf

                    best_f = f
                    best_i = i

            w_lo = grid[max(0, best_i - 1)]
            w_hi = grid[min(29, best_i + 1)]
            for _ in range(30):

                m2 = w_hi - (w_hi - w_lo) / 3.0
                f1 = float(np.max(c + a * m1 + b_vec * m1 * m1))
                f2 = float(np.max(c + a * m2 + b_vec * m2 * m2))
                if f1 < f2:
                    w_hi = m2
                else:
                    w_lo = m1
            
            best_delta = (w_lo + w_hi) / 2.0
            if abs(best_delta) > tol:
                h[p] = float(np.clip(h0 + best_delta, 0.0, 1.0))
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
    
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter=300, deadline=None):
    dx = 2.0 / n
    best_h = h.copy()

            mx = shifted.max()

            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    
    return best_h, best_c
