import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(y.sum() - target) < 1e-14:
        return y
    lo, hi = float(y.min())-1.0, float(y.max())+1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        y = np.clip(x-mid, 0.0, 1.0)
        if y.sum() > target: lo = mid
        else: hi = mid
    y = np.clip(x-0.5*(lo+hi), 0.0, 1.0)
    diff = target - y.sum()
    for _ in range(5):
        if abs(diff) < 1e-13: break
        mask = (y > 1e-12) & (y < 1.0-1e-12)
        cnt = max(1, int(mask.sum()))
        y[mask] += diff/cnt if mask.sum()>0 else diff/n
        y = np.clip(y, 0.0, 1.0)
        diff = target - y.sum()
    return y

def softmax_values(c, beta):
    mc = float(np.max(c))
    z = beta*(c-mc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    s = float(ew.sum())
    if s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mc, w
    return mc + np.log(s)/beta, ew/s

def smooth_grad(h, w, n, dx):
    mid = n-1
    w_pos = np.zeros(n); w_neg = np.zeros(n)
    w_pos[1:] = w[mid+1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pp = np.cumsum(w_pos)
    pn = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pp + pn[n-1-idx]
    kern = np.zeros(2*n-1)
    kern[mid+1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid+n]
    return (w[mid]*(1.0-2.0*h) + consts - neigh) * dx

LS = {}
def l_fg(x):
    n, dx, target, beta, rho = LS['n'], LS['dx'], LS['t'], LS['b'], LS['r']
    c = np.correlate(x, 1.0-x, mode='full') * dx
    f, w = softmax_values(c, beta)
    g = smooth_grad(x, w, n, dx)
    res = float(x.sum()-target)
    if rho != 0:
        f += 0.5*rho*res*res
        g = g + rho*res
    return float(f), g

def refine_lbfgs(h, beta, maxiter, deadline):

    rem = deadline - time.time()

    mx = int(min(maxiter, max(15, rem/3)))
    bounds = [(0.0,1.0)]*n
    try:

                       options={'maxiter':mx,'ftol':1e-18,'gtol':1e-14,'maxls':40})
        cand = project_box_sum(r.x, 0.5*n)
    except:

    return cand, erdos_c5(cand, n)

def coord_sweep(h, n, deadline, n_grid=300):
    dx = 2.0/n
    tg = np.linspace(0.0, 1.0, n_grid)
    c = np.correlate(h, 1.0-h, mode='full')
    s_arr = np.arange(-(n-1), n)
    improved = 0
    for i in range(n):
        if time.time() > deadline: break
        v = h[i]
        A = np.zeros(2*n-1)
        ip = i + s_arr
        im = i - s_arr
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        A[vp] = 1.0 - h[ip[vp]]
        A[vm] -= h[im[vm]]
        B = c - v*A
        vals = B[:,None] + tg[None,:]*A[:,None]
        f = vals.max(axis=0)
        t_opt = tg[int(np.argmin(f))]
        if abs(t_opt - v) > 1e-10:
            c += (t_opt-v)*A
            h[i] = t_opt
            improved += 1
    return h, improved

    cur = erdos_c5(h, n)
    frac = np.where((h>1e-9)&(h<1.0-1e-9))[0]

    for _ in range(3):

        pairs = [(int(top[a]),int(top[b])) for a in range(len(top)) for b in range(a+1,len(top))]

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum()-target)) < 1e-14:
        return y
    lo, hi = float(y.min())-1.0, float(y.max())+1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        y = np.clip(x-mid, 0.0, 1.0)
        if y.sum() > target: lo = mid
        else: hi = mid
    y = np.clip(x-0.5*(lo+hi), 0.0, 1.0)
    diff = target - y.sum()
    for _ in range(5):
        if abs(diff) < 1e-13: break
        mask = (y > 1e-12) & (y < 1.0-1e-12)
        cnt = max(1, int(mask.sum()))
        if mask.sum() > 0: y[mask] += diff/cnt
        else: y += diff/n
        y = np.clip(y, 0.0, 1.0)

def softmax_values(c, beta):
    mc = float(np.max(c))
    z = beta*(c-mc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)

    return mc+np.log(s)/beta, ew/s

def smooth_grad(h, w, n, dx):
    mid = n-1

    wp[1:]=w[mid+1:]; wn[1:]=w[:mid][::-1]
    s=wp+wn; pp=np.cumsum(wp); pn=np.cumsum(wn)
