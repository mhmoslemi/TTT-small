    m = 2*n - 1

    # s offset: idx = s + n - 1
    for s in range(-(n-1), n):

def sqp_linprog(h,n,max_iter,deadline,delta0=0.05,all_constraints=True,top_count=None):
    target = 0.5*n
    h = project_feasible(h,n)
    best_t = compute_tsum(h,n)
    delta = min(0.2, max(1e-4, delta0))
    stall = 0
    m=2*n-1
    if top_count is None: top_count = m if all_constraints else min(m, max(50, n//2))
    obj = np.zeros(n+1); obj[-1]=1.0
    for it in range(max_iter):
        if time.time() > deadline: break
        c = np.correlate(h,1.0-h,mode='full')
        t = float(c.max())
        if t < best_t: best_t=t
        if all_constraints:
            idx = np.arange(m)
        else:
            k = top_count if top_count < m else m
            idx = np.argpartition(c, -k)[-k:]
        G = build_grad_matrix(h,n)
        rows = G[idx]
        b = rows @ h - c[idx]
        A = np.column_stack((rows, -np.ones(len(idx))))
        Aeq = np.zeros(n+1); Aeq[:n]=1.0
        lb = np.minimum(0.0, h - delta)
        ub = np.maximum(1.0, h + delta)
        bounds = list(zip(lb, ub)); bounds.append((0.0, n))
        improved = False
        try:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=target, bounds=bounds, method='highs', options={'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                x1 = project_feasible(res.x[:n], n)
                out = line_search_tsum(h, x1, best_t)
                if out is not None:

import time
import numpy as np
from scipy.optimize import minimize, linprog


def compute_tsum(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def compute_c5(h, n):
    return compute_tsum(h, n) * (2.0 / float(n))


def project_feasible(h, n=None):
    if n is None:
        n = len(h)
    target = 0.5 * float(n)
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(6):
        s = float(x.sum())
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = float((1.0 - x).sum())
            if room <= 1e-18:
                break
            c = min(1.0, diff / room)
            x += c * (1.0 - x)
        else:
            occ = float(x.sum())
            if occ <= 1e-18:
                break
            c = min(1.0, -diff / occ)
            x -= c * x
        x = np.clip(x, 0.0, 1.0)
    if abs(float(x.sum()) - target) > 1e-14:
        for _ in range(10):
            s = float(x.sum())
            diff = target - s
            if abs(diff) < 1e-15:
                break
            if diff > 0.0:
                room = float((1.0 - x).sum())
                if room <= 1e-18:
                    break
                x += (diff / room) * (1.0 - x)
            else:
                occ = float(x.sum())
                if occ <= 1e-18:
                    break
                x -= (-diff / occ) * x
            x = np.clip(x, 0.0, 1.0)
    return x

def interpolate_h(h, n_new):
    n = len(h)
    if n == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)


def make_trapezoid(n, p):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    t = np.abs(x - 1.0)
    p = float(p)
    s = max(1e-12, 1.0 - 2.0 * p)
    h = np.ones(n)
    m = t > p
    h[m] = np.maximum(0.0, 1.0 - (t[m] - p) / s)
    return project_feasible(h, n)


def line_search_tsum(h0, h1, best_t):
    d = h1 - h0
    if d.size == 0:
        return None
    maxd = float(np.max(np.abs(d)))
    if maxd < 1e-13:
        return None
    n = len(h0)
    best_h = None
    best_a = 0.0
    alphas = (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02)
    for a in alphas:
        x = h0 + a * d
        if float(x.min()) < -1e-10 or float(x.max()) > 1.0 + 1e-10:
            continue
        x = project_feasible(x, n)
        t = compute_tsum(x, n)
        if t < best_t - 1e-11:
            best_t = t
            best_h = x.copy()
            best_a = a
    if best_h is None:
        return None
    return best_h, best_t, best_a


def build_grad_matrix(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=float)
    for s in range(-(n - 1), n):
        row = G[s + n - 1]
        lo = max(0, -s)
        hi = min(n, n - s)
        if hi > lo:
            row[lo:hi] += 1.0 - h[s + lo:s + hi]
        lo = max(0, s)
        hi = min(n, n + s)
        if hi > lo:
            row[lo:hi] -= h[lo - s:hi - s]
    return G


def smooth_obj(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = float(c.max())
    y = alpha * (c - mx)
    ym = float(y.max())
    se = np.sum(np.exp(y - ym))
    return (mx + (ym + np.log(se)) / alpha) * dx


def smooth_jac(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = float(c.max())
    y = alpha * (c - mx)
    ym = float(y.max())
    e = np.exp(y - ym)
    se = np.sum(e)
    w = e / se
    G = build_grad_matrix(x, n)
    return dx * (G.T @ w)


def sum_con(x, n):
    return float(x.sum()) - 0.5 * float(n)


def sum_con_jac(x, n):
    return np.ones(len(x))


def slsqp_smooth(h, n, alphas, maxiter, deadline):
    h = project_feasible(h, n)
    best = compute_tsum(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac, "args": (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h, args=(n, alpha), method="SLSQP",
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-13})
            if res.x is not None and np.all(np.isfinite(res.x)):
                nh = project_feasible(res.x, n)
                nt = compute_tsum(nh, n)
                if nt < best:
                    h = nh
                    best = nt
        except Exception:
            pass
    return h, best


def sqp_linprog(h, n, max_iter, deadline, delta0=0.05, all_constraints=True,
                top_count=None, time_budget=None):
    target = 0.5 * float(n)
    h = project_feasible(h, n)
    best_t = compute_tsum(h, n)
    delta = min(0.25, max(1e-5, float(delta0)))
    stall = 0
    m = 2 * n - 1
    t_start = time.time()

    if top_count is None:
        if all_constraints:
            top_count = m
        else:
            top_count = min(m, max(50, n // 2))
    obj = np.zeros(n + 1)
    obj[-1] = 1.0
    Aeq = np.zeros(n + 1)
    Aeq[:n] = 1.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if time_budget is not None and time.time() - t_start > time_budget:
            break

        c = np.correlate(h, 1.0 - h, mode="full")
        t_cur = float(c.max())
        if t_cur < best_t:
            best_t = t_cur

        if top_count >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(c, -top_count)[-top_count:]

        G = build_grad_matrix(h, n)
        rows = G[idx]
        b = rows @ h - c[idx]
        A = np.column_stack((rows, -np.ones(len(idx))))

        lb = np.minimum(0.0, h - delta)
        ub = np.maximum(1.0, h + delta)
        bounds = list(zip(lb, ub))
        bounds.append((0.0, float(n)))

        improved = False
        try:

                              bounds=bounds, method="highs",

                              bounds=bounds, options={"disp": False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                x1 = project_feasible(res.x[:n], n)
                out = line_search_tsum(h, x1, best_t)
                if out is not None:
                    h, best_t, a = out
                    improved = True
                    if a >= 0.5:

                    else:
                        delta = max(1e-6, delta * 0.8)
        except Exception:
            pass

        if not improved:

            except Exception:

        else:

            delta *= 0.5

    h = project_feasible(h, n)
    return h, best_t
