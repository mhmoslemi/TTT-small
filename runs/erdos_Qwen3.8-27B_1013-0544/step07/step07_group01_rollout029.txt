            break

            s = room.sum()
            if s < 1e-12:

            s = h.sum()
            if s < 1e-12:

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

            if s < 1e-13:

            h = h + diff / s * h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        for a in alphas:
            if time.time() > deadline - 30:
                break
            h4 = slsqp(h4, n, a, 2500)
            c4 = compute_c5(h4, n)
            if c4 < best_c:
                best_c, best_h = c4, h4.copy()

    if time.time() < deadline - 10:
        h5, c5 = momentum_refine(best_h, n, deadline - 5, mu_init=0.95, top_k=5)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
