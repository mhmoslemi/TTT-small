import numpy as np
from scipy.optimize import minimize
import time
from itertools import combinations

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    ps = p.sum()
    if ps < 1e-30:
        return smooth_obj(x, n, alpha), np.zeros(n)
    p /= ps
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li] * dx
        if w < 1e-15:
            continue
        if l >= 0 and l < n:
            grad[:n - l] += w * (1.0 - x[l:])
            grad[l:] -= w * x[:n - l]
        elif l < 0:
            ll = -l
            if ll < n:
                grad[ll:] += w * (1.0 - x[:n - ll])
                grad[:n - ll] -= w * x[ll:]
    obj = m + (sm + np.log(ps)) / alpha
    return obj, grad

def lbfgs_opt(h, n, alpha, maxiter, deadline):
    try:
        res = minimize(smooth_grad, h, args=(n, alpha), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'maxls': 30,
                                'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coarse_binary_search(n, deadline, n_rand=50000):
    target = n // 2
    best_h = np.full(n, 0.5)
    best_c = compute_c5(best_h, n)
    if n <= 20:
        for combo in combinations(range(n), target):
            if time.time() > deadline:
                break
            h = np.zeros(n)
            h[list(combo)] = 1.0
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        return best_h, best_c
    rng = np.random.RandomState(99)
    for _ in range(n_rand):
        if time.time() > deadline:
            break
        h = np.zeros(n)
        h[rng.choice(n, size=target, replace=False)] = 1.0
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def subgrad_descent(h, n, deadline, mu=0.93):
    dx = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h, n)
    lr, momentum, stall = dx * 2.0, np.zeros(n), 0
    for _ in range(1000000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        elif lag < 0:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - h[:n - ll])
                grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h = project_feasible(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c, best_h, stall = c, h.copy(), 0
            lr = min(lr * 1.01, dx * 15.0)
        else:
            stall += 1
            if stall > 400:
                lr *= 0.94
                mu = max(mu - 0.003, 0.4)
                if lr < dx * 0.01:
                    lr, stall = dx * 0.5, 0
    return best_h, best_c

def polygonal_descent(h, n, deadline, mu=0.90):
    """Multi-directional subgradient: use top few active constraints."""
    dx = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h, n)
    lr, momentum, stall = dx * 1.5, np.zeros(n), 0
    for _ in range(500000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        top_k = min(5, len(corr))
        idx = np.argsort(corr)[-top_k:]
        wts = corr[idx] - corr.min() + 1e-6
        wts /= wts.sum()
        grad = np.zeros(n)
        for w, li in zip(wts, idx):
            l = li - (n - 1)
            if l >= 0 and l < n:
                grad[:n - l] += w * (1.0 - h[l:])
                grad[l:] -= w * h[:n - l]
            elif l < 0:
                ll = -l
                if ll < n:
                    grad[ll:] += w * (1.0 - h[:n - ll])
                    grad[:n - ll] -= w * h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h = project_feasible(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c, best_h, stall = c, h.copy(), 0
            lr = min(lr * 1.01, dx * 10.0)
        else:
            stall += 1
            if stall > 300:
                lr *= 0.95
                mu = max(mu - 0.002, 0.3)
                if lr < dx * 0.01:
                    lr, stall = dx * 0.5, 0
    return best_h, best_c

def binarize_and_refine(h, n, deadline):
    """Try binarizing and see if it helps."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    excess = h_bin.sum() - target
    if excess > 0:
        idx = np.argsort(h)[:int(excess)]
        h_bin[idx] = 0.0
    elif excess < 0:
        idx = np.argsort(-h)[-int(-excess):]
        h_bin[idx] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < compute_c5(h, n):
        return h_bin, c_bin
    return h, compute_c5(h, n)
