from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_obj_eval(t, base_c, b, old, alpha, mid_idx):
    new_c = base_c + b * (t - old)
    new_c[mid_idx] = base_c[mid_idx] + t * (1.0 - t)
    m = new_c.max()
    shifted = alpha * (new_c - m)
    return (m + np.log(np.exp(shifted).sum())) / alpha

def coord_descent(h, n, alpha, max_sweeps, deadline):
    h = h.copy().astype(float)
    current_c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    s_vals = np.arange(-(n-1), n)
    mid = n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        max_change = 0.0
        for j in range(n):
            old = h[j]
            js = j + s_vals
            jms = j - s_vals
            valid_js = (js >= 0) & (js < n)
            valid_jms = (jms >= 0) & (jms < n)
            
            b = np.zeros(2*n-1)
            mjs = valid_js & (s_vals != 0)
            if mjs.any():
                b[mjs] = 1.0 - h[js[mjs]]
            mjms = valid_jms & (s_vals != 0)
            if mjms.any():
                b[mjms] -= h[jms[mjms]]
            
            base_c = current_c.copy()
            base_c[mid] -= old * (1.0 - old)
            
            res = minimize_scalar(coord_obj_eval, bounds=(0.0, 1.0), method='bounded',
                                  args=(base_c, b, old, alpha, mid),
                                  options={'xatol': 1e-10})
            if res.success and abs(res.x - old) > 1e-12:
                h[j] = res.x
                max_change = max(max_change, abs(res.x - old))
                current_c = base_c + b * (res.x - old)
                current_c[mid] = base_c[mid] + res.x * (1.0 - res.x)
        
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        
        if max_change < 1e-13:
            break
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        return m + np.log(np.exp(shifted).sum()) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

            if c < best_c:

            0.025 * np.cos(2*np.pi*7*x),

        ]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def _coord_obj(t, base_c, b, old, alpha, mid):
    new_c = base_c + b * (t - old)
    new_c[mid] = base_c[mid] + t * (1.0 - t)
    m = new_c.max()
    return (m + np.log(np.exp(alpha * (new_c - m)).sum())) / alpha

def coord_descent(h, n, alpha, max_sweeps, deadline):
    h = h.copy().astype(float)
    current_c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    s_vals = np.arange(-(n - 1), n)
    mid = n - 1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        max_ch = 0.0
        for j in range(n):
            old = h[j]
            js = j + s_vals
            jms = j - s_vals
            b = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n) & (s_vals != 0)
            if m1.any():
                b[m1] = 1.0 - h[js[m1]]
            m2 = (jms >= 0) & (jms < n) & (s_vals != 0)
            if m2.any():
                b[m2] -= h[jms[m2]]
            base_c = current_c.copy()
            base_c[mid] -= old * (1.0 - old)
            res = minimize_scalar(_coord_obj, bounds=(0.0, 1.0), method='bounded',
                                  args=(base_c, b, old, alpha, mid),
                                  options={'xatol': 1e-10})
            if res.success and abs(res.x - old) > 1e-12:
                h[j] = res.x
                max_ch = max(max_ch, abs(res.x - old))
                current_c = base_c + b * (res.x - old)
                current_c[mid] = base_c[mid] + res.x * (1.0 - res.x)
        c_now = compute_c5(h, n)

            best_h = h.copy()
        if max_ch < 1e-13:
            break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})

                h[j] = res.x
                max_ch = max(max_ch, abs(res.x - old))
                current_c = base_c + b * (res.x - old)
                current_c[mid] = base_c[mid] + res.x * (1.0 - res.x)
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        if max_ch < 1e-13:
            break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def _slsqp_obj(x, n, alpha, dx):
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    return m + np.log(np.exp(alpha * (g - m)).sum()) / alpha

def _slsqp_cons(x, n):
    return x.sum() - 0.5 * n

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    obj_fn = partial(_slsqp_obj, n=n, alpha=alpha, dx=dx)
    cons_fn = partial(_slsqp_cons, n=n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fn}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, t_target, maxiter, ftol):
    dx = 2.0 / n
    def _bar_obj(x, n, t_target, dx):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t_target, 0.0)
        return 0.5 * np.sum(excess ** 2)
    obj_fn = partial(_bar_obj, n=n, t_target=t_target, dx=dx)
    cons_fn = partial(_slsqp_cons, n=n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fn}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_push(h, n, c_cur, deadline):
    h = h.copy()
    c = c_cur
    for i in range(15):
        if time.time() > deadline - 3:
            break
        t = c * (1.0 - 0.004 * (i + 1))
        h = barrier_slsqp(h, n, t, 2000, 1e-16)
        c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 8
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent with alpha annealing
    alphas = [3, 10, 40, 150, 600, 2500, 10000, 50000, 200000, 800000]
    p1_end = t0 + 0.52 * total
    for ai, alpha in enumerate(alphas):
        if time.time() > p1_end:
            break
        remaining = p1_end - time.time()
        n_left = len(alphas) - ai
        allocated = remaining / n_left
        if alpha < 100:
            t_lim = time.time() + min(allocated, 0.05 * total)
            max_sw = 300
        elif alpha < 5000:
            t_lim = time.time() + min(allocated, 0.06 * total)
            max_sw = 200
        else:
            t_lim = time.time() + min(allocated, 0.04 * total)
            max_sw = 150
        t_lim = min(t_lim, p1_end)
        use_shuf = ai > 3
        h, c = coord_descent(best_h, n, alpha, max_sw, t_lim,
                             shuffle=use_shuf, seed=seed + ai)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SLSQP polish with increasing alpha
    p2_end = t0 + 0.68 * total
    for alpha in [20000, 100000, 500000]:
        if time.time() > p2_end:
            break
        h = lse_slsqp(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Barrier push
    p3_end = t0 + 0.78 * total
    h, c = barrier_push(best_h, n, best_c, p3_end)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 4: Diverse restarts with coordinate descent
    p4_end = t0 + 0.92 * total
    x = np.linspace(0, 2, n)
    perts = [
        0.02 * rng.randn(n),
        0.03 * np.sin(2 * np.pi * 2.5 * x + 0.3),
        0.02 * np.cos(2 * np.pi * 6.0 * x + 1.1),
        0.04 * rng.randn(n),
        0.015 * np.sin(2 * np.pi * 9.0 * x + 2.0),
        0.025 * rng.randn(n),
    ]
    for pi, pert in enumerate(perts):
        if time.time() > p4_end:
            break
        h_p = project_feasible(best_h + pert, n)
        for alpha in [100, 2000, 30000]:
            if time.time() > p4_end:
                break
            h_p, c_p = coord_descent(h_p, n, alpha, 40, p4_end,
                                     shuffle=True, seed=seed + 100 + pi)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()

    # Phase 5: Final polish
    if time.time() < dl - 12:
        h = lse_slsqp(best_h, n, 1000000, 4000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h, c = barrier_push(best_h, n, best_c, dl - 4)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
