import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective using fast method."""
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax
    shifted = alpha * (g - g.max())
    exp_g = np.exp(shifted)
    a = exp_g / exp_g.sum()
    # w of length n
    w = np.zeros(n)
    w[0] = 2.0 * a[n-1]
    if n > 1:
        w[1:] = a[n:] + a[n-2::-1]
    # b[i]
    prefix = np.zeros(2*n)
    prefix[1:] = np.cumsum(a)
    lo = np.arange(n)
    hi = np.minimum(2*n-1, lo + n)
    b = prefix[hi] - prefix[lo]
    # wh
    wh_full = np.correlate(h, w, mode='full')
    wh = wh_full[n-1:]
    grad = dx * (b - wh)
    return grad, g

def lse_gradient_descent(h, n, deadline, alpha_init=500, alpha_final=1e7, n_iters=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    beta = 0.9
    t = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        t += 1
        # Anneal alpha
        frac = min(1.0, it / (n_iters * 0.8))
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        
        grad, g = compute_grad(h, alpha)
        v = beta * v + (1 - beta) * grad
        # Step size: inverse sqrt of accumulated gradient
        lr = 1.0 / (np.sqrt(t) + 1.0)
        h_new = project_feasible(h - lr * v, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c
