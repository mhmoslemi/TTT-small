```python
import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def cd_sweep(h, n, lam, deadline):
    """One coordinate descent sweep with linearized constraint."""
    dx = 2.0 / n
    h = h.copy()
    fc = np.correlate(h, 1.0 - h, mode='full')
    
    for i in range(n):
        if time.time() > deadline:
            break
        
        # Compute d(fc)/d(h[i])
        slopes = np.zeros(2 * n - 1)
        
        # Term 1: h[i] appears as h[j] in fc[m] when j=i, m = n-1+s, valid if i+s in [0,n-1]
        # d/dh[i] of h[i]*(1-h[i+s]) = (1-h[i+s])
        s_lo, s_hi = -i, n - 1 - i
        if s_lo <= s_hi:
            m_idx = np.arange(n - 1 + s_lo, n - 1 + s_hi + 1)
            j_idx = np.arange(i + s_lo, i + s_hi + 1)
            slopes[m_idx] += (1.0 - h[j_idx])
        
        # Term 2: h[i] appears as h[j+s] in fc[m] when j+s=i, j=i-s
        # d/dh[i] of h[i-s]*(1-h[i]) = -h[i-s]
        s_lo2, s_hi2 = i - (n - 1), i
        if s_lo2 <= s_hi2:
            m_idx2 = np.arange(n - 1 + s_lo2, n - 1 + s_hi2 + 1)
            j_idx2 = np.arange(i - s_lo2, i - s_hi2 + 1)
            slopes[m_idx2] -= h[j_idx2]
        
        # 1D: minimize max_m (fc[m]*dx + (slopes[m]*dx + lam)*delta) over delta in [-h[i], 1-h[i]]
        # = max_m (fc[m]*dx + (slopes[m]*dx + lam)*delta)
        # Transform: t = delta + h[i], t in [0,1]
        # = max_m (fc[m]*dx - (slopes[m]*dx + lam)*h[i] + (slopes[m]*dx + lam)*t)
        
        d_m = slopes * dx + lam
        a2 = fc * dx - d_m * h[i]
        
        lo_t = 0.0
        hi_t = 1.0
        
        # Quick check: if current value is already good, skip
        cur_val = float(np.max(fc * dx))
        
        # Try a few candidate points first
        best_t = h[i]
        best_f = cur_val + lam * (h[i] - h[i])  # = cur_val at current point
        # Actually at t=h[i]: f = max_m(fc[m]*dx) = cur_val
        # The Lagrange term lam*(t-h[i]) = 0 at t=h[i]
        
        res = so.minimize_scalar(
            lambda t: float(np.max(a2 + d_m * t)),
            bounds=(lo_t, hi_t), method='bounded',
            options={'maxiter': 35, 'xatol': 1e-11})
        
        new_t = float(res.x)
        
        if abs(new_t - h[i]) > 1e-12:
            delta = new_t - h[i]
            fc += delta * slopes
            h[i] = new_t
    
    return h

def cd_optimize(h, n, deadline, max_sweeps=200):
    """Coordinate descent with adaptive Lagrange multiplier."""
    target = 0.5 * n
    h = project_box_sum(h, target)
    lam = 0.0
    best_h = h.copy()
    best_c = erdos_c5(h, n)
    eta = 1.0 / n  # step size for lambda
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # periodically reset eta
        if sweep > 0 and sweep % 50 == 0:
            eta = 1.0 / n * 0.5  # decrease step size
        
        h = cd_sweep(h, n, lam, deadline)
        
        # Update lambda
        viol = float(h.sum() - target)
        lam += eta * viol
        
        # Clamp lambda to prevent explosion
        lam = np.clip(lam, -10.0, 10.0)
        
        # Project to feasible
        h = project_box_sum(h, target)
        
        c = erdos_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

```python
def cd_sweep_al(h, n, lam, rho, deadline):

    dx = 2.0 / n
    target = 0.5 * n
    h = h.copy()
    fc = np.correlate(h, 1.0 - h, mode='full')
    cur_sum = float(h.sum())
    
    for i in range(n):
        if time.time() > deadline:
            break
        
        slopes = np.zeros(2 * n - 1)
        
        s_lo, s_hi = -i, n - 1 - i
        if s_lo <= s_hi:
            m_idx = np.arange(n - 1 + s_lo, n - 1 + s_hi + 1)
            j_idx = np.arange(i + s_lo, i + s_hi + 1)
            slopes[m_idx] += (1.0 - h[j_idx])
        
        s_lo2, s_hi2 = i - (n - 1), i
        if s_lo2 <= s_hi2:
            m_idx2 = np.arange(n - 1 + s_lo2, n - 1 + s_hi2 + 1)
            j_idx2 = np.arange(i - s_lo2, i - s_hi2 + 1)
            slopes[m_idx2] -= h[j_idx2]

            fc += delta * slopes
            cur_sum += delta
            h[i] = new_t
    
    return h
```
