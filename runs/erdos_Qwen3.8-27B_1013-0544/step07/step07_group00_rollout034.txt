    h = np.clip(h.copy(), 0.0, 1.0)

    return grad

def slp_step(h, n):
    """One SLP iteration. Returns (h_new, success)."""
    c = np.correlate(h, 1.0-h, mode='full')
    grad = build_grad(h, n)
    d = c - grad @ h
    # LP: min t s.t. -grad@h_new - t <= -d, sum(h_new)=n/2, 0<=h_new<=1
    A_ub = np.column_stack([-grad, -np.ones(2*n-1)])
    b_ub = -d
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    bounds = [(0, 1)]*n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 20000})
        if res.success:
            return res.x[:n], True
    except Exception:
        pass
    return h, False

def slp_optimize(h, n, max_iters=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 1.0
    stall = 0
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        h_new, ok = slp_step(best_h, n)
        if not ok:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-10:
            best_h, best_c, stall = h_new, c_new, 0
            alpha = min(1.0, alpha * 1.5)
        else:
            h_d = (1-alpha)*best_h + alpha*h_new
            h_d = project_feasible(h_d, n)
            c_d = compute_c5(h_d, n)
            if c_d < best_c - 1e-10:
                best_h, best_c, stall = h_d, c_d, 0
            else:
                stall += 1
                alpha = max(0.05, alpha * 0.5)
                if stall > 20:
                    break
    return best_h, best_c

def coord_cd(h, n, max_passes=50, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0-h, mode='full')
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0-1e-14:
                continue
            # Compute a[k] = grad[k, p]
            # a[k] = (1-h[p-lag])*I(0<=p-lag<n) - h[p+lag]*I(0<=p+lag<n) for lag!=0
            # For lag=0: a[n-1] = 1-2*h0
            a = np.zeros(2*n-1)
            # lag > 0: a[n-1+lag] gets (1-h[p-lag]) if p>=lag, and -h[p+lag] if p+lag<n
            # lag < 0 (l=-lag>0): a[n-1-l] gets (1-h[p+l]) if p+l<n, and -h[p-l] if p>=l
            for lag in range(1, n):
                if p >= lag:
                    a[n-1+lag] += 1.0 - h[p-lag]
                if p+lag < n:
                    a[n-1+lag] -= h[p+lag]
                if p+lag < n:
                    a[n-1-lag] += 1.0 - h[p+lag]
                if p >= lag:
                    a[n-1-lag] -= h[p-lag]
            a[n-1] = 1.0 - 2.0*h0
            
            # Bisection to find optimal d
            lo, hi = -h0, 1.0-h0
            for _ in range(40):
                mid = 0.5*(lo+hi)
                vals = c + mid*a
                ks = np.argmax(vals)
                if a[ks] > 1e-12:
                    hi = mid
                elif a[ks] < -1e-12:
                    lo = mid
                else:
                    break
            d = 0.5*(lo+hi)
            if abs(d) > 1e-10:
                h[p] = np.clip(h0+d, 0.0, 1.0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    return h, best_c
