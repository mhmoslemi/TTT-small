def exact_1d_min(a_vec, b_vec, quad_b=0.0, t_lo=0.0, t_hi=1.0):
    """
    Minimize max(max_k (a_k*t + b_k), t - t^2 + quad_b) over [t_lo, t_hi]
    """
    # Find candidate points:
    # 1. t_lo, t_hi
    # 2. Intersection points of pairs of lines that are on the upper envelope
    # 3. Where the quadratic equals a line on the upper envelope
    
    # For simplicity, find all pairwise intersections of lines
    # that are within [t_lo, t_hi] and where both lines are above the others
    
    # Actually, simpler: the minimum of max of convex functions is at a point
    # where at least two functions are equal and are the maximum.
    
    # Candidate points: boundaries + pairwise intersections
    candidates = [t_lo, t_hi]
    m = len(a_vec)
    for i in range(m):
        for j in range(i+1, m):
            if abs(a_vec[i] - a_vec[j]) > 1e-15:
                t_cross = (b_vec[j] - b_vec[i]) / (a_vec[i] - a_vec[j])
                if t_lo <= t_cross <= t_hi:
                    candidates.append(t_cross)
    
    # Also add points where quadratic = each line
    for i in range(m):
        # t - t² + quad_b = a_i*t + b_i
        # t² + (a_i - 1)*t + (b_i - quad_b) = 0
        A = 1.0
        B = a_vec[i] - 1.0
        C = b_vec[i] - quad_b
        disc = B*B - 4*A*C
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            t1 = (-B + sqrt_disc) / (2*A)
            t2 = (-B - sqrt_disc) / (2*A)
            for tc in [t1, t2]:
                if t_lo <= tc <= t_hi:
                    candidates.append(tc)
    
    # Evaluate at all candidates and find the minimum of g
    best_t = t_lo
    best_val = np.inf
    for tc in candidates:
        tc = np.clip(tc, t_lo, t_hi)
        vals = a_vec * tc + b_vec
        quad_val = tc - tc*tc + quad_b
        g = max(vals.max(), quad_val)
        if g < best_val:
            best_val = g
            best_t = tc
    
    return best_t, best_val

Let me define things differently. Let the "shifted" version be:
For lag ℓ (can be negative), define:
(ℓ-shift) overlap = Σ_{x} h(x) * v(x + ℓ) where both x and x+ℓ are in domain.

In the discrete case with h, v ∈ ℝⁿ:
For lag ℓ = k - (n-1) where k = 0, ..., 2n-2:
c[k] = Σ_i h[i] * v[i+ℓ] for i where 0≤i<n and 0≤i+ℓ<n

So c[k] = Σ_i h[i] * (1-h[i+ℓ]) for valid i.

Now, ∂c[k]/∂h[i]:
- Direct term: h[i] appears as the first factor when i is in [0, n-1] and i+ℓ is in [0, n-1]
  Contribution: (1 - h[i+ℓ])
- Cross term: h[i] appears as the second factor (inside 1-h[...]) when i-ℓ is in [0, n-1] and i is in [0, n-1]
  Contribution: -h[i-ℓ]

So: ∂c[k]/∂h[i] = (1-h[i+ℓ]) * 𝟙[0≤i+ℓ<n] - h[i-ℓ] * 𝟙[0≤i-ℓ<n]

where ℓ = k - (n-1).

Let me verify for k = n-1 (ℓ=0):
∂c[n-1]/∂h[i] = (1-h[i]) - h[i] = 1 - 2h[i] ✓

For k = n (ℓ=1):
∂c[n]/∂h[i] = (1-h[i+1]) * 𝟙[i+1<n] - h[i-1] * 𝟙[i-1≥0]
= (1-h[i+1]) if i < n-1 else 0, minus h[i-1] if i > 0 else 0

This looks right.

So the total gradient is:
∇C₅_smoothed(h) = Σ_k w[k] * ∂c[k]/∂h

Let me implement this efficiently:

For each k (with weight w[k] > 0):
ℓ = k - (n-1)
For i in [0, n-1]:
  grad[i] += w[k] * ((1-h[i+ℓ]) if 0≤i+ℓ<n else 0) - (h[i-ℓ] if 0≤i-ℓ<n else 0)

This can be vectorized:
For fixed k, ℓ:
  valid_plus = [i for i in range(n) if 0 ≤ i+ℓ < n] → i in [max(0, -ℓ), min(n-1, n-1-ℓ)]
  grad[max(0,-ℓ) : min(n-1, n-1-ℓ)+1] += w[k] * (1 - h[max(0,-ℓ)+ℓ : min(n-1, n-1-ℓ)+1+ℓ])
  
  valid_minus = [i for i in range(n) if 0 ≤ i-ℓ < n] → i in [max(0, ℓ), min(n-1, n-1+ℓ)]
  grad[max(0,ℓ) : min(n-1, n-1+ℓ)+1] -= w[k] * h[max(0,ℓ)-ℓ : min(n-1, n-1+ℓ)+1-ℓ]

This is O(n) per k, O(n * 2n) = O(n²) total. For n=500, that's 500K operations, very fast.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smoothed_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        lag = k - (n - 1)
        # Term 1: (1-h[i+lag]) for i where 0<=i+lag<n
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += w[k] * (1.0 - h[i_lo+lag : i_hi+1+lag])
        # Term 2: -h[i-lag] for i where 0<=i-lag<n
        i_lo2 = max(0, lag)
        i_hi2 = min(n - 1, n - 1 + lag)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w[k] * h[i_lo2-lag : i_hi2+1-lag]
    
    return grad
