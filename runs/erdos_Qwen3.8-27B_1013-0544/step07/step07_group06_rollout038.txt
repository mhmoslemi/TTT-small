import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def top_n_binary(h, n):
    hb = np.zeros(n)
    t = n // 2
    if t > 0:
        idx = np.argsort(-h)[:t]
        hb[idx] = 1.0
    return hb

def lse_obj_grad(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    v = 1.0 - h
    A = np.convolve(p[::-1], v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    dev = h.sum() - 0.5 * n
    if pen:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def run_lbfgs(h, n, beta, pen, deadline):
    _stop_time = deadline
    res = minimize(lse_obj_grad, h, method="L-BFGS-B", jac=True,
                   bounds=[(0.0, 1.0)] * n, args=(n, beta, pen),
                   callback=stop_callback,
                   options={"maxiter": 200000, "maxls": 40, "ftol": 1e-15, "gtol": 1e-10})
    return res.x if hasattr(res, 'x') else h

def coord_descent(h, n, max_passes=500, deadline=None):
    h = project_feasible(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    lags = np.arange(2 * n - 1) - (n - 1)
    center = n - 1
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        max_d = 0.0
        for i in range(n):
            old = h[i]
            idx1 = i + lags
            idx2 = i - lags
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            eff = np.zeros(2 * n - 1)
            eff[m1] += (1.0 - h[idx1[m1]])
            eff[m2] -= h[idx2[m2]]
            c_eff = 1.0 - 2.0 * old
            a, b = 0.0, 1.0
            fa = float(np.max(corr + (0 - old) * eff)) 
            fa = fa * dx
            # ternary
            for _ in range(30):
                m1v = a + (b - a) / 3.0
                m2v = b - (b - a) / 3.0
                d1 = m1v - old
                d2 = m2v - old
                nc1 = corr + d1 * eff
                nc1[center] = corr[center] + d1 * c_eff - d1 * d1
                f1 = float(np.max(nc1)) * dx
                nc2 = corr + d2 * eff
                nc2[center] = corr[center] + d2 * c_eff - d2 * d2
                f2 = float(np.max(nc2)) * dx
                if f1 < f2:
                    b = m2v
                else:
                    a = m1v
            best_t = (a + b) / 2.0
            delta = best_t - old
            if abs(delta) > max_d:
                max_d = abs(delta)
            h[i] = best_t
            corr += delta * eff
            corr[center] += delta * c_eff - delta * delta
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        if max_d < 1e-12:
            break
    return h
