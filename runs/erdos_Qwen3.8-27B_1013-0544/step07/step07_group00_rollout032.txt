    p /= p.sum()
    
    # Get top-k significant shifts

            for j in range(0, upper):

    p /= p.sum()
    top_idx = np.argsort(p)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        w = p[k]
        if w < 1e-12:
            break
        d = k - (n - 1)
        if d >= 0:
            i1 = np.arange(0, n - d)
            grad[i1] += w * (1.0 - h[i1 + d])
            i2 = np.arange(d, n)
            grad[i2] -= w * h[i2 - d]
        else:
            dp = -d
            i1 = np.arange(dp, n)
            grad[i1] += w * (1.0 - h[i1 - dp])
            i2 = np.arange(0, n - dp)
            grad[i2] -= w * h[i2 + dp]
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def coord_descent(h, n, max_passes=300, deadline=None, n_gs=50, random_order=False, rng=None):
    h = h.copy()
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        prefix_max = np.maximum.accumulate(c)
        suffix_max = np.maximum.accumulate(c[::-1])[::-1]
        order = rng.permutation(n) if (random_order and rng is not None) else range(n)
        improved = False
        for p in order:
            if deadline is not None and time.time() > deadline - 0.01:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            k_lo = min(p, n - 1 - p)
            k_hi = max(p, n - 1 - p) + (n - 1)
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p, n - 1 - p) + (n - 1))
            # Recompute properly
            d_lo = min(-p, p - (n - 1))
            d_hi = max(n - 1 - p, p)
            k_lo = max(0, n - 1 + d_lo)
            k_hi = min(2 * n - 2, n - 1 + d_hi)
            unaff_parts = []
            if k_lo > 0:
                unaff_parts.append(float(prefix_max[k_lo - 1]))
            if k_hi < 2 * n - 2:
                unaff_parts.append(float(suffix_max[k_hi + 1]))
            unaff_max = max(unaff_parts) if unaff_parts else 0.0
            ks = np.arange(k_lo, k_hi + 1)
            ds = ks - (n - 1)
            a = np.zeros(len(ds))
            vf = (p + ds >= 0) & (p + ds < n)
            vb = (p - ds >= 0) & (p - ds < n)
            a[vf] += (1.0 - h[p + ds[vf]])
            a[vb] -= h[p - ds[vb]]
            c_aff = c[ks]
            diag = (ds == 0).astype(float)
            lo_b = -h0
            hi_b = 1.0 - h0
            if hi_b - lo_b < 1e-14:
                continue
            x1 = lo_b + invphi * (hi_b - lo_b)
            x2 = hi_b - invphi * (hi_b - lo_b)
            def fv(dd):
                return max(unaff_max, float((c_aff + dd * a - dd * dd * diag).max()))
            f1, f2 = fv(x1), fv(x2)
            for _ in range(n_gs):
                if f1 < f2:
                    hi_b, x2, f2 = x2, x1, f1
                    x1 = lo_b + invphi * (hi_b - lo_b)
                    f1 = fv(x1)
                else:
                    lo_b, x1, f1 = x1, x2, f2
                    x2 = hi_b - invphi * (hi_b - lo_b)
                    f2 = fv(x2)
            best_d = (lo_b + hi_b) / 2.0
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alpha, maxiter=200):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        return (m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha) * (2.0 / n)
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with alpha annealing
    for alpha in [500, 5000, 50000]:
        if time.time() > deadline - 500:
            break
        h_s = slsqp_opt(best_h, n, alpha, maxiter=150)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # Phase 2: Subgradient descent
    h_sg = best_h.copy()
    for alpha in [200, 1000, 5000]:
        if time.time() > deadline - 350:
            break
        for it in range(400):
            if time.time() > deadline - 350:
                break
            step = 0.06 / (1.0 + 0.008 * it)
            h_sg = subgradient_step(h_sg, n, alpha, step, top_k=30)
            if it % 100 == 99:
                c_cur = compute_c5(h_sg, n)
                if c_cur < best_c:
                    best_h, best_c = h_sg.copy(), c_cur

    # Phase 3: Main coordinate descent with golden section
    h_cd, c_cd = coord_descent(best_h, n, max_passes=500, deadline=deadline - 200, n_gs=50)
    if c_cd < best_c:
        best_h, best_c = h_cd, c_cd

    # Phase 4: Random-order coordinate descent
    h_ro, c_ro = coord_descent(best_h, n, max_passes=200, deadline=deadline - 100,
                               n_gs=50, random_order=True, rng=rng)
    if c_ro < best_c:
        best_h, best_c = h_ro, c_ro

    # Phase 5: Perturbation + coord descent
    for trial in range(6):
        if time.time() > deadline - 30:
            break
        sigma = 0.01 + 0.03 * trial
        hp = project_feasible(best_h + rng.normal(0, sigma, n), n)
        h_o, c_o = coord_descent(hp, n, max_passes=100, deadline=deadline - 20, n_gs=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 6: Final high-precision coordinate descent
    if time.time() < deadline - 5:
        h_f, c_f = coord_descent(best_h, n, max_passes=300, deadline=deadline - 1, n_gs=80)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
