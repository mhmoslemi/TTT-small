```python
    h_R = h.copy()
    # Build constraint matrix: for each k, c_k[i] = h_R[i+k] if valid
    # k ranges from -(n-1) to (n-1), index m from 0 to 2n-2
    # For shift k = m - (n-1):
    #   c_k[i] = h_R[i+k] if 0 <= i+k < n, else 0
    #   valid i: max(0, -k) <= i < min(n, n-k)
    
    n_vars = n + 1  # h_L (n) + t (1)

    # (2n-1) inequalities: c_k^T h_L - t >= 0
    # Rearranged: -c_k^T h_L + t <= 0
    
    A_ineq = np.zeros((2*n-1, n+1))
    # To save memory and time, build row by row
    for m in range(2*n-1):

        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end > i_start:

        A_ineq[m, n] = 1.0  # coefficient of t

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])

    h_L = cp.Variable(n)
    t = cp.Variable()

        A_ineq @ cp.concatenate([h_L, t]) <= 0,
        A_eq @ cp.concatenate([h_L, t]) == b_eq[0],
        h_L >= 0,
        h_L <= 1,
        t >= 0

    prob = cp.Problem(cp.Maximize(t), constraints)
    
    solved = False
    for sol in [solver, 'CLARABEL', 'ECOS', 'SCS']:

            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue

        return h, None
    
    h_new_L = np.clip(h_L.value, 0.0, 1.0)
    h_new_L = project_feasible(h_new_L, n)
    
    # Blend
    h_new = alpha * h_new_L + (1 - alpha) * h_R
    h_new = project_feasible(h_new, n)
    
    t_val = float(t.value) if t.value is not None else None
```

```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def alt_lp_step(h, n, alpha=1.0, solver='CLARABEL'):
    h_R = h.copy()
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n + 1))
    for m in range(n_rows):
        k = m - (n - 1)
        i_s = max(0, -k)
        i_e = min(n, n - k)
        if i_e > i_s:
            A[m, i_s:i_e] = -h_R[i_s + k: i_e + k]
        A[m, n] = 1.0
    
    h_L = cp.Variable(n)
    t = cp.Variable()
    var = cp.hstack([h_L, t])
    cons = [
        A @ var <= 0,
        h_L.sum() == n / 2.0,
        h_L >= 0,
        h_L <= 1,
        t >= 0
    ]
    prob = cp.Problem(cp.Maximize(t), cons)
    solved = False
    for s in [solver, 'CLARABEL', 'ECOS', 'SLS'] if solver != 'SLS' else [solver, 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    if not solved or h_L.value is None:
        return h, -1.0
    h_new = np.clip(h_L.value, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    blended = alpha * h_new + (1.0 - alpha) * h
    blended = project_feasible(blended, n)
    t_val = float(t.value) if t.value is not None else -1.0
    return blended, t_val

def slsqp_refine(h, n, time_limit):
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h, n)
    alphas = [1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > time_limit:
            break
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            logsum = sm + np.log(np.exp(shifted - sm).sum())
            return m + logsum / alpha
        def cons_fun(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h_try = project_feasible(res.x, n)
            val_try = compute_c5(h_try, n)
            if val_try < best_val:
                best_val = val_try
                best_h = h_try
        except Exception:
            pass
    return best_h, best_val
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def alt_lp_step(h, n, alpha=1.0):
    """Alternating LP: fix right, optimize left."""
    h_R = h
    n_rows = 2 * n - 1
    # Build A matrix: row m corresponds to shift k=m-(n-1)
    # A[m, i] = -h_R[i+k] if valid, else 0
    # A[m, n] = 1 (for t)
    A = np.zeros((n_rows, n + 1))
    for m in range(n_rows):
        k = m - (n - 1)
        i_s = max(0, -k)
        i_e = min(n, n - k)
        if i_e > i_s:
            A[m, i_s:i_e] = -h_R[i_s + k: i_e + k]
        A[m, n] = 1.0
    
    h_L = cp.Variable(n)
    t = cp.Variable()
    var = cp.hstack([h_L, t])
    cons = [A @ var <= 0, h_L.sum() == n / 2.0, h_L >= 0, h_L <= 1, t >= 0]
    prob = cp.Problem(cp.Maximize(t), cons)
    
    for solver in ['CLARABEL', 'ECOS', 'SLS', 'CVXOPT']:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_L.value is not None:
                break
        except Exception:
            continue
    
    if h_L.value is None:
        return h, -1.0
    
    h_new = np.clip(h_L.value, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    blended = alpha * h_new + (1.0 - alpha) * h_R
    blended = project_feasible(blended, n)
    t_val = float(t.value) if t.value is not None else -1.0
    return blended, t_val

def slsqp_refine(h, n, max_time):
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h, n)
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > max_time:
            break
        def obj(x):
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        def cons_eq(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 1000, 'ftol': 1e-15})
            h_try = project_feasible(res.x, n)
            v = compute_c5(h_try, n)
            if v < best_val:
                best_val = v
                best_h = h_try.copy()
        except Exception:
            pass
    return best_h, best_val
```

```python
            best_h = h_fin

    # Phase 5: Final small-step LP polish
    if time.time() < deadline - 3:
        h = best_h.copy()
        eps_f = 0.01
        for _ in range(300):
            if time.time() > deadline - 2:
                break
            h_new, mc_new = lp_step_small(h, n, eps_f)
            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            eps_f = max(eps_f * 0.997, 0.001)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
