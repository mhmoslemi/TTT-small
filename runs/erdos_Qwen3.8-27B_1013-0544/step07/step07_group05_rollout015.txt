import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = target - y.sum()
    return y

def _softmax(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def _smooth_max_topk(c, beta, topk):
    if topk > 0 and topk < len(c):
        idx = np.argsort(c)[-topk:]
        f_sub, w_sub = _softmax(c[idx], beta)
        w = np.zeros_like(c)
        w[idx] = w_sub
        return f_sub, w
    return _softmax(c, beta)

class _Optimizer:
    def __init__(self):
        self.n = 1
        self.dx = 1.0
        self.target = 0.5
        self.beta = 1.0
        self.topk = 0
        self.rho = 1e4
        self._cx = None
        self._cg = None

    def compute(self, x):
        n = self.n
        dx = self.dx
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        f, w = _smooth_max_topk(c, self.beta, self.topk)
        v = 1.0 - 2.0 * x
        g = np.correlate(w, v, mode='valid') * dx
        res = float(x.sum() - self.target)
        f = f + self.rho * res * res
        g = g + 2.0 * self.rho * res
        self._cx = x.copy()
        self._cg = g.copy()
        return float(f), g

    def obj(self, x):
        f, _ = self.compute(x)
        return f

    def jac(self, x):
        cx = self._cx
        if cx is not None and np.array_equal(x, cx):
            return self._cg.copy()
        _, g = self.compute(x)
        return g

_opt = _Optimizer()

def _lbfgs(h, beta, topk, maxiter, deadline):
    n = len(h)
   _rem = deadline - time.time()
    if _rem < 5.0:
        return h, erdos_c5(h, n)
    _opt.n = n
    _opt.dx = 2.0 / n
    _opt.target = 0.5 * n
    _opt.beta = beta
    _opt.topk = topk
    _opt.rho = 1e4
    _opt._cx = None
    _opt._cg = None
    mx = int(min(maxiter, max(10, _rem / 4.0)))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(_opt.obj, h, jac=_opt.jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': mx, 'ftol': 1e-18,
                                                   'gtol': 1e-14, 'maxls': 30})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def _pair_polish(h, n, deadline, seed, max_pairs=300, rounds=4):
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    c_full = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_shifts = np.argsort(c_full)[-8:][::-1]
    frac = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if frac.size == 0:
        return h, cur
    scores = np.zeros(len(frac))
    for k in top_shifts[:5]:
        lag = int(k) - (n - 1)
        for idx_i in range(len(frac)):
            i = int(frac[idx_i])
            if 0 <= i + lag < n:
                scores[idx_i] += min(h[i], 1.0 - h[i + lag])
            if 0 <= i - lag < n:
                scores[idx_i] += min(h[i], 1.0 - h[i - lag])
    order = frac[np.argsort(scores)[::-1]]
    m = min(35, len(order))
    top_pts = order[:m]
    if len(top_pts) < 2:
        return h, cur
    rng = np.random.RandomState(seed)
    for rd in range(rounds):
        if time.time() > deadline - 15.0:
            break
        improved = False
        pairs = [(int(top_pts[a]), int(top_pts[b]))
                 for a in range(len(top_pts)) for b in range(a + 1, len(top_pts))]
        rng.shuffle(pairs)
        for i, j in pairs[:max_pairs]:
            if time.time() > deadline - 12.0:
                return h, erdos_c5(h, n)
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-12:
                continue
            def _pn(delta):
                a_val = hi0 + delta
                b_val = hj0 - delta
                if a_val < -1e-10 or a_val > 1.0 + 1e-10 or b_val < -1e-10 or b_val > 1.0 + 1e-10:
                    return 1e10
                hh = h.copy()
                hh[i] = a_val
                hh[j] = b_val
                return erdos_c5(hh, n)
            try:
                rs = so.minimize_scalar(_pn, bounds=(lo, hi), method='bounded',
                                        options={'maxiter': 20, 'xatol': 1e-10})
                if rs.fun < cur - 1e-14:
                    h[i] += rs.x
                    h[j] -= rs.x
                    cur = rs.fun
                    improved = True
            except Exception:
                pass
        h = project_box_sum(h, 0.5 * n)
        cur = erdos_c5(h, n)
        if not improved:
            break
    return h, cur

def _resample(h, n, nh):
    hh = np.interp(np.linspace(0.0, 2.0, nh), np.linspace(0.0, 2.0, n), h)
    return project_box_sum(hh, 0.5 * nh)

    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)

    best_h = h.copy()
    best_c = erdos_c5(best_h, n)

    # Phase 1: L-BFGS with increasing beta (full softmax)
    phase1_end = t0 + 0.35 * total
    schedule1 = [(50, 0), (200, 0), (800, 0), (3000, 0), (10000, 0),

        cand, c = _lbfgs(best_h, beta, topk, 80, deadline)

    phase2_end = t0 + 0.58 * total
    schedule2 = [(20000, 3), (50000, 3), (100000, 5), (300000, 5),

    for beta, topk in schedule2:
        if time.time() > phase2_end - 5.0:

            best_h = cand.copy()

    # Phase 3: Pair polish with shift awareness
    phase3_end = t0 + 0.78 * total

    return best_h, best_c, int(n)
