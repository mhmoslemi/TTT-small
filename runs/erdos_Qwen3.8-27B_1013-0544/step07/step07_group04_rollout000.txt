import numpy as np
from scipy.optimize import minimize

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).ravel())
    if v.size < n:
        w = np.full(n, 0.5)
        w[:v.size] = v
        v = w
    else:
        v = v[:n]
    t = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if float(np.clip(v - m, 0.0, 1.0).sum()) > t:
            lo = m
        else:
            hi = m
    w = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - float(w.sum())
    for _ in range(30):
        if abs(d) < 1e-15:

def fft_conv(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    total = a.size + b.size - 1
    if a.size * b.size < 3000:
        return np.convolve(a, b, "full")
    size = 1
    while size < total:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total]

def lse_weights(c, alpha):
    m = float(c.max())
    if alpha <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    return e / s

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0.0:
        return m
    z = np.clip(alpha * (c - m), -745.0, 0.0)
    return m + float(np.log(float(np.exp(z).sum()))) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    off = n - 1
    a = fft_conv(1.0 - x, q)[off:off + n]
    b = fft_conv(x, q[::-1])[off:off + n]
    return (2.0 / n) * (a - b)

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def pen_obj(x, n, alpha, mu):
    return lse_obj(x, n, alpha) + 0.5 * mu * float(sum_fun(x, n)) ** 2

def pen_grad(x, n, alpha, mu):
    return lse_grad(x, n, alpha) + mu * sum_fun(x, n)

def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=[(0.0, 1.0)] * n,
                       options={"maxiter": maxiter, "maxfun": max(80, 3 * maxiter),
                                "ftol": 1e-15, "gtol": 1e-8})
        if getattr(res, "x", None) is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h

def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP", jac=lse_grad,
                       bounds=[(0.0, 1.0)] * n,

                                     "jac": sum_jac}],
                       options={"maxiter": maxiter, "ftol": 1e-12})
        if getattr(res, "x", None) is not None:

    except Exception:
        pass
    return h

    return np.interp(xn, xo, h)

def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 16)))

    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(np.unique(xs), lo, hi)

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhr = omh[::-1]
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta = np.zeros((xs.size, L), dtype=float)
    delta[:, i:i + n] += ax1[:, None] * omhr[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhr[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, off] += exact_self - base_self
    d = j - i
    ip = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, ip] += exact_p - base_p
    im = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, im] += exact_m - base_m
    return np.max(c_base + dx * delta, axis=1)

def pair_search(h, c_base, i, j, n):
    if i == j:
        return None
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)

        return None

    for _ in range(2):
        if xs.size < 3:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == xs.size - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-11:
            break
        xs2 = np.linspace(a, b, 88)
        vals = pair_eval(h, c_base, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = S - xb
        if yb < 0.0:
            yb = 0.0
            xb = S - yb
        elif yb > 1.0:
            yb = 1.0
            xb = S - yb
        return xb, yb, bv
    return None

                bestn = nh
        if time.time() < deadline - 12.0:
            besth, bestc = pair_local_search(besth, deadline - 3.0, seed + 999)
    except Exception:
        pass
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
