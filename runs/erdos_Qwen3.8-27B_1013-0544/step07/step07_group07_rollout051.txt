from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_lp(h_fixed, n, top_k=None):
    dx = 2.0 / n
    om = (1.0 - h_fixed) * dx
    nc = 2 * n - 1
    A = np.zeros((nc, n))
    for j in range(n - 1, 2 * n - 1):
        k = j - (n - 1)
        A[j, k:n] = om[:n - k]
    for j in range(0, n - 1):
        m = (n - 1) - j
        A[j, :n - m] = om[m:]
    c_vals = np.max(A @ h_fixed, axis=1)
    if top_k is not None and top_k < nc:
        idx = np.argsort(c_vals)[::-1][:top_k]
    else:
        idx = np.arange(nc)
    A_sub = A[idx]
    A_ub = np.hstack([A_sub, -np.ones((idx.size, 1))])
    b_ub = np.zeros(idx.size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    return A_ub, b_ub, A_eq, b_eq, c_obj, bounds

def lp_step(h_fixed, n, top_k=None):

                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:n], 0, 1)
    except Exception:
        pass
    return h_fixed.copy()

def lp_alternate(h_init, n, deadline, max_iters=10000, top_k=None, damp=1.0):

    best_h = h.copy()

            h_new = damp * h_new + (1 - damp) * h

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if abs(prev_c - c_new) < 1e-14:
            break
        prev_c = c_new
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pn_slsqp(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c_pos = np.maximum(c, 1e-12)
        return (np.sum(c_pos ** p)) ** (1.0 / p)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
