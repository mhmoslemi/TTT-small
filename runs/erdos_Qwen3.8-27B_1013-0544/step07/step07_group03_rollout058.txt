```python
    """Fast binary local search with vectorized swap evaluation."""
    dx = 2.0 / n
    c = np.correlate(h_b.astype(np.float64), 1.0 - h_b, mode='full')
    cur_max = float(c.max())
    best_c = cur_max
    best_h = h_b.copy()

    zeros = np.where(h_b == 1)[0]  # will fix
    zeros = np.where(h_b == 0)[0]

    no_improve = 0

            break
        sweeps += 1
        improved = False

        grad = compute_grad_matrix(h_b.astype(np.float64), n)
        grad_ones = grad[ones]  # shape (n//2, 2n-1)

                return best_h, best_c
            # For all possible j in ones: dc = grad[pi] - grad[j]
            dc = grad[pi][None, :] - grad_ones  # shape (n//2, 2n-1)

            new_maxs = new_c.max(axis=1)
            best_j_idx = new_maxs.argmin()
            best_new_max = new_maxs[best_j_idx]

                zeros.sort()

                grad_ones = grad[ones]
                
                improved = True
                no_improve = 0
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h_b.copy()
        
        if not improved:
```

```python
def binary_local_search_v2(h_b, n, deadline, max_sweeps=10000):
    """Fast binary local search with incremental gradient updates."""
    dx = 2.0 / n
    h = h_b.astype(np.float64).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_c = cur_max
    best_h = h_b.copy()
    
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    
    # Initial gradient
    grad = compute_grad_matrix(h, n)
    
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        for pi in range(len(zeros)):
            if time.time() > deadline + 2:  # small overshoot
                return best_h, best_c
            
            p = zeros[pi]
            # Evaluate all possible j in ones
            # dc_j = grad[p] - grad[j] for each j
            # new_c_j = c + dc_j
            # new_max_j = max(new_c_j)
            
            # Vectorized: dc_matrix[j_idx, k] = grad[p, k] - grad[ones[j_idx], k]
            grad_p = grad[p]  # shape (2n-1,)
            grad_ones_arr = np.array([grad[j2] for j2 in ones])  # shape (len(ones), 2n-1)
            dc_mat = grad_p[None, :] - grad_ones_arr  # shape (len(ones), 2n-1)
            new_c_mat = c[None, :] + dc_mat
            new_maxs = new_c_mat.max(axis=1)
            
            best_idx = int(new_maxs.argmin())
            best_new_max = float(new_maxs[best_idx])
            
            if best_new_max < cur_max - 1e-12:
                j = ones[best_idx]
                # Apply swap
                c = c + grad_p - grad[j]
                cur_max = float(c.max())
                
                # Update gradient incrementally
                # For all k: kp = k - (n-1)
                # Update:
                #   grad[pi+kp, k] -= 1 (if valid)
                #   grad[j+kp, k] += 1 (if valid)  
                #   grad[pi-kp, k] -= 1 (if valid)
                #   grad[j-kp, k] += 1 (if valid)
                
                kps = np.arange(-(n-1), n)  # all possible kp values
                # For each kp, the affected (p,k) pairs:
                # k = kp + (n-1)
                # p1 = pi + kp, p2 = j + kp, p3 = pi - kp, p4 = j - kp
                
                for kp in kps:
                    k = kp + (n-1)
                    p1 = pi + kp
                    if 0 <= p1 < n:
                        grad[p1, k] -= 1.0
                    p2 = j + kp
                    if 0 <= p2 < n:
                        grad[p2, k] += 1.0
                    p3 = pi - kp
                    if 0 <= p3 < n:
                        grad[p3, k] -= 1.0
                    p4 = j - kp
                    if 0 <= p4 < n:
                        grad[p4, k] += 1.0
                
                # Update h
                h[pi] = 1.0
                h[j] = 0.0
                
                # Update lists
                ones.remove(j)
                ones.append(pi)
                zeros.remove(pi)
                zeros.append(j)
                
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h_b.astype(np.float64).copy()
                    best_h[pi] = 1.0
                    best_h[j] = 0.0
        
        if not improved:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    
    return best_h.astype(np.float64), best_c
```

```python
def precompute_update_indices(n):
    """For each position p, precompute the (row, col, sign) entries for gradient update."""
    updates = {}
    for p in range(n):
        entries = []
        # h[p] goes 0→1: grad[p+kp, kp+(n-1)] -= 1, grad[p-kp, kp+(n-1)] -= 1
        # valid kp for p+kp: 0<=p+kp<n => kp in [-p, n-1-p], intersect [-(n-1), n-1]
        kp_lo = max(-(n-1), -p)
        kp_hi = min(n-1, n-1-p)
        for kp in range(kp_lo, kp_hi+1):
            k = kp + (n-1)
            pp = p + kp
            if 0 <= pp < n:
                entries.append((pp, k, -1.0))
        # valid kp for p-kp: 0<=p-kp<n => kp in [p-(n-1), p], intersect [-(n-1), n-1]
        kp_lo2 = max(-(n-1), p-(n-1))
        kp_hi2 = min(n-1, p)
        for kp in range(kp_lo2, kp_hi2+1):
            k = kp + (n-1)
            pp = p - kp
            if 0 <= pp < n:
                entries.append((pp, k, -1.0))
        updates[p] = entries  # for flipping 0→1
        # For flipping 1→0, just negate the signs
    return updates
```

            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    grad = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        kp = k - (n - 1)
        lo1 = max(0, kp)
        hi1 = min(n - 1, n - 1 + kp)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1, k] = 1.0 - h[lo1 - kp:hi1 - kp + 1]
        lo2 = max(0, -kp)
        hi2 = min(n - 1, n - 1 - kp)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1, k] -= h[lo2 + kp:hi2 + kp + 1]
    return grad

def precompute_update(n):
    rows_list = []
    cols_list = []
    for p in range(n):
        r, c = [], []
        # Term 1: p+kp valid
        for kp in range(max(-(n-1), -p), min(n-1, n-1-p) + 1):
            k = kp + (n - 1)
            pp = p + kp
            if 0 <= pp < n:
                r.append(pp)
                c.append(k)
        # Term 2: p-kp valid
        for kp in range(max(-(n-1), p - (n-1)), min(n-1, p) + 1):
            k = kp + (n - 1)
            pp = p - kp
            if 0 <= pp < n:
                r.append(pp)
                c.append(k)
        rows_list.append(np.array(r, dtype=np.intp))
        cols_list.append(np.array(c, dtype=np.intp))
    return rows_list, cols_list

def bin_local_search(h_b, n, rows_list, cols_list, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = h_b.astype(np.float64).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(c.max())
    best_c = cur_max
    best_h = h.copy()
    
    ones = set(np.where(h_b == 1)[0].tolist())
    zeros = set(np.where(h_b == 0)[0].tolist())
    
    grad = compute_grad_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zero_list = list(zeros)
        
        for pi in zero_list:
            if time.time() > deadline:
                return best_h, best_c
            if pi not in zeros:
                continue
            
            # Find best j in ones
            one_list = list(ones)
            best_j = -1
            best_nm = cur_max
            
            for j in one_list:
                new_c_max = float((c + grad[pi] - grad[j]).max())
                if new_c_max < best_nm:
                    best_nm = new_c_max
                    best_j = j
            
            if best_j >= 0 and best_nm < cur_max - 1e-12:
                # Apply swap
                c = c + grad[pi] - grad[best_j]
                cur_max = best_nm
                
                # Update gradient
                grad[rows_list[pi], cols_list[pi]] -= 1.0
                grad[rows_list[best_j], cols_list[best_j]] += 1.0
                
                h[pi] = 1.0
                h[best_j] = 0.0
                ones.discard(best_j)
                ones.add(pi)
                zeros.discard(pi)
                zeros.add(best_j)
                
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
        
        if
