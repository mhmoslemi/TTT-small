def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()

            s = room.sum()

            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def get_active(h, n, frac=0.05):
    c = np.correlate(h, 1 - h, mode='full')
    mc = c.max()
    if mc < 1e-10:
        return np.arange(len(c))
    thresh = mc * (1.0 - frac)
    return np.where(c >= thresh)[0]

def lp_step(h, n, eps, act_frac=0.05):

    mc = float(c.max())
    act = get_active(h, n, act_frac)
    G = np.zeros((n, len(act)))
    for ii, m in enumerate(act):
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, ii] = 1.0 - h[j1s + k:j1e + k]
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, ii] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [
        c[act] + G.T @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,

    prob = cp.Problem(cp.Minimize(t), cons)

    return best_h, best_mc

        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
