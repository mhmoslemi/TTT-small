This is a "convolution" of pos_weights with h.

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p /= p.sum()
    
    grad = np.zeros(n)
    for j in range(2*n-1):
        k = j - (n-1)
        w = p[j] * dx
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1 - h)
        elif k > 0:
            # (1-h[m+k]) for m = 0..n-1-k
            grad[:n-k] += w * (1 - h[k:])
            # -h[m-k] for m = k..n-1
            grad[k:] -= w * h[:n-k]
        else:
            ll = -k
            # (1-h[m-ll]) for m = ll..n-1
            grad[ll:] += w * (1 - h[:n-ll])
            # -h[m+ll] for m = 0..n-1-ll
            grad[:n-ll] -= w * h[ll:]
    
    return grad

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, n, alpha, maxiter=500, deadline=None):
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        p = np.exp(shifted - sm)
        p /= p.sum()
        
        grad = np.zeros(n)
        for j in range(2 * n - 1):
            k = j - (n - 1)
            w = p[j] * dx
            if w < 1e-15:
                continue
            if k == 0:
                grad += w * (1.0 - x)
            elif k > 0:
                grad[:n-k] += w * (1.0 - x[k:])
                grad[k:] -= w * x[:n-k]
            else:
                ll = -k
                grad[ll:] += w * (1.0 - x[:n-ll])
                grad[:n-ll] -= w * x[ll:]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,

    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                               'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_nesterov(h, n, deadline, mu=0.9):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    y = h.copy()
    v = np.zeros(n)
    t = 1.0
    lr_base = dx * 3.0
    
    for step in range(500000):
        if time.time() > deadline:
            break
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (1 - t) / t_new
        y = h + beta * (h - y)
        t = t_new
        
        # Compute gradient at y
        corr = np.correlate(y, 1-y, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if lag < n:
                grad[:n-lag] += (1.0 - y[lag:])
                grad[lag:] -= y[:n-lag]
        else:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - y[:n-ll])
                grad[:n-ll] -= y[ll:]
        grad *= dx
        
        lr = lr_base / np.sqrt(t)
        v = mu * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def multi_resolution(h, n, deadline):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _smoothed_max_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    obj = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    s = x.sum() - n / 2.0
    return obj + penalty * s * s

def _smoothed_max_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p /= p.sum()
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)

            grad += w * (1.0 - x)
        elif k > 0:

            grad[k:] -= w * x[:n-k]
        else:
            ll = -k
            grad[ll:] += w * (1.0 - x[:n-ll])
            grad[:n-ll] -= w * x[ll:]
    
    s = x.sum() - n / 2.0
    grad += 2.0 * penalty * s
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(400):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _obj_fn(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    obj = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    s = x.sum() - n / 2.0
    return obj + penalty * s * s

def _grad_fn(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p /= p.sum()
    mask = p > 1e-12
    if not mask.any():
        p = np.zeros(2 * n - 1)
        p[np.argmax(corr)] = 1.0
        mask = p > 0
    grad = np.zeros(n)
    for j in np.where(mask)[0]:
        k = j - (n - 1)
        w = p[j] * dx
        if k == 0:
            grad += w * (1.0 - x)
        elif k > 0:
            grad[:n - k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[:n - k]
        else:
            ll = -k
            grad[ll:] += w * (1.0 - x[:n - ll])
            grad[:n - ll] -= w * x[ll:]
    s = x.sum() - n / 2.0
    grad += 2.0 * penalty * s
    return grad

def lbfgs_optimize(h, n, alpha, maxiter=500, penalty=1e6):
    bounds = [(0.0, 1.0)] * n
    args = (n, alpha, penalty)
    try:
        res = minimize(_obj_fn, h.copy(), args=args, method='L-BFGS-B',
                       jac=_grad_fn, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

    y = h.copy()

    t = 1.0
    lr_base = dx * 4.0
    stall = 0
    for step in range(2000000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - t) / t_new
        y = h + beta * (h - y)
        t = t_new
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n - lag] += (1.0 - y[lag:])
            grad[lag:] -= y[:n - lag]
        elif lag < 0 and -lag < n:
            ll = -lag
            grad[ll:] += (1.0 - y[:n - ll])
            grad[:n - ll] -= y[ll:]
        grad *= dx
        lr = lr_base / np.sqrt(t)
        v = mu * v - lr * grad
        h_new = project_feasible(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-12:

                    stall = 0
        h = h_new
    return best_h, best_c

def multi_resolution(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    xo = np.linspace(0, 2, n)
    for nl in [n // 2, n // 3, n // 4]:
