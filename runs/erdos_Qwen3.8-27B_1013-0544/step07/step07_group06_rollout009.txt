def lse_og(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1)); F = m
    else:
        p = e / s; F = m + np.log(s) / beta
    P = p; R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.all(np.isfinite(g)): g = np.zeros_like(h)
    return float(F), g

def binary_greedy(h, n, deadline, max_sweeps=50, chunk=2048):
    if n % 2 != 0: return h, compute_c5(h, n)
    h = top_n_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy(); bestM = M
    
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        l = np.arange(-(n - 1), n, dtype=int)
        base = np.zeros((n, 2 * n - 1), dtype=float)
        for p in range(n):
            f = p + l; mf = (f >= 0) & (f < n)
            base[p, mf] = 1.0 - h[f[mf]]
            b = p - l; mb = (b >= 0) & (b < n)
            base[p, mb] -= h[b[mb]]
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0: break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start + chunk, zi.size)
            i = zi[start:end]; j = oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            d2 = n - 1 + (i - j)
            arr[:, d1] += 1.0
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv = int(i[loc]); jv = int(j[loc])
                h[iv] = 1.0; h[jv] = 0.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(float(corr.max())))
                if M < bestM: bestM = M; besth = h.copy()
                improved = True
                break
        if not improved: break
    return besth, bestM * (2.0 / float(n))

    return h

            base[p, mb] -= h[b[mb]]
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0: break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline: break
            end = min(start + chunk, zi.size)
            i = zi[start:end]; j = oj[start:end]
            arr = base[i] - base[j] + corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = (n - 1 + (j - i)).astype(int)
            d2 = (n - 1 + (i - j)).astype(int)
            arr[:, d1] += 1.0
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv, jv = int(i[loc]), int(j[loc])
                h[iv] = 1.0; h[jv] = 0.0
                corr = arr[loc].copy()
                M = int(round(float(corr.max())))
                if M < bestM: bestM = M; besth = h.copy()
                improved = True
                break
        if not improved: break
    return besth, bestM * (2.0 / float(n))

def subgradient_polish(h, n, deadline, max_iter=5000):
    h = project_feasible(h.copy(), n)
    best = h.copy(); bc = compute_c5(best, n)
    dx = 2.0 / float(n)
    for it in range(max_iter):
        if time.time() > deadline: break
        v = 1.0 - h
        c = np.correlate(h, v, mode="full")
        kstar = int(np.argmax(c))
        # subgradient of c[kstar] w.r.t. h
        sg = np.zeros(n)
        for i in range(n):
            fi = i + kstar - (n - 1)
            bi = i - kstar + (n - 1)
            term = 0.0
            if 0 <= fi < n: term += (1.0 - h[fi])
            if 0 <= bi < n: term -= h[bi]
            sg[i] = dx * term
        # step
        eta = 0.01 * dx / (1.0 + it * 0.001)
        h_new = project_feasible(h - eta * sg, n)
        cn = compute_c5(h_new, n)
        if cn < bc - 1e-14:
            h = h_new; best = h.copy(); bc = cn
        elif cn <= bc + 1e-12:
            h = h_new
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + max(30.0, float(budget_s) - 12.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)): raise ValueError()
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // 50) % 2).astype(float)

    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    # Phase 1: L-BFGS with fine beta schedule
    betas1 = [5, 15, 40, 100, 250, 600, 1500, 4000, 10000, 25000, 60000, 150000]
    if time.time() < end - 180:
        hh, cc = polish(best_h, best_n, betas1, 1e4, end - 220, 55)
        if cc < best_c: best_c = cc; best_h = hh.copy()

    # Phase 2: Binary greedy
    if best_n % 2 == 0 and time.time() < end - 100:
        hb, cr = binary_greedy(best_h, best_n, end - 130, max_sweeps=60, chunk=2048)
        if cr < best_c: best_c = cr; best_h = hb.copy()

    # Phase 3: Try nearby n values
    for nnew in [396, 404, 420, 380]:
        if nnew % 2 != 0 or time.time() > end - 80: continue
        x0 = np.linspace(0, 2, best_n)
        x1 = np.linspace(0, 2, nnew)
        hr = project_feasible(np.interp(x1, x0, best_h), nnew)
        cr = compute_c5(hr, nnew)
        hh, cc = polish(hr, nnew, [50, 200, 800, 3000, 10000, 40000], 1e4, end - 90, 20)
        if cc < cr: cr = cc; hr = hh.copy()
        if nnew % 2 == 0 and time.time() < end - 35:
            hb2, cr2 = binary_greedy(hr, nnew, end - 35, max_sweeps=25, chunk=2048)
            if cr2 < cr: cr = cr2; hr = hb2.copy()
        if cr < best_c:
            best_c = cr; best_h = hr.copy(); best_n = nnew

    # Phase 4: Subgradient polish
    if time.time() < end - 30:
        hh, cc = subgradient_polish(best_h, best_n, end - 15, max_iter=3000)
        if cc < best_c: best_c = cc; best_h = hh.copy()

    # Phase 5: Random perturb + re-optimize
    if time.time() < end - 50:
        for scale in [0.005, 0.02]:
            if time.time() > end - 40: break
            nh = project_feasible(best_h + scale * rng.randn(best_n), best_n)
            hh, cc = polish(nh, best_n, [5000, 30000, 100000], 1e4, end - 30, 15)
            if cc < best_c: best_c = cc; best_h = hh.copy()
            if best_n % 2 == 0 and time.time() < end - 15:
                hb, cr = binary_greedy(best_h, best_n, end - 10, max_sweeps=30, chunk=2048)
                if cr < best_c: best_c = cr; best_h = hb.copy()

    # Final: binary projection
    if best_n % 2 == 0:
        tb = top_n_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c: best_c = ct; best_h = tb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
