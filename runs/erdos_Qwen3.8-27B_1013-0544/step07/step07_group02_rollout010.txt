def topk_p_fun(x, n, K, p):
    h = np.clip(x, 1e-10, 1.0 - 1e-10)
    C = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(C)[-K:]
    top = C[idx]
    penalty = 100.0 * (h.sum() - n / 2.0) ** 2
    return np.mean(top ** p) + penalty

def topk_p_grad(x, n, K, p):
    h = np.clip(x, 1e-10, 1.0 - 1e-10)
    C = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(C)[-K:]
    top = C[idx]
    G = build_gradient(h, n)
    weights = top ** (p - 1)
    grad = (p / K) * (G[:, idx].T @ weights)
    dp = 200.0 * (h.sum() - n / 2.0)
    grad += dp
    return grad

def cons_eq(x):
    return x.sum() - x.size / 2.0

def cons_eq_jac(x):
    return np.ones(x.size)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_mc = float(np.max(np.correlate(h, 1 - h, mode='full')))
    best_h = h.copy()
    best_c5 = best_mc * 2.0 / n

    # Phase 1: LP with adaptive eps (28% budget)
    p1_dl = t0 + 0.28 * total
    eps = 0.05
    no_imp = 0
    while time.time() < p1_dl:
        h, mc = lp_step(h, n, eps)
        if mc < best_mc - 1e-12:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 6:
                eps *= 0.6
                no_imp = 0
                if eps < 0.0003:
                    break
        h = project_feasible(h, n)

    # Phase 2: Top-K^p smooth optimization (22% budget)
    p2_dl = t0 + 0.50 * total
    for p_val in [4, 8, 16]:
        if time.time() > p2_dl - 30:
            break
        K = 20
        try:
            cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
            res = minimize(topk_p_fun, best_h.copy(), jac=topk_p_grad,
                          args=(n, K, p_val), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            mc_new = float(np.max(np.correlate(h_new, 1 - h_new, mode='full')))
            if mc_new < best_mc:
                best_mc = mc_new
                best_h = h_new.copy()
                best_c5 = mc_new * 2.0 / n
        except Exception:
            pass

    # Phase 3: Re-engage LP after smooth phase (15% budget)
    p3_dl = t0 + 0.65 * total
    h = best_h.copy()
    eps = 0.03
    no_imp = 0
    while time.time() < p3_dl:
        h, mc = lp_step(h, n, eps)
        if mc < best_mc - 1e-12:
            best_mc = mc
            best_h = h.copy()
            best_c5 = mc * 2.0 / n
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                eps *= 0.7
                no_imp = 0
                if eps < 0.0002:
                    break
        h = project_feasible(h, n)

    # Phase 4: Multi-resolution (20% budget)
    p4_dl = t0 + 0.85 * total
    n_best = len(best_h)
    for nh in [200, 300, 500, 800]:
        if time.time() > p4_dl - 15:
            break
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        mc_h = float(np.max(np.correlate(hh, 1 - hh, mode='full')))
        eps_r = 0.04
        improved = False
        for _ in range(100):
            if time.time() > p4_dl - 8:
                break
            hh, mc_new = lp_step(hh, nh, eps_r)
            c5_new = mc_new * 2.0 / nh
            if c5_new < best_c5:
                best_mc = mc_new
                best_h = hh.copy()
                best_c5 = c5_new
                n_best = nh
                improved = True
            hh = project_feasible(hh, nh)
            eps_r = max(eps_r * 0.995, 0.001)
        if not improved:
            n_best = len(best_h)

    # Phase 5: Structured perturbations + LP (10% budget)
    p5_dl = dl - 10
    n_cur = len(best_h)
    for trial in range(40):
        if time.time() > p5_dl:
            break
        if trial % 3 == 0:
            shift = int(rng.randint(1, max(2, n_cur // 8)))
            h_try = np.roll(best_h, shift)
        elif trial % 3 == 1:
            h_try = best_h + 0.01 * rng.randn(n_cur)
        else:
            center = rng.randint(n_cur // 4, 3 * n_cur // 4)
            width = max(3, n_cur // 15)
            mask = (np.arange(n_cur) >= center - width) & (np.arange(n_cur) <= center + width)
            h_try = best_h.copy()
            h_try[mask] += 0.03 * rng.randn(mask.sum())
        h_try = project_feasible(h_try, n_cur)
        eps_r = 0.02
        for _ in range(15):
            if time.time() > p5_dl:
                break
            h_try, mc = lp_step(h_try, n_cur, eps_r)
            c5 = mc * 2.0 / n_cur
            if c5 < best_c5:
                best_mc = mc
                best_h = h_try.copy()
                best_c5 = c5
            h_try = project_feasible(h_try, n_cur)
            eps_r = max(eps_r * 0.99, 0.002)

    # Phase 6: Final fine LP polish
    h = best_h.copy()
    n = len(h)
    eps_f = 0.003
    for _ in range(400):
        if time.time() > dl - 3:
            break
        h, mc = lp_step(h, n, eps_f)
        c5 = mc * 2.0 / n
        if c5 < best_c5:
            best_mc = mc
            best_h = h.copy()
            best_c5 = c5
        h = project_feasible(h, n)
        eps_f = max(eps_f * 0.999, 0.00003)

    best_h = project_feasible(best_h, len(best_h))
    n_final = len(best_h)
    best_c5 = compute_c5(best_h, n_final)
    return best_h, best_c5, n_final
