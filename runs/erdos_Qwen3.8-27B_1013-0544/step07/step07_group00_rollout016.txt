    mx = shifted.max()

    return float((np.sum(c**p))**(1.0/p)) * dx

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def penalty_obj(x, n, alpha, lam, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    s = x.sum() - n / 2.0
    return (m + lse / alpha) * dx + lam * s * s

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def lfbgbs_opt(h, n, alphas, lam, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for alpha in alphas:
        res = minimize(penalty_obj, best_h, args=(n, alpha, lam, dx),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter_each, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    return best_h, best_c

def slsqp_opt(h, n, alphas, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for alpha in alphas:
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        res = minimize(smooth_obj, best_h, args=(n, alpha, dx),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=cons,
                       options={'maxiter': maxiter_each, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    return best_h, best_c

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=60):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max()) * dx
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            mm_lo = max(0, n - 1 - p)
            mm_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(mm_lo, mm_hi + 1)
            q = p + ms - (n - 1)
            r = p - ms + (n - 1)
            a = (1.0 - h[q]) - h[r]
            c_aff = c[ms]
            unaff = 0.0
            if mm_lo > 0:
                unaff = float(c[:mm_lo].max())
            if mm_hi < 2 * n - 2:
                unaff = max(unaff, float(c[mm_hi + 1:].max()))
            diag_mask = (ms == (n - 1))
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = current_max
            for d in d_grid:
                if abs(d) < 1e-15:
                    continue
                nv = c_aff + d * a
                if np.any(diag_mask):
                    nv[diag_mask] -= d * d
                val = max(unaff, float(nv.max())) * dx
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def make_m_sequence(n, seed_val):
    rng = np.random.RandomState(seed_val)
    period = 1
    while period < n:
        period *= 2
    mseq = np.zeros(period)
    state = 1
    for i in range(period):
        mseq[i] = 1.0 if state & 1 else 0.0
        lsb = state & ((1 << 4) - 1)
        new_bit = ((lsb >> 3) ^ (lsb >> 0)) & 1
        state = (state >> 1) | (new_bit << (period - 1)) if period <= 32 else (state >> 1) | new_bit
    mseq = mseq[:n]
    s = mseq.sum()
    target = n // 2
    if s > target:
        ones = np.where(mseq == 1)[0]
        mseq[rng.choice(ones, s - target, replace=False)] = 0
    elif s < target:
        zeros = np.where(mseq == 0)[0]
        mseq[rng.choice(zeros, target - s, replace=False)] = 1
    return mseq

def make_structured_start(n, rng, mode=0):
    if mode == 0:
        return make_m_sequence(n, rng.randint(1000))
    elif mode == 1:
        h = np.zeros(n)
        h[::2] = 1.0
        s = h.sum()
        target = n // 2
        if s > target:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, s - target, replace=False)] = 0
        return h
    elif mode == 2:
        freq = rng.uniform(0.3, 2.0)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        h = np.clip(h, 0, 1)
        return project_feasible(h, n)
    else:
        h = rng.uniform(0, 1, n)
        return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B quick refinement
    h1, c1 = lfbgbs_opt(best_h, n_init, [500, 2000, 10000], 200.0, maxiter_each=200)

    h3, c3 = coord_cd(best_h, n_init, 80, deadline, grid_size=70)

        if n_try == len(best_h):

        h_opt, c_opt = lfbgbs_opt(h_try, n_try, [2000, 8000], 100.0, maxiter_each=150)

            scale = 0.03 + 0.04 * r
            hp = project_feasible(best_h + rng.normal(0, scale, n_best), n_best)
            h_o, c_o = slsqp_opt(hp, n_best, [10000, 50000], maxiter_each=400)

        if cf < best_c:
            best_h, best_c = hf, cf

    # Phase 8: Final coordinate descent polish
    if time.time() < deadline - 3:
        hf, cf = coord_cd(best_h, n_best, 150, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
