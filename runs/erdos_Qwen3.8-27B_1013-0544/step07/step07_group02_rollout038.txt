- Term 2: G[j, m] += -h[j-k] for j such that 0 <= j-k < n, i.e., k <= j < n+k. With j < n: if k>=0: j in [k, n-1], G[j,m] += -h[j-k] which is -h[0:n-k]
  - If k < 0: j in [0, n+k-1], G[j,m] += -h[j-k] which is -h[-k:n]

def build_G(h, n):
    n_c = 2*n - 1
    G = np.zeros((n, n_c))
    for k in range(-(n-1), n):
        m = k + n - 1
        if k >= 0:
            # Term 1: j in [0, n-k-1], value = 1-h[j+k] = 1-h[k:n]
            G[:n-k, m] += 1.0 - h[k:n]
            # Term 2: j in [k, n-1], value = -h[j-k] = -h[0:n-k]
            G[k:, m] -= h[:n-k]
        else:
            # k < 0, let kk = -k > 0
            kk = -k
            # Term 1: j in [kk, n-1], value = 1-h[j-k] = 1-h[j+kk] = 1-h[kk:n]
            # Wait, j+k = j-kk, so we need 1-h[j-kk] for j in [kk, n-1]
            # That's 1-h[0:n-kk] = 1-h[0:n+kk]
            G[kk:, m] += 1.0 - h[:n-kk]
            # Term 2: j in [0, n-kk-1], value = -h[j+jj] = -h[j+kk] for j in [0, n-kk-1]
            # That's -h[kk:n]
            G[:n-kk, m] -= h[kk:]
    return G

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_G(h, n):
    n_c = 2 * n - 1
    G = np.zeros((n, n_c))
    for k in range(-(n - 1), n):
        m = k + n - 1
        if k >= 0:
            G[:n - k, m] += 1.0 - h[k:n]
            G[k:, m] -= h[:n - k]
        else:
            kk = -k
            G[kk:, m] += 1.0 - h[:n - kk]
            G[:n - kk, m] -= h[kk:]
    return G

    C = np.correlate(h, 1 - h, mode='full').astype(float) * dx
    mc = float(C.max())
    n_c = len(C)

    A_ub[:n_c, n] = -1.0
    b_ub[:n_c] = -C
    
    A_ub[n_c:n_c + n, :n] = np.eye(n)
    b_ub[n_c:n_c + n] = np.minimum(eps, 1.0 - h)
    
    A_ub[n_c + n:n_c + 2 * n, :n] = -np.eye(n)
    b_ub[n_c + n:n_c + 2 * n] = np.minimum(eps, h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         method='highs', options={'presolve': True})
    except Exception:
        return h, mc
    
    if not result.success or result.x is None:
        return h, mc

        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try.copy()
    return best_h, best_mc

def smooth_subgrad(h, n, alpha, n_iter):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur, n)

        if c5_step < best_c5:
            best_c5 = c5_step
            best_h = h_step.copy()
        h_cur = h_step
    
    return best_h, best_c5

    phase4_end = deadline - 12
    h = best_h.copy()
    for alpha in [500, 2000, 8000]:
        if time.time() > phase4_end:
            break
        h, c5_val = smooth_subgrad(h, n, alpha, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    # Phase 5: Final fine LP polish
    if time.time() < deadline - 2:
        h = best_h.copy()
        eps_f = 0.004
        for _ in range(300):
            if time.time() > deadline - 2:
                break
            h, mc = lp_step(h, n, eps_f)
            if mc < best_c5 - 1e-14:
                best_c5 = mc
                best_h = h.copy()
            eps_f = max(eps_f * 0.999, 0.0003)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
