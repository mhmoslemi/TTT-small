def project_box_sum(x, target):

    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)

        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def precompute_grad_mats(n):
    """Precompute gradient computation matrices for top-k gradient."""
    mid = n - 1
    w_pos = np.zeros((2 * n - 1, n))
    w_neg = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - mid)
        hi = min(n, k + 1)
        count = hi - lo
        if count > 0:
            w_pos[k, lo:hi] = 1.0
            j_start = lo - (k - mid)
            w_neg[k, j_start:j_start + count] = 1.0
    return w_pos, w_neg

_LS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'k': 10, 'rho': 0.0}

def topk_weighted_grad(h, w, n, dx):
    """Compute gradient of sum_k w[k]*c[k] where c = corr(h,1-h)*dx."""
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def softmax_vals(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return float(maxc + np.log(ws) / beta), w

def pair_polish(h, n, deadline, seed, max_rounds=5):

    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    m = 30 if n <= 500 else 20
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = m - int(order.size)
        if need > 0:
            extra = rng.choice(n, size=min(need, n), replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for _ in range(max_rounds):
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):
                pairs.append((int(top[a]), int(top[b])))
        if not pairs:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 3.0:
                return h, erdos_c5(h, n)
            hi0 = h[i]
            hj0 = h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            best_d = 0.0
            best_c = cur
            for d_try in (lo, 0.5 * (lo + hi), hi):
                hh = h.copy()
                hh[i] = hi0 + d_try
                hh[j] = hj0 - d_try
                c_try = erdos_c5(hh, n)
                if c_try < best_c:
                    best_c = c_try
                    best_d = d_try
            try:
                if time.time() < deadline - 1.0:
                    def _po(d):
                        a = h[i] + d
                        b = h[j] - d
                        if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
                            return 1e100
                        hh = h.copy()
                        hh[i] = a
                        hh[j] = b
                        return erdos_c5(hh, n)
                    rs = so.minimize_scalar(_po, bounds=(lo, hi), method='bounded',
                                            options={'maxiter': 30, 'xatol': 1e-11})
                    if rs.x is not None and rs.fun < best_c:
                        best_c = float(rs.fun)
                        best_d = float(rs.x)
            except Exception:
                pass
            if best_c < cur - 1e-14:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)

                improved = True
        if not improved:

                best_h = hh.copy()
                best_n = nh
                best_c = c

    # Phase 4: Aggressive pair polish
    if time.time() < end4 - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 15.0, seed, max_rounds=6)

    # Phase 5: Final refinement
    if time.time() < end4 - 10.0:
        for k in (3, 10):
            cand, c = refine_lbfgs_topk(best_h, k, 1e5, 30, deadline)
            if c < best_c - 1e-15:
                best_h = cand.copy()
                best_c = c

    # Phase 6: Random restarts (small perturbations)
    rng = np.random.RandomState(int(seed) + 99)
    for sig in (0.02, 0.05, 0.03):
        if time.time() > end4 - 6.0:
            break
        hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
        hh, c = refine_lbfgs_topk(hh, 10, 1e4, 35, deadline)
        if c < best_c - 1e-15:
            best_h = hh.copy()
            best_c = c

    # Final: one more pair polish
    if time.time() < end4 - 3.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 1.0, seed + 7, max_rounds=3)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
