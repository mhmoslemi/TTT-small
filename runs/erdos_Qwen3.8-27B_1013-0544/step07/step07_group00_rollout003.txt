import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_brief(h, n, alphas, maxiter=600):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

    for r in range(n_restarts):

    if time.time() < deadline - 600:
        h1, c1 = slsqp_brief(best_h, n_init, [500, 5000], 800)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Exact CD
    h2, c2 = coord_exact(best_h, n_init, 500, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Multi-restart at n_init
    n_r = max(5, int((deadline - time.time()) / 120))
    if time.time() < deadline - 100:
        h3, c3 = multi_restart(best_h, n_init, rng, min(n_r, 25), deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 300, 600, 250]:
        if time.time() > deadline - 80:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        ho, co = slsqp_brief(h_int, n_try, [1000, 10000], 400)
        ho, co = coord_exact(ho, n_try, 300, deadline)
        if time.time() < deadline - 30:
            ho, co = multi_restart(ho, n_try, rng, 5, deadline)
        if co < best_c:
            best_h, best_c = ho, co
            x_orig = x_new
    # Phase 5: Final polish
    if time.time() < deadline - 10:
        n_fin = len(best_h)
        hf, cf = coord_exact(best_h, n_fin, 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
        if time.time() < deadline - 5:
            hf, cf = multi_restart(best_h, n_fin, rng, 3, deadline)
            if cf < best_c:
                best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
