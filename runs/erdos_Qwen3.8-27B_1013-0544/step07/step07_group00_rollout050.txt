    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu=0.0):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted.max()
    lse = s + np.log(np.sum(np.exp(shifted - s)))
    obj = (m + lse / alpha) * (2.0 / n)
    if mu > 0:
        obj -= mu * (np.sum(x * (1 - x)) / n)
    return obj

def slsqp_opt(h, n, alphas, maxiter=500, mu=0.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
                 'jac': lambda x, n=n: np.ones(n)}]
        try:
            res = minimize(lambda x: smooth_obj(x, n, alpha, mu), best_h,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=300):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            # Compute affected lags and coefficients
            offsets = np.arange(-(n-1), n)
            valid_f = (p + offsets >= 0) & (p + offsets < n)
            valid_b = (p - offsets >= 0) & (p - offsets < n)
            valid = valid_f | valid_b
            if not np.any(valid):
                continue
            ms = offsets[valid] + (n - 1)
            a = np.zeros(len(ms))
            for idx in range(len(ms)):
                o = offsets[valid][idx]
                coef = 0.0
                if 0 <= p + o < n:
                    coef += (1.0 - h[p + o])
                if 0 <= p - o < n:
                    coef -= h[p - o]
                a[idx] = coef
            # Vectorized grid evaluation
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            c_aff = c[ms]
            center_idx = (ms == (n-1))
            # For non-center: c_aff + d*a
            # For center: c_aff + d*a - d^2
            vals = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if np.any(center_idx):
                vals[:, center_idx] -= d_grid[:, None]  # subtract d^2... wait
            # Actually the center correction is: the formula gives d*a[center] which
            # already includes (1-2h0) from the two terms. The extra -d^2 needs to be added.
            # c_center_new = c_center + d*(1-h[p]) - h[p]*d - d^2 = c_center + d*(1-2h0) - d^2
            # And a[center] = (1-h[p]) - h[p] = 1-2h0 (assuming both valid, which they are for o=0)
            # So vals[center] = c_center + d*(1-2h0), and we need to subtract d^2.
            if np.any(center_idx):
                vals[:, center_idx] -= d_grid**2
            row_max = vals.max(axis=1)
            # Unaffected max
            unaff_max = 0.0
            if ms[0] > 0:
                unaff_max = max(unaff_max, float(c[:ms[0]].max()))
            if ms[-1] < 2*n-2:
                unaff_max = max(unaff_max, float(c[ms[-1]+1:].max()))
            total_max = np.maximum(row_max, unaff_max)
            best_idx = total_max.argmin()
            best_d = d_grid[best_idx]
            best_val = total_max[best_idx]
            if best_val < current_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

Let me also think about the 0-1 structure more. The key insight is:
- We want a subset A of {0,...,n-1} with |A| = n/2
- That minimizes max_k |A \ (A+k)| (measuring the "difference")

This is equivalent to maximizing min_k |A ∩ (A+k)| (the autocorrelation).

For the LINEAR correlation (not circular), the valid range of k is -(n-1) to n-1, and |A ∩ (A+k)| is only defined for the overlapping portion.

A key structural insight: if I think of A as a "code" in Z_n (approximately), I want a code with good autocorrelation properties. The best known constructions for this are:
- **Barker sequences** (only exist for n ≤ 13)
- **m-sequences** (cyclic, nearly perfect autocorrelation)
- **Legendre sequences** (based on quadratic residues)
- **Costas arrays**
