Actually, I realize that for the peak lag, the total delta is:
Δc(peak) = A_i + B_j + C(i,j)

where A_i = (1-h[i+m])*1_{0≤i+m<n} - h[i-m]*1_{0≤i-m<n} (contribution from h[i] 0→1)
B_j = -(1-h[j+m])*1_{0≤j+m<n} + h[j-m]*1_{0≤j-m<n} (contribution from h[j] 1→0)
C(i,j) = +1 if i<j and i+m=j, -1 if i>j and i-m=j, 0 otherwise

So Δc(peak) = A_i + B_j + C(i,j)

For fixed i: A_i is fixed. We want to minimize corr[peak] + A_i + B_j + C(i,j) over j in ones.

= (corr[peak] + A_i) + min_j (B_j + C(i,j))

The term B_j + C(i,j) differs from B_j only for j = i+m (C=+1) and j = i-m (C=-1).

So the best j is:
- argmin of B_j over j in ones, UNLESS i+m or i-m is in ones and gives a better value.

- Let j0 = argmin_{j in ones} B_j
- Let j1 = i+m (if in ones), value = B[j1] + 1
- Let j2 = i-m (if in ones), value = B[j2] - 1

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def fast_binary_search(h_init, n, deadline, max_sweeps=50000):

            break

            best_h = h_b.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > no_improve_limit:
            break
        peak_k = int(np.argmax(corr))
        m = peak_k - (n - 1)
        A = np.zeros(n)
        B = np.zeros(n)
        if m == 0:
            A[:] = 1.0 - 2.0 * h_b
            B[:] = A.copy()

def adaptive_subgrad(h, n, deadline, max_iters=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rms = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    stall_limit = 5000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            rng = np.random.RandomState(42 + t % 100)
            h = project_feasible(best_h + 0.01 * rng.randn(n), n)
            rms = np.zeros(n)
            stall = 0
            continue
        c_sorted = np.sort(c)[::-1]
        max_val = c_sorted[0]
        n_close = max(1, int(np.sum(c_sorted > 0.95 * max_val)))
        top_k = min(n, max(1, n_close + 1))
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            if k >= 0:
                if k == 0:
                    g = 1.0 - 2.0 * h
                else:
                    g = np.zeros(n)
                    g[k:] += (1.0 - h[:n - k])
                    g[:n - k] -= h[k:]
            else:
                kk = -k
                g = np.zeros(n)
                g[:n - kk] += (1.0 - h[kk:])
                g[kk:] -= h[:n - kk]
            grad += g
        grad = dx * grad / top_k
        rms = beta * rms + (1 - beta) * grad ** 2
        lr = 0.15 / (np.sqrt(rms) + 1e-8) / (1.0 + 0.005 * t)
        h_new = project_feasible(h - lr * grad, n)
        c_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if c_new < cur_c + 0.002:
            h = h_new
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
