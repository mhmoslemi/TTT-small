    # kmax is the index in the full correlation (0 to 2n-2)

def pair_cd_iteration(h, c_all, n, i, j):
    """One pair step. Updates h and c_all in place. Returns improvement."""
    lags = np.arange(2*n-1) - (n-1)  # precomputed
    
    # Compute slope
    ip = i + lags
    im = i - lags
    jp = j + lags
    jm = j - lags
    
    slope = np.zeros(2*n-1)
    m1 = (ip >= 0) & (ip < n)
    m2 = (im >= 0) & (im < n)
    m3 = (jp >= 0) & (jp < n)
    m4 = (jm >= 0) & (jm < n)
    slope[m1] += 1.0 - h[ip[m1]]
    slope[m2] -= h[im[m2]]
    slope[m3] -= 1.0 - h[jp[m3]]
    slope[m4] += h[jm[m4]]
    
    # Bounds on d
    lo = max(-h[i], 1.0 - h[j])
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0
    
    # Minimize max(c_all + slope*d) over d in [lo, hi]
    # Grid search
    ngrid = 100
    dvals = np.linspace(lo, hi, ngrid)
    # Vectorized: c_all[None,:] + dvals[:,None]*slope[None,:] -> max over axis=1
    # But this creates a ngrid x (2n-1) array. For n=400, that's 100x799 = 79900 elements. Fine.
    vals = np.max(c_all[None, :] + dvals[:, None] * slope[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_d = dvals[best_idx]
    best_val = vals[best_idx]
    
    # Golden section refinement
    if 0 < best_idx < ngrid - 1:
        a, b = dvals[best_idx-1], dvals[best_idx+1]
        for _ in range(25):
            m1gs = a + (b-a)/golden_ratio  # golden section points
            m2gs = a + (b-a)*(1-1/golden_ratio)
            f1 = np.max(c_all + slope*m1gs)
            f2 = np.max(c_all + slope*m2gs)
            if f1 < f2:
                b = m2gs
            else:
                a = m1gs
        best_d = 0.5*(a+b)
        best_val = np.max(c_all + slope*best_d)
    
    cur_max = np.max(c_all)
    if best_val < cur_max - 1e-15:
        h[i] += best_d
        h[j] -= best_d
        c_all += slope * best_d
        return cur_max - best_val
    return 0.0
