def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h_b[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h_b[imn[msk]]
            jm = best_j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h_b[jm[msk]]
            jmn = best_j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h_b[jmn[msk]]
            delta[n - 1] -= 2.0
            delta[n - 1 + (best_j - i)] += 1.0
            delta[n - 1 + (i - best_j)] += 1.0
            h_b[i], h_b[best_j] = 1.0, 0.0
            corr += delta
            cur_c = best_nc
            ones_list.remove(best_j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(best_j)
            improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

def two_swap_search(h_b, n, corr, deadline, max_trials=5000, rng=None):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    best_corr = corr.copy()
    if rng is None:
        rng = np.random.RandomState(999)
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    for _ in range(max_trials):
        if time.time() > deadline:
            break
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break
        i1, i2 = rng.choice(ones_list, size=2, replace=False)
        j1, j2 = rng.choice(zeros_list, size=2, replace=False)
        i1, i2 = int(i1), int(i2)
        j1, j2 = int(j1), int(j2)
        delta = np.zeros(2 * n - 1)
        for pos, sign in [(i1, -1), (i2, -1), (j1, 1), (j2, 1)]:
            im = pos + m_range
            msk = (im >= 0) & (im < n)
            if sign == 1:
                delta[msk] += 1.0 - h_b[im[msk]]
            else:
                delta[msk] -= 1.0 - h_b[im[msk]]
            imn = pos - m_range
            msk = (imn >= 0) & (imn < n)
            if sign == 1:
                delta[msk] -= h_b[imn[msk]]
            else:
                delta[msk] += h_b[imn[msk]]
        positions = [i1, i2, j1, j2]
        for a in range(4):
            for b in range(a + 1, 4):
                pa, pb = positions[a], positions[b]
                sa = -1 if pa in (i1, i2) else 1
                sb = -1 if pb in (i1, i2) else 1
                if sa == sb:
                    continue
                d = (pb + m_range) - pa
                m1 = (n - 1) + d
                if 0 <= m1 < 2 * n - 1:
                    delta[m1] += 1.0
                m2 = (n - 1) - d
                if 0 <= m2 < 2 * n - 1:
                    delta[m2] += 1.0
        nc = (corr + delta).max() * dx
        if nc < cur_c - 1e-14:
            h_b[i1], h_b[i2], h_b[j1], h_b[j2] = 0, 0, 1, 1
            corr += delta
            cur_c = nc
            ones_list = [i for i in ones_list if i not in (i1, i2)]
            ones_list.extend([j1, j2])
            zeros_list = [i for i in zeros_list if i not in (j1, j2)]
            zeros_list.extend([i1, i2])
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
                best_corr = corr.copy()
    return best_h, best_c, best_corr

def subgrad_optimize(h, n, deadline, max_iters=50000, top_k=3, lr_base=0.08):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    stall_limit = 5000
    m_range = np.arange(-(n - 1), n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] += 1.0 - 2.0 * h
                else:
                    m = m_range + k
                    msk = (m >= 0) & (m < n)
                    grad[msk] += (1.0 - h[m[msk]])
                    m2 = m_range + (-k)
                    msk2 = (m2 >= 0) & (m2 < n)
                    grad[msk2] -= h[m2[msk2]]
            else:
                kk = -k
                if kk == 0:
                    grad[:] += 1.0 - 2.0 * h
                else:
                    m = m_range + (-kk)
                    msk = (m >= 0) & (m < n)
                    grad[msk] += (1.0 - h[m[msk]])
                    m2 = m_range + kk
                    msk2 = (m2 >= 0) & (m2 < n)
                    grad[msk2] -= h[m2[msk2]]
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.005:
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    p1 = t0 + 0.15 * total
    alphas = [50000, 500000, 5e6, 5e7, 5e8]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 1500, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    p2 = t0 + 0.55 * total
    hb, cb = binary_local_search(best_h, n, p2, max_sweeps=300, use_sa=False)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    p3 = t0 + 0.70 * total
    h_bin = np.zeros(n)
    h_bin[np.argsort(best_h)[::-1][:n // 2]] = 1.0
    corr_bin = np.correlate(h_bin, 1 - h_bin, mode='full').astype(float)
    rng2 = np.random.RandomState(seed + 1)
    h_bin, c_two, corr_bin = two_swap_search(h_bin, n, corr_bin, p3, max_trials=8000, rng=rng2)
    if c_two < best_c:
        best_c, best_h = c_two, h_bin.copy()

    p4 = t0 + 0.82 * total
    hb, cb = binary_local_search(best_h, n, p4, max_sweeps=200, use_sa=True, sa_temp=0.003)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    p5 = t0 + 0.92 * total
    hs, cs = subgrad_optimize(best_h, n, p5, max_iters=30000, top_k=5, lr_base=0.05)
    if cs < best_c:
        best_c, best_h = cs, hs.copy()

    if time.time() < dl - 15:
        h = best_h + 0.02 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [5e7, 5e8]:
            if time.time() > dl - 5:
                break
            h = slsqp_optimize(h, n, alpha, 1000, 1e-15)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        hs, cs = subgrad_optimize(best_h, n, dl - 1, max_iters=10000, top_k=3, lr_base=0.03)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
