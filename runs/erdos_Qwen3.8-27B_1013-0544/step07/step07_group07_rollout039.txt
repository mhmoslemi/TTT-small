def c5_value(h, n):
    return c_max(h, n) * (2.0 / n)


def project_feasible(h, n):
    h = np.asarray(h, float).reshape(-1)
    if h.size != n:
        x0 = np.linspace(0.0, n, max(h.size, 1))
        h = np.interp(np.arange(n, dtype=float), x0, h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s <= 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s <= 1e-14:
                break
            h += (d / s) * h
        np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if d > 1e-13:
        i = int(np.argmax(1.0 - h))
        h[i] = min(1.0, h[i] + d)
    elif d < -1e-13:
        i = int(np.argmax(h))
        h[i] = max(0.0, h[i] + d)
    return h


def to_binary(h, n):
    h = np.asarray(h, float).reshape(-1)
    k = n // 2
    if k <= 0:
        return np.zeros(n, float)
    if k >= len(h):
        return np.ones(n, float)
    idx = np.argpartition(h, len(h) - k)[-k:]
    b = np.zeros(n, float)
    b[idx] = 1.0
    return b


def weighted_gradient(w, h, n):
    off = n - 1
    s = slice(off, off + n)
    cw = np.empty(2 * n, float)
    cw[0] = 0.0
    np.cumsum(w, out=cw[1:])
    A = cw[n:2 * n] - cw[:n]
    wh = np.convolve(w, h, mode='full')[s]
    wrh = np.convolve(w[::-1], h, mode='full')[s]
    return A - wh - wrh


def obj_fun(x):
    n = G_N
    c = np.maximum(np.correlate(x, 1.0 - x, mode='full'), 0.0).astype(float)
    if G_MODE == 'lse':
        m = float(c.max())
        if m <= 1e-15:
            return 0.0, np.zeros(n, float)
        z = np.clip(G_ALPHA * (c - m), -700.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            return m, weighted_gradient(w, x, n)
        w = e / s
        return m + math.log(s) / G_ALPHA, weighted_gradient(w, x, n)
    if G_MODE == 'pnorm':
        base = float(c.max())
        if base <= 1e-15:
            return 0.0, np.zeros(n, float)
        u = c / base
        p = G_P
        up = np.power(u, p - 1.0)
        S = float(np.dot(u, up))
        if S <= 1e-300:
            S = 1.0
        w = S ** (1.0 / p - 1.0) * up
        return base * S ** (1.0 / p), weighted_gradient(w, x, n)
    ex = c - G_T
    pos = ex > 0.0
    if not np.any(pos):
        return 0.0, np.zeros(n, float)
    e = np.where(pos, ex, 0.0)
    f = float(np.sum(e) + 0.5 * float(np.dot(e, e)))
    w = np.where(pos, 1.0 + e, 0.0)
    return f, weighted_gradient(w, x, n)


def eq_fun(x):
    return float(np.sum(x) - 0.5 * G_N)


def eq_jac(x):
    return G_ONES.copy()


def slsqp_call(h, n, mode, param, maxiter):
    global G_N, G_ALPHA, G_T, G_P, G_ONES, G_BOUNDS
    if maxiter <= 0:
        return h
    G_N = int(n)
    G_ONES = np.ones(G_N)
    G_BOUNDS = [(0.0, 1.0)] * G_N
    if mode == 'lse':
        G_MODE = 'lse'
        G_ALPHA = float(param)
    elif mode == 'pnorm':
        G_MODE = 'pnorm'
        G_P = float(param)
    else:
        G_MODE = 'barrier'
        G_T = float(param)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fun, h, method='SLSQP', bounds=G_BOUNDS, jac=True,
                       constraints=cons, options={'maxiter': int(maxiter),
                                                   'ftol': 1e-11, 'disp': False})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            return project_feasible(res.x, G_N)
    except Exception:
        pass
    return h


def safe_maxiter(n, rem):
    if rem < 20:
        return 15
    if rem < 45:
        return 35
    if rem < 90:
        return 70
    if n >= 700:
        return 60
    if n >= 520:
        return 90
    return 130


def lp_direction(h, n, cfull, eps):
    if linprog is None:
        return None
    maxc = float(cfull.max())
    if maxc <= 1e-12:
        return None
    thresh = maxc - max(1e-5, 0.0002 * maxc)
    idx = np.where(cfull >= thresh)[0]
    if idx.size > 80:
        idx = np.argsort(cfull)[-80:]
    rows = []
    acts = []
    off = n - 1
    for r in idx:
        d = int(r - off)
        g = np.zeros(n, float)
        if d >= 0:
            if d < n:
                g[d:] += 1.0 - h[:n - d]
                g[:n - d] -= h[d:]
        else:
            a = -d
            if a < n:
                g[:n - a] += 1.0 - h[a:]
                g[a:] -= h[:n - a]
        rows.append(g)
        acts.append(float(cfull[r]))
    if not rows:
        return None
    G = np.array(rows, float)
    a = np.array(acts, float)
    A = np.column_stack((G, -np.ones(len(G))))
    cO = np.zeros(n + 1, float)
    cO[-1] = 1.0
    Aeq = np.zeros((1, n + 1), float)
    Aeq[0, :n] = 1.0
    bds = [(-eps, eps)] * n + [(0.0, max(float(maxc), 1.0))]
    try:
        res = linprog(cO, A_ub=A, b_ub=-a, A_eq=Aeq, b_eq=[0.0],
                      bounds=bds, method='highs')
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    tau = float(res.x[n])
    if tau >= maxc - 1e-7:
        return None
    return res.x[:n].copy(), tau


def lp_push(h, n, deadline, iters=12):
    h = project_feasible(h, n).copy()
    cur = c_max(h, n)
    best = cur
    eps = min(0.04, max(0.004, 0.002 * cur))
    stall = 0
    for _ in range(iters):
        if time.time() > deadline - 2:
            break
        cf = np.maximum(np.correlate(h, 1.0 - h, mode='full'), 0.0)
        cur = float(cf.max())
        if cur < best:
            best = cur
        dr = lp_direction(h, n, cf, eps)
        if dr is None:
            eps *= 0.5
            stall += 1
            if stall >= 3 or eps < 1e-6:
                break
            continue
        delta, tau = dr
        accepted = False
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            x = project_feasible(h + alpha * delta, n)
            nc = c_max(x, n)
            if nc < cur - 1e-12:
                h = x
                cur = nc
                accepted = True
                break
        if accepted:
            best = min(best, cur)
            eps = max(1e-5, eps * 0.8)
            stall = 0
        else:
            eps *= 0.5
            stall += 1
            if stall >= 3 or eps < 1e-6:
                break
    return h, best * (2.0 / n)


def do_slsqp_sequence(h, n, deadline, profile):
    h = project_feasible(h, n)
    c = c5_value(h, n)
    if profile == 'strong':
        tasks = [('lse', a) for a in (20.0, 60.0, 200.0, 600.0, 2000.0, 6000.0, 15000.0)]
        tasks += [('pnorm', p) for p in (8.0, 32.0, 128.0)]
        biter = 4
        do_lp = True
        lpi = 12
    elif profile == 'medium':
        tasks = [('lse', a) for a in (10.0, 40.0, 200.0, 800.0, 3000.0)]
        tasks += [('pnorm', p) for p in (8.0, 64.0)]
        biter = 2
        do_lp = False
        lpi = 0
    else:
        tasks = [('lse', a) for a in (2000.0, 8000.0, 30000.0)]
        tasks += [('pnorm', p) for p in (64.0, 256.0)]
        biter = 2
        do_lp = True
        lpi = 8

    for mode, p in tasks:
        rem = deadline - time.time()
        if rem < 10:
            break
        h2 = slsqp_call(h, n, mode, float(p), safe_maxiter(n, rem))
        c2 = c5_value(h2, n)
        if c2 < c - 1e-14:
            h, c = h2, c2

    for _ in range(biter):
        rem = deadline - time.time()
        if rem < 10:
            break
        tcount = c * n / 2.0
        dec = max(0.05, 0.001 * tcount)
        t = max(0.0, tcount - dec)
        h2 = slsqp_call(h, n, 'barrier', float(t), safe_maxiter(n, rem))
        c2 = c5_value(h2, n)
        if c2 < c - 1e-14:
            h, c = h2, c2

    if do_lp and time.time() < deadline - 8:
        h, c = lp_push(h, n, deadline - 3, lpi)
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    global G_N, G_ONES, G_BOUNDS
    t0 = time.time()
    total = max(60.0, float(budget_s) - 25.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    h0 = None
    if 'initial_h_values' in globals():
        try:
            h0 = np.asarray(initial_h_values, float).reshape(-1)
        except Exception:
            h0 = None

    if h0 is None or h0.size < 50:
        n = 400
        h = to_binary(rng.random(n), n)
    else:
        n = int(h0.size)
        if n < 120:
            n2 = 256
            oldx = np.linspace(0.0, 2.0, h0.size)
            newx = np.linspace(0.0, 2.0, n2)
            h = np.interp(newx, oldx, h0)
            n = n2
        elif n > 800:
            n2 = 600
            oldx = np.linspace(0.0, 2.0, h0.size)
            newx = np.linspace(0.0, 2.0, n2)
            h = np.interp(newx, oldx, h0)
            n = n2
        else:
            h = h0.copy()

    h = project_feasible(h, n)
    G_N = n
    G_ONES = np.ones(n)
    G_BOUNDS = [(0.0, 1.0)] * n

    best_h = h.copy()
    best_c = c5_value(best_h, n)

    cands = []
    idx = 0
    cb = to_binary(best_h, n)
    cands.append((c5_value(best_h, n), sidx if False else str(idx), best_h))
    idx += 1
    cands.append((c5_value(cb, n), str(idx), cb))
    idx += 1
    for _ in range(3):
        hb = to_binary(rng.random(n), n)
        cands.append((c5_value(hb, n), str(idx), hb))
        idx += 1

    ii = np.arange(n, float)
    gl = (math.sqrt(5.0) - 1.0) / 2.0
    sq2 = math.sqrt(2.0) - 1.0
    structs = [
        np.mod(ii * gl, 1.0),
        np.mod(ii * ii * gl + 0.29, 1.0),
        np.mod(ii * sq2 + 0.47, 1.0),
        np.mod(ii * ii * sq2 - 0.13, 1.0),
    ]
    for sc in structs:
        hb = to_binary(-sc, n)
        cands.append((c5_value(hb, n), str(idx), hb))
        idx += 1

    xx = np.linspace(0.0, 2.0, n)
    for f in (1, 2, 3):
        amp = 0.45 if f % 2 else 0.30
        ph = rng.uniform(0.0, 2.0 * math.pi)
        vv = project_feasible(np.clip(0.5 + amp * np.sin(math.pi * f * xx + ph), 0.0, 1.0), n)
        cands.append((c5_value(vv, n), str(idx), vv))
        idx += 1

    for amp in (0.03, 0.08):
        vv = project_feasible(best_h + amp * rng.randn(n), n)
        cands.append((c5_value(vv, n), str(idx), vv))
        idx += 1

    cands.sort()

    fracs = (0.42, 0.58, 0.70, 0.80, 0.86)
    for r in range(min(5, len(cands))):
        rem = deadline - time.time()
        if rem < 50:
            break
        edge = t0 + fracs[r] * total if r < len(fracs) else deadline
        start = cands[r][2]
        profile = 'strong' if r == 0 else 'medium'
        if r > 0 and cands[r][0] > best_c + 0.012:
            edge = min(edge, time.time() + 35.0)
        h2, c2 = do_slsqp_sequence(start, n, edge, profile)
        if c2 < best_c:
            best_h, best_c = h2.copy(), c2
        if best_c <= 0.3808 + 1e-9:
            break

    if time.time() < deadline - 80 and n < 720:
        n2 = 560 if n < 560 else 720
        oldx = np.linspace(0.0, 2.0, n)
        newx = np.linspace(0.0, 2.0, n2)
        h2 = project_feasible(np.interp(newx, oldx, best_h), n2)
        c2 = c5_value(h2, n2)
        if c2 < best_c + 0.01:
            h2, c2 = do_slsqp_sequence(h2, n2, deadline - 45, 'medium')
            if c2 < best_c:
                best_h, best_c, n = h2.copy(), c2, n2

    if time.time() < deadline - 14:
        h2, c2 = do_slsqp_sequence(best_h, n, deadline - 6, 'final')
        if c2 < best_c:
            best_h, best_c = h2.copy(), c2

    best_h = project_feasible(best_h, n)
    best_c = c5_value(best_h, n)
    return best_h, float(best_c), int(n)
