**Approach: Frank-Wolfe (conditional gradient) method**

The problem is: min max_k f_k(h) where f_k(h) = sum_i h[i](1-h[i+k]) * dx

This is equivalent to: min t s.t. f_k(h) ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2

The Frank-Wolfe method would:
1. Compute a subgradient g of the objective at current h
2. Solve the linear minimization: min g^T y s.t. 0 ≤ y ≤ 1, sum(y) = n/2
3. Take a step toward y

The linear minimization is easy: y_i = 0 or 1 depending on sign of g_i (with adjustment for the sum constraint).

This is very efficient! Each iteration is O(n²) for the gradient and O(n log n) for the linear step.

def frank_wolfe(h, n, maxiter, gamma):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for it in range(maxiter):
        # Compute objective value and subgradient
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        
        # Subgradient: use top-k averaging
        top_k = min(10, 2*n-1)
        idx = np.argsort(c)[-top_k:]
        
        grad = np.zeros(n)
        for m_idx in idx:
            k = m_idx - (n - 1)
            j1s = max(0, -k)
            j1e = min(n, n - k)
            if j1e > j1s:
                grad[j1s:j1e] += (1.0 - h[j1s+k:j1e+k])
            j2s = max(0, k)
            j2e = min(n, n + k)
            if j2e > j2s:
                grad[j2s:j2e] -= h[j2s-k:j2e-k]
        grad /= top_k
        
        # Linear minimization: min grad^T y s.t. 0<=y<=1, sum(y)=n/2
        # Sort by gradient, set top n/2 to 1, rest to 0
        sorted_idx = np.argsort(grad)
        y = np.zeros(n)
        y[sorted_idx[:n//2]] = 1.0
        
        # Step
        step = 2.0 / (2.0 + it)  # or gamma / (1 + gamma*it)
        h_new = (1.0 - step) * h + step * y
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
    
    return best_h, best_c5

        if time.time() > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_top = min(10, 2 * n - 1)
        idx = np.argsort(c)[-k_top:]
        grad = np.zeros(n)
        for m_idx in idx:
            k = m_idx - (n - 1)
            j1s = max(0, -k)
            j1e = min(n, n - k)
            if j1e > j1s:
                grad[j1s:j1e] += (1.0 - h[j1s + k:j1e + k])
            j2s = max(0, k)
            j2e = min(n, n + k)
            if j2e > j2s:
                grad[j2s:j2e] -= h[j2s - k:j2e - k]
        grad /= k_top
        lr = lr0 / (1.0 + 0.02 * it)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

def make_h(n, kind):
    x = np.linspace(0, 2, n)
    if kind == 0:
        h = np.sin(np.pi * x / 2.0) ** 2
    elif kind == 1:
        h = 0.75 * (1.0 - (x / 2.0) ** 2)
    elif kind == 2:
        h = 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 3:
        h = np.ones(n) * 0.5
    else:
        h = np.exp(-2.0 * (x - 1.0) ** 2)
    return project_feasible(h, n)

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + sm / alpha + np.log(np.exp(shifted - sm).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = make_h(n, 0)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    # Phase 1: Smooth optimization from diverse starts
    phase1_dl = t0 + 0.25 * total
    starts = [h.copy(), make_h(n, 0), make_h(n, 1), make_h(n, 2), make_h(n, 4)]
    for _ in range(3):
        h_r = 0.2 + 0.5 * rng.rand(n)
        starts.append(project_feasible(h_r, n))
    for h_s in starts:
        if time.time() > phase1_dl - 2:
            break
        h_c, c5_c = smooth_opt(h_s, n, 500, 2000, 0.004, phase1_dl - 2)
        if c5_c < best_c5:
            best_h = h_c.copy()
            best_c5 = c5_c

    # Phase 2: LP optimization with restarts
    phase2_dl = t0 + 0.65 * total
    h = best_h.copy()
    eps = 0.05
    no_imp = 0
    while time.time() < phase2_dl - 5:
        h_new, mc = lp_step_fast(h, n, eps)
        c5_new = mc * 2.0 / n
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 8:
                eps *= 0.6
                if eps < 0.0008:
                    h = best_h + 0.02 * rng.randn(n)
                    h = project_feasible(h, n)
                    eps = 0.03
                    no_imp = 0
                else:
                    no_imp = 0
        h = h_new

    # Phase 3: Multi-resolution exploration
    phase3_dl = t0 + 0.87 * total
    for nh in [250, 500, 600, 350]:
        if time.time() > phase3_dl - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        if compute_c5(hh, nh) > best_c5 + 0.005:
            continue
        eps_r = 0.04
        for _ in range(300):
            if time.time() > phase3_dl - 10:
                break
            hh, mc = lp_step_fast(hh, nh, eps_r)
            c5_new = mc * 2.0 / nh
            if c5_new < best_c5 - 1e-12:
                best_h = hh.copy()
                best_c5 = c5_new
                n = nh
            eps_r = max(eps_r * 0.997, 0.002)

    # Phase 4: SLSQP polish
    if time.time() < dl - 8:
        h = best_h.copy()
        for alpha in [5000, 50000, 500000]:
            if time.time() > dl - 8:
                break
            h = slsqp_polish(h, n, alpha, 2000)
            c5_new = compute_c5(h, n)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new

    # Phase 5: Final fine LP
    if time.time() < dl - 3:
        h = best_h.copy()
        eps_f = 0.003
        for _ in range(400):
            if time.time() > dl - 3:
                break
            h, mc = lp_step_fast(h, n, eps_f, n_active=200)
            c5_new = mc * 2.0 / n
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            eps_f = max(eps_f * 0.999, 0.0002)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
