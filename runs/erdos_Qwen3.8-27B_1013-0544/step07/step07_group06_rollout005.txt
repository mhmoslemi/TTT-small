```python
    valid2 = (i - s >= 0) & (i - s < n)
    term1[valid2] += h[i - s[valid2]]
    
    # Term from h[j] going 0->1:
    # adds h[j]*(1-h[j+s]) = 1-h[j+s] where valid
    # removes h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s] where valid (since 1-h[j] was 1, now 0)
    term2 = np.zeros(2*n-1)
    valid3 = (j + s >= 0) & (j + s < n)
    term2[valid3] = (1.0 - h[j + s[valid3]])
    valid4 = (j - s >= 0) & (j - s < n)
    term2[valid4] -= h[j - s[valid4]]
    
    return term1 + term2
```

This is O(n) per swap evaluation. For 40,000 swaps per iteration, that's 40,000 * 400 = 16M ops. In numpy, each swap evaluation involves several array operations on 799-element arrays, so it's probably 10-20 microseconds each. 40,000 * 15μs = 600ms per iteration. That's too slow for many iterations.

Better approach: Vectorize over all swaps at once.

For a fixed i (1→0), the effect on all j (0→1) can be computed as:
effect[i, j] = term1_i + term2_j

where term1_i is the effect of h[i] going 1→0 (independent of j), and term2_j is the effect of h[j] going 0→1 (independent of i).

So:
- Precompute term1 for all i (shape: n_ones × (2n-1))
- Precompute term2 for all j (shape: n_zeros × (2n-1))
- For each pair (i,j): new_corr = corr + term1[i] + term2[j]
- Find the pair that minimizes max(new_corr)

The memory for term1 and term2: 200 × 799 × 8 = 1.28MB each. Fine.

Computing all pairs: we can do this as:
For each i (200 of them):
    base = corr + term1[i]  # shape (799,)
    candidates = base[None, :] + term2  # shape (200, 799)
    maxes = candidates.max(axis=1)  # shape (200,)
    best_j = argmin(maxes)
    track global best

This is 200 × (200 × 799) = 32M operations. In numpy, this should take ~50-100ms total. 

But after each swap, we need to update term1 and term2 for the affected positions. When h[i] changes from 1 to 0 and h[j] from 0 to 1:
- term1[i] is no longer needed (i is no longer a 1)
- term2[j] is no longer needed (j is no longer a 0)
- term1[j] needs to be computed (j is now a 1)
- term2[i] needs to be computed (i is now a 0)
- All other term1[i'] and term2[j'] that reference positions i or j need updating

How many term1 entries reference position i? term1[p] references h[p+s] and h[p-s]. It references h[i] when p+s=i or p-s=i, i.e., p = i-s or p = i+s for various s. So O(n) positions p have term1[p] that references h[i]. Similarly for term2.

So after each swap, we need to update O(n) term1 entries and O(n) term2 entries, each of length O(n). Total: O(n²) = 160K ops per update. Over 10,000 iterations: 1.6B ops. In numpy, each update involves indexing into arrays, so probably 1-2 μs per entry × 400 entries × 2 = ~2ms per iteration. 10,000 iterations × 2ms = 20s. That's manageable.

But actually, the O(n) update per swap is the key cost. Let me think about whether we can avoid it.

Alternative: Don't maintain the term arrays incrementally. Instead, recompute them from scratch after each swap. Computing all term1 and term2 from scratch is O(n²) = 160K ops. Same as the incremental update. So no difference.

```python
def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]

    target = 0.5 * n
    lo, hi = -2.0, 3.0

        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):

        if abs(diff) < 1e-14:

        if diff > 0:

            s = room.sum()
            if s <= 1e-15: break

        else:
            s = h.sum()
            if s <= 1e-15: break

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

    target = n // 2
    hb = np.zeros(n)
    idx = np.argsort(-h)[:target]
    hb[idx] = 1.0
    return hb

def compute_all_terms(h, n):
    """Compute term1 (effect of 1->0) and term2 (effect of 0->1) for all positions."""
    ncorr = 2 * n - 1
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # term1[i, k]: change in corr[k] when h[i] goes 1->0
    # = -(1-h[i+s_k]) for valid i+s_k, + h[i-s_k] for valid i-s_k
    # We compute for all i at once
    # For term1: for each i, valid positions where i+s is in [0,n-1]
    # i+s >= 0 means s >= -i, i+s < n means s < n-i
    # i-s >= 0 means s <= i, i-s < n means s > i-n
    
    # Let's use broadcasting
    # i ranges 0..n-1, s ranges -(n-1)..n-1
    # i+s: shape (n, 2n-1)
    ii = np.arange(n)[:, None]  # (n, 1)
    ss = s[None, :]  # (1, 2n-1)
    is_val = ii + ss  # (n, 2n-1)
    is_valid = (is_val >= 0) & (is_val < n)
    # h[i+s] where valid: need to gather
    # Create index array, clip to [0, n-1] for safety
    is_idx = np.clip(is_val, 0, n-1)
    h_is = h[is_idx]  # (n, 2n-1)
    
    is_val2 = ii - ss  # (n, 2n-1)
    is_valid2 = (is_val2 >= 0) & (is_val2 < n)
    is_idx2 = np.clip(is_val2, 0, n-1)
    h_is2 = h[is_idx2]  # (n, 2n-1)
    
    term1 = np.zeros((n, ncorr))
    term1 = np.where(is_valid, -(1.0 - h_is), 0.0) + np.where(is_valid2, h_is2, 0.0)
    
    # term2[j, k]: change in corr[k] when h[j] goes 0->1

    term2 = np.zeros((n, ncorr))
    term2 = np.where(is_valid, (1.0 - h_is), 0.0) + np.where(is_valid2, -h_is2, 0.0)
    
    return term1, term2

def binary_local_search(h, n, deadline, max_iter=20000, sa=False, T0=5.0, cooling=0.999):
    """Strong local search on binary configurations."""
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = float(corr.max())
    bestM = M
    besth = h.copy()
    
    ones = np.nonzero(h == 1)[0]
    zeros = np.nonzero(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return besth, bestM * (2.0 / float(n))
    
    T = T0
    ncorr = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute all terms
        term1, term2 = compute_all_terms(h, n)
        
        # For each one-position i, find best zero-position j
        # new_corr[i,j] = corr + term1[i] + term2[j]
        # We want min over (i,j) of max(new_corr)
        
        best_max = M
        best_i = -1
        best_j = -1
        best_new_corr = None
        
        t1 = term1[ones]  # (n_ones, ncorr)
        t2 = term2[zeros]  # (n_zeros, ncorr)
        
        for ii in range(len(ones)):
            base = corr + t1[ii]  # (ncorr,)
            candidates = base[None, :] + t2  # (n_zeros, ncorr)
            maxes = candidates.max(axis=1)  # (n_zeros,)
            jk = int(np.argmin(maxes))
            if maxes[jk] < best_max - 1e-12:
                best_max = float(maxes[jk])
                best_i = int(ones[ii])
                best_j = int(zeros[jk])
                best_new_corr = candidates[jk].copy()
        
        if best_i >= 0:
            # Apply swap
            h[best_i] = 0
            h[best_j] = 1
            corr = best_new_corr
            M = best_max
            if M < bestM:
                bestM = M
                besth = h.copy()
            
            # Update ones and zeros
            ones = np.sort(np.delete(ones, np.where(ones == best_i)[0][0]))
            ones = np.sort(np.append(ones, best_j))
            zeros = np.sort(np.delete(zeros, np.where(zeros == best_j)[0][0]))
            zeros = np.sort(np.append(zeros, best_i))
        else:
            # No improvement found
            if sa:
                # Simulated annealing: accept random move
                T *= cooling
                if T > 0.1:
                    ri = int(np.random.randint(len(ones)))
                    rj = int(np.random.randint(len(zeros)))
                    i = int(ones[ri])
                    j = int(zeros[rj])
                    h[i] = 0
                    h[j] = 1
                    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
                    new_M = float(corr.max())
                    # Update tracking
                    ones = np.sort(np.delete(ones, ri))
                    ones = np.sort(np.append(ones, j))
                    zeros = np.sort(np.delete(zeros, rj))
                    zeros = np.sort(np.append(zeros, i))
                    M = new_M
            else:
                break
    
    return besth, bestM * (2.0 / float(n))
```

```python
def lse_obj_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    p = e / s if s > 0 else np.full_like(c, 1.0 / len(c))
    F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    return float(F), g

def lse_lbfgs(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    F, g = lse_obj_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros(n)
    return F, g
```
