    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    C_scaled = alpha * C * dx
    C_scaled -= C_scaled.max()  # for numerical stability
    w = np.exp(C_scaled)
    W = w.sum()
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)  # physical shift
        # Term 1: (1-h[j+k]) for valid j
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w[m] * (1.0 - h[j_lo + k:j_hi + k])
        # Term 2: -h[j-k] for valid j
        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= w[m] * h[j_lo2 - k:j_hi2 - k]
    
    return grad * dx / W

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_box_hyperplane(y, target):
    """Exact projection onto {x: 0<=x<=1, sum(x)=target}"""
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0, 1)

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    # Compute gradient matrix
    G = np.zeros((2*n-1, n))  # G[m, j] = dC[m]/dh[j]
    for m in range(2*n-1):
        k = m - (n-1)
        j1s, j1e = max(0, -k), min(n, n-k)
        if j1e > j1s:
            G[m, j1s:j1e] = 1.0 - h[j1s+k:j1e+k]
        j2s, j2e = max(0, k), min(n, n+k)
        if j2e > j2s:
            G[m, j2s:j2e] -= h[j2s-k:j2e-k]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for s in [solver, 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=s)
            solved = True
            break
        except:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value

import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_box_hyperplane(y, target):
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            G[m, j1s:j1e] = 1.0 - h[j1s + k:j1e + k]
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            G[m, j2s:j2e] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G @ delta <= t,
        delta >= -eps, delta <= eps,
        delta >= -h, delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for s in ['GLOP', 'CLARABEL', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=s)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h, best_mc = h.copy(), mc
    for lam in [1.0, 0.6, 0.3, 0.1]:
        h_try = project_box_hyperplane(h + lam * d, n / 2.0)
        mc_try = float(np.max(np.correlate(h_try, 1 - h_try, mode='full')))
        if mc_try < best_mc - 1e-14:
            best_mc, best_h = mc_try, h_try
    return best_h, best_mc

def momentum_smooth(h, n, alpha, step_size, momentum, max_iters, time_limit):
    dx = 2.0 / n
    v = np.zeros(n)
    best_h, best_mc = h.copy(), float(np.max(np.correlate(h, 1 - h, mode='full')))
    ss = step_size
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        C = np.correlate(h, 1.0 - h, mode='full')
        mc = float(C.max())
        if mc < best_mc:
            best_mc, best_h = mc, h.copy()
        scaled = alpha * C * dx
        scaled -= scaled.max()
        w = np.exp(scaled)
        W = w.sum()
        if W < 1e-300:
            break
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < W * 1e-12:
                continue
            k = m - (n - 1)
            j1s, j1e = max(0, -k), min(n, n - k)
            if j1e > j1s:
                grad[j1s:j1e] += w[m] * (1.0 - h[j1s + k:j1e + k])
            j2s, j2e = max(0, k), min(n, n + k)
            if j2e > j2s:
                grad[j2s:j2e] -= w[m] * h[j2s - k:j2e - k]
        grad *= dx / W
        v = momentum * v - ss * grad
        h_new = project_box_hyperplane(h + v, n / 2.0)
        mc_new = float(np.max(np.correlate(h_new, 1 - h_new, mode='full')))
        if mc_new <= mc + 1e-12:
            h = h_new
            if mc_new < best_mc:
                best_mc, best_h = mc_new, h.copy()
        else:
            ss *= 0.7
            if ss < 1e-8:
                break
            v = momentum * v
        if it % 200 == 0:
            momentum = min(0.95, momentum * 1.001)
    return best_h, best_mc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total

    best_c5 = best_mc * 2.0 / n

    # Phase 1: LP with adaptive epsilon
    phase1_dl = t0 + 0.28 * total
    eps = 0.04
    no_improve = 0
    while time.time() < phase1_dl:
        h_new, mc_new = lp_step(h, n, eps)
        if mc_new < best_mc - 1e-12:
            best_mc, best_h, best_c5, no_improve = mc_new, h_new.copy(), mc_new * 2.0 / n, 0
        else:
            no_improve += 1
            if no_improve > 8:
                eps *= 0.65
                if eps < 0.0008:
                    break
                no_improve = 0
        h = h_new

    # Phase 2: Momentum smooth optimization with increasing alpha
    phase2_dl = t0 + 0.55 * total
    for restart in range(4):
        if time.time() > phase2_dl:
            break
        if restart == 0:
            h = best_h.copy()
        else:
            h = best_h + 0.015 * rng.randn(n)
            h = project_box_hyperplane(h, n / 2.0)
        alpha = 80.0 + restart * 60
        ss = 0.008
        h, mc = momentum_smooth(h, n, alpha, ss, 0.7, 200000, phase2_dl)
        if mc < best_mc - 1e-12:
            best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n
        # Second pass with higher alpha
        if time.time() < phase2_dl - 20:

            h, mc = momentum_smooth(h, n, alpha2, 0.004, 0.8, 100000, phase2_dl)
            if mc < best_mc - 1e-12:
                best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n

    # Phase 3: LP with diverse noise restarts
    phase3_dl = t0 + 0.78 * total
    for trial in range(25):

        if trial % 3 == 0:

        else:

        for _ in range(150):

            h = h_new
            eps_r = max(eps_r * 0.997, 0.0015)

    # Phase 4: Multi-resolution refinement
    phase4_dl = dl - 60

        hh = project_box_hyperplane(hh, nh / 2.0)

        c5_h = mc_h * 2.0 / nh
        if c5_h < best_c5 * 0.999:
            best_mc, best_h, best_c5, n = mc_h, hh.copy(), c5_h, nh
        # Brief LP polish at new resolution
        hh = best_h.copy()

        steps = 0
        while time.time() < phase4_dl and steps < 60:

            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5 - 1e-12:
                best_mc, best_h, best_c5 = mc_new, hh.copy(), c5_new
            hh = hh
            eps_r = max(eps_r * 0.995, 0.001)
            steps += 1

    # Phase 5: Final momentum polish
    if time.time() < dl - 20:
        h = best_h.copy()
        h, mc = momentum_smooth(h, n, 2000, 0.003, 0.85, 200000, dl - 20)
        if mc < best_mc - 1e-12:
            best_mc, best_h, best_c5 = mc, h.copy(), mc * 2.0 / n

    # Phase 6: Tiny LP steps
    if time.time() < dl - 3:
        h = best_h.copy()
        for _ in range(100):

            h_new, mc_new = lp_step(h, n, 0.002)

    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
