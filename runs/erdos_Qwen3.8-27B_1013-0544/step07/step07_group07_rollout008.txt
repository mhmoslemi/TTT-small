For m<0 (mm=-m): nv=n-mm, grad[mm:] += wdx*(1-h[:n-mm]) and grad[:n-mm] -= wdx*h[mm:]. ✓

Now let me think about performance. The loop over k goes from 0 to 2n-2 (799 iterations for n=400). In each iteration, we do 2 vector operations of length ~n. So total is about 799 * 2 * 400 = 640K floating point operations. With numpy overhead, this might take 5-10ms per gradient evaluation.

For 5000 gradient steps, that's 25-50 seconds. Plus the line search (extra evaluations). Should be well within budget.

Actually, I can optimize by pre-filtering: only process k values where p[k] > threshold. For large alpha, only the top few k values have significant probability, so this can be a big speedup.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exps = np.exp(shifted)
    total = exps.sum()
    p = exps / total
    f = max_c + np.log(total) / alpha
    threshold = p.max() * 1e-10
    active = np.where(p > threshold)[0]
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        wdx = p[k] * dx
        if m >= 0:
            nv = n - m
            if nv > 0:
                grad[:nv] += wdx * (1.0 - h[m:])
                grad[m:] -= wdx * h[:nv]
        else:
            mm = -m
            nv = n - mm
            if nv > 0:
                grad[mm:] += wdx * (1.0 - h[:nv])
                grad[:nv] -= wdx * h[mm:]
    return f, grad

def grad_descent_phase(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 80.0
    step = 20.0
    stall = 0
    max_stall = 400
    while time.time() < deadline:
        f_val, grad = lse_obj_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = min(step, 50.0 / (gn + 1e-10))
        improved = False
        c_cur = compute_c5(h, n)
        for _ in range(25):
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_cur - 1e-15:
                h = h_try
                improved = True
                break
            step *= 0.6
        if not improved:
            stall += 1
            if stall > max_stall:
                h = project_feasible(h + 0.02 * rng.randn(n), n)
                step = 20.0
                alpha = max(50.0, alpha * 0.5)
                stall = 0
                continue
        else:
            step = min(step * 1.15, 200.0)
            stall = 0
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
        if (stall + (0 if improved else 1)) % 250 == 0:
            alpha = min(alpha * 1.8, 5e5)
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + sm / alpha + np.log(np.exp(shifted - sm).sum()) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pn_slsqp(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c_pos = np.maximum(c, 1e-14)
        return float(np.sum(c_pos ** p) ** (1.0 / p))
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                i_val = i
                delta = np.zeros(2 * n - 1)
                im = i_val + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i_val - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - i_val)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i_val - best_j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                h[i_val], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i_val)
                ones.append(i_val)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def barrier_slsqp(h, n, t, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0)
        return float(np.sum(excess ** 2))
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
