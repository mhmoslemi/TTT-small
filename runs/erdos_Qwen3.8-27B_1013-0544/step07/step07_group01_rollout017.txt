import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    H = np.fft.fft(h)
    A = np.real(np.fft.ifft(H * np.conj(H)))
    return (n / 2.0 - A.min()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def bottleneck_ascent(h, n, deadline, alpha=200.0, lr=None, mu=0.9):
    if lr is None:
        lr = 1.0 / np.sqrt(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    stall = 0
    for step in range(1000000):
        if time.time() > deadline:
            break
        H = np.fft.fft(h)
        A = np.real(np.fft.ifft(H * np.conj(H)))
        min_A = A.min()
        # Smoothed bottleneck weights (softmax on negative A)
        shifted = -alpha * (A - min_A)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(n)
            weights[np.argmin(A)] = 1.0
        else:
            weights /= ws
        # Gradient via FFT
        W = np.fft.fft(weights)
        grad = np.real(np.fft.ifft(np.conj(W) * H + W * H))
        momentum = mu * momentum + lr * grad
        h = project_feasible(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.9
                if lr < 0.001:
                    lr = 0.5 / np.sqrt(n)
                    mu = min(mu + 0.01, 0.99)
                    stall = 0
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        H = np.fft.fft(x)
        A = np.real(np.fft.ifft(H * np.conj(H)))
        overlap = (n / 2.0 - A) * dx
        m = overlap.max()
        shifted = alpha * (overlap - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
