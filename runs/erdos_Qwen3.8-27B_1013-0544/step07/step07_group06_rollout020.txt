def precompute PN(h, n):
    """Precompute P[a] and N[a] for all positions a."""
    # P[a][d_idx] = 1 - h[a+d] if 0 <= a+d < n, else 0
    # N[a][d_idx] = h[a-d] if 0 <= a-d < n, else 0

    N = np.zeros((n, 2*n-1))

        am = a - l  # a - d
        valid_m = (am >= 0) & (am < n)
        N[a, valid_m] = h[am[valid_m]]
    return P, N

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):

    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)

            if s <= 1e-15: break

def lse_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())

        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]

    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)

    g += pen * dev if pen else 0.0

    return F, g

import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(np.min(a)) - 1.0, float(np.max(a)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0
    return hb

def lse_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, 1.0 - h, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    if pen:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def polish(h, n, betas, pen, deadline, stage_t):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    for beta in betas:
        if time.time() > deadline - 5:
            break
        d = min(deadline - 3, time.time() + stage_t)
        _stop_time = d
        try:
            r = minimize(lse_obj, work, method="L-BFGS-B", jac=True,
                         bounds=[(0.0, 1.0)] * n, args=(n, beta, pen),
                         callback=stop_cb,
                         options={"maxiter": 10**7, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9})
            work = project_feasible(r.x if r is not None and hasattr(r, "x") else work, n)
        except Exception:
            pass
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
        for v in [work[::-1], 1.0 - work[::-1]]:
            cv = project_feasible(v, n)
            cvv = compute_c5(cv, n)
            if cvv < bc:
                bc, best = cvv, cv.copy()
        if bc < cw - 1e-12:
            work = best.copy()
    return best, bc
