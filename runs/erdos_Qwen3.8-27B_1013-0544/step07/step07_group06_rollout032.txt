def binary_sa(h, n, deadline, n_iter = 1000000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = int(np.max(corr))
    best_h = h.copy()
    best_M = M
    
    T = 5.0  # initial temperature
    cooling = 0.9999
    
    for step in range(n_iter):
        if time.time() > deadline:
            break
        if step % 10000 == 0 and T < 0.1:
            break
            
        # Choose random zero and one positions
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]
        
        # Compute Δc_k for this swap
        # Δc_k = [h[i+k]==0] - [h[j+k]==0] - [h[i-k]==1] + [h[j-k]==1]
        dc = compute_delta(h, i, j, n)
        new_corr = corr + dc
        new_M = int(np.max(new_corr))
        
        if new_M <= M:
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            M = new_M
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            # Maybe accept (simulated annealing)
            if np.random.rand() < np.exp(-(new_M - M) / T):
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                M = new_M
        T *= cooling
    
    return best_h, best_M * dx
