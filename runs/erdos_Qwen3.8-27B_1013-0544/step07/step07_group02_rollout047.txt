        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, active_frac=0.0, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    
    if active_frac > 0:
        n_corr = 2 * n - 1
        threshold = C.min() + (1 - active_frac) * (mc - C.min())
        active_idx = np.where(C >= threshold)[0]
        if active_idx.size < 10:
            active_idx = np.argsort(C)[-10:]
        C_use = C[active_idx]
        G = np.zeros((n, active_idx.size))
        for ii, m in enumerate(active_idx):
            k = m - (n - 1)
            j1s = max(0, -k)
            j1e = min(n, n - k)
            if j1e > j1s:
                G[j1s:j1e, ii] = 1.0 - h[j1s+k:j1e+k]
            j2s = max(0, k)
            j2e = min(n, n + k)
            if j2e > j2s:
                G[j2s:j2e, ii] -= h[j2s-k:j2e-k]
    else:
        C_use = C
        G = compute_gradients(h, n)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C_use + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    
    d = delta.value.copy()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.75, 0.5, 0.35, 0.2, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_polish(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def optimize_lp(h, n, t_end, rng, eps_init=0.05, use_active=False):
    h = project_feasible(h, n)
    mc = float(np.correlate(h, 1.0 - h, mode='full').max())
    best_h = h.copy()
    best_mc = mc
    eps = eps_init
    no_impr = 0

            best_h = h_new.copy()

            if no_impr > 15:

    return best_h, best_mc
