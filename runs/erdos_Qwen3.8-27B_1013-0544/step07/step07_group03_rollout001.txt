def project_feasible(h, n):

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0

        if t_true < best_c:
            best_c = t_true
            best_h = h_new.copy()
        t_lp = float(res.x[n])
        violated = []
        top_new = np.argsort(corr_new)[::-1][:25]
        for kidx in top_new:
            if corr_new[kidx] > t_lp + 1e-9:
                k = int(kidx) - (n - 1)
                g_new = compute_grad_lag(k, h_new, n, dx)
                rhs_new = g_new @ h_new - corr_new[kidx]
                violated.append((g_new, rhs_new))
        if violated:
            cuts.extend(violated)
            if len(cuts) > 250:
                cuts, _ = build_cuts(h_new)
        h = h_new
        if it % 300 == 299:
            cuts, _ = build_cuts(h)
        if best_c < prev_best - 1e-8:
            prev_best = best_c
    return best_h, best_c

def binary_search_fast(h_b, n, deadline, max_sweeps=1000, top_lags=40):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    m_range = np.arange(-(n - 1), n)
    h_b = h_b.copy()
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_cm = corr.max()
    best_h = h_b.copy()
    best_c = cur_cm
    ones = list(np.where(h_b == 1.0)[0])
    zeros = list(np.where(h_b == 0.0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argsort(corr)[::-1][:top_lags]
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            di = np.zeros(ncorr)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            di[msk] += 1.0 - h_b[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            di[msk] -= h_b[imn[msk]]
            best_j = -1
            best_nc = cur_cm - 1e-15
            for j in ones:
                dj = np.zeros(ncorr)
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                dj[msk] -= 1.0 - h_b[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                dj[msk] += h_b[jmn[msk]]
                dj[n - 1] -= 2.0
                dj[n - 1 + (j - i)] += 1.0
                dj[n - 1 + (i - j)] += 1.0
                combined = corr + di + dj
                nc = combined[top_idx].max()
                if nc < cur_cm - 1e-12:
                    nc = combined.max()
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                dj = np.zeros(ncorr)
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                dj[msk] -= 1.0 - h_b[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                dj[msk] += h_b[jmn[msk]]
                dj[n - 1] -= 2.0
                dj[n - 1 + (best_j - i)] += 1.0
                dj[n - 1 + (i - best_j)] += 1.0
                h_b[i] = 1.0
                h_b[best_j] = 0.0
                corr += di + dj
                cur_cm = best_nc
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                top_idx = np.argsort(corr)[::-1][:top_lags]
                if cur_cm < best_c:
                    best_c = cur_cm
                    best_h = h_b.copy()
        if not improved:
            break
    return best_h, float(best_c * dx)

def subgrad_optimize(h, n, deadline, max_iters=30000, top_k=3, lr_base=0.08):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    G = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            grad += compute_grad_lag(k, h, n, 1.0)
        grad = grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_check.max() < cur_c + 0.003:
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha (20% time)
    p1 = t0 + 0.20 * total
    alphas = [500, 2000, 10000, 100000, 500000, 5e6, 5e7]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Cutting plane (30% time)
    p2 = t0 + 0.50 * total
    hc, cc = cutting_plane(best_h, n, p2, max_iters=5000, n_cuts=120)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: Binary local search (30% time)
    p3 = t0 + 0.80 * total
    h_bin = to_binary(best_h, n)
    hb, cb = binary_search_fast(h_bin, n, p3, max_sweeps=2000, top_lags=40)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: Cut another round on best (10% time)
    p4 = t0 + 0.90 * total
    hc2, cc2 = cutting_plane(best_h, n, p4, max_iters=2000, n_cuts=80)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 5: Final polish
    if time.time() < dl - 15:
        h = best_h.copy()
        for alpha in [5e6, 5e7, 5e8]:
            if time.time() > dl - 8:
                break
            h = slsqp_optimize(h, n, alpha, 1500, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        hs, cs = subgrad_optimize(best_h, n, dl - 2, max_iters=10000, top_k=3, lr_base=0.05)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()
        # Final binary check
        if time.time() < dl - 5:
            h_bin2 = to_binary(best_h, n)
            hb2, cb2 = binary_search_fast(h_bin2, n, dl - 1, max_sweeps=500, top_lags=30)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
