            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    idx = np.argsort(-h)[:target]
    hb[idx] = 1.0
    return hb

def lse_objective_grad(h, n, beta, pen):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if pen != 0.0:
        dev = float(h.sum() - 0.5 * n)
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def compute_base_binary(h, n):
    """Precompute the base correlation change matrix for binary swaps."""
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def binary_search(h, n, deadline, method='greedy', max_sweeps=50, chunk=2048, rng=None):
    """Binary local search using pair swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    
    h = top_n_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        base = compute_base_binary(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        
        if method == 'greedy':
            for start in range(0, zi.size, chunk):
                if time.time() > deadline:
                    break
                end = min(start + chunk, zi.size)
                i = zi[start:end]
                j = oj[start:end]
                arr = base[i] - base[j]
                arr += corr[None, :]
                arr[:, n - 1] -= 2.0
                d1 = n - 1 + (j - i)
                arr[:, d1] += 1.0
                d2 = n - 1 + (i - j)
                arr[:, d2] += 1.0
                mx = arr.max(axis=1)
                idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
                if idx.size:
                    loc = idx[int(np.argmin(mx[idx]))]
                    iv = int(i[loc])
                    jv = int(j[loc])
                    h[iv] = 1.0
                    h[jv] = 0.0
                    corr = arr[loc].astype(float, copy=True)
                    M = int(round(float(corr.max())))
                    if M < bestM:
                        bestM = M
                        besth = h.copy()
                    improved = True
                    break
        elif method == 'anneal':
            # Allow swaps that increase M by at most 1 with some probability
            for start in range(0, zi.size, chunk):
                if time.time() > deadline:
                    break
                end = min(start + chunk, zi.size)
                i = zi[start:end]
                j = oj[start:end]
                arr = base[i] - base[j]
                arr += corr[None, :]
                arr[:, n - 1] -= 2.0
                d1 = n - 1 + (j - i)
                arr[:, d1] += 1.0
                d2 = n - 1 + (i - j)
                arr[:, d2] += 1.0
                mx = arr.max(axis=1)
                # Accept if mx <= M (improving or neutral) with probability 1
                # Accept if mx == M+1 with probability 0.3
                good = np.nonzero(mx <= M + 1e-12)[0]
                neutral_or_worse = np.nonzero((mx > M + 1e-12) & (mx <= M + 1 + 1e-12))[0]
                
                if good.size > 0:
                    loc = int(np.argmin(mx[good]))
                    loc = good[loc]
                elif neutral_or_worse.size > 0 and rng.rand() < 0.3:
                    loc = neutral_or_worse[int(rng.randint(neutral_or_worse.size))]
                else:
                    continue
                
                iv = int(i[loc])
                jv = int(j[loc])
                h[iv] = 1.0
                h[jv] = 0.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(float(corr.max())))
                if M < bestM:
                    bestM = M
                    besth = h.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return besth, bestM * (2.0 / float(n))

def polish_continuous(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.4 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        options = {"maxiter": 200000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9}
        try:
            res = minimize(
                lse_objective_grad, work, method="L-BFGS-B", jac=True,

                callback=stop_callback, options=options

            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

Wait, I need to be more careful. The correlation c[k] = sum_i h[i](1-h[i+k]) is NOT the same as the autocorrelation. The autocorrelation is sum_i h[i]h[i+k]. For binary h with sum = n/2:

sum_i h[i](1-h[i+k]) = sum_i h[i] - sum_i h[i]h[i+k]

But this is for circular correlation (where the sum is over all i mod n). For aperiodic (linear) correlation, the sums have different ranges.

For the full (aperiodic) correlation:
c[k] for k=0,...,n-1: sum_{i=0}^{n-1-k} h[i](1-h[i+k])
c[k] for k=-(n-1),...,-1: sum_{i=-k}^{n-1} h[i](1-h[i+k]) = sum_{i=0}^{n-1+k} h[i+k](1-h[i]) by substitution

So c[k] (for k >= 0) = (number of i in [0, n-1-k] with h[i]=1) - (number of i in [0, n-1-k] with h[i]=1 and h[i+k]=1)

The first term depends on the position of the window, and the second is a partial autocorrelation.

This makes the problem more complex than just minimizing autocorrelation. The "edge effects" matter.

                    loc = neutral[int(rng.randint(neutral.size))]
                elif one_up.size > 0 and rng.rand() < anneal_prob:
                    loc = one_up[int(rng.randint(one_up.size))]
                else:
                    continue
            else:
                idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
                if idx.size == 0:
                    continue
                loc = idx[int(np.argmin(mx[idx]))]
            iv = int(i[loc])
            jv = int(j[loc])
            h[iv] = 1.0
            h[jv] = 0.0
            corr = arr[loc].astype(float, copy=True)
            M = int(round(float(corr.max())))
            if M < bestM:
                bestM = M
                besth = h.copy()
            improved = True
            break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

def polish_continuous(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 5.0:
            break
        stage_dl = min(deadline - 3.0, now + max(2.0, min(stage_time, 0.4 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        options = {"maxiter": 300000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9}
        try:
            res = minimize(
                lse_objective_grad, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback, options=options
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    end = t0 + max(30.0, total_budget - 10.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad initial")
    except Exception:
        n0 = 400
        h0 = ((np.arange(n0) // 50) % 2).astype(float)

    best_n = n0
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, best_n)

    # Phase 1: Continuous optimization (~35% budget)
    p1_end = t0 + 0.35 * (total_budget - 10.0)
    if time.time() < p1_end:
        hh, cc = polish_continuous(best_h, best_n,
                                   [30.0, 100.0, 400.0, 1500.0, 6000.0, 25000.0],
                                   1e4, p1_end - 5.0, 20.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    # Phase 2: Binary greedy from continuous solution (~25% budget)
    p2_end = t0 + 0.60 * (total_budget - 10.0)
    if best_n % 2 == 0 and time.time() < p2_end:
        hb, cb = binary_local_search(best_h, best_n, p2_end - 5.0,
                                     max_sweeps=60, chunk=4096, rng=rng)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Binary anneal to escape local optima (~15% budget)
    p3_end = t0 + 0.75 * (total_budget - 10.0)
    if best_n % 2 == 0 and time.time() < p3_end:
        hb2, cb2 = binary_local_search(best_h, best_n, p3_end - 5.0,
                                       max_sweeps=80, chunk=4096, rng=rng,
                                       anneal_prob=0.4)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        # Greedy cleanup after anneal
        if time.time() < p3_end + 0.05 * (total_budget - 10.0):
            hb3, cb3 = binary_local_search(best_h, best_n,
                                           p3_end + 0.05 * (total_budget - 10.0) - 3.0,
                                           max_sweeps=30, chunk=4096, rng=rng)
            if cb3 < best_c:
                best_c = cb3
                best_h = hb3.copy()

    # Phase 4: Try alternative resolutions (~10% budget)
    p4_end = t0 + 0.90 * (total_budget - 10.0)
    if time.time() < p4_end:
        for n_new in (300, 200):
            if time.time() > p4_end - 15.0:
                break
            x0 = np.linspace(0.0, 2.0, best_n)
            x1 = np.linspace(0.0, 2.0, n_new)
            hr = np.interp(x1, x0, best_h)
            hr = project_feasible(hr, n_new)
            hh, cc = polish_continuous(hr, n_new, [500.0, 3000.0, 10000.0],
                                       1e4, p4_end - 15.0, 15.0)
            if cc < best_c:
                best_c = cc
                best_h = hh.copy()
                best_n = n_new
            if n_new % 2 == 0 and time.time() < p4_end - 5.0:
                hb, cb = binary_local_search(hr, n_new, p4_end - 5.0,
                                             max_sweeps=20, chunk=2048, rng=rng)
                if cb < best_c:
                    best_c = cb
                    best_h = hb.copy()
                    best_n = n_new

    # Phase 5: Final polish (~5% budget)
    if time.time() < end - 10.0:
        hh, cc = polish_continuous(best_h, best_n, [3000.0, 10000.0, 40000.0],
                                   1e4, end - 8.0, 10.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    if best_n % 2 == 0 and time.time() < end - 3.0:
        hb, cb = binary_local_search(best_h, best_n, end - 2.0,
                                     max_sweeps=15, chunk=4096, rng=rng)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
