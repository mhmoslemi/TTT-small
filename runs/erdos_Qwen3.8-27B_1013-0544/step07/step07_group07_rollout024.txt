import numpy as np
import time
from scipy.optimize import minimize

G_N = 0
G_DX = 0.0
G_ALPHA = 1.0
G_T = 0.0
G_BETA = 1.0
G_LAM = 1.0
G_SCALE = 1.0
G_ONES = None
G_BOUNDS = None
G_LAST_X = None
G_LAST_P = None

def set_globals(n):
    global G_N, G_DX, G_ALPHA, G_T, G_BETA, G_LAM, G_SCALE, G_ONES, G_BOUNDS
    G_N = int(n)
    G_DX = 2.0 / G_N
    G_ALPHA = 1.0
    G_T = 0.0
    G_BETA = 1.0
    G_LAM = 1.0
    G_SCALE = 1.0 / G_N
    G_ONES = np.ones(G_N, dtype=float)
    G_BOUNDS = [(0.0, 1.0)] * G_N

def correlate_c(x):
    return np.correlate(x, 1.0 - x, mode='full') * G_DX

def lse_value(c, alpha):
    z = alpha * c
    zm = float(np.max(z))
    e = np.exp(z - zm)
    return float((zm + np.log(float(np.sum(e)))) / alpha)

def softmax_weights(c, alpha):
    z = alpha * c
    zm = float(np.max(z))
    e = np.exp(z - zm)
    return e / float(np.sum(e))

def weight_grad(x, w):
    m = G_N - 1
    q = w + w[::-1]
    prod = np.convolve(x, q, mode='full')[m:m + G_N]
    lin = np.convolve(w, G_ONES, mode='valid')
    return lin - prod

def shift_sum(h, lam):
    return float(np.clip(h + lam, 0.0, 1.0).sum())

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-12:
        return h
    lo = -float(h.max()); hi = 1.0 - float(h.min())
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if shift_sum(h, mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

    c = correlate_c(x)
    p = softmax_weights(c, G_ALPHA)

def jac_lse(x):

    if G_LAST_X is not None and G_LAST_P is not None and len(G_LAST_X) == len(x):
        if np.array_equal(G_LAST_X, x):
            return weight_grad(x, G_LAST_P)
    c = correlate_c(x)

def objective_barrier(x):
    c = correlate_c(x)
    z = G_BETA * (c - G_T)
    sp = np.empty_like(z)
    pos = z > 0.0
    sp[pos] = z[pos] + np.log1p(np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    sp[~pos] = np.log1p(ez)
    return float(np.sum(sp)) * G_SCALE

def jac_barrier(x):
    c = correlate_c(x)

import numpy as np
import time
from scipy.optimize import minimize

G_N = 0
G_DX = 0.0
G_ALPHA = 1.0
G_T = 0.0
G_BETA = 1.0
G_SCALE = 1.0
G_ONES = None
G_BOUNDS = None
G_LAST_X = None
G_LAST_P = None


def set_globals(n):
    global G_N, G_DX, G_SCALE, G_ONES, G_BOUNDS
    G_N = int(n)
    G_DX = 2.0 / G_N
    G_SCALE = 1.0 / G_N
    G_ONES = np.ones(G_N, dtype=float)
    G_BOUNDS = [(0.0, 1.0)] * G_N


def correlate_c(x):
    return np.correlate(x, 1.0 - x, mode='full') * G_DX


def lse_value(c, alpha):
    z = alpha * c
    zm = float(np.max(z))
    e = np.exp(z - zm)
    return float((zm + np.log(float(np.sum(e)))) / alpha)


def softmax_weights(c, alpha):
    z = alpha * c
    zm = float(np.max(z))
    e = np.exp(z - zm)
    return e / float(np.sum(e))


def weight_grad(x, w):
    m = G_N - 1
    q = w + w[::-1]
    prod = np.convolve(x, q, mode='full')[m:m + G_N]
    lin = np.convolve(w, G_ONES, mode='valid')
    return lin - prod


def shift_sum(h, lam):
    return float(np.clip(h + lam, 0.0, 1.0).sum())


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-12:
        return h
    lo = -float(h.max())
    hi = 1.0 - float(h.min())
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if shift_sum(h, mid) < target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if diff > 1e-15:
        out[int(np.argmin(out))] += diff
    elif diff < -1e-15:
        out[int(np.argmax(out))] += diff
    return np.clip(out, 0.0, 1.0)


def c5_value(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def objective_lse(x):
    global G_LAST_X, G_LAST_P
    c = correlate_c(x)
    p = softmax_weights(c, G_ALPHA)
    val = lse_value(c, G_ALPHA)
    G_LAST_X = x.copy()
    G_LAST_P = p
    return val


def jac_lse(x):
    global G_LAST_X, G_LAST_P
    if G_LAST_X is not None and G_LAST_P is not None and len(G_LAST_X) == len(x):
        if np.array_equal(G_LAST_X, x):
            return weight_grad(x, G_LAST_P)
    c = correlate_c(x)
    p = softmax_weights(c, G_ALPHA)
    G_LAST_X = x.copy()
    G_LAST_P = p
    return weight_grad(x, p)


def objective_barrier(x):
    c = correlate_c(x)
    z = G_BETA * (c - G_T)
    sp = np.empty_like(z)
    pos = z > 0.0
    sp[pos] = z[pos] + np.log1p(np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    sp[~pos] = np.log1p(ez)
    return float(np.sum(sp)) * G_SCALE


def jac_barrier(x):
    c = correlate_c(x)
    z = G_BETA * (c - G_T)
    sig = 0.5 * (1.0 + np.tanh(0.5 * z))
    return G_SCALE * weight_grad(x, sig)


def cons_eq(x):
    return float(np.sum(x) - 0.5 * G_N)


def cons_jac(x):
    return G_ONES.copy()


def slsqp_call(fun, jac, h, maxiter, limit):
    if time.time() >= limit:
        return h
    start = np.asarray(h, dtype=float).copy()
    try:
        res = minimize(
            fun, start, jac=jac, method='SLSQP', bounds=G_BOUNDS,
            constraints={'type': 'eq', 'fun': cons_eq, 'jac': cons_jac},
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'disp': False}
        )
        x = res.x
    except Exception:
        x = None
    if x is None or not np.all(np.isfinite(x)):
        return start
    return project_feasible(x, G_N)


def slsqp_lse(h, alpha, maxiter, limit):
    global G_ALPHA, G_LAST_X, G_LAST_P
    if time.time() >= limit:
        return h
    G_ALPHA = float(alpha)
    G_LAST_X = None
    G_LAST_P = None
    return slsqp_call(objective_lse, jac_lse, h, maxiter, limit)


def barrier_slsqp(h, target, beta, maxiter, limit):
    global G_T, G_BETA
    if time.time() >= limit:
        return h
    G_T = float(target)
    G_BETA = float(beta)
    return slsqp_call(objective_barrier, jac_barrier, h, maxiter, limit)


def barrier_sequence(h, n, c, deadline, steps, beta0):
    global G_N
    h = np.asarray(h, dtype=float).copy()
    stall = 0
    for i in range(int(steps)):

        if c2 < c - 1e-9:
            h = h2
            c = c2
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return h, c


def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=float)
    hb[idx[:n // 2]] = 1.0
    return hb


def binary_gradient_search(h, n, deadline, K, R, alpha):
    global G_N
    h = to_binary(h, n).copy()
    c = correlate_c(h)
    cur_max = float(np.max(c))
    best_h = h.copy()
    best_c = cur_max
    cur_F = lse_value(c, alpha)
    no = 0
    rng = np.random.RandomState(987654321)

    while no < 4 and time.time() < deadline:
        p = softmax_weights(c, alpha)
        g = weight_grad(h, p)
        z = np.flatnonzero(h < 0.5)
        o = np.flatnonzero(h > 0.5)
        if z.size < 1 or o.size < 1:
            break
        Kz = min(int(K), z.size)
        Ko = min(int(K), o.size)
        zi = z[np.argsort(g[z])[:Kz]]
        oj = o[np.argsort(-g[o])[:Ko]]
        pairs = []
        for a in zi:
            for b in oj:
                pairs.append((int(a), int(b)))
        for _ in range(int(R)):
            pairs.append((int(rng.choice(z)), int(rng.choice(o))))

        bkey = None
        bi = -1
        bj = -1
        c2b = None
        for i, j in pairs:
            if i == j:
                continue
            h[i] = 1.0
            h[j] = 0.0
            c2 = correlate_c(h)
            F2 = lse_value(c2, alpha)
            m2 = float(np.max(c2))
            h[i] = 0.0
            h[j] = 1.0
            if F2 < cur_F - 1e-13 or m2 < cur_max - 1e-12:
                key = (F2, m2)
                if bkey is None or key < bkey:
                    bkey = key
                    bi = i
                    bj = j
                    c2b = c2.copy()

        if bi >= 0:
            h[bi] = 1.0
            h[bj] = 0.0
            c = c2b
            cur_F = bkey[0]
            cur_max = bkey[1]
            no = 0
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
        else:
            no += 1
    return best_h, best_c
