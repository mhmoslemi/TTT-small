def project_box_sum(x, target):
    """Project x onto {0<=y<=1, sum(y)=target}."""
    n = len(x)
    # Binary search on lambda
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2, 0, 1)
    # Final adjustment for exact sum
    diff = target - y.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = (y < 1) & (y > 0)
            if mask.any():
                y[mask] += diff / mask.sum()
            else:
                y = y + diff / n
        else:
            mask = (y > 0) & (y < 1)
            if mask.any():
                y[mask] += diff / mask.sum()
    return np.clip(y, 0, 1)

        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: PGD exploration (adaptive K)
    p1_dl = t0 + 0.30 * total
    h_work = h0.copy()
    for k_top in [10, 7, 5, 3]:
        if time.time() > p1_dl:
            break
        h_work, c5_w, h_work = pgd_optimize(h_work, best_n, min(p1_dl, time.time() + total*0.06),
                                             400, k_top, 0.04, 50000)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()
    
    # Phase 2: SLSQP with increasing sharpness
    p2_dl = t0 + 0.50 * total
    best_h, best_c5, h_work = slsqp_polish(h_work, best_n, p2_dl, [800, 4000, 15000])
    
    # Phase 3: Multi-resolution exploration
    p3_dl = t0 + 0.80 * total
    for nh in [250, 500, 750]:
        if time.time() > p3_dl - 30:
            break
        if nh == best_n:
            continue
        xo = np.linspace(0, 2, best_n, endpoint=False) + 1.0 / best_n
        xn = np.linspace(0, 2, nh, endpoint=False) + 1.0 / nh
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        c5_h = compute_c5(hh, nh)
        
        hh, c5_opt, hh_end = pgd_optimize(hh, nh, min(p3_dl - 20, time.time() + 100),
                                           400, 5, 0.03, 50000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = hh.copy()
            best_n = nh
    
    # Phase 4: Noise restarts for escape
    p4_dl = dl - 50
    for trial in range(40):
        if time.time() > p4_dl:
            break
        noise = 0.015 * (1.0 + 0.5 * rng.rand())
        h_try = best_h + noise * rng.randn(best_n)
        h_try = project_feasible(h_try, best_n)
        h_try, c5_try, _ = pgd_optimize(h_try, best_n, min(p4_dl, time.time() + 60),
                                         500, 5, 0.03, 30000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 5: Final SLSQP + PGD polish
    if time.time() < dl - 15:
        best_h, best_c5, h_work = slsqp_polish(best_h, best_n, dl - 15, [8000, 30000])
    if time.time() < dl - 3:
        best_h, best_c5, _ = pgd_optimize(h_work if 'h_work' in dir() else best_h,
                                           best_n, dl - 2, 2000, 2, 0.003, 100000)
    
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
