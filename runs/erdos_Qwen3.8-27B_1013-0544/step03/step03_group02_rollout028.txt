from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_h(h, target_sum):
    """Project onto {x: sum(x)=target_sum, 0<=x<=1} using bisection on lambda"""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx, target_sum, penalty):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    sm = smooth_max(c, beta) * dx
    s = h_in.sum() - target_sum
    return sm + penalty * s * s

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0/1)"""
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def dekorrelate(h, n, target):
    """Try to decorrelate by permuting bits of a binary pattern"""
    # Start from Thue-Morse
    h = thue_morse(n)
    # Adjust to get correct sum

    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    # If too many ones, flip some; if too few, flip some zeros to ones

def project_h(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid

def smooth_max_obj(h, beta, dx, target_sum, penalty):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta
    s = h.sum() - target_sum
    return sm * dx + penalty * s * s

def smooth_max_grad(h, beta, dx, target_sum, penalty):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new),
                     np.linspace(0.0, 2.0, len(h)), h)

def smooth_max_obj_and_grad(h, beta, dx, target_sum, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(xs)
    total = exps.sum()
    weights = exps / total
    sm = m + np.log(total / len(corr)) / beta
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] -= w * h[la:]
    grad *= dx
    
    # Penalty for sum constraint
    s = h.sum() - target_sum
    grad += 2.0 * penalty * s
    obj = sm * dx + penalty * s * s
    return obj, grad
