def binary_local_search(h, max_time, rng):
    """Efficient binary local search with pair flips."""
    n = len(h)
    target = n / 2.0
    h = (h >= 0.5).astype(float)
    # Fix sum
    deficit = int(round(target - h.sum()))
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        for k in range(min(deficit, len(zeros))):
            h[zeros[k]] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        for k in range(min(-deficit, len(ones))):
            h[ones[k]] = 0.0
    
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

    no_improve = 0
    
    while time.time() - t_start < max_time:
        # Pick random pair
        if not ones_list or not zeros_list:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        corr = np.correlate(h, 1.0 - h, mode="full")

        # Δresult[m] = (h[j+lag_m] - h[i+lag_m]) + (h[i-lag_m] - h[j-lag_m])

        n_full = 2 * n - 1

        for m in range(n_full):

            ip = i + lag
            jm = j - lag
            im = i - lag
            d = 0.0

            if 0 <= ip < n:
                d -= h[ip]
            if 0 <= im < n:
                d += h[im]
            if 0 <= jm < n:
                d -= h[jm]
            delta[m] = d

        if new_max < np.max(corr) - 1e-10:

            h[i] = 0.0
            h[j] = 1.0
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            no_improve = 0
            cur_c5 = new_max * (2.0 / n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                break
    
    return best_h, best_c5

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_pass(h, n_pts, target, dx, beta, max_iter, t0, budget_s):
    """One SLSQP optimization pass. Returns improved h or None."""
    if time.time() - t0 > budget_s - 20:
        return None
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n_pts)
    
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        h_new = project_h(res.x, target)
        return h_new
    except Exception:
        return None

def binary_local_search(h_init, max_time_s, seed):
    """Efficient binary local search with pair flips and annealing."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    
    h = (h_init >= 0.5).astype(float)
    # Fix sum to be exactly target
    deficit = int(round(target - h.sum()))
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= deficit:
            idx = np.random.RandomState(seed+1).choice(len(zeros), deficit, replace=False)
            h[zeros[idx]] = 1.0
        else:
            h[zeros] = 1.0
            deficit -= len(zeros)
            if deficit > 0:
                h[:deficit] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        if len(ones) >= -deficit:
            idx = np.random.RandomState(seed+1).choice(len(ones), -deficit, replace=False)
            h[ones[idx]] = 0.0
        else:
            h[ones] = 0.0
            deficit += len(ones)
            if deficit < 0:
                h[deficit:] = 0.0
    
    # Recompute corr
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    cur_max = np.max(corr)
    dx = 2.0 / n
    best_h = h.copy()
    best_max = cur_max
    
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    t_start = time.time()
    no_improve = 0
    total_moves = 0
    
    while time.time() - t_start < max_time_s:
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        
        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]
        
        # Compute delta vector
        jp = j + lags
        ip = i + lags
        jm = j - lags
        im = i - lags
        
        # Safe indexing
        delta = np.zeros(2*n - 1)
        valid = (jp >= 0) & (jp < n)
        delta[valid] += h[jp[valid]]
        valid = (ip >= 0) & (ip < n)
        delta[valid] -= h[ip[valid]]
        valid = (im >= 0) & (im < n)
        delta[valid] += h[im[valid]]
        valid = (jm >= 0) & (jm < n)
        delta[valid] -= h[jm[valid]]
        
        new_max = np.max(corr + delta)
        improvement = cur_max - new_max
        
        # Annealing: accept if improvement > 0 or with probability exp(-|worse|/T)
        elapsed_frac = (time.time() - t_start) / max_time_s
        T = 0.001 * (1.0 - elapsed_frac) + 0.0001
        
        if improvement > 1e-12:
            accept = True
        elif improvement > -T:
            accept = rng.random() < np.exp(improvement / T)
        else:
            accept = False
        
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            cur_max = new_max
            total_moves += 1
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20000:
            break

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_beta = 100.0
_g_dx = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _g_beta) * _g_dx

def _con_fun(x):
    return np.sum(x) - _g_target

def _con_jac(x):
    return np.ones(len(x))

def _slsqp_pass(cur_h, n_pts, target, dx, beta, max_iter):
    global _g_beta, _g_dx, _g_target
    _g_beta = float(beta)
    _g_dx = dx
    _g_target = target
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, cur_h, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        return project_h(res.x, target)
    except Exception:
        return None

def fix_binary_sum(h, target, seed):
    h = h.copy()
    rng = np.random.RandomState(seed)
    deficit = int(round(target - h.sum()))
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= deficit:
            idx = rng.choice(len(zeros), deficit, replace=False)
            h[zeros[idx]] = 1.0
        else:
            h[zeros] = 1.0
            for k in range(deficit - len(zeros)):
                h[rng.randint(len(h))] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        if len(ones) >= -deficit:
            idx = rng.choice(len(ones), -deficit, replace=False)
            h[ones[idx]] = 0.0
        else:
            h[ones] = 0.0
            for k in range(-deficit - len(ones)):
                h[rng.randint(len(h))] = 0.0
    return h

def binary_local_search(h_init, max_time_s, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    h = (h_init >= 0.5).astype(float)
    h = fix_binary_sum(h, target, seed)

    corr = np.correlate(h, 1.0 - h, mode="full").astype(float).copy()
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max

    lags = np.arange(-(n - 1), n)
    t_start = time.time()
    no_improve = 0

    while time.time() - t_start < max_time_s:
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break

        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]

        jp = j + lags
        ip = i + lags
        jm = j - lags
        im = i - lags

        delta = np.zeros(2 * n - 1)
        v = (jp >= 0) & (jp < n)
        delta[v] += h[jp[v]]
        v = (ip >= 0) & (ip < n)
        delta[v] -= h[ip[v]]
        v = (im >= 0) & (im < n)
        delta[v] += h[im[v]]
        v = (jm >= 0) & (jm < n)
        delta[v] -= h[jm[v]]

        new_max = np.max(corr + delta)
        improvement = cur_max - new_max

        elapsed_frac = (time.time() - t_start) / max_time_s
        T = 0.0005 * (1.0 - elapsed_frac) + 0.00005

        if improvement > 1e-12:
            accept = True
        elif improvement > -T:
            accept = rng.random() < np.exp(improvement / T)
        else:
            accept = False

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1

        if no_improve > 30000:
            break

    return best_h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    global _g_beta, _g_dx, _g_target
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP continuation at parent resolution
    if budget_s - elapsed() > 100:
        target = n_base / 2.0
        dx = 2.0 / n_base
        cur = h_parent.copy()
        for beta in [80, 300, 1200, 5000]:
            if elapsed() > budget_s - 20:
                break
            h_new = _slsqp_pass(cur, n_base, target, dx, beta, 800)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(cur):
                    cur = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new

    # Phase 2: Multi-start binary local search
    remaining = budget_s - elapsed() - 20
    if remaining > 30:
        n_searches = min(int(remaining / 25) + 1, 14)
        time_per = remaining / n_searches

        for i in range(n_searches):
            if elapsed() > budget_s - 10:
                break
            # Vary threshold
            t_val = 0.35 + 0.3 * (i % 7) / 6.0
            s = seed + i * 13 + 7

            # Try from best continuous and from parent
            h_b1 = (best_h >= t_val).astype(float)
            h_b2 = (h_parent >= t_val).astype(float)
            c1 = compute_c5(fix_binary_sum(h_b1, n_base / 2.0, s))
            c2 = compute_c5(fix_binary_sum(h_b2, n_base / 2.0, s))
            src = h_b1 if c1 <= c2 else h_b2

            h_found, c5_found = binary_local_search(src, time_per, s)
            if c5_found < best_c5:
                best_h = h_found.copy()
                best_c5 = c5_found

    # Phase 3: Final SLSQP polish
    if elapsed() < budget_s - 25:
        n_f = len(best_h)
        h_new = _slsqp_pass(best_h, n_f, n_f / 2.0, 2.0 / n_f, 10000, 5000)
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Phase 4: One more binary search on best
    if elapsed() < budget_s - 8:
        h_b = (best_h >= 0.5).astype(float)
        h_found, c5_found = binary_local_search(h_b, budget_s - elapsed() - 3, seed + 7777)
        if c5_found < best_c5:
            best_h = h_found.copy()
            best_c5 = c5_found

    return best_h, best_c5, int(len(best_h))
