```python
def active_set_lp(h, n, n_active=10):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Find active k values
    sorted_idx = np.argsort(c)[::-1]
    active_k = sorted_idx[:n_active]
    
    # Linearize c[k] around h
    # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])  [using np.correlate convention]
    # dc[k]/dh[m] = (1-h[m-k]) - h[m+k] for valid indices
    
    # Build LP: minimize t subject to c_lin[k] <= t for k in active, 0<=h<=1, sum(h)=n/2
    import cvxpy as cp
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for k in active_k:
        # Compute linearized c[k] at h
        # c_lin[k] = c[k] + sum_m dc[k]/dh[m] * (h_var[m] - h[m])
        shift = k - (n-1)
        grad = np.zeros(n)
        for m in range(n):
            g = 0.0
            if 0 <= m - shift < n:
                g += (1 - h[m - shift])
            if 0 <= m + shift < n:
                g += (-h[m + shift])
            grad[m] = g
        c_lin = c[k] + np.dot(grad, h_var - h)
        constraints.append(c_lin <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1)
    return h
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-13: break
            h += (diff / rs) * room
        else:
            hs = h.sum()
            if hs < 1e-13: break
            h *= target / hs
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Softmax for weights
    smax = shifted.max()
    if smax > 500:
        w = np.exp(shifted - smax)
        w_sum = w.sum()
        obj = float(m + (smax + np.log(w_sum)) / alpha)
    else:
        w = np.exp(shifted)
        w_sum = w.sum()
        obj = float(m + np.log(w_sum) / alpha)
    w = w / w_sum
    
    # Gradient
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-16:
            continue
        shift = j - (n - 1)
        wj = w[j] * dx
        # d c[j] / d x[m] for all m
        m_arr = np.arange(n)
        i1 = m_arr - shift
        mask1 = (i1 >= 0) & (i1 < n)
        grad[mask1] += wj * (1.0 - x[i1[mask1]])
        i2 = m_arr + shift
        mask2 = (i2 >= 0) & (i2 < n)
        grad[mask2] -= wj * x[i2[mask2]]
    
    return obj, grad

def slsqp_h(h, n, alpha, maxiter=2000, ftol=1e-15):
    def fun(x):
        return obj_and_grad(x, n, alpha)[0]
    def jac(x):
        return obj_and_grad(x, n, alpha)[1]
    
    bounds = [(0.0, 1.0)] * n
    def sum_con(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except:
        pass
    return h

        return h, compute_c5(h, n)
```

```python
def pair_update(h, n, i, j, delta):
    """Compute new max correlation after changing h[i]+=delta, h[j]-=delta."""
    s_range = np.arange(-(n-1), n)  # all shifts
    # For shift s:
    # Δc[s] = delta * [(1-h[i-s]) - h[i+s]] - delta * [(1-h[j-s]) - h[j+s]]
    # where terms are 0 if index out of range
    
    di = np.zeros(2*n-1)
    dj = np.zeros(2*n-1)
    
    # For index i: valid shifts where i-s is in [0,n) or i+s is in [0,n)
    # i-s valid when s in [i-n+1, i]
    # i+s valid when s in [-i, n-1-i]
    s_arr = np.arange(-(n-1), n)
    
    # (1-h[i-s]) contribution: valid when 0 <= i-s < n, i.e., i-n+1 <= s <= i
    mask1 = (i - s_arr >= 0) & (i - s_arr < n)
    di[mask1] = 1 - h[i - s_arr[mask1]]
    # -h[i+s] contribution: valid when 0 <= i+s < n, i.e., -i <= s <= n-1-i
    mask2 = (i + s_arr >= 0) & (i + s_arr < n)
    di[mask2] -= h[i + s_arr[mask2]]
    
    mask3 = (j - s_arr >= 0) & (j - s_arr < n)
    dj[mask3] = 1 - h[j - s_arr[mask3]]
    mask4 = (j + s_arr >= 0) & (j + s_arr < n)
    dj[mask4] -= h[j + s_arr[mask4]]
    
    dc = delta * (di - dj)
    return dc  # Change in raw correlation values (not yet multiplied by dx)
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-13: break
            h += (diff / rs) * room
        else:
            hs = h.sum()
            if hs < 1e-13: break
            h *= target / hs
    return np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    """Compute gradient vectors for all coordinates.
    Returns G[i, s] = dc[shift s]/dh[i] (without dx factor)
    """
    n_shifts = 2 * n - 1
    G = np.zeros((n, n_shifts))
    s_arr = np.arange(-(n-1), n)
    for i in range(n):
        mask1 = (i - s_arr >= 0) & (i - s_arr < n)
        G[i, mask1] += 1.0 - h[i - s_arr[mask1]]
        mask2 = (i + s_arr >= 0) & (i + s_arr < n)
        G[i, mask2] -= h[i + s_arr[mask2]]
    return G

def obj_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    if smax > 500:
        w = np.exp(shifted - smax)
        obj = m + (smax + np.log(w.sum())) / alpha
    else:
        w = np.exp(shifted)
        obj = m + np.log(w.sum()) / alpha
    w = w / w.sum()
    
    # Gradient using precomputed structure
    grad = np.zeros(n)
    s_arr = np.arange(-(n-1), n)
    for j in range(2*n-1):
        if w[j] < 1e-15: continue
        shift = s_arr[j]
        wj = w[j] * dx
        m_arr = np.arange(n)
        i1 = m_arr - shift
        mask1 = (i1 >= 0) & (i1 < n)
        grad[mask1] += wj * (1.0 - x[i1[mask1]])
        i2 = m_arr + shift
        mask2 = (i2 >= 0) & (i2 < n)
        grad[mask2] -= wj * x[i2[mask2]]
    return float(obj), grad

def slsqp_opt(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    def fun(x): return obj_grad(x, n, alpha)[0]
    def jac(x): return obj_grad(x, n, alpha)[1]
    def con_fun(x): return x.sum() - n/2.0
    def con_jac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:

    except Exception:
        pass
    return h

            i2 = m_arr + shift
