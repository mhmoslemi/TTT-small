```python
def c5_obj_grad(h, n, alpha):
    """Compute log-sum-exp of correlations and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    obj = m + np.log(exp_vals.sum()) / alpha
    p = exp_vals / exp_vals.sum()  # length 2n-1
    
    # Gradient: grad[j] = dx * sum_k p[k] * (1{j+k<n} - h[j+k]*1{j+k<n} - h[j-k]*1{j-k>=0})
    # where k = k_idx - (n-1) is the lag
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    for k_idx in range(2*n-1):
        if p[k_idx] < 1e-16:
            continue
        lag = lags[k_idx]
        # Term 1: 1{j+k<n} for all j
        # j + lag < n => j < n - lag => j <= n-1-lag (if lag < n)
        # Valid j: 0 <= j < n-lag (if lag < n), or none (if lag >= n)
        # Also need j >= 0
        # j + lag >= 0 is not required for this term
        
        # Actually, let me just compute for each j:
        # term = 1{j+lag < n} - h[j+lag]*1{j+lag < n} - h[j-lag]*1{j-lag >= 0}
        #      = (1 - h[j+lag])*1{0 <= j+lag < n} - h[j-lag]*1{0 <= j-lag < n}
        
        # For vectorized computation:
        # j+lag ranges from 0+lag to n-1+lag
        # Valid when 0 <= j+lag < n, i.e., -lag <= j < n-lag
        # j valid range: max(0, -lag) to min(n-1, n-1-lag) = max(0, -lag) to min(n-1, n-1-lag)
        
        j_start1 = max(0, -lag)
        j_end1 = min(n, n - lag)  # exclusive
        if j_end1 > j_start1:
            grad[j_start1:j_end1] += p[k_idx] * (1.0 - h[j_start1:j_end1 + lag])
        
        j_start2 = max(0, lag)
        j_end2 = min(n, n + lag)  # exclusive... wait
        # j-lag valid when 0 <= j-lag < n, i.e., lag <= j < n+lag
        j_start2 = max(0, lag)
        j_end2 = min(n, n + lag)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= p[k_idx] * h[j_start2:j_end2 - lag]
    
    grad *= dx
    return obj, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def c5_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -600, 600)
    exp_v = np.exp(shifted)
    obj = m + np.log(exp_v.sum()) / alpha
    p = exp_v / exp_v.sum()
    nc = 2 * n - 1
    lags = np.arange(nc) - (n - 1)
    grad = np.zeros(n)
    for k_idx in range(nc):
        if p[k_idx] < 1e-16:
            continue
        lag = lags[k_idx]
        js1 = max(0, -lag)
        je1 = min(n, n - lag)
        if je1 > js1:
            grad[js1:je1] += p[k_idx] * (1.0 - h[js1:je1 + lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] -= p[k_idx] * h[js2:je2 - lag]
    grad *= dx
    return obj, grad

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    def sum_fun(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    res = minimize(c5_obj_grad, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def subgradient_step(h, n, steps=2000, gamma0=1.0):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    nc = 2 * n - 1
    lags = np.arange(nc) - (n - 1)
    for t in range(steps):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        active = np.where(np.abs(corr - corr.max()) < 1e-12)[0]
        n_active = len(active)
        g = np.zeros(n)
        for k_idx in active:
            lag = lags[k_idx]
            js1 = max(0, -lag)
            je1 = min(n, n - lag)
            if je1 > js1:
                g[js1:je1] += (1.0 - h[js1:je1 + lag])
            js2 = max(0, lag)
            je2 = min(n, n + lag)
            if je2 > js2:
                g[js2:je2] -= h[js2:je2 - lag]
        g *= dx / n_active
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        eta = gamma0 / (1.0 + 0.1 * t)
        h_new = h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def pair_local(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    nc = 2 * n - 1
    for d in deltas:
        for _ in range(3):
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        break

                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
            if not improved:
                break
    return h, best_c
