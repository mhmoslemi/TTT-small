    dx = 2.0 / n
    # Gradient of corr[s] with respect to h

    # corr[s] = sum_i h[i](1-h[i+s]) for valid i
    # d/dh[j] = (1-h[j+s]) if 0<=j+s<n, minus h[j-s] if 0<=j-s<n

            order = np.argsort(h_new)
            added = 0
            for idx in order:

            for idx in order:

                h_new[idx] -= rem
                removed += rem
                if removed >= -diff - 1e-12:
                    break
    return h_new, v

import numpy as np
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def corr_grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    elif lag < 0:
        p = -lag
        grad[p:] += (1.0 - h[:n - p])
        grad[:n - p] -= h[p:]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(400):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def topk_smoothmax_val_grad(h, k, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:k]
    lags = top_idx - (n - 1)
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-14:
            continue
        grad += w[i] * corr_grad_lag(h, lags[i])
    grad *= dx
    return val, grad


def momentum_grad_descent(h, n_iter, k, beta, lr0, lr_decay, momentum, target):
    n = len(h)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(n_iter):
        val, grad = topk_smoothmax_val_grad(h, k, beta)
        v = momentum * v - lr * grad
        h = project_feasible(h + v, target)
        if it > 0 and it % 100 == (n_iter % 100 - 1 if n_iter % 100 != 0 else 99):
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
        lr = lr0 * (lr_decay ** (it / max(1, n_iter // 10)))
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5


def binary_round(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:k]] = 1.0
    return h_bin


def compute_grad_matrix(h):
    """Compute G[i, s] = d corr[s] / d h[i] for all i and s."""
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    for s_idx in range(2 * n - 1):
        lag = s_idx - (n - 1)
        if lag > 0:
            G[:n - lag, s_idx] = (1.0 - h[lag:])
            G[lag:, s_idx] = -h[:n - lag]
        elif lag < 0:
            p = -lag
            G[p:, s_idx] = (1.0 - h[:n - p])
            G[:n - p, s_idx] = -h[p:]
        else:
            G[:, s_idx] = 1.0 - 2.0 * h
    return G


def binary_local_search(h_bin, max_rounds=80):
    n = len(h_bin)
    target = n / 2.0
    h = h_bin.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = corr.max() * dx

    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        G = compute_grad_matrix(h)
        H = corr[None, :] + G  # H[j, s] = corr[s] + G[j,s]

        best_improve = 0
        best_i = None
        best_j = None

        for i in ones:
            base = corr - G[i]
            if len(zeros) == 0:
                break
            new_maxs = (base[None, :] + G[zeros, :]).max(axis=1)
            j_local = zeros[new_maxs.argmin()]
            cur_max = new_maxs.min()
            if cur_max < corr.max() - best_improve:
                best_improve = corr.max() - cur_max
                best_i = i
                best_j = j_local

        if best_i is not None:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = compute_corr(h)
            cur_c5 = corr.max() * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                improved = True
        if not improved:
            break
    return h, best_c5


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def basin_hop(h_best, best_c5, n_hops, beta, k, lr, momentum, max_iter, noise_base, noise_decay, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 30:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = project_feasible(h_best + rng.randn(n) * noise, target)
        h_local, c_local = momentum_grad_descent(h_pert, max_iter, k, beta, lr, 0.97, momentum, target)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5
