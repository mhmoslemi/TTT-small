For coordinate descent: if we fix all h[j] for j≠i and want to minimize the smooth max of the top-k lags, we need to find the h[i] that minimizes:
S(h[i]) = (1/k) ∑_{m∈top_k} (corr[m] + Δ_corr[m](h[i]))

where Δ_corr[m] is the change in corr[m] due to changing h[i].

Since corr[m] is linear in h[i] (for m≠0) and quadratic in h[i] (for m=0), S(h[i]) is a quadratic function of h[i]. We can find its minimum analytically!

Let me formalize this. For a given h[i] = t:
- corr[0] changes by: t*(1-t) - h[i]*(1-h[i]) (quadratic in t)
- corr[m] (m>0, i+m<n) changes by: h[i+m]*(1-t) - h[i+m]*(1-h[i]) = h[i+m]*(h[i]-t) (linear in t)
- corr[-m] (m>0, i-m>=0) changes by: t*(1-h[i-m]) - h[i]*(1-h[i-m]) = (t-h[i])*(1-h[i-m]) (linear in t)

So if the top-k lags include lag 0, S(t) is quadratic. Otherwise, it's linear.

For the quadratic case: S(t) = a*t² + b*t + c, minimum at t = -b/(2a) (clipped to [0,1]).
For the linear case: S(t) is minimized at t=0 or t=1 (boundary).

This gives us an efficient coordinate descent method!

```python
def coordinate_step(h, top_lags, n, dx):
    """Perform one coordinate descent sweep."""
    for i in range(n):
        old = h[i]
        # Compute the effect of changing h[i] on each top lag
        a = 0.0  # quadratic coefficient
        b = 0.0  # linear coefficient
        for t_idx in top_lags:
            lag = t_idx - (n - 1)
            if lag == 0:
                # h[i]*(1-h[i]) changes to t*(1-t)
                a += 1.0
                b -= 1.0
            elif lag > 0:
                if i - lag >= 0:
                    # h[i] is the "first" element: h[i]*(1-h[i-lag])
                    b += (1 - h[i-lag])
                if i + lag < n:
                    # h[i] is the "second" element: h[i+lag]*(1-h[i])
                    b -= h[i+lag]
            else:
                l = -lag
                if i + l < n:
                    # h[i+l]*(1-h[i]): h[i] is "second"
                    b -= h[i+l]
                if i - l >= 0:
                    # h[i]*(1-h[i+l]): h[i] is "first"
                    b += (1 - h[i+l])
        # Minimize a*t^2 + b*t over t in [0,1]
        if abs(a) > 1e-15:
            t_opt = -b / (2*a)
        else:
            t_opt = 0.0 if b > 0 else 1.0
        t_opt = np.clip(t_opt, 0.0, 1.0)
        h[i] = t_opt
    return h
```

```python
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")

    """Smooth max using only top-k values."""
    k = min(topk, len(x))
    top = np.partition(x, -k)[-k:]
    m = np.max(top)
    shifted = np.clip(beta * (top - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) / k) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

        return h_old.copy()

    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_topk(corr, beta, topk) * dx

    return np.sum(h) - target

    return np.ones(h.shape)

def coord_descent(h, top_k, n, max_sweeps=10):
    """Coordinate descent on top-k lags."""
    center = n - 1
    for sweep in range(max_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        changed = False
        for i in range(n):
            old = h[i]
            if old <= 1e-10 or old >= 1.0 - 1e-10:
                continue
            # Compute linear/quadratic coefficients for top lags
            a = 0.0
            b = 0.0

                if lag == 0:
                    a += 1.0
                    b -= 1.0
                elif lag > 0:
                    if i - lag >= 0:
                        b += (1.0 - h[i - lag])
                    if i + lag < n:
                        b -= h[i + lag]
                else:
                    l = -lag

                        b -= h[i + l]
                    if i - l >= 0:
                        b += (1.0 - h[i + l])
            if abs(a) > 1e-15:
                t_opt = -b / (2.0 * a)
            else:
                t_opt = 0.0 if b > 0 else 1.0
            t_opt = np.clip(t_opt, 0.0, 1.0)
            if abs(t_opt - old) > 1e-10:
                h[i] = t_opt
                changed = True
        if not changed:
            break
    return h

def subgradient_step(h, n, dx, top_k, step, momentum_buf=None, mu=0.7):
    """One subgradient step with momentum."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_val = corr * dx
    center = n - 1
    top_idx = np.argsort(corr_val)[::-1][:top_k]
    vals = corr_val[top_idx]
    
    # Soft weights based on gap to max
    gap = vals[0] - vals
    weights = np.exp(-gap * 200.0)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for t_idx, w in zip(top_idx, weights):
        lag = int(t_idx) - center
        if lag == 0:
            grad[:] += w * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag]) * dx
            grad[:n - lag] += w * (-h[lag:]) * dx
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:]) * dx
            grad[l:] += w * (-h[:n - l]) * dx
    
    if momentum_buf is not None:
        grad = mu * momentum_buf + (1 - mu) * grad
        momentum_buf[:] = grad
    
    return h, grad, momentum_buf
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_smooth(h, dx, beta, topk):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(topk, len(corr))
    top = np.partition(corr, -k)[-k:]
    m = top.max()
    s = np.clip(beta * (top - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s)) / k) / beta) * dx

def proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def grad_topk(h, n, dx, topk, temp=100.0):
    """Gradient of smooth max of top-k correlation values."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    cv = corr * dx
    center = n - 1
    top = np.argsort(cv)[::-1][:topk]
    vals = cv[top]
    w = np.exp(-(vals[0] - vals) * temp)
    w /= w.sum()
    g = np.zeros(n)
    for idx, wi in zip(top, w):
        lag = int(idx) - center
        if lag == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wi * (1.0 - h[:n-lag])
            g[:n-lag] += wi * (-h[lag:])
        else:
            l = -lag
            g[:n-l] += wi * (1.0 - h[l:])
            g[l:] += wi * (-h[:n-l])
    return g

def _coord_cd(h, n, target, max_sweeps=10):
    """Coordinate descent on top-2 lags."""
    center = n - 1
    for _ in range(max_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top2 = np.argsort(corr)[::-1][:2]
        for i in range(n):
            old = h[i]
            if old <= 1e-8 or old >= 1.0 - 1e-8:
                continue
            a = 0.0
            b = 0.0
            for idx in top2:
                lag = int(idx) - center
                if lag == 0:
                    a += 1.0
                    b -= 1.0
                elif lag > 0:
                    if i + lag < n:
                        b += (1.0 - h[i+lag])
                    if i - lag >= 0:
                        b -= h[i-lag]
                else:
                    l = -lag
                    if i - l >= 0:
                        b += (1.0 - h[i-l])
                    if i + l < n:
                        b -= h[i+l]
            if abs(a) > 1e-14:
                t = -b / (2.0 * a)
            else:
                t = 0.0 if b > 0 else 1.0
            h[i] = np.clip(t, 0.0, 1.0)
        h = proj(h, target)
    return h
```
