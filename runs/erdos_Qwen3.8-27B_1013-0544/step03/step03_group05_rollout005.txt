                max_iters=3000, time_fn=time_left, time_limit=8,
                best_state=best_state,
                momentum_init=0.85, momentum_decay=0.999,
                step_init=0.03, top_k=5, temp_init=300.0, temp_growth=1.5,
                perturb_every=100
            )

    # Final high-precision polish
    if time_left() > 8:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target_f = best_n / 2.0
        dx_f = 2.0 / best_n
        best_h, best_c5 = fine_polish(best_h, best_n, target_f, dx_f,
                                       time_left, 2)
        if best_c5 < best_state[0]:
            best_state[0] = best_c5
            best_state[1] = best_h

    return best_state[1], best_state[0], best_state[2]
