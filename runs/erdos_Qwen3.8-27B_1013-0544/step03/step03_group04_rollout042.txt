Actually, I realize I should also try a different strategy for the perturbation: instead of random Gaussian perturbations, try structured perturbations that specifically target the worst lag.

For example, if the worst lag is l*, I can try to "shift" some of the mass of h in a direction that reduces the overlap at lag l*.

Let me think about what direction reduces the overlap at lag l*. The overlap at lag l is:
O(l) = sum_i h[i] * (1-h[i+l])

The gradient of O(l) w.r.t. h[j] is:
(1-h[j-l]) - h[j+l] (with boundary conditions)

So to decrease O(l), we should move h in the direction -gradient(O(l)).

But we need to do this while keeping the other overlaps from increasing. This is exactly what the smooth max gradient does - it's a weighted combination of the gradients of the top overlaps.

So the projected gradient method with the smooth max gradient is already doing the right thing. The key is to have a good approximation of which overlaps are "active" (near the max), which is controlled by beta.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = free.sum()
        if nf == 0:
            h += diff / len(h)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(x))) / beta - np.log(len(corr)) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -300, 300)
    exp_x = np.exp(x)
    s = np.sum(exp_x)
    w = exp_x / s
    active = np.where(w > 1e-5)[0]
    grad = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        i_lo = max(0, -l)
        i_hi = min(n - 1, n - 1 - l)
        if i_lo > i_hi:
            continue
        wk = dx * w[k]
        j1_lo = max(0, i_lo + l)
        j1_hi = min(n - 1, i_hi + l)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += wk * (1.0 - h[j1_lo-l:j1_hi-l+1])
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] -= wk * h[i_lo+l:i_hi+l+1]
    return grad

def pgd_solve(h, target, beta, max_iter, time_limit):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    cur = h.copy()
    cur_f = smooth_max_val(np.correlate(cur, 1.0-cur, mode='full'), beta) * dx
    step = 0.1
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = smooth_max_grad(cur, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        alpha = step
        improved = False
        for _ in range(25):
            h_t = project_h(cur - alpha * g, target)
            f_t = smooth_max_val(np.correlate(h_t, 1.0-h_t, mode='full'), beta) * dx
            if f_t < cur_f - 1e-14:
                cur = h_t
                cur_f = f_t
                improved = True
                break
            alpha *= 0.5
        if improved:
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.4
            if step < 1e-10:
                break
    return cur, cur_f

def slsqp_solve(h, target, beta, max_iter=1000):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        return smooth_max_val(corr, beta) * dx
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(x.shape)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constr,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_h(res.x, target)
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        h_parent = np.full(280, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: PGD at parent resolution with increasing beta
    n = len(best_h)
    target = n / 2.0
    for beta in [50, 100, 200, 500, 1000, 3000, 8000]:
        if rem() < 10:

        h_new, f = pgd_solve(best_h, target, beta, max_iter=500, time_limit=min(30, rem()-5))
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
    
    # Phase 2: SLSQP at parent resolution

        c5 = compute_c5(h_new)
        if c5 < best_c5:

    # Phase 3: Upsample to 2x and optimize
    n2 = n * 2

        x_new = np.linspace(0, 2, n2)
        h_up = project_h(np.interp(x_new, x_old, best_h), target2)
        cur_c5 = compute_c5(h_up)
        
        for beta in [100, 200, 500, 1000, 3000, 8000]:
            if rem() < 10:

            c5 = compute_c5(h_new)
            if c5 < cur_c5:
                h_up, cur_c5 = h_new, c5
            if c5 < best_c5:
                best_h, best_c5 = h_up.copy(), c5
        
        for beta in [1000, 5000]:
            if rem() < 20:
                break
            h_new = slsqp_solve(h_up, target2, beta, max_iter=500)
            c5 = compute_c5(h_new)
            if c5 < cur_c5:
                h_up, cur_c5 = h_new, c5
            if c5 < best_c5:
                best_h, best_c5 = h_up.copy(), c5
    
    # Phase 4: Try 3x upsampling if time
    n3 = n * 3
    if n3 <= 1000 and rem() > 30:
        target3 = n3 / 2.0

        x_new = np.linspace(0, 2, n3)
        h_up = project_h(np.interp(x_new, x_old, best_h), target3)
        cur_c5 = compute_c5(h_up)

                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_h, best_c5 = h_r, c5_r
    
    # Final
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = free.sum()
        if nf == 0:
            h += diff / len(h)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(x))) / beta - np.log(len(corr)) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -300, 300)
    exp_x = np.exp(x)
    s = np.sum(exp_x)
    w = exp_x / s
    active = np.where(w > 1e-5)[0]
    grad = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        i_lo = max(0, -l)
        i_hi = min(n - 1, n - 1 - l)
        if i_lo > i_hi:
            continue
        wk = dx * w[k]
        j1_lo = max(0, i_lo + l)
        j1_hi = min(n - 1, i_hi + l)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += wk * (1.0 - h[j1_lo - l:j1_hi - l + 1])
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] -= wk * h[i_lo + l:i_hi + l + 1]
    return grad

def pgd_solve(h, target, beta, max_iter, time_limit):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    cur = h.copy()
    cur_f = smooth_max_val(np.correlate(cur, 1.0 - cur, mode='full'), beta) * dx
    step = 0.1
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = smooth_max_grad(cur, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        alpha = step
        improved = False
        for _ in range(30):
            h_t = project_h(cur - alpha * g, target)
            f_t = smooth_max_val(np.correlate(h_t, 1.0 - h_t, mode='full'), beta) * dx
            if f_t < cur_f - 1e-14:
                cur = h_t
                cur_f = f_t
                improved = True
                break
            alpha *= 0.5
        if improved:
            step = min(step * 1.03, 0.5)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            if stall > 30 or step < 1e-9:
                break
    return cur, cur_f

def slsqp_solve(h, target, beta, max_iter=800):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, beta) * dx
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(x.shape)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constr,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_h(res.x, target)
    except Exception:
        return h
