def project_h(h, target_sum):

        if abs(diff) < 1e-14:

def lag_gradient(h, lag, n):
    """Gradient of correlation at a specific lag w.r.t. h"""
    grad = np.zeros(n)
    # corr at lag: sum_i h[i] * (1 - h[i+lag]) for valid i
    # d/dh[i] = (1-h[i+lag]) if i+lag valid
    #           - h[i-lag] if i-lag valid (from the term where j=i-lag: h[j]*(1-h[j+lag]) = h[i-lag]*(1-h[i]))
    for i in range(n):
        j = i + lag
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
        j = i - lag
        if 0 <= j < n:
            grad[i] -= h[j]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def make_obj(target, dx, beta):
    def obj(x):
        h = project_h(x, target)
        corr = compute_corr(h)
        return smooth_max(corr, beta) * dx
    return obj

def make_con(target):
    def con(x):
        return x.sum() - target
    return con

def make_jac(n):
    def jac(x):
        return np.ones(n)
    return jac

def fix_binary_sum(hb, target, rng):
    hb = hb.copy()
    deficit = int(round(target - hb.sum()))
    if deficit > 0:
        zeros = np.where(hb == 0)[0]
        idx = rng.choice(zeros, min(deficit, len(zeros)), replace=False)
        hb[idx] = 1.0
    elif deficit < 0:
        ones = np.where(hb == 1)[0]
        idx = rng.choice(ones, min(-deficit, len(ones)), replace=False)
        hb[idx] = 0.0
    return hb

def binary_local_search(hb, target, n_iter=5000, n_trials=25, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(hb)
    hb = hb.copy()
    corr = compute_corr(hb)
    best_c5 = float(np.max(corr))
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    stale = 0
    for it in range(n_iter):
        improved = False
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            hb[i] = 0
            hb[j] = 1
            new_max = float(np.max(compute_corr(hb)))
            if new_max < best_c5:
                best_c5 = new_max
                ones = np.where(hb == 1)[0]
                zeros = np.where(hb == 0)[0]
                improved = True
            else:
                hb[i] = 1
                hb[j] = 0
        if not improved:
            stale += 1
            if stale > 3:
                break
        else:
            stale = 0
    return hb, best_c5

def block_swap_search(hb, target, n_iter=3000, rng=None):
    """Try shifting blocks of consecutive 1s or 0s"""
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(hb)
    hb = hb.copy()
    best_c5 = float(np.max(compute_corr(hb)))
    for it in range(n_iter):
        improved = False
        for _ in range(15):
            start = rng.randint(n)
            length = rng.randint(1, min(8, n // 4))
            end = min(start + length, n)
            if end - start < 1:
                continue
            block = hb[start:end].copy()
            shift = rng.choice([-2, -1, 1, 2])
            new_start = start + shift
            new_end = end + shift
            if new_start < 0 or new_end > n:
                continue
            if new_end <= start or new_start >= end:
                hb[start:end] = 0.0
                hb[new_start:new_end] = block
                new_max = float(np.max(compute_corr(hb)))
                if new_max < best_c5:
                    best_c5 = new_max
                    improved = True
                else:
                    hb[new_start:new_end] = 0.0
                    hb[start:end] = block
        if not improved:
            break
    return hb, best_c5

def simulated_annealing(hb, target, t_start=0.3, t_end=0.005, n_iter=8000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(hb)
    hb = hb.copy()
    current_c5 = float(np.max(compute_corr(hb)))
    best_c5 = current_c5
    best_hb = hb.copy()
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / n_iter)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i] = 0
        hb[j] = 1
        new_c5 = float(np.max(compute_corr(hb)))
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            current_c5 = new_c5
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_hb = hb.copy()
        else:
            hb[i] = 1
            hb[j] = 0
    return best_hb, best_c5

def multi_resolution_slsqp(h_start, n_target, target, dx, elapsed_fn, remaining_fn, seed):
    h_cur = h_start.copy()
    cur_c5 = compute_c5(h_cur)
    n_pts = n_target
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    con = make_con(target)
    jac = make_jac(n_pts)
    for beta in [60, 150, 400, 1000, 2500]:
        if remaining_fn() < 15:
            break
        obj = make_obj(target, dx, beta)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h_cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return h_cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at base resolution
    if remaining() > 80:
        n1 = n_base
        target1 = n1 / 2.0
        dx1 = 2.0 / n1
        h1 = project_h(h_parent.copy(), target1)
        h1, c5_1 = multi_resolution_slsqp(h1, n1, target1, dx1, elapsed, remaining, seed)
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    # Phase 2: Binary local search at base and doubled resolution
    for n_res in [n_base, 2 * n_base]:
        if remaining() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_interp = np.interp(x_new, x_old, best_h)
        target = n_res / 2.0

        # Multiple threshold levels
        for thresh in [0.35, 0.45, 0.5, 0.55, 0.65]:
            if remaining() < 20:
                break
            hb = (h_interp >= thresh).astype(float)
            hb = fix_binary_sum(hb, target, rng)
            c5_try = compute_c5(hb)
            if c5_try < best_c5:
                best_h, best_c5 = hb.copy(), c5_try
            # Local search
            hb_ls, c5_ls = binary_local_search(hb, target, n_iter=2000, n_trials=20, rng=rng)
            if c5_ls < best_c5:
                best_h, best_c5 = hb_ls.astype(float).copy(), c5_ls
            # Block swap
            hb_bl, c5_bl = block_swap_search(hb_ls, target, n_iter=1500, rng=rng)
            if c5_bl < best_c5:
                best_h, best_c5 = hb_bl.astype(float).copy(), c5_bl

    # Phase 3: SLSQP at higher resolution from best
    if remaining() > 80:
        n2 = 2 * n_base
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)

    # Phase 4: Simulated annealing with restarts
    if remaining() > 100:
        n_sa = 2 * n_base
        target_sa = n_sa / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_sa)
        h_interp = np.interp(x_new, x_old, best_h)

        for restart in range(4):
            if remaining() < 35:
                break
            if restart == 0:
                hb = (h_interp >= 0.5).astype(float)
            else:
                hb = (h_interp >= 0.5).astype(float)
                n_flips = int(n_sa * 0.03 * (restart + 1))
                ones = np.where(hb == 1)[0]
                zeros = np.where(hb == 0)[0]
                for _ in range(n_flips):
                    if len(ones) == 0 or len(zeros) == 0:
                        break

        n_fin = len(best_h)
        target_fin = n_fin / 2.0

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
