```python
                I1 = 1.0 if 0 <= j_plus < n else 0.0
                I2 = 1.0 if 0 <= j_minus < n else 0.0
                a_vals[idx] = I1 - hjp * I1 - hjm * I2
                b_vals[idx] = corr[m_arr[idx]] - t0 * a_vals[idx]
            
            # Now minimize max(max_unaffected, max(a_vals * t + b_vals)) over t in [0,1]
            # This is min_t max(M, a_1*t+b_1, ..., a_m*t+b_m)
            # = min_t max(M, a_1*t+b_1, ..., a_m*t+b_m)
            # Add M as a constant "line": a=0, b=M
            
            a_all = np.append(a_vals, 0.0)
            b_all = np.append(b_vals, max_unaffected)
            
            # Find optimal t: candidates are 0, 1, and pairwise intersections
            candidates = [0.0, 1.0]
            for i in range(len(a_all)):
                for k in range(i + 1, len(a_all)):
                    da = a_all[i] - a_all[k]
                    if abs(da) > 1e-12:
                        t_int = (b_all[k] - b_all[i]) / da
                        if -1e-9 <= t_int <= 1.0 + 1e-9:
                            candidates.append(np.clip(t_int, 0.0, 1.0))
            
            # Evaluate
            best_t = t0
            best_val = np.max(a_all * t0 + b_all)
            for t in candidates:
                val = np.max(a_all * t + b_all)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
            
            delta = best_t - t0
            if abs(delta) > 1e-12:
                # Update h[j] and corr
                h[j] = best_t
                # Update corr at affected indices
                for idx in range(len(m_arr)):
                    corr[m_arr[idx]] += delta * a_vals[idx]
                total_improvement += abs(best_val - np.max(a_all * t0 + b_all))
        
        # Project to maintain constraint
        h = project_h(h, target_sum)
        # Recompute corr after projection (since projection might have changed values)
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        if total_improvement < tol * max_corr:
            break
    
    return h
```

```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _de_obj(x, target, dx, beta):
    h = np.clip(x, 0, 1)
    h = project_h(h, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def coord_descent(h_init, target_sum, max_sweeps=200, time_limit=None, t0=None, perturb_every=50, perturb_mag=0.05):
    n = len(h_init)
    dx = 2.0 / n
    h = project_h(h_init.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        max_corr = float(np.max(corr))
        
        for j in range(n):
            lo = min(n - 1 - j, j)
            hi = max(2 * n - 2 - j, n - 1 + j)
            
            # Max of unaffected corr
            if lo > 0 and hi < 2 * n - 2:
                max_unaff = max(float(np.max(corr[:lo])), float(np.max(corr[hi+1:])))
            elif lo > 0:
                max_unaff = float(np.max(corr[:lo]))
            elif hi < 2 * n - 2:
                max_unaff = float(np.max(corr[hi+1:]))
            else:
                max_unaff = 0.0
            
            # Build coefficients for affected range
            m_arr = np.arange(lo, hi + 1)
            lags = m_arr - (n - 1)
            j_plus = j + lags
            j_minus = j - lags
            valid_plus = (j_plus >= 0) & (j_plus < n)
            valid_minus = (j_minus >= 0) & (j_minus < n)
            
            # Safe indexing
            jp_safe = np.clip(j_plus, 0, n-1)
            jm_safe = np.clip(j_minus, 0, n-1)
            hjp = np.where(valid_plus, h[jp_safe], 0.0)
            hjm = np.where(valid_minus, h[jm_safe], 0.0)
            
            a_vals = valid_plus.astype(float) - hjp - hjm
            t0_val = h[j]
            b_vals = corr[m_arr].copy() - t0_val * a_vals
            
            # Include constant for unaffected
            a_all = np.empty(len(a_vals) + 1)
            b_all = np.empty(len(b_vals) + 1)
            a_all[:-1] = a_vals
            b_all[:-1] = b_vals
            a_all[-1] = 0.0
            b_all[-1] = max_unaff
            
            # Ternary search for min of max(a_all*t + b_all) over t in [0,1]
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = np.max(a_all * m1 + b_all)
                f2 = np.max(a_all * m2 + b_all)
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_opt = (lo_t + hi_t) / 2.0
            
            delta = t_opt - t0_val
            if abs(delta) > 1e-12:
                h[j] = t_opt
                corr[m_arr] += delta * a_vals
        
        # Project to maintain sum constraint
        h = project_h(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_now = float(np.max(corr) * dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Periodic perturbation to escape local optima
        if perturb_every > 0 and (sweep + 1) % perturb_every == 0 and c5_now > best_c5 * 1.001:
            h = project_h(h + perturb_mag * np.random.randn(n) * 0.1, target_sum)
            corr = np.correlate(h, 1.0 - h, mode="full")
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)
    
    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: DE for global exploration (if budget allows)
    if budget_s - elapsed() > 200:
        n_de = n_base
        target = n_de / 2.0
        dx_de = 2.0 / n_de
        bounds_de = [(0.0, 1.0)] * n_de
        
        try:
            de_res = differential_evolution(
                lambda x: _de_obj(x, target, dx_de, 100.0),
                bounds_de, seed=seed, maxiter=50,
                popsize=20, tol=1e-6, mutation=(0.5, 1.2),
                recombination=0.8, polish=False, updating='immediate',
                init='latinhypercube'
            )
            h_de = project_h(de_res.x, target)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h, best_c5 = h_de.copy(), c5_de
        except Exception:
            pass
    
    # Phase 2: Coordinate descent at base resolution
    if budget_s - elapsed() > 100:
        h_cd, c5_cd = coord_descent(
            best_h, n_base / 2.0,
            max_sweeps=200, time_limit=budget_s - 100, t0=t0,
            perturb_every=30, perturb_mag=0.03
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    
    # Phase 3: Resolution bump + coordinate descent
    for mult in [1.5, 2.0, 3.0]:
        n_new = int(n_base * mult)
        if n_new > 1000 or budget_s - elapsed() < 60:
            break
        
        target_new = n_new / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_h(h_interp, target_new)
        
        h_cd, c5_cd = coord_descent(
            h_interp, target_new,
            max_sweeps=100, time_limit=budget_s - 60, t0=t0,
            perturb_every=40, perturb_mag=0.02
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    
    # Phase 4: Final polish at best resolution
    if budget_s - elapsed() > 30:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_cd, c5_cd = coord_descent(
            best_h, target_final,
            max_sweeps=100, time_limit=budget_s - 10, t0=t0,
            perturb_every=0
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    
    # Final projection and evaluation
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, int(len(best_h))
```

```python
import numpy as np
from scipy.optimize import differential_evolution
import time

_g_target = 0.0
_g_dx = 0.0
_g_beta = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _de_objective(x):
    h = project_h(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, _g_beta) * _g_dx

                  perturb_every=0, perturb_mag=0.02, seed=42):
    n = len(h_init)
    dx = 2.0 / n
    h = project_h(h_init.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        for j in range(n):
            lo = min(n - 1 - j, j)
            hi = max(2 * n - 2 - j, n - 1 + j)
            
            # Max of unaffected
            if lo > 0 and hi < 2 * n - 2:
                max_unaff = max(float(np.max(corr[:lo])), float(np.max(corr[hi+1:])))
            elif lo > 0:
                max_unaff = float(np.max(corr[:lo]))
            elif hi < 2 * n - 2:
                max_unaff = float(np.max(corr[hi+1:]))
            else:
                max_unaff = 0.0
            
            m_arr = np.arange(lo, hi + 1)
            lags = m_arr - (n - 1)
            j_plus = j + lags
            j_minus = j - lags
            vp = (j_plus >= 0) & (j_plus < n)
            vm = (j_minus >= 0) & (j_minus < n)
            jp_s = np.clip(j_plus, 0, n-1)
            jm_s = np.clip(j_minus, 0, n-1)
            hjp = np.where(vp, h[jp_s], 0.0)
            hjm = np.where(vm, h[jm_s], 0.0)
            
            a_vals = vp.astype(float) - hjp - hjm
            t0v = h[j]
            b_vals = corr[m_arr].copy() - t0v * a_vals
            
            a_all = np.empty(len(a_vals) + 1)
            b_all = np.empty(len(b_vals) + 1)
            a_all[:-1] = a_vals
            b_all[:-1] = b_vals
            a_all[-1] = 0.0
            b_all[-1] = max_unaff
            
            # Ternary search: 40 iterations gives precision ~(2/3)^40 ≈ 5e-8
            lo_t, hi_t = 0.0, 1.0

                    hi_t = m2
                else:

            d = t_opt - t0v
            if abs(d) > 1e-12:
                h[j] = t_opt
                corr[m_arr] += d * a_vals
        
        # Project
        h = project_h(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_now = float(np.max(corr) * dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Periodic perturbation
        if perturb_every > 0 and (sweep + 1) % perturb_every == 0:
            if time.time() - t0 > time_limit - 5 if time_limit else False:
                break
            h = project_h(h + perturb_mag * rng.randn(n), target_sum)
            corr = np.correlate(h, 1.0 - h, mode="full")
    
    return best_h, best_c5
```

```python
        h_interp = project_h(np.interp(x_new, x_old, best_h), target_new)

        h_cd, c5_cd = coord_descent(
            h_interp, target_new, max_sweeps=300,
            time_limit=budget_s - 30, t0=t0,
            perturb_every=80, perturb_mag=0.02, rng_seed=seed + 500
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 4: Continued refinement at best resolution with smaller perturbations
    if budget_s - (time.time() - t0) > 20:
        n_f = len(best_h)
        h_cd, c5_cd = coord_descent(
            best_h, n_f / 2.0, max_sweeps=300,
            time_limit=budget_s - 10, t0=t0,
            perturb_every=100, perturb_mag=0.01, rng_seed=seed + 999
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 5: Final clean polish (no perturbation)
    if budget_s - (time.time() - t0) > 10:
        n_f = len(best_h)
        h_cd, c5_cd = coord_descent(
            best_h, n_f / 2.0, max_sweeps=200,
            time_limit=budget_s - 5, t0=t0,
            perturb_every=0
        )
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(len(best_h))
```
