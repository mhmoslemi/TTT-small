        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def corr_grad_single(h, lag, n, dx):
    """Gradient of corr at given lag with respect to h."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[n - lag:] += (1.0 - h[:lag]) * dx
        grad[:n - lag] += (-h[lag:]) * dx
        # Fix: for full correlation, lag meaning depends on indexing
        # corr[k] where k = n-1+lag corresponds to shift by lag
        # Actually let me use the standard: full corr index i corresponds to lag = i-(n-1)
        # corr[i] = sum_j h[j] * (1-h[j + i - (n-1)])
        # For lag > 0: corr[n-1+lag] = sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
        # d/dh[j] for j in [0, n-1-lag]: (1-h[j+lag])
        # d/dh[j] for j in [lag, n-1]: -h[j-lag]
        grad[:n - lag] += (1.0 - h[lag:]) * dx
        grad[lag:] += (-h[:n - lag]) * dx
    else:
        l = -lag
        grad[l:] += (1.0 - h[:n - l]) * dx
        grad[:n - l] += (-h[l:]) * dx
    return grad

def obj_grad_lbfgs(h, dx, beta, target, pen_w):
    """Objective and gradient for L-BFGS-B with penalty on sum constraint."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    val = smooth_max_val(c * dx, beta)
    w = smooth_max_weights(c * dx, beta)
    # Gradient of smooth max of (c*dx) w.r.t. h
    grad = np.zeros(n)
    center = n - 1
    for idx, wi in enumerate(w):
        if wi < 1e-14:
            continue
        lag = idx - center
        if abs(wi) < 1e-12:
            continue
        g = corr_grad_single(h, lag, n, dx)
        grad += wi * g
    # Penalty for sum constraint
    cv = h.sum() - target
    val += 0.5 * pen_w * cv * cv
    grad += pen_w * cv * np.ones(n)
    return val, grad

def momentum_subgrad(h, n, target, dx, max_iters, time_fn, min_time, best_state, momentum_init=0.9):
    """Projected subgradient with Nesterov momentum."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    vel = np.zeros(n)
    mu = momentum_init
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, len(corr_val))
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 300.0 + 200.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * corr_grad_single(cur, lag, n, dx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Nesterov lookahead
        y = cur - mu * vel
        step = 0.05 / (1.0 + 0.003 * it)
        improved = False
        for _ in range(25):
            h_trial = project_h(y - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                vel_new = h_trial - cur
                cur = h_trial
                cur_c5 = c5_trial
                vel = mu * vel + (1 - mu) * vel_new
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved:
            T = 0.004 * max(0.005, np.exp(-it / 800.0))
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T * 2:
                vel_new = h_trial - cur
                cur = h_trial
                cur_c5 = c5_trial
                vel = mu * vel + (1 - mu) * vel_new
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
        # Decay momentum slowly
        if it % 500 == 0 and it > 0:
            mu = max(0.5, mu * 0.95)
    return cur, cur_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent
    best_state = [best_c5, best_h.copy(), best_n]
    
    # Multi-resolution optimization
    for n in [n_parent, 300, 350, 400, 500]:
        if time_left() < 60:
            break
        target = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        
        # Start from best resampled
        h_cur = project_h(resample(best_state[1], n), target)
        cur_c5 = compute_c5(h_cur)
        if cur_c5 < best_state[0]:
            best_state[0] = cur_c5
            best_state[1] = h_cur.copy()
            best_state[2] = n
        
        # L-BFGS-B phase with increasing beta
        pen_w = 50.0
        for beta in [20, 40, 80, 150, 300, 600]:
            if time_left() < 25:
                break
            def lbf_obj(x):
                v, _ = obj_grad_lbfgs(x, dx, beta, target, pen_w)
                return v
            def lbf_grad(x):
                _, g = obj_grad_lbfgs(x, dx, beta, target, pen_w)
                return g
            try:
                res = minimize(lbf_obj, h_cur, jac=lbf_grad, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': 300, 'ftol': 1e-18,
                                                     'gtol': 1e-12, 'maxls': 40, 'maxcor': 30})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    h_cur = h_new
                    cur_c5 = c5_new
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = h_cur.copy()
                    best_state[2] = n
            except Exception:
                pass
            # Increase penalty as beta increases
            pen_w = min(pen_w * 1.5, 200.0)
        
        # Momentum subgradient phase
        if time_left() < 20:
            break
        h_cur, cur_c5, _ = momentum_subgrad(
            h_cur, n, target, dx,
            max_iters=5000, time_fn=time_left, min_time=8,
            best_state=best_state
        )
    
    # Final fine polish
    if time_left() > 15:
        h_f = best_state[1].copy()
        n_f = len(h_f)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        c5_f = best_state[0]
        
        # Extremely fine subgradient
        for it in range(80000):
            if time_left() < 3:
                break
            corr_raw = np.correlate(h_f, 1.0 - h_f, mode="full")
            corr_val = corr_raw * dx_f
            top_idx = np.argsort(corr_val)[::-1][:2]
            vals = corr_val[top_idx]
            weights = np.exp((vals - vals[0]) * 800.0)
            weights /= weights.sum()
            
            grad = np.zeros(n_f)
            center = n_f - 1
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                grad += w * corr_grad_single(h_f, lag, n_f, dx_f)
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            step = 0.003
            for _ in range(20):
                h_trial = project_h(h_f - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_f - 1e-16:
                    h_f = h_trial
                    c5_f = c5_trial
                    best_state[0] = c5_f
                    best_state[1] = h_f.copy()
                    best_state[2] = n_f
                    break
                step *= 0.5
            if it % 200 == 0 and it > 0:
                # Tiny perturbation to escape plateaus
                noise = np.random.normal(0, 0.0005, n_f)
                h_trial = project_h(h_f + noise, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_f:
                    h_f = h_trial
                    c5_f = c5_trial
                    best_state[0] = c5_f
                    best_state[1] = h_f.copy()
                    best_state[2] = n_f
    
    best_h = best_state[1]
    best_c5 = best_state[0]
    best_n = best_state[2]
    return best_h, best_c5, best_n
