from scipy.optimize import minimize, Bounds
from scipy.special import expit
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            # All at bounds, just add to any
            h[0] = np.clip(h[0] + diff, 0, 1)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_values(h, dx):
    """Get all correlation values"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_for_lag(h, lag, n, dx):
    """Gradient of f_lag(h) = dx * sum_i h[i]*(1-h[i+lag]) with respect to h"""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        # f_lag = dx * sum_{i=0}^{n-lag-1} h[i] * (1-h[i+lag])
        # d/dh[i] for i in [0, n-lag): (1-h[i+lag])*dx
        # d/dh[i+lag] for i in [0, n-lag): -h[i]*dx
        grad[:n-lag] += (1.0 - h[lag:]) * dx
        grad[lag:] += -h[:n-lag] * dx
    else:
        l = -lag
        # f_lag = dx * sum_{i=0}^{n-l-1} h[i+l] * (1-h[i])
        # Same as lag = l by symmetry of correlation
        grad[:n-l] -= h[l:] * dx  # d/dh[i] for the (1-h[i]) term
        grad[l:] += (1.0 - h[:n-l]) * dx  # d/dh[i+l] for the h[i+l] term
    return grad

def smooth_max_obj(h, dx, beta, target_sum):
    """Smooth max of correlation values + penalty for sum constraint"""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # Numerically stable log-sum-exp
    vals = np.clip(beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(vals))) / beta
    # Penalty for sum constraint
    penalty = 1e6 * (np.sum(h) - target_sum) ** 2
    return sm + penalty

    """Gradient of smooth max objective"""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    vals = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(vals)
    weights = exp_vals / exp_vals.sum()

def slsqp_phase(h, n, target, dx, beta, maxiter, time_fn, min_time):
    """SLSQP with smooth max objective"""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        vals = np.clip(beta * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(vals))) / beta
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        vals = np.clip(beta * (corr - m), -500, 500)
        w = np.exp(vals)
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for k in range(len(corr)):
            if w[k] < 1e-12:
                continue
            lag = k - center
            if lag == 0:
                g += w[k] * (1.0 - 2.0 * x) * dx
            elif 0 < lag < n:
                g[:n-lag] += w[k] * (1.0 - x[lag:]) * dx
                g[lag:] += w[k] * (-x[:n-lag]) * dx
            else:
                l = -lag
                g[:n-l] += w[k] * (-x[l:]) * dx
                g[l:] += w[k] * (1.0 - x[:n-l]) * dx
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, target)
    except:
        return h

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            idx = int(np.argmax(h < 1.0))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def full_gradient(h, dx, top_k=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    if top_k >= 2 * n - 1:
        idxs = np.arange(2 * n - 1)
    else:
        idxs = np.argsort(corr)[::-1][:top_k]
    m = float(corr[idxs[0]])
    beta = 150.0
    vals = np.clip(beta * (corr[idxs] - m), -500, 500)
    w = np.exp(vals)
    w /= w.sum()
    g = np.zeros(n)
    for idx, wi in zip(idxs, w):
        lag = int(idx) - center
        if lag == 0:
            g += wi * (1.0 - 2.0 * h) * dx
        elif 0 < lag < n:
            g[:n - lag] += wi * (1.0 - h[lag:]) * dx
            g[lag:] += -wi * h[:n - lag] * dx
        else:
            l = -lag
            g[:n - l] += -wi * h[l:] * dx
            g[l:] += wi * (1.0 - h[:n - l]) * dx
    return g

def nesterov_phase(h, n, target, dx, max_iters, time_fn, min_time, best_state, top_k=20, momentum=0.85):

        if time_fn() < min_time:
            break
        y = np.clip(cur + momentum * v, 0.0, 1.0)
        g = full_gradient(y, dx, top_k=top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = 0.06 / (1.0 + 0.004 * it)
        improved = False

            c5 = compute_c5(trial)

                v = trial - cur
                cur = trial
                cur_c5 = c5
                improved = True
                if cur_c5 < best_state[0]:

                    best_state[1] = cur.copy()
