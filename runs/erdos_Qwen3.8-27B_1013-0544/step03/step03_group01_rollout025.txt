    # For index i:
    # First term: (1-h[i+m]) where 0<=i+m<n
    term1_i = np.zeros(2*n-1)
    mask1 = (i + m >= 0) & (i + m < n)
    term1_i[mask1] = 1.0 - h[i + m[mask1]]

    # Valid when max(0,m) <= i <= n-1, i.e., m <= i and m >= i-n+1... 
    # Actually: for m>=0, need m<=i (and i<=n-1 which is true)
    # For m<0, need i <= n-1+m (and i>=0 which is true)
    term2_i = np.zeros(2*n-1)
    mask2 = (m <= i) & (m >= i - (n-1))  # i-m >= 0 and i-m < n
    # Wait: 0 <= i-m < n means -n < -m < n-i, i.e., i-n < m <= i
    # So: m > i-n and m <= i, i.e., m >= i-n+1 (since m is integer) and m <= i
    # Actually 0 <= i-m means m <= i, and i-m < n means m > i-n, i.e., m >= i-n+1
    mask2 = (m >= i - (n-1)) & (m <= i)
    # Hmm wait: 0 <= i-m < n
    # i-m >= 0 => m <= i
    # i-m < n => m > i-n => m >= i-n+1 (for integer m)
    # But m ranges from -(n-1) to n-1, so i-n+1 >= -(n-1) means i >= 0 ✓
    # And i <= n-1, so m <= i <= n-1 ✓
    term2_i[mask2] = -h[i - m[mask2]]
    
    # Similarly for j:
    term1_j = np.zeros(2*n-1)
    mask1j = (j + m >= 0) & (j + m < n)
    term1_j[mask1j] = 1.0 - h[j + m[mask1j]]
    
    term2_j = np.zeros(2*n-1)
    mask2j = (m >= j - (n-1)) & (m <= j)
    term2_j[mask2j] = -h[j - m[mask2j]]
    
    return delta * (term1_i + term2_i - term1_j - term2_j)

    return float(np.max(O))

    target = n / 2.0

def compute_swap_D(h, i, j, n):
    """Compute D array for swap: ΔO = delta * D"""
    m = np.arange(-(n-1), n)
    D = np.zeros(2*n-1)
    m1 = (i + m >= 0) & (i + m < n)
    D[m1] += 1.0 - h[i + m[m1]]
    m2 = (i - m >= 0) & (i - m < n)

    D[m3] -= 1.0 - h[j + m[m3]]
    m4 = (j - m >= 0) & (j - m < n)
    D[m4] += h[j - m[m4]]
    return D

def best_delta(O, D, h_i, h_j, n):
    """Find best delta for this swap to minimize max(O + delta*D)"""
    # delta must satisfy: 0 <= h_i + delta <= 1 and 0 <= h_j - delta <= 1
    d_min = max(-h_i, -(1.0 - h_j))  # h_i + delta >= 0 => delta >= -h_i; h_j - delta >= 0 => delta <= h_j
    d_max = min(1.0 - h_i, h_j)  # h_i + delta <= 1 => delta <= 1-h_i; h_j - delta <= 1 => delta >= h_j-1

        return 0.0

    # Also try the point where the two largest O values are equalized
    base = O.copy()

    candidates = np.linspace(d_min, d_max, 50)
    # Add endpoints
    candidates = np.concatenate([candidates, [d_min, d_max, 0.0]])
    candidates = candidates[(candidates >= d_min) & (candidates <= d_max)]

    for d in candidates:
        c = c5_from_O(base + d * D)
        if c < best_c:
            best_c = c
            best_d = d
    return best_d

def sa_loop(h, n, O, deadline, T_start, T_end, n_iter_target, rng):
    """Simulated annealing loop with pair swaps"""
    C = c5_from_O(O)
    best_C = C
    best_h = h.copy()
    
    total_iters = 0
    T = T_start
    
    while time.time() < deadline:
        total_iters += 1
        # Annealing schedule
        frac = min(1.0, total_iters / n_iter_target)
        T = T_start * (T_end / T_start) ** frac

        j = rng.randint(n)

            continue
        
        hi = h[i]
        hj = h[j]
        
        # Determine valid delta range
        d_min = max(-hi, hj - 1.0)
        d_max = min(hj, 1.0 - hi)
        if d_min >= d_max - 1e-15:
            continue
        
        # Pick delta
        # Mix: sometimes random, sometimes optimal
        if rng.random() < 0.3:
            # Try to find good delta
            D = compute_swap_D(h, i, j, n)
            # Find best delta on a coarse grid
            grid = np.linspace(d_min, d_max, 20)
            best_d = 0.0
            best_c = C
            for d in grid:
                c = np.max(O + d * D)
                if c < best_c:
                    best_c = c
                    best_d = d
            delta = best_d
        else:
            # Random delta
            scale = 0.01 + 0.05 * (1 - frac)
            delta = (d_max - d_min) * rng.uniform(0, 1) * (2*rng.random()-1)
            if delta < d_min or delta > d_max:

        D = compute_swap_D(h, i, j, n)
        new_O = O + delta * D
        new_C = float(np.max(new_O))

        if new_C <= C or rng.random() < np.exp(-(new_C - C) / max(T, 1e-10)):

            h[j] -= delta
            O = new_O
            C = new_C
            if C < best_C:
                best_C = C
                best_h = h.copy()
    
    return best_h, best_C, O

If the optimal is binary, it's a subset of 50 positions. The key is to minimize the maximum, over all shifts m, of the number of "1→0" transitions at distance m.

This is related to the concept of a "difference set" or "perfect difference set" in combinatorial design theory. A (v, k, λ) difference set has the property that every non-zero difference occurs exactly λ times. If we could find such a set, then for each shift m≠0, the number of pairs at distance m would be constant.

For v=100, k=50, we'd need λ = k(k-1)/(v-1) = 50*49/99 ≈ 24.75, which is not an integer. So a perfect difference set doesn't exist for these parameters. But we can get close.

Actually, the relationship is more subtle because we're not considering circular differences but linear (one-sided) differences. For a linear array, the number of pairs at distance m depends on the position (pairs near the edges have fewer neighbors).

import numpy as np
import time

def compute_O(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_max(O):
    return float(np.max(O))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

# Precompute shift array (module-level for efficiency)
def get_shifts(n):
    return np.arange(-(n-1), n)

def swap_D(h, i, j, n, shifts):
    """Compute D array: O_new[m] = O[m] + delta * D[m]"""
    m = shifts
    D = np.zeros(len(m))
    m1 = (i + m >= 0) & (i + m < n)
    D[m1] += 1.0 - h[i + m[m1]]
    m2 = (i - m >= 0) & (i - m < n)
    D[m2] -= h[i - m[m2]]
    m3 = (j + m >= 0) & (j + m < n)
    D[m3] -= 1.0 - h[j + m[m3]]
    m4 = (j - m >= 0) & (j - m < n)
    D[m4] += h[j - m[m4]]
    return D

def sa_greedy(h, n, O, deadline, T_start, T_end, rng, greedy_frac=0.0):
    """SA loop. If greedy_frac > 0, after that fraction, only accept improvements."""
    C = c5_max(O)
    best_C = C
    best_h = h.copy()
    shifts = get_shifts(n)
    iters = 0
    T = T_start
    
    while time.time() < deadline:
        iters += 1
        frac = min(1.0, iters / 100000.0)
        T = T_start * (T_end / T_start) ** frac
        if greedy_frac > 0 and frac > greedy_frac:
            T = 0.0
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        hi = h[i]
        hj = h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(hj, 1.0 - hi)
        if d_min >= d_max - 1e-12:
            continue
        
        # Choose delta
        if rng.random() < 0.4 and d_max - d_min > 0.01:
            # Try to find a good delta
            D = swap_D(h, i, j, n, shifts)
            # Sample a few delta values
            n_samples = 8
            deltas = np.linspace(d_min + 1e-6, d_max - 1e-6, n_samples)
            best_d = 0.0
            best_c = C
            for d in deltas:
                c = np.max(O + d * D)
                if c < best_c:
                    best_c = c
                    best_d = d
            delta = best_d
            if abs(delta) < 1e-12:
                continue
        else:
            # Random delta
            span = d_max - d_min
            if span < 1e-12:
                continue
            delta = span * (rng.random() * 2 - 1) * 0.3
            if delta < d_min: delta = d_min
            if delta > d_max: delta = d_max
            if abs(delta) < 1e-12:
                continue
        
        D = swap_D(h, i, j, n, shifts)
        new_C = float(np.max(O + delta * D))
        
        if new_C <= C or (T > 0 and rng.random() < np.exp(-(new_C - C) / T)):
            h[i] += delta
            h[j] -= delta
            O += delta * D
            C = new_C
            if C < best_C:
                best_C = C
                best_h = h.copy()
    
    return best_h, best_C, O

def greedy_local(h, n, O, deadline, rng, n_passes=100):
    """Greedy local search: always accept improvements"""
    C = c5_max(O)
    best_C = C
    best_h = h.copy()
    shifts = get_shifts(n)
    
    for _ in range(n_passes):
        if time.time() > deadline - 5:
            break
        improved = False
        # Try random pairs for speed
        for _ in range(2000):
            if time.time() > deadline - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_min = max(-h[i], h[j]-1.0)
            d_max = min(h[j], 1.0-h[i])
            if d_min >= d_max - 1e-10:
                continue
            D = swap_D(h, i, j, n, shifts)
            # Try optimal delta on a fine grid
            grid = np.linspace(d_min + 1e-8, d_max - 1e-8, 20)
            best_d = 0.0
            best_c = C
            for d in grid:
                c = np.max(O + d * D)
                if c < best_c:
                    best_c = c
                    best_d = d
            if best_d != 0.0 and best_c < C - 1e-12:
                h[i] += best_d
                h[j] -= best_d
                O += best_d * D
                C = best_c
                improved = True
                if C < best_C:
                    best_C = C
                    best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_C, O

def make_binary_start(n, rng, seed_offset=0):

    h = np.zeros(n)
    positions = rng.permutation(n)[:n//2]
