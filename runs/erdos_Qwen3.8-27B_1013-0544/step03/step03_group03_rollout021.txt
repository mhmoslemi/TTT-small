import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(shifted.sum() + 1e-300) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, beta) * dx

def _smooth_grad(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -700, 700))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            m_ = n - lag
            if m_ > 0:
                idx = np.arange(m_)
                grad[idx] += w[j] * (1.0 - x[idx + lag])
                if lag > 0:
                    grad[lag:] -= w[j] * x[:m_]
        else:
            la = -lag
            m_ = n - la
            if m_ > 0:
                idx = np.arange(m_)
                grad[idx + la:idx + la + m_] += w[j] * (1.0 - x[idx])
                grad[:m_] -= w[j] * x[la:la + m_]
    grad *= dx
    return grad

    if k <= 0:
        return np.zeros(n)
    idx = np.argpartition(grad, k)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v
