import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def obj_penalty(h, dx, beta, lam, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max(corr, beta)
    pen = lam * (h.sum() - target_sum) ** 2
    return sm * dx + pen

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones_like(h)

def grad_worst_lag(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_worst = np.argmax(corr)
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad, corr

def grad_topk(h, n, k_top, sharpness):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_top:]
    w = np.exp(sharpness * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += w[j] * (1.0 - h[idx + lag])
            grad[idx + lag] -= w[j] * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += w[j] * (1.0 - h[idx])
            grad[idx] -= w[j] * h[idx + la]
    return grad, corr

def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_func, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0

def lbfgsb_optimize(h0, dx, target_sum, beta, lam, maxiter):
    n = len(h0)
    bnds = list(zip(np.zeros(n), np.ones(n)))
    try:
        res = minimize(obj_penalty, h0, args=(dx, beta, lam, target_sum),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP multi-beta from initial
    for beta in [30, 100, 500, 2000, 8000, 30000]:
        if budget_s - (time.time() - t0) < 30:
            break
        h0 = best_h.copy()
        h_new = slsqp_optimize(h0, dx, target_sum, beta, 4000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
        # Second trial with perturbation
        if budget_s - (time.time() - t0) < 20:
            break
        h0 = best_h + np.random.randn(n) * 0.003
        h0 = project_feasible(h0, target_sum)
        h_new = slsqp_optimize(h0, dx, target_sum, beta, 3000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 2: L-BFGS-B with increasing penalty
    for lam in [10, 100, 1000, 10000]:
        if budget_s - (time.time() - t0) < 30:
            break
        h_new = lbfgsb_optimize(best_h, dx, target_sum, 5000, lam, 3000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 3: Directional subgradient with line search
    if budget_s - (time.time() - t0) > 150:
        h = best_h.copy()
        vel = np.zeros(n)
        beta_mom = 0.85
        stuck = 0
        for it in range(80000):
            if budget_s - (time.time() - t0) < 100:
                break
            grad, corr = grad_topk(h, n, 4, 4.0)
            lr = 0.006 / (1.0 + 0.0005 * it)
            vel = beta_mom * vel - lr * grad
            h_new = project_feasible(h + vel, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
                stuck = 0

                stuck = 0

    # Phase 4: Resolution sweep
    if budget_s - (time.time() - t0) > 100:

            dx_t = 2.0 / n_try

            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            # Optimize at this resolution

                if budget_s - (time.time() - t0) < 30:
                    break
                h_new = slsqp_optimize(h_t, dx_t, ts_t, beta, 3000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                    n = n_try
                    dx = dx_t
                    target_sum = ts_t
                    h_t = h_new
            # Also try L-BFGS-B
            if budget_s - (time.time() - t0) < 20:

            h_new = lbfgsb_optimize(h_t, dx_t, ts_t, 3000, 500, 2000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
                n = n_try
                dx = dx_t
                target_sum = ts_t

    # Phase 5: High-beta SLSQP polish
    for beta in [50000, 200000, 500000]:
        if budget_s - (time.time() - t0) < 15:

            best_h = h_new

    if budget_s - (time.time() - t0) > 30:
        for trial in range(5):

                break

            if c5 < best_c5:

                best_c5 = c5

        no_improve = 0
        for it in range(20000):
            if budget_s - (time.time() - t0) < 2:

                    best_c5 = c_new

                        break

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
