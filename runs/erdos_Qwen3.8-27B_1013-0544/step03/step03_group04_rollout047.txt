import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max_and_grad(corr, beta):
    n = len(corr)
    m = np.max(corr)
    x_sh = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_sh = np.exp(x_sh)
    w = exp_sh / np.sum(exp_sh)
    sm = m + np.log(np.sum(exp_sh) / n) / beta
    return sm, w

def _corr_grad(h, weights):
    n = len(h)
    active = weights > 1e-7
    active_idx = np.where(active)[0]
    grad = np.zeros(n)
    for mi in active_idx:
        lag = mi - (n - 1)
        wi = weights[mi]
        # d c[mi]/d h[i] = I[0<=i-lag<n]*(1-h[i-lag]) - I[0<=i+lag<n]*h[i+lag]
        i_lo = max(0, lag)
        i_hi = min(n - 1, n - 1 + lag)
        if i_lo <= i_hi:

        i_lo2 = max(0, -lag)
        i_hi2 = min(n - 1, n - 1 - lag)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= wi * h[i_lo2+lag:i_hi2+1+lag]
    return grad

def _obj_pen(x, n, target, beta, pen):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, _ = _smooth_max_and_grad(corr, beta)
    return sm * dx + pen * (h.sum() - target) ** 2

def _obj_grad_pen(x, n, target, beta, pen):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = _smooth_max_and_grad(corr, beta)
    obj = sm * dx + pen * (h.sum() - target) ** 2
    gs = _corr_grad(h, w)
    grad = gs * dx + 2.0 * pen * (h.sum() - target) * np.ones(n)
    return obj, grad

def _lbfgs_optimize(h0, n, target, beta, pen, maxiter=500, gtol=1e-10):
    def fun(x):
        return _obj_grad_pen(x, n, target, beta, pen)[0]
    def fprime(x):
        return _obj_grad_pen(x, n, target, beta, pen)[1]
    res = minimize(fun, h0, method='L-BFGS-B', jac=fprime,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': 2000})
    return project_h(res.x, target)

def _subgrad_polish(h, n, target, n_iter=5000, t0=None, budget_s=1100):

    cur = h.copy()
    cur_c5 = compute_c5(cur)
    momentum = np.zeros(n)
    best = cur.copy()
    best_c5 = cur_c5
    for it in range(n_iter):

            break

        gn = np.linalg.norm(grad)
        if gn < 1e-14:

                    best_c5 = cur_c5

def _try_binary_round(h, target):
    n = len(h)
    base_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), base_c5
    for t in np.linspace(0.3, 0.7, 11):

            if len(zeros) == 0:

            order = zeros[np.argsort(h[zeros])[::-1]]
            k = int(np.ceil(deficit))
            if k <= len(order):
                hb[order[:k]] = 1.0
        elif deficit < 0:
