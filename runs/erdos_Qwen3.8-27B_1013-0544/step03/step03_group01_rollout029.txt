import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    return float(m + (s.max() + np.log(np.exp(s).sum())) / alpha)

def soft_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    e = np.exp(s)
    return e / e.sum()

def grad_c(h, n, p):
    """Gradient of the vector c = correlate(h, 1-h, 'full') weighted by p.
    Returns grad[i] = sum_k p[k] * d(c[k])/d(h[i]) for all i.
    p has length 2n-1, indexed 0..2n-2 corresponding to lag -(n-1)..(n-1).
    """
    grad = np.zeros(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # d(c[k])/d(h[i]) = (1-h[i+lag]) for i in [0, n-1-lag]
            #                  - h[i-lag] for i in [lag, n-1]
            for i in range(n - lag):
                grad[i] += pk * (1.0 - h[i + lag])
            for i in range(lag, n):
                grad[i] -= pk * h[i - lag]
        else:
            L = -lag
            for i in range(L, n):
                grad[i] += pk * (1.0 - h[i - L])  # h[i+lag] = h[i-L]
            for i in range(n - L):
                grad[i] -= pk * h[i + L]  # h[i-lag] = h[i+L]
    return grad

def obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_raw = c / dx  # raw correlate values
    val = soft_max(c, alpha)
    p = soft_max_weights(c_raw, alpha)
    g = grad_c(h, n, p) * dx
    return val, g

def slsqp_refine(h, n, alpha, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    def objective(x):
        return obj_and_grad(x, n, alpha)[0]
    def gradient(x):
        return obj_and_grad(x, n, alpha)[1]
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def coord_descent(h, n, lam, max_passes=100, tol=1e-10):
    """Coordinate descent with 1D minimax for each variable."""
    h = h.copy()
    for path in range(max_passes):
        improved = False
        order = np.arange(n)
        for i in order:
            # Compute B[k] for all k
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                b = 0.0
                if 0 <= i + lag < n:
                    b += (1.0 - h[i + lag])
                if 0 <= i - lag < n:
                    b -= h[i - lag]
                B[k] = b
            # C0[k] = c[k] - h[i]*B[k]
            c = np.correlate(h, 1 - h, mode='full')
            C0 = c - h[i] * B
            # Minimize f(t) = max_k(C0[k] + t*B[k]) + lam*t over t in [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(C0 + m1 * B) + lam * m1
                v2 = np.max(C0 + m2 * B) + lam * m2
                if v1 <= v2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) / 2.0
            if abs(new_val - h[i]) > tol:
                h[i] = new_val
                improved = True
        if not improved:
            break
    # Project to satisfy sum constraint exactly
    h = project_feasible(h, n)
    return h

def gradient_descent(h, n, alpha, lam, lr, max_iter, deadline):
    """Projected gradient descent with momentum."""
    h = h.copy()
    v = np.zeros(n)
    mu = 0.9  # momentum
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, g = obj_and_grad(h, n, alpha)
        # Add gradient of constraint penalty
        g = g + lam * np.ones(n)
        v = mu * v + lr * g
        h_new = h - v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        h_new = project_feasible(h_new, n)
        h = h_new
    return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d
                # Avoid O(n^2) being too slow; break early if no improvement found
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    return float(m + (s.max() + np.log(np.exp(s).sum())) / alpha)

def soft_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    e = np.exp(s)
    return e / e.sum()

def weighted_corr_grad(h, n, p):
    """grad[i] = sum_k p[k] * d(c_raw[k])/d(h[i]), p has length 2n-1."""
    grad = np.zeros(n)
    half = n - 1
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-16:
            continue
        lag = k - half
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += pk * (1.0 - h[i + lag])
            for i in range(lag, n):
                grad[i] -= pk * h[i - lag]
        else:
            L = -lag
            for i in range(L, n):
                grad[i] += pk * (1.0 - h[i - L])
            for i in range(n - L):
                grad[i] -= pk * h[i + L]
    return grad

def obj_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    val = soft_max_val(c, alpha)
    p = soft_max_weights(c_raw, alpha)
    g = weighted_corr_grad(h, n, p) * dx
    return val, g

def slsqp_refine(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return obj_grad(x, n, alpha)[0]
    def g(x):
        return obj_grad(x, n, alpha)[1]
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def coord_descent_minimax(h, n, lam, max_passes=200, tol=1e-12):
    """Coordinate descent exploiting 1D minimax structure."""
    h = h.copy()
    half = n - 1
    for _ in range(max_passes):
        changed = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - half
                b = 0.0
                if 0 <= i + lag < n:
                    b += 1.0 - h[i + lag]
                if 0 <= i - lag < n:
                    b -= h[i - lag]
                B[k] = b
            c = np.correlate(h, 1 - h, mode='full')
            C0 = c - h[i] * B
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(C0 + m1 * B) + lam * m1
                v2 = np.max(C0 + m2 * B) + lam * m2
                if v1 <= v2:
                    hi = m2
                else:
                    lo = m1
            new_t = (lo + hi) / 2.0
            if abs(new_t - h[i]) > tol:
                h[i] = new_t
                changed = True
        if not changed:
            break
    return project_feasible(h, n)

def grad_descent_momentum(h, n, alpha, lam, lr, max_iter, deadline):
    h = h.copy()
    v = np.zeros(n)
    mu = 0.85
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        _, g = obj_grad(h, n, alpha)
        g = g + lam * np.ones(n)
        v = mu * v + lr * g
        h = h - v
        h = project_feasible(h, n)
    return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                            improved_any = True
                        else:
                            h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c
