                            h = h_try
                            best_c5 = c5_t
                            improved = True
                            break
                if improved:
                    break
            elif abs(lag) < n:
                al = abs(lag)
                valid = n - al
                for _ in range(8):
                    i = np.random.randint(valid)
                    j = i + al if lag > 0 else i - lag
                    for d in [0.03, -0.03, 0.015, -0.015, 0.01, -0.01, 0.005, -0.005]:
                        h_try = h.copy()
                        h_try[i] += d
                        h_try[j] -= d
                        if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                            h_try = project_h(h_try, target)
                            c5_t = compute_c5(h_try)
                            if c5_t < best_c5:
                                h = h_try
                                best_c5 = c5_t
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
        if not improved and it % 500 == 499:
            p = np.random.randn(n) * 0.006
            h_try = project_h(h + p, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at starting resolution with beta sweep
    n0 = len(best_h)
    target0 = n0 / 2.0
    h_work = best_h.copy()
    for beta in [50, 100, 200, 400, 800, 1500, 3000]:
        if time.time() > t_end - 200:
            break
        h_new, c5_new = slsqp_refine(h_work, target0, n0, beta, 3000, t_end - 200)
        if c5_new < compute_c5(h_work):
            h_work = h_new
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 2: Try other resolutions
    for n_try in [160, 180, 200, 220, 250, 300]:
        if time.time() > t_end - 150:
            break
        target_t = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_h(np.interp(x_new, x_old, best_h), target_t)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_h = h_interp.copy()
            best_c5 = c5_interp
        h_t = h_interp.copy()
        for beta in [100, 300, 800, 2000]:
            if time.time() > t_end - 100:
                break
            h_new, c5_new = slsqp_refine(h_t, target_t, n_try, beta, 2000, t_end - 100)
            if c5_new < compute_c5(h_t):
                h_t = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Phase 3: Subgradient descent
    n_f = len(best_h)
    target_f = n_f / 2.0
    best_h, best_c5 = subgrad_phase(best_h, target_f, t_end - 60, max_iter=40000)

    # Phase 4: Pairwise moves
    best_h, best_c5 = pairwise_phase(best_h, target_f, t_end - 30, max_iter=30000)

    # Phase 5: Final SLSQP polish
    if time.time() < t_end - 8:
        n_p = len(best_h)
        target_p = n_p / 2.0
        h_new, c5_new = slsqp_refine(best_h, target_p, n_p, 5000, 15000, t_end - 3)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Final subgradient sweep
    if time.time() < t_end - 2:
        best_h, best_c5 = subgrad_phase(best_h, target_f, t_end, max_iter=10000)

    return best_h, best_c5, len(best_h)
