        g[:n - lag] -= h[lag:]
    else:
        g[:n + lag] += (1.0 - h[-lag:])
        g[-lag:] -= h[:n + lag]
    return g

def sum_con(x, target):
    return np.sum(x) - target

def sum_con_jac(x):
    return np.ones(x.shape)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    # Phase 1: SLSQP with smooth max, adaptive beta, multiple resolutions
    resolutions = [n_base]
    for mult in [1.5, 2.0, 2.5]:
        nr = int(n_base * mult)
        if nr <= 1000 and nr not in resolutions:
            resolutions.append(nr)
    resolutions.sort()

    for n_pts in resolutions:
        if elapsed() > budget_s - 120:
            break
        target = n_pts / 2.0
        dx = 2.0 / n_pts

        if n_pts != best_n:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_pts)
            h_cur = np.interp(x_new, x_old, best_h)
        else:
            h_cur = best_h.copy()
        h_cur = project_h(h_cur, target)
        cur_c5 = compute_c5(h_cur)

        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (target,)}]

        for beta in [50, 100, 200, 400, 800, 1600, 3200, 6400]:
            if elapsed() > budget_s - 60:
                break

            def obj(x, b=beta):
                corr = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(corr, b) * dx

            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    h_cur = h_new
                    cur_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_new
                    best_n = n_pts
            except Exception:
                pass

    # Phase 2: Active set refinement with explicit constraints
    if elapsed() < budget_s - 100:
        n_cur = len(best_h)
        target = n_cur / 2.0
        dx = 2.0 / n_cur
        h_cur = best_h.copy()
        cur_c5 = best_c5

        for n_active in [5, 7, 10]:
            if elapsed() > budget_s - 50:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            top_idx = np.argsort(corr)[::-1][:n_active]
            current_max = corr.max() * dx

            for t_frac in [1.0, 0.995, 0.99, 0.985]:
                if elapsed() > budget_s - 30:
                    break
                t_val = current_max * t_frac

                def qcon_fun(x, ti=top_idx, tv=t_val):
                    h2 = np.clip(x, 0, 1)
                    c = np.correlate(h2, 1.0 - h2, mode="full")
                    return tv - c[ti] * dx

                def qcon_jac(x, ti=top_idx):
                    h2 = np.clip(x, 0, 1)
                    n = len(h2)
                    J = np.zeros((len(ti), n))
                    for ai, li in enumerate(ti):
                        lag = int(li) - (n - 1)
                        g = corr_grad(h2, lag)
                        J[ai] = -g * dx
                    return J

                cons_q = [{'type': 'ineq', 'fun': qcon_fun, 'jac': qcon_jac}]
                cons_all = cons_q + [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (target,)}]

                def obj_feat(x):
                    return float(np.sum(x * x))

                bounds2 = Bounds(np.zeros(n_cur), np.ones(n_cur))
                try:
                    res = minimize(obj_feat, h_cur, method='SLSQP', bounds=bounds2,
                                  constraints=cons_all, options={'maxiter': 5000, 'ftol': 1e-12, 'iprint': 0})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < cur_c5:
                        h_cur = h_new
                        cur_c5 = c5_new
                    if c5_new < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_new
                        best_n = n_cur
                except Exception:
                    pass

    # Phase 3: Gradient-based local search with momentum
    if elapsed() < budget_s - 15:
        h_cur = best_h.copy()
        n_cur = len(h_cur)
        target = n_cur / 2.0
        dx = 2.0 / n_cur
        cur_c5 = compute_c5(h_cur)
        momentum = np.zeros(n_cur)
        n_top = 15

        for it in range(80000):
            if elapsed() > budget_s - 5:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            beta = 800.0
            w = smooth_max_grad(corr, beta)
            top_pos = np.argsort(w)[::-1][:n_top]

            grad = np.zeros(n_cur)
            for pi in top_pos:
                wi = w[pi]
                if wi < 1e-9:
                    break
                lag = int(pi) - (n_cur - 1)
                grad += wi * corr_grad(h_cur, lag)

            # Project to satisfy sum constraint
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break

            lr = 0.05 / (1.0 + 0.005 * it)
            momentum = 0.75 * momentum - 0.25 * (grad / gn)
            h_trial = project_h(h_cur + lr * momentum, target)
            c5_trial = compute_c5(h_trial)

            if c5_trial < cur_c5 - 1e-16:
                h_cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = cur_c5
                    best_n = n_cur
            else:
                lr *= 0.3
                h_trial = project_h(h_cur + lr * momentum, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-16:
                    h_cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = cur_c5
                        best_n = n_cur
                else:
                    momentum *= 0.5

    # Final: try a few perturbed restarts if time remains
    if elapsed() < budget_s - 30:
        for restart in range(3):
            if elapsed() > budget_s - 20:
                break
            n_r = best_n
            target_r = n_r / 2.0
            dx_r = 2.0 / n_r
            h_r = best_h + 0.05 * np.random.randn(n_r)
            h_r = project_h(h_r, target_r)
            cr = compute_c5(h_r)

            bounds_r = Bounds(np.zeros(n_r), np.ones(n_r))
            cons_r = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (target_r,)}]
            for beta in [200, 500, 1000]:
                if elapsed() > budget_s - 15:
                    break
                def obj_r(x, b=beta):
                    corr = np.correlate(x, 1.0 - x, mode="full")
                    return smooth_max(corr, b) * dx_r
                try:
                    res = minimize(obj_r, h_r, method='SLSQP', bounds=bounds_r,
                                  constraints=cons_r, options={'maxiter': 2000, 'ftol': 1e-14, 'iprint': 0})
                    h_new = project_h(res.x, target_r)
                    c5_new = compute_c5(h_new)
                    if c5_new < cr:
                        h_r = h_new
                        cr = c5_new
                    if c5_new < best_c5:
                        best_h = h_r.copy()
                        best_c5 = c5_new
                        best_n = n_r
                except Exception:
                    pass

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
