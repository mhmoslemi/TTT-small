```python
    n = len(h_bin)
    dx = 2.0 / n
    c5 = compute_c5(h_bin)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for _ in range(max_passes):
        improved = False
        best_delta = 0
        best_swap = None
        for i in ones:
            for j in zeros:
                # Try swapping i and j
                h_bin[i] = 0
                h_bin[j] = 1
                new_c5 = compute_c5(h_bin)
                delta = new_c5 - c5
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_swap = (i, j)
                h_bin[i] = 1
                h_bin[j] = 0
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            c5 = compute_c5(h_bin)
            improved = True
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
```

```python
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def swap_delta(h, i, j):
    """Compute delta in correlation when h[i]: 1->0 and h[j]: 0->1."""
    n = len(h)
    dc = np.zeros(2 * n - 1)
    # h[i]: 1 -> 0
    # As h[n+k], n=i-k: k in [max(0,i-n+1), min(2n-2,i)]
    lo = max(0, i - n + 1)
    hi = min(2 * n - 2, i)
    if hi >= lo:
        nvals = i - np.arange(lo, hi + 1)
        dc[lo:hi + 1] -= (1.0 - h[nvals])
    # In (1-h[n]), n=i: k in [0, n-1-i]
    hi = min(2 * n - 2, n - 1 - i)
    if hi >= 0:
        dc[0:hi + 1] += h[i:i + hi + 1]
    # h[j]: 0 -> 1
    # As h[n+k], n=j-k: k in [max(0,j-n+1), min(2n-2,j)]
    lo = max(0, j - n + 1)
    hi = min(2 * n - 2, j)
    if hi >= lo:
        nvals = j - np.arange(lo, hi + 1)
        dc[lo:hi + 1] += (1.0 - h[nvals])
    # In (1-h[n]), n=j: k in [0, n-1-j]
    hi = min(2 * n - 2, n - 1 - j)
    if hi >= 0:
        dc[0:hi + 1] -= h[j:j + hi + 1]
    return dc


def binary_local_search(h_bin, seed=42, max_time=30.0):
    """First-improvement local search on binary solutions."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy().astype(float)
    c = compute_corr(h)
    c5 = float(np.max(c) * dx)
    best_c5 = c5
    best_h = h.copy()
    t0 = time.time()

    while time.time() - t0 < max_time:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                dc = swap_delta(h, int(i), int(j))
                new_max = float(np.max(c + dc))
                if new_max < c5 - 1e-12:
                    # Accept
                    c += dc
                    h[i] = 0.0
                    h[j] = 1.0
                    c5 = new_max
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                if time.time() - t0 > max_time:
                    break
        if not improved:
            # Stuck - try a random move (simulated annealing style)
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            dc = swap_delta(h, int(i), int(j))
            c += dc
            h[i] = 0.0
            h[j] = 1.0
            c5 = float(np.max(c) * dx)
    return best_h, best_c5


def to_binary(h):
    """Project to binary with exactly n/2 ones."""
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def basin_hop_binary(h_best, best_c5, n_hops, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 30:
            break
        h_bin = to_binary(h_best)
        # Random perturbation: flip k random bits
        n_flips = rng.randint(2, min(8, n // 10))
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        for _ in range(n_flips):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i], h_bin[j] = 0.0, 1.0
            # Update sets
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
        h_new, c_new = binary_local_search(h_bin, seed=seed + hop, max_time=10)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
    return h_best, best_c5
```

Let me just use a simpler approach: 
- For each i in ones (100 choices), pre-compute the "contribution" of flipping i
- For each j in zeros (100 choices), pre-compute the "contribution" of flipping j
- The total delta for swap (i,j) is the sum of the two contributions
- Then I just need to find the (i,j) pair that minimizes max_k (c[k] + contrib_i[k] + contrib_j[k])

The contrib_i[k] (for flipping h[i] from 1 to 0) is the delta from the first two terms in swap_delta.
The contrib_j[k] (for flipping h[j] from 0 to 1) is the delta from the last two terms.

So I have:
- A: 100 x 399 matrix (contrib for each one-position)
- B: 100 x 399 matrix (contrib for each zero-position)
- c: 399 vector

For each (i,j): new_max = max_k (c[k] + A[i,k] + B[j,k])

I want to minimize this over all (i,j).

This is a 2D minimization problem. A naive approach is O(100*100*399) = 4M, which is fast in numpy.

```python
def binary_best_swap(h, c, c5):
    """Find the best swap (i,j) to minimize C5."""
    n = len(h)
    ones = np.where(h == 1)[0].astype(int)
    zeros = np.where(h == 0)[0].astype(int)
    
    # Precompute contributions
    # A[i_idx, k] = delta from flipping h[i] from 1 to 0
    # B[j_idx, k] = delta from flipping h[j] from 0 to 1
    n_ones = len(ones)
    n_zeros = len(zeros)
    corr_len = 2 * n - 1
    
    A = np.zeros((n_ones, corr_len))
    B = np.zeros((n_zeros, corr_len))
    
    for idx, i in enumerate(ones):
        lo = max(0, i - n + 1)
        hi = min(corr_len - 1, i)
        if hi >= lo:
            nvals = i - np.arange(lo, hi + 1)
            A[idx, lo:hi+1] -= (1.0 - h[nvals])
        hi = min(corr_len - 1, n - 1 - i)
        if hi >= 0:
            A[idx, 0:hi+1] += h[i:i+hi+1]
    
    for idx, j in enumerate(zeros):
        lo = max(0, j - n + 1)
        hi = min(corr_len - 1, j)
        if hi >= lo:
            nvals = j - np.arange(lo, hi + 1)
            B[idx, lo:hi+1] += (1.0 - h[nvals])
        hi = min(corr_len - 1, n - 1 - j)
        if hi >= 0:
            B[idx, 0:hi+1] -= h[j:j+hi+1]
    
    # For each (i, j), compute max_k (c[k] + A[i,k] + B[j,k])
    # This is (n_ones x n_zeros) where each entry is a max over corr_len
    # Vectorized: (n_ones, 1, corr_len) + (1, n_zeros, corr_len) + (1, 1, corr_len)
    # = (n_ones, n_zeros, corr_len)
    # Then max over last axis = (n_ones, n_zeros)
    
    total = A[:, np.newaxis, :] + B[np.newaxis, :, :] + c[np.newaxis, np.newaxis, :]
    new_maxes = np.max(total, axis=2)  # shape (n_ones, n_zeros)
    
    # Find the minimum
    flat_idx = np.argmin(new_maxes)
    i_idx = flat_idx // n_zeros
    j_idx = flat_idx % n_zeros
    best_new_c5 = new_maxes[i_idx, j_idx]
    
    return int(ones[i_idx]), int(zeros[j_idx]), float(best_new_c5)
```
