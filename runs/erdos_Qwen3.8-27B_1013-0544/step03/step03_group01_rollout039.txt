def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_max(a, b, lo=0.0, hi=1.0):
    """Find t in [lo, hi] minimizing max_k(a[k]*t + b[k])
    a, b: arrays of slopes and intercepts
    """
    # Evaluate at boundaries
    f_lo = np.max(a * lo + b)
    f_hi = np.max(a * hi + b)
    
    # Check if minimum is at a boundary
    # Subgradient at lo: max of a[k] where a[k]*lo + b[k] is maximal
    vals_lo = a * lo + b
    max_lo = np.max(vals_lo)
    active_lo = np.where(np.abs(vals_lo - max_lo) < 1e-14)[0]
    if len(active_lo) > 0:
        if np.max(a[active_lo]) <= 1e-14:
            return lo, f_lo
    vals_hi = a * hi + b
    max_hi = np.max(vals_hi)
    active_hi = np.where(np.abs(vals_hi - max_hi) < 1e-14)[0]
    if len(active_hi) > 0:
        if np.min(a[active_hi]) >= -1e-14:
            return hi, f_hi
    
    # Bisection
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        vmax = np.max(vals)
        active = np.where(np.abs(vals - vmax) < 1e-13)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        max_slope = np.max(a[active])
        min_slope = np.min(a[active])
        if max_slope <= 1e-14 and min_slope >= -1e-14:
            # 0 in subdifferential
            break
        elif max_slope < 0:
            lo = mid
        else:
            hi = mid
    t_opt = (lo + hi) / 2.0
    return t_opt, np.max(a * t_opt + b)

def pair_cd(h, n, max_cycles, deadline):
    """Pair coordinate descent: for each pair (i,j), find optimal transfer s
    such that h[i] -> h[i]+s, h[j] -> h[j]-s minimizes max overlap."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    for cyc in range(max_cycles):
        if time.time() > deadline:
            break
        improved = False
        # Randomize pair order
        indices = list(range(n))
        np.random.shuffle(indices)  # Use a fixed rng
        
        for ii, i in enumerate(indices):
            if time.time() > deadline:
                break
            for jj in range(ii + 1, n):
                j = indices[jj]
                # Compute coefficients for transfer s from j to i
                # h[i] -> h[i] + s, h[j] -> h[j] - s
                # O_d(s) = O_d + c_d * s
                # c_d = (1-h[i+d])*I(i+d valid) - (1-h[j+d])*I(j+d valid) 
                #       - h[i-d]*I(i-d valid) + h[j-d]*I(j-d valid)
                
                # Valid range for d in correlate(h, 1-h, 'full'):
                # d ranges from -(n-1) to (n-1), index = d + (n-1)
                offset = n - 1
                c_d = np.zeros(n_lags)
                for d in range(-n + 1, n):
                    idx = d + offset
                    # i+d valid?
                    if 0 <= i + d < n:
                        c_d[idx] += (1.0 - h[i + d])
                    # j+d valid?
                    if 0 <= j + d < n:
                        c_d[idx] -= (1.0 - h[j + d])
                    # i-d valid? (h[i-d]*(1-h[i]) term: derivative w.r.t. h[i] is -h[i-d])
                    if 0 <= i - d < n:
                        c_d[idx] -= h[i - d]
                    # j-d valid? (h[j-d]*(1-h[j]) term: derivative w.r.t. h[j] is -h[j-d], but h[j] decreases)
                    if 0 <= j - d < n:
                        c_d[idx] += h[j - d]
                
                O_d = np.correlate(h, 1 - h, mode='full')
                # O_d(s) = O_d + c_d * s
                # Constraint: 0 <= h[i]+s <= 1 and 0 <= h[j]-s <= 1
                # => s >= -h[i], s <= 1-h[i], s >= h[j]-1, s <= h[j]
                s_lo = max(-h[i], h[j] - 1.0)
                s_hi = min(1.0 - h[i], h[j])
                
                if s_hi - s_lo < 1e-15:
                    continue
                
                s_opt, _ = solve_1d_max(c_d, O_d, s_lo, s_hi)
                
                new_hi = h[i] + s_opt
                new_hj = h[j] - s_opt
                if 0 <= new_hi <= 1 and 0 <= new_hj <= 1:
                    h[i] = new_hi
                    h[j] = new_hj
                    improved = True
        
        if not improved:
            break
    
    c_final = compute_c5(h, n)
    return h, c_final
