def pair_local(h, n, deadline, deltas):

    improved = True
    while improved and time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            oi, oj = h[i], h[j]
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                improved = True
                            else:
                                h[i] = oi
                                h[j] = oj
    return h, best_c
