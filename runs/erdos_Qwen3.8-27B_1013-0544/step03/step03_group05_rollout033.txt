```python
def project_h(h, target_sum):

    for _ in range(200):

        if abs(diff) < 1e-14:

            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def grad_for_lag(h, lag, n, dx):

    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n-l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n-l] * dx
    return grad

def nesterov_phase(h, target, n, dx, max_iters, time_fn, min_time, best_state):
    """Nesterov-accelerated projected gradient on smooth max."""
    cur = h.copy()
    y = h.copy()  # Nesterov extrapolation point
    momentum = 0.9
    cur_c5 = compute_c5(cur)
    beta = 100.0
    
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        
        # Compute gradient at y
        corr_y = np.correlate(y, 1.0 - y, mode="full") * dx
        m = np.max(corr_y)
        logits = np.clip(beta * (corr_y - m), -500, 500)
        weights = np.exp(logits)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        center = n - 1
        for k in range(2*n-1):
            if weights[k] < 1e-12:
                continue
            lag = k - center
            grad += weights[k] * grad_for_lag(y, lag, n, dx)
        
        # Step
        step = 1.0 / (beta + it * 0.1)
        new_cur = project_h(cur - step * grad, target)
        
        # Nesterov update
        t_val = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
        y = new_cur + momentum * (new_cur - cur)
        y = np.clip(y, 0, 1)
        
        cur_c5_new = compute_c5(new_cur)
        if cur_c5_new < cur_c5:
            cur = new_cur
            cur_c5 = cur_c5_new
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n
        else:
            # Revert
            y = cur.copy()
    
    return cur, cur_c5
```

Wait, one more idea: **alternating projections / Frank-Wolfe**. The Frank-Wolfe algorithm (a.k.a. conditional gradient) for minimizing a smooth function over a polytope:
- At each iteration, solve a linear oracle: min_{v in polytope} ∇f(x)^T v
- Take a step towards the solution

The polytope here is {h : 0 ≤ h ≤ 1, ∑h = n/2}. The linear oracle for min ∇f(x)^T v subject to 0≤v≤1, ∑v=n/2 is:
- Sort the gradient components
- Set v[i] = 1 for the smallest gradients, v[i] = 0 for the largest
- Set v[i] = θ for one coordinate (to exactly satisfy the sum constraint)

This is a "thresholding" operation. Let me implement this:

```python
def linear_oracle(grad, target, n):
    """Solve min grad^T v s.t. 0<=v<=1, sum(v)=target"""
    order = np.argsort(grad)
    # Greedy: assign 1 to smallest gradient components
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining >= 1.0:
            v[idx] = 1.0
            remaining -= 1.0
        elif remaining > 0:
            v[idx] = remaining
            remaining = 0
            break
        else:
            break
    return v
```

Wait, this isn't quite right. The linear oracle for the constraint set {v: 0≤v≤1, ∑v=target} is:
- Sort grad in ascending order: grad[σ(1)] ≤ grad[σ(2)] ≤ ... ≤ grad[σ(n)]
- Set v[σ(1)] = ... = v[σ(k)] = 1, v[σ(k+1)] = θ, v[σ(k+2)] = ... = v[σ(n)] = 0
- Where k = floor(target) and θ = target - k (if target is not integer)

Actually, target = n/2, which is an integer (when n is even). So v = 1 for the n/2 smallest gradient components, 0 for the rest.

This is the "hard thresholding" Frank-Wolfe step! Let me use this.

Frank-Wolfe iteration:
1. Compute gradient g = ∇f(x) (or subgradient if f is not smooth)
2. v = linear_oracle(g) (the "vertex" of the polytope)
3. x_new = (1-γ)x + γv where γ = 2/(t+2) (FW step size)
4. Repeat

This is guaranteed to converge, and it's very simple! The advantage is that it naturally handles the constraints.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n-l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n-l] * dx
    return grad

def smooth_obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    logits = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(logits)
    w /= w.sum()
    n = len(h)
    center = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        lag = k - center
        grad += w[k] * grad_for_lag(h, lag, n, dx)
    return smooth_max_val(corr, beta) * 0 + np.max(corr) + np.log(np.sum(np.clip(np.exp(logits), 1e-300, 1e300)) / len(corr)) / beta, grad

def smooth_max_val(corr, beta):
    m = np.max(corr)
    logits = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(logits)) / len(corr)) / beta

def fw_organic(h, target, n, dx, max_iters, time_fn, min_time, best_state, beta=100.0):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for t in range(max_iters):
        if time_fn() < min_time:
            break
        gamma = 2.0 / (t + 2.0)
        obj_val, grad = smooth_obj_grad(cur, dx, beta)
        order = np.argsort(grad)
        v = np.zeros(n)
        k = min(int(target), n)
        v[order[:k]] = 1.0
        new = (1.0 - gamma) * cur + gamma * v
        new = np.clip(new, 0, 1)
        new = project_h(new, target)
        c5_new = compute_c5(new)
        if c5_new < cur_c5 - 1e-16:
            cur = new
            cur_c5 = c5_new
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n
    return cur, cur_c5

def cd_pass(h, target, time_fn, min_time, max_sweeps=150):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    order = np.random.permutation(n)
    for sweep in range(max_sweeps):
        if time_fn() < min_time:
            break
        improved = False
        for i in order:
            old = cur[i]
            cands = [0.0, 1.0, old * 0.5, (old + 1.0) * 0.5,
                     max(0.0, old - 0.1), min(1.0, old + 0.1)]
            cands = sorted(set(np.clip(cands, 0, 1)))
            best_c = cur_c5
            best_h = None
            for t in cands:
                if abs(t - old) < 1e-10:
                    continue
                trial = cur.copy()
                trial[i] = t
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = trial.copy()
            if best_h is not None:
                cur = best_h
                cur_c5 = best_c
                improved = True
        if not improved:
            break
    return cur, cur_c5

def pair_pass(h, target, time_fn, min_time, max_iters=5000):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        d_min = max(-cur[i], cur[j] - 1.0)
        d_max = min(1.0 - cur[i], cur[j])
        if d_max - d_min < 1e-10:
            continue
        span = d_max - d_min
        d_cands = [0.3 * span, -0.3 * span, 0.7 * span, -0.7 * span, span, -span]
        best_c = cur_c5
        best_h = None
        for dc in d_cands:
            d = 0.5 * span + dc
            if d <= d_min + 1e-12 or d >= d_max - 1e-12:
                continue
            trial = cur.copy()
            trial[i] += d
            trial[j] -= d
            trial = np.clip(trial, 0, 1)
            c = compute_c5(trial)
            if c < best_c - 1e-15:
                best_c = c
                best_h = trial.copy()
        if best_h is not None:
            cur = best_h
            cur_c5 = best_c
    return cur, cur_c5

    center = n - 1
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        top_k = min(4, 2 * n // 100 + 2)
        top_idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[top_idx]
        temp = 200.0
        w = np.exp((vals - vals[0]) * temp)
        w /= w.sum()
        grad = np.zeros(n)
        for t_idx, wi in zip(top_idx, w):
            lag = int(t_idx) - center
            grad += wi * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        step = 0.05 / (1.0 + 0.002 * it)
        for _ in range(20):
            trial = project_h(cur - step * grad, target)

                    best_state[1] = cur.copy()
                    best_state[2] = n

            step *= 0.5
    return cur, cur_c5
```
