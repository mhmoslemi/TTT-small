def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)

        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])

    return val, grad

        h_new = project_feasible(h_new, target)
        h_cur = h_new
    return best_h

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

        return smoothmax_val_grad(x, beta, top_k)[1]

                       options={'maxiter': maxiter, 'ftol': 1e-16})

        return project_feasible(h, target)

        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def _constraint_fun(x, target):
    return np.sum(x) - target


def _constraint_jac(x, target):
    return np.ones_like(x)


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = {'type': 'eq', 'fun': obj, 'jac': objg}
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def _slsqp_obj(x, beta, target, top_k):
    return smoothmax_val_grad(x, beta, top_k)[0]


def _slsqp_grad(x, beta, target, top_k):
    return smoothmax_val_grad(x, beta, top_k)[1]


def slsqp_solve2(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        return g

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def momentum_optimize(h, beta, max_iter=5000, lr=0.05, momentum=0.9,
                      top_k=None, t0=None, budget_s=None, patience_frac=0.2):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    v = np.zeros(n)
    val0, _ = smoothmax_val_grad(h_cur, beta, top_k)
    best_val = val0
    best_h = h_cur.copy()
    patience = 0
    max_patience = max(100, int(max_iter * patience_frac))
    for it in range(max_iter):
        if t0 is not None and budget_s - (time.time() - t0) < 5:
            break
        val, grad = smoothmax_val_grad(h_cur, beta, top_k)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_cur.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        h_cur = h_new
    return best_h


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)

    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Quick SLSQP continuation
    for beta in [10, 50, 150]:
        if time_left() < 80:
            break
        h_new = slsqp_solve2(h, beta, 250, top_k=None)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        h = h_new

    # Phase 2: Momentum optimization with increasing beta (fast iterations)
    h = best_h.copy()
    momentum_betas = [(50, 0.1, 0.85), (100, 0.08, 0.88), (200, 0.05, 0.90),
                      (400, 0.03, 0.92), (600, 0.02, 0.94)]
    for beta, lr, mom in momentum_betas:
        if time_left() < 50:
            break
        h = momentum_optimize(h, beta, max_iter=4000, lr=lr, momentum=mom,
                              top_k=None, t0=t0, budget_s=budget_s)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: Basin hopping with momentum local optimizer
    rng = np.random.RandomState(seed + 7)
    n_hops = 30
    noise_base = 0.04
    noise_decay = 0.88
    for hop in range(n_hops):
        if time_left() < 35:
            break
        noise = noise_base * (noise_decay ** hop)
        # Mix of Gaussian and structured perturbations
        if hop % 3 == 0:
            h_pert = best_h + rng.randn(n) * noise
        elif hop % 3 == 1:
            # Sinusoidal perturbation
            freq = rng.randint(2, max(3, n // 10)) + 1
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = best_h + noise * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        else:
            # Sparse perturbation
            n_flip = max(5, n // 10)
            idx = rng.choice(n, size=n_flip, replace=False)
            h_pert = best_h.copy()
            h_pert[idx] += rng.uniform(-2 * noise, 2 * noise, size=n_flip)
        h_pert = project_feasible(h_pert, target)
        h_local = h_pert
        for beta in [150, 400]:
            if time_left() < 15:
                break
            h_local = momentum_optimize(h_local, beta, max_iter=2500, lr=0.04,
                                        momentum=0.91, t0=t0, budget_s=budget_s)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            best_h = h_local.copy()

    # Phase 4: Try different resolutions
    for n_try in [150, 250, 300, 400, 500]:
        if time_left() < 100:
            break
        h_interp = interp_to_n(best_h, n_try)
        h_c = h_interp
        for beta in [50, 200, 500]:
            if time_left() < 40:
                break
            h_c = momentum_optimize(h_c, beta, max_iter=3000, lr=0.04,
                                    momentum=0.9, t0=t0, budget_s=budget_s)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_c.copy()
            n = len(best_h)
            target = n / 2.0

    # Phase 5: Fine basin hopping at best resolution
    rng2 = np.random.RandomState(seed + 100)
    for hop in range(15):
        if time_left() < 25:
            break
        noise = 0.015 * (0.85 ** hop)
        h_pert = best_h + rng2.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = momentum_optimize(h_pert, 500, max_iter=3000, lr=0.03,
                                    momentum=0.93, t0=t0, budget_s=budget_s)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            best_h = h_local.copy()

    # Phase 6: Final SLSQP polish
    if time_left() > 25:
        h_fin = slsqp_solve2(best_h, 500, 400, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    # Phase 7: Ultra-fine momentum
    if time_left() > 20:
        h_fin2 = momentum_optimize(best_h, 800, max_iter=6000, lr=0.015,
                                   momentum=0.95, t0=t0, budget_s=budget_s)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            best_h = h_fin2

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
