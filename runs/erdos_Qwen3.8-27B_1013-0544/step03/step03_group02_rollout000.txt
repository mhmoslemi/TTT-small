        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        length = n - lag
        if length > 0:
            grad[:length] += (1.0 - h[lag:lag + length])
            grad[lag:lag + length] -= h[:length]
    else:
        la = -lag
        length = n - la
        if length > 0:
            grad[la:la + length] += (1.0 - h[:length])
            grad[:length] -= h[la:la + length]
    return grad

def gradient_top_k(h, top_idx, n):
    g = np.zeros(n)
    n_minus_1 = n - 1
    for idx in top_idx:
        lag = int(idx) - n_minus_1
        g += grad_for_lag(h, lag, n)
    return g

def projected_gradient_opt(h, target_sum, n, t0, budget, max_iter=30000):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    step = 0.008
    mom = np.zeros(n)
    mood = 0.8
    stall = 0
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = compute_correlations(h)
        K = max(8, min(n // 2, int(3 * n * np.exp(-it / 2000.0) + 3)))
        top_idx = np.argsort(corr)[::-1][:K]
        
        g = gradient_top_k(h, top_idx, n) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = mood * mom - step * g
        
        # Backtracking
        accepted = False
        for bt in range(30):
            trial = project_h(h + mom, target_sum)
            c_trial = compute_c5(trial)
            if c_trial < best_c5 - 1e-15:
                h = trial
                best_c5 = c_trial
                best_h = h.copy()
                step = min(step * 1.02, 0.03)
                accepted = True
                break
            mom = mood * mom - (step * 0.5) * g
        if not accepted:
            stall += 1
            step *= 0.6
            if stall > 50:
                mom = np.zeros(n)
                step = 0.008
                stall = 0
            elif stall > 300:
                break
        else:
            stall = 0
    return best_h, best_c5

def coord_descent_fine(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def make_square_wave(n, period_frac, duty_cycle=0.5):
    x = np.linspace(0, 2, n)
    t = (x / (2.0 * period_frac)) % 1.0
    h = np.where(t < duty_cycle, 1.0, 0.0).astype(float)
    return h

def make_sawtooth(n, periods):
    x = np.linspace(0, 2, n)
    t = (periods * x / 2.0) % 1.0
    return np.clip(t * 2.0 - 0.5, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(160, 0.5)
    
    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base
    
    # Phase 1: Explore at multiple resolutions
    resolutions = []
    for r in [64, 96, 128, 160, 200, 240, 320, 384]:
        if r not in resolutions:
            resolutions.append(r)
    if n_base not in resolutions:
        resolutions.append(n_base)
    resolutions.sort()
    
    # Generate diverse initializations
    init_candidates = []
    init_candidates.append(('parent', project_h(resample_h(h_parent, resolutions[0]), resolutions[0] / 2.0)))
    
    for r in resolutions[:4]:
        for pf in [0.25, 0.33, 0.4, 0.5, 0.6, 0.75]:
            h_sq = make_square_wave(r, pf)
            h_sq = project_h(h_sq, r / 2.0)
            init_candidates.append(('sq_f' + str(pf), h_sq))
        for n_sp in [2, 3, 5, 8]:
            h_st = make_sawtooth(r, n_sp)
            h_st = project_h(h_st, r / 2.0)
            init_candidates.append(('st_n' + str(n_sp), h_st))
        h_rand = project_h(np.random.rand(r) * 0.6 + 0.2, r / 2.0)
        init_candidates.append(('rand', h_rand))
    
    # Phase 2: Optimize diverse candidates at low-to-mid resolution
    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 30:
            break
        target = n_pts / 2.0
        time_per_cand = max(5, min(25, remaining / max(1, len(init_candidates))))
        
        for idx, (name, h_init) in enumerate(init_candidates):
            if len(h_init) != n_pts:
                h_init = project_h(resample_h(h_init, n_pts), target)
            elif 'parent' not in name:
                h_init = project_h(h_init, target)
            
            elapsed = time.time() - t0
            if elapsed > budget_s - 15:
                break
            
            h_opt, c5_opt = projected_gradient_opt(h_init, target, n_pts, t0, budget_s - 15)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n_pts
                # Also update parent for future upscaling
                h_parent = h_opt.copy()
        
        # Also optimize the best found so far at this resolution
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            h_cur = project_h(resample_h(best_h, n_pts), target)
            h_opt, c5_opt = projected_gradient_opt(h_cur, target, n_pts, t0, budget_s - 10)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n_pts
                h_parent = h_opt.copy()
    
    # Phase 3: High-resolution refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        for n_pts in [best_n, best_n + 64, best_n + 128]:
            if n_pts > 512 or remaining < 15:
                break
            target = n_pts / 2.0
            h_cur = project_h(resample_h(best_h, n_pts), target)
            h_opt, c5_opt = projected_gradient_opt(h_cur, target, n_pts, t0, budget_s - 5)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n_pts
            remaining = budget_s - (time.time() - t0)
    
    # Phase 4: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        n_f = len(best_h)
        t_f = n_f / 2.0
        best_h, best_c5 = coord_descent_fine(best_h, t_f, n_f, 30, t0, budget_s - 2)
    
    # Phase 5: Final fine-tuning with very small steps
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        n_f = len(best_h)
        t_f = n_f / 2.0
        # Extra fine coordinate descent
        for j in range(n_f):
            if remaining - (time.time() - t0) < 1:
                break
            for d in [0.003, -0.003, 0.001, -0.001]:
                trial = best_h.copy()
                trial[j] += d
                trial = project_h(trial, t_f)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c_try
                    break
    
    # Final check and cleanup
    n_final = len(best_h)
    t_final = n_final / 2.0
    best_h = project_h(best_h, t_final)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_final
