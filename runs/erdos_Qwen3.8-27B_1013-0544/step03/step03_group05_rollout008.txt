import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n_ones):
    n = len(h)
    n_ones = min(n_ones, n)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgradient_polish(h, n, target, dx, max_iters=5000, time_fn=None, min_time=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    for it in range(max_iters):
        if time_fn is not None and time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, n // 30 + 3)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 200.0 + 100.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            if lag == 0:
                grad[:] = (1.0 - 2.0 * cur) * dx
            elif lag > 0:
                grad[lag:] += (1.0 - cur[:n - lag]) * dx
                grad[:n - lag] += -cur[lag:] * dx
            else:
                l = -lag
                grad[:n - l] += (1.0 - cur[l:]) * dx
                grad[l:] += -cur[:n - l] * dx
            grad *= w
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        step = 0.04 / (1.0 + 0.005 * it)
        improved = False
        for _ in range(30):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)

                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                break

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

                # Complement
                h_start = 1.0 - h_bin
                # Ensure correct sum (complement of n/2 ones is n/2 zeros, so sum is still n/2)
                h_start = np.where(h_start > 0.5, 1.0, 0.0)
            else:
                # Random binary with correct sum
                h_start = np.zeros(n)
                positions = np.random.choice(n, int(target), replace=False)
                h_start[positions] = 1.0

            h_b, c5_b = binary_local_search(
                h_start, n, max_sweeps=300,
                time_fn=time_left, min_time=30, seed=seed + attempt
            )
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n

    # Phase 2: SLSQP smooth max refinement
    if time_left() > 80:
        h_work = best_h.copy()
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [100, 300, 800, 2000, 5000]:
            if time_left() < 15:
                break
            try:
                obj = lambda x: obj_smooth_max(x, dx, beta)
                cons = [{'type': 'eq',
                         'fun': lambda x: np.sum(x) - target,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(obj, h_work, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                h_work = h_new
            except Exception:
                pass

    # Phase 3: Subgradient polish
    if time_left() > 30:
        h_pol, c5_pol = subgradient_polish(
            best_h, best_n, best_n / 2.0, 2.0 / best_n,
            max_iters=10000, time_fn=time_left, min_time=10
        )
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = len(h_pol)

    # Phase 4: Try a different resolution (n=200)
    if time_left() > 60:
        n2 = 200
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        h2 = project_h(resample(best_h, n2), target2)
        h2_bin = to_binary(h2, int(target2))
        if time_left() > 40:
            h2_b, c5_2b = binary_local_search(
                h2_bin, n2, max_sweeps=200,
                time_fn=time_left, min_time=15, seed=seed + 100
            )
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2_b.copy()
                best_n = n2
        if time_left() > 20:
            h_pol2, c5_pol2 = subgradient_polish(
                best_h, best_n, best_n / 2.0, 2.0 / best_n,
                max_iters=5000, time_fn=time_left, min_time=5
            )
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
                best_n = len(h_pol2)

    # Final: ensure feasibility
    best_h = project_h(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        # Projection slightly hurt, try to fix
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
