```python
def smooth_max(x, beta):
    m = np.max(x)
    if beta * (np.max(x) - m) > 500:
        beta = 500 / (np.max(x) - m) if np.max(x) > m else beta
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def sml_objective(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

    return np.sum(h) - target

def sum_jac(h, target):
    return np.ones(len(h))
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def topk_smooth_max(x, beta, k):
    """Smooth max focusing on top k values."""
    # Sort and take top k
    idx = np.argpartition(x, -k)[-k:]
    top = x[idx]
    m = np.max(top)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + (1.0/beta) * np.log(np.sum(np.exp(shifted)) / len(x))

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

    c = np.correlate(h, 1.0 - h, mode="full")
    if topk and topk < len(c):
        return topk_smooth_max(c, beta, topk) * dx
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -500, 500)
    return (m + (1.0/beta)*np.log(np.sum(np.exp(shifted))/len(c))) * dx

def sum_eq(h, target):
    return np.sum(h) - target

def sum_eq_jac(h, target):
    return np.ones(len(h))

    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except:
        return h0

        anti_grad[lag:] -= (1.0 - h[:n-lag])

        anti_grad[la:] += h[:n-la]
        anti_grad[:n-la] -= (1.0 - h[la:])
    
    hp += 0.01 * anti_grad
    return project_h(hp, target)
```

```python
    weights[top_idx] *= beta * 0.5
    wsum = np.sum(weights)
    return m + (1.0 / beta) * np.log(np.sum(weights * np.exp(shifted)) / wsum)

def obj_fn(h, beta, dx, mode):
    c = np.correlate(h, 1.0 - h, mode="full")
    if mode == 0:
        return smooth_max_all(c, beta) * dx
    else:
        k = mode
        return smooth_max_topk(c, beta, k) * dx

    return np.sum(h) - target

def sum_eq_jac(h, target):
    return np.ones(len(h))

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol, mode=0):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, 2.0 / n, mode), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except:
        return h0

def make_perturbation(h, n, target, s):
    """Generate diverse perturbations."""
    if s % 5 == 0:
        return project_h(h + np.random.randn(n) * 0.02, target)
    elif s % 5 == 1:
        f = np.random.randint(2, 8)
        p = np.random.uniform(0, 2 * np.pi)
        return project_h(h + 0.03 * np.sin(2 * np.pi * f * np.linspace(0, 2, n) + p), target)
    elif s % 5 == 2:
        hp = h.copy()
        blk = np.random.permutation(n)[:n // 4]
        hp[blk] += np.random.choice([-1, 1]) * 0.05
        return project_h(hp, target)
    elif s % 5 == 3:
        # Binary push
        hp = h.copy()
        hp += 0.15 * np.sign(hp - 0.5) * np.random.random(n)
        return project_h(hp, target)
    else:
        # Informed by worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst = int(np.argmax(corr))
        lag = worst - (n - 1)
        hp = h.copy()
        if lag >= 0 and lag < n:
            hp[:n - lag] -= 0.02 * h[lag:]
            hp[lag:] += 0.02 * (1.0 - h[:n - lag])
        elif lag < 0:
            la = -lag
            hp[la:] -= 0.02 * h[:n - la]
            hp[:n - la] += 0.02 * (1.0 - h[la:])
        return project_h(hp + np.random.randn(n) * 0.01, target)

def coord_polish(cur, target, t0, budget, max_sweeps=30):
    n = len(cur)
    best_c5 = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005)
    for _ in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(cur[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def subgrad_refine(cur, target, t0, budget, n_iter, step0):
    n = len(cur)
    best = compute_c5(cur)
    mom = np.zeros(n)
    stall = 0
    step = step0
    for _ in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        # Top 6 lags
        top_idx = np.argsort(corr)[::-1][:6]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.82 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best - 1e-14:
            cur = trial
            best = c_try
            stall = 0
            step = min(step * 1.02, 0.08)
        else:
            stall += 1
            step *= 0.85
            mom *= 0.4
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best - 1e-14:
                cur = trial
                best = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 80:
            cur = project_h(cur + np.random.randn(n) * 0.015, target)
            mom = np.zeros(n)
            stall = 0
    return cur, best
```

Wait, I need to double-check my `smooth_max_topk` function. The idea is to weight the top-k values more heavily in the log-sum-exp, making the gradient focus on the worst lags.

Actually, let me reconsider. The standard smooth max is:
smooth_max(x) ≈ m + (1/β) log(Σ exp(β(x_i - m)))

This gives equal weight to all x_i in the exponential. The top-k variant should give MORE weight to the top-k values.

A cleaner way: 
- Compute the log-sum-exp only over the top-k values
- This gives a differentiable approximation to max of top-k

Let me simplify: just use the standard smooth max but with a larger beta for the "focus" effect. A larger beta already concentrates the gradient on the max.

Actually, the issue with the standard smooth max is:
- Small β: gradient spread over all lags → too diffuse
- Large β: gradient on single worst lag → can oscillate

The top-k approach gives a middle ground: gradient on top-k lags.

Let me implement it properly:

```python
def smooth_max_topk(x, beta, k):
    """Approximate max of top-k elements using log-sum-exp on top-k."""
    # Get indices of top-k
    if k >= len(x):
        k = len(x)
    top_idx = np.argpartition(x, -k)[-k:]
    top_vals = x[top_idx]
    m = np.max(top_vals)
    shifted = beta * (top_vals - m)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)) / k)
```

This is differentiable with respect to x (the gradient is concentrated on the top-k elements). But for SLSQP, we need the gradient of the FULL function, which scipy computes numerically. So the top-k selection inside the objective is fine - scipy will compute the numerical gradient correctly.

Actually wait - there's a subtlety. If we select top-k based on the current x, and x changes slightly, the selection might change discontinuously. This could cause issues with numerical gradient.

A safer approach: use a "soft" top-k selection:
```python
def smooth_max_topk(x, beta, k):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    # Soft weight: emphasize high values
    weights = np.exp(shifted * 0.5)  # Softer emphasis
    wsum = np.sum(weights)
    return m + (1.0 / beta) * np.log(np.sum(weights * np.exp(shifted)) / wsum)
```
