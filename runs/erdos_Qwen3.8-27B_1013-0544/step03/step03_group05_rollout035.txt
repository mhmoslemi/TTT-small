        B[valid_vj] -= (1.0 - h[vj_idx[valid_vj]])
    if valid_uj.any():
        B[valid_uj] += h[uj_idx[valid_uj]]
    
    # Minimize max_m(corr_full[m] + s*B[m]) over s in [s_lo, s_hi]
    def f_of_s(s_val):
        return np.max(corr_full + s_val * B)
    
    res = minimize_scalar(f_of_s, bounds=(s_lo, s_hi), method='bounded',
                          options={'xtol': 1e-12})
    s_opt = res.x
    
    delta = s_opt * B
    return hi + s_opt, s_opt, delta
