    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    w_sel = exps / sw

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_TARGET_SUM = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def project_waterfill(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(len(corr))
    c_sel = corr[idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wk = w_sel[ii]
        if wk < 1e-14:
            continue
        k = idx[ii]
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def _cons_fun(x):
    return np.sum(x) - _TARGET_SUM

def _cons_jac(x):
    return np.ones(len(x))

class SmoothMaxObj:
    def __init__(self, beta, top_k):
        self.beta = beta
        self.top_k = top_k
    def value(self, x):
        return smoothmax_val_grad(x, self.beta, self.top_k)[0]
    def gradient(self, x):
        return smoothmax_val_grad(x, self.beta, self.top_k)[1]

def slsqp_solve(h, beta, maxiter, top_k):
    global _TARGET_SUM
    n = len(h)
    _TARGET_SUM = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac})
    of = SmoothMaxObj(beta, top_k)
    try:
        res = minimize(of.value, h, jac=of.gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)

def nesterov_optimize(h0, beta, n_steps, lr, top_k):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    x_prev = h0.copy()
    t = 1.0
    best_val = smoothmax_val_grad(x, beta, top_k)[0]
    for k in range(n_steps):
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x + momentum * (x - x_prev)
        y = project_waterfill(y, target)
        val, grad = smoothmax_val_grad(y, beta, top_k)
        step = lr
        x_new = project_waterfill(y - step * grad, target)
        val_new, _ = smoothmax_val_grad(x_new, beta, top_k)
        if val_new > val + 1e-12:
            step *= 0.5
            x_new = project_waterfill(y - step * grad, target)
            val_new, _ = smoothmax_val_grad(x_new, beta, top_k)
        if val_new < best_val:
            best_val = val_new
        x_prev = x
        x = x_new
        t = t_new
    return x

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def block_perturb(h, rng, n_blocks, min_bs, max_bs, scale):
    n = len(h)
    target = n / 2.0
    h_pert = h.copy()
    for _ in range(n_blocks):
        bs = rng.randint(min_bs, max_bs + 1)
        start = rng.randint(0, max(1, n - bs))
        end = min(start + bs, n)
        if end > start:
            h_pert[start:end] += rng.randn(end - start) * scale
    return project_feasible(h_pert, target)

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET_SUM
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)
        h_cur = project_feasible(h_cur, 100)

    n = len(h_cur)
    target = n / 2.0
    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: SLSQP continuation with increasing beta
    for beta in [40, 100, 300, 600, 1000]:
        if budget_s - (time.time() - t0) < 60:
            break
        h_new = slsqp_solve(h_cur, beta, 500, top_k=None)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: Nesterov accelerated gradient with adaptive beta
    for beta in [300, 700, 1500]:
        if budget_s - (time.time() - t0) < 80:
            break
        h_cur = nesterov_optimize(h_cur, beta, 8000, 0.008, top_k=30)
        h_cur = slsqp_solve(h_cur, beta, 300, top_k=None)
        c_new = compute_c5(h_cur)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_cur.copy()

    # Phase 3: Block-based basin hopping (structured perturbations)
    rng = np.random.RandomState(seed)
    for hop in range(40):
        if budget_s - (time.time() - t0) < 50:
            break
        # Vary the perturbation style
        if hop % 3 == 0:
            h_pert = block_perturb(h_best, rng, 4, 5, 25, 0.08)
        elif hop % 3 == 1:
            h_pert = h_best + rng.randn(n) * 0.015
            h_pert = project_feasible(h_pert, target)
        else:
            h_pert = block_perturb(h_best, rng, 2, 10, 50, 0.12)
        # Optimize the perturbation
        h_local = nesterov_optimize(h_pert, 400, 5000, 0.005, top_k=20)
        h_local = slsqp_solve(h_local, 600, 300, top_k=None)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 4: Multi-resolution exploration
    for n_try in [100, 150, 250, 350]:
        if budget_s - (time.time() - t0) < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_c = nesterov_optimize(h_interp, 400, 4000, 0.01, top_k=25)
        h_c = slsqp_solve(h_c, 800, 400, top_k=None)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_c.copy()

    # Phase 5: Fine basin hopping at best resolution
    if len(h_best) != n:
        h_best = interp_to_n(h_best, n)
        n = len(h_best)
        target = n / 2.0
    rng2 = np.random.RandomState(seed + 999)
    for hop in range(25):
        if budget_s - (time.time() - t0) < 30:
            break
        noise_scale = 0.004 + 0.003 * (hop / 25.0)
        h_pert = h_best + rng2.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_local = nesterov_optimize(h_pert, 800, 6000, 0.003, top_k=10)
        h_local = slsqp_solve(h_local, 1200, 400, top_k=None)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 6: Final high-beta polish
    if budget_s - (time.time() - t0) > 40:
        h_fin = nesterov_optimize(h_best, 2000, 8000, 0.002, top_k=5)
        h_fin = slsqp_solve(h_fin, 2000, 600, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 7: Final squeeze
    if budget_s - (time.time() - t0) > 20:
        h_fin2 = slsqp_solve(h_best, 3000, 500, top_k=None)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    _TARGET_SUM = len(h_best) / 2.0
    return h_best, best_c5, len(h_best)
