def topk_grad(h, top_k):
    """Compute c5 approximation and gradient using top-k shifts"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # top_k shifts
    top_idx = np.argsort(-corr)[:top_k]
    # Average correlation of top k
    val = corr[top_idx].mean()
    # Gradient: average of gradients at each top-k shift
    grad = np.zeros(n)
    for ks in top_idx:
        s = int(ks) - n + 1
        i_start = max(0, -s)
        i_end = min(n, n - s)
        # d corr[k]/dh[j] for j in range:
        # contributions from h[j] as "a" (i.e., j = i+s for i in [i_start, i_end)): (1-h[i])
        # contributions from h[j] as "b" (i.e., j = i for i in [i_start, i_end)): -h[i+s]
        # First: j = i+s, so i = j-s, valid when i_start <= j-s < i_end
        #   j in [i_start+s, i_end+s) intersect [0, n)
        j1_lo = max(0, i_start + s)
        j1_hi = min(n, i_end + s)
        if j1_hi > j1_lo:
            i1_lo = j1_lo - s
            i1_hi = j1_hi - s
            grad[j1_lo:j1_hi] += (1.0 - h[i1_lo:i1_hi])
        # Second: j = i, valid when i_start <= j < i_end
        #   j in [i_start, i_end) intersect [0, n)
        j2_lo = max(0, i_start)
        j2_hi = min(n, i_end)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] += (-h[j2_lo + s:j2_hi + s])
    grad *= dx / top_k
    return val, grad

def topk_val_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(-corr)[:top_k]
    val = corr[top_idx].mean()
    grad = np.zeros(n)
    for ks in top_idx:
        s = int(ks) - n + 1
        i_start = max(0, -s)
        i_end = min(n, n - s)
        j1_lo = max(0, i_start + s)
        j1_hi = min(n, i_end + s)
        if j1_hi > j1_lo:
            grad[j1_lo:j1_hi] += (1.0 - h[j1_lo - s:j1_hi - s])
        j2_lo = max(0, i_start)
        j2_hi = min(n, i_end)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] += (-h[j2_lo + s:j2_hi + s])
    grad *= dx / top_k
    return val, grad

def momentum_opt(h, target, max_iter, top_k_init, top_k_min, lr_init, momentum):
    """Momentum-based projected subgradient with adaptive top_k"""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_val = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        top_k = max(top_k_min, int(top_k_init * (0.97 ** it)))
        val, grad = topk_val_grad(h, top_k)
        # Heavy ball momentum
        v = momentum * v + (1.0 - momentum) * grad
        # Take step
        h_new = project_feasible(h - lr * v, target)
        val_new = compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        if val_new < val:
            h = h_new
        else:
            # Backtrack
            lr *= 0.5
            v = v * 0.5  # reduce momentum effect
        # Decay learning rate
        lr = lr_init * (0.999 ** it)
    return best_h, best_val

def slsqq_polish(h, target, beta, maxiter):
    """SLSQP with smoothmax for final polish"""
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        exps = np.exp(beta * (corr - m))
        return (m + np.log(exps.sum()) / beta) * dx
    
    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        exps = np.exp(beta * (corr - m))
        w = exps / exps.sum()
        # gradient of smoothmax
        grad = np.zeros(n)
        for i, k in enumerate(range(len(corr))):
            if w[i] < 1e-14: continue
            s = k - n + 1
            i_start = max(0, -s)
            i_end = min(n, n - s)
            j1_lo = max(0, i_start + s)
            j1_hi = min(n, i_end + s)
            if j1_hi > j1_lo:
                grad[j1_lo:j1_hi] += w[i] * (1.0 - x[j1_lo - s:j1_hi - s])
            j2_lo = max(0, i_start)
            j2_hi = min(n, i_end)
            if j2_hi > j2_lo:
                grad[j2_lo:j2_hi] += w[i] * (-x[j2_lo + s:j2_hi + s])
        return grad * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except:
        return h

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def topk_val_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(-corr)[:top_k]
    val = corr[top_idx].mean()
    grad = np.zeros(n)
    for ks in top_idx:
        s = int(ks) - n + 1
        i_start = max(0, -s)
        i_end = min(n, n - s)
        j1lo = max(0, i_start + s)
        j1hi = min(n, i_end + s)
        if j1hi > j1lo:
            grad[j1lo:j1hi] += (1.0 - h[j1lo - s:j1hi - s])
        j2lo = max(0, i_start)
        j2hi = min(n, i_end)
        if j2hi > j2lo:
            grad[j2lo:j2hi] += (-h[j2lo + s:j2hi + s])
    grad *= dx / top_k
    return val, grad

def momentum_opt(h, target, max_iter, tk_init, tk_min, lr_init, mom, decay=0.998):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_val = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        tk = max(tk_min, int(tk_init * (0.96 ** it)))
        val, grad = topk_val_grad(h, tk)
        v = mom * v + (1.0 - mom) * grad
        h_new = project_feasible(h - lr * v, target)
        val_new = compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        if val_new <= val + 1e-14:
            h = h_new
        else:
            lr *= 0.5
            v *= 0.3
        lr = max(lr, lr_init * (decay ** it))
    return best_h, best_val

def slsqp_smoothmax(h, target, beta, top_k, maxiter):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if top_k is not None and top_k < len(corr):
            top = np.argsort(-corr)[:top_k]
            cs = corr[top]
        else:
            cs = corr
        m = cs.max()
        if m * beta > 500:
            exps = np.exp(beta * (cs - m))
        else:
            exps = np.exp(beta * (cs - m))
        sw = exps.sum()
        return (m + np.log(sw) / beta) * dx

    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if top_k is not None and top_k < len(corr):
            top = np.argsort(-corr)[:top_k]
        else:
            top = np.arange(len(corr))
        cs = corr[top]
        m = cs.max()
        exps = np.exp(beta * (cs - m))
        w = exps / exps.sum()
        grad = np.zeros(n)
        for ii, k in enumerate(top):
            wi = w[ii]
            if wi < 1e-14:
                continue
            s = int(k) - n + 1
            i_start = max(0, -s)
            i_end = min(n, n - s)
            j1lo = max(0, i_start + s)
            j1hi = min(n, i_end + s)
            if j1hi > j1lo:
                grad[j1lo:j1hi] += wi * (1.0 - x[j1lo - s:j1hi - s])
            j2lo = max(0, i_start)
            j2hi = min(n, i_end)
            if j2hi > j2lo:
                grad[j2lo:j2hi] += wi * (-x[j2lo + s:j2hi + s])
        return grad * dx

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h

def make_structured_starts(n, rng):
    starts = []
    # Ramp
    x = np.linspace(0, 1, n)
    starts.append(project_feasible(np.clip(2.5 * x - 0.5, 0, 1), n / 2.0))
    # Cosine
    for f in [1, 2, 3]:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * f * np.arange(n) / n)
        starts.append(project_feasible(h, n / 2.0))
    # Sine with offset
    for f in [1, 2]:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * f * np.arange(n) / n + 0.5)
        starts.append(project_feasible(np.clip(h, 0, 1), n / 2.0))
    # Blocks
    block_len = max(3, n // 10)
    for phase in range(3):
        h = np.zeros(n)
        for i in range(n):
            cycle = (i + phase) // block_len
            h[i] = 1.0 if cycle % 2 == 0 else 0.0
        starts.append(project_feasible(h, n / 2.0))
    # Random
    starts.append(project_feasible(rng.rand(n), n / 2.0))
    # Smooth random
    h_r = rng.rand(n)
    kernel = np.ones(7) / 7
    h_s = np.convolve(h_r, kernel, mode='same')
    starts.append(project_feasible(h_s, n / 2.0))
    return starts
