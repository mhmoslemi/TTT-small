    dx = 2.0 / n
    # Compute current C values
    C = np.correlate(h, 1-h, mode='full') * dx
    n_c = len(C)  # 2n-1
    
    for i in range(n):
        # Compute slope of each C[k] w.r.t. h[i]
        slope = np.zeros(n_c)
        # h[i] appears as first factor: h[i]*(1-h[i-s]) for s where 0<=i-s<n
        # This is in C[k] where k = (n-1)+s, so s = k-(n-1)
        # i.e., C[k] where k ranges from (n-1)+max(0,i-n+1) to (n-1)+min(i, n-1)
        s_vals = np.arange(max(0, i-(n-1)), min(i, n-1)+1)  # valid s values
        k_vals = (n-1) + s_vals
        slope[k_vals] += (1 - h[i - s_vals]) * dx  # wait, i-s_vals must be valid
        
        # h[i] appears as second factor: h[i+s]*(1-h[i]) for s where 0<=i+s<n
        # This is in C[k] where k = (n-1)-s, so s = (n-1)-k
        s_vals2 = np.arange(0, min(n-i, n))  # s such that i+s < n
        k_vals2 = (n-1) - s_vals2
        slope[k_vals2] -= h[i + s_vals2] * dx
        
        # Now C[k] = base[k] + slope[k] * h[i]
        # But we already have C[k] = C_current[k], so:
        # base[k] = C_current[k] - slope[k] * h[i]
        base = C - slope * h[i]
        
        # Find x that minimizes max_k (base[k] + slope[k]*x) for x in [0,1]
        # Check boundaries
        f0 = np.max(base)
        f1 = np.max(base + slope)
        
        # Find all pair intersections
        best_x = 0 if f0 <= f1 else 1
        best_f = min(f0, f1)
        
        # For pairs where slopes have opposite signs
        neg = slope < 0
        pos = slope > 0
        if np.any(neg) and np.any(pos):
            # For each pos line j and neg line i, intersection is at:
            # (base[i] - base[j]) / (slope[j] - slope[i])
            # We want to find the one that minimizes the max value at that point
            # But we only need to check the "critical" pairs
            # Actually, the minimum of the upper envelope is at the intersection
            # of the two lines that define the "valley"
            # Simple approach: check all pairs (O(n²) but n=100 so fine)
            # Actually for efficiency, let's just do a fine grid + refinement
            # Or: the min is at one of the intersection points
            # Let's compute all intersections and evaluate
            
            # For efficiency with n=100: 100*100/2 = 5000 pairs, each O(1) to compute x
            # Then O(n) to evaluate. Total O(n³) = 10^6. Fine.
            
            neg_idx = np.where(neg)[0]
            pos_idx = np.where(pos)[0]
            
            # For each pair, compute intersection
            for ji in pos_idx:

                    denom = slope[ji] - slope[ni]
                    if abs(denom) < 1e-15:
                        continue
                    x_int = (base[ni] - base[ji]) / denom
                    if 0 <= x_int <= 1:

                        if f_val < best_f - 1e-15:
                            best_f = f_val
                            best_x = x_int
        
        h[i] = best_x
        # Recompute C
        C = np.correlate(h, 1-h, mode='full') * dx
    
    return h

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum
            # Can only increase where h < 1
            mask = h < 1
            if not np.any(mask):
                break
            room = 1 - h
            room[~mask] = 0
            total_room = room.sum()
            if total_room < 1e-15:
                break
            h += diff * room / total_room
        else:
            # Need to decrease sum
            mask = h > 0
            if not np.any(mask):
                break
            total = h.sum()
            if total < 1e-15:
                break
            h += diff * h / total  # proportional decrease
    return np.clip(h, 0, 1)

def fourier_to_h(coeffs, n):
    K = len(coeffs) // 2
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(K):
        h += coeffs[k] * np.cos((k+1) * np.pi * x)
        h += coeffs[K+k] * np.sin((k+1) * np.pi * x)
    return h

def h_to_fourier(h, n, K):
    x = np.linspace(0, 2, n)
    coeffs = np.zeros(2*K)
    for k in range(K):
        coeffs[k] = (2.0 / n) * np.sum(h * np.cos((k+1)*np.pi*x))
        coeffs[K+k] = (2.0 / n) * np.sum(h * np.sin((k+1)*np.pi*x))
    return coeffs

def obj_fourier(coeffs, n, alpha, n_active):
    h = fourier_to_h(coeffs, n)
    # Penalty for box constraint
    penalty = 0.0
    if h.min() < 0:
        penalty += alpha * np.sum(h[h < 0]**2)
    if h.max() > 1:
        penalty += alpha * np.sum((h[h > 1] - 1)**2)
    
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    # Log-sum-exp for the max
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return lse + penalty

def coord_descent_with_sum(h, n, n_passes=5):

            # Valid s: 0 <= i+s < n, so 0 <= s < n-i

                k_vals = (n-1) - s_vals
                slope[k_vals] -= h[i + s_vals] * dx

                C = np.correlate(h, 1 - h, mode='full') * dx
        
        # Project to feasible set
        h = project_feasible(h, n)
        C = np.correlate(h, 1 - h, mode='full') * dx
    
    return h

                        sb = slope[b]
                        denom = sa - sb
                        if abs(denom) < 1e-15:
                            continue
                        xi = (base[b] - ba) / denom
                        if xi < 0 or xi > 1:
                            continue
                        fv = ba + sa * xi
                        if fv < best_f - 1e-15:
                            best_f = fv
                            best_x = xi
            if best_x != h[i]:
                h[i] = best_x
                C = np.correlate(h, 1 - h, mode='full') * dx
        h = project_feasible(h, n)
        C = np.correlate(h, 1 - h, mode='full') * dx
    return h

def fourier_to_h(coeffs, n):
    K = len(coeffs) // 2
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(K):
        h += coeffs[k] * np.cos((k + 1) * np.pi * x)
        h += coeffs[K + k] * np.sin((k + 1) * np.pi * x)
    return h

def h_to_fourier(h, n, K):
    x = np.linspace(0, 2, n)
    coeffs = np.zeros(2 * K)
    for k in range(K):
        coeffs[k] = (2.0 / n) * np.dot(h, np.cos((k + 1) * np.pi * x))
        coeffs[K + k] = (2.0 / n) * np.dot(h, np.sin((k + 1) * np.pi * x))
    return coeffs

def obj_fourier(coeffs, n, alpha):
    h = fourier_to_h(coeffs, n)
    pen = 0.0
    below = h[h < 0]
    if len(below) > 0:
        pen += alpha * np.sum(below * below)
    above = h[h > 1]
    if len(above) > 0:
        pen += alpha * np.sum((above - 1) * (above - 1))
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        lse = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = m + np.log(np.sum(np.exp(shifted))) / alpha
    return lse + pen

def slsqp_refine(h, n, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        a = 200000.0
        sh = a * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a
    def jac(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        a = 200000.0
        sh = a * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad_c = np.zeros(2 * n - 1)
        slope = np.zeros(len(x))
        for i in range(len(x)):
            nc = 2 * len(x) - 1
            s1 = max(0, i - (len(x) - 1))
            s2 = min(i, len(x) - 1)
            if s1 <= s2:
                sv = np.arange(s1, s2 + 1)
                grad_c[(len(x)-1) + sv] += (1 - hx[i - sv]) * dx
            sm = len(x) - 1 - i
            if sm >= 0:
                sv = np.arange(0, sm + 1)
                grad_c[(len(x)-1) - sv] -= hx[i + sv] * dx
            slope[i] = np.dot(w, grad_c)
        return slope
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_local(h, n, deadline):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for outer in range(20):
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = None
        n = 100

    h = h_init.copy() if h_init is not None else np.full(n, 0.5)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    def rem():
        return deadline - time.time()

    # Phase 1: Coordinate descent (exact 1D steps)
    for _ in range(5):
        if rem() < 15:
            break
        h_cd = coord_descent(best_h, n, n_passes=2)
        c = compute_c5(h_cd, n)
        if c < best_c:
            best_c = c
            best_h = h_cd.copy()

    # Phase 2: Fourier optimization to explore different basins
    for K in [8, 12, 16, 20]:
        if rem() < 25:
            break
        coeffs = h_to_fourier(best_h, n, K)
        for alpha in [500, 5000, 50000]:
            if rem() < 10:
                break
            try:
                res = minimize(obj_fourier, coeffs, args=(n, alpha),
                              method='L-BFGS-B', maxiter=800,
                              options={'ftol': 1e-15, 'disp': False})
                h_try = fourier_to_h(res.x, n)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 3: More coordinate descent after Fourier
    if rem() > 20:
        h_cd = coord_descent(best_h, n, n_passes=5)
        c = compute_c5(h_cd, n)
        if c < best_c:
            best_c = c
            best_h = h_cd.copy()

    # Phase 4: Perturbed Fourier restarts
    for trial in range(4):
        if rem() < 25:
            break
        K = 16
        coeffs = h_to_fourier(best_h, n, K) + rng.normal(0, 0.03, 2 * K)
        for alpha in [10000, 100000]:
            if rem() < 10:
                break
            try:
                res = minimize(obj_fourier, coeffs, args=(n, alpha),
                              method='L-BFGS-B', maxiter=1000,
                              options={'ftol': 1e-16, 'disp': False})
                h_try = fourier_to_h(res.x, n)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 5: Final coordinate descent
    if rem() > 15:
        n_passes = min(10, max(2, int(rem() // 8)))
        h_cd = coord_descent(best_h, n, n_passes=n_passes)
        c = compute_c5(h_cd, n)
        if c < best_c:
            best_c = c
            best_h = h_cd.copy()

    # Phase 6: Swap local search
    if rem() > 10:
        best_h, best_c = swap_local(best_h, n, deadline)

    # Phase 7: SLSQP final polish
    if rem() > 15:
        h_try = slsqp_refine(best_h, n, maxiter=3000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Final coordinate descent pass
    if rem() > 5:
        h_cd = coord_descent(best_h, n, n_passes=2)
        c = compute_c5(h_cd, n)
        if c < best_c:
            best_c = c
            best_h = h_cd.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
