import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def exact_coord_sweep(h, n, max_sweeps=50, t0=None, budget=0):

    target = n / 2.0

    for sweep in range(max_sweeps):
        if t0 is not None and budget > 0:
            if budget - (time.time() - t0) < 2:
                break
        
        improved = False
        for j in range(n):
            # Compute gradient vector g_L for this j
            # g_L = dx * [(1-h[j+L]) - h[j-L]] for L = -(n-1)..(n-1)
            # We need c_L for all L
            corr = np.correlate(h, 1.0 - h, mode="full")
            c = corr * dx  # length 2n-1, index m corresponds to lag L = m-(n-1)

            # term1: (1 - h[j+L]) if 0 <= j+L < n
            # term2: h[j-L] if 0 <= j-L < n

            j_plus_L = j + L_vals  # indices for h[j+L]
            j_minus_L = j - L_vals  # indices for h[j-L]
            
            term1 = np.zeros(2*n-1)
            mask1 = (j_plus_L >= 0) & (j_plus_L < n)

            term2 = np.zeros(2*n-1)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new),
                     np.linspace(0.0, 2.0, len(h)), h)

def exact_coord_single(h, j, n):
    """Find exact optimal value for h[j] given all other coords fixed."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    
    L_vals = np.arange(-(n-1), n)
    j_plus_L = j + L_vals
    j_minus_L = j - L_vals
    
    g = np.zeros(2*n-1)
    mask1 = (j_plus_L >= 0) & (j_plus_L < n)
    g[mask1] += 1.0 - h[j_plus_L[mask1]]
    mask2 = (j_minus_L >= 0) & (j_minus_L < n)
    g[mask2] -= h[j_minus_L[mask2]]
    g *= dx
    
    lo = -h[j]
    hi = 1.0 - h[j]
    if hi - lo < 1e-12:
        return h[j]
    
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c + m1 * g) < np.max(c + m2 * g):
            hi = m2
        else:
            lo = m1
    
    return np.clip(h[j] + (lo + hi) / 2.0, 0.0, 1.0)

def exact_coord_sweep(h, n, max_sweeps=100, t0=None, budget=0, min_sweeps=5):
    target = n / 2.0
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if t0 is not None and budget > 0:
            if budget - (time.time() - t0) < 2 and sweep >= min_sweeps:
                break
        
        max_change = 0.0
        for j in range(n):
            new_val = exact_coord_single(h, j, n)
            change = abs(new_val - h[j])
            max_change = max(max_change, change)
            if change > 1e-12:
                h[j] = new_val
        
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if max_change < 1e-10:
            break
    
    h = project_h(h, target)
    return h, compute_c5(h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_step(h, n, top_k=5):
    """Compute subgradient of C5 at h, focused on top_k worst lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    dx = 2.0 / n
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n-la])
            grad[:n-la] -= h[la:]
    return grad * dx

def nesterov_smooth(h0, n, target, beta, steps, t0, budget):
    """Nesterov-accelerated gradient descent on smooth approximation."""
    dx = 2.0 / n
    x = h0.copy()
    v = np.zeros(n)
    mu = 0.5
    best_local = compute_c5(x)
    
    for it in range(steps):
        if budget - (time.time() - t0) < 3:
            break
        
        # Nesterov lookahead
        y = x + mu * v
        x_shift = np.clip(y, 0, 1)
        
        # Gradient of smooth max
        c = np.correlate(x_shift, 1.0 - x_shift, mode="full")
        cm = np.max(c)
        w = np.exp(beta * (c - cm))
        w /= w.sum()
        
        # Gradient of c_k w.r.t. h
        grad = np.zeros(n)
        for idx in range(len(c)):
            if w[idx] < 1e-10:
                continue
            lag = idx - (n - 1)
            if lag >= 0:
                grad[:n-lag] += w[idx] * (1.0 - x_shift[lag:])
                grad[lag:] -= w[idx] * x_shift[:n-lag]
            else:
                la = -lag
                grad[la:] += w[idx] * (1.0 - x_shift[:n-la])
                grad[:n-la] -= w[idx] * x_shift[la:]
        grad *= dx
        
        # Step
        lr = 1.0 / (beta * np.linalg.norm(grad) + 1e-10)
        v = (it / (it + 1 + mu)) * (v - lr * grad)
        x_new = project_h(x + v, target)
        
        c_try = compute_c5(x_new)
        if c_try < best_local:
            best_local = c_try
        
        x = x_new
        mu = (1 + np.sqrt(1 + 4 * mu * mu)) / 2
    
    return x, best_local

            g *= dx

            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-13:
                continue
            for _ in range(55):
                m1v = lo + (hi - lo) / 3.0
                m2v = hi - (hi - lo) / 3.0
                if np.max(c + m1v * g) < np.max(c + m2v * g):
                    hi = m2v
                else:
                    lo = m1v

            delta = (lo + hi) / 2.0
            new_val = np.clip(h[j] + delta, 0.0, 1.0)
            d_act = new_val - h[j]
            if abs(d_act) > 1e-13:
                h[j] = new_val
                c += d_act * g
                max_change = max(max_change, abs(d_act))

        c5_now = float(np.max(c))
        if c5_now < best_c5:
            best_c5 = c5_now
        if max_change < 1e-11:
            break

    h = project_h(h, target)
    return h, compute_c5(h)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def random_binary_init(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    n0 = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Exact coordinate descent at native resolution
    best_h, best_c5 = exact_coord_sweep_fast(best_h.copy(), n0,
                                             max_sweeps=80, t0=t0, budget=budget_s)

    # Phase 2: Multi-resolution (downsample, optimize, upsample)
    for n_pts in [240, 200, 160, 120, 100]:
        if n_pts >= n0:
            continue
        if budget_s - (time.time() - t0) < 40:
            break
        cur = project_h(resample(best_h, n_pts), n_pts / 2.0)
        cur, cur_c5 = exact_coord_sweep_fast(cur, n_pts,
                                             max_sweeps=300, t0=t0, budget=budget_s)
        cur_up = project_h(resample(cur, n0), n0 / 2.0)
        cur_up, cur_up_c5 = exact_coord_sweep_fast(cur_up, n0,
                                                   max_sweeps=60, t0=t0, budget=budget_s)
        if cur_up_c5 < best_c5:
            best_h, best_c5 = cur_up.copy(), cur_up_c5

    # Phase 3: SLSQP refinement
    if budget_s - (time.time() - t0) > 25:
        n_pts = len(best_h)
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        for beta in [300, 800, 2000]:
            if budget_s - (time.time() - t0) < 10:
                break
            cur = do_slsqp(best_h.copy(), n_pts, target, beta, bounds, 8000, 1e-15)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

    # Phase 4: More exact coordinate descent
    if budget_s - (time.time() - t0) > 10:
        best_h, best_c5 = exact_coord_sweep_fast(best_h.copy(), len(best_h),
                                                 max_sweeps=150, t0=t0, budget=budget_s)

    # Phase 5: Perturbation + re-optimize
    n_pts = len(best_h)
    target = n_pts / 2.0
    for att in range(8):
        if budget_s - (time.time() - t0) < 8:
            break
        if att < 5:
            scale = 0.04 / (att + 1)
            hp = project_h(best_h + rng.randn(n_pts) * scale, target)
        else:
            hp = project_h(random_binary_init(n_pts, rng), target)
            hp = project_h(hp + rng.randn(n_pts) * 0.05, target)
        hp, hp_c5 = exact_coord_sweep_fast(hp, n_pts,
                                           max_sweeps=120, t0=t0, budget=budget_s)
        if hp_c5 < best_c5:
            best_h, best_c5 = hp.copy(), hp_c5
        if budget_s - (time.time() - t0) < 15:
            break
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        hp2 = do_slsqp(hp.copy(), n_pts, target, 1000, bounds, 5000, 1e-14)
        hp2_c5 = compute_c5(hp2)
        if hp2_c5 < best_c5:
            best_h, best_c5 = hp2.copy(), hp2_c5

    # Phase 6: Final polish
    if budget_s - (time.time() - t0) > 8:
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = do_slsqp(best_h.copy(), n_pts, target, 3000, bounds, 10000, 1e-15)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5
    if budget_s - (time.time() - t0) > 5:
        best_h, best_c5 = exact_coord_sweep_fast(best_h.copy(), n_pts,
                                                 max_sweeps=300, t0=t0, budget=budget_s)

    return best_h, best_c5, n_pts
