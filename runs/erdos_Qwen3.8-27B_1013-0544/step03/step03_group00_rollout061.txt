import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    shifted = beta * (c_sel - m)
    shifted -= np.max(shifted)  # for numerical stability
    exps = np.exp(shifted)
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def slsqp_smoothmax(h, beta, maxiter, top_k=None, ftol=1e-16):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def penalty_val_grad(h, T, gamma):
    """Penalty objective: sum of (max(0, corr[k]*dx - T))^2 * gamma + small smoothmax"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(0, corr - T)
    val = gamma * np.sum(excess**2)
    # gradient: d/dh[j] of sum_k gamma * max(0, corr_k*dx - T)^2
    # = sum_k 2*gamma*max(0, corr_k*dx - T)*dx * d(corr_k)/dh[j]
    if np.max(excess) < 1e-14:
        return val, np.zeros(n)
    active = excess > 1e-14
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        w = 2.0 * gamma * excess[k] * dx
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += w * (-h[i_lo2 + s:i_hi2 + s])
    return val, grad

def slsqp_penalty(h, T, gamma, maxiter):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return penalty_val_grad(x, T, gamma)[0]
    def objg(x):
        return penalty_val_grad(x, T, gamma)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def swap_perturb(h, n_swaps, rng):
    """Random swap perturbation that preserves the sum."""
    h = h.copy()
    n = len(h)
    for _ in range(n_swaps):
        i, j = rng.randint(0, n, 2)
        if i != j:
            h[i], h[j] = h[j], h[i]
    return project_feasible(h, n / 2.0)

def block_perturb(h, rng, n_blocks=3, block_size_max=None):
    """Perturb by adding noise to random blocks."""
    n = len(h)
    if block_size_max is None:
        block_size_max = max(3, n // 10)
    h = h.copy()
    for _ in range(n_blocks):
        bs = rng.randint(2, block_size_max + 1)
        start = rng.randint(0, n - bs)
        h[start:start+bs] += rng.uniform(-0.1, 0.1, bs)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    n = len(h_cur)
    cont_betas = [10, 30, 80, 200, 500]
    for beta in cont_betas:

        h_new = slsqp_smoothmax(h_cur, beta, 400, top_k=None)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new
    
    # Phase 2: Interpolate to n=400 and optimize there
    if remaining() > 200:
        h400 = interp_to_n(h_best, 400)
        c400 = compute_c5(h400)
        if c400 < best_c5:

            h_best = h400.copy()
        # Optimize at n=400
        for beta in [50, 150, 400, 800]:
            if remaining() < 60:

            h400 = slsqp_smoothmax(h400, beta, 500, top_k=None)
            c400 = compute_c5(h400)
            if c400 < best_c5:

                h_best = h400.copy()
        # Penalty-based refinement
        if remaining() > 120:
            T = best_c5 + 0.001
            for _ in range(3):

                c400 = compute_c5(h400)
                if c400 < best_c5:
                    best_c5 = c400
                    h_best = h400.copy()
                T = max(best_c5 * 0.999, T - 0.001)
    
    # Phase 3: Basin hopping at best resolution
    if remaining() > 100:
        n_h = len(h_best)
        rng = np.random.RandomState(seed + 42)

        if c600 < best_c5:

            h_best = h600.copy()
        for beta in [100, 300, 700]:

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    shifted = beta * (c_sel - m)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        k = top_idx[i]
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def penalty_val_grad(h, T, gamma):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(0.0, corr - T)
    val = gamma * np.sum(excess * excess)
    if np.max(excess) < 1e-14:
        return val, np.zeros(n)
    active_idx = np.where(excess > 1e-14)[0]
    grad = np.zeros(n)
    for k in active_idx:
        w = 2.0 * gamma * excess[k] * dx
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += w * (-h[i_lo2 + s:i_hi2 + s])
    return val, grad

def slsqp_solve(h, beta, maxiter, top_k=None, use_penalty=False, T=None, gamma=500.0):
    n = len(h)
    target = n / 2.0
    cp_constraint = ({'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac})
    if use_penalty:
        def obj_fn(x):
            return penalty_val_grad(x, T, gamma)[0]
        def obj_grad(x):
            return penalty_val_grad(x, T, gamma)[1]
    else:
        def obj_fn(x):
            return smoothmax_val_grad(x, beta, top_k)[0]
        def obj_grad(x):
            return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cp_constraint,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def swap_perturb(h, n_swaps, rng):
    h = h.copy()
    n = len(h)
    for _ in range(n_swaps):
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i != j:
            h[i], h[j] = h[j], h[i]
    return project_feasible(h, n / 2.0)

def block_perturb(h, rng, n_blocks=3, amp=0.05):
    n = len(h)
    h = h.copy()
    bs_max = max(3, n // 12)
    for _ in range(n_blocks):
        bs = rng.randint(2, bs_max + 1)
        start = rng.randint(0, n - bs)
        h[start:start + bs] += rng.uniform(-amp, amp, bs)
    return project_feasible(h, n / 2.0)

def gaussian_perturb(h, noise, rng):
    h_pert = h + rng.randn(len(h)) * noise
    return project_feasible(h_pert, len(h) / 2.0)
