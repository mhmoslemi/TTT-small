import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        else:
            t = -s
            ms = np.arange(0, n - t)
            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def con_func(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def nesterov_opt(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        lags = idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                ms = np.arange(s, n)
                grad[ms] += (1.0 - y[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            else:
                tt = -s
                ms = np.arange(0, n - tt)
                grad[ms] += (1.0 - y[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += (-y[mp - tt])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_smooth(h, max_iters, lr0, decay, beta):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        _, grad = smoothmax_val_grad(y, beta)
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v
