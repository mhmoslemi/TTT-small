def penalty_grad(h, beta, dx, lam, target):
    # Gradient of smooth_max
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(np.clip(beta * (corr - m), -500, 500))
    weights /= weights.sum()
    
    # Gradient of corr with respect to h
    # corr[k] = sum_i h[i] * (1 - h[i + lag]) where lag = k - (n-1)
    # d(corr[k])/d(h[j]) = (1-h[j+lag]) (if j+lag < n) - h[j-lag] (if j >= lag)
    
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(len(corr)):
        lag = k - (n - 1)
        w = weights[k] * dx
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j < n-lag (from first term)

            for j in range(lag, n):
                grad[j] += w * (-h[j - lag])

                grad[j] += w * (1 - h[j + la])
            for j in range(la, n):

    # Add penalty gradient

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def smooth_max_topk(x, beta, k):
    """Smooth max of only the top k elements."""
    top = np.sort(x)[::-1][:k]
    m = np.max(top)
    xs = np.clip(beta * (top - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / k) / beta

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def slsqp_opt(h0, target, beta, maxiter, focus_topk=0):
    n = len(h0)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    if focus_topk == 0:
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode="full")
            return smooth_max(corr, beta) * dx
    else:
        def obj(v):
            corr = np.correlate(v, 1.0 - v, mode="full")
            return smooth_max_topk(corr, beta, focus_topk) * dx
    
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - target,
             'jac': lambda v: np.ones(n)}]

        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(r.x, target)
    except Exception:
        return h0

def subgrad_step(h, target, step, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n-la])
            grad[:n-la] -= h[la:]
    grad /= (top_k * n)  # normalize
    return grad

    mom = np.zeros(len(h))
    step = step0
    best_c = compute_c5(h)
    stall = 0

                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(r.x, target)
    except Exception:
        return h0

def subgrad_gradient(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad / top_k

def subgrad_loop(h, target, n_iter, step0, t0, budget):
    n = len(h)
    mom = np.zeros(n)
    step = step0
    best_c = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        g = subgrad_gradient(h, 5)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.88 * mom - step * g
        trial = project_sum(h + mom, target)
        c = compute_c5(trial)
        if c < best_c - 1e-15:
            h = trial
            best_c = c
            stall = 0
        else:
            stall += 1
            step *= 0.7
            mom *= 0.2
            trial = project_sum(h + mom, target)
            c = compute_c5(trial)
            if c < best_c - 1e-15:
                h = trial
                best_c = c
                stall = 0
                step *= 1.2
            else:
                mom = np.zeros(n)
                step = max(step, 0.001)
        if stall > 400:
            break
    return h

def coord_polish(h, target, max_sweeps, t0, budget):
    n = len(h)
    best_c = compute_c5(h)
    base_d = 0.012
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 1:
            break
        d = base_d / (1 + 0.3 * sw)
        if d < 0.002:
            d = 0.002
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 0.5:
                break
            for delta in (d, -d):
                trial = h.copy()
                trial[j] += delta
                trial = project_sum(trial, target)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    h = trial
                    best_c = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(200, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_init)

    # Try native resolution and a slightly coarser one
    n_list = [best_n]
    if best_n > 200:
        n_list.append(160)

    for n_pts in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        target = n_pts / 2.0
        cur = project_sum(resample(h_init, n_pts), target)

        # SLSQP with increasing smoothness
        for beta in [25, 50, 100, 200, 400, 800]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            maxit = 8000 if beta >= 200 else 4000
            cur = slsqp_opt(cur, target, beta, maxit)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

        # Subgradient descent with momentum
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur = subgrad_loop(cur, target, 10000, 0.018, t0, budget_s)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

        # Coordinate descent polish
        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            cur, c = coord_polish(cur, target, 60, t0, budget_s)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Perturbation restarts
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        n = len(best_h)
        target = n / 2.0
        for att in range(8):
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            scale = 0.012 / (att + 1)
            hp = project_sum(best_h + np.random.randn(n) * scale, target)
            hp = slsqp_opt(hp, target, 600, 4000)
            hp = subgrad_loop(hp, target, 2000, 0.01, t0, budget_s)
            c = compute_c5(hp)
            if c < best_c5:
                best_h, best_c5, best_n = hp.copy(), c, n

    # Final high-precision polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        n = len(best_h)
        target = n / 2.0
        for beta in [2000, 4000, 8000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 4:
                break
            best_h = slsqp_opt(best_h, target, beta, 15000)
            best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining > 4:
        n = len(best_h)
        target = n / 2.0
        best_h = subgrad_loop(best_h, target, 8000, 0.006, t0, budget_s)
        best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        n = len(best_h)
        target = n / 2.0
        best_h, best_c5 = coord_polish(best_h, target, 100, t0, budget_s)

    # Final exact projection
    n = len(best_h)
    best_h = project_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
