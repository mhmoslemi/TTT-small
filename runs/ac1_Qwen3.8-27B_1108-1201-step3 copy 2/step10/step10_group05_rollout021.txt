    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def parametric_init(n, rng, param_type="cosine"):
    if param_type == "cosine":
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif param_type == "hann":
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif param_type == "parzen":
        x = np.linspace(0, 1, n)
        xm = np.abs(x - 0.5) * 2
        a = np.where(xm <= 1.0, 1.5*(1-xm**2) - 0.5*(1-xm**2)**3, 0.0)
        a = np.maximum(a, 0)
    elif param_type == "gaussian":
        x = np.linspace(0, 1, n)
        sigma = rng.uniform(0.25, 0.4)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif param_type == "polynomial":
        x = np.linspace(0, 1, n)
        p = rng.uniform(0.5, 1.5)
        a = (1.0 - (2*x - 1)**2) ** p
        a = np.maximum(a, 0)
    elif param_type == "uniform":
        a = np.ones(n)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a * (1.0 + 0.005 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return symmetrize(a)

def iterative_best_response(a, b_init=None, iters=30, lr_scale=0.5):
    """Iteratively find b that minimizes max_k (a*b)_k using mirror descent."""
    n = len(a)
    if b_init is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(b_init, 0).copy()
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / s

    best_b = b.copy()
    best_M = float(cross_convolve(a, b).max())

    for it in range(iters):
        ab = cross_convolve(a, b)
        M = float(ab.max())
        if M < best_M:
            best_M = M
            best_b = b.copy()

        # Subgradient: uniform on active set
        threshold = M - max(1e-12, 0.001 * M)
        active = ab >= threshold
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / max(active.sum(), 1)

        # Gradient wrt b: grad[j] = sum_k p[k] * a[k-j]
        # = cross-correlation of p and a at lag j
        a_flip = a[::-1]
        fft_size = 1 << (2 * n - 2).bit_length()
        Pf = rfft(p, fft_size)
        Af = rfft(a_flip, fft_size)
        grad = irfft(Pf * Af, fft_size)[:n]

        # Mirror descent step
        lr = lr_scale / (it + 3)
        b = b * np.exp(-lr * grad)
        b = np.maximum(b, 1e-20)
        b = b / b.sum()

    return best_b, best_M

def lp_fictitious_play(a0, time_limit, warm_start_b=None, verbose=False, label=""):
    """LP fictitious play with symmetry enforcement and adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = symmetrize(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    b = warm_start_b if warm_start_b is not None else np.ones(n) / n

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Best response
        b_new, M_b = iterative_best_response(a, b_init=b, iters=min(30, max(10, 60 // (iters // 10 + 1))))
        b = b_new

        # Adaptive damping
        alpha = min(1.0, 8.0 / (iters + 3.0))
        a_new = (1 - alpha) * a + alpha * b
        a_new = symmetrize(np.maximum(a_new, 0))
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * float(fast_convolve(b).max())
        if score_b < best_score - 1e-15:
            best_score = score_b
            best_a = symmetrize(b).copy() / max(symmetrize(b).sum(), 1e-15)

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score, iters

def smoothed_gradient_descent(a0, time_limit, beta_init=30.0, lr_init=0.01, verbose=False, label=""):
    """Smoothed mirror descent with Nesterov momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = symmetrize(a) / max(symmetrize(a).sum(), 1e-15)

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Smoothed gradient via log-sum-exp
        conv_y = fast_convolve(y)
        max_c = float(conv_y.max())
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient: grad[j] = 2 * sum_k weights[k] * y[k-j]
        y_flip = y[::-1]
        fft_size = 1 << (2 * n - 2).bit_length()
        Wf = rfft(weights, fft_size)
        Yf = rfft(y_flip, fft_size)
        grad = 2.0 * irfft(Wf * Yf, fft_size)[:n]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)
        a = symmetrize(a)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss

        score = 2.0 * n * float(fast_convolve(a).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            beta *= 1.005

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Priority-based coordinate descent with exact per-coordinate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = symmetrize(a)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Process coordinates in order of descending value (priority)
        order = np.argsort(a)[::-1]
        # Only process non-negligible coordinates
        active_coords = order[a[order] > 1e-10]

        for i in active_coords:
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Generate candidates
            candidates = [0.0]

            # Where lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai < d < ai:
                        candidates.append(d)

            # Top-line crossings
            top_k = min(4, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai < d < ai:
                            candidates.append(d)

            # Fractional moves
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(0.5 * frac * ai)

            for delta in candidates:
                if delta <= -ai + 1e-12 or abs(delta) < 1e-18:
                    continue
                delta = max(delta, -ai)
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                # Symmetrize periodically
                if sweeps % 5 == 0:
                    a = symmetrize(a)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
                    conv = fast_convolve(a)
                    M = float(conv.max())

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f}")

    # Final symmetrization
    a = symmetrize(a)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss
    score = 2.0 * n * float(fast_convolve(a).max())
    if score < best_score - 1e-14:
        best_score = score
        best_a = a.copy()

    return best_a, best_score, sweeps

def fine_polish(a0, time_limit):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = symmetrize(a)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 50:
            break

    return best_a, best_score

def peak_shaving(a0, time_limit, verbose=False, label=""):
    """Directly target peaks of conv(a,a) by redistributing mass."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = symmetrize(a)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())

        # Focus on near-peak region
        threshold = M * 0.85
        near_peak = np.where(conv >= threshold)[0]
        rho = 1.0 / len(near_peak)

        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                pressure[i] += rho * a[k - i]

        # Move mass from high-pressure to low-pressure coordinates
        p_mean = pressure.mean()
        lr = 0.02 / np.sqrt(it + 1)

        a_new = a * (1.0 - lr * (pressure - p_mean))
        a_new = np.maximum(a_new, 0)
        a_new = simplex_project(a_new)
        a_new = symmetrize(a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {label} it={it} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score
