        if time.monotonic() - t0 > total_budget:
            break

        time_for_beta = min(
            total_budget * np.exp(-beta_schedule.index(beta) * 0.2) / len(beta_schedule),
            (time_limit - 60 - (time.monotonic() - t0))
        )
        time_for_beta = max(time_for_beta, 5.0)

        def fun(x):
            return make_smoothed_fn(x, beta, N, n)[0]

        def jac(x):
            return make_smoothed_fn(x, beta, N, n)[1]

        def fun_and_grad(x):
            val, g = make_smoothed_fn(x, beta, N, n)
            return val, g

        try:
            result = minimize(
                fun_and_grad, x0, method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': 5000,
                    'gtol': 1e-6,
                    'maxfun': 20000,
                    'iprint': 0
                }
            )
            x_opt = result.x
            a_new = np.exp(np.clip(x_opt, -60, 60) - np.clip(x_opt, -60, 60).max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            x0 = x_opt.copy()

            if verbose:
                print(f"    L-BFGS beta={beta}: score={score:.6f}, iters={result.nit}, status={result.status}")

        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")
            # Try continuing from current x0
            continue

    return best_a, best_score


def smart_coord_descent(a0, time_limit, label=""):
    """Smart 1D line-search coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                candidates.append(-frac * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

    return best_a, best_score


def nesterov_grad_descent(a0, time_limit, beta_init=50.0, label=""):
    """Nesterov-accelerated gradient descent with adaptive step size."""
    n = len(a0)
    N = 1 << (2 * n - 2).bit_length()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15).copy()
    a = a / a.sum()
    x = np.log(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    gamma = 1.0
    y = x.copy()
    beta = beta_init
    lr = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * x - gamma * y

        # Compute gradient at y
        y_clip = np.clip(y, -60, 60)
        y_shifted = y_clip - y_clip.max()
        y_a = np.exp(y_shifted)
        y_a = y_a / y_a.sum()

        A = fft(y_a, N)
        c = ifft(A * A, N)[:2 * n - 1].real
        c = np.maximum(c, 0)

        c_max = c.max()
        c_shifted = beta * (c - c_max)
        c_shifted = np.clip(c_shifted, -50, 0)
        w = np.exp(c_shifted)
        W = w.sum()
        if W < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            W = 1.0
        w = w / W

        W_fft = fft(w, N)
        r = ifft(W_fft * np.conj(A), N)[:n].real
        W_c = np.dot(w, c)
        grad = 2.0 * y_a * (r - W_c)

        # Backtracking line search
        obj_y, _ = make_smoothed_fn(y, beta, N, n)
        lr_try = lr
        for _ in range(20):
            x_new = y - lr_try * grad
            x_clip = np.clip(x_new, -60, 60)
            x_shifted = x_clip - x_clip.max()
            x_a = np.exp(x_shifted)
            x_a = x_a / x_a.sum()
            obj_new, _ = make_smoothed_fn(x_new, beta, N, n)
            if obj_new <= obj_y - 1e-4 * lr_try * np.dot(grad, grad):
                break
            lr_try *= 0.5
        else:
            lr_try = 1e-10

        x = y - lr_try * grad
        lr = lr_try * 1.5  # increase step if backtracking didn't reduce much

        # Evaluate
        x_clip = np.clip(x, -60, 60)
        x_shifted = x_clip - x_clip.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            beta *= 1.5
            lr *= 0.7
            no_improve = 0
            if beta > 5000:
                break

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def cosine_init(n, freq=1, amplitude=0.3, rng=None):
    """Create a cosine-modulated initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * (j - center) / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gaussian_init(n, sigma_frac=0.3, rng=None):
    """Gaussian bump initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    return a / a.sum()


def dual_gaussian_init(n, rng):
    """Two Gaussian bumps."""
    j = np.arange(n)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    s1 = n * rng.uniform(0.05, 0.15)
    s2 = n * rng.uniform(0.05, 0.15)
    a = np.exp(-0.5 * ((j - c1) / s1) ** 2) + np.exp(-0.5 * ((j - c2) / s2) ** 2)
    return a / a.sum()


def spectral_flat_init(n, rng):
    """Initialize by trying to flatten the spectrum."""
    N = 1 << (2 * n - 2).bit_length()
    # Start with random phase
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0
    # Add random phases at other frequencies with equal magnitude
    mags = np.ones(N) * (1.0 / np.sqrt(N))
    A = mags * np.exp(2j * np.pi * rng.random(N))
    A[0] = 1.0
    a = ifft(A).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gs_phase_retrieval(a_init, n_iter, N, n, target_c=None):
    """Gerchberg-Saxton style phase retrieval to flatten autocorrelation."""
    a = np.maximum(a_init.copy(), 0)
    a = a / a.sum()

    for _ in range(n_iter):
        A = fft(a, N)
        mag = np.abs(A)

        # Compute current autocorrelation
        c = ifft(A * A, N)[:2 * n - 1].real
        c_max = c.max()

        # Modify magnitude to reduce peaks
        # Reduce frequencies that contribute most to peaks
        # Simple approach: cap the magnitude
        c_target = np.ones(2 * n - 1) / (2 * n - 1)
        # Desired |A|^2 for flat c
        desired_power = fft(c_target, N)
        desired_mag = np.abs(desired_power)
        # Blend
        new_mag = 0.7 * mag + 0.3 * np.maximum(desired_mag, 1e-10)
        new_mag[0] = 1.0  # Keep DC

        # Keep phase, change magnitude
        A_new = new_mag * np.exp(1j * np.angle(A))
        A_new[0] = 1.0
        a = ifft(A_new).real[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    return a


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_targeted_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate():
    """
    Multi-phase optimization:
    1. FFT-accelerated L-BFGS on smoothed objective (softmax parameterization)
    2. Nesterov gradient descent with adaptive beta
    3. Multi-start with diverse initializations
    4. Smart coordinate descent polishing
    5. Multi-length exploration
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    # Define lengths to try
    lengths_to_try = [768, 1024, 1536]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
        # Try one larger
        if len(parent) < 2000:
            lengths_to_try.append(2000)

    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for L in lengths_to_try:
        if L not in seen and 256 <= L <= 2500:
            seen.add(L)
            unique_lengths.append(L)
    lengths_to_try = unique_lengths[:4]

    print(f"Lengths to try: {lengths_to_try}")

    # Allocate time: 60% to L-BFGS phase, 25% to CD polish, 15% for exploration
    total_time = 1790.0
    lbfgs_time_frac = 0.50
    cd_time_frac = 0.30
    explore_time_frac = 0.20

    for n in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        N = 1 << (2 * n - 2).bit_length()
        print(f"\n=== Processing n={n}, N={N}, time_rem={rem:.0f}s ===")

        t_lbfgs = min(rem * lbfgs_time_frac, 400.0)
        t_cd = min(rem * cd_time_frac, 200.0)

        # Generate initializations
        inits = []
        labels = []

        if parent is not None and len(parent) == n:
            inits.append(parent.copy())
            labels.append("parent")
        elif parent is not None:
            inits.append(interpolate_seq(parent, n))
            labels.append("parent_interp")

        # Always try these
        inits.append(np.ones(n) / n)
        labels.append("uniform")

        for freq in [1, 2, 3]:
            for amp in [0.2, 0.5, 0.8]:
                inits.append(cosine_init(n, freq=freq, amplitude=amp))
                labels.append(f"cos_f{freq}_a{amp}")

        inits.append(gaussian_init(n, sigma_frac=0.2))
        labels.append("gauss_0.2")
        inits.append(gaussian_init(n, sigma_frac=0.4))
        labels.append("gauss_0.4")

        inits.append(dual_gaussian_init(n, rng))
        labels.append("dual_gauss")

        inits.append(spectral_flat_init(n, rng))
        labels.append("spectral_flat")

        # Random
        for _ in range(3):
            r = np.abs(rng.standard_normal(n)) + 0.1
            inits.append(r / r.sum())
            labels.append(f"random_{rng.integers(10000)}")

        # Truncation: limit number of inits based on time
        max_inits = max(4, int(t_lbfgs / 30))
        inits = inits[:max_inits]
        labels = labels[:max_inits]

        print(f"  Trying {len(inits)} initializations with L-BFGS (budget={t_lbfgs:.0f}s)")
        t_init0 = time.monotonic()

        local_best_a = None
        local_best_score = float('inf')

        for idx, (a_init, label) in enumerate(zip(inits, labels)):
            t_rem_inits = deadline - time.monotonic() - 60
            if t_rem_inits < 20:
                break
            time_for_this = min(t_rem_inits / (len(inits) - idx), 60.0)
            if time_for_this < 5:
                continue

            # Optionally run GS phase retrieval first for a better start
            if label in ("uniform", "spectral_flat") and time_for_this > 20:
                a_init = gs_phase_retrieval(a_init, n_iter=50, N=N, n=n)

            # Run L-BFGS
            a_opt, score_opt = optimize_with_lbfgs(
                a_init, time_for_this, N, n, verbose=(idx == 0)
            )

            if score_opt < local_best_score:
                local_best_score = score_opt
                local_best_a = a_opt.copy()
                print(f"    L-BFGS [{label}] n={n}: score={score_opt:.6f}")

            # Also try Nesterov GD from L-BFGS result if time permits
            t_rem_ngd = deadline - time.monotonic() - 60
            if t_rem_ngd > 30 and time_for_this > 15:
                t_ngd = min(t_rem_ngd * 0.3, 20.0)
                a_ngd, score_ngd = nesterov_grad_descent(
                    a_opt, t_ngd, beta_init=50.0, label=f"ngd_{label}"
                )
                if score_ngd < local_best_score:
                    local_best_score = score_ngd
                    local_best_a = a_ngd.copy()
                    print(f"    NGD [{label}] n={n}: score={score_ngd:.6f}")

        # CD polish on the best from L-BFGS
        if local_best_a is not None:
            t_cd_actual = min(rem * cd_time_frac, 100.0)
            t_cd_actual = max(t_cd_actual, 10.0)
            t_cd_actual = min(t_cd_actual, deadline - time.monotonic() - 30)
            if t_cd_actual > 10:
                a_cd, score_cd = smart_coord_descent(local_best_a, t_cd_actual, label=f"cd_n{n}")
                if score_cd < local_best_score:
                    local_best_score = score_cd
                    local_best_a = a_cd.copy()
                    print(f"    CD [{label}] n={n}: score={score_cd:.6f}")

            # Verify and update global best
            best_seq, best_score = check_and_update(local_best_a, best_seq, best_score, f"n{n}")

        # Also perturb and try again
        if local_best_a is not None and (deadline - time.monotonic()) > 60:
            t_pert = min((deadline - time.monotonic()) * 0.3, 60.0)
            perturbed = peak_targeted_perturb(local_best_a, rng, strength=0.04)
            a_pert, score_pert = smart_coord_descent(perturbed, t_pert, label=f"pert_n{n}")
            if score_pert < local_best_score:
                local_best_score = score_pert
                local_best_a = a_pert.copy()
            best_seq, best_score = check_and_update(local_best_a, best_seq, best_score, f"pert_n{n}")

    # Final intensive polishing on the best found
    print(f"\n=== Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            # Multiple CD passes
            for pass_idx in range(3):
                t_rem = deadline - time.monotonic() - 20
                if t_rem < 15:
                    break
                t_pass = min(t_rem * 0.4, 80.0)
                a_final, score_final = smart_coord_descent(
                    best_seq, t_pass, label=f"final_{pass_idx}"
                )
                best_seq, best_score = check_and_update(a_final, best_seq, best_score, f"final_{pass_idx}")

                # Small perturbation between passes
                if pass_idx < 2:
                    best_seq = peak_targeted_perturb(best_seq, rng, strength=0.02)

        # Nesterov GD polish
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            N = 1 << (2 * len(best_seq) - 2).bit_length()
            a_ngd_final, score_ngd_final = nesterov_grad_descent(
                best_seq, min(rem * 0.5, 40.0), beta_init=100.0, label="final_ngd"
            )
            best_seq, best_score = check_and_update(a_ngd_final, best_seq, best_score, "final_ngd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
