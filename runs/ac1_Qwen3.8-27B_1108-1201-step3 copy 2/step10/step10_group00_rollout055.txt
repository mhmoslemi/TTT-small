    return float(2.0 * n * b.max() / (s * s))

def lse_objective_and_grad(z, n, T):
    """LogSumExp objective and gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    # Gradient w.r.t. a
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)

            continue

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a[idx] += wk * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbgfs_refine(a0, time_limit, T_stages, iters_per_stage=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = log_of(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    if iters_per_stage is None:
        iters_per_stage = max(30, 3000 // len(T_stages))

    def func(z_cur):
        if time.monotonic() > deadline:
            return 1e15
        return lse_obj_grad(z_cur, n, T_stages[_cur_stage[0]])[0]

    def jac(z_cur):
        if time.monotonic() > deadline:
            return np.zeros_like(z_cur)
        return lse_obj_grad(z_cur, n, T_stages[_cur_stage[0]])[1]

    _cur_stage = [0]
    for i, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        _cur_stage[0] = i
        res = minimize(func, z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                'disp': False})
        z = res.x
        if i % 3 == 0:
            a_check = softmax_of(z)
            sc = eval_local(a_check)
            print(f"      LSE T={T:.4f} ({time.monotonic()-t0:.0f}s): score={sc:.8f}")

    return softmax_of(z)


def nesterov_peak_subgradient(a0, time_limit, max_iters=500000):
    """Nesterov-accelerated subgradient method targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    deadline = t0 + time_limit

    eta0 = 0.02
    active_frac = 0.005  # peaks within 0.5% of max are "active"

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]

        # Subgradient: average over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)

        # Remove mean (stays on simplex)
        g -= g.mean()

        # Nesterov step
        alpha = 2.0 / (it + 2)
        # Update a (primal)
        a = a - eta0 * alpha * g
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        # Update y (dual/lookahead)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        t_old = (1.0 + np.sqrt(1.0 + 4.0 * it ** 2)) / 2.0 if it > 0 else 1.0
        y = a + ((t_old - 1.0) / t_new) * (a - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Track best
        sc_a = eval_local(a)
        sc_y = eval_local(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 5000 == 0:
            print(f"      NesterSubGrad it={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def directed_peak_perturb(a0, rng, n_pert=5):
    """Generate perturbations that specifically target active peaks."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 0.01)
    active = np.where(b >= threshold)[0]

    candidates = []
    for i in range(n_pert):
        a_pert = a.copy()
        # Reduce mass at positions contributing most to active peaks
        for k in active[:min(3, len(active))]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib = a[j] * a[k - j]
                if contrib > 0.1 * max_b:
                    a_pert[j] *= (1.0 - 0.02 * rng.random())
                    a_pert[k - j] *= (1.0 - 0.02 * rng.random()) if k - j != j else (1.0 - 0.01 * rng.random())
        a_pert = np.maximum(a_pert, 0)
        sp = a_pert.sum()
        if sp > 1e-15:
            a_pert = a_pert / sp
        candidates.append(a_pert)
    return candidates


def cosine_perturb(a0, rng, n_pert=5):
    """Generate cosine-based perturbations at various frequencies."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    a = a / s
    x = np.linspace(0, 2 * np.pi, n)
    candidates = []
    for i in range(n_pert):
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        strength = 0.01 + 0.03 * rng.random()
        a_pert = a + strength * np.cos(freq * x + phase)
        a_pert = np.maximum(a_pert, 0)
        sp = a_pert.sum()
        if sp > 1e-15:
            a_pert = a_pert / sp
        candidates.append(a_pert)
    return candidates


def make_palindromic(a):
    """Make a sequence palindromic (symmetric)."""
    n = len(a)
    half = (n + 1) // 2
    a_half = a[:half].copy()
    if n % 2 == 0:
        a_full = np.concatenate([a_half, a_half[::-1]])
    else:
        a_full = np.concatenate([a_half, a_half[-2::-1]])
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s > 1e-15:
        a_full = a_full / s
    return a_full


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.3 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coordinate_polish(a0, time_limit, max_rounds=30):
    """Simple coordinate descent: try scaling each coordinate by factors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    deadline = t0 + time_limit
    rng = np.random.default_rng(777)

    factors = [0.6, 0.8, 1.2, 1.4]

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved_any = False
        for j in order:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-8:
                continue
            best_factor = 1.0
            best_local_score = best_score
            for f in factors:
                a[j] = old * f
                sn = a.sum()
                if sn < 1e-15:
                    a[j] = old
                    continue
                a_norm = a / sn
                sc = 2.0 * n * fast_convolve(a_norm).max()
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_factor = f
            a[j] = old * best_factor
            if best_factor != 1.0:
                improved_any = True
                sn = a.sum()
                if sn > 1e-15:
                    a = a / sn
                    sc = eval_local(a)
                    if sc < best_score:
                        best_score = sc
                        best = a.copy()

        if not improved_any and rd > 3:
            break
        if rd % 5 == 0:
            print(f"    CoordPolish round {rd} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Ultra-fine LogSumExp L-BFGS on parent (n=1200)
    Phase 2: Nesterov peak-directed subgradient (different landscape navigation)
    Phase 3: Multi-start with directed perturbations
    Phase 4: Palindromic constrained optimization
    Phase 5: Try n=1500
    Phase 6: Try n=1800
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_parent = len(best_seq)

    # === Phase 1: Ultra-fine LogSumExp L-BFGS ===
    print(f"\n=== Phase 1: Fine L-BFGS n={n_parent} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 540.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
        a_out = lse_lbgfs_refine(best_seq, t1, T_stages, iters_per_stage=150)
        local_update(a_out, "lbgfs_fine1")

    # === Phase 2: Nesterov Peak Subgradient ===
    print(f"\n=== Phase 2: Nesterov SubGrad n={len(best_seq)} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_ns, _ = nesterov_peak_subgradient(best_seq, t2, max_iters=300000)
        local_update(a_ns, "nesterov_subgrad")

    # === Phase 3: Multi-start with directed perturbations ===
    print(f"\n=== Phase 3: Multi-Start Perturbations score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 260.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        n_trials = 5
        t_per_trial = t3 / n_trials
        for trial in range(n_trials):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            # Mix directed and cosine perturbations
            if trial % 2 == 0:
                perts = directed_peak_perturb(best_seq, rng, n_pert=1)
            else:
                perts = cosine_perturb(best_seq, rng, n_pert=1)
            a_pert = perts[0]
            t_sub = min(t_per_trial, rem * 0.4)
            if t_sub < 8:
                break
            # Quick L-BFGS refinement
            T_quick = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_opt = lse_lbgfs_refine(a_pert, t_sub, T_quick, iters_per_stage=80)
            local_update(a_opt, f"pert_trial{trial}")

    # === Phase 4: Palindromic optimization ===
    print(f"\n=== Phase 4: Palindromic Constrained score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a_pal = make_palindromic(best_seq)
        T_pal = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_pal_opt = lse_lbgfs_refine(a_pal, t4, T_pal, iters_per_stage=100)
        local_update(a_pal_opt, "palindromic")

    # === Phase 5: Try n=1500 ===
    print(f"\n=== Phase 5: n=1500 score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 220.0)
    if t5 > 40:
        print(f"  Budget: {t5:.0f}s")
        init_1500 = interpolate_seq(best_seq, 1500)
        T_1500 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_1500 = lse_lbgfs_refine(init_1500, t5 * 0.6, T_1500, iters_per_stage=100)
        local_update(a_1500, "lbgfs_n1500")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_1500_ns, _ = nesterov_peak_subgradient(best_seq if len(best_seq) == 1500 else a_1500,
                                                      min(rem, t5 * 0.35), max_iters=100000)
            local_update(a_1500_ns, "nsg_n1500")

    # === Phase 6: Try n=1800 ===
    print(f"\n=== Phase 6: n=1800 score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 180.0)
    if t6 > 40:
        print(f"  Budget: {t6:.0f}s")
        init_1800 = interpolate_seq(best_seq, 1800)
        T_1800 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_1800 = lse_lbgfs_refine(init_1800, t6 * 0.7, T_1800, iters_per_stage=80)
        local_update(a_1800, "lbgfs_n1800")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Coordinate polish
        t_cp = min(rem * 0.3, 50.0)
        if t_cp > 10:
            a_cp, _ = coordinate_polish(best_seq, t_cp, max_rounds=20)
            local_update(a_cp, "coord_polish")

        # Final ultra-fine L-BFGS
        rem = deadline - time.monotonic()
        if rem > 20:
            t_fin = min(rem * 0.4, 40.0)
            T_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lse_lbgfs_refine(best_seq, t_fin, T_fin, iters_per_stage=60)
            local_update(a_fin, "final_lbgfs")

        # Final Nesterov subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin_ns, _ = nesterov_peak_subgradient(best_seq, min(rem, 25.0), max_iters=50000)
            local_update(a_fin_ns, "final_nsg")

    # Final verification
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
        # Also verify with the actual evaluate_sequence
        try:
            actual = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"Verified with evaluate_sequence: {actual:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    if best_seq is None:
        best_seq = np.ones(100) / 100

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
