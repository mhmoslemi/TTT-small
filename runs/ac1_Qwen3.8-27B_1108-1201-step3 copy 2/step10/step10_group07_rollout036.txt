import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:

    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def fenchel_lp_iteration(a0, time_limit, solver, verbose=True):
    """
    LP best-response with Fenchel uniform averaging.
    """
    n = len(a0)

    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Fenchel average
    fenchel_avg = a.copy()
    fenchel_weight = 0.0
    
    # Also track running average
    running_avg = a.copy()
    
    # Current "center" for LP (using Fenchel average)
    center = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: best response to current center
        b_new, M_lp = lp_best_response(center, solver)
        if not np.isfinite(M_lp):

        # Update Fenchel average
        fenchel_weight += 1.0
        fenchel_avg = fenchel_avg + (b_new - fenchel_avg) / fenchel_weight
        # Actually, Fenchel uses weighted average: w_i = i

        elif it <= 30:
            alpha = 0.8
        else:
            alpha = max(0.3, 20.0 / (it + 20.0))
        center = (1 - alpha) * center + alpha * b_new
        center = np.maximum(center, 0)
        cs = center.sum()
        if cs < 1e-15:
            center = a.copy()
        else:
            center = center / cs

        # Evaluate multiple candidates
        candidates = [b_new, running_avg, fenchel_avg, center]
        for c in candidates:

            sc = c.sum()
            if sc < 1e-15:

            c = c / sc
            score = 2.0 * n * fast_convolve(c).max()
            if score < best_score:

                best_a = c.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FenchelLP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Smooth logsumexp optimization with momentum-based projected gradient
3. Coordinate descent with exact 1D optimization for fine polishing
4. Multi-restart with diverse initializations and length exploration

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M*1, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_fenchel(a0, time_limit, solver, verbose=True):
    """
    LP best-response with Fenchel uniform averaging and adaptive damping.
    """
    n = len(a0)

    t0 = time.monotonic()
    it = 0

    # Fenchel average with weights w_i = i
    fenchel_avg = a.copy()
    fenchel_num = 0.0

    # Simple running average
    run_avg = a.copy()

    # Current center for LP queries
    center = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:

        # Solve LP: best response to current center
        b_new, M_lp = lp_best_response(center, solver)
        if not np.isfinite(M_lp):
            break

        # Update Fenchel average (weighted by iteration number)
        fenchel_num += 1.0
        fenchel_avg = fenchel_avg + (b_new - fenchel_avg) / fenchel_num

        # Update running average
        run_avg = run_avg + (b_new - run_avg) / it

        # Adaptive damping for center update
        if it <= 5:
            alpha = 1.0
        elif it <= 20:
            alpha = 0.9
        elif it <= 50:
            alpha = 0.7
        else:
            alpha = max(0.4, 30.0 / (it + 10.0))

        if cs < 1e-15:
            center = a.copy()
        else:
            center = center / cs

            c_clean = np.maximum(c, 0.0)

    temp = temp_init

    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # a[k-i] for k in [k_start, k_end]
            idx_start = k_start - i  # = 0
            idx_end = k_end - i      # = n-1 or less
            a_shifted = a[idx_start:idx_end + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Try various multiplicative adjustments
            base = max(a[i], 1e-10)
            factors = [-0.7, -0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5, 0.7]

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                # Fix quadratic term at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    qidx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[qidx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                actual_delta = a[i] - old_val
                changes = 2.0 * actual_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    qidx = 2 * i - k_start
                    changes = changes.copy()
                    changes[qidx] = 2.0 * actual_delta * old_val + actual_delta * actual_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:n]  # a[0..n-1], used for k-i in [0, n-1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine multiplicative steps
            for factor in [0.995, 0.997, 0.999, 0.998, 1.001, 1.003, 1.005, 1.002]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    qidx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[qidx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                actual_delta = a[i] - old_val
                changes = 2.0 * actual_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    qidx = 2 * i - k_start
                    changes = changes.copy()
                    changes[qidx] = 2.0 * actual_delta * old_val + actual_delta * actual_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n)
    if init_type == 0:
        # Triangle (peaked in middle)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        # Cosine window
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(0.5, 2.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Periodic
        period = int(rng.integers(3, 12))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # W-shape
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Smooth random (low-frequency)
        n_freq = int(rng.integers(2, 6))
        a = np.ones(n)
        for _ in range(n_freq):
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.cos(2 * np.pi * freq * t / n + phase)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def dual_lp_solve(a, solver):
    """
    Solve the dual LP to get a lower bound on max(conv(a,b)).
    Dual: max sum(y) s.t. T^T y <= 1, y >= 0.
    The optimal dual value = optimal primal value.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    y_var = cp.Variable(size, nonneg=True)
    constraint = T.T @ y_var <= np.ones(n)
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), [constraint])
    try:
        prob.solve(solver=solver)
    except Exception:
        return float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return float('inf')

    return float(cp.sum(y_var).value)


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv)/(sum)^2.
    Uses LP best-response with Fenchel averaging as primary driver,
    smooth logsumexp optimization, and coordinate descent polishing.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    # Find LP solver
    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Best-Response with Fenchel Averaging (50% time) ===
    print(f"\n=== Phase 1: LP-Fenchel (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.50

    # Main LP run from parent
    t_alloc = min(t1_end - time.monotonic(), 600.0)
    print(f"  LP-Fenchel from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_fenchel(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Try from diverse initializations
    if time.monotonic() < t1_end - 60:
        inits = [
            diverse_init(n, rng, 0),
            diverse_init(n, rng, 1),
            diverse_init(n, rng, 2),
        ]
        for idx, init in enumerate(inits):
            remaining = t1_end - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.3, 100.0)
            print(f"  LP-Fenchel init {idx+1}/{len(inits)}, budget={t_alloc:.0f}s")
            a_tmp, s_tmp = lp_fenchel(init, t_alloc, solver, verbose=False)
            try_update(a_tmp, f"lp_init{idx+1}")

    # Second round on best so far
    if time.monotonic() < t1_end - 30:
        t_alloc = min(t1_end - time.monotonic(), 200.0)
        print(f"  LP-Fenchel round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_fenchel(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_round2")

    # === Phase 2: Smooth LogSumExp Optimization (20% time) ===
    print(f"\n=== Phase 2: Smooth (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t2_end - time.monotonic(), 200.0)
    if t_alloc > 20:
        print(f"  Smooth momentum, budget={t_alloc:.0f}s")
        a_sm1, s_sm1 = smooth_momentum(best_seq, t_alloc, temp_init=0.05, temp_min=0.003, verbose=True)
        try_update(a_sm1, "smooth1")

        if time.monotonic() < t2_end - 20:
            t_alloc2 = min(t2_end - time.monotonic(), 100.0)
            a_sm2, s_sm2 = smooth_momentum(a_lp, t_alloc2, temp_init=0.02, temp_min=0.001, verbose=False)
            try_update(a_sm2, "smooth2")

    # === Phase 3: Coordinate Descent (20% time) ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t3_end - time.monotonic(), 250.0)
    if t_alloc > 20:
        print(f"  CD from best, budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_alloc, verbose=True)
        try_update(a_cd, "cd_best")

        if time.monotonic() < t3_end - 30:
            t_alloc2 = min(t3_end - time.monotonic(), 120.0)
            a_cd2, s_cd2 = coord_descent(a_lp, t_alloc2, verbose=False)
            try_update(a_cd2, "cd_lp")

    # === Phase 4: Fine Polish + Final Checks (10% time) ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 10

    if time.monotonic() < t4_end - 20:
        t_fp = min(t4_end - time.monotonic() - 10, 80.0)
        if t_fp > 10:
            print(f"  FinePolish, budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine")

    # Final CD pass
    if time.monotonic() < t4_end - 5:
        t_final = min(t4_end - time.monotonic() - 3, 30.0)
        if t_final > 5:
            a_fin, s_fin = coord_descent(best_seq, t_final, verbose=False)
            try_update(a_fin, "final_cd")

    # One more fine polish at the very end
    if time.monotonic() < deadline - 3:
        t_last = min(deadline - time.monotonic() - 2, 15.0)
        if t_last > 3:
            a_last, s_last = fine_polish(best_seq, t_last, verbose=False)
            try_update(a_last, "last_polish")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
