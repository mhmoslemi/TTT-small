import time

_LP_SOLVER = None

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:

    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:

            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name

    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]

    if s < 0.01:
        return float('inf')

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def subgradient_descent(a0, time_limit, verbose=False):

    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        perturb_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.35, 90.0)

        pert_type = perturb_count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Peak-focused reduction
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                ii_start = max(0, int(k) - n + 1)
                ii_end = min(n - 1, int(k))
                idx = np.arange(ii_start, ii_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.04 * rng.random())
        elif pert_type == 1:
            # Gaussian noise
            sigma = 0.015 + 0.025 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            # Block swap
            blen = max(4, rng.integers(n // 10, n // 4))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            # Shifted mix
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, shift)
            perturbed = 0.75 * a + 0.25 * shifted
        else:
            # Two-site swap
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            amt = a[i1] * rng.uniform(0.1, 0.5)
            perturbed[i1] -= amt
            perturbed[i2] += amt

        perturbed = normalize_seq(perturbed)

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_{perturb_count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {perturb_count}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.25, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.1, 0.2)
    w2 = n * rng.uniform(0.1, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
