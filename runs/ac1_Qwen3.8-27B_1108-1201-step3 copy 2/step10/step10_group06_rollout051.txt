    conv = fast_convolve(a)
    max_val = conv.max()
    # Active lags
    active = np.where(conv >= max_val * (1 - 1e-10))[0]
    # Subgradient
    grad = np.zeros(len(a))
    for k in active:
        for i in range(len(a)):

            if 0 <= j < len(a):
                if k == 2*i:
                    grad[i] += 2 * a[i]
                else:
                    grad[i] += 2 * a[j]
    grad /= len(active)
    # Mirror descent step (multiplicative weights)
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            # Determine bracket
            upper = max(4.0 * v, 8.0 / n, 0.5 / n)
            best_x = golden_section_search(_score_at, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def pair_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent that updates pairs of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try pairs (i, j) where j is near i
        for i in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            j = i + 1
            vi = a[i]
            vj = a[j]
            if vi < 1e-15 and vj < 1e-15:
                continue
            
            # For a pair update, we keep vi+vj constant and vary the ratio
            total_v = vi + vj
            if total_v < 1e-15:
                continue
            # Try alpha in [0, 1]: a_i = alpha*total_v, a_j = (1-alpha)*total_v
            def _pair_score(alpha):
                new_ai = alpha * total_v
                new_aj = (1 - alpha) * total_v
                delta_i = new_ai - vi
                delta_j = new_aj - vj
                # Update conv
                conv_new = conv.copy()
                # Effect of changing a[i]
                for k in range(i, min(i + n, 2 * n - 1)):
                    partner = k - i
                    if 0 <= partner < n:
                        conv_new[k] += delta_i * a[partner]
                for k in range(j, min(j + n, 2 * n - 1)):
                    partner = k - j
                    if 0 <= partner < n:
                        conv_new[k] += delta_j * a[partner]
                # Fix the cross term (i,j pair counted twice)
                for k in range(max(i, j), min(i + n, j + n, 2 * n - 1)):
                    pass  # Already handled correctly since we used original a
                # Actually this is getting complex; let's just do it properly
                a_trial = a.copy()
                a_trial[i] = new_ai
                a_trial[j] = new_aj
                conv_trial = fast_convolve(a_trial)
                return 2.0 * n * float(conv_trial.max()) / (current_sum * current_sum)
            
            # Golden section on alpha
            # Current alpha
            cur_alpha = vi / total_v if total_v > 0 else 0.5
            best_alpha = golden_section_search(_pair_score, 0.0, 1.0, max_iter=25)
            
            if abs(best_alpha - cur_alpha) > 1e-10:
                new_ai = best_alpha * total_v
                new_aj = (1 - best_alpha) * total_v
                a[i] = new_ai
                a[j] = new_aj
                conv = fast_convolve(a)
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose:
            print(f"    PCD {label}: best={best_score:.8f}")

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False, use_dual=True):
    """LP Fictitious Play with adaptive step sizes and dual information."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    a_ema = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b, M_val, dual = solve_lp_best_response(a, solver)
        if b is None:
            b, M_val = solve_lp_best_response_fast(a, solver)
            if b is None:
                continue

        # Adaptive step size: larger when progress is being made
        if no_improve < 5:
            base_t = 0.4
        elif no_improve < 15:
            base_t = 0.2
        elif no_improve < 30:
            base_t = 0.1
        else:
            base_t = 0.05

        # Line search over step sizes
        best_t = 0.0
        best_ls = best_score
        trial_ts = np.linspace(0.05, 1.0, 8)
        for trial_t in trial_ts:
            t_actual = trial_t * base_t
            cand = (1.0 - t_actual) * a + t_actual * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = t_actual

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-13:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            # Use exponential averaging when no direct improvement
            alpha = 2.0 / (iters + 10.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        # EMA tracking
        a_ema = 0.95 * a_ema + 0.05 * a
        sc_ema = compute_score(a_ema)
        if sc_ema < best_score:
            best_score = sc_ema
            best_a = a_ema.copy()

        # Also check b itself
        sc_b = compute_score(b)
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP it={iters}: best={best_score:.8f}, no_improve={no_improve}")

        if no_improve > 50:
            break

    return best_a, best_score, iters

def multiplicative_weights(a0, time_limit, verbose=False, label=""):
    """Multiplicative weights / mirror descent approach."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.1
    iters = 0
    no_improve = 0
    a_avg = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Find active lags (top few)
        top_k = min(20, 2 * n - 1)
        active_lags = np.argpartition(conv, -top_k)[-top_k:]
        active_vals = conv[active_lags]
        # Weight by proximity to max
        temp = 0.01 * max_val
        w = np.exp((active_vals - active_vals.max()) / temp)
        w = w / w.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(active_lags)):
            wi = w[idx]
            if wi < 1e-12:
                continue
            k = int(active_lags[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if i == j:
                    grad[i] += wi * 2.0 * a[i]
                else:
                    grad[i] += wi * a[j]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        ns = a_new.sum()
        if ns < 1e-15:
            a_new = a
        else:
            a_new = a_new / ns

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_avg = (iters * a_avg + a_new) / (iters + 1)
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new
        # Adaptive learning rate
        if no_improve > 50:
            eta *= 0.7
            no_improve = 0
        eta = max(eta * 0.999, 0.001)
        
        if no_improve > 300:
            break

        if verbose and iters % 50 == 0:
            print(f"    MW {label} it={iters}: best={best_score:.8f}")

    return best_a, best_score

def spectral_refinement(a0, time_limit, verbose=False, label=""):
    """Flatten the autocorrelation spectrum using iterative spectral methods."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Compute "pressure" at each position based on which lags are high
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            excess = max(0, conv[k] - 0.98 * max_val)
            if excess > 0:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if i == j:
                        pressure[i] += 2.0 * excess * a[i]
                    else:
                        pressure[i] += excess * a[j]

        # Update: reduce pressure
        a_new = a / (1.0 + 0.05 * pressure / max_val)
        a_new = np.maximum(a_new, 0)
        ns = a_new.sum()
        if ns < 1e-15:
            a_new = a
        else:
            a_new = a_new / ns

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score

def symmetric_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent on the half-sequence (exploiting palindromic symmetry)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    # Make symmetric
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    a_sym = a_sym / s
    
    # Work with half sequence
    half_n = (n + 1) // 2
    half = a_sym[:half_n].copy()
    
    def expand(h):
        if n % 2 == 0:
            return np.concatenate([h[:n//2], h[:n//2][::-1]])
        else:
            return np.concatenate([h[:n//2], h[n//2], h[:n//2][::-1]])
    
    a_full = expand(half)
    best_score = compute_score(a_full)
    best_half = half.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(half_n):
            if time.monotonic() - t0 > time_limit:
                break
            if half[i] < 1e-15:
                continue
            
            # Try varying half[i] with golden section
            v = half[i]
            def _sc(alpha):
                h_new = half.copy()
                h_new[i] = alpha
                h_new = np.maximum(h_new, 0)
                h_new_s = h_new.sum()
                if h_new_s < 1e-15:
                    return 1e10
                h_new = h_new / h_new_s
                f = expand(h_new)
                return compute_score(f)
            
            upper = max(4.0 * v, 2.0 / n)
            best_v = golden_section_search(_sc, 0.0, upper, max_iter=20)
            
            if abs(best_v - v) > 1e-12:
                half[i] = best_v
                a_full = expand(half)
                sc = compute_score(a_full)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_half = half.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

    return expand(best_half), best_score

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 60.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise (moderate)
            sigma = 0.01 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Localized perturbation
            center = rng.integers(n // 4, 3 * n // 4)
            width = rng.integers(n // 16, n // 4)
            mask = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            perturbed = a * (1.0 + 0.1 * rng.standard_normal(n) * mask)
        elif pert_type == 2:
            # Frequency-domain perturbation
            fft_size = 1 << (n - 1).bit_length()
            F = rfft(a, fft_size)
            F += rng.standard_normal(fft_size // 2 + 1) * 0.005
            perturbed = irfft(F, fft_size)[:n]
        elif pert_type == 3:
            # Swap two random segments
            seg_len = rng.integers(n // 10, n // 4)
            i1 = rng.integers(0, n - seg_len)
            i2 = rng.integers(0, n - seg_len)
            if abs(i1 - i2) > seg_len:
                temp = perturbed[i1:i1+seg_len].copy()
                perturbed[i1:i1+seg_len] = perturbed[i2:i2+seg_len]
                perturbed[i2:i2+seg_len] = temp
        else:
            # Exponential tilt
            tilt = rng.uniform(-0.5, 0.5)
            x = np.linspace(0, 1, n)
            perturbed = a * np.exp(tilt * (x - 0.5))

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-11:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")
            # Also refine with multiplicative weights
            t_rem = time_limit - (time.monotonic() - t0)
            if t_rem > 30:
                a_opt2, s_opt2 = multiplicative_weights(a_opt, min(t_rem * 0.3, 30), label=f"mw_{count}")
                if s_opt2 < best_score - 1e-11:
                    best_score = s_opt2
                    best_a = a_opt2.copy()

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3, width_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_peak(n, rng):
    """Initialize with two peaks."""
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    w1 = n * rng.uniform(0.1, 0.2)
    w2 = n * rng.uniform(0.1, 0.2)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    return a / s

def init_cosine_modulated(n, rng):
    """Cosine-modulated window."""
    t = np.linspace(0, 1, n)
    freq = rng.uniform(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    envelope = np.sin(np.pi * t)
    a = envelope * (0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        print(f"  ECD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph3_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD r{rnd+1}: {sc:.8f}")
        else:
            print(f"  ECD r{rnd+1}: no improvement (got {sc:.8f})")
            # Try perturbation to escape
            perturbed = best_seq + rng.standard_normal(n_work) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 40.0))
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST ECD-pert r{rnd+1}: {sc2:.8f}")

    # Phase 4: Multiplicative weights
    print(f"\n=== Phase 4: Multiplicative Weights ===")
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t_mw = min(rem * 0.4, 80.0)
        print(f"  MW budget={t_mw:.0f}s")
        a_mw, s_mw = multiplicative_weights(best_seq, t_mw, verbose=True, label="main")
        sc = float(evaluate_sequence(a_mw.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_mw.copy()
            print(f"  NEW BEST MW: {sc:.8f}")

    # Phase 5: Spectral refinement
    print(f"\n=== Phase 5: Spectral Refinement ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t_sr = min(rem * 0.3, 50.0)
        print(f"  SR budget={t_sr:.0f}s")
        a_sr, s_sr = spectral_refinement(best_seq, t_sr, verbose=True, label="main")
        sc = float(evaluate_sequence(a_sr.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sr.copy()
            print(f"  NEW BEST SR: {sc:.8f}")
        # Polish after spectral
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 10:
            a_cd3, s_cd3, _ = exact_coord_descent(a_sr, min(rem2 * 0.4, 30.0), label="sr_polish")
            sc3 = float(evaluate_sequence(a_cd3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:
                best_score = sc3
                best_seq = a_cd3.copy()
                print(f"  NEW BEST SR+CD: {sc3:.8f}")

    # Phase 6: Basin hopping
    print(f"\n=== Phase 6: Basin Hopping ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_bh = min(rem * 0.6, 180.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 7: Symmetric variant
    print(f"\n=== Phase 7: Symmetric Optimization ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sym = min(rem * 0.5, 60.0)
        print(f"  Sym budget={t_sym:.0f}s")
        a_sym, s_sym = symmetric_coord_descent(best_seq, t_sym, verbose=True, label="sym")
        sc = float(evaluate_sequence(a_sym.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sym.copy()
            print(f"  NEW BEST Sym: {sc:.8f}")

    # Phase 8: Final intensive polishing
    print(f"\n=== Phase 8: Final Polish ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_fin = min(rem * 0.5, 60.0)
        print(f"  Final ECD budget={t_fin:.0f}s")
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_fin, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST Final: {sc:.8f}")

    # One more LP pass if time
    rem = deadline - time.monotonic() - 20
    if rem > 30:
        t_lp = min(rem * 0.6, 40.0)
        print(f"  Final LP budget={t_lp:.0f}s")
        a_lp, s_lp, _ = lp_fictitious_play(best_seq, t_lp, solver=solver)
        sc = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(sc) and sc < best_score:

            best_seq = a_lp.copy()
            print(f"  NEW BEST Final LP: {sc:.8f}")
        # Quick ECD after
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 5:
            a_last, s_last, _ = exact_coord_descent(best_seq, min(rem2 * 0.8, 20.0), label="last")
            sc2 = float(evaluate_sequence(a_last.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_last.copy()

    # Final: one more ECD pass
    rem = deadline - time.monotonic() - 5
    if rem > 3:
        a_fin2, s_fin2, _ = exact_coord_descent(best_seq, min(rem * 0.7, 10.0), label="fin2")
        sc = float(evaluate_sequence(a_fin2.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin2.copy()

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
