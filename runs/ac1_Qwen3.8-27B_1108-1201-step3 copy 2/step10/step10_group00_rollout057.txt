from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Multi-stage temperature annealing (T: 10 -> 0.005)
3. Multi-length exploration (n=1000, 1200, 1400, 1600)
4. Active-set peeling and local refinement
Key innovation: O(n log n) gradient via FFT cross-correlation,
enabling far more iterations than the previous O(n^2) approach.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_grad_fft(a, T):
    """
    Compute LogSumExp of conv(a,a)/T and its gradient w.r.t. a.
    O(n log n) via FFT.
    Returns (lse_value, grad_a) where grad_a = d(2*n*LSE)/d(a).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = exp_shifted / sum_exp
    lse = max_b + T * np.log(sum_exp)
    
    # Gradient: d(2*n*LSE)/d(a[i]) = (4*n/T) * sum_k w[k]*a[k-i]
    # Compute cross-correlation via FFT
    # w has length 2n-1, a has length n
    # cross[i] = sum_k w[k] * a[k-i] (valid k, valid k-i)
    N = 1
    while N < 3 * n:
        N *= 2
    W = rfft(w, N)
    A = rfft(a, N)
    cross = irfft(W * A.conj(), N)[:n]
    
    grad_a = (4.0 * n / T) * cross
    return 2.0 * n * lse, grad_a


def compute_lse_grad_z(z, T):
    """Compute objective and gradient in z-space (log-parameterization)."""
    a = softmax_from_z(z)
    obj, grad_a = compute_lse_grad_fft(a, T)
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_lse_optimize(z0, T_stages, time_limit, lr_base=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer on LogSumExp with temperature annealing.
    z0: initial log-parameters
    T_stages: list of temperatures to anneal through
    Returns: final z
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    # Allocate more iterations to lower temperatures
    total_iters = 100000  # will be limited by time
    # Distribute iterations: more to lower T
    n_stages = len(T_stages)
    # Give 70% of iterations to the last 30% of stages
    weights = np.ones(n_stages)
    weights[-min(3, n_stages):] *= 3.0
    weights = weights / weights.sum()
    
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        stage_iter_budget = int(total_iters * weights[stage_idx])
        
        for _ in range(stage_iter_budget):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            
            obj, grad_z = compute_lse_grad_z(z, T)
            
            # Adaptive learning rate
            lr = lr_base / (1.0 + 0.1 * t)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic print
            if t % 5000 == 0:
                a = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a).max()
                elapsed = time.monotonic() - t0
                print(f"      Adam T={T:.4f} iter={t} ({elapsed:.0f}s): obj={obj:.8f} score={score:.8f}")
    
    return z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Adam optimizer on LogSumExp surrogate (O(n log n) per step)
2. Multi-stage temperature annealing (T: 15 -> 0.005)
3. Multi-length exploration (n=1000, 1200, 1400, 1600)
4. Nesterov FISTA as complementary optimizer
5. Peak-shaving local refinement
Key innovation: O(n log n) gradient via FFT cross-correlation enables
10x+ more optimization iterations than previous O(n^2) approach.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_grad_fft(a, T):
    """
    Compute 2*n*LSE(conv(a,a)/T) and its gradient w.r.t. a.
    O(n log n) via FFT cross-correlation.
    Returns (objective, grad_a).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    shifted = np.clip(shifted, -60, 60)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = exp_shifted / sum_exp
    lse = max_b + T * np.log(sum_exp)

    # Cross-correlation via FFT: cross[i] = sum_k w[k]*a[k-i]
    N = 1
    while N < 3 * n:
        N *= 2
    W = rfft(w, N)
    A = rfft(a, N)
    cross = irfft(W * A.conj(), N)[:n]

    grad_a = (4.0 * n / T) * cross
    return 2.0 * n * lse, grad_a


def compute_obj_grad_z(z, T):
    """Objective and gradient in z-space (log-parameterization)."""
    a = softmax_from_z(z)
    obj, grad_a = compute_lse_grad_fft(a, T)
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_optimize(z0, T_stages, time_limit, lr_base=0.1, beta1=0.85, beta2=0.999, eps=1e-8,
                  print_interval=3000):
    """
    Adam optimizer on LogSumExp with temperature annealing.
    Returns (final_z, best_score).
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    best_score = float('inf')
    best_z = z.copy()
    a = softmax_from_z(z)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_z = z.copy()

    n_stages = len(T_stages)
    # Allocate iterations: more to lower temperatures
    weights = np.linspace(0.5, 2.0, n_stages)
    weights = weights / weights.sum()

    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        # Adaptive lr per stage: higher T allows larger steps
        lr = lr_base * min(1.0, 2.0 / (T + 1.0))
        stage_budget = 80000  # max iters per stage (time-limited anyway)

        for _ in range(stage_budget):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1

            obj, grad_z = compute_obj_grad_z(z, T)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Check progress periodically
            if t % print_interval == 0:
                a = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_z = z.copy()
                elapsed = time.monotonic() - t0
                print(f"        Adam T={T:.4f} t={t} ({elapsed:.0f}s): score={best_score:.8f}")

    # Final check
    a = softmax_from_z(z)
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_z = z.copy()

    return best_z, best_score


def fista_optimize(z0, T_stages, time_limit, lr0=0.01, print_interval=5000):
    """
    FISTA (Nesterov-accelerated) gradient descent on LogSumExp.
    Returns (final_z, best_score).
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    y = z0.copy().astype(np.float64)
    t_k = 1.0
    t0_time = time.monotonic()

    best_score = float('inf')
    best_z = z.copy()
    a = softmax_from_z(z)
    best_score = 2.0 * n * fast_convolve(a).max()

    n_stages = len(T_stages)
    weights = np.linspace(0.5, 2.5, n_stages)
    weights = weights / weights.sum()

    it = 0
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0_time > time_limit:
            break
        lr = lr0 * min(1.0, 1.5 / (T + 0.5))
        stage_budget = 40000

        for _ in range(stage_budget):
            if time.monotonic() - t0_time > time_limit:
                break
            it += 1

            obj_y, grad_y = compute_obj_grad_z(y, T)
            z_new = y - lr * grad_y

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = z_new + ((t_k - 1.0) / t_new) * (z_new - z)
            z = z_new
            t_k = t_new

            if it % print_interval == 0:
                a = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_z = z.copy()
                elapsed = time.monotonic() - t0_time
                print(f"        FISTA T={T:.4f} it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    a = softmax_from_z(z)
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_z = z.copy()

    return best_z, best_score


def peak_shaving(a, n_steps=5000, strength=0.01):
    """
    Local peak-shaving: identify coordinates contributing most to the peak
    and reduce them, redistributing mass to less-contributing coordinates.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = np.argmax(b)

        # Find coordinates that contribute most to the peak
        # b[peak_idx] = sum_j a[j]*a[peak_idx - j]
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contributions = a[js:je + 1] * a[peak_idx - (js:je + 1)]
        total_contrib = contributions.sum()
        if total_contrib < 1e-30:
            break

        # Reduce top contributors, increase others
        n_reduce = max(1, n // 20)
        top_idx_in_range = np.argsort(contributions)[-n_reduce:]
        reduce_mask = np.zeros(n, dtype=bool)
        reduce_mask[js + top_idx_in_range] = True

        # Redistribute: move mass from reduce_mask to others
        reduce_amount = strength * a[reduce_mask].sum()
        if reduce_amount < 1e-15:
            break

        a[reduce_mask] -= reduce_amount / max(1, reduce_mask.sum())
        a[reduce_mask] = np.maximum(a[reduce_mask], 0)

        # Add to non-reduced
        add_mask = ~reduce_mask
        a[add_mask] += reduce_amount / max(1, add_mask.sum())

        s_new = a.sum()
        if s_new < 1e-15:
            break
        a = a / s_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        elif step > 100 and score > best_score * 1.01:
            # Revert if too bad
            a = best_a.copy()

        # Decay strength
        strength *= 0.999

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if s < 1e-15:

            z_pert_adv, _ = adam_optimize(z_pert, T_sub, t_sub, lr_base=0.05)
            a_pert_adv = softmax_from_z(z_pert_adv)
            try_update(a_pert_adv, f"pert{trial}")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n=== Phase 9: Final Ultra-Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 50.0)
        if t_fin > 10:
            z_fin = z_from_a(best_seq)
            T_fin = [0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
            z_final, _ = adam_optimize(z_fin, T_fin, t_fin, lr_base=0.03)
            a_final = softmax_from_z(z_final)
            try_update(a_final, "final_polish")

    # Final peak shaving on best
    rem = deadline - time.monotonic()
    if rem > 15:
        a_last = peak_shaving(best_seq, n_steps=10000, strength=0.003)
        try_update(a_last, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
