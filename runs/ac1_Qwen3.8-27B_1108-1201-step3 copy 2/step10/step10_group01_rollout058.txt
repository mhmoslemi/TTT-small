SOTA: uses the correct LogSumExp smoothing f_tau = 2n*tau*log(sum exp(c_k/tau))

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smoothed_grad(a, tau):
    """Compute gradient of 2n*tau*log(sum_k exp(c_k/tau)) where c=conv(a,a).
    Returns (grad, c, objective_value)."""
    n = len(a)
    c = fast_convolve(a)
    c_scaled = c / tau
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / exp_c.sum()
    obj = 2.0 * n * tau * (c_max + np.log(exp_c.sum()))
    # d_j = sum_m a_m * w_{m+j} = cross_correlation(a, w)[j]
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    d = irfft(A * np.conj(W), fft_size)[:n]
    grad = 4.0 * n * d
    return grad, c, obj, w

    while time.monotonic() - t0 < time_limit:
        grad, c = compute_true_grad(a)
        grad_centered = grad - grad.mean()

        a = a - lr * grad_centered
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            s = 1.0
        else:
            a = a / s
        iters += 1
        lr *= lr_decay

        if iters % check_interval == 0:
            c_fresh = fast_convolve(a)
            score = 2.0 * n * c_fresh.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.8
                    no_improve = 0

    return best_a, best_score


def coordinate_polish(a0, time_limit, top_k=5, check_interval=100):
    """Simple coordinate descent on the true objective targeting top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        max_c = c.max()
        top_peaks = np.argsort(c)[::-1][:top_k]

        improved = False
        for peak_k in top_peaks:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = idx[np.argmax(contribs)]

            for factor in [0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[top_contrib] *= factor
                diff = a[top_contrib] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[top_contrib] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                c_trial = fast_convolve(a_trial)
                score = 2.0 * n * c_trial.max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation to escape
                a_pert = a + 0.01 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                if 2.0 * n * fast_convolve(a_pert).max() < best_score:
                    a = a_pert
                    stall = 0
                elif stall > 500:
                    break
        else:
            stall = 0

    return best_a, best_score


def get_lp_solver():
    """Try to find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    except ImportError:
        return None


def quick_lp_alternation(a0, time_limit, n_iters=200):
    """Quick LP alternation for initialization."""
    solver = get_lp_solver()
    if solver is None:
        return a0, 2.0 * len(a0) * fast_convolve(a0).max()

    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = 0.1

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        size = 2 * n - 1
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = a

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb > 1e-15:
                    b_val = b_val / sb
                    a_new = (1.0 - beta) * a + beta * b_val
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    score = 2.0 * n * fast_convolve(a_new).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                    a = a_new
        except Exception:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses Adam+LogSumExp with correct temperature annealing,
    subgradient polish, coordinate descent, and multi-start at different lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = 2.0 * n_init * fast_convolve(best_seq).max()
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Quick LP for initialization (30s) ===
    rem = deadline - time.monotonic()
    if rem > 50:
        print(f"\n=== Phase 1: Quick LP init (score={best_score:.8f}) ===")
        t1 = min(rem * 0.08, 35.0)
        a_lp, _ = quick_lp_alternation(best_seq, t1, n_iters=100)
        try_update(a_lp, "lp_init")

    # === Phase 2: Main Adam + LogSumExp optimization (longest phase) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 2: Adam+LogSumExp (score={best_score:.8f}) ===")
    t2 = min(rem * 0.55, 900.0)
    if t2 > 50:
        print(f"  Budget: {t2:.0f}s, n={n_best}")
        a_main, _ = adam_logsumexp_optimize(
            best_seq, t2,
            lr_init=0.01, lr_decay=0.99997,
            beta1=0.9, beta2=0.999,
            tau_schedule=[3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.65, 0.55, 0.45,
                          0.38, 0.32, 0.27, 0.22, 0.18, 0.15, 0.12, 0.10,
                          0.08, 0.065, 0.05, 0.04, 0.032, 0.025, 0.02,
                          0.015, 0.012, 0.009, 0.006, 0.004, 0.003, 0.002,
                          0.0015, 0.001, 0.0006, 0.0003, 0.0001]
        )
        try_update(a_main, "adam_main")

    # === Phase 3: Subgradient polish on true max (100s) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 3: Subgradient polish (score={best_score:.8f}) ===")
    t3 = min(rem * 0.15, 120.0)
    if t3 > 15:
        a_sg, _ = subgradient_polish(best_seq, t3, lr_init=0.005, lr_decay=0.999)
        try_update(a_sg, "subgrad")

    # === Phase 4: Coordinate descent (60s) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 4: Coordinate descent (score={best_score:.8f}) ===")
    t4 = min(rem * 0.08, 60.0)
    if t4 > 10:
        a_cd, _ = coordinate_polish(best_seq, t4, top_k=5)
        try_update(a_cd, "coord")

    # === Phase 5: Second Adam pass from best (60s) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 5: Adam pass 2 (score={best_score:.8f}) ===")
    t5 = min(rem * 0.12, 80.0)
    if t5 > 20:
        a_am2, _ = adam_logsumexp_optimize(
            best_seq, t5,
            lr_init=0.005, lr_decay=0.99998,
            tau_schedule=[0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03,
                          0.02, 0.012, 0.007, 0.004, 0.002, 0.001, 0.0005,
                          0.0002, 0.0001]
        )
        try_update(a_am2, "adam2")

    # === Phase 6: Multi-start perturbations (60s) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Multi-start (score={best_score:.8f}) ===")
    if rem > 50:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 40.0)
        for pi in range(3):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            x = np.linspace(0, 2 * np.pi, n_cur)
            if pi == 0:
                pert = 0.02 * np.cos(2 * x) + 0.015 * np.cos(5 * x)
            elif pi == 1:
                pert = 0.015 * rng.standard_normal(n_cur)
            else:
                pert = 0.02 * np.sin(3 * x) + 0.01 * np.sin(7 * x)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            a_res, _ = adam_logsumexp_optimize(
                a_pert, t_each,
                lr_init=0.008, lr_decay=0.99995,
                tau_schedule=[1.0, 0.7, 0.5, 0.35, 0.25, 0.17, 0.12,
                              0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0001]
            )
            try_update(a_res, f"perturb_{pi}")

    # === Phase 7: Different length exploration (60s) ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 7: Alt length (score={best_score:.8f}) ===")
    if rem > 50:
        n_cur = len(best_seq)
        for alt_n in [int(n_cur * 1.15), int(n_cur * 0.85)]:
            if time.monotonic() > deadline - 20:
                break
            t_alt = min(rem * 0.4, 50.0)
            if t_alt < 15:
                break
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            old_x = np.linspace(0, 1, n_cur)
            new_x = np.linspace(0, 1, alt_n)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()
            a_alt, _ = adam_logsumexp_optimize(
                a_interp, t_alt,
                lr_init=0.01, lr_decay=0.99995,
                tau_schedule=[2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.17,
                              0.12, 0.08, 0.05, 0.03, 0.02, 0.01, 0.006,
                              0.003, 0.001, 0.0005]
            )
            try_update(a_alt, f"alt_n{alt_n}")
            rem = deadline - time.monotonic()

    # === Phase 8: Final polish ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 8: Final polish (score={best_score:.8f}) ===")
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        a_fin, _ = subgradient_polish(best_seq, t_fin, lr_init=0.002, lr_decay=0.998)
        try_update(a_fin, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cdf, _ = coordinate_polish(best_seq, min(rem * 0.5, 25.0), top_k=4)
        try_update(a_cdf, "final_coord")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_amf, _ = adam_logsumexp_optimize(
            best_seq, min(rem * 0.7, 20.0),
            lr_init=0.003, lr_decay=0.99999,
            tau_schedule=[0.1, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0001]
        )
        try_update(a_amf, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = 2.0 * len(best_seq) * fast_convolve(best_seq).max()
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
