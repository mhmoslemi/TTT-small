import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = 'ECOS'
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        for s in ['SCS', 'CLARABEL', 'PDLP', 'ECOS']:
            try:
                prob.solve(solver=s)
                break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def spectral_init(n, num_iters=200, rng=None):
    """Gerchberg-Saxton style: flatten spectrum, enforce non-negativity."""
    if rng is None:
        rng = np.random.default_rng(42)
    # Start with a reasonable shape
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for it in range(num_iters):
        # FFT
        A = fft(a, 2*n-1)
        # Flatten magnitude (but keep DC fixed)
        mag = np.abs(A)
        mag[0] = 1.0  # sum(a) = 1
        # Target: flat magnitude
        target_mag = np.ones_like(mag)
        # Blend
        mag_new = 0.5 * mag + 0.5 * target_mag
        # Keep phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        # IFFT
        a_new = np.real(ifft(A_new, 2*n-1))[:n]
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a

def mirror_descent(a0, time_limit, eta0=0.05, active_frac=0.999, nesterov=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

    a = a / s
    y = a.copy()

    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # BB step
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.3 * eta + 0.7 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        # Update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        a_new = a_new / a_new_sum

            no_improve = 0

            y = a.copy()
            no_improve = 0

def interpolate_sequence(f, new_n):

    ss = f_interp.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating with Nesterov
acceleration, spectral initialization, symmetric projection, and targeted
peak-aware perturbations. Exploits multiple sequence lengths and strategic
restarts to escape local minima.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def spectral_init(n, num_iters=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x)
    a = np.maximum(a, 0)
    a = a / a.sum()
    for it in range(num_iters):
        A = fft(a, 2 * n - 1)
        mag = np.abs(A)
        mag[0] = 1.0
        target_mag = np.ones_like(mag)
        blend = 0.3 + 0.4 * (it / num_iters)
        mag_new = (1 - blend) * mag + blend * target_mag
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, 2 * n - 1))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a


def symmetric_project(a):
    n = len(a)
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def lp_alternating(a0, time_limit, solver, use_symmetry=False):

def mirror_descent(a0, time_limit, eta0=0.05, active_frac=0.999):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(7777)

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.4 * eta + 0.6 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        a_new = a_new / a_new_sum

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys

        a = a_new
        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 600:
            noise = rng.standard_normal(n) * 0.004
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f} |A|={len(active)}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            print(f"    CD {time.monotonic()-t0:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
