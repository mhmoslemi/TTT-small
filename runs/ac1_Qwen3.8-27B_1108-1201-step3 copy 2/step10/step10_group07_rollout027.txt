import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_alternate(a0, time_limit, solver, verbose=True):
    """LP-based alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    it = 0
    prev_score = best_score
    oscillation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        b_new, M_lp, _ = solve_lp(a, solver)
        if not np.isfinite(M_lp):
            break
        
        # Evaluate b_new
        score_b = 2.0 * n * fast_conv(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Adaptive update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 15.0 / (it + 20.0))
        
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        a = a_new
        
        score_a = 2.0 * n * fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Detect oscillation
        if score_a > prev_score * 1.001:
            oscillation += 1
        else:
            oscillation = max(0, oscillation - 1)
        
        # If oscillating, reduce alpha more aggressively
        if oscillation > 5:
            alpha = 0.2
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            oscillation = 0
        
        prev_score = score_a
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} cur={score_a:.8f}")
    
    return best_a, best_score


def coord_descent_optimal(a0, time_limit, verbose=True, label=""):
    """Coordinate descent with optimal step via bisection on convex 1D subproblem."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-14:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())
            
            # Precompute: a_shifted[j] = a[k_start-i+j] for j in range(affected_len)
            a_shifted = a[k_start - i : k_end - i + 1].copy()
            conv_affected = conv[k_start : k_end + 1].copy()
            quad_idx = 2 * i - k_start  # index in affected where k=2i (quadratic term)
            
            def eval_M(delta):
                # Compute new M given delta change to a[i]
                new_aff = conv_affected + 2.0 * delta * a_shifted
                if 0 <= quad_idx < affected_len:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] += delta * delta
                return max(unaff_max, float(new_aff.max()))
            
            # Bisection to find optimal delta in [-a[i], a[i]*2]
            lo = -a[i]
            hi = a[i] * 2.0
            
            # Check endpoints first
            M_lo = eval_M(lo)
            M_hi = eval_M(hi)
            M_0 = M  # current value (delta=0)
            
            best_delta = 0.0
            best_M = M_0
            if M_lo < best_M:
                best_M = M_lo
                best_delta = lo
            if M_hi < best_M:
                best_M = M_hi
                best_delta = hi
            
            # Bisection (20 iterations)
            if lo < 0 < hi:
                for _ in range(20):
                    mid1 = lo + (hi - lo) / 3.0
                    mid2 = hi - (hi - lo) / 3.0
                    m1 = eval_M(mid1)
                    m2 = eval_M(mid2)
                    if m1 < m2:
                        hi = mid2
                    else:
                        lo = mid1
                    mid = (lo + hi) / 2.0
                    mm = eval_M(mid)
                    if mm < best_M:
                        best_M = mm
                        best_delta = mid
            
            if best_delta != 0.0 and best_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update conv
                changes = 2.0 * best_delta * a_shifted
                if 0 <= quad_idx < affected_len:
                    changes = changes.copy()
                    changes[quad_idx] += best_delta * best_delta
                conv[k_start : k_end + 1] += changes
                M = best_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def smoothed_gradient(a0, time_limit, verbose=True):
    """
    Gradient descent on smoothed objective (log-sum-exp approximation to max).
    Uses adaptive temperature and backtracking line search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    it = 0
    temperature = 0.01  # start close to max
    lr = 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Compute softmax weights with adaptive temperature
        # Temperature: start small (close to max), increase if stuck
        scaled = (conv - M) / max(temperature, 1e-10)
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute subgradient: sg[i] = 2 * sum_k w[k] * a[k-i] (valid bounds)
        sg = np.zeros(n)
        # Vectorized computation
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end+1] += 2.0 * w[k] * a[k - i_end : k - i_start + 1][::-1]
        
        # Projected gradient step with backtracking
        step = lr
        a_new = a - step * sg
        a_new = simplex_project(a_new)
        
        new_score = 2.0 * n * fast_conv(a_new).max()
        
        # Backtracking
        while new_score > best_score * 1.001 and step > 1e-6:
            step *= 0.5
            a_new = a - step * sg
            a_new = simplex_project(a_new)
            new_score = 2.0 * n * fast_conv(a_new).max()
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            # Slowly increase temperature (make smoother)
            temperature *= 1.002
            lr = min(lr * 1.01, 0.1)
        else:
            temperature *= 0.95  # make less smooth (closer to max)
            lr *= 0.8
            a = a  # don't update
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGrad it={it} ({elapsed:.0f}s): best={best_score:.8f} temp={temperature:.4f}")
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

def cross_corr_fft(w, a):
    """Compute cross-correlation: result[i] = sum_k w[k]*a[k-i] (valid bounds)."""
    n = len(a)
    m = len(w)
    # cross_corr[i] = conv(w, reverse(a))[i + n - 1]

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """Compute cross-correlation: result[i] = sum_k w[k]*a[k-i] (valid bounds). O(n log n)."""
    n = len(a)
    m = len(w)
    a_rev = a[::-1].copy()
    size = m + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    result_full = irfft(W * Ar, fft_size)[:size]
    return result_full[n - 1:n - 1 + n]


def simplex_project(v):
    """Project onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def solve_lp(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:

            print(f"    FinePolish sweep={sweeps} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        c = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - c) / max(c, 1.0), 0.0)
    elif init_type == 1:
        t = np.arange(n)
        c = (n - 1) / 2.0
        x = (t - c) / max(c, 1.0)
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        c = (n - 1) / 2.0
        x = (t - c) / max(c, 1.0)
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0) ** 1.5
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * 0.3
        c2 = n * 0.7
        w = n * 0.12
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    else:
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * t / 7.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resize_sequence(a, new_n):
    """Resize sequence using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0.0, float(old_n - 1), new_n)
    return np.interp(new_x, old_x, a)


def propose_candidate():
    """
    Multi-phase optimization:
    1. LP best-response alternating (main driver, ~50% time)
    2. FFT-accelerated smoothed gradient descent (~20% time)
    3. Optimal-step coordinate descent (~25% time)
    4. Fine polishing (~5% time)
    With multi-length exploration.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = get_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, lbl):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{lbl}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alt (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.50

    t_alloc = min(t1_end - time.monotonic(), 450.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternate(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Try different lengths
    for trial_n in [n * 15 // 16, n * 13 // 12]:
        if trial_n == n or time.monotonic() > t1_end - 60:
            continue
        if trial_n < 64:
            continue
        a_resized = resize_sequence(best_seq, trial_n)
        t_rem = t1_end - time.monotonic()
        if t_rem < 40:
            break
        t_a = min(t_rem * 0.5, 120.0)
        print(f"  LP n={trial_n}, budget={t_a:.0f}s")
        a_i, s_i = lp_alternate(a_resized, t_a, solver, verbose=False)
        try_update(a_i, f"lp_n{trial_n}")

    # Second LP round on best
    if time.monotonic() < t1_end - 40:
        t_a = min(t1_end - time.monotonic(), 150.0)
        if t_a > 30:
            print(f"  LP round 2, budget={t_a:.0f}s")
            a2, s2 = lp_alternate(best_seq, t_a, solver, verbose=False)
            try_update(a2, "lp_r2")

    # === Phase 2: Smoothed Gradient ===
    print(f"\n=== Phase 2: Smoothed Gradient (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_a = min(t2_end - time.monotonic(), 200.0)
    if t_a > 30:
        print(f"  SGrad budget={t_a:.0f}s")
        a_sg, s_sg = fft_smoothed_gradient(best_seq, t_a, verbose=True)
        try_update(a_sg, "sgrad")
        if time.monotonic() < t2_end - 30:
            t_a2 = min(t2_end - time.monotonic(), 80.0)
            a_sg2, s_sg2 = fft_smoothed_gradient(a_lp, t_a2, verbose=False)
            try_update(a_sg2, "sgrad_lp")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.25

    t_a = min(t3_end - time.monotonic(), 350.0)
    if t_a > 30:
        print(f"  CD budget={t_a:.0f}s")
        a_cd, s_cd = coord_descent_optimal(best_seq, t_a, verbose=True, label="main")
        try_update(a_cd, "cd_main")
        if time.monotonic() < t3_end - 60:
            t_a2 = min(t3_end - time.monotonic(), 150.0)
            a_cd2, s_cd2 = coord_descent_optimal(a_lp, t_a2, verbose=False, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 4: Fine Polish ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 10

    t_a = min(t4_end - time.monotonic(), 120.0)
    if t_a > 15:
        print(f"  FinePolish budget={t_a:.0f}s")
        a_fp, s_fp = fine_polish(best_seq, t_a, verbose=True)
        try_update(a_fp, "fine")

    # Final CD pass
    if time.monotonic() < t4_end - 5:
        t_f = min(t4_end - time.monotonic(), 50.0)
        if t_f > 10:
            a_fin, s_fin = coord_descent_optimal(best_seq, t_f, verbose=False, label="fin")
            try_update(a_fin, "final")

    # Final polish again
    if time.monotonic() < t4_end - 3:
        t_fp2 = min(t4_end - time.monotonic(), 30.0)
        if t_fp2 > 8:
            a_fp2, s_fp2 = fine_polish(best_seq, t_fp2, verbose=False)
            try_update(a_fp2, "fine2")

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
