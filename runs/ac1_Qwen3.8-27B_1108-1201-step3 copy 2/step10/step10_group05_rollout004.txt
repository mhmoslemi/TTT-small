def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:

    import cvxpy as cp

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.GLPK, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(7, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.3 * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:

            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue

        # Harmonic step size with adaptive adjustment
        alpha = 2.0 / (iters + 2.0)
        # Also try a slower schedule periodically
        if iters % 50 == 0:
            alpha = max(alpha, 0.05)

        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        a = a_new

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Also try the LP solution directly (sometimes it's already well-optimized)
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if stall_count > 30:
            # Restart with a perturbation
            a = a + 0.01 * np.random.default_rng(iters).standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

    return best_a, best_score, iters


def simulate_annealing_cd(a0, time_limit, rng, label=""):
    """Simulated annealing with coordinate descent as local search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    current_score = best_score
    temp = 0.01
    cooling = 0.97
    cd_budget = min(5.0, time_limit * 0.05)

    while time.monotonic() - t0 < time_limit:
        # Local search: run CD for a short budget
        a_local, score_local, _ = smart_coord_descent(a, cd_budget, label=f"sa_{label}")
        if score_local < best_score - 1e-14:
            best_score = score_local
            best_a = a_local.copy()
        a = a_local
        current_score = score_local

        # Perturbation
        if time.monotonic() - t0 >= time_limit:
            break

        pert_type = rng.random()
        if pert_type < 0.4:
            # Peak-targeted perturbation
            conv = fast_convolve(a)
            M = conv.max()
            peak_threshold = M * 0.85
            peak_positions = np.where(conv >= peak_threshold)[0]
            pressure = np.zeros(n)
            for k in peak_positions:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    pressure[i] += a[i] * a[j]
            p_max = pressure.max()
            if p_max > 1e-20:
                pressure = pressure / p_max
            noise = rng.standard_normal(n)
            strength = 0.02 + 0.03 * temp * 100
            a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
        elif pert_type < 0.7:
            # Random swap perturbation
            idx = rng.choice(n, size=2, replace=False)
            swap_amt = 0.1 * a[idx[0]]
            a_new = a.copy()
            a_new[idx[0]] = max(a[idx[0]] - swap_amt, 0)
            a_new[idx[1]] = a[idx[1]] + swap_amt
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
        else:
            # Gaussian perturbation
            a_new = a + 0.05 * temp * 100 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss

        # Evaluate perturbed
        pert_score = 2.0 * n * fast_convolve(a_new).max()
        delta = pert_score - current_score

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_new
            current_score = pert_score
        else:
            # Keep current
            pass

        temp *= cooling
        if temp < 1e-6:
            temp = 0.005
            # Reset temperature periodically to keep exploring

    return best_a, best_score


def block_cd(a0, time_limit, rng, label=""):
    """Block coordinate descent updating 2 adjacent coordinates at once."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(0, n - 1, 2):
            a1, a2 = a[i], a[i + 1]
            if a1 < 1e-15 and a2 < 1e-15:
                continue

            # Affected range is union of affected ranges for i and i+1
            k_start = i
            k_end = min(i + n, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1].copy()

            # Try grid of (delta1, delta2)
            best_d1, best_d2 = 0.0, 0.0
            best_new_max = M

            # Sample on a small grid
            for f1 in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5]:
                d1 = f1 * a1
                if d1 < -a1:
                    continue
                for f2 in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5]:
                    d2 = f2 * a2
                    if d2 < -a2:
                        continue
                    if abs(d1) < 1e-15 and abs(d2) < 1e-15:
                        continue

                    # Compute new affected
                    new_aff = affected.copy()
                    # Contribution from d1
                    c1_start = i - k_start
                    c1_end = min(i + n - 1, k_end) - k_start
                    c1_indices = np.arange(max(0, c1_start), c1_end + 1)
                    c1_coeffs = a[(max(0, c1_start) + k_start - i):(c1_end + 1 - i)] + a1
                    new_aff[c1_indices] += d1 * c1_coeffs
                    # Quadratic term for i
                    if 2 * i >= k_start and 2 * i <= k_end:
                        new_aff[2 * i - k_start] += d1 * d1

                    # Contribution from d2
                    c2_start = (i + 1) - k_start
                    c2_end = min(i + n, k_end) - k_start
                    c2_indices = np.arange(max(0, c2_start), c2_end + 1)
                    if len(c2_indices) > 0:
                        c2_coeffs = a[(max(0, c2_start) + k_start - i - 1):(c2_end + 1 - i - 1)] + a2
                        new_aff[c2_indices] += d2 * c2_coeffs
                    if 2 * (i + 1) >= k_start and 2 * (i + 1) <= k_end:
                        new_aff[2 * (i + 1) - k_start] += d2 * d2

                    new_max = max(unaff_max, float(new_aff.max()))
                    if new_max < best_new_max - 1e-14:
                        best_new_max = new_max
                        best_d1, best_d2 = d1, d2

            if best_d1 != 0.0 or best_d2 != 0.0:
                a[i] = max(a1 + best_d1, 0.0)
                a[i + 1] = max(a2 + best_d2, 0.0)
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score


def try_symmetry(a0, label=""):
    """Try palindromic averaging."""
    n = len(a0)
    a = a0.copy()
    a_rev = a[::-1].copy()
    a_avg = 0.5 * (a + a_rev)
    s = a_avg.sum()
    if s < 1e-15:
        return a0, float('inf')
    a_avg = a_avg / s
    score = 2.0 * n * fast_convolve(a_avg).max()
    return a_avg, score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_perturb(a0, rng, strength=0.1):
    """Perturb in frequency domain to create diverse candidates."""
    n = len(a0)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a0, fft_size)
    # Add phase noise (preserving magnitudes approximately)
    phase = np.angle(A)
    mag = np.abs(A)
    noise = rng.standard_normal(fft_size // 2 + 1) * strength * 0.01
    A_new = A + noise * mag
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient with Nesterov and near-peak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M - 0.005 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / max(len(near_peak), 1)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def nesterov_smoothed_md(a0, time_limit, beta_init=30.0, lr_init=0.008, label=""):
    """Nesterov-accelerated smoothed mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.85
            beta *= 1.5
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 50 == 0:
            beta *= 1.02

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

    if rem > 15:
        a8e, s8e = fine_coord_descent(best_seq, min(rem * 0.7, 30.0), label="p8e")
        best_seq, best_score = check_and_update(a8e, best_seq, best_score, "p8e_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
