Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated mirror descent
with adaptive active sets, targeted peak-pair reduction, multi-dimension search,
and analytic coordinate polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Evaluate 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_normalize(z):
    """Stable softmax to get a probability vector."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage strategy:
1. FFT-accelerated mirror descent with adaptive step sizes and Nesterov momentum
2. LogSumExp-smoothed L-BFGS for precise optimization
3. Novel targeted peak-pair reduction (analytic 2-coordinate optimization)
4. Multi-dimension exploration
5. Final coordinate descent polish
Key innovation: peak-pair reduction directly identifies and optimizes the
coordinate pairs contributing most to the convolution peak.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Evaluate 2*n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_vec(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_prob(a):
    """Project to probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_descent_opt(a0, time_limit, eta0=0.05, active_frac=0.999,
                       use_nesterov=True, use_bb=True, max_iters=2000000,
                       restart_thresh=300):
    """Accelerated mirror descent with BB step size and Nesterov momentum."""
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        if use_bb and it >= 3 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_ds = float(np.dot(s_vec, s_vec))
            s_dy = float(np.dot(s_vec, y_vec))
            y_dy = float(np.dot(y_vec, y_vec))
            if s_ds > 1e-18 and y_dy > 1e-18:
                if abs(s_dy) > 1e-18:
                    eta_bb = s_ds / abs(s_dy)
                else:
                    eta_bb = s_ds / y_dy
                eta = 0.6 * eta + 0.4 * eta_bb
                eta = max(1e-6, min(eta, 3.0))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))
        g_shift = g - g.min()
        log_a_new = log_a - eta * g_shift
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        if use_nesterov:
            alpha = 2.0 / (it + 2.0)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        score = eval_score(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        cand = min(score, score_y)
        if cand < best_score - 1e-14:
            best_score = cand
            if score <= score_y:
                best_a = a.copy()
            else:
                best_a = normalize_prob(y)
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= restart_thresh:
            scale = 0.002 + 0.004 * min(no_improve / restart_thresh, 2.0)
            noise = rng.standard_normal(n) * scale
            a = a + noise
            a = normalize_prob(a)
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.4)

    return best_a, best_score


def logsumexp_smooth_obj(z, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-parameterization)."""
    a = softmax_vec(z)
    n = len(a)
    b = fast_convolve(a)
    shifted = b - b.max()
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(b)) / len(b)
    else:
        weights = weights / wsum

    logsumexp_val = b.max() + np.log(wsum)
    obj = (2.0 * n / T) * logsumexp_val

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if weights[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * weights[k] * a[k - idx]

    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)

    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS on LogSumExp-smoothed objective with annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_smooth_obj(z_inner, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_smooth_obj(z_inner, T)
            return grad

        try:
            iters = min(maxiter_per_stage, max(20, int((time_limit - (time.monotonic() - t0)) / (len(T_stages) - T_stages.index(T)) / 0.05)))
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_vec(z)


def peak_pair_reduction(a0, time_limit, n_rounds=500000):
    """
    Novel: Targeted peak-pair reduction.
    Identifies coordinate pairs (i, j) with i+j=k_peak that have the largest
    product a[i]*a[j], and reduces them by redistributing mass to "safe"
    coordinates (those that contribute minimally to all peaks).
    """
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find all peaks within 0.1% of max
        threshold = max_b * 0.999
        peaks = np.where(b >= threshold)[0]

        # For each peak, find top contributing pairs
        improved = False
        for pk in peaks[:5]:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) < 2:
                continue
            products = a[idx] * a[pk - idx]
            top_m = min(4, len(idx))
            top_pos = idx[np.argsort(products)[::-1][:top_m]]

            for pos in top_pos:
                partner = pk - pos
                if partner < 0 or partner >= n or partner == pos:
                    continue
                if a[pos] < 1e-14 or a[partner] < 1e-14:
                    continue

                # Try reducing both pos and partner
                for frac in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    delta = a_trial[pos] * (1 - frac) + a_trial[partner] * (1 - frac)
                    a_trial[pos] *= frac
                    a_trial[partner] *= frac

                    # Redistribute delta to safe coordinates
                    # Safe = coordinates where a[k-i] is small for all active peaks
                    safe_scores = np.zeros(n)
                    for pk2 in peaks[:3]:
                        js2 = max(0, pk2 - n + 1)
                        je2 = min(n - 1, pk2)
                        for i in range(js2, je2 + 1):
                            safe_scores[i] += a[pk2 - i]
                    safe_scores[pos] = 1e10
                    safe_scores[partner] = 1e10

                    n_safe = min(10, n - 2)
                    safe_idx = np.argsort(safe_scores)[:n_safe]
                    safe_sum = a_trial[safe_idx].sum()
                    if safe_sum < 1e-15:
                        # Distribute uniformly to remaining
                        mask = np.ones(n, dtype=bool)
                        mask[pos] = False
                        mask[partner] = False
                        safe_idx = np.where(mask)[0]
                        safe_sum = a_trial[safe_idx].sum()
                        if safe_sum < 1e-15:
                            break

                    a_trial[safe_idx] += delta * a_trial[safe_idx] / safe_sum
                    a_trial = normalize_prob(a_trial)

                    sc = eval_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 50:
                # Try a batch reduction
                a_trial = a.copy()
                for pk in peaks[:3]:
                    js = max(0, pk - n + 1)
                    je = min(n - 1, pk)
                    idx = np.arange(js, je + 1)
                    products = a_trial[idx] * a_trial[pk - idx]
                    top_pos = idx[np.argsort(products)[::-1][:2]]
                    for pos in top_pos:
                        if a_trial[pos] > 1e-12:
                            a_trial[pos] *= 0.999
                a_trial = normalize_prob(a_trial)
                sc = eval_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_a = a.copy()
                    best_score = sc
                    stall = 0
                elif stall > 500:
                    # Perturb and reset stall
                    a = a + rng.standard_normal(n) * 0.005
                    a = normalize_prob(a)
                    stall = 0
            if stall > 3000:
                break
        else:
            stall = 0

        if (rnd + 1) % 50000 == 0:
            print(f"    PeakPair {time.monotonic()-t0:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, n_steps=200000):
    """Fine-grained coordinate descent for final polish."""
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_k - idx]
        top_m = min(8, len(idx))
        top_pos = idx[np.argsort(products)[::-1][:top_m]]

        improved = False
        for pos in top_pos:
            if a[pos] < 1e-14:
                continue
            for factor in [0.9999, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.99]:
                a_trial = a.copy()
                diff = a_trial[pos] - a_trial[pos] * factor
                a_trial[pos] *= factor

                others = a_trial[mask]
                os_ = others.sum()
                if os_ < 1e-15:
                    break
                a_trial[mask] = others + diff * others / os_
                a_trial = a_trial / a_trial.sum()

                if sc < best_score - 1e-16:

def lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. (a*b)[k] <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = cp.ECOS
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None


def nesterov_lp_opt(a0, time_limit, solver=None, max_iters=20000):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize_prob(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize_prob(a)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = normalize_prob(y)

        if (it + 1) % 50 == 0:
            if stall > 15:
                a = a + rng.standard_normal(n) * 0.01 / np.sqrt(stall / 10)
                a = normalize_prob(a)
                y = a.copy()
                stall = 0
            stall = 1 if best_score > 2.0 * n * fast_convolve(a).max() - 1e-12 else 0

    return best_a, best_score


def interpolate_to_n(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum() if f.sum() > 1e-15 else np.ones(old_n) / old_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_opt(best_seq, t_alloc, solver=cp.ECOS, max_iters=15000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 15:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=3000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Peak-Pair Reduction (novel) ===
    print(f"\n=== Phase 4: Peak-Pair Reduction (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 320.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pp, _ = peak_pair_reduction(best_seq, t_alloc, n_rounds=400000)
        try_update(a_pp, "peak_pair")

    # === Phase 5: Try different dimension ===
    print(f"\n=== Phase 5: Multi-dimension (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        for n_try in [1500, 1000]:
            if time.monotonic() > deadline - 30:
                break
            rem2 = deadline - time.monotonic()
            t_sub = min(rem2 * 0.5, 100.0)
            if t_sub < 15:
                break
            a_interp = interpolate_to_n(best_seq, n_try)
            print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
            a_md2, _ = mirror_descent_opt(a_interp, t_sub * 0.6, eta0=0.04,
                                          active_frac=0.999, use_nesterov=True,
                                          use_bb=True, restart_thresh=200)
            if time.monotonic() < deadline - 10:
                a_pol = coordinate_polish(a_md2, min(t_sub * 0.4, 30.0), n_steps=50000)
                a_md2 = a_pol
            try_update(a_md2, f"dim_{n_try}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_pol, _ = coordinate_polish(best_seq, t_alloc, n_steps=100000)
        try_update(a_pol, "final_polish")

    # === Phase 7: One more mirror descent pass with lower eta ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 50.0)
    if t_alloc > 10:
        a_final, _ = mirror_descent_opt(best_seq, t_alloc, eta0=0.01,
                                        active_frac=0.9999, use_nesterov=True,
                                        use_bb=True, restart_thresh=100)
        try_update(a_final, "final_md")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
