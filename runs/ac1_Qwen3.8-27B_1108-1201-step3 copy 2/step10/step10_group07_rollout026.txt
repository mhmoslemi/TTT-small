def build_conv_matrix_row(a, k):
    """Build row k of the convolution matrix T where T @ b = conv(a, b)."""
    n = len(a)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    for j in range(j_start, j_end + 1):
        row[j] = a[k - j]
    return row


def build_conv_matrix_rows(a, indices):
    """Build selected rows of the convolution matrix."""
    n = len(a)
    m = len(indices)
    result = np.zeros((m, n))
    for idx, k in enumerate(indices):

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

    return result


def lp_best_response(a, solver, K=50):
    """
    Solve LP: min M s.t. conv(a,b)_k <= M for k in top-K, sum(b)=1, b>=0.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    
    if K < 2 * n - 1:
        peak_indices = np.argpartition(conv_a, -K)[-K:]
    else:
        peak_indices = np.arange(2 * n - 1)
    
    T = build_conv_matrix_rows(a, peak_indices)
    m = len(peak_indices)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(m)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating_improved(a0, time_limit, solver, K_init=30, K_max=100, verbose=True):

def pairwise_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Pairwise coordinate descent: update pairs (i,j) that share a peak.
    Uses exact 1D line search on the parameter t: a[i]+t, a[j]-t (or similar).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Find top peaks
        K_peaks = 10
        peak_idx = np.argpartition(conv, -K_peaks)[-K_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        for k_star in peak_idx:
            k_star = int(k_star)
            if conv[k_star] < M * 0.9:
                continue

            # Find contributing pairs
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Get the pairs sorted by contribution
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

            # Try the top few pairs
            for contrib, i, j in contribs[:5]:
                if contrib < 1e-12:
                    continue
                if i == j:
                    continue

                # Try redistributing mass between i and j
                # Parameterize: a[i] -> a[i] + t, a[j] -> a[j] - t
                # (or both decrease, or one increases and other decreases)
                # We want to keep sum approximately constant
                for t_frac in [0.05, 0.1, 0.15, 0.2, 0.3, -0.05, -0.1, -0.15, -0.2, -0.3]:
                    t = t_frac * min(a[i], a[j])
                    if a[i] + t < -1e-15 or a[j] - t < -1e-15:
                        continue

                    # Compute new conv incrementally
                    # Changing a[i] by t affects conv[k] for k in [i, i+n-1]
                    # Changing a[j] by -t affects conv[k] for k in [j, j+n-1]
                    new_conv = conv.copy()
                    
                    # Effect of a[i] -> a[i] + t
                    ki_s = i
                    ki_e = min(i + n - 1, 2 * n - 2)
                    new_conv[ki_s:ki_e+1] += 2.0 * t * a[ki_s-i:ki_e-i+1]
                    # Quadratic correction at k = 2i
                    if 2*i >= ki_s and 2*i <= ki_e:
                        new_conv[2*i] += t * t

                    # Effect of a[j] -> a[j] - t
                    kj_s = j
                    kj_e = min(j + n - 1, 2 * n - 2)
                    new_conv[kj_s:kj_e+1] += 2.0 * (-t) * a[kj_s-j:kj_e-j+1]
                    # Quadratic correction at k = 2j
                    if 2*j >= kj_s and 2*j <= kj_e:
                        new_conv[2*j] += t * t  # (-t)^2 = t^2

                    new_M = float(new_conv.max())
                    if new_M < M - 1e-13:
                        a[i] = max(a[i] + t, 0.0)
                        a[j] = max(a[j] - t, 0.0)
                        conv = new_conv
                        M = new_M
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        improved = True
                        break  # Move to next peak

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, verbose=True, top_k=30):
    """Nesterov momentum subgradient descent with simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    beta = 0.9  # momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.9
            if lr < 1e-10:
                break

        # Compute subgradient from top peaks
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end+1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Nesterov step
        v = y - a
        y = a + beta * v - lr * sg
        old_a = a
        a = simplex_project(y)

        conv = fast_convolve(a)

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    Nesterov it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} K={K} alpha={alpha:.2f}")
    return best_a, best_score


def pairwise_coord_descent(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        K_peaks = 8
        peak_idx = np.argpartition(conv, -K_peaks)[-K_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        for k_star in peak_idx:
            k_star = int(k_star)
            if conv[k_star] < M * 0.85:
                continue
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                if i != j:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            found_update = False
            for contrib, i, j in contribs[:6]:
                if contrib < 1e-13 or found_update:
                    continue
                min_val = min(a[i], a[j])
                if min_val < 1e-14:
                    continue
                for t_frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, -0.05, -0.1, -0.2, -0.3]:
                    t = t_frac * min_val
                    if a[i] + t < -1e-15:
                        continue
                    if a[j] - t < -1e-15:
                        continue
                    new_conv = conv.copy()
                    ki_s = i
                    ki_e = min(i + n - 1, 2 * n - 2)
                    new_conv[ki_s:ki_e + 1] += 2.0 * t * a[ki_s - i:ki_e - i + 1]
                    if 2 * i >= ki_s and 2 * i <= ki_e:
                        new_conv[2 * i] += t * t
                    kj_s = j
                    kj_e = min(j + n - 1, 2 * n - 2)
                    new_conv[kj_s:kj_e + 1] += 2.0 * (-t) * a[kj_s - j:kj_e - j + 1]
                    if 2 * j >= kj_s and 2 * j <= kj_e:
                        new_conv[2 * j] += t * t
                    new_M = float(new_conv.max())
                    if new_M < M - 1e-13:
                        a[i] = max(a[i] + t, 0.0)
                        a[j] = max(a[j] - t, 0.0)
                        conv = new_conv
                        M = new_M
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        improved = True
                        found_update = True
                        break
        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def nesterov_subgradient(a0, time_limit, verbose=True, top_k=30):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.015
    beta = 0.9
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            lr *= 0.88
            if lr < 1e-10:
                break
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.004 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        v = beta * v - lr * sg
        y = a + v
        a = simplex_project(y)
        conv = fast_convolve(a)
        if it % 80 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    return best_a, best_score


def single_coord_descent(a0, time_limit, verbose=True, fine=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    if fine:
        factors = [0.997, 0.994, 0.99, 0.98, 1.01, 1.02, 1.04, 1.08, 0.5, 1.5]
    else:
        factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if a[i] < 1e-16:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            best_delta = 0.0
            best_new_M = M
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_local = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_local] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_local = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_local] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > (10 if fine else 20):
                break
        else:
            no_improve = 0
        if verbose and sweeps % 8 == 0:
            elapsed = time.monotonic() - t0
            label = "Fine" if fine else "Coarse"
            print(f"    {label}CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def spectral_init(n, rng):
    """Initialize using flat-spectrum (Gerchberg-Saxton style) approach."""
    for attempt in range(50):
        phases = rng.uniform(0, 2 * np.pi, size=n)
        A = np.ones(n) * np.exp(1j * phases)
        a = np.real(irfft(A, n) if False else np.fft.irfft(np.fft.rfft(np.exp(1j * phases)), n))
        # Actually let's do it more carefully
        # We want a real non-negative sequence
        # Start with random phases, IFFT, take real part, project to non-negative
        freq_k = n // 2 + 1
        complex_spec = np.exp(1j * rng.uniform(0, 2 * np.pi, size=freq_k))
        a_raw = np.fft.irfft(complex_spec, n)
        a_pos = np.maximum(a_raw, 0.0)
        s = a_pos.sum()
        if s > 1e-10:
            a_pos = a_pos / s
            sc = score_of(a_pos)
            if sc < 2.2:
                return a_pos
    # Fallback: smooth random
    t = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n * rng.integers(1, 5)) + 0.2 * np.cos(4 * np.pi * t / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(2, 7))
        t = np.arange(n)
        a = 0.6 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.7 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Zigzag / sawtooth
        t = np.arange(n)
        a = np.abs(np.sin(np.pi * t / (rng.integers(2, 6))))
        a = 0.3 + 0.7 * a
    else:
        a = rng.uniform(0.1, 1.0, size=n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def hill_climb_diff(a0, time_limit, verbose=False):
    """
    Differential-evolution style local search.
    Maintain a small population and mutate.
    """
    n = len(a0)
    pop = [a0.copy() / a0.sum()]
    pop[0] = np.maximum(pop[0], 0)
    pop[0] = pop[0] / pop[0].sum()
    best_score = float(score_of(pop[0]))
    best_a = pop[0].copy()
    t0 = time.monotonic()
    gen = 0

    # Initialize population
    for _ in range(14):
        a_new = pop[0] + np.random.standard_normal(n) * (0.05 / np.sqrt(n))
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            pop.append(a_new)
            sc = score_of(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    pop_scores = [score_of(p) for p in pop]

    while time.monotonic() - t0 < time_limit:
        gen += 1
        for i in range(len(pop)):
            if len(pop) < 3:
                break
            j, k = [x for x in range(len(pop)) if x != i]
            if len(pop) > 2:
                j, k = np.random.choice([x for x in range(len(pop)) if x != i], size=2, replace=False)
            else:
                j = (i + 1) % len(pop)
                k = (i + 2) % len(pop)
            F = np.random.uniform(0.5, 1.2)
            trial = pop[i] + F * (pop[j] - pop[k])
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            sc = score_of(trial)
            if sc < pop_scores[i]:
                pop[i] = trial
                pop_scores[i] = sc
            if sc < best_score:
                best_score = sc
                best_a = trial.copy()
                if verbose and gen % 200 == 0:
                    print(f"    DE gen={gen}: best={best_score:.8f}")
    return best_a, best_score


def propose_candidate():
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    solver = find_lp_solver()
    print(f"LP solver: {solver}")
    print(f"Initial best: {best_score:.8f} (n={n})")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating on parent ===
    print(f"\n=== Phase 1: LP Alternating ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.40
    t_alloc = min(t1_end - time.monotonic(), 500.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, K_init=20, K_max=100, verbose=True)
    try_update(a_lp, "lp_parent")

    # LP from diverse inits
    inits_to_try = [
        diverse_init(n, rng, 1),  # U-shape
        diverse_init(n, rng, 5),  # W-shape
        diverse_init(n, rng, 6),  # Zigzag
        spectral_init(n, rng),    # Spectral
    ]
    for idx, init in enumerate(inits_to_try):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(remaining * 0.35, 100.0)
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, K_init=20, K_max=80, verbose=False)
        try_update(a_tmp, f"lp_init{idx}")

    # Second LP round from best
    if time.monotonic() < t1_end - 40:
        t_alloc = min(t1_end - time.monotonic(), 150.0)
        a_lp2, s_lp2 = lp_alternating(best_seq, t_alloc, solver, K_init=40, K_max=150, verbose=True)
        try_update(a_lp2, "lp_r2")

    # === Phase 2: Pairwise Coordinate Descent ===
    print(f"\n=== Phase 2: Pairwise CD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.18
    t_alloc = min(t2_end - time.monotonic(), 200.0)
    if t_alloc > 30:
        a_pc, s_pc = pairwise_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_pc, "pair_cd")
        if time.monotonic() < t2_end - 30:
            a_pc2, s_pc2 = pairwise_coord_descent(a_lp, min(t2_end - time.monotonic(), 80.0), verbose=False, label="lp")
            try_update(a_pc2, "pair_cd_lp")

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov SGD (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.12
    t_alloc = min(t3_end - time.monotonic(), 120.0)
    if t_alloc > 20:
        a_ng, s_ng = nesterov_subgradient(best_seq, t_alloc, verbose=True, top_k=40)
        try_update(a_ng, "nesterov")
        if time.monotonic() < t3_end - 20:
            a_ng2, s_ng2 = nesterov_subgradient(a_lp, min(t3_end - time.monotonic(), 60.0), verbose=False, top_k=50)
            try_update(a_ng2, "nesterov_lp")

    # === Phase 4: Differential Evolution (diversification) ===
    print(f"\n=== Phase 4: Diff-Evo (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.12
    t_alloc = min(t4_end - time.monotonic(), 120.0)
    if t_alloc > 20:
        a_de, s_de = hill_climb_diff(best_seq, t_alloc, verbose=True)
        try_update(a_de, "diff_evo")

    # === Phase 5: Fine Polishing ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 30

    # Coarse CD first
    if time.monotonic() < t5_end - 60:
        t_alloc = min(t5_end - time.monotonic() - 60, 80.0)
        if t_alloc > 20:
            a_cc, s_cc = single_coord_descent(best_seq, t_alloc, verbose=True, fine=False)
            try_update(a_cc, "coarse_cd")

    # Fine CD
    if time.monotonic() < t5_end - 20:
        t_alloc = min(t5_end - time.monotonic() - 20, 100.0)
        if t_alloc > 15:
            a_fc, s_fc = single_coord_descent(best_seq, t_alloc, verbose=True, fine=True)
            try_update(a_fc, "fine_cd")

    # Final micro-pair CD
    if time.monotonic() < t5_end - 10:
        t_alloc = min(t5_end - time.monotonic() - 10, 40.0)
        if t_alloc > 10:
            a_mp, s_mp = pairwise_coord_descent(best_seq, t_alloc, verbose=True, label="final")
            try_update(a_mp, "final_pair")

    # Final LP polish
    if time.monotonic() < t5_end - 5:
        t_alloc = min(t5_end - time.monotonic() - 5, 20.0)
        if t_alloc > 5:
            a_fl, s_fl = lp_alternating(best_seq, t_alloc, solver, K_init=60, K_max=150, verbose=False)
            try_update(a_fl, "final_lp")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
