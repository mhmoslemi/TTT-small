from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def score_of(a):
    """Compute the objective value (normalized to sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    M = fast_convolve(a_norm).max()
    return 2.0 * n * M

            if k_end < 2 * n - 2:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast ternary-search coordinate descent
2. Simulated annealing with adaptive block moves (key innovation for non-convex problem)
3. LP-based fictitious play for game-theoretic refinement
4. Multi-length and multi-initialization exploration
5. Nesterov-accelerated subgradient for final polishing
The key insight is the problem is non-convex (max of indefinite quadratic forms),
so simulated annealing is essential to escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def block_project(v, S):
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - S) / (rho + 1.0)
        result = np.maximum(v - theta, 0.0)
    else:
        result = np.full(n, S / n)
    return result

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    M = float(fast_convolve(a_norm).max())
    return 2.0 * n * M

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def coord_descent(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            affected = conv[k_start:k_end + 1]
            idx = np.arange(k_start - i, k_end - i + 1)
            coeffs = a[idx] + ai
            has_quad = (k_start <= 2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            d_lo = -ai
            d_hi = 2.0 * ai
            for _ in range(22):
                d1 = d_lo + (d_hi - d_lo) / 3.0
                d2 = d_hi - (d_hi - d_lo) / 3.0
                na1 = affected + d1 * coeffs
                if has_quad:
                    na1 = na1.copy()
                    na1[quad_idx] += d1 * d1
                g1 = max(unaff_max, float(na1.max()))
                na2 = affected + d2 * coeffs
                if has_quad:
                    na2 = na2.copy()
                    na2[quad_idx] += d2 * d2
                g2 = max(unaff_max, float(na2.max()))
                if g1 < g2:
                    d_hi = d2
                else:
                    d_lo = d1
            d_opt = (d_lo + d_hi) / 2.0
            if d_opt < -ai:
                d_opt = -ai
            na = affected + d_opt * coeffs
            if has_quad:
                na = na.copy()
                na[quad_idx] += d_opt * d_opt
            new_M = max(unaff_max, float(na.max()))
            if new_M < M - 1e-13:
                a[i] = max(ai + d_opt, 0.0)
                conv[k_start:k_end + 1] = na
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            break
    return best_a, best_score
