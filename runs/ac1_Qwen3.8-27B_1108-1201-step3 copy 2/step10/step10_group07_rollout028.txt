"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization with adaptive damping and tight tolerances
2. Exact convex 1D coordinate descent exploiting the structure of the subproblem
3. KKT-condition-based refinement (pressure equalization)
4. Multi-peak subgradient descent with adaptive temperature
5. Diverse initializations and strategic restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def _find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(_find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    _find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def _build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt, dual_values).
    """
    n = len(a)
    T = _build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, rel_gap=1e-10, abs_gap=1e-12, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-10, reltol=1e-10, verbose=False)
        except Exception:
            return a.copy(), float('inf'), None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf'), None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf'), None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val, None


def _lp_alternating(a0, time_limit, solver):
    """
    LP-based best-response alternating minimization with adaptive damping.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp, _ = _lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping
        if it <= 30:
            alpha = 1.0
        elif it <= 80:
            alpha = 0.8
        elif it <= 150:
            alpha = 0.6
        else:
            alpha = max(0.35, 15.0 / (it + 20.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum

        conv = fast_convolve(a)
        score_a = 2.0 * n * conv.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection
        if score_a > prev_score * (1 + 1e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        prev_score = score_a

        # Periodic kicks
        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            noise = np.random.standard_normal(n) * noise_scale
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert_sum = a_pert.sum()
            if a_pert_sum > 1e-10:
                a_pert = a_pert / a_pert_sum
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.005:
                    a = a_pert
            stagnation = 0

    return best_a, best_score


def _exact_coord_descent(a0, time_limit):
    """
    Exact coordinate descent exploiting convexity of 1D subproblem.
    For each coordinate, tries multiple candidate deltas including
    analytically motivated ones, then refines.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        any_improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            U = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            has_quad = (2 * i >= k_start and 2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            a_i = a[i]

            # Candidate deltas: include analytically motivated values
            # Key insight: quadratic c[2i]+2*delta*a_i+delta^2 minimized at delta=-a_i
            candidates = [
                -a_i, -0.7 * a_i, -0.5 * a_i, -0.3 * a_i, -0.15 * a_i,
                -0.05 * a_i, 0.05 * a_i, 0.15 * a_i, 0.3 * a_i, 0.5 * a_i,
                0.7 * a_i, a_i, 1.5 * a_i
            ]

            best_delta = 0.0
            best_new_M = M

            for delta in candidates:
                new_val = a_i + delta
                if new_val < -1e-15:
                    continue
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] = conv[2 * i] + 2.0 * delta * a_i + delta * delta
                new_M = max(U, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            # Refine around best_delta
            if best_delta != 0.0:
                step = abs(best_delta) * 0.15
                for frac in np.arange(-1.0, 1.01, 0.25):
                    delta = best_delta + frac * step
                    new_val = a_i + delta
                    if new_val < -1e-15:
                        continue
                    new_affected = conv_affected + 2.0 * delta * a_shifted
                    if has_quad:
                        new_affected = new_affected.copy()
                        new_affected[quad_idx] = conv[2 * i] + 2.0 * delta * a_i + delta * delta
                    new_M = max(U, float(new_affected.max()))
                    if new_M < best_new_M - 1e-15:
                        best_new_M = new_M
                        best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a_i + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = 2.0 * best_delta * (a_i - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(U, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                any_improved = True

        if not any_improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _kkt_refinement(a0, time_limit):
    """
    Refine using KKT conditions: at optimum, the 'pressure' from active peaks
    should be constant across all non-zero coordinates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())

        # Identify active peaks (within 1e-6 of max)
        active = np.where(conv >= M * (1 - 1e-6))[0]
        active = active[(active >= 0) & (active < 2 * n - 1)]
        K = len(active)
        if K == 0:
            break

        # Compute pressure: for each i, pressure[i] = sum_{k in active} a[k-i] / K
        # where a[k-i] is valid (0 <= k-i < n)
        pressure = np.zeros(n)
        for k in active:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            pressure[i_start:i_end + 1] += a[k - i_end:k - i_start + 1][::-1]
        pressure /= K

        # For non-zero a[i], we want pressure[i] to be constant
        # Update: a_new[i] = a[i] * (pressure[i] / mean_pressure)^beta
        nonzero = a > 1e-10
        if nonzero.sum() < 2:
            break
        mean_p = pressure[nonzero].mean()
        if mean_p < 1e-15:
            break

        # damped update
        beta = 0.3
        ratio = np.ones(n)
        ratio[nonzero] = (pressure[nonzero] / mean_p) ** beta
        a_new = a * ratio
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = (0.7) * a + (0.3) * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

    return best_a, best_score


def _subgradient_descent(a0, time_limit, lr_init=0.015, top_k=40):
    """
    Subgradient descent with multi-peak weighting and adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    temp = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.88
            temp *= 0.95
            no_improve = 0
            if lr < 1e-11:
                break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

    return best_a, best_score


def _diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.15, 0.3)
        c2 = n * rng.uniform(0.7, 0.85)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(4, 16))
        t = np.arange(n)
        a = 0.5 + 0.45 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # Spectral init: random low-pass filter in frequency domain
        N = 1 << (2 * n - 2).bit_length()
        freq = np.zeros(N // 2 + 1)
        bandwidth = int(N * 0.1)
        idx = rng.integers(0, bandwidth, size=min(bandwidth, 50))
        freq[idx] = 1.0
        freq[0] = 1.0
        a = irfft(freq * N, N)[:n]
        a = np.abs(a)
    else:
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n * rng.uniform(2, 8))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a


def _fine_polish(a0, time_limit):

    n = len(a0)

    else:

    conv = fast_convolve(a)

def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = len(v)
