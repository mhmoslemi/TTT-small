    n = len(a)
    size = 2 * n - 1
    
    if active_set is None:
        active_set = set([n-1])  # Start with center
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    for _ in range(max_lazy_iters):
        K = sorted(active_set)
        # Build constraint matrix for active set
        T_active = np.zeros((len(K), n))
        for i, k in enumerate(K):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    T_active[i, j] = a[idx]
        
        constraints = [T_active @ b_var - M_var * np.ones(len(K)) <= 0,
                       cp.sum(b_var) == 1]

        prob.solve(solver=solver, verbose=False)

def peak_shave(a0, budget, n_steps=200000):

    n = len(a0)

    s = a.sum()
    if s < 1e-15:

        if time.monotonic() - t0 > budget:
            break

        c = fast_convolve(a)
        max_c = c.max()

        # Get top peaks
        top_k = min(6, len(c))
        peak_positions = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_positions[:3]:
            # Find contributing indices
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_contrib_idx = idx[np.argsort(contribs)[::-1][:4]]

            for pos in top_contrib_idx:
                if a[pos] < 1e-6:
                    continue
                for factor in [0.995, 0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-14:
                        break
                    # Redistribute to low-contribution positions
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    # Add to positions far from peak
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc

                if improved:
                    break

def subgradient_refine(a0, budget, n_steps=100000):
    """Subgradient method with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > budget:
            break

        c = fast_convolve(a)
        max_c = c.max()
        # Active set: peaks within 1% of max
        threshold = max_c * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Average subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_lbgfs(a0, budget):
    """L-BFGS on LogSumExp-smoothed objective with cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]

    for T in T_stages:
        if time.monotonic() - t0 > budget:
            break
        remaining = budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        maxiter = max(10, int(remaining * 2))

        def func_loc(z_loc):
            if time.monotonic() - t0 > budget:
                return 1e15
            a_loc = np.exp(z_loc - z_loc.max())
            a_loc = a_loc / a_loc.sum()
            b_loc = fast_convolve(a_loc)
            max_b = b_loc.max()
            exp_b = np.exp(b_loc - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T) * lse

        def jac_loc(z_loc):
            if time.monotonic() - t0 > budget:
                return np.zeros_like(z_loc)
            a_loc = np.exp(z_loc - z_loc.max())
            a_loc = a_loc / a_loc.sum()
            b_loc = fast_convolve(a_loc)
            max_b = b_loc.max()
            exp_b = np.exp(b_loc - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_loc)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_loc[k - idx]
            grad_a *= 2.0 / T
            S = np.dot(grad_a, a_loc)
            return a_loc * (grad_a - S)

        try:
            res = minimize(fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': maxiter * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: Nesterov LP (lazy) -> Peak Shave -> Subgradient ->
    Multi-length exploration -> LogSumExp L-BFGS polish.
    """
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Main Nesterov LP ===
    print(f"\n=== Phase 1: Nesterov LP n={len(best_seq)}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.50, 900.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t1, solver, max_iters=500000, verbose=True)
        try_update(a_lp, "nlp_main")

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shave, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 250.0)
    if t2 > 15:
        a_ps, _ = peak_shave(best_seq, t2, n_steps=300000)
        try_update(a_ps, "peak_shave")

    # === Phase 3: Subgradient Refinement ===
    print(f"\n=== Phase 3: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 10:
        a_sg, _ = subgradient_refine(best_seq, t3, n_steps=80000)
        try_update(a_sg, "subgrad")

    # === Phase 4: Second LP pass ===
    print(f"\n=== Phase 4: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 250.0)
    if t4 > 40:
        a_lp2, _ = nesterov_lp(best_seq, t4, solver, max_iters=200000, verbose=True)
        try_update(a_lp2, "nlp_2")

    # === Phase 5: Peak Shave Again ===
    print(f"\n=== Phase 5: Peak Shave 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 120.0)
    if t5 > 10:
        a_ps2, _ = peak_shave(best_seq, t5, n_steps=200000)
        try_update(a_ps2, "peak_shave_2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 0.8, 1.5]:
        alt_n = int(n_cur * factor)
        if alt_n != n_cur and alt_n >= 400 and alt_n <= 4000:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        t_alt = min(rem * 0.25, 180.0)
        if t_alt < 30:
            break
        print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt, _ = nesterov_lp(init_alt, t_alt * 0.6, solver, max_iters=100000, verbose=False)
        try_update(a_alt, f"lp_n{alt_n}")

    # === Phase 7: LogSumExp L-BFGS Polish ===
    print(f"\n=== Phase 7: L-BFGS Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 150.0)
    if t7 > 10:
        a_lb = logsumexp_lbgfs(best_seq, t7)
        try_update(a_lb, "lbgfs_polish")

    # === Phase 8: Final Peak Shave ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.4, 80.0)
    if t8 > 8:
        a_fin, _ = peak_shave(best_seq, t8, n_steps=100000)
        try_update(a_fin, "final_shave")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg2, _ = subgradient_refine(best_seq, min(rem * 0.6, 20.0), n_steps=30000)
        try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
