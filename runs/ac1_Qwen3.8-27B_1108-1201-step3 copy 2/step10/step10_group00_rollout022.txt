def best_response_mw(a, n_inner, eta_init=0.5, eta_decay=0.999):
    """Solve the best response LP using multiplicative weights on the dual.
    Given a (unit sum, non-negative), find b (unit sum, non-negative) that
    minimizes max_k (a*b)[k].
    
    Uses the dual: max nu s.t. (T^T lambda)_j >= nu, sum lambda = 1, lambda >= 0
    where T[k,j] = a[k-j] (valid indices).
    
    The multiplicative weights update for the dual:
    lambda[k] *= exp(eta * T[k, j*]) where j* = argmin_j (T^T lambda)_j
    """
    n = len(a)
    # Initialize lambda uniformly over 2n-1 positions
    lambda_vec = np.ones(2*n-1) / (2*n-1)
    
    # Track average j* for the primal solution
    # After T iterations, b[j] = frequency of j being chosen as j*
    j_counts = np.zeros(n)
    
    eta = eta_init
    for it in range(n_inner):
        # Compute (T^T lambda)_j = sum_k lambda[k] * a[k-j] for j=0..n-1
        # = cross-correlation of lambda and a
        # (T^T lambda)_j = sum_{k=j}^{j+n-1} lambda[k] * a[k-j]
        # = sum_{m=0}^{n-1} lambda[j+m] * a[m]
        # This is the cross-correlation at lag j
        # = ifft(fft(lambda) * conj(fft(a)))[j] (with padding)
        
        # Use FFT for speed
        fft_size = 1 << (3*n - 2 - 1).bit_length()
        L = rfft(lambda_vec, fft_size)
        A = rfft(a, fft_size)
        g = irfft(A * np.conj(L), fft_size)[:n]  # g[j] = sum_m a[m] * lambda[j+m]
        
        # j* = argmin g
        j_star = np.argmin(g)
        j_counts[j_star] += 1
        
        # Multiplicative weights update for lambda
        # lambda[k] *= exp(eta * T[k, j_star])
        # T[k, j_star] = a[k - j_star] if 0 <= k-j_star <= n-1, else 0
        # So T[:, j_star] = [0, ..., 0, a[0], a[1], ..., a[n-1], 0, ..., 0]
        # with a[0] at position j_star
        
        # We can compute the update efficiently:
        # log_update[k] = eta * (a[k - j_star] if 0 <= k-j_star <= n-1 else 0)
        log_update = np.zeros(2*n-1)
        log_update[j_star:j_star+n] = a  # T[k, j_star] = a[k-j_star] for k in [j_star, j_star+n-1]
        
        lambda_vec *= np.exp(eta * log_update)
        s = lambda_vec.sum()
        if s > 0 and np.isfinite(s):
            lambda_vec /= s
        else:
            lambda_vec = np.ones(2*n-1) / (2*n-1)
        
        eta *= eta_decay
    
    # Primal solution: b[j] = fraction of time j was chosen
    b = j_counts / j_counts.sum()
    return b

def lse_gradient_step(a, tau, lr):
    """One step of gradient descent on the LogSumExp-smoothed objective.
    Minimize 2n * LSE(tau * (a*a)) / (sum a)^2
    where LSE(x) = log(sum exp(x))
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    
    b = fast_convolve(a)
    max_b = b.max()
    
    # Compute p = softmax(tau * b)
    exp_b = np.exp(tau * (b - max_b))
    p = exp_b / exp_b.sum()
    
    L = max_b + (1.0/tau) * np.log(exp_b.sum())
    # Actually, LSE(tau * b) = max(tau*b) + log(sum exp(tau*b - max(tau*b)))
    # = tau * max_b + log(sum exp(tau*(b - max_b)))
    L = tau * max_b + np.log(exp_b.sum())
    
    # Gradient of LSE w.r.t. a_i = tau * sum_k p_k * a_{k-i}
    # = tau * (p ⋆ a)_i
    grad_L = tau * fast_xcorr(p, a)
    
    # Full gradient of f = 2n * L / (tau * s^2)
    # df/da_i = 2n/(tau*s^2) * grad_L_i - 2n*L/(tau*s^3)
    grad = 2.0 * n / (tau * s * s) * grad_L - 2.0 * n * L / (tau * s**3)
    
    # Gradient step with projection onto non-negative (and renormalize)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new / s_new
    
    return a_new

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
