Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using diverse initialization with
spectrum-flattening, multi-scale mirror descent with adaptive step sizes,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def spectrum_flatten_init(n, iterations=500, rng=None, flatten_rate=0.3):
    """Initialize using spectrum flattening (Gerchberg-Saxton style)."""
    if rng is None:
        rng = np.random.default_rng()
    a = rng.random(n) + 0.05
    a = a / a.sum()
    
    N = 1 << (2 * n - 1).bit_length()
    for it in range(iterations):
        A = rfft(a, N)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Partial flattening
        target_mag = avg_mag + flatten_rate * (mag - avg_mag)
        target_mag = np.maximum(target_mag, 1e-10)
        phase = A / (mag + 1e-30)
        A_new = phase * target_mag
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        # Decrease flatten rate over time
        flatten_rate *= 0.999
    
    return a

def logsumexp_objective_and_grad(z, T):
    """Compute LogSumExp smoothed objective and gradient in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) * T)
    w = exp_b / exp_b.sum()
    
    # Gradient with respect to a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 / T
    
    # Objective
    logsumexp = max_b + np.log(exp_b.sum()) / T
    obj = 2.0 * n * logsumexp
    
    # Convert to z-gradient (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_objective_and_grad(z_inner, T)
            return obj
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_objective_and_grad(z_inner, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    return np.exp(z_shifted) / np.exp(z_shifted).sum()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
