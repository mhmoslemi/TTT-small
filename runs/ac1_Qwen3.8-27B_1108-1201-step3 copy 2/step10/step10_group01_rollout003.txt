"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp
gradient descent with adaptive temperature cooling, peak-directed coordinate
descent, multi-start diversity, and length exploration.
Key innovation: O(n log n) gradient computation via FFT, enabling 10x+ more

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def fft_cross_corr(w, a):
    """Compute g_i = sum_k w[k] * a[k-i] using FFT.
    w has length 2n-1, a has length n.
    Returns g of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W_fft = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    return irfft(np.conj(A_fft) * W_fft, fft_size)[:n]

def logsumexp_objective_and_gradient(z, n, T):
    """Compute objective and gradient in log-domain (softmax parameterization).
    z is the log-parameter, a = softmax(z).
    """
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    
    # Objective: (2n/T) * (max_b + T * log(sum_exp))
    obj = (2.0 * n / T) * (max_b + T * np.log(sum_exp))
    
    # Gradient
    w = exp_b / sum_exp
    g = fft_cross_corr(w, a)  # g_i = sum_k w_k * a_{k-i}
    ga = (2.0 * n / T) * g  # gradient w.r.t. a
    S = np.dot(a, ga)
    grad = a * (ga - S)  # gradient w.r.t. z (softmax chain rule)
    
    return obj, grad

def logsumexp_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """Multi-stage LogSumExp optimization with FFT-based gradients."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_local):
            obj, _ = logsumexp_objective_and_gradient(z_local, n, T)
            return obj
        
        def grad(z_local):
            _, g = logsumexp_objective_and_gradient(z_local, n, T)
            return g
        
        maxiter = min(maxiter_per_stage, int(remaining * 2000))  # adaptive
        maxiter = max(maxiter, 100)
        
        try:
            res = minimize(
                func, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 20, 'disp': False}
            )
            z = res.x

            pass

    return best_a, best_score

def peak_coordinate_descent(a0, time_limit, n_steps=100000):
    """Fast peak-directed coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find the top contributor to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = idx[np.argmax(contribs)]
        
        # Try reducing the top contributor
        for factor in [0.999, 0.995, 0.99, 0.98, 0.97]:
            a_trial = a.copy()
            diff = a[top_idx] * (1.0 - factor)
            if diff < 1e-15:
                continue
            a_trial[top_idx] -= diff
            # Redistribute to the element with lowest contribution to peak
            mask = np.ones(n, dtype=bool)
            mask[top_idx] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            a_trial[mask] = others + diff * others / others_sum
            a_trial = a_trial / a_trial.sum()
            
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                break
        else:
            stall += 1
            if stall > 500:
                # Try a different strategy: reduce the peak contributor
                # and add to a random position
                pos = np.random.randint(n)
                a_trial = a.copy()
                diff = a[top_idx] * 0.01
                a_trial[top_idx] -= diff
                a_trial[pos] += diff
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall = 1000  # force break next iteration
                        if stall > 1000:
                            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
