"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(fixed_seq, solver):
    """Given fixed sequence (sum=1), find x >= 0, sum(x)=1 minimizing max_k conv(fixed, x)."""
    n = len(fixed_seq)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed_seq)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            x_val = np.asarray(x_var.value, dtype=np.float64)
            x_val = np.maximum(x_val, 0)
            s = x_val.sum()
            if s > 1e-15:
                return x_val / s, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                s = x_val.sum()
                if s > 1e-15:
                    return x_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def bidi_lp_alternation(a0, time_limit, solver, use_nesterov=True):
    """Alternating minimization with optional Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    while time.monotonic() - t0 < time_limit:
        it = int((time.monotonic() - t0) * 100)  # approximate iter count

        # Step 1: Best b given y_a
        b_new, M1 = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            if use_nesterov:
                alpha = 2.0 / (it + 2)
                y_b = (1.0 - alpha) * y_b + alpha * b_new
                y_b = normalize(y_b)
                b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            else:
                b = b_new
                y_b = b_new.copy()
            b = normalize(b)

        # Step 2: Best a given y_b
        a_new, M2 = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            if use_nesterov:
                alpha = 2.0 / (it + 3)
                y_a = (1.0 - alpha) * y_a + alpha * a_new
                y_a = normalize(y_a)
                a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            else:
                a = a_new
                y_a = a_new.copy()
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if it % 500 == 0 and it > 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 5:
                # Switch to direct mode for a while
                y_a = a.copy()
                y_b = b.copy()
                stall = 0

def peak_flattening_direction(a, solver):

    delta_var = cp.Variable(n)
    eps_var = cp.Variable()

    obj = cp.Minimize(cp.sum(G @ delta_var) / m_active + eps_var)

    constraints = [
        cp.sum(delta_var) == 0,
        a + delta_var >= 0,
        G @ delta_var <= m_active * eps_var,
    ]

    # Also constrain near-peak positions not to increase
    near_threshold = M * 0.90
    near_k = np.where((b < threshold) & (b >= near_threshold))[0]
    if 0 < len(near_k) <= 200:
        G2 = np.zeros((len(near_k), n))
        for i, k in enumerate(near_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                G2[i, idx] = 2.0 * a[k - idx]
        constraints.append(G2 @ delta_var <= 0)

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            delta_val = np.asarray(delta_var.value, dtype=np.float64)
            return delta_val, M
    except Exception:
        pass
    return None, M


def peak_flatten_iterate(a0, solver, time_limit, n_iters=200):
    """Iteratively apply peak-flattening direction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        delta, M = peak_flattening_direction(a, solver)
        if delta is None:
            break

        # Try different step sizes
        improved = False
        for step in [0.5, 0.3, 0.1, 0.05, 0.02]:
            a_trial = a + step * delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 15))

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    idx = np.arange(js, je + 1)
                    grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def asymmetric_init(n, rng=None):
    """Create an asymmetric initial sequence."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        # Skewed Gaussian
        a = np.exp(-3.0 * (x - 0.35) ** 2) + 0.3 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 1:
        # Power law decay
        a = (1 - x) ** 2.5 + 0.05
    elif choice == 2:
        # Asymmetric bump
        a = np.where(x < 0.5,
                     np.exp(-10 * (x - 0.3) ** 2),
                     np.exp(-5 * (x - 0.7) ** 2))
        a = a + 0.02
    elif choice == 3:
        # Sine-squared shifted
        a = np.sin(np.pi * (x + 0.2)) ** 2
        a = np.maximum(a, 0)
    elif choice == 4:
        # Exponential ramp
        a = np.exp(-2 * x) + 0.1 * np.exp(2 * (x - 1))
    else:
        # Linear ramp
        a = (1 - x) ** 1.5 + 0.05 * x
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def uniform_init(n):
    """Uniform initialization."""
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_step(a, lr, n_active=5):
    """One subgradient step towards the active peaks."""
    n = len(a)
    b = fast_convolve(a)
    M = b.max()

    # Get top n_active peaks
    top_idx = np.argsort(b)[-n_active:]
    top_idx = np.sort(top_idx)

    g = np.zeros(n)
    for k in top_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
    g /= len(top_idx)

    a_trial = a - lr * g
    a_trial = np.maximum(a_trial, 0)
    st = a_trial.sum()
    if st < 1e-15:
        return a.copy()
    return a_trial / st


def subgradient_method(a0, time_limit, n_steps=50000):
    """Adaptive subgradient method."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        a_trial = subgradient_step(a, lr, n_active=3)
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def direct_active_set_solve(a0, solver, n_active_candidates=None):
    """Try to solve the active set system directly."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    if n_active_candidates is None:
        n_active_candidates = [2, 3, 4, 5]

    for n_act in n_active_candidates:
        b = fast_convolve(a)
        M = b.max()
        # Get top n_act positions
        top_idx = np.argsort(b)[-n_act:]
        top_idx = np.sort(top_idx)

        # LP: minimize M subject to conv(a, x)_k <= M for k in top_idx,
        #     conv(a, x)_k <= M for all k, sum(x)=1, x>=0
        # This is the same as the standard LP but focusing on the active set
        x_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(a)
        constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                s = x_val.sum()
                if s > 1e-15:
                    x_val = x_val / s
                    sc = 2.0 * n * fast_convolve(x_val).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = x_val.copy()
        except Exception:
            pass

    return best_a, best_score


def fourier_refine(a0, time_limit, n_freq=30):
    """Refine in the low-frequency Fourier domain."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass

    # L-BFGS from the NM solution
    if time.monotonic() - t0 < time_limit - 5:
        try:
            res2 = minimize(
                fourier_obj, x_opt, method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12}
            )
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = res2.x[0]
            for i in range(1, n_freq + 1):
                A_new[i] = res2.x[2 * i - 1] + 1j * res2.x[2 * i]
            A_new[n_freq + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s > 1e-10:
                a_candidate = a_candidate / s
                sc = 2.0 * n * fast_convolve(a_candidate).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key improvements over previous:
    - Asymmetric initialization (exploits that optimal must be asymmetric)
    - Faster LP alternation with adaptive Nesterov/direct switching
    - Peak-flattening with proper step-size backtracking
    - Very low T LogSumExp polishing
    - Multi-n exploration
    - Fourier domain refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = asymmetric_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Coarse LP alternation ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 300.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = bidi_lp_alternation(best_seq, t1, solver, use_nesterov=True)
        try_update(a_lp, "lp_nesterov")

    # === Phase 2: Direct LP (no averaging) ===
    print(f"\n=== Phase 2: Direct LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 150.0)
    if t2 > 20:
        a_direct, sc_direct = bidi_lp_alternation(best_seq, t2, solver, use_nesterov=False)
        try_update(a_direct, "lp_direct")

    # === Phase 3: Peak flattening ===
    print(f"\n=== Phase 3: Peak Flattening (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 180.0)
    if t3 > 15:
        a_pf, sc_pf = peak_flatten_iterate(best_seq, solver, t3, n_iters=500)
        try_update(a_pf, "peak_flat")

    # === Phase 4: Subgradient ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 10:
        a_sg, sc_sg = subgradient_method(best_seq, t4, n_steps=30000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Very low T LogSumExp ===
    print(f"\n=== Phase 5: Low-T L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 300.0)
    if t5 > 20:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb, sc_lb = logsumexp_lbgfs(best_seq, t5, T_stages=T_stages)
        try_update(a_lb, "lowT_lbgfs")

    # === Phase 6: Fourier refinement ===
    print(f"\n=== Phase 6: Fourier Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if t6 > 15:
        for nf in [20, 40, 60]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            a_four, sc_four = fourier_refine(best_seq, min(rem * 0.3, 35.0), n_freq=nf)
            try_update(a_four, f"fourier_K{nf}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = n_best
        for mult in [0.8, 0.9, 1.0, 1.1, 1.25, 1.5]:
            cand_n = int(base_n * mult)
            if 400 <= cand_n <= 3000 and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if cand_n == n_best:
                continue
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 25:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP
            a_cand, _ = bidi_lp_alternation(init_cand, t_sub * 0.5, solver, use_nesterov=True)
            try_update(a_cand, f"lp_n{cand_n}")
            # Quick L-BFGS
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.3, 40.0),
                                             T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05])
                try_update(a_cand2, f"lbgfs_n{cand_n}")

    # === Phase 8: Second round of intensive polishing ===
    print(f"\n=== Phase 8: Intensive Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # More LP
        t_lp2 = min(rem * 0.3, 60.0)
        if t_lp2 > 10:
            a_lp2, _ = bidi_lp_alternation(best_seq, t_lp2, solver, use_nesterov=False)
            try_update(a_lp2, "lp_polish2")

        # Peak flatten
        rem = deadline - time.monotonic()
        if rem > 20:
            a_pf2, _ = peak_flatten_iterate(best_seq, solver, min(rem * 0.3, 40.0), n_iters=300)
            try_update(a_pf2, "pf_polish2")

        # Very low T
        rem = deadline - time.monotonic()
        if rem > 15:
            a_lbgfs2, _ = logsumexp_lbgfs(best_seq, min(rem * 0.4, 50.0),
                                          T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_lbgfs2, "lbgfs_polish2")

        # Subgradient final
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sg2, _ = subgradient_method(best_seq, min(rem * 0.3, 20.0), n_steps=10000)
            try_update(a_sg2, "sg_polish2")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
