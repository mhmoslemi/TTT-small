import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def spectral_flatten(a0, time_limit, verbose=True):
    """
    Frequency-domain approach: iteratively flatten the power spectrum.
    The autocorrelation is ifft(|FFT(a)|^2). To minimize its peak,
    we want |FFT(a)|^2 to be as flat as possible.
    
    Approach: In frequency domain, modify the phase of FFT(a) to reduce
    the peak of the inverse FFT. This is similar to "spectral balancing".
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        autocorr = irfft(power, fft_size)[:2*n-1]
        M = float(autocorr.max())
        
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Strategy: modify the amplitude of A to flatten the autocorrelation
        # The peak of autocorrelation is at lag 0 (since a >= 0), which equals sum(a^2)
        # So we need to reduce sum(a^2) while keeping sum(a) = 1
        # This means making a more uniform, which conflicts with reducing other peaks
        
        # Alternative: use the "successive over-relaxation" approach
        # Compute target power spectrum (flat), then adjust
        target_power = np.full(len(power), power.sum() / len(power))
        # Blend current and target power
        alpha = 0.1 / (1 + it * 0.01)
        new_power = (1 - alpha) * power + alpha * target_power
        # Keep phase the same
        phase = A / np.abs(A + 1e-30)
        new_A = np.sqrt(new_power) * phase
        new_a = irfft(new_A, fft_size)
        new_a = np.maximum(new_a[:n], 0)
        new_s = new_a.sum()
        if new_s < 1e-15:
            break
        new_a = new_a / new_s
        
        a = new_a
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            score_now = 2.0 * n * fast_convolve(a).max()
            print(f"    Spectral it={it} ({elapsed:.0f}s): current={score_now:.8f} best={best_score:.8f}")
        
        if it > 200:
            break
            
    return best_a, best_score


def lp_alternating_v2(a0, time_limit, solver, verbose=True):
    """
    Improved LP-based alternating with line search and restart.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Line search: find best alpha in (0, 1]
        best_alpha = 1.0
        best_trial_score = 2.0 * n * fast_convolve(a).max()
        
        for alpha in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3]:
            trial = (1 - alpha) * a + alpha * b_new
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            trial_score = 2.0 * n * fast_convolve(trial).max()
            if trial_score < best_trial_score - 1e-14:
                best_trial_score = trial_score
                best_alpha = alpha
            if alpha < best_alpha:
                break  # Don't waste time on smaller alphas if we already found improvement
        
        a = (1 - best_alpha) * a + best_alpha * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Simulated annealing: occasionally accept worse moves
        if stagnation > 10 and np.random.random() < 0.1:
            noise_scale = 0.05 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps > 1e-15:
                a_pert = a_pert / ps
                pert_score = 2.0 * n * fast_convolve(a_pert).max()
                # Accept if not much worse
                if pert_score < best_score * 1.01:
                    a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP2 it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f} alpha={best_alpha}")

    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

    """Build the convolution matrix T such that T @ b = conv(a, b)."""

        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1

    try:
        prob.solve(solver=solver, verbose=False)

            return a.copy(), float('inf')

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                changes = changes.copy()
                changes[quad_offset] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
                sweep_improvements += 1

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 40:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} improvements={sweep_improvements}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Bisection-refined coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            if unaff_max >= M - 1e-16:
                continue

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            quad_offset = 2 * i - k_start

            best_delta = 0.0
            best_new_M = M

            # Coarse: multiplicative factors
            for factor in [0.99, 0.995, 0.998, 1.002, 1.005, 1.01]:
                delta = a[i] * (factor - 1.0)
                new_affected = conv_affected + 2.0 * delta * a_shifted
                new_affected = new_affected.copy()
                new_affected[quad_offset] += delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            # Bisection refinement around best_delta
            if best_delta != 0.0:
                lo = best_delta * 0.3
                hi = best_delta * 1.7
                for _ in range(15):
                    mid = (lo + hi) * 0.5
                    new_affected = conv_affected + 2.0 * mid * a_shifted
                    new_affected = new_affected.copy()
                    new_affected[quad_offset] += mid * mid
                    new_M = max(unaff_max, float(new_affected.max()))
                    # Compare mid vs current best
                    new_affected_best = conv_affected + 2.0 * best_delta * a_shifted
                    new_affected_best = new_affected_best.copy()
                    new_affected_best[quad_offset] += best_delta * best_delta
                    best_M_check = max(unaff_max, float(new_affected_best.max()))
                    if new_M < best_M_check - 1e-18:
                        lo = mid
                        best_delta = mid
                        best_new_M = new_M
                    else:
                        hi = mid

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                changes = changes.copy()
                changes[quad_offset] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def spectral_flatten(a0, time_limit, verbose=True):
    """
    Frequency-domain approach: iteratively modify the spectrum to flatten autocorrelation.
    Uses a combination of amplitude modulation and phase adjustment.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 500:
        it += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        autocorr = irfft(power, fft_size)[:2 * n - 1]
        M = float(autocorr.max())

        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Strategy: reduce power at frequencies that contribute to peaks
        # The peak at lag 0 is sum(a^2) - we want to reduce concentrated energy
        # Use "water-filling" in frequency domain
        avg_power = power.sum() / len(power)

        # Dampen high-power frequencies, boost low-power ones
        alpha = 0.05 / (1.0 + 0.05 * it)
        ratio = np.sqrt(np.maximum(avg_power / np.maximum(power, 1e-30), 0.1))
        ratio = np.minimum(ratio, 2.0)  # Cap the adjustment

        new_power = power * ratio
        # Preserve total power (approximately)
        scale = np.sqrt(power.sum() / new_power.sum())
        new_power = new_power * scale * scale

        # Keep phase
        phase = A / (np.abs(A) + 1e-30)
        new_A = np.sqrt(new_power) * phase
        new_a = irfft(new_A, fft_size)
        new_a = np.maximum(new_a[:n], 0.0)
        new_s = new_a.sum()
        if new_s < 1e-15:
            break
        new_a = new_a / new_s

        # Only accept if it improves (greedy)
        new_score = 2.0 * n * fast_convolve(new_a).max()
        if new_score < score - 1e-14:
            a = new_a

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * fast_convolve(a).max()
            print(f"    Spectral it={it} ({elapsed:.0f}s): current={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, lr_init=0.01, top_k=30, verbose=True):
    """Subgradient descent with Nesterov acceleration and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    y = a.copy()  # Nesterov lookahead

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-12:
                break

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]

        temp = 0.004 * max(float(pv.max()), 1e-10)
        w = np.exp((pv - float(pv.max())) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * y[k - i_end:k - i_start + 1][::-1]

        # Nesterov update
        a_old = a.copy()
        a_new = y - lr * sg
        a_new = simplex_project(a_new)
        momentum = 0.85
        y = a_new + momentum * (a_new - a_old)
        y = simplex_project(y)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD-N it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def diverse_init(n, rng, init_type):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.25, 0.35)
        c2 = n * rng.uniform(0.65, 0.75)
        w = n * rng.uniform(0.1, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(4, 12))
        t = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n, dtype=np.float64)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    else:
        t = np.arange(n, dtype=np.float64)
        freqs = rng.uniform(2.0, 20.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.15 * np.cos(2 * np.pi * f * t / n + p)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization: LP alternating with line search, exact coordinate
    descent with peak-aware skipping, spectral flattening, Nesterov SGD, and
    bisection fine-polish. Targets sub-1.503 score.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating with Line Search (35%) ===
    remaining = deadline - time.monotonic()
    t1_budget = remaining * 0.35
    print(f"\n=== Phase 1: LP Alt v2 ({t1_budget:.0f}s, best={best_score:.8f}) ===")

    t_alloc = min(t1_budget * 0.6, 500.0)
    a_lp, s_lp = lp_alternating_v2(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp1")

    remaining1 = min(deadline - t1_budget - time.monotonic(), 300.0)
    if remaining1 > 30:
        a_lp2, s_lp2 = lp_alternating_v2(best_seq, remaining1, solver, verbose=True)
        try_update(a_lp2, "lp2")

    # === Phase 2: Exact Coordinate Descent (25%) ===
    remaining = deadline - time.monotonic()
    t2_budget = remaining * 0.25
    print(f"\n=== Phase 2: Exact CD ({t2_budget:.0f}s, best={best_score:.8f}) ===")

    t_alloc = min(t2_budget * 0.6, 300.0)
    a_cd, s_cd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
    try_update(a_cd, "cd_main")

    remaining2 = min(deadline - t2_budget - time.monotonic(), 200.0)
    if remaining2 > 30:
        a_cd2, s_cd2 = exact_coord_descent(best_seq, remaining2, verbose=True, label="round2")
        try_update(a_cd2, "cd_r2")

    # === Phase 3: Spectral + Nesterov SGD (15%) ===
    remaining = deadline - time.monotonic()
    t3_budget = remaining * 0.15
    print(f"\n=== Phase 3: Spectral + SGD ({t3_budget:.0f}s, best={best_score:.8f}) ===")

    t_spec = min(t3_budget * 0.5, 120.0)
    if t_spec > 20:
        a_spec, s_spec = spectral_flatten(best_seq, t_spec, verbose=True)
        try_update(a_spec, "spectral")

    remaining3 = min(deadline - t3_budget - time.monotonic(), 150.0)
    if remaining3 > 20:
        a_sgd, s_sgd = subgradient_nesterov(best_seq, remaining3, lr_init=0.008, top_k=40, verbose=True)
        try_update(a_sgd, "nesterov")

    # === Phase 4: Fine Polish (25%) ===
    remaining = deadline - time.monotonic()
    t4_budget = remaining * 0.25
    print(f"\n=== Phase 4: Fine Polish ({t4_budget:.0f}s, best={best_score:.8f}) ===")

    t_fp = min(t4_budget * 0.5, 250.0)
    a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
    try_update(a_fp, "fine1")

    remaining4 = min(deadline - t4_budget - time.monotonic(), 200.0)
    if remaining4 > 30:
        a_cd3, s_cd3 = exact_coord_descent(best_seq, min(remaining4 * 0.6, 150.0), verbose=True, label="late")
        try_update(a_cd3, "cd_late")

    # Final polish rounds
    remaining5 = deadline - time.monotonic() - 5
    if remaining5 > 10:
        a_fp2, s_fp2 = fine_polish(best_seq, min(remaining5, 100.0), verbose=True)
        try_update(a_fp2, "fine2")

    remaining6 = deadline - time.monotonic() - 2
    if remaining6 > 5:
        a_fp3, s_fp3 = fine_polish(best_seq, min(remaining6, 50.0), verbose=False)
        try_update(a_fp3, "fine3")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
