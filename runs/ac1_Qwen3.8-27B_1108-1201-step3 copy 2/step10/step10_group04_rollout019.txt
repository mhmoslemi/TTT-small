def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_cross_matrix(a):
    n = len(a)
    size = 2 * n - 1
    A = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_br_lp(a, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    A = build_cross_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def softmax_from_z(z):

    return exp_z / s

def lse_lbgfs_optimize(a0, time_limit, T_start=200.0, T_end=0.02, n_stages=15):
    n = len(a0)
    a_init = normalize(a0)
    z = z_from_a(a_init).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = score_normalized(a_init)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters = max(15, min(80, int(remaining / max(n_stages - si, 1))))

        def func_z(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_tmp = softmax_from_z(z_in)
            b_tmp = fast_convolve(a_tmp)
            Tb = stage_T * b_tmp
            max_Tb = min(Tb.max(), 700.0)
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac_z(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_tmp = softmax_from_z(z_in)
            b_tmp = fast_convolve(a_tmp)
            Tb = stage_T * b_tmp
            max_Tb = min(Tb.max(), 700.0)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_tmp)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_tmp[k - idx]
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)

        try:
            res = minimize(
                fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 200,
                         'disp': False}
            )
            z = res.x
            a_chk = softmax_from_z(z)
            sc_chk = score_normalized(a_chk)
            if sc_chk < best_score:
                best_score = sc_chk
                best_z = z.copy()
        except Exception:
            pass

    return softmax_from_z(best_z)

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)})")

    print(f"\n=== Phase 1: Main optimization at n={n_base} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.55, 900.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_single_length(best_seq, t1, solver, f"n{n_base}")
        global_update(a_out, f"main_n{n_base}")

    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    candidate_lengths = []
    for mult in [0.8, 1.0, 1.2, 1.5]:
        ln = int(n_base * mult)
        if 200 <= ln <= 3000 and ln != n_base:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))[:4]

    for ln in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.20, 200.0)
        if t_alloc < 40:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        init_long = interpolate_seq(best_seq, ln)
        sc_init = evaluate_local(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
        a_out, _ = optimize_single_length(init_long, t_alloc, solver, f"n{ln}")
        global_update(a_out, f"long_n{ln}")

    print(f"\n=== Phase 3: Final polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp = min(rem * 0.35, 150.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_optimize(best_seq, t_lp, solver)
            global_update(a_fin, "polish_lp")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_lse = min(rem * 0.30, 100.0)
            if t_lse > 8:
                a_fin2 = lse_lbgfs_optimize(best_seq, t_lse, T_start=5.0, T_end=0.01, n_stages=10)
                global_update(a_fin2, "polish_lse")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_pe = min(rem * 0.30, 80.0)
            if t_pe > 5:
                a_fin3, _ = peak_shave_optimize(best_seq, t_pe)
                global_update(a_fin3, "polish_pe")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_lp2 = min(rem * 0.40, 60.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_lp_optimize(best_seq, t_lp2, solver)
                global_update(a_fin4, "polish_lp2")

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
