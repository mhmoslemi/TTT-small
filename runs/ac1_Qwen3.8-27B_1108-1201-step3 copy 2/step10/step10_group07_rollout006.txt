from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)"""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response_scipy(a):
    """LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0 using scipy linprog"""
    n = len(a)
    T = build_conv_matrix(a)
    # Variables: b_0, ..., b_{n-1}, M (total n+1 variables)
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])

    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={"presolve": True, "dual_feasibility_tolerance": 1e-9,
                           "primal_feasibility_tolerance": 1e-9})
    if res.status != 0:
        return a.copy(), float('inf')
    b_val = np.maximum(res.x[:n], 0.0)
    M_val = float(res.x[-1])
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    return b_val, M_val


def diverse_init(n, rng, init_type=0):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / max(center, 1), 0)
    elif init_type == 1:
        x = (t - center) / max(center, 1)
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / max(center, 1)
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Raised cosine with variable width
        x = (t - center) / max(center, 1)
        width = rng.uniform(0.5, 1.0)
        a = 0.5 * (1 + np.cos(np.pi * x / width)) ** 2
        a = np.where(np.abs(x) < width, a, 0.0)
    else:
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def lp_alternating_averaged(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    avg_a = a.copy()
    avg_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        b_new, M_lp = lp_best_response(a)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Update average (exponential moving average with increasing weight)
        avg_count += 1
        alpha_avg = 1.0 / avg_count
        avg_a = (1 - alpha_avg) * avg_a + alpha_avg * b_new

        # Check average score
        avg_score = 2.0 * n * fast_convolve(avg_a).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_a.copy()

        # Damping schedule: start aggressive, become conservative
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 80:
            alpha = 0.5
        else:
            alpha = max(0.2, 8.0 / (it + 20.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection: try perturbation
        if it > 50 and it % 25 == 0:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.default_rng(it * 31).standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.002:
                a = a_pert

        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f} avg={avg_score:.10f}")

    # Final check of average
    avg_score = 2.0 * n * fast_convolve(avg_a).max()
    if avg_score < best_score:
        best_score = avg_score
        best_a = avg_a.copy()

    return best_a, best_score


def smooth_max_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    momentum = 0.9
    temp = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

        # Compute smooth max gradient
        # w_k = exp(conv[k]/temp) / sum(exp(conv[j]/temp))
        # Use top-K to avoid underflow
        K = min(50, 2 * n - 1)
        peak_idx = np.argpartition(conv, -K)[-K:]
        peak_vals = conv[peak_idx]
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w = w / ws

        # Gradient: grad_j = sum_k w_k * 2 * a[k-j]
        grad = np.zeros(n)
        for idx in range(K):
            if w[idx] < 1e-14:
                continue
            k = int(peak_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Nesterov momentum
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = simplex_project(a_new)

        # Adaptive lr
        if it % 50 == 0:
            lr *= 0.95
        if lr < 1e-8:
            break

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def adaptive_coord_descent(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    factors_list = [-0.4, -0.25, -0.15, -0.08, -0.03, 0.03, 0.08, 0.15, 0.25, 0.4]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            base = max(a[i], 1e-8)
            for factor in factors_list:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 40:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    fine_factors = [0.997, 0.994, 0.991, 0.9955, 1.001, 1.003, 1.005, 1.0045, 1.002]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in fine_factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def two_coord_search(a0, time_limit, verbose=True):
    """Try pairs of coordinates simultaneously for better local moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = 8
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.98:
                continue

            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Find top contributing pairs
            contributions = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contributions.append((i, j, a[i] * a[j]))
            contributions.sort(key=lambda x: x[2], reverse=True)

            for (i1, j1, c1) in contributions[:5]:
                for (i2, j2, c2) in contributions[5:10]:
                    if c2 < 1e-14:
                        continue
                    # Try transferring mass from pair (i1,j1) to reduce the peak
                    # Reduce a[max(i1,j1)] and increase a[min(i1,j2), min(j1,i2)] etc.
                    # Simpler: try reducing the larger contributor and see
                    targets = set()
                    targets.add(i1)
                    targets.add(j1)
                    if i2 != i1 and j2 != j1:
                        targets.add(i2)
                        targets.add(j2)

                    for t in targets:
                        for frac in [0.05, 0.1, 0.15, 0.2]:
                            delta = -a[t] * frac
                            if abs(delta) < 1e-15:
                                continue
                            new_a = a.copy()
                            new_a[t] = max(new_a[t] + delta, 0)
                            # Redistribute to neighbors
                            delta_redist = -delta  # positive
                            # Spread to adjacent coordinates
                            spread_idx = [t - 1, t + 1, t - 2, t + 2]
                            valid_spread = [x for x in spread_idx if 0 <= x < n]
                            if not valid_spread:
                                continue
                            per_share = delta_redist / len(valid_spread)
                            for sx in valid_spread:
                                new_a[sx] += per_share

                            # Quick score check
                            new_conv = fast_convolve(new_a)
                            new_M = float(new_conv.max())
                            if new_M < M - 1e-12:
                                a = new_a
                                conv = new_conv
                                M = new_M
                                improved = True
                                cur_score = 2.0 * n * M
                                if cur_score < best_score:
                                    best_score = cur_score
                                    best_a = a.copy()
                                break
                        if improved:
                            break

        if not improved and rounds > 100:
            break

        if verbose and rounds % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TwoCoord round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(2024)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)}) [prev={old_score:.8f}]")

    # === Phase 1: LP Alternating (55% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    p1_time = (deadline - time.monotonic()) * 0.55

    # Main LP run from parent
    t_alloc = min(p1_time * 0.5, 500.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating_averaged(best_seq, t_alloc, verbose=True)
    try_update(a_lp, "lp_parent")

    # LP from diverse inits (shorter runs)
    inits = [
        diverse_init(n, rng, 0),
        diverse_init(n, rng, 1),
        diverse_init(n, rng, 2),
        diverse_init(n, rng, 6),
    ]
    for idx, init in enumerate(inits):
        if time.monotonic() > time.monotonic() + p1_time - 60:
            break
        if time.monotonic() + 40 > deadline - p1_time - time.monotonic():
            break
        remaining_p1 = (time.monotonic() + p1_time) - time.monotonic()
        if remaining_p1 < 50:
            break
        t_alloc_init = min(remaining_p1 * 0.2, 60.0)
        print(f"  LP init {idx+1}/{len(inits)}, budget={t_alloc_init:.0f}s")
        a_tmp, s_tmp = lp_alternating_averaged(init, t_alloc_init, verbose=False)
        try_update(a_tmp, f"lp_init{idx+1}")

    # Second LP round on best
    remaining_p1 = (time.monotonic() + p1_time) - time.monotonic()
    if remaining_p1 > 60:
        t_alloc2 = min(remaining_p1, 200.0)
        print(f"  LP round 2, budget={t_alloc2:.0f}s")
        a_lp2, s_lp2 = lp_alternating_averaged(best_seq, t_alloc2, verbose=True)
        try_update(a_lp2, "lp_round2")

    # === Phase 2: Smooth-max GD (15% time) ===
    print(f"\n=== Phase 2: Smooth-max GD (best={best_score:.8f}) ===")
    p2_time = (deadline - time.monotonic()) * 0.15

    if p2_time > 30:
        t_alloc = min(p2_time * 0.7, 120.0)
        print(f"  SmoothGD budget={t_alloc:.0f}s")
        a_gd, s_gd = smooth_max_gradient_descent(best_seq, t_alloc, verbose=True)
        try_update(a_gd, "smooth_gd")

        if p2_time - t_alloc > 20:
            a_gd2, s_gd2 = smooth_max_gradient_descent(a_lp, min(p2_time - t_alloc - time.monotonic() + time.monotonic(), 40.0), verbose=False)
            try_update(a_gd2, "smooth_gd_lp")

    # === Phase 3: Coordinate Descent (20% time) ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    p3_time = (deadline - time.monotonic()) * 0.20

    t_alloc = min(p3_time * 0.5, 200.0)
    if t_alloc > 20:
        print(f"  CD main budget={t_alloc:.0f}s")
        a_cd, s_cd = adaptive_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

    remaining_p3 = (time.monotonic() + p3_time) - time.monotonic()
    if remaining_p3 > 40:
        t_alloc2 = min(remaining_p3 * 0.5, 80.0)
        if t_alloc2 > 15:
            a_cd2, s_cd2 = adaptive_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 4: Fine Polishing (10% time) ===
    print(f"\n=== Phase 4: Fine Polishing (best={best_score:.8f}) ===")
    p4_time = deadline - time.monotonic() - 15

    # Fine polish
    if p4_time > 40:
        t_fp = min(p4_time * 0.5, 100.0)
        print(f"  FinePolish budget={t_fp:.0f}s")
        a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
        try_update(a_fp, "fine")

    # Two-coordinate search
    remaining_p4 = p4_time - t_fp
    if remaining_p4 > 30:
        t_tc = min(remaining_p4 * 0.5, 60.0)
        print(f"  TwoCoord budget={t_tc:.0f}s")
        a_tc, s_tc = two_coord_search(best_seq, t_tc, verbose=True)
        try_update(a_tc, "two_coord")

    # Final fine polish
    remaining_p4b = p4_time - t_fp - t_tc
    if remaining_p4b > 15:
        t_final = min(remaining_p4b, 50.0)
        print(f"  Final polish budget={t_final:.0f}s")
        a_fin, s_fin = fine_polish(best_seq, t_final, verbose=False)
        try_update(a_fin, "final")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
