def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def fictitious_play_mirror_descent(a0, time_limit, eta0=0.1, nesterov=True):
    """
    Fictitious play with mirror descent for the zero-sum game:
    min_a max_k q_k(a) where q_k(a) = conv(a,a)_k, a in simplex.
    
    Dual: max_y min_a a^T M_y a where M_y[i,j] = y[i+j] (Hankel).

    """
    n = len(a0)
    a = normalize(a0).copy()
    y_avg = np.ones(2 * n - 1) / (2 * n - 1)  # uniform over peaks
    a_avg = a.copy()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    # Nesterov state
    y_nest = a.copy()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        batch_size = 50 if it < 100 else 20
        
        for _ in range(batch_size):
            # Compute convolution at current a (or Nesterov point)
            eval_a = y_nest if nesterov else a
            b = fast_convolve(eval_a)
            max_b = b.max()
            
            # Best response of y: find the peak
            # Use a small active set for stability
            threshold = max_b * 0.999
            active = np.where(b >= threshold)[0]
            
            # Update y: point mass on active peaks (uniform over them)
            y_br = np.zeros(2 * n - 1)
            y_br[active] = 1.0 / len(active)
            
            # Update y_avg (running average)
            t_step = 1.0 / (it * batch_size + 1)
            y_avg = (1.0 - t_step) * y_avg + t_step * y_br
            
            # Compute M_y @ a using Hankel structure
            Ma = hankel_matvec(y_avg, a)
            
            # Mirror descent update
            log_a = np.log(np.maximum(a, 1e-300))
            # Gradient of a^T M a is 2 M a
            grad = 2.0 * Ma
            # Normalize gradient for stability
            grad = grad - grad.mean()
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-300:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            # Nesterov momentum
            if nesterov:
                alpha = 2.0 / (it * batch_size + 2)
                y_nest = (1.0 - alpha) * y_nest + alpha * a_new
                y_nest = np.maximum(y_nest, 0)
                sy = y_nest.sum()
                if sy > 1e-15:
                    y_nest = y_nest / sy
            
            a = a_new
            a_avg = (1.0 - t_step) * a_avg + t_step * a
        
        # Evaluate periodically
        if it % 10 == 0 or it < 5:
            score = evaluate_local(a)
            score_avg = evaluate_local(a_avg)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            elif score_avg < best_score - 1e-12:
                best_score = score_avg
                best_a = a_avg.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FP iter={it} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")
            
            # Adapt eta
            if no_improve >= 100:
                eta *= 0.95
                no_improve = 0
            elif no_improve >= 50:
                eta *= 0.99
        
        # Periodic perturbation

            a_avg = a.copy()
            no_improve = 0

    # If i = k-i (i.e., k = 2i), then d b_k / d a_i = 2 a_i (the two terms are the same).

    #   Term where k-j = i, i.e., j = k-i: a_i (if k-i is a valid j, i.e., 0 <= k-i < n and k-i ≠ i)
    #   If k-i = i (i.e., k = 2i): only one term, contributes 2 * a_i (since a_i appears twice)
    #   Wait no, if j = i and k-j = i, that's the same term a_i * a_i = a_i^2, and its derivative is 2 a_i.
    #   If j = i but k-j ≠ i: contributes a_{k-i}
    #   If k-j = i but j ≠ i: contributes a_j = a_{k-i}

    # In both cases: 2 a_{k-i} (with the convention that a_{k-i} is the same as a_i when k=2i).
    # But wait, when i = k-i, the term is a_i^2, and d(a_i^2)/d(a_i) = 2 a_i. And 2 a_{k-i} = 2 a_i. ✓
    # When i ≠ k-i: the terms are a_i * a_{k-i} (from j=i) and a_{k-i} * a_i (from j=k-i).
    # d/d a_i [a_i * a_{k-i}] = a_{k-i}
    # d/d a_i [a_{k-i} * a_i] = a_{k-i}
    # Total: 2 a_{k-i}. ✓
    
    # So grad b_k / d a_i = 2 a_{k-i} (valid indices only)
    
    # grad obj / d a_i = (2n/T) * sum_k w_k * 2 a_{k-i}
    #                    = (4n/T) * sum_k w_k * a_{k-i}

    # sum_k w_k * a_{k-i} = sum_k w_k * a_{k-i}
    # Let's define v_j = a_j. Then sum_k w_k v_{k-i} is a cross-correlation.

    # Since len(w) = 2n-1 and len(a) = n, we get n values.
    
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # The objective (for minimizing):

    # = 2n * (max_b + T * log(sum(exp((b-max_b)/T)) / T))
    # Hmm, let me just use: obj = 2n * (max_b + T * log(np.mean(exp((b-max_b)/T))))
    # This is a valid smooth approximation of 2n * max_b.
    
    obj = 2.0 * n * (max_b + T * np.log(np.mean(np.exp((b - max_b) / T))))
    
    return obj, grad_a


def logsumexp_lbfgs(a0, time_limit, T_stages=None):
    """L-BFGS on the log-sum-exp smoothed objective."""
    n = len(a0)
    a = normalize(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Use L-BFGS on the simplex via reparameterization
        # Work with z = log(a), then a = softmax(z)
        z = np.log(np.maximum(a, 1e-300))
        
        def func(z):
            a_test = np.exp(z - z.max())
            a_test = a_test / a_test.sum()
            obj, _ = logsumexp_objective(a_test, T)
            return obj
        
        def jac(z):
            a_test = np.exp(z - z.max())
            a_test = a_test / a_test.sum()
            _, grad_a = logsumexp_objective(a_test, T)
            # Convert to z-gradient: grad_z = a * (grad_a - grad_a @ a)
            S = np.dot(grad_a, a_test)
            return a_test * (grad_a - S)
        
        try:
            maxiter = max(50, int(remaining / (T_stages.index(T) + 1) / 10))
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                   'disp': False})
            z = res.x
        except Exception:
            pass
        
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score
