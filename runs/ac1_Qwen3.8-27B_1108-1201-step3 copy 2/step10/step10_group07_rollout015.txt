"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a, n):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver, tol=1e-8):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, tol=tol)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, tol=tol)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, verbose=True, label=""):
    """
    LP-based best-response alternating minimization with adaptive damping.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_a = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_a = b_new.copy()
            if verbose and it % 20 == 0:
                print(f"    LP[{label}] it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

        if it <= 10:
            a = b_new.copy()
        elif it <= 30:
            alpha = 0.6
            a = (1 - alpha) * a + alpha * b_new
            a = normalize(a)
        elif it <= 80:
            alpha = max(0.4, 15.0 / (it + 5.0))
            a = (1 - alpha) * a + alpha * b_new
            a = normalize(a)
        else:
            alpha = max(0.25, 20.0 / (it + 10.0))
            a = (1 - alpha) * a + alpha * b_new
            a = normalize(a)

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            if verbose and it % 20 == 0:
                print(f"    LP[{label}] it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} (updated)")

        # Stagnation detection and restart
        if prev_a is not None:
            diff = np.linalg.norm(a - prev_a)
            if diff < 1e-6:
                stagnation += 1
            else:
                stagnation = max(0, stagnation - 1)
        prev_a = a.copy()

        if stagnation > 15:
            # Targeted perturbation: identify peak-contributing coordinates
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            # Reduce the largest contributor to the peak
            max_contrib_idx = -1
            max_contrib = 0
            for i in range(i_start, i_end + 1):
                j = peak_k - i
                c = a[i] * a[j]

                a = normalize(a + noise)
            stagnation = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with analytical 1D optimization per coordinate.
    For each coordinate i, finds the exact delta that minimizes the new max convolution.
    """
    n = len(a0)
    a = normalize(a0)
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_M = M

        for i in range(n):
            ai = a[i]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1].copy()
            c_aff = conv[k_start:k_end + 1].copy()

            r_quad = 2 * i - k_start
            has_quad = (0 <= r_quad < affected_len)

            # Find active positions (those near the current max)
            threshold = M * 0.99
            active = np.where(c_aff >= threshold)[0]
            if len(active) == 0:
                top_k = min(5, affected_len)
                active = np.argpartition(c_aff, -top_k)[-top_k:]

            # Build candidate deltas
            candidates = set()
            candidates.add(0.0)
            candidates.add(-ai)  # boundary: a[i] -> 0

            # For each active r, delta that makes c_aff[r] + 2*delta*a_shifted[r] = unaff_max
            for r in active:
                if a_shifted[r] > 1e-18:
                    d = (unaff_max - c_aff[r]) / (2.0 * a_shifted[r])
                    candidates.add(max(d, -ai))

            # Intersections between pairs of active lines
            for idx1 in range(min(len(active), 8)):
                r1 = active[idx1]
                for idx2 in range(idx1 + 1, min(len(active), 8)):
                    r2 = active[idx2]
                    denom = 2.0 * (a_shifted[r2] - a_shifted[r1])
                    if abs(denom) > 1e-18:
                        d = (c_aff[r1] - c_aff[r2]) / denom
                        candidates.add(max(d, -ai))

            # If has_quad: intersection of parabola with lines
            if has_quad:
                # Parabola vertex
                d_vertex = -a_shifted[r_quad]
                candidates.add(max(d_vertex, -ai))

                # Parabola = line intersections
                for r in active:
                    if r == r_quad:
                        continue
                    # c_quad + 2*s_quad*d + d^2 = c_r + 2*s_r*d
                    # d^2 + 2*(s_quad - s_r)*d + (c_quad - c_r) = 0
                    A_q = 1.0
                    B_q = 2.0 * (a_shifted[r_quad] - a_shifted[r])
                    C_q = c_aff[r_quad] - c_aff[r]
                    disc = B_q * B_q - 4.0 * C_q
                    if disc >= 0:
                        sqrt_d = np.sqrt(disc)
                        candidates.add(max((-B_q + sqrt_d) / 2.0, -ai))
                        candidates.add(max((-B_q - sqrt_d) / 2.0, -ai))

                # Parabola = unaff_max
                A_q = 1.0
                B_q = 2.0 * a_shifted[r_quad]
                C_q = c_aff[r_quad] - unaff_max
                disc = B_q * B_q - 4.0 * C_q
                if disc >= 0:
                    sqrt_d = np.sqrt(disc)
                    candidates.add(max((-B_q + sqrt_d) / 2.0, -ai))
                    candidates.add(max((-B_q - sqrt_d) / 2.0, -ai))

            # Evaluate candidates
            best_delta = 0.0
            best_new_M = M

            for d in candidates:
                d = max(d, -ai + 1e-16)
                new_c = c_aff + 2.0 * d * a_shifted
                if has_quad:
                    new_c[r_quad] += d * d
                new_M_val = max(unaff_max, float(new_c.max()))
                if new_M_val < best_new_M - 1e-15:
                    best_new_M = new_M_val
                    best_delta = d

            if best_delta < -1e-16 and best_new_M < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if has_quad:
                    changes[r_quad] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved or (M > sweep_start_M * (1 - 1e-12)):
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.12f}")

    return best_a, best_score

    M = float(conv.max())

    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Multiplicative factors for fine adjustment
    factors = []
    for base in [0.1, 0.3, 0.5, 1.0, 3.0, 5.0, 10.0, 30.0]:
        for sign in [-1, 1]:
            factors.append(1.0 + sign * base * 0.001)
            factors.append(1.0 + sign * base * 0.0001)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            c_aff = conv[k_start:k_end + 1]
            r_quad = 2 * i - k_start
            has_quad = (0 <= r_quad < k_end - k_start + 1)

            best_delta = 0.0
            best_new_M = M

            for f in factors:
                delta = ai * (f - 1.0)
                if ai + delta < -1e-15:
                    continue
                new_c = c_aff + 2.0 * delta * a_shifted
                if has_quad:
                    new_c = new_c.copy()
                    new_c[r_quad] += delta * delta
                new_M_val = max(unaff_max, float(new_c.max()))
                if new_M_val < best_new_M - 1e-16:
                    best_new_M = new_M_val
                    best_delta = delta

            if abs(best_delta) > 1e-18 and best_new_M < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if has_quad:
                    changes = changes.copy()
                    changes[r_quad] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Fine[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, label=""):
    """
    Mirror descent with entropic mirror (exponential weights).
    a_new[i] = a[i] * exp(-lr * sg[i]) / Z
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        M = float(conv.max())
        score = 2.0 * n * M
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-8:
                break

        # Compute subgradient: gradient of max_k c[k] w.r.t. a
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[j] = a[k-j] + a[j] (for valid k-j, j)
        # Subgradient of max: 0.5 * (a[k*-j] + a[j]) for k* = argmax

        # Use softmax-weighted subgradient over top peaks
        top_k = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                sg[j] += w[idx] * (a[k - j] + a[j])

        # Mirror descent step: a_new = a * exp(-lr * sg) / Z
        log_a = np.log(a + 1e-30) - lr * sg
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a = a_new / a_new.sum()

        conv = fast_convolve(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0, parent=None):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        x = (t - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** 2
    elif init_type == 3:
        # Upsample parent
        if parent is not None:
            from scipy.interpolate import interp1d
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            f = interp1d(t_old, parent, kind='cubic', fill_value='extrapolate')
            a = np.maximum(f(t_new), 0)
        else:
            t = np.arange(n)
            a = np.exp(-0.5 * ((t - n/2) / (n/4))**2)
    elif init_type == 4:
        # Smooth random
        freqs = rng.integers(2, 15, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amplitudes = rng.uniform(0.3, 1.0, size=5)
        a = np.zeros(n)
        t = np.arange(n)
        for f_idx in range(5):
            a += amplitudes[f_idx] * np.cos(2 * np.pi * freqs[f_idx] * t / n + phases[f_idx])
        a = np.maximum(a, 0)
    elif init_type == 5:
        # Based on parent with noise
        if parent is not None:
            from scipy.interpolate import interp1d
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            f = interp1d(t_old, parent, kind='cubic', fill_value='extrapolate')
            a = np.maximum(f(t_new) * (1 + 0.1 * rng.standard_normal(n)), 0)
        else:
            a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
            a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    return normalize(a)


def peak_targeted_restart(a, rng, strength=0.3):
    """Perturb the solution by reducing peak-contributing coordinates."""
    n = len(a)
    a_pert = a.copy()
    conv = fast_convolve(a)
    
    # Find top 3 peaks
    K = min(3, 2 * n - 1)
    peak_indices = np.argpartition(conv, -K)[-K:]
    
    for k_star in peak_indices:
        k_star = int(k_star)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        
        # Find the top contributor
        max_contrib = 0
        max_i = -1
        for i in range(i_start, i_end + 1):
            j = k_star - i
            c = a[i] * a[j]
            if c > max_contrib:
                max_contrib = c
                max_i = i
        
        if max_i >= 0:
            reduction = a[max_i] * strength
            a_pert[max_i] -= reduction
            # Add to random positions
            n_add = min(20, n)
            add_idx = rng.choice(n, n_add, replace=False)
            a_pert[add_idx] += reduction / n_add
    
    return normalize(a_pert)


def propose_candidate():
    """
    Multi-phase optimization with multi-n exploration:
    1. LP alternating on parent (n=1024)
    2. Exact CD refinement
    3. LP alternating on n=2048 (upsampled parent)
    4. CD on n=2048
    5. Mirror descent
    6. Final polish on best solution
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        n_parent = len(parent)
    else:
        n_parent = 1024
        best_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n_parent}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(candidate)}) ***")

    # === Phase 1: LP Alternating on parent ===
    print(f"\n=== Phase 1: LP Alternating n={n_parent} (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.30
    t_alloc = min(t1_end - time.monotonic(), 350.0)
    print(f"  Budget: {t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True, label="parent")
    try_update(a_lp, "lp_parent")

    # === Phase 2: Exact CD on parent ===
    print(f"\n=== Phase 2: Exact CD n={n_parent} (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20
    t_alloc = min(t2_end - time.monotonic(), 250.0)
    print(f"  Budget: {t_alloc:.0f}s")
    a_cd, s_cd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
    try_update(a_cd, "cd_main")

    # Also run CD from LP result
    if time.monotonic() < t2_end - 30:
        t_alloc2 = min(t2_end - time.monotonic(), 120.0)
        a_cd2, s_cd2 = exact_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
        try_update(a_cd2, "cd_lp")

    # === Phase 3: LP Alternating on n=2048 ===
    print(f"\n=== Phase 3: LP Alternating n=2048 (best={best_score:.8f}) ===")
    n_big = 2048
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.22
    t_alloc = min(t3_end - time.monotonic(), 280.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        # Upsample parent to n_big
        a_big_init = diverse_init(n_big, rng, 3, parent=parent)
        a_big_lp, s_big_lp = lp_alternating(a_big_init, t_alloc * 0.6, solver, verbose=True, label="big")
        try_update(a_big_lp, "lp_2048")

        # CD on n=2048
        if time.monotonic() < t3_end - 30:
            t_cd_big = min(t3_end - time.monotonic(), t_alloc * 0.4)
            if t_cd_big > 30:
                a_big_cd, s_big_cd = exact_coord_descent(a_big_lp, t_cd_big, verbose=True, label="big_cd")
                try_update(a_big_cd, "cd_2048")
    else:
        print("  Skipped (insufficient time)")

    # === Phase 4: Mirror Descent ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.10
    t_alloc = min(t4_end - time.monotonic(), 120.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mirror, s_mirror = mirror_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_mirror, "mirror")

    # === Phase 5: Peak-targeted restarts + LP ===
    print(f"\n=== Phase 5: Targeted Restarts (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + (deadline - time.monotonic()) * 0.10
    restarts = 0
    while time.monotonic() < t5_end - 10:
        restarts += 1
        strength = 0.2 + 0.1 * (restarts % 3)
        a_pert = peak_targeted_restart(best_seq, rng, strength)
        # Quick LP from perturbed start
        t_sub = min(t5_end - time.monotonic() - 5, 30.0)
        if t_sub < 10:
            break
        a_r, s_r = lp_alternating(a_pert, t_sub, solver, verbose=False, label=f"rst{restarts}")
        try_update(a_r, f"restart{restarts}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    t6_end = deadline - 30
    t_alloc = min(t6_end - time.monotonic(), 150.0)
    if t_alloc > 20:
        print(f"  Fine polish budget: {t_alloc:.0f}s")
        a_fp, s_fp = fine_polish(best_seq, t_alloc, verbose=True, label="final")
        try_update(a_fp, "fine")

    # One more CD pass on the absolute best
    if time.monotonic() < t6_end - 20:
        t_final = min(t6_end - time.monotonic() - 15, 60.0)
        if t_final > 15:
            a_fin, s_fin = exact_coord_descent(best_seq, t_final, verbose=False, label="last")
            try_update(a_fin, "last_cd")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
