"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based gradient computation for LogSumExp-smoothed objective
2. Multi-stage temperature annealing with L-BFGS (log-parameterization)
3. Quick LP-based initialization via alternating best-response
4. Coordinate descent refinement
5. Multi-n exploration with interpolation
6. Diverse restarts to escape local minima
Key innovation: O(n log n) gradient via cross-correlation, enabling
very fine-grained T annealing (25+ stages) within the time budget.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

    import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via rFFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of 2*n*max(conv(a,a)) for sum(a)=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize_simplex(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_lse_gradient(z, T):
    """Compute gradient of F_T(z) = 2n * soft_max_T(conv(softmax(z), softmax(z)))
    Returns (objective, gradient_wrt_z, a, b)."""
    n = len(z)
    # a = softmax(z)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    # b = conv(a, a)
    b = fast_convolve(a)
    # LogSumExp of b/T
    bt = T * b
    bt_max = bt.max()
    exp_bt = np.exp(bt - bt_max)
    Z = exp_bt.sum()
    obj = (2.0 * n / T) * (bt_max + np.log(Z))
    if not np.isfinite(obj):
        return obj, np.zeros(n), a, b
    # softmax weights
    w = exp_bt / Z
    # g[m] = sum_k w[k] * a[k-m] via FFT cross-correlation
    total_len = len(b) + n - 1
    fft_len = 1 << (total_len - 1).bit_length()
    W = rfft(w, fft_len)
    A = rfft(a, fft_len)
    g = irfft(W * np.conj(A), fft_len)[:n]
    # Gradient w.r.t. a: 4n * g
    grad_a = 4.0 * n * g
    # Chain rule for softmax: grad_z = a * (grad_a - <a, grad_a>)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return obj, grad_z, a, b


def lse_lbgfs_multistage(a0, T_stages, max_iter_per_stage, time_limit):
    """Multi-stage LogSumExp L-BFGS optimization on the simplex via log-param."""
    n = len(a0)
    a = normalize_simplex(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z = z - z.max()
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()

    for stage, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_iter = min(max_iter_per_stage, max(50, int(remaining * 15)))

        def obj_func(z_in):
            o, _, _, _ = compute_lse_gradient(z_in, T)
            return o if np.isfinite(o) else 1e15

        def grad_func(z_in):
            _, g, _, _ = compute_lse_gradient(z_in, T)
            if not np.all(np.isfinite(g)):
                return np.zeros(len(z_in))
            return g

        try:
            res = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': stage_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': stage_iter * 500,
                         'maxls': 50}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = local_eval(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            # Update z from new a
            a_clipped = np.maximum(a_new, 1e-300)
            z = np.log(a_clipped) - np.log(a_clipped).max()
        except Exception:
            pass

        if (stage + 1) % 5 == 0:
            print(f"    LSE stage {stage+1}/{len(T_stages)} T={T:.4f}: best={best_score:.10f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method on the exact max objective with adaptive step size."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    lr = 0.02
    vel = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    step_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        M = b.max()
        # Find active positions (within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Weighted subgradient
        w = b[active]
        w = w / w.sum()
        g = np.zeros(n)
        for idx, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_arr = np.arange(js, je + 1)
            g[j_arr] += w[idx] * 2.0 * a[k - j_arr]
        # Adam-style update
        step_count += 1
        vel = beta1 * vel + (1 - beta1) * g
        vhat = vel / (1 - beta1 ** step_count)
        a_trial = a - lr * vhat
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st
        sc = local_eval(a_trial)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        if step % 5000 == 0 and step > 0:
            lr *= 0.8

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, max_rounds=50):
    """Coordinate descent: for each coordinate, try small perturbations."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Random order
        order = rng.permutation(n)
        for m in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try a few delta values
            for scale in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                delta = scale * a[m]
                if a[m] + delta < 0 or a[m] + delta > 0.999:
                    continue
                a_trial = a.copy()
                a_trial[m] += delta
                a_trial = normalize_simplex(a_trial)
                sc = local_eval(a_trial)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
        if not improved and rnd > 5:
            break

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b>=0, sum(b)=1 minimizing max conv(a,b)."""
    if not HAS_CVXPY:
        return None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    # Try alternative solvers
    for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None


def quick_lp_alternation(a0, solver, time_limit, max_iters=200):
    """Quick alternating LP with Nesterov averaging."""
    n = len(a0)
    a = normalize_simplex(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Best response for b given y_a
        b_new = lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1 - alpha) * y_b + alpha * b_new
            y_b = normalize_simplex(y_b)
            b = ((it + 1) * b + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize_simplex(b)
        # Best response for a given y_b
        a_new = lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1 - alpha) * y_a + alpha * a_new
            y_a = normalize_simplex(y_a)
            a = ((it + 1) * a + alpha * (it + 3) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize_simplex(a)

        sc_a = local_eval(a)
        sc_b = local_eval(b)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 50 == 0:
            print(f"    LP iter {it+1}: best={best_score:.10f} ({time.monotonic()-t0:.1f}s)")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

    elif choice == 3:

        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-30 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:

    else:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

    print(f"\n=== Diverse restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        for restart in range(3):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_r = min(rem * 0.3, 80.0)
            # Perturb the best sequence
            noise = rng.standard_normal(n_best) * 0.05
            a_pert = best_seq + noise
            a_pert = normalize_simplex(a_pert)
            T_stages_r = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_r, _ = lse_lbgfs_multistage(a_pert, T_stages_r, max_iter_per_stage=150, time_limit=t_r)
            sc_r = local_eval(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.10f}")

    # === Final polish ===
    print(f"\n=== Final polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = rem * 0.7
        T_stages_fin = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        a_fin, _ = lse_lbgfs_multistage(best_seq, T_stages_fin, max_iter_per_stage=200, time_limit=t_fin)
        sc_fin = local_eval(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()

        # Final subgradient
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fs, _ = subgradient_refine(best_seq, rem * 0.5, n_steps=10000)
            sc_fs = local_eval(a_fs)
            if sc_fs < best_score:
                best_score = sc_fs
                best_seq = a_fs.copy()

    # Final verification
    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    try:
        eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
