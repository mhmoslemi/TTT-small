def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:

    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:

def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,  # T @ b has shape (2n-1,), M_var is scalar, so M_var broadcasts
        cp.sum(b_var) == 1
    ]
    # Wait, T @ b_var <= M_var means each row of T @ b_var <= M_var
    # That's exactly what we want: conv(a,b)[k] <= M for all k
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    return b_val, float(M_var.value)

def exact_coord_descent(a0, time_limit, verbose=True, max_sweeps=1000000):

    n = len(a0)

    M = float(conv.max())

        sweeps += 1
        improved = False

            if a[i] < 1e-18:

            if k_start > 0:

            a_sh = a[k_start - i:k_end - i + 1]
            quad_idx = i  # where k = 2i in the affected array

            def g_and_sg(delta):
                vals = affected + 2.0 * delta * a_sh
                vals[quad_idx] += delta * delta
                g = max(U, float(vals.max()))
                
                tol = 1e-11 * max(1.0, abs(g))
                sg_lo = np.inf
                sg_hi = -np.inf
                
                if U >= g - tol and U > 0:
                    sg_lo = min(sg_lo, 0.0)
                    sg_hi = max(sg_hi, 0.0)

                            # Find position in active slopes
                            pass

                        slope = 2.0 * a_sh[j]
                        if j == quad_idx:
                            slope += 2.0 * delta
                        sg_lo = min(sg_lo, slope)
                        sg_hi = max(sg_hi, slope)
                
                if np.isinf(sg_lo):
                    sg_lo, sg_hi = 0.0, 0.0
                
                return g, sg_lo, sg_hi

                    break
                hi *= 2.0
                if hi > 10.0:
                    break

            for _ in range(50):

                g_mid, sg_lo, sg_hi = g_and_sg(mid)

                    lo = hi = mid
                    break
                if sg_lo > 1e-9:
                    hi = mid
                else:
                    lo = mid

            best_delta = max(best_delta, -a[i])

                a[i] = old_val + best_delta
                # Update convolution
                changes = 2.0 * best_delta * a_sh
                changes[quad_idx] = 2.0 * best_delta * (old_val - best_delta) + best_delta * best_delta
                # Wait, let me redo the quadratic term
                # old conv[2i] = ... + a[i]^2 + ...
                # new conv[2i] = ... + (a[i]+delta)^2 + ...
                # change = (a[i]+delta)^2 - a[i]^2 = 2*a[i]*delta + delta^2

                # The linear part gives 2*delta*a_sh[quad_idx] = 2*delta*a[i] (since a_sh[quad_idx] = a[2i-i] = a[i])
                # The quadratic part adds delta^2
                # So total = 2*delta*a[i] + delta^2
                # But we already set changes = 2*delta*a_sh, so changes[quad_idx] = 2*delta*a[i]
                # We need to add delta^2: changes[quad_idx] += delta^2
                # But wait, changes is computed from old a_sh which uses old a[i]

                # where a_sh is computed from OLD a (before update)
                # This is correct because:
                # conv_new[k] = sum_j a_new[j]*a_new[k-j]
                # = sum_j (a[j] + delta*[j==i]) * (a[k-j] + delta*[k-j==i])
                # = conv_old[k] + delta * a[k-i] + delta * a[i] * [k-i valid] + delta^2 * [k==2i]

                #   = conv_old[k] + 2*delta*a[k-i] + delta^2*[k==2i]
                #   (for valid k-i, i.e., 0 <= k-i < n)

                # k-i ranges from 0 to n-1, all valid.
                # So: changes[j] = 2*delta*a_sh[j] + delta^2*(j==quad_idx)
                # where j indexes the affected array.
                
                # I already have changes = 2.0 * best_delta * a_sh
                # I need to add delta^2 at quad_idx:
                changes = changes.copy()
                changes[quad_idx] += best_delta * best_delta
                
                conv[k_start:k_end + 1] += changes
                M = max(U, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:

            print(f"    ExactCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f} M={M:.12f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, verbose=True, T_init=50.0, T_end=200.0):
    """Mirror descent with smooth max approximation (softmax)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        total_time = time_limit
        
        # Adaptive temperature
        progress = min(1.0, elapsed / total_time)
        T = T_init + (T_end - T_init) * progress
        lr = 0.5 / (T + 1.0)
        
        # Compute smooth approximation of max
        conv = fast_convolve(a)
        M = float(conv.max())
        
        # Softmax weights (normalized to sum to 1)
        shifted = conv - M
        weights = np.exp(shifted / (0.01 + 1.0/T))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.ones_like(conv) / len(conv)
        else:
            w = weights / w_sum
        
        # Gradient of sum_k w[k]*conv[k] w.r.t. a[j]
        # conv[k] = sum_i a[i]*a[k-i]
        # d/d a[j] = sum_k w[k] * (a[k-j] + a[j-(k-j)] * [k-j != j])
        # = sum_k w[k] * a[k-j] + ... (for j != k-j)
        # Actually: d conv[k] / d a[j] = a[k-j] + (a[j-(k-j)] if j != k-j else a[k-j])
        # = a[k-j] + a[2j-k] * [j != k-j] (when 2j-k is valid)
        # Hmm, let me think more carefully.
        # conv[k] = sum_i a[i] * a[k-i]
        # d/d a[j] conv[k] = a[k-j] (from the i=j term) + a[2j-k] (from the k-i=j term, i.e., i=k-j)
        # but only if these indices are valid and j != k-j
        # If j = k-j (i.e., k = 2j), then it's 2*a[j]
        
        # So: grad[j] = sum_k w[k] * (a[k-j] * [0<=k-j<n] + a[2j-k] * [0<=2j-k<n and 2j-k != k-j])
        # Simplifying: grad[j] = sum_k w[k] * a[k-j] * [0<=k-j<n] + sum_k w[k] * a[2j-k] * [0<=2j-k<n]
        # Wait that double counts when k-j = 2j-k, i.e., k=2j.
        # Actually the correct formula is:
        # grad[j] = sum_k w[k] * (a[k-j] + a[2j-k]) where we only include valid terms
        # But when k=2j, both terms are a[j], so it's 2*w[2j]*a[j]
        # When k!=2j, a[k-j] and a[2j-k] are different terms.
        
        # Alternative: just compute it directly for efficiency.
        # grad[j] = sum over k of w[k] * a[k-j] + sum over k of w[k] * a[2j-k]
        # First sum: for fixed j, k-j ranges over valid values. Let m = k-j, then k = j+m.
        # sum_m w[j+m] * a[m] for valid m and j+m
        # Second sum: for fixed j, 2j-k = m means k = 2j-m.
        # sum_m w[2j-m] * a[m] for valid m and 2j-m
        
        # This can be computed as:
        # grad[j] = (w * a)[j] reversed-ish... actually it's a convolution-like operation.
        # grad = convolve(w, a) + convolve(w_reversed, a_reversed) ... no, this is getting complicated.
        
        # Let me just do it naively for now (n=1024 is small enough)
        grad = np.zeros(n)
        for j in range(n):
            # Sum over k where 0 <= k-j < n (i.e., j <= k <= j+n-1)
            k_min = j
            k_max = min(j + n - 1, 2*n - 2)
            if k_max >= k_min:
                grad[j] += np.dot(w[k_min:k_max+1], a[k_min-j:k_max-j+1])
            # Sum over k where 0 <= 2j-k < n (i.e., 2j-n+1 <= k <= 2j)
            k_min2 = max(0, 2*j - n + 1)
            k_max2 = min(2*j, 2*n - 2)
            if k_max2 >= k_min2:
                grad[j] += np.dot(w[k_min2:k_max2+1], a[2*j-k_max2:2*j-k_min2+1][::-1])
        
        # Mirror descent: a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad))
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        # Normalize
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        a = a_new
        conv = fast_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            print(f"    MirrorDescent it={it} ({elapsed:.0f}s): best={best_score:.10f} T={T:.1f}")
    
    return best_a, best_score

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

    """Solve 1D subproblem for coordinate i. Returns optimal delta."""
    n = len(a)

                if not any_active:
                    sg_lo = sl
                    sg_hi = sl
                    any_active = True
                else:
                    sg_lo = min(sg_lo, sl)
                    sg_hi = max(sg_hi, sl)

        return 0.0
    
    # Bracket

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response alternating with adaptive damping

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv)."""

def normalize_vec(a):

        return np.ones(len(a)) / len(a)
    return a / s


def build_T(a):
    """Build convolution matrix T such that T@b = conv(a,b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_solver():
    cached = getattr(get_solver, '_cached', None)
    if cached is not None:
        return cached
    for name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_solver._cached = name
                return name
        except Exception:
            continue
    get_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    """LP: min M s.t. conv(a,b)<=M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [T @ b <= M, cp.sum(b) == 1])
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    return b_val / s, float(M.value)


def subgrad_interval(delta, affected, a_sh, qi, U, length):
    """Compute g(delta) and subgradient interval [sg_lo, sg_hi]."""
    vals = affected + 2.0 * delta * a_sh
    vals = vals.copy()
    vals[qi] += delta * delta
    g = max(U, float(vals.max()))
    tol = 1e-10 * max(1.0, abs(g))
    sg_lo = 0.0
    sg_hi = 0.0
    any_act = False
    if U >= g - tol and U > 1e-30:
        sg_lo = 0.0
        sg_hi = 0.0
        any_act = True
    for j in range(length):
        if vals[j] >= g - tol:
            sl = 2.0 * a_sh[j]
            if j == qi:
                sl += 2.0 * delta
            if not any_act:
                sg_lo = sl
                sg_hi = sl
                any_act = True
            else:
                if sl < sg_lo:
                    sg_lo = sl
                if sl > sg_hi:
                    sg_hi = sl
    if not any_act:
        sg_lo = 0.0
        sg_hi = 0.0
    return g, sg_lo, sg_hi


def coord_solve(i, a, conv):
    """Solve 1D subproblem for coordinate i via subgradient bisection. Returns delta."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    length = k_end - k_start + 1
    U = 0.0
    if k_start > 0:
        U = max(U, float(conv[:k_start].max()))
    if k_end < 2 * n - 2:
        U = max(U, float(conv[k_end + 1:].max()))
    affected = conv[k_start:k_end + 1]
    a_sh = a[k_start - i:k_end - i + 1]
    qi = i
    # Check delta=0
    g0, sgl0, sgh0 = subgrad_interval(0.0, affected, a_sh, qi, U, length)
    if sgl0 <= 1e-7 and sgh0 >= -1e-7:
        return 0.0
    # Bracket
    lo = -a[i]
    hi = max(2.0 * a[i], 0.05)
    for _ in range(20):
        _, sgl_h, _ = subgrad_interval(hi, affected, a_sh, qi, U, length)
        if sgl_h >= 0:
            break
        hi *= 2.0
    # Bisection
    for _ in range(45):
        mid = (lo + hi) / 2.0
        _, sgl, sgh = subgrad_interval(mid, affected, a_sh, qi, U, length)
        if sgl <= 1e-7 and sgh >= -1e-7:
            break
        if sgl > 1e-7:
            hi = mid
        else:
            lo = mid
    delta = (lo + hi) / 2.0
    if delta < -a[i]:
        delta = -a[i]
    return delta


def exact_cd(a0, time_limit, verbose=True):
    """Exact 1D coordinate descent using subgradient bisection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize_vec(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    best = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    stall = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for i in range(n):
            if a[i] < 1e-18:
                continue
            delta = coord_solve(i, a, conv)
            if abs(delta) > 1e-14:
                k_s = i
                k_e = min(i + n - 1, 2 * n - 2)
                a_sh = a[k_s - i:k_e - i + 1].copy()
                qi = i
                ch = 2.0 * delta * a_sh
                ch[qi] += delta * delta
                conv[k_s:k_e + 1] += ch
                U = 0.0
                if k_s > 0:
                    U = max(U, float(conv[:k_s].max()))
                if k_e < 2 * n - 2:
                    U = max(U, float(conv[k_e + 1:].max()))
                M = max(U, float(conv[k_s:k_e + 1].max()))
                a[i] += delta
                cur = 2.0 * n * M
                if cur < best:
                    best = cur
                    best_a = a.copy()
                improved = True
        if not improved:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
        if verbose and sweep % 50 == 0:
            print(f"    ECD sweep={sweep} ({time.monotonic()-t0:.0f}s): best={best:.10f}")
    return best_a, best


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP best-response alternating with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize_vec(a)
    best_a = a.copy()
    best = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3.0:
            break
        b, Mlp = lp_best_response(a, solver)
        if not np.isfinite(Mlp):
            break
        sb = 2.0 * n * fast_convolve(b).max()
        if sb < best:
            best = sb
            best_a = b.copy()
        prog = (time.monotonic() - t0) / time_limit
        if it <= 5:
            alpha = 1.0
        elif prog < 0.25:
            alpha = 0.75
        elif prog < 0.5:
            alpha = 0.5
        elif prog < 0.8:
            alpha = 0.3
        else:
            alpha = max(0.15, 0.3 - 0.4 * prog)
        a = (1.0 - alpha) * a + alpha * b
        a = np.maximum(a, 0.0)
        a = normalize_vec(a)
        sa = 2.0 * n * fast_convolve(a).max()
        if sa < best:
            best = sa
            best_a = a.copy()
        if verbose and it % 10 == 0:
            print(f"    LP it={it} ({time.monotonic()-t0:.0f}s): best={best:.10f} alpha={alpha:.3f}")
    return best_a, best


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with softmax-smoothed max."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize_vec(a)
    best_a = a.copy()
    best = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        prog = min(1.0, (time.monotonic() - t0) / time_limit)
        T = 20.0 + 300.0 * prog * prog
        lr = 1.0 / (T + 1.0)
        conv = fast_convolve(a)
        M = float(conv.max())
        temp = max(0.005, 1.0 / T)
        w = np.exp((conv - M) / temp)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / ws
        grad = np.zeros(n)
        for j in range(n):
            k1 = j
            k2 = min(j + n - 1, 2 * n - 2)
            if k2 >= k1:
                grad[j] += float(np.dot(w[k1:k2 + 1], a[0:k2 - k1 + 1]))
            k1b = max(0, 2 * j - n + 1)
            k2b = min(2 * j, 2 * n - 2)
            if k2b >= k1b:
                grad[j] += float(np.dot(w[k1b:k2b + 1], a[2 * j - k2b:2 * j - k1b + 1][::-1]))
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()
        cur = 2.0 * n * fast_convolve(a).max()
        if cur < best:
            best = cur
            best_a = a.copy()
        if verbose and it % 100 == 0:
            print(f"    Mirror it={it} ({time.monotonic()-t0:.0f}s): best={best:.10f} T={T:.1f}")
    return best_a, best


def subgrad_ascent(a0, time_limit, verbose=True):
    """Subgradient method with multi-peak softmax weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize_vec(a)
    best_a = a.copy()
    best = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.05
    top_k = 30
    no_imp = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        prog = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = max(0.001, 0.05 * (1.0 - prog))
        conv = fast_convolve(a)
        M = float(conv.max())
        cur = 2.0 * n * M
        if cur < best:
            best = cur
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 500:
            lr *= 0.8
            no_imp = 0
            if lr < 1e-6:
                break
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.002 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            grad[i_s:i_e + 1] += w[idx] * 2.0 * a[k - i_e:k - i_s + 1][::-1]
        a = a - lr * grad
        # Project onto simplex
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u)
        rho = np.arange(1, n + 1)
        cond = u * rho > (cssv - 1)
        if cond.any():
            kk = int(np.nonzero(cond)[0][-1])
            theta = (cssv[kk] - 1.0) / (kk + 1.0)
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.ones(n) / n
        if verbose and it % 200 == 0:
            print(f"    SubGrad it={it} ({time.monotonic()-t0:.0f}s): best={best:.10f} lr={lr:.6f}")
    return best_a, best


def diverse_init(n, rng, typ):
    t = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    if typ == 0:
        a = np.maximum(1.0 - np.abs(t - c) / max(c, 1e-6), 0)
    elif typ == 1:
        x = (t - c) / max(c, 1e-6)
        a = 0.2 + 0.8 * x * x
    elif typ == 2:
        x = (t - c) / max(c, 1e-6)
        a = np.power(np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 1e-10), 1.5)
    elif typ == 3:
        a = np.exp(-((t - 0.3 * n) / (0.12 * n))**2) + np.exp(-((t - 0.7 * n) / (0.12 * n))**2)
    elif typ == 4:
        period = int(rng.integers(3, 8))
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * t / period)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
