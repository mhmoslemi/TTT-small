def fast_convolve(a):

def compute_score(a):

def build_conv_matrix(a):

    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

        # Actually this is wrong. Let's do:
        # We want |A_new|^2 = C_modified, phase(A_new) = phase(A)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue

    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(hash(label) % (2**31))

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(5.0 * v, 10.0 / n)

                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        conv = fast_convolve(a)
        max_val = float(conv.max())
        peak_k = int(np.argmax(conv))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_end < i_start:
            break
        contributors = np.arange(i_start, i_end + 1)
        contrib_vals = np.array([a[i] * a[peak_k - i] if 0 <= peak_k - i < n else 0 for i in contributors])
        top_idx = np.argsort(contrib_vals)[::-1][:min(8, len(contributors))]
        improved_this_iter = False
        for ti in top_idx:
            i = contributors[ti]
            partner = peak_k - i
            if partner < 0 or partner >= n:
                continue
            if a[i] < 1e-14:
                continue
            base_val = a[i]
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80, 0.75]:
                a[i] = base_val * factor
                ns = a.sum()
                if ns < 0.01:
                    a[i] = base_val
                    break
                sc = compute_score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                    improved_this_iter = True
                else:
                    a[i] = base_val
                    break
        if not improved_this_iter:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0
        if iters > 100:
            break
        if verbose and iters % 10 == 0:
            print(f"    PS it={iters}: best={best_score:.10f}")
    return best_a, best_score

def basin_hopping_refined(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = 0.015
    T_min = 0.001
    cooling = 0.997
    count = 0
    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 60.0)
        r = rng.random()
        perturbed = a.copy()
        if r < 0.25:
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            n_red = max(3, (i_end - i_start + 1) // 5)
            contrib = rng.choice(np.arange(i_start, i_end + 1),
                                 size=min(n_red, i_end - i_start + 1), replace=False)
            for ii in contrib:
                perturbed[ii] *= (1.0 - T * rng.random() * 4.0)
        elif r < 0.5:
            perturbed = a + rng.standard_normal(n) * T
        elif r < 0.7:
            i, j = rng.choice(n, size=2, replace=False)
            perturbed[i], perturbed[j] = a[j], a[i]
        elif r < 0.85:
            blen = rng.integers(3, max(4, n // 16))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        else:
            shift = rng.integers(1, max(2, n // 8))
            shifted = np.roll(a, shift)
            perturbed = (1.0 - 3.0*T) * a + 3.0*T * shifted
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            T = max(T * cooling, T_min)
            continue
        perturbed = perturbed / ps
        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{count}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
        T = max(T * cooling, T_min)
        if verbose and count % 5 == 0:
            print(f"    BH {count}: best={best_score:.10f}, T={T:.5f}")
    return best_a, best_score

def pairwise_coord_opt(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        pairs = []
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(key=lambda x: -x[0])
        improved = False
        for contrib, i, j in pairs[:8]:
            if rem < 2:
                break
            if a[i] < 1e-14:
                continue
            vi = a[i]
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80, 0.75, 0.70]:
                a[i] = vi * factor
                ns = a.sum()
                if ns < 0.01:
                    a[i] = vi
                    break
                sc = compute_score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = vi
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 15:
                break
        else:
            no_improve = 0
        if verbose and iters % 10 == 0:
            print(f"    PCO {label} it={iters}: best={best_score:.10f}")
    return best_a, best_score

def spectral_refinement(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    while time.monotonic() - t0 < time_limit:
        iters += 1
        if iters > 300:
            break
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        C = mag * mag
        c = irfft(C, fft_size)[:2*n-1]
        max_c = float(c.max())
        if max_c < 1e-15:
            break
        decay = np.exp(-0.05 * (c / max_c) ** 2)
        new_mag = mag * np.sqrt(decay)
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ns = a_new.sum()
        if ns < 1e-10:
            break
        a_new = a_new / ns
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        elif verbose and iters % 50 == 0:
            print(f"    SR it={iters}: best={best_score:.10f}")
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
