In the frequency domain, `fft(a)` has `n` "free" parameters (the rest are determined by `a` being real, but we're zero-padding to `2n-1` or `2n` for the FFT).

Actually, let's use FFT size `N >= 2n-1`. Then:
- `A = fft(a, N)` (zero-padded)
- `C = ifft(|A|^2, N)` gives the circular auto-correlation
- For `N >= 2n-1`, the first `2n-1` elements of `C` equal the linear auto-correlation

We want to minimize `max(C[0:2n-1])`.

Now, `|A|^2` is a non-negative vector in frequency domain. The constraint is that its inverse FFT (first `n` elements) is non-negative and sums to 1.

This is like designing a "power spectrum" `|A|^2` that has a small-peak inverse FFT, subject to constraints.

**Practical algorithm: Iterative spectral peak reduction**

1. Initialize `a` (e.g., from the parent sequence)
2. Compute `C = conv(a,a)`
3. Find the peak position `k*`
4. Construct a "mask" in frequency domain that reduces the peak at `k*`
   - The peak at lag `k*` corresponds to a sinusoidal component `exp(i*2π*f*k*/N)` in frequency
   - We can "notch" this frequency component
5. Update `a` by inverse FFT of the modified spectrum
6. Project back to non-negative (clip negatives to 0) and renormalize
7. Iterate

This is a form of "frequency domain equalization."

**Even better: Use the "cycling projections" method.**

We want to find `a` in the intersection of:

For set C, we can approximate it as a union of halfspaces (for small increments of M) or use the subgradient.

import numpy as np
from scipy.fft import rfft, irfft
import time

# Helper functions (all top-level)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_best_response(a, solver):
    """Solve LP for best response."""
    import cvxpy as cp
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [T @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)

    if s < 1e-15:

        score_b = 2.0 * n * fast_convolve(b_new).max()

def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact 1D minimization via ternary search.
    For each coordinate i, find the value that minimizes max(conv) exactly.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()

        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    # Smart ordering: coordinates that contribute most to peaks go first
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        # Determine coordinate order based on peak contributions
        peak_idx = int(np.argmax(conv))
        order = list(range(n))
        # Sort by contribution to the top peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = peak_idx - i
            contrib[i] = a[i] * a[j]
        # Put high-contribution coords first
        order_key = np.zeros(n)
        order_key[i_start:i_end+1] = contrib[i_start:i_end+1]
        order = sorted(range(n), key=lambda x: -order_key[x])
        
        for i in order:
            if a[i] < 1e-18:
                # Try a small positive value
                candidates = [1e-6, 1e-5, 1e-4]
            else:
                # Ternary search on [0, 2*a[i]] for the best value of a[i]
                lo = 0.0
                hi = 2.0 * a[i]
                # Evaluate the objective as a function of a[i]
                # f(x) = max(unaff_max, max_k (conv_affected_k + 2*(x-a[i])*a_shifted_k + (x-a[i])^2 * [k=2i-i]))
                # Actually, let's parameterize: new_a[i] = x, delta = x - a[i]
                # new conv at k: conv[k] + 2*delta*a[k-i] + delta^2 * [k == 2i]
                # (where a[k-i] is the OLD value, and [k==2i] means the quadratic term)
                
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                
                if k_start > 0:
                    unaff_left = conv[:k_start].max()
                else:
                    unaff_left = -1e30
                if k_end < 2 * n - 2:
                    unaff_right = conv[k_end + 1:].max()
                else:
                    unaff_right = -1e30
                unaff_max = max(unaff_left, unaff_right)
                
                a_shifted = a[k_start - i:k_end - i + 1].copy()
                conv_affected = conv[k_start:k_end + 1].copy()
                has_quad = (2 * i >= k_start) and (2 * i <= k_end)
                quad_idx = 2 * i - k_start if has_quad else -1
                
                def f_of_x(x):
                    """Max convolution value when a[i] = x (others fixed)."""
                    delta = x - a[i]
                    new_aff = conv_affected + 2.0 * delta * a_shifted
                    if has_quad:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] = conv_affected[quad_idx] + 2.0 * delta * a_shifted[quad_idx] + delta * delta
                    return max(unaff_max, float(new_aff.max()))
                
                # Ternary search
                for _ in range(30):  # ~30 iterations gives precision ~ (2/3)^30 * range
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = f_of_x(m1)
                    f2 = f_of_x(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                best_x = (lo + hi) / 2.0
                best_x = max(0.0, best_x)
                new_M = f_of_x(best_x)
                
                if new_M < M - 1e-15:
                    delta = best_x - a[i]
                    a[i] = best_x
                    changes = 2.0 * delta * a_shifted
                    if has_quad:
                        changes = changes.copy()
                        changes[quad_idx] = 2.0 * delta * a[i] + delta * delta  # Note: a[i] is now updated
                        # Actually need to be careful: the quadratic term is delta^2

                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            print(f"  ECD[{label}] sweep={sweeps}: best={best_score:.8f}")
    return best_a, best_score

def subgradient_descent_v2(a0, time_limit, lr_init=0.01, top_k=30, verbose=True):
    """Enhanced subgradient descent with adaptive temperature and line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_conv = conv.copy()  # Running average for smoothing
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-12:
                break
        
        # Use smoothed convolution for direction
        avg_conv = 0.9 * avg_conv + 0.1 * conv
        M_sm = avg_conv.max()
        
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(avg_conv, -k_act)[-k_act:]
        pv = avg_conv[peaks]
        
        # Adaptive temperature
        temp = 0.005 * max(M_sm, 1e-10) + 0.001 * (it / 1000.0)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Step with projection
        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            print(f"  SGD2 it={it}: best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score
