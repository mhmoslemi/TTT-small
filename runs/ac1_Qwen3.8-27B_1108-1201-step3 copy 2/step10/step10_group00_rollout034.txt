def active_set_lp(a, eps=1e-3):
    """Find a perturbation direction using active set LP."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    S = np.where(b >= max_b * (1 - eps))[0]
    
    # We want to find delta that minimizes max_{k in S} linearized_b_k
    # linearized_b_k = b_k + 2 * sum_{i+j=k} a_i * delta_j
    # Subject to: sum(delta) = 0, delta >= -a (to keep a+delta >= 0), ||delta||_inf <= step_size
    # 
    # This is: min t s.t. 2 * sum_{i+j=k} a_i * delta_j - t <= -b_k for k in S
    #                       sum(delta) = 0
    #                       -a <= delta <= step_size (box constraint)
    
    # This is a small LP with n+1 variables and |S|+1 constraints
    # For efficiency, we can use scipy's linprog or just do a simple projected gradient
    
    # Actually, let's use a simpler approach: projected subgradient
    # The subgradient of the linearized objective w.r.t. delta_j is:
    # g_j = (2/|S|) * sum_{k in S, i: i+j=k} a_i
    #     = (2/|S|) * sum_{k in S} a_{k-j} (for valid k-j)
    
    g = np.zeros(n)
    for k in S:
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                g[j] += 2 * a[i]
    g /= len(S)
    
    # Projected gradient step
    step = 0.01  # small step
    delta = -step * g
    # Project onto {sum=0, -a <= delta <= 0.01}
    delta -= delta.mean()  # ensure sum = 0
    delta = np.clip(delta, -a, 0.01)
    delta -= delta.mean()  # re-project sum = 0
    
    a_new = a + delta
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Weighted subgradient (variance-based) for fast coarse descent
2. Fine LogSumExp L-BFGS with aggressive temperature annealing (down to T=0.001)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def logsumexp_obj_grad(z, n, T):
    """LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    iters_per_stage = max(30, maxiter_per_stage // len(T_stages))
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Scale iterations based on remaining time
        stage_iter = iters_per_stage
        if remaining < time_limit * 0.3:
            stage_iter = min(stage_iter, iters_per_stage * 2)
        
        def func(z_inner):
            obj, _ = logsumexp_obj_grad(z_inner, n, stage_T)
            return obj
        
        def jac(z_inner):
            _, g = logsumexp_obj_grad(z_inner, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': stage_iter, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': stage_iter * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def active_set_refine(a0, time_limit, n_iters=2000):
    """Active-set refinement: identify top peaks and do targeted optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    t0 = time.monotonic()
    
    step_size = 0.01

        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient from active set
        g = np.zeros(n)
        for k in active:

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g *= 2.0 / len(active)
        
        # Line search
        best_alpha = step_size
        best_trial_score = best_score
        for alpha_mult in [1.0, 0.5, 0.25, 0.1]:
            alpha = step_size * alpha_mult
            a_trial = a - alpha * g
            a_trial = np.maximum(a_trial, 0)

def spectral_refine(a0, time_limit, n_iters=500):
    """Spectral domain equalization: flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    beta = 0.1  # spectral equalization strength
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute circular convolution (via FFT)
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2  # power spectrum
        
        # Equalize: reduce dynamic range
        P_mean = P[P > 1e-20].mean()
        P_clipped = np.maximum(P, 1e-20)
        P_new = P_clipped * (P_mean / P_clipped) ** beta
        P_new = P_new / P_new.sum() * P.sum()  # preserve total power
        
        # Reconstruct: use original phase with new magnitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        
        # For real signal, need to enforce Hermitian symmetry
        # rfft gives us the non-negative frequency part
        # We need to be careful here
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        if it % 100 == 0:

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.3 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-3 * (x - 0.5) ** 2)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured noise."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    # Add some random noise too
    perturbation += strength * 0.3 * rng.standard_normal(n)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def spectral_refine(a0, time_limit, n_iters=500):
    """Spectral domain equalization: reduce dynamic range of power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    beta = 0.05
    blend = 0.3

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute circular convolution via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_clipped = np.maximum(P, 1e-20)
        P_mean = P_clipped.mean()

        # Equalize power spectrum
        P_new = P_clipped * (P_mean / P_clipped) ** beta
        P_new /= P_new.sum()
        P_new *= P_clipped.sum()

        # Blend with original to maintain validity
        P_blend = (1 - blend) * P_clipped + blend * P_new
        mag_new = np.sqrt(P_blend)

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        # Blend with current to avoid drastic changes
        a_candidate = (1 - blend * 0.5) * a + blend * 0.5 * a_new
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc > 1e-15:
            a_candidate = a_candidate / sc
            score = 2.0 * n * fast_convolve(a_candidate).max()
            if score < best_score - 1e-13:
                best_score = score
                best_a = a_candidate.copy()
                a = a_candidate

        # Gradually increase alpha
        if it % 50 == 0:
            beta = min(beta * 1.15, 0.4)
            blend = min(blend * 1.1, 0.7)

    return best_a, best_score

        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.3 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-3 * (x - 0.5) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create perturbed version with structured + random noise."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    perturbation += strength * 0.2 * rng.standard_normal(n)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def phase2_combo(a, rem_time):
    """Combine focused subgradient + L-BFGS for a trial."""
    t1 = min(rem_time * 0.35, 50.0)
    a1, _ = focused_subgradient(a, t1, n_steps=15000, top_k=5)
    rem2 = deadline_global - time.monotonic()
    if rem2 < 10:
        return a1
    t2 = min(rem2 * 0.6, 60.0)
    if t2 < 5:
        return a1
    T_quick = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    return lbgfs_refine(a1, t2, T_quick, maxiter_per_stage=6000)


# Global deadline for phase2_combo
deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses focused subgradient with line search, ultra-fine L-BFGS,
    spectral equalization, active-set polishing, and basin hopping.
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    deadline_global = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Focused subgradient (coarse descent) ===
    print(f"\n=== Phase 1: Focused Subgradient (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 10:
        a_fs, _ = focused_subgradient(best_seq, t_alloc, n_steps=30000, top_k=5)
        local_update(a_fs, "focused_sg")

    # === Phase 2: Top-k weighted subgradient ===
    print(f"\n=== Phase 2: Weighted SubGrad (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_w, _ = topk_weighted_subgradient(best_seq, t_alloc, n_steps=20000, top_k=10)
        local_update(a_w, "weighted_sg")

    # === Phase 3: Fine L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 3: Fine L-BFGS (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=18000)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Spectral equalization ===
    print(f"\n=== Phase 4: Spectral Refine (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_sp, _ = spectral_refine(best_seq, t_alloc, n_iters=800)
        local_update(a_sp, "spectral")
        # Follow up with L-BFGS if spectral helped
        rem = deadline_global - time.monotonic()
        if best_score < float('inf') and rem > 30:
            T_sp = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_sp2 = lbgfs_refine(best_seq, min(rem * 0.4, 60.0), T_sp, maxiter_per_stage=5000)
            local_update(a_sp2, "spectral_lbgfs")

    # === Phase 5: Active-set polishing ===
    print(f"\n=== Phase 5: Active Set Polish (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_as, _ = active_set_polish(best_seq, t_alloc, n_iters=8000)
        local_update(a_as, "active_set")

    # === Phase 6: Basin hopping (multi-start) ===
    print(f"\n=== Phase 6: Basin Hopping (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    if rem > 120:
        n_trials = 6
        t_per_trial = min(rem / (n_trials + 2), 100.0)
        for trial in range(n_trials):
            rem_sub = deadline_global - time.monotonic()
            if rem_sub < 25:
                break
            strengths = [0.008, 0.012, 0.018, 0.025, 0.035, 0.05]
            strength = strengths[trial % len(strengths)]
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)

            deadline_sub = time.monotonic() + min(rem_sub * 0.7, t_per_trial)
            global deadline_global
            old_deadline = deadline_global
            deadline_global = deadline_sub

            a_opt = phase2_combo(a_pert, t_per_trial)

            deadline_global = old_deadline
            local_update(a_opt, f"basin_{trial}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    n_cur = len(best_seq)
    # Try a few nearby lengths
    candidates = []
    for delta in [200, 400, -200, 600]:
        target_n = n_cur + delta
        if 600 <= target_n <= 3000:
            candidates.append(target_n)
    candidates = list(set(candidates))[:3]

    for target_n in candidates:
        rem_sub = deadline_global - time.monotonic()
        if rem_sub < 80:
            break
        t_alloc_n = min(rem_sub * 0.3, 150.0)
        print(f"  Trying n={target_n}, budget={t_alloc_n:.0f}s")
        init_n = interpolate_sequence(best_seq, target_n)

        # Quick optimization at new length
        a_fs_n, _ = focused_subgradient(init_n, t_alloc_n * 0.35, n_steps=8000, top_k=5)
        rem_sub2 = deadline_global - time.monotonic()
        if rem_sub2 > 30:
            T_quick_n = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
            a_lb_n = lbgfs_refine(a_fs_n, min(rem_sub2 * 0.5, 80.0), T_quick_n, maxiter_per_stage=5000)
            local_update(a_lb_n, f"len_{target_n}")

    # === Phase 8: Final ultra-fine L-BFGS ===
    print(f"\n=== Phase 8: Final Ultra-Fine L-BFGS (score={best_score:.8f}) ===")
    rem = deadline_global - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 180.0)
        T_final = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        a_fin = lbgfs_refine(best_seq, t_fin, T_final, maxiter_per_stage=12000)
        local_update(a_fin, "final_lbgfs")

    # === Phase 9: Final active-set polish ===
    rem = deadline_global - time.monotonic()
    if rem > 20:
        a_as_fin, _ = active_set_polish(best_seq, min(rem * 0.4, 50.0), n_iters=3000)
        local_update(a_as_fin, "final_active")

    # === Phase 10: Ultra-fine focused subgradient ===
    rem = deadline_global - time.monotonic()
    if rem > 15:
        a_fs_fin, _ = focused_subgradient(best_seq, min(rem * 0.5, 40.0), n_steps=5000, top_k=3)
        local_update(a_fs_fin, "final_focused")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        # Verify with evaluate_sequence
        try:
            test_result = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"  evaluate_sequence confirms: {test_result:.8f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
