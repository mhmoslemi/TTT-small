    """FFT-based full convolution of a with itself."""
    n = len(a)

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    return irfft(A * B, fsize)[:size]


def build_T(a):
    """Build convolution matrix: T[k,j] = a[k-j] if valid, else 0.
    Shape: (2n-1, n)"""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_vec(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def log_vec(a):
    return np.log(np.maximum(a, 1e-300))


def interp_to_len(f, new_n):
    """Linearly interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

_lp_solver_name = None

def get_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = name
                return name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_br(a_fix, solver):
    """Solve LP: min M s.t. T(a_fix)@b <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None)."""
    n = len(a_fix)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T(a_fix)
    cons = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), cons)
    status = None
    try:
        prob.solve(solver=solver)
        status = prob.status
    except Exception:
        status = 'error'
    if status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    status = prob.status
                    break
            except Exception:
                continue
    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    bv = np.asarray(b_var.value, dtype=np.float64)
    bv = np.maximum(bv, 0)
    s = bv.sum()
    if s < 1e-15:
        return None, None
    bv = bv / s
    return bv, float(M_var.value)


def nesterov_lp_iter(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated best-response iteration for the simplex game."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_br(y, solver)
        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = 2.0 * n * fast_conv(a).max()
        sc_y = 2.0 * n * fast_conv(y).max()

        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 8:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.random(n) * 0.02
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0
                prev_best = best_score
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M={M_val:.6f}")

    return best_a, best_score


def spectral_refine(a0, time_limit, n_iters=1000):
    """Spectral flatness pursuit: flatten the power spectrum while
    maintaining non-negativity. Uses softmax parameterization."""
    n = len(a0)
    z = log_vec(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_conv(a0).max()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_vec(z)
        # Compute power spectrum
        fsize = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fsize)
        P = np.abs(A) ** 2

        # Gradient of max autocorrelation w.r.t. z
        b = fast_conv(a)
        max_b = b.max()
        # Find active lags
        thresh = max_b * 0.9
        active = np.where(b >= thresh)[0]

        # Gradient of max_k (a*b)_k w.r.t. a (using active set)
        grad_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]
        grad_a /= len(active)

        # Convert to z-space: grad_z = a * (grad_a - sum(a * grad_a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Spectral flattening: move power from high to low in spectrum
        # This is a heuristic: reduce peaks in P, boost valleys
        P_mean = P.mean()
        P_flat = P + 0.1 * (P_mean - P)  # pull toward flat
        P_flat = np.maximum(P_flat, 1e-10)
        
        # Reconstruct a from flattened spectrum (keep phase)
        A_mag = np.sqrt(P_flat)
        A_phase = np.angle(A)
        A_new = A_mag * np.exp(1j * A_phase)
        a_spectral = irfft(A_new, fsize)[:n]
        
        # Project to non-negative, sum=1
        a_spectral = np.maximum(a_spectral, 0)
        ss = a_spectral.sum()
        if ss > 1e-15:
            a_spectral = a_spectral / ss
            sc = 2.0 * n * fast_conv(a_spectral).max()
            if sc < best_score:
                best_score = sc
                best_a = a_spectral.copy()
                z = log_vec(a_spectral)

        # Also try gradient step in z-space
        lr = 0.5 / (1 + 0.01 * it)
        z_new = z - lr * grad_z
        a_grad = softmax_vec(z_new)
        sc_grad = 2.0 * n * fast_conv(a_grad).max()
        if sc_grad < best_score:
            best_score = sc_grad
            best_a = a_grad.copy()
            z = z_new

    return best_a, best_score


def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """L-BFGS with LogSumExp smoothing, multi-stage annealing."""
    n = len(a0)
    z = log_vec(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        max_iter = max(20, int(remaining * 5))

        def func(z_in):
            a = softmax_vec(z_in)
            b = fast_conv(a)
            Tb = stage_T * b
            m = Tb.max()
            if m > 500:
                m = 500
            return (2.0 * n / stage_T) * (m + np.log(np.exp(Tb - m).sum()))

        def jac(z_in):
            a = softmax_vec(z_in)
            b = fast_conv(a)
            Tb = stage_T * b
            m = Tb.max()
            if m > 500:
                m = 500
            e = np.exp(Tb - m)
            w = e / e.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': max_iter * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_vec(z)


def peak_shave(a0, time_limit, n_steps=20000):
    """Local search: repeatedly reduce the peak autocorrelation by
    shifting mass from peak-contributing positions to safe positions."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n * fast_conv(a0).max()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        peak_k = int(np.argmax(b))

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if contrib.max() < 1e-15:
            break

        src = idx[int(np.argmax(contrib))]

        # Find safe destinations (not contributing to top peaks)
        top_lags = np.argsort(b)[::-1][:5]
        peak_indices = set()
        for k in top_lags:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            peak_indices.update(range(js2, je2 + 1))

        safe = [j for j in range(n) if j not in peak_indices and j != src]
        if not safe:
            safe = [j for j in range(n) if j != src]

        # Try several fractions
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            candidates = safe[:20]
            dst = candidates[rng.integers(len(candidates))]
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = 2.0 * n * fast_conv(a_trial).max()
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        if not improved and step > 5000:
            break

    return best_a, best_score

The problem is: min_a max_k (a*a)_k subject to a >= 0, sum(a) = 1.

This is: min M subject to (a*a)_k <= M for all k, a >= 0, sum(a) = 1.

def interp_to_len(f, new_n):
    old_n = len(f)

    if s < 1e-15:
        return np.ones(new_n) / new_n
