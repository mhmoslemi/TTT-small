import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def gs_spectral_flatten(a0, n_iters=5000, time_limit=60.0):
    """Gerchberg-Saxton style: alternate between flattening power spectrum
    and projecting to non-negative. Novel approach for this problem."""
    n = len(a0)
    a = normalize(a0.copy())
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # FFT of zero-padded signal
        A = rfft(a, fft_size)

        # Target: make non-DC magnitudes as equal as possible
        # DC component A[0] = sum(a) = 1 (fixed by normalization)
        # We want |A[m]| to be equal for m = 1, ..., fft_size//2
        abs_A = np.abs(A)
        # Compute target magnitude: geometric mean of non-DC magnitudes
        # (keep DC fixed)
        non_dc = abs_A[1:]
        target_mag = np.sqrt(np.prod(np.maximum(non_dc, 1e-30)))  # geometric mean
        # A soft version: pull magnitudes toward target
        # Use a soft update to avoid destroying the signal
        for m in range(1, fft_size // 2 + 1):
            if abs_A[m] > 1e-30:
                phase = A[m] / abs_A[m]
                new_mag = 0.7 * target_mag + 0.3 * abs_A[m]
                A[m] = phase * new_mag
            else:
                A[m] = phase * target_mag if abs_A[m] > 0 else target_mag

        # Inverse FFT
        a_new = irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def gs_alternating(a0, n_iters=3000, time_limit=60.0):
    """Improved GS: alternate between spectrum flattening and direct peak reduction.
    At each step, either flatten the spectrum or reduce the current peak."""
    n = len(a0)
    a = normalize(a0.copy())
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if it % 2 == 0:
            # Flatten spectrum
            A = rfft(a, fft_size)
            abs_A = np.abs(A)
            non_dc = abs_A[1:]
            target_mag = np.median(non_dc)  # use median for robustness
            for m in range(1, fft_size // 2 + 1):
                if abs_A[m] > 1e-30:
                    phase = A[m] / abs_A[m]
                    new_mag = 0.5 * target_mag + 0.5 * abs_A[m]
                    A[m] = phase * new_mag
            a_new = irfft(A, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                a = a_new
        else:
            # Reduce peak: find peak position, move mass away
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            M = b[peak_k]

            # Find the largest contributor to the peak
            # b[peak_k] = sum_{i+j=peak_k} a[i]*a[j]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            if je >= js:
                # Find pair (i, peak_k - i) with largest product
                candidates = [(i, peak_k - i) for i in range(js, je + 1)]
                best_prod = 0
                best_pair = None
                for i, j in candidates:
                    prod = a[i] * a[j]
                    if prod > best_prod:
                        best_prod = prod
                        best_pair = (i, j)

                if best_pair is not None:
                    i, j = best_pair
                    # Move small amount from the larger of a[i], a[j]
                    # to a random position that doesn't increase the peak
                    if a[i] > a[j]:
                        src = i
                    else:
                        src = j
                    # Target: a position far from the peak
                    # Try moving to position that minimizes impact
                    epsilon = 0.001 * a[src]
                    a[src] -= epsilon

                    # Find best target position
                    best_target = None
                    best_sc = local_eval_raw = 2.0 * n * b.max()
                    # Try a few candidate targets
                    for t_pos in [0, n // 4, 3 * n // 4, n - 1, src + n // 3]:
                        t_pos = min(max(t_pos, 0), n - 1)
                        a[t_pos] += epsilon
                        sc = 2.0 * n * fast_convolve(a).max()
                        if sc < best_sc:
                            best_sc = sc
                            best_target = t_pos
                        a[t_pos] -= epsilon

                    if best_target is not None:
                        a[best_target] += epsilon
                        s = a.sum()
                        a = a / s

        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def logsumexp_lbfgs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization.
    Uses FFT-based convolution for speed. Very aggressive temperature schedule."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    # Remove mean for numerical stability (scale invariance)
    z = z - z.mean()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # More iterations at lower T
        if T > 5.0:
            max_iter = max(30, int(remaining * 5))
        elif T > 0.5:
            max_iter = max(50, int(remaining * 10))
        else:
            max_iter = max(100, int(remaining * 30))

        def obj(z_in):
            z_max = z_in.max()
            exp_z = np.exp(z_in - z_max)
            a_cur = exp_z / exp_z.sum()
            A = rfft(a_cur, fft_size)
            b = irfft(A * A, fft_size)
            b = b[:2 * n - 1]
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
            return 2.0 * n * lse

        def grad(z_in):
            z_max = z_in.max()
            exp_z = np.exp(z_in - z_max)
            a_cur = exp_z / exp_z.sum()
            A = rfft(a_cur, fft_size)
            b = irfft(A * A, fft_size)
            b = b[:2 * n - 1]
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            # Gradient of b[k] w.r.t. a[j]: 2*a[k-j] if valid
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            # Chain rule for softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxfun': max_iter * 250, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def active_set_lp(a0, solver, time_limit, n_active_max=30):
    """Active-set LP alternation: solve LP using only active peak constraints.
    Much faster than full LP since we only include ~n_active constraints."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    active_set = set()
    stall = 0
    prev_score = best_score

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Update active set: positions within 0.5% of max
        threshold = M * 0.9995
        new_active = set(np.where(b >= threshold)[0].tolist())

        # Merge with previous active set (for stability)
        if it > 0:
            active_set = new_active | (active_set & new_active)
        else:
            active_set = new_active

        # Cap the active set size
        if len(active_set) > n_active_max:
            # Keep only the top n_active_max
            peaks_in_set = [(k, b[k]) for k in active_set]
            peaks_in_set.sort(key=lambda x: -x[1])
            active_set = set(k for k, _ in peaks_in_set[:n_active_max])

        # Build LP with active constraints
        delta_var = cp.Variable(n)
        M_var = cp.Variable()

        constraints = [
            cp.sum(delta_var) == 0,
            a + delta_var >= 0,
        ]

        # Add active peak constraints (linearized)
        for k in active_set:
            # b[k] + 2 * sum_j a[k-j] * delta_j <= M_var
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            row = np.zeros(n)
            row[idx] = 2.0 * a[k - idx]
            constraints.append(row @ delta_var <= M_var - b[k])

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                delta_val = np.asarray(delta_var.value, dtype=np.float64)
                M_val = float(M_var.value)

                # Take a step (with backtracking)
                for step_size in [1.0, 0.5, 0.25, 0.1]:
                    a_trial = a + step_size * delta_val
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        stall = 0
                        break

            if solved and not (best_score < prev_score - 1e-10):
                stall += 1
            else:
                stall = 0
        except Exception:
            stall += 1

        # If stalled, perturb and reset active set
        if stall > 20:
            rng = np.random.default_rng(it)
            a = a + rng.standard_normal(n) * 0.01
            a = normalize(a)
            active_set = set()
            stall = 0

        prev_score = best_score

        if (it + 1) % 500 == 0:
            print(f"    ASLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}, active={len(active_set)}")

    return best_a, best_score


def fourier_magnitude_optimize(a0, n_freqs=50, time_limit=120.0):
    """Optimize the first n_freqs Fourier magnitudes (keeping phases fixed).
    This directly shapes the power spectrum to be flatter."""
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    A = rfft(a, fft_size)
    # Keep phases fixed, optimize magnitudes of first n_freqs bins (non-DC)
    phases = np.angle(A[1:n_freqs + 1])
    A_rest = A[n_freqs + 1:].copy()

    n_params = n_freqs  # number of magnitudes to optimize

    def obj(mags):
        mags = np.maximum(mags, 1e-10)
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = A[0]  # DC fixed
        for i in range(n_freqs):
            A_new[i + 1] = mags[i] * np.exp(1j * phases[i])
        A_new[n_freqs + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    # Initial magnitudes
    x0 = np.abs(A[1:n_freqs + 1]).copy()

    try:
        res = minimize(
            obj, x0, method='Nelder-Mead',
            options={'maxiter': 10000, 'xatol': 1e-12, 'fatol': 1e-14,
                     'adaptive': True, 'maxfev': 50000}
        )
        x_opt = res.x
        mags = np.maximum(x_opt, 1e-10)
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = A[0]
        for i in range(n_freqs):
            A_new[i + 1] = mags[i] * np.exp(1j * phases[i])
        A_new[n_freqs + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass

    # Also try L-BFGS with numerical gradient
    if time.monotonic() - t0 < time_limit - 10:
        try:
            res2 = minimize(
                obj, x_opt, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxfun': 50000}
            )
            x_opt2 = np.maximum(res2.x, 1e-10)
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = A[0]
            for i in range(n_freqs):
                A_new[i + 1] = x_opt2[i] * np.exp(1j * phases[i])
            A_new[n_freqs + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s > 1e-10:
                a_candidate = a_candidate / s
                sc = 2.0 * n * fast_convolve(a_candidate).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=100):
    """Coordinate-wise optimization: try adjusting one coordinate at a time.
    For each coordinate, find the optimal value via 1D minimization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Random order of coordinates
        order = rng.permutation(n)
        improved = False

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Save current value
            old_val = a[idx]

            # Try a few perturbation sizes
            for frac in [0.8, 0.6, 0.4, 1.2, 1.5, 2.0, 0.1]:
                new_val = max(0.0, old_val * frac)
                a[idx] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[idx] = old_val
                    continue
                # Quick evaluation (full conv is expensive, use approximation)
                # For speed, only check if the peak at the relevant positions decreased
                # But for correctness, do full eval for small n
                if n <= 2000:
                    b = fast_convolve(a)
                    sc = 2.0 * n * b.max() / (s * s)
                else:
                    a[idx] = old_val
                    break

                if sc < best_score - 1e-12:
                    a = a / s
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[idx] = old_val

            a[idx] = old_val  # restore if not improved

        if not improved and rnd > 0:
            # Add small noise to escape local minimum
            a = a + rng.standard_normal(n) * (0.001 / math.sqrt(rnd + 1))
            a = normalize(a)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, choice=None):
    """Create structured initial sequence."""
    if choice is None:
        choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        a = np.sinc(3 * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
    else:
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 2 * x) * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def power_method_refine(a0, n_iters=200, time_limit=60.0):
    """Iterative best-response via power method: at each step, solve the
    dual problem approximately by gradient descent on the dual variables."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    # Dual variables: weights on each lag
    w = np.ones(2 * n - 1) / (2 * n - 1)

    lr = 0.1
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Update a: minimize sum_k w[k] * b[k] + entropy regularization
        # gradient of sum_k w[k] * b[k] w.r.t. a[j] = 2 * sum_k w[k] * a[k-j]
        # This is a convolution-based gradient
        grad_a = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        # Gradient step with projection
        a_trial = a - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s

        # Update dual weights: emphasize peaks
        b = fast_convolve(a_trial)
        b_shift = b / 0.1  # temperature for sharpness
        w = np.exp(b_shift - b_shift.max())
        w = w / w.sum()

        sc = 2.0 * n * b.max()
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial

        # Decay learning rate
        lr = max(0.001, lr * 0.999)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses GS spectral flattening, very aggressive LogSumExp L-BFGS,
    active-set LP, Fourier magnitude optimization, power method,
    and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: GS spectral flattening (fast, many iterations) ===
    print(f"\n=== Phase 1: GS Spectral Flattening (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 150.0)
    if t1 > 10:
        print(f"  Time budget: {t1:.0f}s")
        a_gs, _ = gs_spectral_flatten(best_seq, n_iters=10000, time_limit=t1 * 0.5)
        try_update(a_gs, "gs_flat")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_gs2, _ = gs_alternating(best_seq, n_iters=5000, time_limit=min(rem * 0.5, 60.0))
            try_update(a_gs2, "gs_alt")

    # === Phase 2: Very aggressive LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 30:
        T_stages = [30.0, 15.0, 7.0, 3.0, 1.5, 0.7, 0.3, 0.15, 0.07, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  T_stages: {T_stages}, budget={t2:.0f}s")
        a_lb, _ = logsumexp_lbfgs(best_seq, t2, T_stages=T_stages)
        try_update(a_lb, "lbgfs_aggressive")

    # === Phase 3: Active-set LP ===
    print(f"\n=== Phase 3: Active-Set LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        a_aslp, _ = active_set_lp(best_seq, solver, t3, n_active_max=20)
        try_update(a_aslp, "active_set_lp")

    # === Phase 4: Fourier magnitude optimization ===
    print(f"\n=== Phase 4: Fourier Magnitude Opt (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        for nf in [30, 50, 80]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_four, _ = fourier_magnitude_optimize(best_seq, n_freqs=nf, time_limit=min(rem * 0.3, 40.0))
            try_update(a_four, f"fourier_K{nf}")

    # === Phase 5: Power method refinement ===
    print(f"\n=== Phase 5: Power Method (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 10:
        a_pm, _ = power_method_refine(best_seq, n_iters=500, time_limit=t5)
        try_update(a_pm, "power_method")

    # === Phase 6: Second round of L-BFGS (lower T) ===
    print(f"\n=== Phase 6: L-BFGS Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if t6 > 20:
        T_stages2 = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb2, _ = logsumexp_lbfgs(best_seq, t6, T_stages=T_stages2)
        try_update(a_lb2, "lbgfs_round2")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = len(best_seq)
        for mult in [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.6, 2.0, 2.5, 3.0]:
            cand_n = int(base_n * mult)
            if 100 <= cand_n <= 8000 and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.15, 60.0)
            if t_sub < 15:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick GS + L-BFGS
            a_cand, _ = gs_spectral_flatten(init_cand, n_iters=2000, time_limit=t_sub * 0.3)
            try_update(a_cand, f"gs_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cand2, _ = logsumexp_lbfgs(a_cand, min(rem * 0.4, 30.0),
                                            T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05])
                try_update(a_cand2, f"lbgfs_n{cand_n}")

    # === Phase 8: Final intensive polish on best ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low T L-BFGS
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            a_fin, _ = logsumexp_lbfgs(best_seq, t_fin,
                                       T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_fin, "final_lbgfs")

        # Short active-set LP
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = active_set_lp(best_seq, solver, min(rem * 0.3, 40.0), n_active_max=15)
            try_update(a_fin2, "final_aslp")

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15 and len(best_seq) <= 2000:
            a_fin3, _ = coordinate_descent(best_seq, min(rem * 0.3, 40.0), n_rounds=20)
            try_update(a_fin3, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
