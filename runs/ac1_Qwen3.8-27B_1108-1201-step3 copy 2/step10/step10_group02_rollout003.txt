import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def lp_best_response(a_fixed):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] using direct optimization."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build the convolution matrix T such that T @ b = conv(a_fixed, b)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        T[rows, j] = a_fixed
    
    # Solve: min M s.t. T @ b <= M * 1, b >= 0, sum b = 1
    # This is an LP. Use scipy's linprog.
    from scipy.optimize import linprog
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: T @ b - M <= 0 -> [T, -1] @ [b, M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum b = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b_val = result.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(result.x[-1])
    return b_val, M_val

                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    
                    # Quick evaluation
                    b_trial = fast_convolve(a_trial)
                    score = 2.0 * n * b_trial.max()
                    
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            # Also try increasing elements that are currently small at the peak
            if not improved:
                bottom_m = min(3, len(idx))
                bottom_positions = idx[np.argsort(contribs)[:bottom_m]]
                for pos in bottom_positions:
                    if a[pos] > 0.9 * max(a):  # skip if already large
                        continue
                    for factor in [1.01, 1.05, 1.1, 1.2]:
                        a_trial = a.copy()
                        old_val = a_trial[pos]
                        new_val = old_val * factor
                        a_trial[pos] = new_val
                        diff = new_val - old_val
                        # Reduce others proportionally
                        mask = np.ones(n, dtype=bool)
                        mask[pos] = False
                        scale = (a[mask].sum() - diff) / a[mask].sum()
                        if scale < 0:
                            break
                        a_trial[mask] *= scale
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        b_trial = fast_convolve(a_trial)
                        score = 2.0 * n * b_trial.max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            stall += 1
            if stall > 500:
                # Random structured perturbation
                n_perturb = max(3, n // 50)
                positions = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for pos in positions:
                    a_trial[pos] *= (0.95 + 0.1 * rng.random())
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                stall = 0
        else:
            stall = 0
        
        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD-v2 {elapsed:.0f}s: score={best_score:.8f} stall={stall}")
    
    return best_a, best_score


def logsumexp_smoothed_grad(z, T_temp):
    """
    Compute gradient of the smoothed objective:
    (2n/T) * log(sum_k exp(T * (a*a)[k]))
    where a = softmax(z).
    """
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    
    # Compute the weighted average of subgradients
    # w_k = exp(T * b[k]) / sum_j exp(T * b[j])
    T_b = T_temp * b
    T_b_shifted = T_b - T_b.max()
    exp_Tb = np.exp(T_b_shifted)
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of (a*a)[k] w.r.t. a is: for each position j,
    # d(a*a)[k]/da[j] = a[k-j] + a[j-(k-j)] (with appropriate bounds)
    # Simplified: the subgradient at peak k is (1/|A|) * (a reflected around k)
    
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # d(a*a)[k]/da[idx] = a[k - idx]
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Convert to z-space: grad_z = a * (grad_a - dot(a, grad_a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    # Objective value
    logsumexp = b.max() + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_temp) * logsumexp
    
    return obj, grad_z


def logsumexp_lbgfs_v2(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    from scipy.optimize import minimize
    
    max_stages = len(T_stages)
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_smoothed_grad(z_inner, T)
            return obj
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_smoothed_grad(z_inner, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax_safe(z)


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_start_optimize(best_seq, best_score, time_limit, rng):
    """
    Try multiple starting points and optimization strategies.
    Returns the best found.
    """
    deadline = t0 + time_limit  # t0 is set externally
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    n_best = len(best_seq)
    
    # Try different n values
    n_candidates = [n_best]
    for factor in [0.5, 0.75, 1.5, 2.0, 3.0]:
        n_try = int(n_best * factor)
        if 100 <= n_try <= 5000:
            n_candidates.append(n_try)
    
    n_candidates = sorted(set(n_candidates))
    print(f"    Trying n values: {n_candidates}")
    
    best_overall = best_seq.copy()
    best_overall_score = best_score
    
    # Allocate time proportional to number of candidates
    time_per = max(30, time_limit / len(n_candidates))
    
    for n_try in n_candidates:
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        if n_try == n_best:
            a_init = best_seq.copy()
        else:
            a_init = interpolate_to_length(best_seq, n_try)
        
        # Run a short optimization from this starting point
        t_alloc = min(time_per, time_limit - (time.monotonic() - t0) - 10)
        if t_alloc < 10:
            break
        
        a_opt, sc = mirror_descent_v2(a_init, t_alloc, eta0=0.05,
                                       active_frac=0.999, nesterov=True,
                                       polyak=True, cycle=True)
        
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall = a_opt.copy()
            print(f"    NEW BEST: n={n_try}, score={sc:.8f}")
    
    return best_overall, best_overall_score
