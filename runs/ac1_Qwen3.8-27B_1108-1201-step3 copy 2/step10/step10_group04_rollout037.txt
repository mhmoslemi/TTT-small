def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def _compute_score(a):
    """Compute 2n*max(a*a) assuming sum(a)=1."""
    b = _fast_convolve(a)
    return 2.0 * len(a) * b.max()


def _softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def _z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def _central_binomial_init(n, rng):
    """
    Initialize using the central binomial coefficient pattern.
    The recursion (a*a)_k = const for k=0..n-1 gives a_k prop to C(2k,k)/4^k.
    We create several variants: decreasing, increasing, symmetric.
    """
    k = np.arange(n, dtype=np.float64)
    # Use log for numerical stability
    # log(C(2k,k)/4^k) = log(C(2k,k)) - k*log(4)
    # For large k, use Stirling: C(2k,k)/4^k ≈ 1/sqrt(pi*k)
    if n <= 500:
        from math import lgammal
        log_vals = np.array([lgammal(2*int(ki)+1) - 2*lgammal(int(ki)+1) - int(ki)*np.log(4) for ki in range(n)])
        vals = np.exp(log_vals - log_vals.max())
    else:
        # Asymptotic: 1/sqrt(pi*k) for k>=1, 1 for k=0
        vals = np.ones(n)
        for ki in range(1, n):
            vals[ki] = 1.0 / np.sqrt(np.pi * ki)
    
    variant = rng.integers(0, 4)
    if variant == 0:
        # Decreasing
        a = vals.copy()
    elif variant == 1:
        # Increasing (reverse)
        a = vals[::-1].copy()
    elif variant == 2:
        # Symmetric
        a = (vals + vals[::-1]).copy()
    else:
        # Decaying exponential blend
        a = vals * np.exp(-0.5 * np.linspace(0, 1, n))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    # Add small random perturbation to break symmetry
    a += rng.random(n) * 0.01
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def _structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5  # Square root of sin - peaked in middle
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2  # Decreasing
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 1.0
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)  # Gaussian bump
    elif choice == 4:
        # Two bumps at edges
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x**2) + np.exp(-8 * (1-x)**2)
    else:
        # Power law decay
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5 + 0.1
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    return a


def _subgradient_descent(a0, time_limit, n_active_frac=0.05, max_steps=2000000):
    """Subgradient descent with heavy-ball momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    lr = 0.5
    momentum = 0.9
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        # Compute subgradient using active set
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Adaptive step size: decrease if stuck
        alpha = lr / (1 + 0.01 * no_improve)

        # Heavy ball update
        v = momentum * v - alpha * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = _compute_score(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 5.0)
        else:
            no_improve += 1
            lr = max(lr * 0.95, 0.01)
            if no_improve >= 50:
                # Perturbation to escape
                rng = np.random.default_rng(step * 7 + 3)
                a_new = a + rng.random(n) * 0.02
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                no_improve = 0
                lr = 0.5
                v = np.zeros(n)

        a = a_new

    return best_a, best_score


def _pair_coordinate_descent(a0, time_limit, max_steps=50000, search_iters=40):
    """
    Pair coordinate descent: pick pairs (j, m), optimize the split between them
    using ternary search on t (where a_j = t, a_m = c - t).
    Exploits the fact that for fixed other coordinates, the convolution
    entries are at most quadratic in t.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_convolve(a)
        # Find peak-contributing positions
        peak_k = int(np.argmax(b))
        
        # Find which positions contribute to the top peaks
        top_peaks = np.argsort(b)[::-1][:3]
        contrib = np.zeros(n)
        for pk in top_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[pk - idx]
        
        # Source: highest contributor
        src = int(np.argmax(contrib))
        
        # Destination: low-contributing, low-value position
        low_contrib = np.argsort(contrib)[:n//4]
        candidates = [j for j in low_contrib if j != src and a[j] < a.max() * 0.5]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        dst = candidates[rng.integers(len(candidates))]

        # Ternary search on t = a_src_new, with a_dst_new = a[src] + a[dst] - t
        c = a[src] + a[dst]
        lo, hi = 0.0, c

        def objective(t_val):
            a_trial = a.copy()
            a_trial[src] = t_val
            a_trial[dst] = c - t_val
            sc = _compute_score(a_trial)
            return sc

        # Golden section search (faster convergence than ternary)
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = objective(c1)
        f2 = objective(c2)

        for _ in range(search_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = objective(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = objective(c2)

        t_opt = (lo + hi) / 2
        a_trial = a.copy()
        a_trial[src] = t_opt
        a_trial[dst] = c - t_opt
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        score = _compute_score(a_trial)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try a random pair as exploration
            if step % 10 == 0:
                j = rng.integers(n)
                m = rng.integers(n)
                if j != m:
                    c2 = a[j] + a[m]
                    t_rand = rng.uniform(0, c2)
                    a_trial2 = a.copy()
                    a_trial2[j] = t_rand
                    a_trial2[m] = c2 - t_rand
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 > 1e-15:
                        a_trial2 = a_trial2 / ss2
                        sc2 = _compute_score(a_trial2)
                        if sc2 < best_score - 1e-14:
                            best_score = sc2
                            best_a = a_trial2.copy()
                            a = a_trial2

    return best_a, best_score


def _lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=8):
    """L-BFGS with LogSumExp approximation."""
    n = len(a0)
    z = _z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_func(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = _softmax_from_z(z_in)
        b = _fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = _softmax_from_z(z_in)
        b = _fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Gradient of max_k (a*a)_k w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(10, int(60 * (0.5 ** si)))

        def func(z_in):
            return lse_func(z_in, stage_T)

        def jac(z_in):
            return lse_jac(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return _softmax_from_z(z)

def _get_lp_solver():

    if _lp_solver is not None:

    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

def _build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def _solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None or not HAS_CVXPY:
        return None, None
    n = len(a_fixed)
    if n > 3000:  # LP too slow for very large n
        return None, None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _lp_nash_iteration(a0, time_limit, solver, max_iters=500000):
    """LP-based Nash iteration (best response method with averaging)."""
    if solver is None or not HAS_CVXPY:
        return a0, _compute_score(a0)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    y = a.copy()  # running average
    avg_count = 0.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = _solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Also track the "Nesterov" point
        if it == 0:
            a = b_new.copy()
        else:
            a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = _compute_score(a)
        score_y = _compute_score(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        # Stall detection with perturbation
        if (it + 1) % 500 == 0:
            b_check = _fast_convolve(best_a)
            # Check gradient: if best_a is close to y, we might be at a fixed point
            if np.linalg.norm(best_a - y) < 0.01 * best_score:
                rng = np.random.default_rng(it * 17 + 41)
                y = y + rng.random(n) * 0.05
                y = np.maximum(y, 0)
                y = y / y.sum()

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Central binomial coefficient initialization (based on flat-convolution recursion)
2. Subgradient descent with heavy-ball momentum
3. Pair coordinate descent with golden-section search (exploits quadratic structure)
4. LP-based Nash best-response iteration
5. LogSumExp smoothed L-BFGS for smooth refinement
6. Multi-length exploration with interpolation and adaptive time allocation
Key new ideas: combinatorial initialization from C(2k,k)/4^k pattern,
coordinate descent exploiting the 2-variable quadratic structure of convolution,
and aggressive multi-n search with smart budget allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _compute_score(a):
    """Compute 2n*max(a*a) assuming sum(a)=1."""
    b = _fast_convolve(a)
    return 2.0 * len(a) * b.max()


def _softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def _z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def _central_binomial_init(n, rng):
    """Initialize using C(2k,k)/4^k pattern from flat-convolution recursion."""
    k = np.arange(n, dtype=np.float64)
    if n <= 1000:
        from math import lgammal
        log_vals = np.array([lgammal(2 * int(ki) + 1) - 2 * lgammal(int(ki) + 1) - int(ki) * np.log(4) for ki in range(n)])
        vals = np.exp(log_vals - log_vals.max())
    else:
        vals = np.ones(n)
        for ki in range(1, n):
            vals[ki] = 1.0 / np.sqrt(np.pi * ki)

    variant = rng.integers(0, 5)
    if variant == 0:
        a = vals.copy()
    elif variant == 1:
        a = vals[::-1].copy()
    elif variant == 2:
        a = (vals + vals[::-1]).copy()
    elif variant == 3:
        a = vals * np.exp(-0.3 * np.linspace(0, 1, n))
    else:
        # Two-edge bump: large at both ends
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x**2) + np.exp(-4 * (1-x)**2) + 0.1 * vals

    a = np.maximum(a, 0)
    a += rng.random(n) * 0.01
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x))
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x**2) + np.exp(-6 * (1-x)**2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _subgradient_descent(a0, time_limit, n_active_frac=0.05, max_steps=2000000):
    """Subgradient descent with heavy-ball momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    lr = 0.5
    momentum = 0.9
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        alpha = lr / (1.0 + 0.05 * no_improve)
        v = momentum * v - alpha * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = _compute_score(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 5.0)
        else:
            no_improve += 1
            lr = max(lr * 0.95, 0.01)
            if no_improve >= 50:
                rng = np.random.default_rng(step * 7 + 3)
                a_new = a + rng.random(n) * 0.02
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                no_improve = 0
                lr = 0.5
                v = np.zeros(n)

        a = a_new

    return best_a, best_score


def _pair_coordinate_descent(a0, time_limit, max_steps=50000, search_iters=40):
    """
    Pair coordinate descent: pick pairs (j, m), optimize the split using
    golden-section search. Exploits that for fixed other coords, each
    (a*a)_k is at most quadratic in the split variable.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_convolve(a)
        top_peaks = np.argsort(b)[::-1][:3]

        contrib = np.zeros(n)
        for pk in top_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[pk - idx]

        src = int(np.argmax(contrib))
        low_contrib = np.argsort(contrib)[:max(1, n // 4)]
        candidates = [j for j in low_contrib if j != src and a[j] < a.max() * 0.5]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        dst = candidates[rng.integers(len(candidates))]

        c = a[src] + a[dst]
        lo, hi = 0.0, c

        def obj(t_val):
            a_trial = a.copy()
            a_trial[src] = t_val
            a_trial[dst] = c - t_val
            return _compute_score(a_trial)

        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = obj(c1)
        f2 = obj(c2)

        for _ in range(search_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = obj(c2)

        t_opt = (lo + hi) / 2.0
        a_trial = a.copy()
        a_trial[src] = t_opt
        a_trial[dst] = c - t_opt
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        score = _compute_score(a_trial)

        improved = False
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            improved = True

        if not improved and step % 5 == 0:
            # Random exploration pair
            j = rng.integers(n)
            m = rng.integers(n)
            if j != m:
                c2 = a[j] + a[m]
                t_rand = rng.uniform(0.1, 0.9) * c2
                a_trial2 = a.copy()
                a_trial2[j] = t_rand
                a_trial2[m] = c2 - t_rand
                a_trial2 = np.maximum(a_trial2, 0)
                ss2 = a_trial2.sum()
                if ss2 > 1e-15:
                    a_trial2 = a_trial2 / ss2
                    sc2 = _compute_score(a_trial2)
                    if sc2 < best_score - 1e-14:
                        best_score = sc2
                        best_a = a_trial2.copy()
                        a = a_trial2

    return best_a, best_score


def _lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=8):
    """L-BFGS with LogSumExp approximation for smooth optimization."""
    n = len(a0)
    z = _z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_func(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        aa = _softmax_from_z(z_in)
        b = _fast_convolve(aa)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        aa = _softmax_from_z(z_in)
        b = _fast_convolve(aa)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * aa[k - idx]
        S = np.dot(grad_a, aa)
        return aa * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(10, int(40 * (0.6 ** si)))

        def func(z_in):
            return lse_func(z_in, stage_T)

        def jac(z_in):
            return lse_jac(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return _softmax_from_z(z)


_lp_solver = None

def _get_lp_solver():
    global _lp_solver
    if not HAS_CVXPY:
        return None
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def _build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def _solve_lp_best_response(a_fixed, solver):
    if solver is None or not HAS_CVXPY:
        return None, None
    n = len(a_fixed)
    if n > 2500:
        return None, None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _lp_nash_iteration(a0, time_limit, solver, max_iters=500000):
    """LP-based Nash iteration with Nesterov-style averaging."""
    if solver is None or not HAS_CVXPY:
        return a0, _compute_score(a0)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    y = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = _solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        if it == 0:
            a = b_new.copy()
        else:
            a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = _compute_score(a)
        score_y = _compute_score(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 1000 == 0:
            if np.linalg.norm(best_a - y) < 0.005:
                rng = np.random.default_rng(it * 17 + 41)
                y = y + rng.random(n) * 0.05
                y = np.maximum(y, 0)
                y = y / y.sum()

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def _evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
