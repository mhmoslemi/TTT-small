    """Compute gradient of LogSumExp(beta * conv(a,a)) w.r.t. a.
    Returns gradient (length n)."""
    n = len(a)
    # Compute b = conv(a, a)
    N = 1 << (3*n-2).bit_length()  # Pad to next power of 2
    A = rfft(a)  # This gives length N//2+1
    B = A * A
    b = irfft(B, N)[:2*n-1]  # conv(a,a)
    
    # Compute softmax weights

    grad = 2.0 * conv_result[n-1:2*n-1]

    Uses FFT for O(n log n) computation.

        a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    y = a.copy()
    k = 0

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        stage_iters = 0
        while time.monotonic() - t0 < time_limit:
            k += 1
            stage_iters += 1

            need = 3 * n - 2
            N = 1 << (need - 1).bit_length()
            A = rfft(y, N)
            B = irfft(A * A, N)[:2 * n - 1]

            B_pos = np.maximum(B, 1e-30)
            Bp = B_pos ** p
            sum_Bp = Bp.sum()
            w = Bp / sum_Bp

            flip_a = y[::-1]
            FA = rfft(flip_a, N)
            FW = rfft(w, N)
            conv_aw = irfft(FA * FW, N)
            grad = 2.0 * p * conv_aw[n - 1:2 * n - 1] / (sum_Bp ** (1.0 / p)) * B_pos ** (p - 1)

            lr = 0.003 / (1.0 + 0.02 * np.sqrt(k + 1.0))
            y_new = y - lr * grad
            y_new = project_simplex(y_new)

            alpha = 2.0 / (2.0 + np.sqrt(1.0 + 4.0 * max(0, k - 1) ** 2))
            a_new = (1.0 - alpha) * a + alpha * y_new
            a_new = project_simplex(a_new)

            y = y_new
            a = a_new

            if stage_iters % 100 == 0:
                score = evaluate_local(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

                if stage_iters % 500 == 0:
                    print(f"    PS k={k} p={p} iter={stage_iters}: score={best_score:.8f}")

    return best_a, best_score


def peak_coord_descent(a0, time_limit, n_steps):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for peak_k in top_peaks[:3]:
            pk = int(peak_k)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = int(idx[oi])
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.96]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            pk = int(top_peaks[0])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            if stall > 300:
                break

        if step % 5000 == 0:
            print(f"    PeakCD {time.monotonic() - t0:.0f}s step={step}: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps):
    """Subgradient method with Nesterov acceleration targeting active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    y = a.copy()
    k = 0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        k += 1

        need = 3 * n - 2
        N = 1 << (need - 1).bit_length()
        A = rfft(y, N)
        B = irfft(A * A, N)[:2 * n - 1]
        max_b = B.max()
        threshold = max_b * 0.999
        active = np.where(B >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(B))])

        flip_a = y[::-1]
        FA = rfft(flip_a, N)
        active_w = np.zeros(2 * n - 1)
        active_w[active] = 1.0 / len(active)
        FW = rfft(active_w, N)
        conv_aw = irfft(FA * FW, N)
        grad = 2.0 * conv_aw[n - 1:2 * n - 1]

        lr_t = lr / (1.0 + 0.01 * np.sqrt(k))
        y_new = y - lr_t * grad
        y_new = project_simplex(y_new)

        alpha = 2.0 / (2.0 + np.sqrt(1.0 + 4.0 * (k - 1)))
        a_new = (1.0 - alpha) * a + alpha * y_new
        a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        if step % 50 == 0:
            score = evaluate_local(a)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-6, lr)

        if step % 2000 == 0:
            print(f"    SubGrad {time.monotonic() - t0:.0f}s step={step} lr={lr:.6f}: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """Initialize with a sequence designed to have flat power spectrum."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.2 + 0.1 * np.exp(-8 * (x - 0.5) ** 2)
    else:
        a = np.exp(-3.5 * (x - 0.5) ** 2 / (0.5 * (1 - x * 0)))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
