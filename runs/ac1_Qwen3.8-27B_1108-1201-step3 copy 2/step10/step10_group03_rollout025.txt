"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP alternating minimization with warm-starting
2. Nesterov-accelerated subgradient with FFT-based convolution
3. LogSumExp-smoothed L-BFGS with adaptive temperature annealing
4. Multi-n exploration with interpolation
5. Coordinate descent polish
Key improvements: efficient active-set reduction, better time management,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (sum(a)=1 assumed)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def obj_normalized(a):
    """Objective assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response_lp(fixed_a, active_k, solver):
    """
    Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_{k in active_k} conv(a,b)[k].
    Only uses constraints for positions in active_k.
    """
    n = len(fixed_a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]

    # Build constraints efficiently using the convolution structure
    for k in active_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        coeff = fixed_a[k - idx]
        constraints.append(coeff @ b_var[idx] <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solver_list = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt not in solver_list:
            solver_list.append(alt)

    for s in solver_list:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def active_set_lp_alternation(a0, time_limit, solver, max_iters=50000, active_frac=0.15):
    """
    Bidirectional LP alternation with active set reduction.
    At each iteration, only include constraints near the current max.
    Uses Nesterov-type averaging for convergence.
    """
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution and active set
        conv_b = fast_convolve(b)
        M_b = conv_b.max()
        # Active set: positions where conv is within active_frac of max
        threshold = M_b * (1.0 - active_frac)
        active_b = np.where(conv_b >= threshold)[0]
        if len(active_b) == 0:
            active_b = np.array([int(np.argmax(conv_b))])
        # Limit active set size
        if len(active_b) > 200:
            top_idx = np.argsort(conv_b)[::-1][:200]
            active_b = top_idx

        # Solve LP: best a given y_b
        a_new, M1 = solve_best_response_lp(y_b, active_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            # Nesterov combined
            a = ((it + 1) * a + alpha * (it + 3) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        # Recompute active set based on a
        conv_a = fast_convolve(a)
        M_a = conv_a.max()
        threshold_a = M_a * (1.0 - active_frac)
        active_a = np.where(conv_a >= threshold_a)[0]
        if len(active_a) == 0:
            active_a = np.array([int(np.argmax(conv_a))])
        if len(active_a) > 200:
            top_idx = np.argsort(conv_a)[::-1][:200]
            active_a = top_idx

        # Solve LP: best b given y_a
        b_new, M2 = solve_best_response_lp(y_a, active_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 4)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + alpha * (it + 4) * b_new) / (it + 1 + alpha * (it + 4))
            b = normalize(b)

        # Track best
        sc_a = obj_normalized(a)
        sc_b = obj_normalized(b)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        # Stall detection and perturbation
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                noise = 0.03 / math.sqrt(min(stall_count, 20))
                y_a = y_a + rng.standard_normal(n) * noise
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise
                y_b = normalize(y_b)
                # Also perturb the running average
                a = a + rng.standard_normal(n) * noise * 0.5
                a = normalize(a)
                b = b + rng.standard_normal(n) * noise * 0.5
                b = normalize(b)
                stall_count = 0

            if (it + 1) % 500 == 0:
                print(f"    LPAlt iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, n_steps=500000):
    """
    Nesterov-accelerated subgradient method for minimizing max_k conv(a,a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    y = normalize(a0).copy()
    x = y.copy()
    best_a = x.copy()
    best_score = obj_normalized(x)
    t0 = time.monotonic()
    gamma = 1.0  # step size parameter

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y
        b = fast_convolve(y)
        M = b.max()
        # Find active set (within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Weight by proximity to max (soft active set)
        w = b[active] - b[active].min() + 1e-10
        w = w / w.sum()

        # Subgradient: average of gradients of active functions
        # grad f_k(a)[j] = 2*a[k-j] for valid k-j
        grad = np.zeros(n)
        for i, k in enumerate(active):
            if w[i] < 1e-8:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * y[k - idx]

        # Step size: gamma / sqrt(it + 1)
        step = gamma / math.sqrt(it + 1)

        # Update y (Nesterov)
        y_new = y - step * grad
        # Project onto simplex: a >= 0, sum(a) = 1
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        # Nesterov momentum
        t_val = (1.0 + math.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        t_prev = (1.0 + math.sqrt(1.0 + 4.0 * it ** 2)) / 2.0 if it > 0 else 1.0
        momentum = (it + 1 - t_val) / t_prev
        y = y_new + momentum * (y_new - y)
        y = np.maximum(y, 0)
        s = y.sum()
        if s > 1e-15:
            y = y / s

        x = y_new  # x is the "primal" point

        # Track best
        sc = obj_normalized(x)
        if sc < best_score:
            best_score = sc
            best_a = x.copy()

        # Also check y
        sc_y = obj_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # Adaptive gamma
        if (it + 1) % 500 == 0:
            if best_score < prev_gs - 1e-12:
                gamma = min(gamma * 1.05, 5.0)
            elif best_score > prev_gs + 1e-8:
                gamma = max(gamma * 0.9, 0.1)
            prev_gs = best_score

    return best_a, best_score


prev_gs = 0.0


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    global prev_gs
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z = z - z.max()  # for numerical stability
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 15))

        def lse_obj(z_vec):
            z_max = z_vec.max()
            exp_z = np.exp(z_vec - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.sum(np.exp(b_shift - b_max)))
            return 2.0 * n * lse

        def lse_grad(z_vec):
            z_max = z_vec.max()
            exp_z = np.exp(z_vec - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()

            # Gradient w.r.t. a: 2 * sum_k w[k] * a[k-j]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]

            # Chain rule: softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = obj_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=50000):
    """
    Coordinate descent: at each step, optimize one element of a while keeping
    others fixed. Uses the fact that conv(a,a) is quadratic in each a[i].
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Pre-compute: for each i, conv(a,a) as a function of a[i]
    # conv(a,a)[k] = a[i]^2 * [k==2i] + a[i] * (sum of cross terms) + (terms not involving a[i])
    # More precisely: conv(a,a)[k] = sum_j a[j]*a[k-j]
    # If we change a[i] to a[i]+delta:
    #   conv(a',a')[k] = conv(a,a)[k] + delta*(a[k-i] + a[i-k]) + delta^2*[k==2i]
    #   (with a[j]=0 for j out of range)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick coordinate to optimize (cycling with random restarts)
        if rnd % 1000 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute current conv
            b = fast_convolve(a)
            M = b.max()

            # Try a few values of a[i]
            # The optimal a[i] given others: minimize max_k (b[k] + delta*c[k] + delta^2*d[k])
            # where c[k] = a[k-i] + a[i-k] (with out-of-range = 0), d[k] = 1 if k==2i else 0
            ai_old = a[i]

            # Compute c[k] = a[k-i] + a[i-k] for valid indices
            c = np.zeros(2 * n - 1)
            # a[k-i] is valid when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
            js1 = max(0, i)
            je1 = min(2 * n - 2, i + n - 1)
            if js1 <= je1:
                c[js1:je1 + 1] += a[i:(je1 - i + 1)]
            # a[i-k] is valid when 0 <= i-k <= n-1, i.e., i-n+1 <= k <= i
            js2 = max(0, i - n + 1)
            je2 = min(2 * n - 2, i)
            if js2 <= je2:
                c[js2:je2 + 1] += a[i - (je2 - js2 + 1):i + 1][::-1]

            # d[k] = 1 if k == 2i and 2i < 2n-1
            d = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                d[2 * i] = 1.0

            # We want to find delta that minimizes max_k (b[k] + delta*c[k] + delta^2*d[k])
            # subject to a[i] + delta >= 0, sum(a) + delta = 1 (so delta = -sum of other changes... 
            # actually we should renormalize)
            # For simplicity, try a few candidate values:
            # - Current value (delta=0)
            # - Value that would make the peak entry zero (if possible)
            # - Value that equalizes the top few entries
            # - 0
            # - A few values around current

            candidates = [0.0, ai_old]
            # Try to reduce the current max
            k_max = int(np.argmax(b))
            c_kmax = c[k_max]
            d_kmax = d[k_max]
            if d_kmax > 0:
                # Quadratic: b[kmax] + delta*c_kmax + delta^2 = M_target
                # delta = (-c_kmax + sqrt(c_kmax^2 - 4*(b[kmax]-M_target))) / 2
                for target_frac in [0.95, 0.9, 0.85]:
                    target = M * target_frac
                    disc = c_kmax ** 2 - 4.0 * (b[k_max] - target)
                    if disc >= 0:
                        delta1 = (-c_kmax + math.sqrt(disc)) / 2.0
                        delta2 = (-c_kmax - math.sqrt(disc)) / 2.0
                        if ai_old + delta1 >= 0:
                            candidates.append(delta1)
                        if ai_old + delta2 >= 0:
                            candidates.append(delta2)
            elif c_kmax != 0:
                delta_to_reduce = (M * 0.9 - b[k_max]) / c_kmax
                candidates.append(delta_to_reduce)

            # Try a few small perturbations
            for pert in [0.01, -0.01, 0.05, -0.05]:
                candidates.append(ai_old + pert * ai_old if ai_old > 0 else pert)

            best_delta = None
            best_trial_score = best_score

            for delta in candidates:
                ai_new = ai_old + delta
                if ai_new < 0 or ai_new > 10:
                    continue
                # Quick evaluation: compute new max
                new_b = b + delta * c + delta * delta * d
                new_max = new_b.max()
                # Renormalize: a[i] changes from ai_old to ai_new
                # sum changes by delta, so we need to scale
                new_sum = 1.0 + delta  # since we're changing one element
                if new_sum < 0.01:
                    continue
                # After renormalization, the conv scales by 1/new_sum^2
                trial_score = 2.0 * n * new_max / (new_sum * new_sum)
                if trial_score < best_trial_score:
                    best_trial_score = trial_score
                    best_delta = delta

            if best_delta is not None:
                a[i] = a[i] + best_delta
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = ai_old
                    continue
                a = a / s
                sc = obj_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def direct_subgradient_fast(a0, time_limit, n_steps=200000):
    """
    Fast subgradient method with FFT-based gradient computation.
    Uses a smarter active set and adaptive step size.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    step_size = 1.0
    no_improve = 0

    # Use a "dual" variable for step size adaptation
    beta = 1.0  # initial step size

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Soft active set: weight by exp(b/T) for small T
        T_active = M * 0.01
        b_shift = b / T_active
        b_max = b_shift.max()
        w = np.exp(b_shift - b_max)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        # This is 2 * (w * a)[j] where w has length 2n-1 and a has length n
        # Compute via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # (w * a)[j] for j in [0, n-1]: this is the convolution of w and a
        # But w has length 2n-1 and a has length n
        # conv(w, a)[j] = sum_k w[k] * a[j-k] ... 
        # Actually we want sum_k w[k] * a[k-j]
        # = sum_k w[k] * a[(k-j)] = (w flipped * a)[j] ... 
        # Let me just do it directly using the FFT of a and the weights
        # grad[j] = 2 * sum_k w[k] * a[k-j]
        # For each j: sum over k where k-j is in [0,n-1], i.e., j <= k <= j+n-1
        # This is a cross-correlation type computation
        # Let's use: grad = 2 * ifft( W_full * A_conj ) ... no that's not right either
        # 
        # Actually: sum_k w[k] * a[k-j] = (w * a_flipped)[j] where a_flipped[i] = a[n-1-i]
        # Hmm, let me just compute it directly for the active set
        grad = np.zeros(n)
        active_idx = np.where(w > 1e-10)[0]
        for k in active_idx:
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * a[k - idx]

        # Projected gradient step with adaptive step size
        a_trial = a - beta * grad
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s

        sc = obj_normalized(a_trial)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            beta = min(beta * 1.02, 3.0)
        else:
            no_improve += 1
            if no_improve >= 50:
                beta = max(beta * 0.9, 0.01)
                no_improve = 0
                # Occasionally add noise
                if no_improve % 5 == 0:
                    a = a + np.random.standard_normal(n) * 0.005
                    a = normalize(a)

    return best_a, best_score


def peak_shaving(a0, time_limit, n_rounds=10000):
    """
    Direct peak shaving: at each round, find the peak position k*, identify
    the elements of a that contribute most to conv(a,a)[k*], and reduce them.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        k_peak = int(np.argmax(b))

        # Find contributing elements
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k_peak - idx]
        total_contrib = contributions.sum()

        if total_contrib < 1e-15:
            continue

        # Reduce the top contributors
        n_reduce = max(1, len(idx) // 4)
        top_local = np.argsort(contributions)[::-1][:n_reduce]

        # Compute the "cost" of reducing each element: how much it contributes to other peaks
        # For element a[j], it contributes to conv[k] for k in [j, j+n-1] (as a[j]*a[k-j])
        # and to conv[k] for k in [n-1-j, n-1+j] (as a[k-j]*a[j])
        # Simplification: just try reducing and check

        for _ in range(3):  # Try a few reduction amounts
            reduction_frac = rng.uniform(0.01, 0.1)
            a_trial = a.copy()
            for li in top_local:
                j = idx[li]
                a_trial[j] *= (1.0 - reduction_frac)
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = obj_normalized(a_trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break

        # Also try: boost elements that are at zero but would help flatten
        # (this is less reliable, skip for now)

        if (rnd + 1) % 1000 == 0:
            print(f"    PeakShave round={rnd+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP alternation, subgradient, LogSumExp L-BFGS, coordinate descent,
    peak shaving, and multi-n exploration.
    """
    global prev_gs
    prev_gs = 0.0

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        a = np.exp(-4 * x * (1 - x))
        best_seq = normalize(a)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = obj_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternation (primary optimizer) ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 450.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = active_set_lp_alternation(best_seq, t1, solver, max_iters=50000, active_frac=0.12)
        try_update(a_lp, "lp_alt_1")

    # === Phase 2: Nesterov Subgradient ===
    print(f"\n=== Phase 2: Nesterov Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 20:
        a_ng, sc_ng = nesterov_subgradient(best_seq, t2, n_steps=300000)
        try_update(a_ng, "nesterov_sg")

    # === Phase 3: LP Alternation Round 2 ===
    print(f"\n=== Phase 3: LP Alternation R2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 300.0)
    if t3 > 20:
        # Use smaller active set for more focus
        a_lp2, sc_lp2 = active_set_lp_alternation(best_seq, t3, solver, max_iters=50000, active_frac=0.05)
        try_update(a_lp2, "lp_alt_2")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 300.0)
    if t4 > 20:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb, sc_lb = logsumexp_lbgfs(best_seq, t4, T_stages=T_stages)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 10:
        a_ps, sc_ps = peak_shaving(best_seq, t5, n_rounds=20000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        # Try nearby n values
        n_candidates = []
        base_n = n_best
        for mult in [0.8, 0.9, 1.1, 1.25, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 300 <= cand_n <= 5000 and cand_n not in n_candidates and cand_n != n_best:
                n_candidates.append(cand_n)
        # Also try some absolute values
        for cn in [800, 1000, 1500, 2000, 2500]:
            if cn not in n_candidates and cn != n_best:
                n_candidates.append(cn)

        n_candidates.sort(key=lambda x: abs(x - n_best))
        n_candidates = n_candidates[:5]  # Limit to 5 candidates

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_sub = min(rem * 0.2, 120.0)
            if t_sub < 40:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)

            # Quick LP
            a_cand, _ = active_set_lp_alternation(init_cand, t_sub * 0.5, solver, max_iters=20000, active_frac=0.15)
            try_update(a_cand, f"lp_n{cand_n}")

            # Quick L-BFGS
            rem = deadline - time.monotonic()
            if rem > 30:
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.3, 60.0),
                                             T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05])
                try_update(a_cand2, f"lbgfs_n{cand_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # LP with very small active set
        t_p1 = min(rem * 0.3, 60.0)
        if t_p1 > 10:
            a_fin1, _ = active_set_lp_alternation(best_seq, t_p1, solver, max_iters=20000, active_frac=0.03)
            try_update(a_fin1, "final_lp_tight")

        # Very low T L-BFGS
        rem = deadline - time.monotonic()
        if rem > 20:
            t_p2 = min(rem * 0.4, 80.0)
            a_fin2, _ = logsumexp_lbgfs(best_seq, t_p2,
                                       T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_fin2, "final_lbgfs_lowT")

        # Peak shaving
        rem = deadline - time.monotonic()
        if rem > 15:
            t_p3 = min(rem * 0.3, 40.0)
            a_fin3, _ = peak_shaving(best_seq, t_p3, n_rounds=10000)
            try_update(a_fin3, "final_peakshave")

        # Direct subgradient (fast)
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin4, _ = direct_subgradient_fast(best_seq, min(rem * 0.5, 30.0), n_steps=50000)
            try_update(a_fin4, "final_direct_sg")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
