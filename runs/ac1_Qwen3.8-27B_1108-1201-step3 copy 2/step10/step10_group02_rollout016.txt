import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_fast(a):
    """Fast evaluation of the objective (assumes a >= 0, sum > 0)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(a):
    """Project onto the probability simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if a.sum() <= 0:
        return np.ones(n) / n
    # Use sorting-based projection
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def interp_to_len(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    return normalize(a_new)

    eta = eta0

        it += 1
        # Compute convolution at Nesterov point

            no_improve = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD iter={it} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None, verbose=False):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if solver else [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for sol in solvers_to_try:
        if sol is None:
            continue
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_alternating(a0, time_limit, solver=None, verbose=True):
    """LP alternating method for peak reduction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_fast(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver=solver)
        if b_new is None:
            continue

        # Mirror-type update
        score_a = eval_fast(a)
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update a toward b_new with diminishing step
        t = 2.0 / (it + 2.0)
        a_new = (1.0 - t) * a + t * b_new
        a_new = normalize(a_new)
        a = a_new

        if no_improve > 200:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = normalize(a)
            no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it} ({elapsed:.0f}s): score={best_score:.8f}")
            if no_improve > 500:
                break

    return best_a, best_score


def peak_deflation_cd(a0, time_limit, verbose=True):
    """
    Coordinate descent that explicitly targets the convolution peak.
    For each coordinate contributing to the peak, try reducing it and
    redistributing mass to low-correlation positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_fast(a)
    t0 = time.monotonic()
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]

        # Find coordinates contributing to the peak, sorted by contribution
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top_k = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_k]]

        # Find low-correlation positions for redistribution
        full_correlation = np.zeros(n)
        full_correlation[js:je+1] = a[k_peak - idx]
        low_positions = np.where(full_correlation < np.median(full_correlation)[0] if len(full_correlation) > 0 else 0)[0]
        if len(low_positions) == 0:
            low_positions = np.argsort(full_correlation)[:5]
        low_positions = low_positions[low_positions >= 0][:10]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            # Try reducing this position
            for frac in [0.999, 0.995, 0.99, 0.98, 0.95]:
                delta = a[pos] * (1.0 - frac)
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[pos] -= delta

                # Redistribute to low-correlation positions
                if len(low_positions) > 0:
                    redist = low_positions.astype(float) / low_positions.sum() if low_positions.sum() > 0 else None
                    if redist is not None:
                        for lp in low_positions:
                            if lp >= 0 and lp < n:
                                a_trial[lp] += delta * (1.0 / len(low_positions))

                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                score = eval_fast(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Try a random small perturbation
                rng = np.random.default_rng(step * 13)
                mask = np.random.choice(n, size=min(5, n), replace=False)
                a_trial = a.copy()
                for m in mask:
                    a_trial[m] *= (1.0 + rng.standard_normal() * 0.01)
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    score = eval_fast(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = a_trial  # accept anyway for exploration
                        stall = 0
                if stall > 1000:
                    break
        else:
            stall = 0

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakCD step={step} ({elapsed:.0f}s): score={best_score:.8f} stall={stall}")

    return best_a, best_score


def power_sum_descent(a0, time_limit, p_seq=None, verbose=True):
    """
    Optimizes the power-sum (sum_k conv^p)^(1/p) as a smooth surrogate for max.
    Increases p over time to sharpen toward the max.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_fast(a)
    t0 = time.monotonic()

    if p_seq is None:
        p_seq = [2, 4, 8, 16, 32, 64, 128]

    for p in p_seq:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        # Optimize power-sum for this p using gradient descent
        a_sub, _ = _power_sum_refine(a, remaining / len(p_seq), p, verbose=(p == p_seq[-1]))
        sc = eval_fast(a_sub)
        if sc < best_score:
            best_score = sc
            best_a = a_sub.copy()
            a = a_sub

    return best_a, best_score


def _power_sum_refine(a0, time_limit, p, verbose=False):
    """Refine a by minimizing (sum_k conv(a,a)^p)^(1/p) via gradient descent."""
    n = len(a0)
    a = normalize(a0)
    t0 = time.monotonic()
    lr = 0.01
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        # Power-sum objective and gradient
        bp = b ** p
        obj = (bp.sum()) ** (1.0 / p)
        # Gradient: d/da_i (sum_k conv^p)^(1/p) = (1/p) * (sum_k conv^p)^(1/p - 1) * p * sum_k conv^{p-1} * d(conv)_k/da_i
        # = (sum_k conv^p)^(1/p - 1) * sum_k conv_k^{p-1} * 2*a_{k-i}
        if obj < 1e-15:
            break
        grad = np.zeros(n)
        for k in range(len(b)):
            if b[k] < 1e-15:
                continue
            coeff = b[k] ** (p - 1)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += coeff * a[k - i]
        grad *= 2.0
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        score = eval_fast(a_new)
        if score < obj * 2 * n:  # rough check
            a = a_new
        lr *= 0.999

    return normalize(a), 0


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses targeted subgradient descent, LP alternating, multi-n exploration,
    and peak-deflation coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_fast(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 1.5)
        best_score = eval_fast(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = normalize(np.asarray(a, dtype=np.float64))
        sc = eval_fast(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    print(f"\n{'='*60}")
    print(f"Starting optimization. Initial score: {best_score:.8f}")
    print(f"{'='*60}")

    # === Phase 1: Targeted subgradient descent (main workhorse) ===
    print(f"\n=== Phase 1: Subgradient Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = subgradient_descent_peak(best_seq, t_alloc, eta0=0.03,
                                           polyak=True, nesterov_alpha=2.0)
        try_update(a_md, "subgrad_p1")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver=cp.ECOS)
        try_update(a_lp, "lp_alt")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_per_n = max(5, t_alloc * 0.15 if rem > 0 else 5)
    n_candidates = [800, 1000, 1200, 1500, 1800, 2000, 2500]
    n_base = len(best_seq)
    for n_try in n_candidates:
        if time.monotonic() > deadline:
            break
        if abs(n_try - n_base) < 50:
            continue  # Skip if too close to current
        t_rem_each = min(rem * 0.1, 120.0)
        t_elapsed_probe = time.monotonic()
        a_interp = interp_to_len(best_seq, n_try)
        # Quick subgradient descent at this n
        a_opt, _ = subgradient_descent_peak(a_interp, min(t_rem_each, 60.0),
                                            eta0=0.02, polyak=False, nesterov_alpha=2.0,
                                            verbose=False)
        try_update(a_opt, f"multi_n_{n_try}")

    # === Phase 4: Peak deflation coordinate descent ===
    print(f"\n=== Phase 4: Peak Deflation CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = peak_deflation_cd(best_seq, t_alloc)
        try_update(a_cd, "peak_cd")

    # === Phase 5: Second subgradient pass with different params ===
    print(f"\n=== Phase 5: Refine Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 120.0)
    if t_alloc > 10:
        a_md2, _ = subgradient_descent_peak(best_seq, t_alloc, eta0=0.005,
                                            polyak=True, nesterov_alpha=3.0)
        try_update(a_md2, "subgrad_p2")

    # === Phase 6: Power-sum smoothing optimization ===
    print(f"\n=== Phase 6: Power-Sum Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ps, _ = power_sum_descent(best_seq, t_alloc, p_seq=[4, 8, 16, 32])
        try_update(a_ps, "power_sum")

    # === Phase 7: Final coordinate descent polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_polish, _ = peak_deflation_cd(best_seq, t_alloc)
        try_update(a_polish, "final_polish")

    # === Phase 8: Try one more LP round if time permits ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_lp2, _ = lp_alternating(best_seq, t_alloc, solver=cp.ECOS)
        try_update(a_lp2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = eval_fast(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"{'='*60}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
