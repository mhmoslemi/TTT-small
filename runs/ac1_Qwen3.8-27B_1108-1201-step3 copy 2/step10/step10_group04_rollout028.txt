        if src == dst:
            continue
        # Line search for optimal fraction
        best_frac = 0.0
        best_sc = best_score
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = min(a[src] * frac, a[src] - 1e-15)
            if amt <= 0:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = score_of(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_frac = frac
        if best_frac > 0:
            amt = a[src] * best_frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        # Additional random moves
        for _ in range(3):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            frac = float(rng.uniform(0.01, 0.1))
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = score_of(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
    return best_a, best_score


def simulated_annealing_opt(a0, time_limit, T_start=0.05, T_end=0.001):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    current_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    n_steps = max(500, int(time_limit * 80))
    temps = np.geomspace(T_start, T_end, n_steps)

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        T = temps[it]
        move_type = int(rng.integers(0, 4))

        if move_type == 0:
            k = int(rng.integers(max(2, n // 10), max(3, n // 3)))
            idx = rng.choice(n, k, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, T * 0.05, k)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
        elif move_type == 1:
            b = fast_conv(a)
            peak_k = int(np.argmax(b))
            contrib = np.zeros(n)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contrib[idx] = a[idx] * a[peak_k - idx]
            src = int(np.argmax(contrib))
            dst = int(rng.integers(n))
            if dst == src:
                dst = (dst + 1) % n
            frac = float(rng.uniform(0.02, 0.15))
            a_trial = a.copy()
            amt = a[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
        elif move_type == 2:
            i = int(rng.integers(0, n - 1))
            j = int(rng.integers(i + 1, n))
            factor = float(rng.uniform(0.6, 1.4))
            a_trial = a.copy()
            a_trial[i:j + 1] *= factor
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if a[i] > 1e-12:
                frac = float(rng.uniform(0.01, 0.08))
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
            else:
                a_trial = a.copy()

        score_trial = score_of(a_trial)
        delta = score_trial - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * x * (1 - x))
    elif choice == 4:
        a = (0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))) ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 * (1 + 0.1 * np.cos(20 * np.pi * x))
    else:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3.0 * x ** 2) * (1 + 0.2 * np.cos(8 * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Multi-phase optimization at a fixed length."""
    best_a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(len(a_init)) / len(a_init)
    best_a = best_a / best_a.sum()
    best_score = score_of(best_a)
    t0 = time.monotonic()

    def try_update(a_cand, src_label):
        nonlocal best_a, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(a_cand, 0).astype(np.float64)
        ss = a_cand.sum()
        if ss < 1e-15:
            return
        a_cand = a_cand / ss
        sc = score_of(a_cand)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cand.copy()
            print(f"  UPDATE [{src_label}]: {sc:.6f} (n={len(a_cand)}, t={time.monotonic()-t0:.0f}s)")

    # Phase A: Mirror descent
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.15, 60.0)
    if t_a > 10:
        a_md, _ = mirror_descent_opt(best_a, t_a, active_frac=0.05)
        try_update(a_md, f"md_{label}")

    # Phase B: LP game iteration (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.40, 400.0)
    if t_b > 20:
        a_lp, _ = lp_game_iteration(best_a, t_b, solver=solver)
        try_update(a_lp, f"lp_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.25, 150.0)
    if t_c > 15:
        a_lb = lse_lbfgs_opt(best_a, t_c, T_start=30.0, T_end=0.02, n_stages=10)
        try_update(a_lb, f"lse_{label}")

    # Phase D: Coordinate descent
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.10, 60.0)
    if t_d > 10:
        a_cd, _ = coordinate_descent_opt(best_a, t_d, n_sweeps=100)
        try_update(a_cd, f"cd_{label}")

    # Phase E: More LP
    rem = time_budget - (time.monotonic() - t0)
    t_e = min(rem * 0.10, 60.0)
    if t_e > 15:
        a_lp2, _ = lp_game_iteration(best_a, t_e, solver=solver)
        try_update(a_lp2, f"lp2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-length exploration with multi-phase optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    solver = _get_solver()
    print(f"Solver: {solver}, budget: {budget_s:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = score_of(best_seq)
        print(f"Init: n={n_init}, score={best_score:.6f}")

    def try_update_global(a_cand, label):
        nonlocal best_seq, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(a_cand, 0).astype(np.float64)
        ss = a_cand.sum()
        if ss < 1e-15:
            return
        a_cand = a_cand / ss
        sc = score_of(a_cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cand.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a_cand)}, t={time.monotonic()-(deadline-budget_s+60):.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: Optimize at base length ===
    print(f"\n=== Phase 1: Optimize n={n_base} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, sc_out = optimize_at_length(best_seq, t1, solver, label=f"n{n_base}")
        try_update_global(a_out, f"opt_n{n_base}")

    # === Phase 2: Try different lengths ===
    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    candidates = []
    for mult in [0.6, 0.8, 1.2, 1.5, 2.0]:
        ln = int(n_base * mult)
        if 200 <= ln <= 5000 and ln != n_base:
            candidates.append(ln)
    candidates = sorted(set(candidates))

    for ln in candidates:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.20, 200.0)
        if t_alloc < 50:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        init_long = interpolate_sequence(best_seq, ln)
        sc_init = score_of(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolation improved to {sc_init:.6f}")
        a_out, _ = optimize_at_length(init_long, t_alloc, solver, label=f"n{ln}")
        try_update_global(a_out, f"opt_n{ln}")

    # === Phase 3: Refine at best length ===
    print(f"\n=== Phase 3: Final refinement (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Additional LP
        t_lp = min(rem * 0.30, 120.0)
        if t_lp > 15:
            a_fin, _ = lp_game_iteration(best_seq, t_lp, solver=solver)
            try_update_global(a_fin, "final_lp")

        # LSE with very low final temp
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.30, 80.0)
            if t_lse > 10:
                a_fin2 = lse_lbfgs_opt(best_seq, t_lse, T_start=5.0, T_end=0.01, n_stages=10)
                try_update_global(a_fin2, "final_lse")

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.25, 50.0)
            if t_cd > 10:
                a_fin3, _ = coordinate_descent_opt(best_seq, t_cd, n_sweeps=200)
                try_update_global(a_fin3, "final_cd")

        # One more LP pass
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lp2 = min(rem * 0.40, 60.0)
            if t_lp2 > 10:
                a_fin4, _ = lp_game_iteration(best_seq, t_lp2, solver=solver)
                try_update_global(a_fin4, "final_lp2")

        # SA for escaping
        rem = deadline - time.monotonic()
        if rem > 10:
            t_sa = min(rem * 0.50, 40.0)
            if t_sa > 5:
                a_fin5, _ = simulated_annealing_opt(best_seq, t_sa, T_start=0.03, T_end=0.0005)
                try_update_global(a_fin5, "final_sa")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
