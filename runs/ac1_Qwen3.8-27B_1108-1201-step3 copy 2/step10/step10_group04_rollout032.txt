def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):

    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)

def build_conv_matrix(a_fixed):

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    v = v / target_sum
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient_structured(a, active_fraction=0.05):
    """
    Compute subgradient of max_k conv(a,a)[k].
    Uses the exact structure: d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n.
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    if max_c < 1e-300:
        return np.zeros(n), max_c, np.array([int(np.argmax(c))])

    n_active = max(1, int(active_fraction * len(c)))
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_active]

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    return g, max_c, active

    """
    Peak-shaving using the exact gradient structure.
    The gradient of max_k conv(a,a)[k] is a blurred/reversed version of a.
    We move in the negative gradient direction with careful line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n, 2.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_c, active = compute_subgradient_structured(a, active_frac)
        g_norm = np.sqrt(np.dot(g, g))
        if g_norm < 1e-15:
            break

        # Compute the directional derivative on the simplex
        # The projected gradient direction is: -g + (mean of g over support)
        g_proj = g - np.mean(g[a > 1e-15]) * (a > 1e-15) if np.any(a > 1e-15) else g

        # Line search
        alpha = lr
        improved = False
        for bt in range(15):
            a_trial = a - alpha * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / st
            c_trial = fast_convolve(a_trial)
            score_trial = 2.0 * n * c_trial.max()

            if score_trial < best_score - 1e-14:
                best_score = score_trial
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                lr = min(alpha * 1.05, 1.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.005)
            if no_improve >= 50:
                # Perturb to escape
                rng = np.random.default_rng(step * 31 + 7)
                k_pert = rng.integers(n // 10, n // 3)
                idx = rng.choice(n, k_pert, replace=False)
                a[idx] += rng.normal(0, 0.005, k_pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                lr = 0.1

    return best_a, best_score


def pairwise_exchange(a0, time_limit, n_sweeps=200):
    """
    Targeted pairwise exchange: move mass from elements that contribute
    to peak correlations to elements that don't.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        top_k = np.argsort(c)[::-1][:min(5, len(c))]

        # Build contribution map for top peaks
        contrib_to_peak = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                contrib_to_peak[i] += c[k] * a[i] * a[k - i]

        # Try exchanges: from high-contrib to low-contrib
        high_mask = contrib_to_peak > np.median(contrib_to_peak)
        low_mask = contrib_to_peak <= np.median(contrib_to_peak)
        high_idx = np.where(high_mask)[0]
        low_idx = np.where(low_mask)[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            high_idx = np.argsort(contrib_to_peak)[::-1][:n // 3]
            low_idx = np.argsort(contrib_to_peak)[:n // 3]

        improved_this_sweep = False
        n_tries = min(200, len(high_idx) * len(low_idx))

        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            src = rng.choice(high_idx)
            dst = rng.choice(low_idx)
            if src == dst or a[src] < 1e-12:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved_this_sweep = True
                    break

        if not improved_this_sweep and sweep > 50:
            # Add random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                for frac in [0.003, 0.01, 0.03]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            break

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.03, n_stages=8):
    """L-BFGS with LogSumExp smooth approximation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_loc = np.exp(z_in - z_in.max())
        a_loc = a_loc / a_loc.sum()
        b = fast_convolve(a_loc)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a_loc = np.exp(z_in - z_in.max())
        a_loc = a_loc / a_loc.sum()
        b = fast_convolve(a_loc)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_loc[k - idx]
        S = np.dot(grad_a, a_loc)
        return a_loc * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, 300 // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_lse(z_in):
            return lse_objective(z_in, stage_T)

        def jac_lse(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.5 * x ** 0.3
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a_new, src):
        nonlocal best_a, best_score
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0).astype(np.float64)
        ss = a_new.sum()
        if ss < 0.01:
            return
        a_new = a_new / ss
        sc = 2.0 * n * fast_convolve(a_new).max()
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            elapsed = time.monotonic() - t0
            print(f"    [{src}] new best: {sc:.7f} (n={n}, t={elapsed:.0f}s)")

    # Phase A: LP Nesterov
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.35, 400.0)
    if t_a > 30:
        a_out, _ = nesterov_lp(best_a, t_a, solver=solver)
        try_update(a_out, "lp")

    # Phase B: Peak shaving
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.25, 200.0)
    if t_b > 15:
        a_out, _ = peak_shaving(best_a, t_b, active_frac=0.03)
        try_update(a_out, "peak")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.20, 150.0)
    if t_c > 10:
        a_out = lbgfs_logsumexp(best_a, t_c, T_start=30.0, T_end=0.02, n_stages=8)
        try_update(a_out, "lse")

    # Phase D: Pairwise exchange
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.20, 100.0)
    if t_d > 5:
        a_out, _ = pairwise_exchange(best_a, t_d, n_sweeps=100)
        try_update(a_out, "pair")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.7f}")

    def try_global_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.7f} (n={len(a)})")

    # Phase 1: Optimize at base length (main workhorse)
    print(f"\n=== Phase 1: Optimize at n={n_base} (score={best_score:.7f}) ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t1, solver, label=f"n{n_base}")
        try_global_update(a_out, f"opt_n{n_base}")

    # Phase 2: Peak shaving refinement on current best
    print(f"\n=== Phase 2: Peak shaving (score={best_score:.7f}) ===")
    rem = time_limit_global - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 30:
        a_out, _ = peak_shaving(best_seq, t2, active_frac=0.02)
        try_global_update(a_out, "peak_refine")

    # Phase 3: LP refinement
    rem = time_limit_global - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 20:
        a_out, _ = nesterov_lp(best_seq, t3, solver=solver)
        try_global_update(a_out, "lp_refine")

    # Phase 4: Try nearby lengths
    print(f"\n=== Phase 4: Nearby lengths (score={best_score:.7f}) ===")
    nearby_lengths = []
    for delta in [-100, -50, 50, 100, 150, -150, 200]:
        ln = n_base + delta
        if 400 <= ln <= 3000 and ln != n_base:
            nearby_lengths.append(ln)
    nearby_lengths = sorted(set(nearby_lengths))[:4]

    for ln in nearby_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.20, 150.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        init_long = interpolate_sequence(best_seq, ln)
        a_out, _ = optimize_at_length(init_long, t_alloc, solver, label=f"n{ln}")
        try_global_update(a_out, f"opt_n{ln}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final Polish (score={best_score:.7f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 20:
        # Peak shaving with very small active set
        t_pe = min(rem * 0.30, 80.0)
        if t_pe > 10:
            a_out, _ = peak_shaving(best_seq, t_pe, active_frac=0.01)
            try_global_update(a_out, "final_peak")

        # Pairwise exchange
        rem = time_limit_global - time.monotonic()
        t_pair = min(rem * 0.30, 60.0)
        if t_pair > 5:
            a_out, _ = pairwise_exchange(best_seq, t_pair, n_sweeps=50)
            try_global_update(a_out, "final_pair")

        # One more LP
        rem = time_limit_global - time.monotonic()
        t_lp = min(rem * 0.30, 40.0)
        if t_lp > 10:
            a_out, _ = nesterov_lp(best_seq, t_lp, solver=solver)
            try_global_update(a_out, "final_lp")

        # LogSumExp fine tune
        rem = time_limit_global - time.monotonic()
        t_lse = min(rem * 0.30, 30.0)
        if t_lse > 5:
            a_out = lbgfs_logsumexp(best_seq, t_lse, T_start=5.0, T_end=0.01, n_stages=6)
            try_global_update(a_out, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
