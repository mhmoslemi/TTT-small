```python
def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[0]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[0]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

            break

        P = np.abs(A)**2

        P_target[1:] = target_mean

        a_new = project_simplex(a_new, 1.0)
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (Gerchberg-Saxton) for fast basin-finding
2. Power-sum FISTA with adaptive p-increase for smooth gradient refinement
3. Multi-length exploration (n=1200 to n=3000)
4. Exact subgradient polishing for final convergence
Key innovation: power-sum smoothing with O(n log n) gradient via FFT cross-correlation,
replacing LP-based approaches with pure gradient methods for better convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[0]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    """Score = 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_power_sum_grad(y, p, N):
    """
    Compute gradient of 2*n*(sum_k b_k^p)^(1/p) at point y (on simplex, sum=1).
    Uses FFT cross-correlation for O(N log N) computation.
    """
    n = len(y)
    b = fast_convolve(y)
    b = np.maximum(b, 0.0)
    h = np.power(b, p - 1.0)

    A = fft(y, N)
    H = fft(h, N)
    corr = ifft(np.conj(A) * H).real

    indices = np.mod(-np.arange(n), N)
    c = corr[indices]

    sum_bp = np.sum(np.power(b, p))
    if sum_bp < 1e-300:
        return np.zeros(n)

    coef = 4.0 * n * np.power(sum_bp, 1.0 / p - 1.0)
    return coef * c


def spectral_flatten(a, n_iters=300, alpha=0.5, deadline=None, verbose=False):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    a = project_simplex(a, 1.0).copy()
    best_a = a.copy()
    best_score = evaluate_local(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.fft(a, n)
        P = np.abs(A) ** 2

        if n > 1:
            target_mean = P[1:].mean()
        else:
            target_mean = P[0]

        P_target = P.copy()
        P_target[1:] = target_mean

        P_new = alpha * P + (1.0 - alpha) * P_target
        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(A_new))
        a_new = project_simplex(a_new, 1.0)

        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 50 == 0:
            print(f"      Spectral iter {it}: score={best_score:.8f}")

    return best_a, best_score


def power_sum_fista(a0, p_start, p_end, n_iters, dt, N, deadline=None,
                    backtracking=True, verbose=False):
    """
    FISTA with power-sum smoothed objective.
    p is annealed from p_start to p_end over n_iters.
    dt is the base step size.
    """
    n = len(a0)
    a = project_simplex(a0, 1.0).copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = evaluate_local(a)

    gamma = dt
    no_improve = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Anneal p
        frac = it / max(1, n_iters - 1)
        p = p_start + (p_end - p_start) * frac

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = y + (t - 1.0) / (t_new + 1.0) * (y - a)
        y = y_new

        # Gradient at y
        grad = compute_power_sum_grad(y, p, N)

        # Gradient step
        z = y - gamma * grad

        # Project onto simplex
        a_new = project_simplex(z, 1.0)
        t = t_new

        # Evaluate
        score = evaluate_local(a_new)

        if backtracking and score > best_score * 1.01:
            gamma *= 0.5
            continue

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and it % 100 == 0:
            print(f"      FISTA iter {it} (p={p:.1f}): score={best_score:.8f} gamma={gamma:.6f}")

    return best_a, best_score


def subgradient_polish(a0, n_steps, deadline=None, verbose=False):
    """
    Subgradient polishing: use exact subgradient (active max index)
    with decreasing step size and averaging.
    """
    n = len(a0)
    a = project_simplex(a0, 1.0).copy()
    best_a = a.copy()
    best_score = evaluate_local(a)

    eta = 0.1 / n
    avg_len = min(50, n_steps // 10)
    avg_grad = np.zeros(n)
    step_count = 0

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-5)
        active = np.where(b >= threshold)[0]

        # Compute subgradient: average over active peaks
        grad = np.zeros(n)
        for k in active[:20]:  # cap at 20 active indices
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active[:20])

        # Exponential averaging
        avg_grad = (step_count * avg_grad + grad) / (step_count + 1)
        step_count += 1

        # Step size decay
        eta_cur = eta / (1.0 + step * 0.01)

        z = a - eta_cur * avg_grad
        a_new = project_simplex(z, 1.0)

        score = evaluate_local(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and step % 1000 == 0:
            print(f"      SubGrad step {step}: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

                        best_seq, p_start=8.0, p_end=32.0,
                        n_iters=1000, dt=0.4 / n, N=N,
                        deadline=time.monotonic() + t_f
                    )
                    local_update(a_f, f"fista_after_spec_a{alpha_try}")

    # === Phase 3: Try n=1500 ===
    print(f"\n=== Phase 3: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t_n1500 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, sc_1500 = full_optimize(init_1500, t_n1500, deadline, verbose=True)
        local_update(a_1500, "full_n1500")

    # === Phase 4: Try n=2000 ===
    print(f"\n=== Phase 4: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_n2000 = min(rem * 0.20, 200.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, sc_2000 = full_optimize(init_2000, t_n2000, deadline, verbose=True)
        local_update(a_2000, "full_n2000")

    # === Phase 5: Try n=2500 ===
    print(f"\n=== Phase 5: Try n=2500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n2500 = min(rem * 0.18, 180.0)
        print(f"  Budget: {t_n2500:.0f}s")
        init_2500 = interpolate_sequence(best_seq, 2500)
        a_2500, sc_2500 = full_optimize(init_2500, t_n2500, deadline, verbose=True)
        local_update(a_2500, "full_n2500")

    # === Phase 6: Perturbation escape from best ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        N_cur = 1 << (3 * n_cur).bit_length()
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 60.0)
            # Quick FISTA on perturbed
            a_opt, sc_opt = power_sum_fista(
                a_pert, p_start=8.0, p_end=32.0,
                n_iters=1500, dt=0.4 / n_cur, N=N_cur,
                deadline=time.monotonic() + t_sub
            )
            local_update(a_opt, f"pert{trial}")

    # === Phase 7: Final fine FISTA on best ===
    print(f"\n=== Phase 7: Final fine FISTA (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_fin = len(best_seq)
        N_fin = 1 << (3 * n_fin).bit_length()
        t_fin = min(rem * 0.35, 80.0)
        a_fin, sc_fin = power_sum_fista(
            best_seq, p_start=32.0, p_end=128.0,
            n_iters=3000, dt=0.2 / n_fin, N=N_fin,
            deadline=time.monotonic() + t_fin
        )
        local_update(a_fin, "final_fista_fine")

    # === Phase 8: Final subgradient ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.4, 25.0)
        a_sg, sc_sg = subgradient_polish(
            best_seq, n_steps=10000,
            deadline=time.monotonic() + t_sg
        )
        local_update(a_sg, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
