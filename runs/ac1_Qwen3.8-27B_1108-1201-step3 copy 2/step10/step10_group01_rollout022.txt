    lp_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new, M_b = solve_lp_best_response(y, T, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue
        lp_count += 1
        
        # Adaptive Nesterov momentum
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Primal iterated average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= stall_patience:
                noise_scale = 0.03 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0
            
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP alt iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_LP={M_b:.8f} lps={lp_count}")
    
    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per = max(30, maxiter_per_stage // n_stages)

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_stage(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = np.exp(zz - zz.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_bb = bb.max()
            exp_bb = np.exp(T_stage * (bb - max_bb))
            lse = max_bb + np.log(exp_bb.sum()) / T_stage
            return 2.0 * n * lse

        def jac_stage(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            aa = np.exp(zz - zz.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_bb = bb.max()
            exp_bb = np.exp(T_stage * (bb - max_bb))
            w = exp_bb / exp_bb.sum()
            # grad[j] = 2/(T) * sum_k w[k] * a[k-j]
            # = 2/T * cross_corr(w, a)[j]
            grad_a = 2.0 / T_stage * cross_convolve(a=aa, y=w)
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)

        try:
            res = minimize(
                fun=func_stage, x0=z, jac=jac_stage, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return np.maximum(a_out, 0)


def peak_coord_descent(a0, time_limit, n_steps=100000, top_m=6):
    """Coordinate descent targeting top convolution peaks with adaptive steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]
        improved = False

        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.997
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=80000):
    """Subgradient method with adaptive learning rate on active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Hann window
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    else:
        # Flat-top window approximation
        a = 1.0 - 0.6 * np.cos(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x) - 0.1 * np.cos(6 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP alternation -> L-BFGS -> Peak CD ->
    Subgradient -> Multi-n -> Perturbation -> Final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Primary LP alternation ===
    print(f"\n=== Phase 1: LP alternation n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 700.0)
    if t1 > 40:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = lp_alternation(best_seq, t1, solver=solver,
                                  max_iters=100000, stall_patience=12)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: L-BFGS LogSumExp refinement ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        T_stages = [15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]
        print(f"  {len(T_stages)} stages, budget={t2:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t2, T_stages=T_stages, maxiter_per_stage=20000)
        try_update(a_lb, "lbgfs_phase2")

    # === Phase 3: Peak-directed coordinate descent ===
    print(f"\n=== Phase 3: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 15:
        a_cd, _ = peak_coord_descent(best_seq, t3, n_steps=150000, top_m=6)
        try_update(a_cd, "peak_cd")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 100.0)
    if t4 > 10:
        a_sg, _ = subgradient_refine(best_seq, t4, n_steps=100000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Second LP pass ===
    print(f"\n=== Phase 5: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 160.0)
    if t5 > 30:
        a_lp2, _ = lp_alternation(best_seq, t5, solver=solver,
                                   max_iters=80000, stall_patience=15)
        try_update(a_lp2, "lp_phase5")

    # === Phase 6: Multi-start perturbation ===
    print(f"\n=== Phase 6: Multi-start perturbation, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 60.0)
        for pi in range(4):
            if time.monotonic() > deadline - 40:
                break
            if t_each < 20:
                break
            if pi == 0:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.04 * np.cos(2 * x) + 0.02 * np.sin(3 * x)
            elif pi == 1:
                pert = 0.03 * rng.standard_normal(n_cur)
            elif pi == 2:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.03 * np.sin(5 * x) + 0.01 * np.sin(7 * x)
            else:
                # Structured perturbation targeting the peak
                b_cur = fast_convolve(best_seq)
                peak_pos = np.argmax(b_cur)
                js = max(0, peak_pos - n_cur + 1)
                je = min(n_cur - 1, peak_pos)
                pert = np.zeros(n_cur)
                pert[js:je+1] = -0.02
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            a_res, _ = lp_alternation(a_pert, t_each, solver=solver,
                                       max_iters=30000, stall_patience=10)
            try_update(a_res, f"perturb_{pi}")

    # === Phase 7: Alternative length exploration ===
    print(f"\n=== Phase 7: Alt length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.3)
        t_alt = min(rem * 0.35, 200.0)
        if t_alt > 30:
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = lp_alternation(init_alt, t_alt * 0.6, solver=solver,
                                       max_iters=30000, stall_patience=12)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                a_alt2 = lbgfs_logsumexp(best_seq, rem * 0.4,
                                          T_stages=[5.0, 3.0, 2.0, 1.0, 0.6, 0.3, 0.15, 0.08, 0.04],
                                          maxiter_per_stage=5000)
                try_update(a_alt2, f"lbgfs_alt")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_logsumexp(best_seq, t_fin,
                                     T_stages=[2.0, 1.0, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01],
                                     maxiter_per_stage=8000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.4, 30.0),
                                       n_steps=50000, top_m=5)
        try_update(a_cd2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg2, _ = subgradient_refine(best_seq, min(rem * 0.5, 20.0), n_steps=30000)
        try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
