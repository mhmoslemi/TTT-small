"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated LogSumExp
gradient descent with FFT-based gradient computation, aggressive temperature
annealing, adaptive restarts, and final LP-based polishing.
Key improvements: O(n log n) per iteration via FFT, millions of iterations
possible, and aggressive multi-stage annealing for precise convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

# Global cache for FFT sizes
def _get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def _get_corr_fft_size(n):
    size = 3 * n - 2
    return 1 << (size - 1).bit_length()

def fast_autoconv(a):
    n = len(a)
    fft_size = _get_fft_size(n)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2*n-1]

def fast_cross_corr(w, a):
    """c[j] = sum_k w[k] * a[k-j] for j=0..n-1"""
    n = len(a)
    fft_size = _get_corr_fft_size(n)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W.conj() * A, fft_size)
    return corr[:n]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_obj_and_grad(a, T):
    """Compute LogSumExp objective and gradient (in a-space)."""
    n = len(a)
    b = fast_autoconv(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp
    lse = max_b + T * np.log(sum_exp)
    # Gradient in a-space: 2n * T * cross_corr(w, a)
    c = fast_cross_corr(w, a)
    grad_a = 2.0 * n * T * c
    # Gradient in z-space (log-domain): a * (grad_a - dot(a, grad_a))
    dot_a = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_a)
    # The actual objective value (2n * max_b / sum^2), but for normalized a:
    obj = 2.0 * n * lse  # This is the smoothed objective (not exactly 2n*max_b)
    return obj, grad_z

def coordinate_descent_polish(a0, time_limit, n_steps=100000):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autoconv(a)
        max_idx = np.argmax(b)
        max_b = b[max_idx]
        
        # Find top contributors to the max peak
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        
        # Try reducing the top contributor
        order = np.argsort(contribs)[::-1]
        improved = False
        for oi in order[:8]:
            pos = idx[oi]
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.995, 0.999, 0.985]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1.0 - factor)
                if diff < 1e-15:
                    break
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score

    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        if time.monotonic() - t0 > max_time:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                b_val = b_val / s
                return b_val
        except Exception:
            continue
    return None

def lp_polish(a0, time_limit):
    """Polish using a few LP best-response steps with Nesterov averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b_new = lp_best_response(y, max_time=3.0)
        if b_new is None:
            break
        it += 1
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        a = a / sa
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 2:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 3:
        a = (1 - x) ** 1.2 + 0.1
    elif choice == 4:
        a = 1.0 - 0.3 * x + 0.1 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) * (1.0 / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
