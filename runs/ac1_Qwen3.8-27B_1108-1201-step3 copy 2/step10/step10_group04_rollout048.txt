```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a, solver):
    """Solve LP: min M s.t. T(a) @ b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
```

The key insight from the literature on approximation algorithms for Minimax is that the time-average of iterates in a regret-minimizing algorithm converges to the game value. Specifically:


```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fixed Nesterov-accelerated LP best-response (main workhorse)
2. Time-averaged iterates for better convergence guarantees
3. Multi-length exploration around the parent sequence length
4. Online Mirror Descent with adaptive step for polishing
5. Direct subgradient local search for final refinement
Key improvements: correct Nesterov acceleration, time-averaging,
adaptive LP constraint reduction, and multi-scale exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_reduced(a, top_k=50):
    """Build reduced convolution matrix using only top_k most important rows."""
    n = len(a)
    b = fast_convolve(a)
    sorted_idx = np.argsort(b)[::-1][:top_k]
    size = 2 * n - 1
    T = np.zeros((len(sorted_idx), n))
    for i, k in enumerate(sorted_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a[k - idx]
    return T, sorted_idx


def evaluate_local(a):
    """Locally evaluate the objective (scale-invariant)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return 1e10
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_fast(a, solver, use_reduced=False, top_k=60):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if use_reduced and n > 200:
        T, _ = build_conv_matrix_reduced(a, top_k)
    else:
        T = build_conv_matrix(a)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, use_reduced=True, top_k=60):
    """
    Nesterov-accelerated LP best-response with time-averaging.
    Properly implements Nesterov's acceleration scheme.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()  # Nesterov lookahead point
    a_prev = a.copy()
    t_k = 1.0  # Nesterov momentum parameter

    best_a = a.copy()
    best_score = evaluate_local(a)

    # Time-averaged iterate
    a_avg = a.copy()
    avg_count = 1

    t0 = time.monotonic()
    iter_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        b, M = solve_lp_fast(y, solver, use_reduced=use_reduced, top_k=top_k)
        if b is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                # Try without reduced constraints
                b, M = solve_lp_fast(y, solver, use_reduced=False)
                if b is None:
                    break
                lp_fail_count = 0
            continue

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new

        # Extrapolate: y_new = b + alpha * (b - a_prev)
        y_new = b + alpha * (b - a_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b.copy()
        else:
            y_new = y_new / sy

        a_prev = a.copy()
        a = b.copy()
        y = y_new
        t_k = t_new

        # Update time average
        a_avg = a_avg + (a - a_avg) / avg_count
        avg_count += 1

        # Evaluate current iterate
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Evaluate time average (every 10 iterations for speed)
        if iter_count % 10 == 0:
            score_avg = evaluate_local(a_avg)
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()

        # Evaluate y (lookahead) every 50 iterations
        if iter_count % 50 == 0:
            score_y = evaluate_local(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        # Periodic reporting
        if iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={iter_count} ({elapsed:.0f}s): score={best_score:.6f}")

        # Adaptive: if toasting, try reducing top_k
        if iter_count % 5000 == 0 and iter_count > 5000:
            top_k = max(30, top_k - 10)

    # Final check of time average
    score_avg = evaluate_local(a_avg)
    if score_avg < best_score:
        best_score = score_avg
        best_a = a_avg.copy()

    return best_a, best_score, iter_count


def mirror_descent_optimize(a0, time_limit, n_active_frac=0.05):
    """Online Mirror Descent with backtracking in log-domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Backtracking line search
        alpha = lr
        improved = False
        for bt in range(8):
            z_new = z - alpha * g
            z_new_shifted = z_new - z_new.max()
            exp_z = np.exp(z_new_shifted)
            a_new = exp_z / exp_z.sum()
            score_new = evaluate_local(a_new)

            if score_new < best_score - 1e-15:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.85, 0.05)
            if no_improve >= 50:
                rng = np.random.default_rng(step * 31 + 7)
                noise = rng.random(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = np.log(np.maximum(a, 1e-300))
                no_improve = 0
                lr = 0.5

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=100000):
    """Direct subgradient method with momentum for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    v = np.zeros(n)  # momentum
    momentum = 0.9
    t0 = time.monotonic()
    lr0 = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.array([int(np.argmax(b))])
        # Add nearby active peaks
        k_best = int(np.argmax(b))
        js = max(0, k_best - n + 1)
        je = min(n - 1, k_best)

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Project gradient onto tangent space (sum to 0)
        g = g - g.mean()

        # Heavy-ball momentum
        v = momentum * v - lr0 / (1.0 + step * 0.01) * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                # Partial step
                v = momentum * v * 0.5

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_start_lp(a_init, n, time_budget, solver, label=""):
    """Run LP-based optimization with multiple strategies."""
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()

    # Phase 1: Nesterov LP with reduced constraints (fast)
    rem = time_budget - (time.monotonic() - t0)
    t1 = rem * 0.5
    if t1 > 5:
        a1, s1, iters = nesterov_lp_optimize(best_a, t1, solver, use_reduced=True, top_k=80)
        if s1 < best_score:
            best_score = s1
            best_a = a1
            print(f"  [{label}] Phase1 (reduced LP, {iters} iters): {best_score:.6f}")

    # Phase 2: Nesterov LP with full constraints (more accurate)
    rem = time_budget - (time.monotonic() - t0)
    t2 = rem * 0.5
    if t2 > 5:
        a2, s2, iters = nesterov_lp_optimize(best_a, t2, solver, use_reduced=False)
        if s2 < best_score:
            best_score = s2
            best_a = a2
            print(f"  [{label}] Phase2 (full LP, {iters} iters): {best_score:.6f}")

    # Phase 3: Mirror descent polishing
    rem = time_budget - (time.monotonic() - t0)
    t3 = rem * 0.5
    if t3 > 3:
        a3, s3 = mirror_descent_optimize(best_a, t3, n_active_frac=0.03)

    candidate_lengths = sorted(set(candidate_lengths))

    for cl in candidate_lengths[:4]:
        rem = time_limit - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 180.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={cl}, budget={t_alloc:.0f}s")
        init = interpolate_sequence(best_seq, cl)
        a_out, s_out = multi_start_lp(init, cl, t_alloc, solver, label=f"n{cl}")
        try_update(a_out, f"long_n{cl}")

    # === PHASE 3: Second pass at best length with more time ===
    print(f"\n=== Phase 3: Extended LP at n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 60:
        t3 = rem * 0.35
        print(f"  Budget: {t3:.0f}s")
        # Run Nesterov LP with full constraints for extended time
        a_ext, s_ext, iters = nesterov_lp_optimize(best_seq, t3, solver, use_reduced=False)
        try_update(a_ext, f"ext_fullLP_n{len(best_seq)}")

        # Also try reduced with smaller top_k for speed
        rem = time_limit - time.monotonic()
        if rem > 30:
            t3b = rem * 0.5
            a_ext2, s_ext2, iters2 = nesterov_lp_optimize(best_seq, t3b, solver, use_reduced=True, top_k=40)
            try_update(a_ext2, f"ext_redLP_n{len(best_seq)}")

    # === PHASE 4: Mirror descent + subgradient polish ===
    print(f"\n=== Phase 4: MD + SG polish (best={best_score:.6f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 30:
        t4 = rem * 0.4
        if t4 > 10:
            a_md, s_md = mirror_descent_optimize(best_seq, t4, n_active_frac=0.02)
            try_update(a_md, "final_MD")

        rem = time_limit - time.monotonic()
        if rem > 20:
            t5 = rem * 0.5
            a_sg, s_sg = subgradient_polish(best_seq, t5)
            try_update(a_sg, "final_SG")

    # === PHASE 5: Final verification ===
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    return [float(x) for x in best_seq]
```
