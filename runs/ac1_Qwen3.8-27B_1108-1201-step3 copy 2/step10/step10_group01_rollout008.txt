Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using extended FISTA-style LP alternation
with systematic n-sweep, power-smoothed L-BFGS refinement, peak equalization

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_T(n):
    """Build (2n-1) x n index structure for convolution matrix rows."""
    # For a fixed 'a', T[k,j] = a[k-j] if 0<=k-j<n, else 0
    # We precompute the row indices
    rows = []
    cols = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
    return np.array(rows), np.array(cols)


def solve_lp_response(a_fixed, solver, T_rows=None, T_cols=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    if T_rows is None:
        T_rows, T_cols = build_T(n)
    # T @ b where T[k,j] = a[k-j] if valid
    # Build as sparse-style: we need T @ b_var <= M_var
    # T is (2n-1) x n, T[k,j] = a[k-j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for j in range(n):
        T[T_rows[j*n:(j+1)*n], T_cols[j*n:(j+1)*n]] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val /= s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val /= s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

def lp_alternate(a0, time_limit, solver, max_iters=100000, verbose=True):
    """Nesterov-accelerated LP alternation with averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_fast(y, solver, n)
        if b_new is None:
            stall_count += 1
            if stall_count > 40:
                break
            continue
        stall_count = 0

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if (it + 1) % 2000 == 0:
            if best_score < prev_score - 1e-10:
                stall_count = 0
            else:
                stall_count += 1

            if stall_count >= 15:
                # Restart with perturbation
                noise_scale = 0.01 / np.sqrt(max(1, stall_count))
                y = y + rng.standard_normal(n) * noise_scale
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            prev_score = best_score
            if verbose:
                print(f"    LP it={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.10f} M={M_b:.10f}")

    return best_a, best_score


def power_smooth_obj(z, n, p):
    """Power-smoothed objective: (sum conv^p)^(1/p) * 2n."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    # Clip small negative values from FFT
    b = np.maximum(b, 0)
    if p == 1:
        val = b.sum()
    else:
        max_b = b.max()
        if max_b < 1e-30:
            return 0.0
        scaled = b / max_b
        val = max_b * (scaled ** p).sum() ** (1.0 / p)
    return 2.0 * n * val


def power_smooth_grad(z, n, p):
    """Gradient of power-smoothed objective."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    b = np.maximum(b, 0)
    max_b = b.max()
    if max_b < 1e-30:
        return np.zeros(n)
    scaled = b / max_b
    if p == 1:
        w = np.ones(len(b)) / len(b)
    else:
        w = scaled ** (p - 1) / (scaled ** p).sum()
        w = w / max_b * (max_b ** p) ** (1.0/p) * max_b / (scaled**p).sum()
        # Actually let's be more careful
        # f = (sum b_i^p)^(1/p)
        # df/db_i = (1/p) * (sum b^p)^(1/p - 1) * p * b_i^(p-1)
        #         = (sum b^p)^(1/p - 1) * b_i^(p-1)
        #         = b_i^(p-1) / (sum b^p)^(1 - 1/p)
        #         = b_i^(p-1) / (max_b^p * sum scaled^p)^(1-1/p)
        #         = b_i^(p-1) / (max_b * (sum scaled^p)^(1-1/p))
        sum_p = (scaled ** p).sum()
        denom = max_b * (sum_p ** (1.0 - 1.0/p)) if sum_p > 0 else 1.0
        w = (b ** (p - 1)) / denom
        w = w / w.sum()

    # Gradient of f = 2n * (sum conv^p)^(1/p) w.r.t. a
    # conv(a,a)[k] = sum_j a[j]*a[k-j]
    # d(conv[a,a]/da[i]) = 2 * a[k-i] for the k-th component (for i in valid range)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Normalize: gradient of f(a/||a||) where a = exp(z)/sum(exp(z))
    # Use the standard log-barrier style: grad_z = a * (grad_a / a - dot(grad_a, a) / sum(a^2))
    # Actually simpler: we parameterize a = softmax(z), so da/dz = a*(I - a^T)
    # d f / d z = a * (grad_a - a * dot(grad_a, a) / dot(a, a))
    # But since sum(a) = 1 is our constraint, we use:
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def lbgfs_power_smooth(a0, time_limit, p_stages=None):
    """L-BFGS with power smoothing, multi-stage p increase."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if p_stages is None:
        p_stages = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for p in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time.monotonic() - t0
        if time_limit - remaining < 3:
            break

        def func(z):
            return power_smooth_obj(z, n, p)

        def jac(z):
            return power_smooth_grad(z, n, p)

        iters = max(30, 5000 // len(p_stages))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]

    def lse_obj(z, T):
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        b = np.maximum(b, 0)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse

    def lse_grad(z, T):
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        b = np.maximum(b, 0)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-16:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    n_stages = len(T_stages)
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if T_stages.index(T) if T in T_stages else 0 != 0:
            pass
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(20, 8000 // n_stages)
        try:
            res = minimize(
                fun=lambda z: lse_obj(z, T), x0=z,
                jac=lambda z: lse_grad(z, T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_equalize_cd(a0, time_limit, n_steps=100000):
    """Coordinate descent that specifically targets peak equalization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    lr = 0.02
    rng = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b = np.maximum(b, 0)
        max_b = b.max()

        # Find top peaks (within 0.1% of max)
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # For each active peak, compute gradient contribution
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Subgradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.9
                stall = 0
            if lr < 1e-8:
                # Try random perturbation
                a_pert = a + rng.standard_normal(n) * 0.005
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sp = 2.0 * n * fast_convolve(a_pert).max()
                if sp < best_score:
                    a = a_pert
                    best_a = a.copy()
                    best_score = sp
                    lr = 0.02
                    stall = 0
                else:
                    if step % 5000 == 0:
                        break

        if step % 10000 == 0:
            print(f"    PeakEq {time.monotonic()-t0:.0f}s: score={best_score:.10f} lr={lr:.6f}")

    return best_a, best_score


def coordinate_descent_shave(a0, time_limit, n_steps=50000):
    """Direct coordinate descent: reduce elements contributing to max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b = np.maximum(b, 0)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        # Try reducing each top contributor
        order = np.argsort(contribs)[::-1][:8]
        for oi in order:
            pos = int(idx[oi])
            if a[pos] < 1e-14:
                continue
            for factor in [0.995, 0.99, 0.985, 0.98, 0.97]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1.0 - factor)
                if diff < 1e-16:
                    break
                # Redistribute to other elements proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top 3
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-14:
                    a_trial[pos] *= 0.998
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            if stall > 500:
                break

        if step % 10000 == 0:
            print(f"    CDShave {time.monotonic()-t0:.0f}s: score={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = (1 - x) ** 1.2 + 0.05
    elif choice == 5:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.exp(-3 * (x - 0.5) ** 2 / 0.15) + 0.1
    elif choice == 7:
        # Bimodal
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 8:
        # Power law
        a = (1 - x) ** 0.5 + 0.1
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.random()
            width = 0.1 + 0.2 * rng.random()
            amp = 0.5 + rng.random()
            a += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def quick_lp(a0, solver, n_lp_iters=20):
    """Quick LP alternation for evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    y = a.copy()
    for it in range(n_lp_iters):
        b_new, M_b = solve_lp_fast(y, solver, n)
        if b_new is None:
            break
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
    return a
