import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (sum-normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, M_bound, solver=None):
    """Given fixed a (sum=1) and M upper bound, find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if M_bound is not None:
        constraints.append(M_var <= M_bound)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    # Fallback solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:

    return None, None


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

                            nesterov=True, bb_adapt=True, max_iters=5000000):

    Mirror descent with Barzilai-Borwein step size and Nesterov acceleration.

    a = normalize_nonneg(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient(y, active)

            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.998

        prev_g = g.copy()

        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.003 + 0.003 * min(1.0, no_improve / 2000.0)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = normalize_nonneg(a)
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=30000):
    """Nesterov LP alternating for refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_rounds = 0
    prev_best = best_score
    rng = np.random.default_rng(42424)
    M_current = fast_convolve(a).max()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, M_current * 1.01, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize_nonneg(y)

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize_nonneg(a)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-11:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            if stall_rounds >= 15:
                noise_scale = 0.01 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = normalize_nonneg(y + noise)
                stall_rounds = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_and_grad(z, T):
    """Compute LogSumExp objective and gradient in softmax parameterization."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()

    size = 2 * n - 1
    grad_a = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    logsumexp = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35,
                    0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.005]

    n_stages = len(T_stages)
    # Allocate more iterations to later (lower T) stages
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Increase iterations for lower temperature stages
        T_frac = T / T_stages[0]
        iters_this = int(iters_per_stage * (1.5 / (1.0 + T_frac)))

        def func_lse(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            o, _ = logsumexp_obj_and_grad(zz, T)
            return o

        def jac_lse(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            _, g = logsumexp_obj_and_grad(zz, T)
            return g

        try:
            res = minimize(fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                           options={'maxiter': iters_this, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_this * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def dual_guided_refine(a0, time_limit, n_steps=100000):
    """
    Projected gradient descent with dual variable (active set) tracking.
    Maintains a soft active set and uses it for more precise subgradient estimation.
    """
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.02
    active_weights = None
    stagnation = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Soft active set using exponential weighting
        if active_weights is None or stagnation > 50:
            # Use a wider window for active set
            threshold = max_b * 0.9999
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(b))])
            # Exponential weights based on proximity to max
            w = np.exp((b[active] - max_b) * 50.0)
            w = w / w.sum()
            active_weights = (active, w)

        active_idx, active_w = active_weights

        # Weighted subgradient
        g = np.zeros(n)
        for i, (k, wi) in enumerate(zip(active_idx, active_w)):
            if wi < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wi * a[k - idx]

        # Projected gradient step on the simplex
        g_proj = g - g.mean()
        a_new = a - eta * a * g_proj
        a_new = normalize_nonneg(a_new)

        # Adaptive step size
        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < score:
            a = a_new
            eta = min(eta * 1.01, 0.1)
        else:
            eta = max(eta * 0.9, 1e-5)
            # Try smaller step
            a_new2 = a - eta * 0.5 * a * g_proj
            a_new2 = normalize_nonneg(a_new2)
            if 2.0 * n * fast_convolve(a_new2).max() < score:
                a = a_new2

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DualRefine step={step+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

        if stagnation > 5000:
            break

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * 0.99995
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9999, 0.9997, 0.9995, 0.999, 0.998, 0.995, 0.99]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:4]]
                for pos in top_positions:
                    if a[pos] > 1e-14:
                        a_trial[pos] *= 0.9995
                a_trial = normalize_nonneg(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 2000:
                        break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def window_init(n, window_type=0, rng=None):
    """Create window-function initialized sequence."""
    x = np.linspace(0, 1, n)
    if window_type == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif window_type == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif window_type == 2:
        a = np.exp(-4 * x * (1 - x))
    elif window_type == 3:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif window_type == 4:
        # Blackman
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 5:
        # Flat top-ish: elevated in middle
        a = 0.3 + 0.7 * np.exp(-4 * (x - 0.5) ** 2)
    elif window_type == 6:
        # Asymmetric: heavier on one side
        a = np.exp(-3 * x) * (1 + 0.3 * np.sin(2 * np.pi * x))
        a = np.maximum(a, 0)
    elif window_type == 7:
        # Two-hump
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)

    if rng is not None and window_type >= 0:
        # Add small random perturbation for diversity
        a = a + rng.standard_normal(n) * 0.01
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sparse_selector_init(n, m, rng):
    """Create a sparse sequence with m non-zero entries."""
    a = np.zeros(n)
    positions = rng.choice(n, size=m, replace=False)
    # Use random weights
    weights = rng.standard_normal(m)
    weights = np.abs(weights)
    weights = weights / weights.sum()
    a[positions] = weights
    return a


def optimize_candidate(a0, time_budget, solver=None):
    """Run the full optimization pipeline on a single candidate."""
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    def try_update(cand):
        nonlocal best_a, best_score
        cand = normalize_nonneg(cand)
        sc = 2.0 * n * fast_convolve(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 1: Mirror descent
    if rem > 20:
        t_md = min(rem * 0.3, 200.0)
        a_md, _ = mirror_descent_optimize(a, t_md, eta0=0.05, active_frac=0.999)
        try_update(a_md)
        a = best_a.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 2: LP Alternating
    if rem > 30:
        t_lp = min(rem * 0.25, 250.0)
        a_lp, _ = nesterov_lp_refine(a, t_lp, solver=solver, n_max_iters=20000)
        try_update(a_lp)
        a = best_a.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 3: LogSumExp L-BFGS (fine temperature schedule)
    if rem > 20:
        t_lse = min(rem * 0.35, 350.0)
        T_stages = [15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4,
                    0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.008, 0.004]
        a_lse = logsumexp_lbgfs(a, t_lse, T_stages=T_stages, maxiter_per_stage=8000)
        try_update(a_lse)
        a = best_a.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 4: Dual-guided refinement
    if rem > 15:
        t_dual = min(rem * 0.15, 120.0)
        a_dual, _ = dual_guided_refine(a, t_dual, n_steps=80000)
        try_update(a_dual)
        a = best_a.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 5: Coordinate descent polish
    if rem > 10:
        t_cd = min(rem * 0.15, 80.0)
        a_cd, _ = coordinate_descent_refine(a, t_cd, n_steps=40000)
        try_update(a_cd)
        a = best_a.copy()

    rem = time_budget - (time.monotonic() - t0)

    # Stage 6: Final fine LogSumExp
    if rem > 15:
        t_fin = min(rem * 0.2, 60.0)
        T_fin = [1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
        a_fin = logsumexp_lbgfs(a, t_fin, T_stages=T_fin, maxiter_per_stage=3000)
        try_update(a_fin)

    rem = time_budget - (time.monotonic() - t0)

    # Stage 7: Final coordinate descent
    if rem > 5:
        a_polish, _ = coordinate_descent_refine(best_a, rem * 0.5, n_steps=20000)
        try_update(a_polish)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization searching over different n values and initializations.
    Uses mirror descent, LP alternating, LogSumExp L-BFGS, dual-guided refinement,
    and coordinate descent in a pipeline for each candidate.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')
    best_n = 0

    def try_update(a, label):
        nonlocal best_seq, best_score, best_n
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Load parent sequence ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
            best_n = len(parent)
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = window_init(n_init, 0)
        best_score = evaluate_local(best_seq)
        best_n = n_init
        print(f"Default init: n={n_init}, score={best_score:.10f}")

    # === Phase 1: Optimize parent at its native n ===
    print(f"\n=== Phase 1: Optimize parent (n={best_n}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 50:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc = optimize_candidate(best_seq, t_alloc, solver=solver)
        try_update(a_opt, "parent_opt")

    # === Phase 2: Try different n values ===
    print(f"\n=== Phase 2: N-sweep (best so far={best_score:.10f}) ===")
    rem = deadline - time.monotonic()

    # Determine which n values to try
    n_candidates = [800, 1000, 1200, 1400, 1600, 1800, 2000, 2400]
    # Filter out n too close to best_n (already tried)
    n_candidates = [n for n in n_candidates if abs(n - best_n) > 50]
    # Sort by priority: prefer n near 1200-2000
    n_candidates.sort(key=lambda n: abs(n - 1500))

    # Allocate time per n based on remaining budget
    n_to_try = n_candidates[:4]  # Try at most 4 different n values
    if rem > 0 and n_to_try:
        per_n_time = min(rem * 0.25 / len(n_to_try), 250.0)
        print(f"  Trying n values: {n_to_try}, {per_n_time:.0f}s each")

        for n_try in n_to_try:
            if time.monotonic() > deadline:
                break
            rem_this = deadline - time.monotonic()
            if rem_this < 30:
                break

            # Interpolate parent to this n
            a_interp = interpolate_sequence(best_seq, n_try)
            # Also try a few window inits
            best_for_n = a_interp.copy()
            sc_for_n = 2.0 * n_try * fast_convolve(a_interp).max()

            for wt in [0, 2, 5]:
                a_w = window_init(n_try, wt, rng=rng)
                sc_w = 2.0 * n_try * fast_convolve(a_w).max()
                if sc_w < sc_for_n:
                    sc_for_n = sc_w
                    best_for_n = a_w.copy()

            print(f"  n={n_try}: init score={sc_for_n:.10f}")
            t_this = min(per_n_time, rem_this - 10)
            if t_this > 30:
                a_opt, sc = optimize_candidate(best_for_n, t_this, solver=solver)
                try_update(a_opt, f"n={n_try}")
                # If this n is better, also try sparse init at this n
                if best_n != n_try and rem_this > 60:
                    m_sparse = max(100, n_try // 4)
                    a_sparse = sparse_selector_init(n_try, m_sparse, rng)
                    a_sp, sc_sp = optimize_candidate(a_sparse, t_this * 0.4, solver=solver)
                    try_update(a_sp, f"n={n_try}_sparse")

    # === Phase 3: Re-optimize best with more time ===
    print(f"\n=== Phase 3: Re-optimize best (n={best_n}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.30, 400.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc = optimize_candidate(best_seq, t_alloc, solver=solver)
        try_update(a_opt, "reopt")

    # === Phase 4: Diversity-based exploration ===
    print(f"\n=== Phase 4: Diversity search (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_div = min(rem * 0.20, 200.0)
        n_div = len(best_seq)
        per_candidate = max(20.0, t_div / 6)

        for trial in range(6):
            if time.monotonic() > deadline:
                break
            rem_this = deadline - time.monotonic()
            if rem_this < 25:
                break

            # Generate diverse candidate
            choice = rng.integers(0, 4)
            if choice == 0:
                # Perturbed best
                noise = rng.standard_normal(n_div) * 0.05
                a_div = normalize_nonneg(best_seq + noise)
            elif choice == 1:
                # Different window
                a_div = window_init(n_div, int(rng.integers(0, 8)), rng=rng)
            elif choice == 2:
                # Blend best with window
                a_w = window_init(n_div, int(rng.integers(0, 8)))
                alpha = rng.uniform(0.2, 0.8)
                a_div = normalize_nonneg(alpha * best_seq + (1 - alpha) * a_w)
            else:
                # Sparse with random structure
                m = max(50, n_div // int(rng.integers(3, 6)))
                a_div = sparse_selector_init(n_div, m, rng)

            t_this = min(per_candidate, rem_this - 10)
            if t_this > 20:
                a_opt, sc = optimize_candidate(a_div, t_this, solver=solver)
                try_update(a_opt, f"div_{trial}")

    # === Phase 5: Final intensive refinement ===
    print(f"\n=== Phase 5: Final refinement (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = rem * 0.4
        print(f"  Budget: {t_final:.0f}s")
        # LogSumExp with very fine schedule
        T_final = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.02, 0.01,
                   0.006, 0.003, 0.0015, 0.0008]
        a_fin = logsumexp_lbgfs(best_seq, t_final * 0.7, T_stages=T_final,
                                maxiter_per_stage=10000)
        try_update(a_fin, "final_lse")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_cd, _ = coordinate_descent_refine(best_seq, rem * 0.5, n_steps=30000)
            try_update(a_cd, "final_cd")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
        # One more check with the actual evaluator
        try:
            eval_result = evaluate_sequence([float(x) for x in best_seq])
            print(f"  evaluate_sequence confirms: {eval_result:.10f}")
            if eval_result < best_score:
                best_score = eval_result
        except Exception:
            pass

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
