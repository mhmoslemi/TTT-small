import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return 'ECOS'

def build_conv_matrix_rows(a, k_list):
    """Build rows of the convolution matrix for specific lags."""
    n = len(a)
    T = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            T[row, i] = a[k - i]
    return T

def solve_lp_best_response(a_fixed, k_list=None, solver=None):
    """Solve the LP: min M s.t. T_k @ b <= M, sum(b)=1, b>=0 for k in k_list."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    
    if k_list is None:
        k_list = list(range(2 * n - 1))
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, k_list)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def golden_section_search(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=40)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 10 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            # Use momentum update
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        # Nesterov lookahead
        a_ys = a + beta * (a - a_vel)
        a_vel = a
        a_ys = np.maximum(a_ys, 0)
        ys_sum = a_ys.sum()
        if ys_sum > 1e-10:
            a_ys = a_ys / ys_sum

        for cand in [a, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP-M n={n} it={iters}: best={best_score:.8f}")

        if no_improve > 50:
            beta = max(0.5, beta * 0.95)
            no_improve = 0

    return best_a, best_score, iters

def frequency_domain_refinement(a0, time_limit, verbose=False):
    """Reduce autocorrelation peaks by adjusting in frequency domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    lr = 0.01
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = rfft(a_padded)
        power = np.abs(A)**2
        max_power = power.max()
        
        # Create a "mask" that reduces high-power frequency bins
        # This is like the SOLVED/active-set method for PAPR
        threshold = max_power * 0.9
        mask = np.ones(fft_size // 2 + 1)
        # Reduce the amplitude at high-power bins
        for k in range(len(mask)):
            if power[k] > threshold:
                # Reduce amplitude proportionally
                mask[k] = np.sqrt(threshold / power[k])
        
        A_new = A * mask
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            continue
        a_new = a_new / cs
        
        # Blend with original
        blend = 0.5
        a_candidate = (1 - blend) * a + blend * a_new
        a_candidate = np.maximum(a_candidate, 0)
        a_candidate = a_candidate / a_candidate.sum()
        
        sc = compute_score(a_candidate)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_candidate.copy()
            a = a_candidate
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            lr *= 0.9
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            print(f"    FDR it={iters}: best={best_score:.8f}")

    return best_a, best_score

def pairwise_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent that updates pairs of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            j = i + 1
            if j >= n:
                # Handle last element
                v_i = a[i]
                if v_i < 1e-14:
                    continue
                # Just do single coordinate for the last one
                k_lo_i = i
                k_hi_i = min(i + n - 1, 2 * n - 2)
                unaff_max = 0.0
                if k_lo_i > 0:
                    unaff_max = max(unaff_max, float(conv[:k_lo_i].max()))
                if k_hi_i < 2 * n - 2:
                    unaff_max = max(unaff_max, float(conv[k_hi_i + 1:].max()))
                partners = np.arange(k_lo_i, k_hi_i + 1) - i
                a_partners = a[partners].copy()
                a_partners[i - i] = 0  # Remove self
                has_quad = (2 * i >= k_lo_i) and (2 * i <= k_hi_i)
                n_aff = k_hi_i - k_lo_i + 1
                c_vals = np.empty(n_aff)
                for idx in range(n_aff):
                    k = k_lo_i + idx
                    if k == 2 * i:
                        c_vals[idx] = conv[k] - v_i * v_i
                    else:
                        c_vals[idx] = conv[k] - 2.0 * v_i * a_partners[idx]
                if has_quad:
                    quad_idx = 2 * i - k_lo_i
                v = v_i
                def _score_single(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                               _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                               _v=v, _n=n):
                    ns = _cs + (x - _v)
                    if ns < 0.01:
                        return 1e10
                    nv = _c + 2.0 * x * _ap
                    if _hq:
                        nv = nv.copy()
                        nv[_qi] = _c[_qi] + x * x
                    nm = max(_qm, float(nv.max()))
                    return 2.0 * _n * nm / (ns * ns)
                upper = max(3.0 * v, 5.0 / n)
                best_x = golden_section_search(_score_single, 0.0, upper, max_iter=40)
                if abs(best_x - v) > 1e-13:
                    delta = best_x - v
                    a[i] = best_x
                    current_sum += delta
                    changes = 2.0 * delta * a_partners
                    if has_quad:
                        changes = changes.copy()
                        changes[quad_idx] = delta * (2.0 * v + delta)
                    conv[k_lo_i:k_hi_i + 1] += changes
                    cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                continue

            # Pairwise update for (i, j)
            v_i = a[i]
            v_j = a[j]
            if v_i < 1e-14 and v_j < 1e-14:
                continue
            
            # For pairwise, we need to optimize over (x_i, x_j) simultaneously
            # This is a 2D problem. Use a grid search + refinement.
            # For speed, we do a 1D search along the direction that reduces the objective
            
            # First try: scale both by the same factor
            # Then try: transfer mass between i and j
            
            best_score_pair = best_score
            best_xi, best_xj = v_i, v_j
            
            # Try scaling
            for scale in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                xi_new = v_i * scale
                xj_new = v_j * scale
                xi_new = max(xi_new, 0)
                xj_new = max(xj_new, 0)
                
                # Compute new convolution
                di = xi_new - v_i
                dj = xj_new - v_j
                conv_new = conv.copy()
                # Update from i
                k_lo_i = i
                k_hi_i = min(i + n - 1, 2 * n - 2)
                partners_i = np.arange(k_lo_i, k_hi_i + 1) - i
                a_p_i = a[partners_i].copy()
                # Remove j's contribution (will be handled separately)
                if j in partners_i:
                    a_p_i[partners_i == j] = 0
                has_quad_i = (2 * i >= k_lo_i) and (2 * i <= k_hi_i)
                changes_i = 2.0 * di * a_p_i
                if has_quad_i:
                    qi = 2 * i - k_lo_i
                    changes_i = changes_i.copy()
                    changes_i[qi] = di * (2.0 * v_i + di)
                conv_new[k_lo_i:k_hi_i + 1] += changes_i
                
                # Update from j
                k_lo_j = j
                k_hi_j = min(j + n - 1, 2 * n - 2)
                partners_j = np.arange(k_lo_j, k_hi_j + 1) - j
                a_p_j = a[partners_j].copy()
                if i in partners_j:
                    a_p_j[partners_j == i] = 0
                has_quad_j = (2 * j >= k_lo_j) and (2 * j <= k_hi_j)
                changes_j = 2.0 * dj * a_p_j
                if has_quad_j:
                    qj = 2 * j - k_lo_j
                    changes_j = changes_j.copy()
                    changes_j[qj] = dj * (2.0 * v_j + dj)
                conv_new[k_lo_j:k_hi_j + 1] += changes_j
                
                # Cross term: di * a[j] and dj * a[i] contributions
                # Actually this is already handled if we include each other's values
                # Let me redo this more carefully...
                
                # Actually, the convolution change from updating a[i] by di:
                # For lag k: a[k-i] * di + a[i] * a[k-i]... 
                # This is getting complicated. Let me just compute the score directly.
                a_trial = a.copy()
                a_trial[i] = xi_new
                a_trial[j] = xj_new
                sc = compute_score(a_trial)
                if sc < best_score_pair - 1e-14:
                    best_score_pair = sc
                    best_xi, best_xj = xi_new, xj_new
            
            # Try transferring mass
            for frac in [0.1, 0.2, 0.3, 0.5, -0.1, -0.2, -0.3, -0.5]:
                transfer = frac * min(v_i, v_j)
                xi_new = max(v_i + transfer, 0)
                xj_new = max(v_j - transfer, 0)
                a_trial = a.copy()
                a_trial[i] = xi_new
                a_trial[j] = xj_new
                sc = compute_score(a_trial)
                if sc < best_score_pair - 1e-14:
                    best_score_pair = sc
                    best_xi, best_xj = xi_new, xj_new
            
            if best_xi != v_i or best_xj != v_j:
                a[i] = best_xi
                a[j] = best_xj
                current_sum = a.sum()
                conv = fast_convolve(a)
                if best_score_pair < best_score:
                    best_score = best_score_pair
                    best_a = a.copy()
                improved = True
                # Recompute to keep conv consistent
                conv = fast_convolve(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    PCD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def subgradient_descent_v2(a0, time_limit, verbose=False):
    """Improved subgradient with adaptive focus and restarting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    iters = 0
    a_avg = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())

        # Focus on top peaks with adaptive temperature
        k_act = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.002 * max_val + 1e-10
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_avg = (iters * a_avg + a_new) / (iters + 1)
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new
        
        # Adaptive learning rate
        if no_improve > 0:
            lr *= (1.0 - 0.01 * min(no_improve, 50))
        
        if no_improve > 50:
            lr = max(0.001, lr * 0.5)
            no_improve = 0
        if no_improve > 200:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_optimal_shape(n, rng):
    """Initialize with a shape that approximates the optimal solution."""
    # The optimal shape tends to be slightly asymmetric
    # with a "bump" that's not too narrow or too wide
    t = np.linspace(0, 1, n)
    # Try a few different shapes
    shape_type = rng.integers(0, 4)
    if shape_type == 0:
        # Bell-like
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.15, 0.35)
        a = np.exp(-((t - center) / width) ** 2)
    elif shape_type == 1:
        # Two bumps
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.1, 0.2)
        a = 0.5 * np.exp(-((t - c1) / w) ** 2) + 0.5 * np.exp(-((t - c2) / w) ** 2)
    elif shape_type == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    else:
        # Power law decay from center
        center = rng.uniform(0.3, 0.7)
        a = (1 + np.abs(t - center) * 3) ** (-rng.uniform(1, 3))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def build_conv_matrix_rows(a, k_list):
    n = len(a)
    T = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            T[row, i] = a[k - i]
    return T


def solve_lp_best_response(a_fixed, k_list, solver):
    import cvxpy as cp
    n = len(a_fixed)

    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, k_list)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break

                continue

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

    M_val = float(M_var.value)
    return b_val, M_val


def golden_section_search(f, lo, hi, max_iter=50, *args):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c, *args)
    fd = f(d, *args)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c, *args)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d, *args)
    return (lo + hi) * 0.5

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0
    iters = 0
    a_avg = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

        conv = fast_convolve(a)
        max_val = float(conv.max())

        k_act = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.0015 * max_val + 1e-10
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_avg = (iters * a_avg + a_new) / (iters + 1)
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new

        if no_improve > 0:
            lr *= (1.0 - 0.005 * min(no_improve, 100))

        if no_improve > 30:
            lr = max(0.0005, lr * 0.6)
            no_improve = 0
        if no_improve > 150:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_optimal_shape(n, rng):
    t = np.linspace(0, 1, n)
    shape_type = rng.integers(0, 5)
    if shape_type == 0:
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.15, 0.35)
        a = np.exp(-((t - center) / width) ** 2)
    elif shape_type == 1:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.1, 0.2)
        a = 0.5 * np.exp(-((t - c1) / w) ** 2) + 0.5 * np.exp(-((t - c2) / w) ** 2)
    elif shape_type == 2:
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif shape_type == 3:
        center = rng.uniform(0.3, 0.7)
        a = (1 + np.abs(t - center) * 3) ** (-rng.uniform(1, 3))
    else:
        center = rng.uniform(0.3, 0.7)
        sigma = rng.uniform(0.1, 0.25)
        a = np.exp(-((t - center) ** 2) / (2 * sigma ** 2))
        a += 0.3 * np.exp(-((t - (1 - center)) ** 2) / (2 * (sigma * 1.5) ** 2))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Multi-phase optimization:
    1. Multi-start diverse initialization with quick ECD screening
    2. Enhanced LP Fictitious Play with Nesterov momentum (main driver)
    3. Frequency-domain PAPR-style refinement
    4. Subgradient descent with adaptive LR
    5. Multiple ECD polishing rounds
    6. Final LP FP pass + ECD finishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(777)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Phase 1: Diverse initialization + quick screening
    print("\n=== Phase 1: Diverse Init ===")
    t1_start = time.monotonic()

    if parent is not None:
        n_parent = len(parent)
    else:
        n_parent = 1024

    n_values = set()
    n_values.add(n_parent)
    for offset in [-64, -128, 64, 128, 256]:
        alt = n_parent + offset
        if 256 <= alt <= 2500:
            n_values.add(alt)
    for alt in [768, 1024, 1280, 1536, 2048]:
        if 256 <= alt <= 2500:
            n_values.add(alt)
    n_values = sorted(n_values)[:6]

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 120:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_asymmetric(n, rng, 0.3))
        inits.append(init_asymmetric(n, rng, 0.45))
        inits.append(init_flat_spectrum(n, rng))
        inits.append(init_optimal_shape(n, rng))
        inits.append(init_optimal_shape(n, rng))

        for j, cand in enumerate(inits[:5]):
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(1.0, min(4.0, t_left / 6.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{j}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.8f}")
    for s_c, a_c, n_c in candidates[:4]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    n_work = len(best_seq) if best_seq is not None else 1024
    print(f"  Working n={n_work}, best={best_score:.8f}")

    # Phase 2: LP FP with Momentum (primary)
    print(f"\n=== Phase 2: LP FP Momentum (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.35, 250.0)
        print(f"  Budget={t_fp:.0f}s")
        a_fp, s_fp, iters = lp_fictitious_play_momentum(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP-M: {sc:.8f}")

    # Quick ECD after FP
    rem = deadline - time.monotonic() - 500
    if rem > 15:
        t_cd = min(rem * 0.15, 40.0)
        a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd, label="ph2_cd")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ph2_cd: {sc:.8f}")

    # Phase 3: Frequency domain
    print(f"\n=== Phase 3: FDR (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 15:
        t_fdr = min(rem * 0.15, 50.0)
        a_fdr, s_fdr = frequency_domain_refinement(best_seq, t_fdr, verbose=True)
        sc = float(evaluate_sequence(a_fdr.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fdr.copy()
            n_work = len(best_seq)
            print(f"  NEW BEST FDR: {sc:.8f}")

    # Phase 4: Subgradient
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 300
    if rem > 15:
        t_sg = min(rem * 0.2, 60.0)
        a_sg, s_sg = subgradient_descent_v2(best_seq, t_sg, verbose=True)
        sc = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sg.copy()
            print(f"  NEW BEST SG: {sc:.8f}")

    # Phase 5: ECD Polishing (multiple rounds with perturbation)
    print(f"\n=== Phase 5: ECD Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(5):
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            break
        t_cd = min(rem * 0.15, 60.0)
        print(f"  ECD r{rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph5_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            n_work = len(best_seq)
            print(f"  NEW BEST ECD: {sc:.8f}")
        else:
            # Perturb to escape
            sigma = 0.01 + 0.01 * rnd
            perturbed = best_seq + rng.standard_normal(n_work) * sigma
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_pert = min(t_cd * 0.5, 30.0)
            a_cd2, s_cd2, _ = exact_coord_descent(perturbed, t_pert)
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                n_work = len(best_seq)
                print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # Phase 6: Second LP FP round
    print(f"\n=== Phase 6: LP FP Round 2 (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        t_fp2 = min(rem * 0.4, 120.0)
        a_fp2, s_fp2, _ = lp_fictitious_play_momentum(best_seq, t_fp2, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp2.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp2.copy()
            print(f"  NEW BEST FP2: {sc:.8f}")

    # Phase 7: Try different n from current best
    print(f"\n=== Phase 7: Multi-n (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for alt_n in [n_work + 64, n_work - 64, n_work + 128, n_work - 128]:
        if alt_n < 200 or alt_n > 2500:
            continue
        rem = deadline - time.monotonic() - 50
        if rem < 20:
            break
        a_init = interpolate_seq(best_seq, alt_n)
        t_alt = min(rem * 0.15, 40.0)
        a_alt, s_alt, _ = exact_coord_descent(a_init, t_alt, label=f"n{alt_n}")
        sc = float(evaluate_sequence(a_alt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            n_work = alt_n
            print(f"  NEW BEST n={alt_n}: {sc:.8f}")
            # Quick FP on the new n
            rem2 = deadline - time.monotonic() - 40
            if rem2 > 20:
                t_fp3 = min(rem2 * 0.3, 30.0)
                a_fp3, s_fp3, _ = lp_fictitious_play_momentum(best_seq, t_fp3, solver=solver)
                sc3 = float(evaluate_sequence(a_fp3.tolist()))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = a_fp3.copy()
                    print(f"  NEW BEST n={alt_n} FP: {sc3:.8f}")

    # Phase 8: Final ECD
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 15
    if rem > 10:
        t_final = min(rem * 0.5, 40.0)
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_final, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    # Last second pass
    rem = deadline - time.monotonic() - 3
    if rem > 3:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.8, 10.0))
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
