    """Compute corr[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j=0..n-1"""
    n = len(a)
    a_rev = a[::-1]

    conv = irfft(A * W, fft_size)
    # Extract positions n-1 to 2n-2

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated log-sum-exp smooth optimization with L-BFGS
2. Fourier series parameterization for reduced-dimension optimization
3. Direct coordinate descent for final polish
4. Multi-n exploration
Key innovations: O(n log n) gradient via FFT cross-correlation,
Fourier parameterization for precise low-dim optimization,
and aggressive temperature annealing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_corr(a, w):
    """
    Compute corr[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j=0..n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    a_rev = a[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    return conv[n-1:n-1+n]


def logsumexp_obj_grad_z(z, T, n):
    """
    Compute objective and gradient in z-space where a = softmax(z).
    Objective: n/T * log(sum_k exp(T * conv(a,a)_k / T))
             = n * logsumexp(conv(a,a))  [soft max approximation]
    Actually: we want (1/T) * log(sum_k exp(T * conv_k)) as T->inf gives max_k conv_k.
    The full objective to minimize is: n * (1/T) * log(sum_k exp(T * conv_k))
    But since n is constant, we minimize (1/T) * log(sum_k exp(T * conv_k)).
    """
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    size = len(b)
    
    # Soft max weights
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shifted)
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of conv w.r.t. a
    corr = compute_corr(a, w)
    k_diag = 2 * np.arange(n)
    valid = k_diag < size
    diag = np.zeros(n)
    diag[valid] = w[k_diag[valid]] * a[valid]
    grad_a = corr + diag
    
    # Convert to z-gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    # Objective: (1/T) * log(sum exp(T*b))
    logsumexp = Tb.max() / T + np.log(exp_Tb.sum()) / T
    obj = n * logsumexp  # Scale by n for consistency with the actual objective
    
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z = z - z.max()  # Center for numerical stability
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50, 30, 20, 15, 10, 7, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
    
    best_score = evaluate_local(a0)
    best_z = z.copy()
    
    n_stages = len(T_stages)
    base_iters = 3000
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # More iterations at lower temperatures (more important)
        frac = si / n_stages
        iters = int(base_iters * (0.5 + 1.5 * frac))
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_grad_z(z_inner, T, n)
            return obj
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_obj_grad_z(z_inner, T, n)
            return grad
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
        
        # Evaluate actual (non-smooth) objective
        a = np.exp(z - z.max())
        a = a / a.sum()
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_z = z.copy()
        
        if (si + 1) % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE L-BFGS stage {si+1}/{n_stages} T={T}: score={best_score:.8f} ({elapsed:.0f}s)")
    
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out


def fourier_param_obj_grad(coeffs, n, T, t_grid, n_harmonics):
    """
    Objective and gradient for Fourier parameterization.
    coeffs = [c0, a1, b1, a2, b2, ..., aK, bK]
    z_i = c0 + sum_{k=1}^K (a_k * cos(2*pi*k*t_i) + b_k * sin(2*pi*k*t_i))
    a = softmax(z)
    """
    c0 = coeffs[0]
    z = np.full(n, c0)
    for k in range(1, n_harmonics + 1):
        ak = coeffs[2*k - 1]
        bk = coeffs[2*k]
        z += ak * np.cos(2 * np.pi * k * t_grid)
        z += bk * np.sin(2 * np.pi * k * t_grid)
    
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    size = len(b)
    
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shifted)
    w = exp_Tb / exp_Tb.sum()
    
    corr = compute_corr(a, w)
    k_diag = 2 * np.arange(n)
    valid = k_diag < size
    diag = np.zeros(n)
    diag[valid] = w[k_diag[valid]] * a[valid]
    grad_a = corr + diag
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    # Chain rule: grad_coeffs = Dz/dc @ grad_z
    # z = c0 * 1 + sum_k (a_k cos(2pi k t) + b_k sin(2pi k t))
    # dz/dc0 = 1
    # dz/d(a_k) = cos(2pi k t)
    # dz/d(b_k) = sin(2pi k t)
    
    grad_coeffs = np.zeros_like(coeffs)
    grad_coeffs[0] = np.sum(grad_z)
    for k in range(1, n_harmonics + 1):
        grad_coeffs[2*k - 1] = np.dot(grad_z, np.cos(2 * np.pi * k * t_grid))
        grad_coeffs[2*k] = np.dot(grad_z, np.sin(2 * np.pi * k * t_grid))
    
    logsumexp = Tb.max() / T + np.log(exp_Tb.sum()) / T
    obj = n * logsumexp
    
    return obj, grad_coeffs


def fourier_lbgfs(a0, time_limit, max_harmonics=30):
    """Optimize using Fourier series parameterization."""
    n = len(a0)
    t_grid = np.arange(n) / (n - 1) if n > 1 else np.array([0.0])
    t0 = time.monotonic()
    
    best_score = evaluate_local(a0)
    best_a = a0.copy()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated log-sum-exp smooth optimization with L-BFGS (O(n log n) gradients)
2. Fourier series parameterization for reduced-dimension precise optimization
3. Direct coordinate descent for final polish
4. Multi-n exploration and diverse initialization
Key innovations: FFT cross-correlation for fast gradients, Fourier parameterization
reducing effective dimension to ~2K+1, aggressive multi-stage temperature annealing,
and diversified restarts to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_corr(a, w):
    """
    Compute corr[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j=0..n-1.
    Uses FFT: corr[j] = conv(a[::-1], w)[j+n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    return conv[n-1:n-1+n]


def logsumexp_obj_grad_z(z, T, n):
    """
    Compute objective and gradient in z-space where a = softmax(z).
    Objective: n * (1/T) * log(sum_k exp(T * conv_k))
    Gradient via FFT cross-correlation for O(n log n) complexity.
    """
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    size = len(b)

    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shifted)
    w = exp_Tb / exp_Tb.sum()

    corr = compute_corr(a, w)
    k_diag = 2 * np.arange(n)
    valid = k_diag < size
    diag = np.zeros(n)
    diag[valid] = w[k_diag[valid]] * a[valid]
    grad_a = corr + diag

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    logsumexp = Tb.max() / T + np.log(exp_Tb.sum()) / T
    obj = n * logsumexp

    return obj, grad_z


def _lse_func(z, n, T, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad_z(z, T, n)
    return obj


def _lse_jac(z, n, T, deadline):

    _, grad = logsumexp_obj_grad_z(z, T, n)

def fourier_z_from_coeffs(coeffs, n, n_harmonics):
    """Compute z values from Fourier coefficients."""
    t_grid = np.arange(n) / (n - 1) if n > 1 else np.array([0.0])
    c0 = coeffs[0]
    z = np.full(n, c0)
    for k in range(1, n_harmonics + 1):
        ak = coeffs[2 * k - 1]
        bk = coeffs[2 * k]
        z += ak * np.cos(2.0 * np.pi * k * t_grid)
        z += bk * np.sin(2.0 * np.pi * k * t_grid)
    return z


def fourier_obj_grad(coeffs, n, T, n_harmonics):
    """Objective and gradient for Fourier parameterization."""
    t_grid = np.arange(n) / (n - 1) if n > 1 else np.array([0.0])
    z = fourier_z_from_coeffs(coeffs, n, n_harmonics)

    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    size = len(b)

    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shifted)
    w = exp_Tb / exp_Tb.sum()

    corr = compute_corr(a, w)
    k_diag = 2 * np.arange(n)
    valid = k_diag < size
    diag = np.zeros(n)
    diag[valid] = w[k_diag[valid]] * a[valid]
    grad_a = corr + diag

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    grad_coeffs = np.zeros_like(coeffs)
    grad_coeffs[0] = np.sum(grad_z)
    for k in range(1, n_harmonics + 1):
        ck = np.cos(2.0 * np.pi * k * t_grid)
        sk = np.sin(2.0 * np.pi * k * t_grid)
        grad_coeffs[2 * k - 1] = np.dot(grad_z, ck)
        grad_coeffs[2 * k] = np.dot(grad_z, sk)

    logsumexp = Tb.max() / T + np.log(exp_Tb.sum()) / T
    obj = n * logsumexp

    return obj, grad_coeffs


def _fourier_func(coeffs, n, T, n_harmonics, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = fourier_obj_grad(coeffs, n, T, n_harmonics)
    return obj


def _fourier_jac(coeffs, n, T, n_harmonics, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(coeffs)
    _, grad = fourier_obj_grad(coeffs, n, T, n_harmonics)
    return grad


def fourier_lbgfs(a0, time_limit, max_harmonics=40):
    """Optimize using Fourier series parameterization with increasing harmonics."""
    n = len(a0)
    t0 = time.monotonic()

    best_score = evaluate_local(a0)
    best_a = a0.copy()
    t_grid = np.arange(n) / (n - 1) if n > 1 else np.array([0.0])

    harmonic_levels = [3, 5, 8, 12, 18, 25, max_harmonics]

    for n_h in harmonic_levels:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        # Initialize coefficients via least squares
        z_init = np.log(np.maximum(best_a, 1e-300))
        z_init = z_init - z_init.mean()

        n_coeffs = 2 * n_h + 1
        X = np.ones((n, n_coeffs))
        for k in range(1, n_h + 1):
            X[:, 2 * k - 1] = np.cos(2.0 * np.pi * k * t_grid)
            X[:, 2 * k] = np.sin(2.0 * np.pi * k * t_grid)

        coeffs_init, _, _, _ = np.linalg.lstsq(X, z_init, rcond=None)
        coeffs = coeffs_init.copy()

        T_stages = [8, 4, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02]

        for T in T_stages:
            if time.monotonic() - t0 > time_limit:
                break
            rem2 = time_limit - (time.monotonic() - t0)
            if rem2 < 5:
                break

            maxit = min(3000, max(100, int(rem2 * 80)))
            try:
                res = minimize(fun=_fourier_func, x0=coeffs, jac=_fourier_jac,
                               args=(n, T, n_h, t0 + time_limit),
                               method='L-BFGS-B',
                               options={'maxiter': maxit, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'maxfun': maxit * 100,
                                        'disp': False})
                coeffs = res.x
            except Exception:
                pass

            # Evaluate
            z_eval = fourier_z_from_coeffs(coeffs, n, n_h)
            a_eval = np.exp(z_eval - z_eval.max())
            a_eval = a_eval / a_eval.sum()
            score = evaluate_local(a_eval)
            if score < best_score:
                best_score = score
                best_a = a_eval.copy()

        elapsed = time.monotonic() - t0
        print(f"    Fourier (K={n_h}): score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a


def coordinate_descent(a0, time_limit, n_steps=100000, seed=12345):
    """Direct coordinate descent on the non-smooth objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_m = min(8, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.997, 0.995, 0.99, 0.985, 0.98, 0.97, 0.95, 0.93]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    break
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                n_perturb = int(rng.integers(1, 4))
                positions = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for pos in positions:
                    a_trial[pos] *= (0.995 + 0.01 * rng.random())
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 1500:
                    noise = rng.standard_normal(n) * 0.008
                    a_trial = a + noise
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score * 1.005:
                        a = a_trial
                        stall = 0
                    elif stall > 4000:
                        break
        else:
            stall = 0

        if (step + 1) % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a


def structured_init(n, rng):
    """Create various structured initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 10)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-3.0 * x * (1.0 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * 2.0 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    elif choice == 5:
        a = 1.0 + 2.0 * (2.0 * x - 1.0) ** 2
    elif choice == 6:
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
    elif choice == 7:
        a = 1.0 + 0.15 * np.sin(4.0 * np.pi * x)
    elif choice == 8:
        a = (1.0 + np.cos(2.0 * np.pi * x)) ** 0.5
    else:
        a = np.exp(-1.5 * x) * (1.0 + 0.5 * np.cos(3.0 * np.pi * x))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate():
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: FFT-accelerated LSE L-BFGS (main workhorse)
    Phase 2: Fourier parameterization (reduced-dim precision)
    Phase 3: Coordinate descent (targeted peak reduction)
    Phase 4: Second LSE pass (different schedule)
    Phase 5: Multi-n exploration
    Phase 6: Final polish
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + 1785.0

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: LSE L-BFGS (main workhorse, longest allocation) ===
    print(f"\n=== Phase 1: LSE L-BFGS (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.32, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [40, 25, 15, 10, 7, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01]
        a_lse = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages)

            best_score = sc

    # === Phase 2: Fourier parameterization ===
    print(f"\n=== Phase 2: Fourier L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_four = fourier_lbgfs(best_seq, t_alloc, max_harmonics=50)

        if sc < best_score:

            best_seq = a_four.copy()
            print(f"  NEW BEST [fourier]: {sc:.8f}")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 15:
        a_cd = coordinate_descent(best_seq, t_alloc, n_steps=200000, seed=777)
        sc = evaluate_local(a_cd)
        if sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST [coord]: {sc:.8f}")

    # === Phase 4: Second LSE pass (finer T resolution) ===
    print(f"\n=== Phase 4: LSE L-BFGS pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 180.0)
    if t_alloc > 15:
        T_stages2 = [15, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01]
        a_lse2 = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages2)
        sc = evaluate_local(a_lse2)
        if sc < best_score:
            best_score = sc
            best_seq = a_lse2.copy()
            print(f"  NEW BEST [lse_p2]: {sc:.8f}")

    rem = deadline - time.monotonic()

        base_n = len(best_seq)
        n_options = []
        for dn in [-200, -100, 100, 200, 400]:
            nn = base_n + dn
            if 500 <= nn <= 3000:

        per_n_time = max(12, min((rem - 20) / max(1, len(n_options)), 50))

        for nn in n_options:

            if rem < per_n_time + 5:

        if np.isfinite(fs):

        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
