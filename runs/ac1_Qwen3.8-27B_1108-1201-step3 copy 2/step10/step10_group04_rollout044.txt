import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, active_frac=0.1):
    """Compute subgradient of max_k b_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), 0.0
    # Use active set: all k where b_k >= max_b * (1 - active_frac)
    threshold = max_b * (1 - active_frac)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # g[m] = (1/|A|) * sum_{k in A} 2*a[k-m] (for valid k-m)
    g = np.zeros(n)
    for k in active:
        # a[k-m] is valid when 0 <= k-m < n, i.e., max(0, k-n+1) <= m <= min(k, n-1)
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            g[m_start:m_end+1] += 2.0 * a[k - (m_end + 1):k - m_start + 1][::-1]
    g /= len(active)
    return g, max_b
