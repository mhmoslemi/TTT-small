on the LogSumExp-smoothed objective with FFT-based O(n log n) gradients,
adaptive temperature cooling, periodic LP refinement, and multi-start diversity.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_local(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def project_simplex(v):

    Objective: f_T(a) = logsumexp(T * conv(a,a)) / T

    # where cross(j) = sum_i a[i] * w[i+j]

    grad -= correction

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)

def project_simplex(v):
    """Project v onto {x: sum x = 1, x >= 0} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)

            contribs = a[idx] * a[pk - idx]
            top3 = np.argsort(contribs)[::-1][:3]
            a_trial = a.copy()
            for oi in top3:
                pos = idx[oi]
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.998
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f}, stall={stall}")

    return best_a, best_score

def multiplicative_weights_lp(a_fixed, n_iters=500):
    """
    Solve min max_k conv(a_fixed, x) s.t. sum x = 1, x >= 0
    using multiplicative weights on the dual.
    Dual: max sum(y) s.t. (T^T y)_j <= 1 for all j, y >= 0
    where (T^T y)_j = sum_k a_fixed[k-j] * y[k] (cross-correlation)
    """
    n = len(a_fixed)
    size = 2 * n - 1
    y = np.ones(size) / size  # Initialize dual variables
    eta = 0.1

    for it in range(n_iters):
        # Compute (T^T y)_j = cross-correlation of a_fixed and y
        y_rev = y[::-1]
        conv_ay = fast_convolve(a_fixed, y_rev)
        # (T^T y)_j = conv_ay[2n-2-j]
        indices = 2 * (n - 1) - np.arange(n)
        constraints = conv_ay[indices]
        max_constraint = constraints.max()

        if max_constraint < 1e-15:
            break

        # Multiplicative weights update
        ratio = constraints / max_constraint
        y = y * np.exp(eta * (1.0 - ratio))
        # Actually, for the dual max sum(y) s.t. T^T y <= 1:
        # The constraint violation at j is (T^T y)_j / 1 - 1
        # MW update: y_k *= exp(-eta * max_j violation_j * a[j relevant])
        # Simpler: just scale
        pass

    # Extract primal from dual (averaged)
    # This is approximate; use the dual solution to guide
    # For now, return a uniform-ish solution weighted by y
    x = np.ones(n) / n
    return x

def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method on the active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: positions within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Average subgradient over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Gradient step
        a_trial = a - lr * g
        a_trial = project_simplex(a_trial)

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            lr *= 0.95
            lr = max(lr, 1e-8)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 2:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2.7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 0.5 + 0.1
    else:
        a = np.exp(-2 * x * (1 - x)) + 0.1 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Nesterov GD with cooling (primary)
    2. Multi-start at different lengths
    3. Coordinate descent polish
    4. Subgradient refinement
    5. Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    print(f"Budget: {budget_s:.0f}s, deadline in {budget_s-20:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Primary Nesterov GD (longest, 500s) ===
    print(f"\n=== Phase 1: Nesterov GD n={n_best} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 50:
        print(f"  Budget: {t1:.0f}s, T: 30 -> 0.5")
        a_res, _ = nesterov_cd(best_seq, T_start=30.0, T_end=0.5,
                               n_steps=5000000, eta0=0.02, eta_decay=0.001,
                               time_limit=t1, lp_blend_interval=0,
                               print_every=100000)
        try_update(a_res, "ngd_p1")

    # Continue with lower temperature
    rem = deadline - time.monotonic()
    t1b = min(rem * 0.15, 250.0)
    if t1b > 30:
        print(f"  Phase 1b: T: 0.5 -> 0.05, budget={t1b:.0f}s")
        a_res, _ = nesterov_cd(best_seq, T_start=0.5, T_end=0.05,
                               n_steps=3000000, eta0=0.005, eta_decay=0.002,
                               time_limit=t1b, print_every=100000)
        try_update(a_res, "ngd_p1b")

    # === Phase 2: Coordinate Descent Polish (300s) ===
    print(f"\n=== Phase 2: CD Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 20:
        a_cd, _ = coord_descent_polish(best_seq, t2, n_steps=200000)
        try_update(a_cd, "cd_p2")

    # === Phase 3: Subgradient refinement (150s) ===
    print(f"\n=== Phase 3: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 15:
        a_sg, _ = subgradient_refine(best_seq, t3, n_steps=100000)
        try_update(a_sg, "subgrad_p3")

    # === Phase 4: Another Nesterov pass at very low T (200s) ===
    print(f"\n=== Phase 4: Fine Nesterov, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        a_res, _ = nesterov_cd(best_seq, T_start=0.1, T_end=0.01,
                               n_steps=2000000, eta0=0.002, eta_decay=0.005,
                               time_limit=t4, print_every=100000)
        try_update(a_res, "ngd_fine_p4")

    # === Phase 5: Multi-start at different lengths (300s) ===
    print(f"\n=== Phase 5: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_lengths = []
        if n_cur < 1500:
            alt_lengths.append(1500)
        if n_cur < 2000:
            alt_lengths.append(2000)
        alt_lengths.append(int(n_cur * 0.8))

        t_each = min(rem / max(1, len(alt_lengths) + 1), 100.0)
        for alt_n in alt_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            if t_each < 20:
                break
            print(f"  Trying n={alt_n}, t={t_each:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_cd(init_alt, T_start=20.0, T_end=0.3,
                                   n_steps=1000000, eta0=0.015, eta_decay=0.001,
                                   time_limit=t_each * 0.7, print_every=50000)
            try_update(a_alt, f"ngd_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_cd2, _ = coord_descent_polish(a_alt, min(rem * 0.3, t_each * 0.25),
                                                 n_steps=30000)
                try_update(a_cd2, f"cd_n{alt_n}")

    # === Phase 6: Perturbation restarts (150s) ===
    print(f"\n=== Phase 6: Perturbations, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_pert = min(rem / 3.0, 50.0)
        if t_pert > 15:
            for pi in range(3):
                if time.monotonic() > deadline - 15:
                    break
                if pi == 0:
                    x = np.linspace(0, 2 * np.pi, n_cur)
                    pert = 0.02 * np.cos(3 * x) + 0.01 * np.cos(7 * x)
                elif pi == 1:
                    pert = 0.015 * rng.standard_normal(n_cur)
                else:
                    x = np.linspace(0, 2 * np.pi, n_cur)
                    pert = 0.02 * np.sin(5 * x + 1.0)
                a_pert = best_seq + pert
                a_pert = np.maximum(a_pert, 0)
                ps = a_pert.sum()
                if ps < 1e-15:
                    continue
                a_pert = a_pert / ps
                a_res, _ = nesterov_cd(a_pert, T_start=5.0, T_end=0.1,
                                       n_steps=500000, eta0=0.01, eta_decay=0.002,
                                       time_limit=t_pert, print_every=50000)
                try_update(a_res, f"pert_{pi}")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 60.0)
        if t_fin > 10:
            a_res, _ = nesterov_cd(best_seq, T_start=0.05, T_end=0.005,
                                   n_steps=1000000, eta0=0.001, eta_decay=0.01,
                                   time_limit=t_fin, print_every=50000)
            try_update(a_res, "final_nesterov")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd, _ = coord_descent_polish(best_seq, min(rem * 0.4, 25.0), n_steps=50000)
        try_update(a_cd, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg, _ = subgradient_refine(best_seq, min(rem * 0.5, 15.0), n_steps=30000)
        try_update(a_sg, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
