def fast_convolve_auto(a):
    """Linear auto-convolution via FFT: conv(a, a)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(x, y):
    """Linear cross-convolution via FFT: conv(x, y)."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Key new idea: the LP dual gives optimal weights c on convolution positions.

projected-gradient phase that's much cheaper per iteration than full LP solves.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve_auto(a):
    """Linear auto-convolution via FFT: conv(a, a)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(x, y):
    """Linear cross-convolution via FFT: conv(x, y)."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:size]


def compute_ca(c, a):
    """Compute (c*a)_i = sum_j a[j]*c[i+j] for i=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    conv = fast_convolve_cross(a_rev, c)
    return conv[n - 1: 2 * n - 1]


def compute_tw_qual(c, v):
    """Compute (T_c^T v)_i = sum_j v[j]*c[i+j] using FFT."""
    n = len(v)
    v_rev = v[::-1]
    conv = fast_convolve_cross(v_rev, c)
    return conv[n - 1: 2 * n - 1]


def simplex_project(v):
    """Project v onto probability simplex {x: x>=0, sum(x)=1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    succeeded = False
    for s in [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                succeeded = True
                break
        except Exception:
            continue

    if not succeeded:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(max(0, M_var.value))
    return b_val, M_val


def solve_lp_dual(a_fixed, solver):
    """Solve dual LP: max lambda s.t. T_a^T c >= lambda*1, sum(c)=1, c>=0."""
    n = len(a_fixed)
    c_var = cp.Variable(2 * n - 1, nonneg=True)
    lambda_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat.T @ c_var >= lambda_var, cp.sum(c_var) == 1]
    prob = cp.Problem(cp.Maximize(lambda_var), constraints)

    succeeded = False
    for s in [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                succeeded = True
                break
        except Exception:
            continue

    if not succeeded:
        return None, None

    c_val = np.asarray(c_var.value, dtype=np.float64)
    c_val = np.maximum(c_val, 0)
    sc = c_val.sum()
    if sc < 1e-15:
        return None, None
    c_val = c_val / sc
    lambda_val = float(max(0, lambda_var.value))
    return c_val, lambda_val


def evaluate_local(a):
    """Evaluate sequence score locally (fast, no external call)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    n = len(a)
    b = fast_convolve_auto(a)
    return float(2.0 * n * b.max() / (s * s))


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with stall detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve_auto(a).max()
        score_y = 2.0 * n * fast_convolve_auto(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 12:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                if rng.integers(0, 2) == 0:
                    noise += 0.02 * np.sin(2 * freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def dual_deconvolve_refine(a0, time_limit, solver, n_inner=80, n_outer=500):
    """Dual deconvolution: alternate LP dual solve with projected gradient to flatten c*a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    eta = 0.08
    no_improve = 0

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break

        c, lambda_val = solve_lp_dual(a, solver)
        if c is None:
            b = fast_convolve_auto(a)
            max_b = b.max()
            threshold = max_b * (1.0 - 1e-4)
            active = (b >= threshold).astype(float)
            sc = active.sum()
            if sc > 0:
                c = active / sc
            else:
                c = np.ones(2 * n - 1) / (2 * n - 1)

        for inner in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break

            ca = compute_ca(c, a)
            ca_centered = ca - ca.mean()
            grad = 2.0 * compute_tw_qual(c, ca_centered)

            grad_std = grad.std()
            if grad_std < 1e-15:
                break
            eta_eff = eta / grad_std

            a_new = simplex_project(a - eta_eff * grad)
            score = 2.0 * n * fast_convolve_auto(a_new).max()

            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new
            eta *= 0.9995

        if no_improve > n_inner * 3:
            eta = 0.05
            no_improve = 0

    return best_a, best_score


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z (log-domain)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s

    b = fast_convolve_auto(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(a0, 1e-300)
    z = np.log(a / a.sum()).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        deadline = time.monotonic() + remaining

        def func(z_inner):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_inner, n, stage_T)
            return obj

        def jac(z_inner):
            if time.monotonic() > deadline:
                return np.zeros(n)
            _, g = logsumexp_obj_grad(z_inner, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return exp_z / s


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    eta_base = 0.06
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve_auto(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve_auto(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.4 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.sinc(3 * (x - 0.5)) ** 2 + 0.1
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
