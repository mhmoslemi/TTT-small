def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a

def golden_section_search_1d(f, lo, hi, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_line_search_t(a, b, conv_aa, conv_ab, conv_bb, n):
    """Find optimal t in [0,1] minimizing max_k [(1-t)^2 C_aa_k + 2t(1-t) C_ab_k + t^2 C_bb_k]"""
    def f_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ab + t*t * conv_bb
        return float(val.max())
    best_t = golden_section_search_1d(f_t, 0.0, 1.0, max_iter=60)
    return best_t

def simplex_project(v):
    n = len(v)
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def exact_coord_descent(a0, time_limit, verbose=False, label=""):

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            quad_idx = -1

                quad_idx = 2 * i - k_lo

            upper = max(4.0 * v, 10.0 / n)
            
            def _score_x(x):
                ns = current_sum + (x - v)

                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

                    best_a = a.copy()
                improved = True

                break
        else:
            no_improve = 0
        if verbose and sweeps % 20 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def fast_convolve_cross(a, b):
    """Compute cross-convolution a * b using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3, width_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_power_spectral(n, rng, beta=1.0):

    amps *= rng.uniform(0.9, 1.1, n_freq)

    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
