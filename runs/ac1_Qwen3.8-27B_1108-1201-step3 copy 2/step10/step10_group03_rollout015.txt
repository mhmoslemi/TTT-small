"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using successive linear programming (SLP)
with active-set management, parametric initialization, and multi-phase local search.
Key novel component: at each SLP iteration, linearize the auto-convolution at the
current point and solve the resulting LP to find the best "linearized" update,
enabling globally-informed local steps that escape shallow local minima.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def parametric_init(n, rng):
    """Try several parametric forms and return the best."""
    x = np.linspace(0, 1, n)
    candidates = []
    
    # Gaussian variants
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.15, 0.2, 0.25, 0.3]:
            a = np.exp(-((x - center) / width) ** 2)
            candidates.append(a)
    
    # Beta-like
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for q in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = x**p * (1-x)**q
            a = np.maximum(a, 1e-10)
            candidates.append(a)
    
    # Power-law
    for alpha in [0.3, 0.5, 0.7, 1.0]:
        a = (x + 0.1) ** alpha
        candidates.append(a)
        a2 = (1-x + 0.1) ** alpha
        candidates.append(a2)
    
    # Shot noise / mixture
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.5, 0.6, 0.7]:
            a = np.exp(-5*((x-c1)**2)) + 0.5*np.exp(-5*((x-c2)**2))
            candidates.append(a)
    
    # Cosine bump
    a = np.cos(np.pi * x / 2) ** 3
    candidates.append(a)
    a = np.cos(np.pi * x / 2) ** 5
    candidates.append(a)
    
    # Linear
    a = x + 0.1
    candidates.append(a)
    a = (1-x) + 0.1
    candidates.append(a)
    
    # Evaluate all
    best_a = None
    best_score = float('inf')
    for a in candidates:
        a = np.maximum(a, 0)
        a = normalize(a)
        sc = local_eval(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def successive_lp_optimize(a0, solver, time_limit, max_iters=1000):
    """
    Successive Linear Programming: at each iteration, linearize the auto-convolution
    at the current point and solve the resulting LP.
    
    The LP at current a is:
        minimize M_new
        subject to: 2 * conv(a_fixed, a_new)[k] <= M + C[k]  for k in active set
                    a_new >= 0
                    sum a_new = 1
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = fast_convolve(a)
        M = C.max()
        size = 2 * n - 1
        
        # Active set: positions where C[k] >= M - delta
        delta = 0.02 * M
        active = np.where(C >= M - delta)[0]
        
        # Build the LP: minimize M_new
        # s.t. 2 * T_a @ a_new <= M + C[k] for k in active
        #      sum a_new = 1
        #      a_new >= 0
        # where T_a is the (2n-1) x n convolution matrix with a fixed
        
        a_new_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build constraint matrix for active positions
        # conv(a, a_new)[k] = sum_j a[k-j] * a_new[j] (with appropriate bounds)
        # This is the same as (T_a @ a_new)[k] where T_a has rows from the convolution
        T = build_conv_matrix(a)
        T_active = T[active, :]  # (|active|) x n
        
        constraints = [
            2.0 * T_active @ a_new_var <= M_var + C[active],
            cp.sum(a_new_var) == 1,
        ]
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_new_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s = a_val.sum()
                if s > 1e-15:
                    a_val = a_val / s
                    sc = 2.0 * n * fast_convolve(a_val).max()
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_a = a_val.copy()
                        # Damped update
                        t = 0.5
                        a = (1 - t) * a + t * a_val
                        a = normalize(a)
                    else:
                        # No improvement, try smaller step
                        t = 0.1
                        a_trial = (1 - t) * a + t * a_val
                        a_trial = normalize(a_trial)
                        sc_trial = 2.0 * n * fast_convolve(a_trial).max()
                        if sc_trial < best_score:
                            best_score = sc_trial
                            best_a = a_trial.copy()
                            a = a_trial
                        else:
                            # Stall: add noise
                            rng = np.random.default_rng(it)
                            a = a + rng.standard_normal(n) * 0.01 / np.sqrt(n)
                            a = normalize(a)
            else:
                # LP didn't solve, break
                break
        except Exception:
            break
        
        if (it + 1) % 100 == 0:
            print(f"    SLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score
