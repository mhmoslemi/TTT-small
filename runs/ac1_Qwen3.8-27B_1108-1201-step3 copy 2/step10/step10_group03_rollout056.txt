import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (normalized sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_subgradient(a0, time_limit, n_steps=200000):
    """Nesterov-accelerated subgradient method with adaptive active set."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    y = a.copy()
    momentum = 0.9
    base_lr = 0.1
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution at y
        b = fast_convolve(y)
        M = b.max()
        
        # Active set: positions within 0.1% of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of gradients at active positions
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)
        
        # Nesterov update
        t_k = (1 + math.sqrt(1 + 4 * step * step)) / 2
        beta = (t_k - 1) / (t_k + 1)
        
        # Step size
        lr = base_lr / (1 + step * 0.01)
        
        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Update y (Nesterov extrapolation)
        y = a_new + beta * (a_new - a)
        y = normalize(y)
        a = a_new
        
        # Track best
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Also check y
        sc_y = 2.0 * n * fast_convolve(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with very fine T schedule."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25,
                    0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Determine max iterations based on remaining time and T
        # Lower T needs more iterations for convergence
        if T > 1:
            max_iter = max(50, int(remaining * 30))
        elif T > 0.1:
            max_iter = max(100, int(remaining * 50))
        else:
            max_iter = max(200, int(remaining * 80))
        
        def obj(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            
            # Gradient w.r.t. a: sum_k w[k] * d(a*a)[k]/d a[j]
            # d(a*a)[k]/d a[j] = 2*a[k-j] (when valid)
            # Use FFT for efficiency: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            # This is 2 * convolve(w_padded, a) evaluated at specific positions
            # Actually: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            # Let's compute this via convolution of w and a
            # w has length 2n-1, a has length n
            # We want: for each j, sum_k w[k] * a[k-j] where 0<=k-j<n
            # This is (w * a_flipped)[j] type operation... 
            # Actually it's cross-correlation: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            # = 2 * (w ⊛ a)[j] where ⊛ is cross-correlation
            # In FFT terms: IFFT(FFT(w) * conj(FFT(a)))... 
            # Simpler: just do it directly for small active sets or use the formula
            
            # Direct computation (efficient when w is concentrated)
            grad_a = np.zeros(n)
            # Use the fact that w is approximately concentrated near the peak
            # For efficiency, only process k where w[k] > threshold
            w_thresh = w.max() * 1e-10
            active_k = np.where(w > w_thresh)[0]
            for k in active_k:
                wk = w[k]
                if wk < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a_cur[k - idx]
            
            # Chain rule for softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def coordinate_descent_active(a0, time_limit, n_rounds=100):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()

            for j in range(js, je + 1):

                    hot_coords.add(j)
        
        if not hot_coords:
            hot_coords = set(range(n))
        
        hot_list = sorted(hot_coords)
        
        # Coordinate descent on hot coordinates
        improved = False
        for j in hot_list:
            if time.monotonic() - t0 > time_limit:
                break
            
            v = a[j]
            # Try a few candidate values for a[j]
            candidates = [v * 0.5, v * 0.8, v * 1.2, v * 1.5, v * 0.3, 0.0, v * 2.0]

                if sc < best_sc:
                    best_sc = sc
                    best_v = cv
            
            if best_v != v:
                a[j] = best_v
                a = normalize(a)
                improved = True
        
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if not improved:

                a_trial[j] = cv

                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
    
    return best_a, best_score


def peak_shaving_quadratic(a0, time_limit):
    """Targeted peak shaving using local quadratic approximation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(50000):

            break
        
        b = fast_convolve(a)
        M = b.max()
        peak_k = int(np.argmax(b))
        
        # The peak is at position peak_k
        # (a*a)[peak_k] = sum_j a[j] * a[peak_k - j]
        # To reduce this, we want to decrease the "hot" pairs
        
        # Find the top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_k - idx]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Find the coordinate with the largest "excess" contribution
        # We want to redistribute mass from high-contrib pairs to low-contrib ones
        
        # Simple approach: reduce the coordinate with highest a[j]*a[peak_k-j]
        max_idx_local = np.argmax(contributions)
        j_hot = idx[max_idx_local]
        other_hot = peak_k - j_hot
        
        # Reduce a[j_hot] slightly and add to a random "cold" coordinate
        v = a[j_hot]
        if v < 1e-10:
            break
        
        # Try reducing a[j_hot] by various amounts
        for frac in [0.01, 0.005, 0.002, 0.001]:
            delta = v * frac
            # Find a cold coordinate to add the mass to
            # Cold = low contribution to peak
            min_idx_local = np.argmin(contributions)
            j_cold = idx[min_idx_local]
            
            a_trial = a.copy()
            a_trial[j_hot] = v - delta
            a_trial[j_cold] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:

            sc = 2.0 * n * fast_convolve(a_trial).max()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_subgradient_v2(a0, time_limit, n_steps=500000):
    """Improved Nesterov subgradient with adaptive step size and FISTA-like restart."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    t_k = 1.0
    lr = 0.5
    prev_f = local_eval(a)
    stall = 0
    rng = np.random.default_rng(123)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        M = b.max()
        
        # Adaptive active set: positions within adaptive threshold of max
        # Use a wider active set early, narrower later
        frac = min(0.005, 0.01 * (1 - step / min(n_steps, 10000)))
        threshold = M * (1 - frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Cap active set size for efficiency
        if len(active) > 200:
            # Keep only the top 200
            idx_sorted = np.argsort(b[active])[::-1][:200]
            active = active[idx_sorted]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)
        
        # Adaptive learning rate (based on gradient norm)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            step_size = lr / (g_norm * math.sqrt(step + 1))
        else:
            step_size = lr
        
        # Update
        a_new = y - step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # FISTA update
        t_new = (1 + math.sqrt(1 + 4 * t_k * t_k)) / 2
        y = a_new + ((t_k - 1) / t_new) * (a_new - a)
        y = normalize(y)
        a = a_new
        t_k = t_new
        
        # Track best
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalling
        if stall > 200:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.01
            y = normalize(a + noise)
            a = y.copy()
            t_k = 1.0
            lr = max(0.1, lr * 0.9)
            stall = 0
    
    return best_a, best_score


def logsumexp_fast_grad(a0, time_limit, T_stages=None):

    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25,
                    0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    # Precompute FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # More iterations for lower T
        if T > 1:
            max_iter = max(30, int(remaining * 40))
        elif T > 0.1:
            max_iter = max(80, int(remaining * 60))
        else:
            max_iter = max(150, int(remaining * 100))
        
        def obj(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            A = rfft(a_cur, fft_size)
            b = irfft(A * A, fft_size)[:2*n-1]
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            
            # FFT-based convolution
            A = rfft(a_cur, fft_size)
            b = irfft(A * A, fft_size)[:2*n-1]
            
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            
            # grad_a[j] = 2 * sum_k w[k] * a_cur[k-j]
            # This is 2 * (w cross-correlated with a_cur)[j]
            # In FFT: IFFT(FFT(w) * conj(FFT(a_cur, fft_size)))
            # But we need to be careful about the indices
            # Actually: sum_k w[k] * a_cur[k-j] for k in [0, 2n-2], j in [0, n-1]
            # Let's use the FFT approach:
            W = rfft(w, fft_size)
            # We want c[j] = sum_k w[k] * a_cur[k-j]
            # This is the cross-correlation of w and a_cur
            # cross_corr(m, n)[j] = sum_k m[k] * n[k-j]
            # In FFT: IFFT(FFT(m) * conj(FFT(n)))
            # But we need to be careful with the zero-padding
            W_full = np.zeros(fft_size // 2 + 1, dtype=complex)
            W_full[:len(w)] = rfft(w, fft_size)
            A_full = rfft(a_cur, fft_size)
            # cross-correlation: IFFT(conj(W) * A) gives sum_k w[k+j] * a[k]
            # Or: IFFT(W * conj(A)) gives sum_k w[k] * a[k-j]... 
            # Let me just verify: 
            # If W = FFT(w) and A = FFT(a), then IFFT(W * conj(A)) = circular cross-correlation
            # For our purposes with proper zero-padding, this should work
            
            # Actually, let's just do it the simple way since n might not be too large
            # For n=1200, the active set of w is small, so direct computation is fine
            grad_a = np.zeros(n)
            w_thresh = w.max() * 1e-12
            active_k = np.where(w > w_thresh)[0]
            for k in active_k:
                wk = w[k]
                if wk < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a_cur[k - idx]
            
            # Chain rule for softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-18,
                         'maxfun': max_iter * 1000, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def fista_subgradient(a0, time_limit, n_steps=500000, seed=123):
    """FISTA (Nesterov accelerated) subgradient with adaptive restart."""
    n = len(a0)

            step_size = lr

            a_new = a_new / s

        a = a_new

            stall = 0

            a = y.copy()
            t_k = 1.0
            lr = max(0.05, lr * 0.95)

def logsumexp_lbgfs(a0, time_limit, T_stages=None, seed=456):
    """L-BFGS on LogSumExp-smoothed objective with FFT-accelerated gradient."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25,
                    0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    # Precompute cross-correlation FFT size (need L >= 3n-2)
    L = 1 << (3 * n - 2).bit_length()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        if T > 1.0:
            max_iter = max(30, int(remaining * 35))
        elif T > 0.1:
            max_iter = max(80, int(remaining * 50))
        else:
            max_iter = max(150, int(remaining * 80))

        T_cur = T  # Capture for use in nested-free approach

        def lse_obj(z_arr, T_val=T_cur):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T_val
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def lse_grad(z_arr, T_val=T_cur):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T_val
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()

            # FFT-based cross-correlation: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            W = rfft(w, L)
            A = rfft(a_cur, L)
            grad_a = 2.0 * irfft(W * np.conj(A), L)[:n]

            # Softmax chain rule
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-18,
                         'maxfun': max_iter * 1000, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        # Print progress for each T stage
        elapsed = time.monotonic() - t0
        print(f"    T={T:.4f}: score={best_score:.10f} ({elapsed:.0f}s)")

    return best_a, best_score

def peak_shaving(a0, time_limit, n_steps=100000, seed=321):
    """Targeted peak shaving: move mass from hot to cold coordinates."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

        b = fast_convolve(a)
        M = b.max()
        peak_k = int(np.argmax(b))

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_k - idx]
        total = contributions.sum()
        if total < 1e-15:
            break

        # Find hot and cold coordinates
        max_local = np.argmax(contributions)
        min_local = np.argmin(contributions)
        j_hot = idx[max_local]
        j_cold = idx[min_local]

        v_hot = a[j_hot]
        if v_hot < 1e-10:
            # Try random coordinate
            j_hot = rng.integers(0, n)
            v_hot = a[j_hot]
            if v_hot < 1e-10:
                continue

        # Try small reductions of hot coordinate, adding to cold
        best_found = False
        for frac in [0.02, 0.01, 0.005, 0.002, 0.001]:
            delta = v_hot * frac
            for j_cand_cold in [j_cold, int(rng.integers(0, n))]:
                a_trial = a.copy()
                a_trial[j_hot] = v_hot - delta
                a_trial[j_cand_cold] += delta
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = 2.0 * n * fast_convolve(a_trial).max()

                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    best_found = True
                    break
            if best_found:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """

    Uses FISTA subgradient, ultra-fine LogSumExp L-BFGS with FFT gradients,
    coordinate descent, peak shaving, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    state = {'seq': None, 'score': float('inf')}

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            state['seq'] = parent.copy()
            state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        init_seq = structured_init(n_init, rng)
        state['seq'] = init_seq
        state['score'] = local_eval(init_seq)
        print(f"Default init: n={len(state['seq'])}, score={state['score']:.10f}")

    n_best = len(state['seq'])

    # === Phase 1: FISTA Subgradient ===
    print(f"\n=== Phase 1: FISTA (n={n_best}, score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_fista, _ = fista_subgradient(state['seq'], t1, seed=123)
        try_update_global(state, a_fista, "fista1")

    # === Phase 2: LogSumExp L-BFGS (main phase) ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 30:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005]
        print(f"  T_stages (len={len(T_stages)}), budget={t2:.0f}s")
        a_lse, _ = logsumexp_lbgfs(state['seq'], t2, T_stages=T_stages)
        try_update_global(state, a_lse, "logsumexp")

    # === Phase 3: Second FISTA ===
    print(f"\n=== Phase 3: FISTA Round 2 (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 20:
        a_f2, _ = fista_subgradient(state['seq'], t3, seed=456)
        try_update_global(state, a_f2, "fista2")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coord Descent (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 15:
        a_cd, _ = coordinate_descent_active(state['seq'], t4, n_rounds=200)
        try_update_global(state, a_cd, "coord_desc")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 10:
        a_ps, _ = peak_shaving(state['seq'], t5, n_steps=80000)
        try_update_global(state, a_ps, "peak_shave")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        base_n = len(state['seq'])
        n_candidates = []
        for delta in [-50, -20, 20, 50, 100, -100, 200, -200]:
            cand_n = base_n + delta
            if 400 <= cand_n <= 5000 and cand_n != base_n and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        for cand_n in n_candidates[:6]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(state['seq'], cand_n)
            # FISTA + quick LogSumExp
            a_cand, _ = fista_subgradient(init_cand, t_sub * 0.45, seed=cand_n)
            try_update_global(state, a_cand, f"fista_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 25:
                T_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.4, 80.0), T_stages=T_short)
                try_update_global(state, a_cand2, f"lse_n{cand_n}")

    # === Phase 7: Final LogSumExp Polish ===
    print(f"\n=== Phase 7: Final LogSumExp (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 150.0)
        T_fin = [1.0, 0.5, 0.25, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        a_fin, _ = logsumexp_lbgfs(state['seq'], t_fin, T_stages=T_fin)
        try_update_global(state, a_fin, "final_lse")

    # === Phase 8: Final FISTA ===
    print(f"\n=== Phase 8: Final FISTA (score={state['score']:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin2, _ = fista_subgradient(state['seq'], min(rem * 0.6, 100.0), seed=999)
        try_update_global(state, a_fin2, "final_fista")

    # Final verification
    if state['seq'] is not None:
        fs = local_eval(state['seq'])
        if np.isfinite(fs):
            state['score'] = fs
        print(f"\nFINAL SCORE: {state['score']:.10f} (n={len(state['seq'])})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in state['seq']])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in state['seq']]
