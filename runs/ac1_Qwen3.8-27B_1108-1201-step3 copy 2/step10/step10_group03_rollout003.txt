import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT, returns full (2n-1) result."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_response(fixed, solver):
    """Given fixed vector (sum=1), find other vector minimizing max conv(fixed, other)."""
    n = len(fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None

            y_b = normalize(y_b)

    T_start = 0.05
    T_end = 0.005
    T = T_start

    # Precompute index mapping for fast gradient computation
    # For peak k, gradient contribution to a[j] is a[k-j] if 0 <= k-j < n
    step = 0
    while time.monotonic() - t0 < time_limit and step < n_steps:
        b = fast_convolve(a)
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Softmax weights over peaks (concentrates on top peaks as T decreases)
        b_scaled = b / T
        b_shifted = b_scaled - b_scaled.max()
        w = np.exp(b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

        # Weighted subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[k] * a[k - idx]

        # Momentum update
        velocity = beta * velocity + (1.0 - beta) * g
        a_trial = a - lr * velocity
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            # Slightly increase lr on improvement
            lr = min(lr * 1.002, 0.05)
        else:
            lr = max(lr * 0.998, 0.001)

        # Anneal temperature
        progress = step / max(1, n_steps)
        T = T_start * (T_end / T_start) ** progress

        step += 1
        if step % 5000 == 0:
            print(f"    MomSG step={step}: score={best_score:.10f}, T={T:.5f}, lr={lr:.5f}")

    return best_a, best_score


def peak_peeling(a0, time_limit, n_rounds=50):
    """
    Peak-peeling: iteratively identify the dominant peak and take a step
    to reduce it, then re-identify. Uses a "peel one at a time" strategy.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find all peaks within 0.1% of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute the "ideal" direction: we want to reduce all active peaks
        # The gradient of peak k at a is: g_k[j] = a[k-j] if valid
        # We want to move in direction -g (to reduce the peaks)
        # But we also need to maintain sum(a) = 1 and a >= 0

        # Compute combined gradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Center the gradient (project out the mean to maintain sum constraint)
        g_centered = g - g.mean()

        # Adaptive step size: use line search
        best_step_score = best_score
        best_step_a = a.copy()
        base_lr = 0.01 / (1 + rnd * 0.1)

        for lr_try in [base_lr, base_lr * 0.5, base_lr * 0.2, base_lr * 2, base_lr * 5]:
            a_trial = a - lr_try * g_centered
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_step_score:
                best_step_score = sc
                best_step_a = a_trial.copy()

        if best_step_score < best_score - 1e-14:
            best_score = best_step_score
            best_a = best_step_a.copy()
            a = best_step_a
        else:
            # Try a different strategy: push mass from peak-contributing positions
            # to non-contributing positions
            peak_idx = int(np.argmax(b))
            # Positions that contribute most to this peak
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            peak_contrib = a[js:je+1].copy()
            total_contrib = peak_contrib.sum()
            if total_contrib < 1e-10:
                break
            # Redistribute: reduce the largest contributors, increase the smallest
            contrib_order = np.argsort(peak_contrib)
            n_move = max(1, n // 10)
            # Move mass from top contributors to bottom contributors
            for i in range(min(n_move, len(contrib_order))):
                src = js + contrib_order[len(contrib_order) - 1 - i]  # largest
                dst = js + contrib_order[i]  # smallest
                if src == dst:
                    continue
                amt = a[src] * 0.01 * (1 + 0.1 * i)
                amt = min(amt, a[src] * 0.5)
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    break

    return best_a, best_score


def logsumexp_lbgfss(a0, time_limit):
    """
    LogSumExp-smoothed objective optimized with L-BFGS in log-space.
    Uses an adaptive T schedule that's more aggressive at low T.
    """
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 15))

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-300)) - np.log(a_new.sum())
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    # Apply a slight smoothing to avoid oscillations
    if new_n > 10:
        kernel_size = min(5, new_n // 4 * 2 + 1)
        if kernel_size >= 3:
            kernel = np.ones(kernel_size) / kernel_size
            f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.2) ** 2) + 0.5 * np.exp(-2 * (x - 0.8) ** 2) + 0.1
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 2 * x) + 0.08 * np.cos(2 * np.pi * 4 * x)
    elif choice == 8:
        # Edge-peaked
        a = 0.3 + 0.7 * (np.abs(x - 0.5) * 2) ** 3
    elif choice == 9:
        # Triangular
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif choice == 10:
        # Flat with slight edge boost
        a = 1.0 + 0.2 * (np.abs(x - 0.5) * 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def symmetry_lp(a0, solver, time_limit):
    """Optimize with symmetry constraint a[i] = a[n-1-i]."""
    n = len(a0)
    a = normalize(a0)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = normalize(a_sym)
    best_a = a_sym.copy()
    best_score = 2.0 * n * fast_convolve(a_sym).max()
    t0 = time.monotonic()
    half_n = (n + 1) // 2

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(a_sym)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        # Symmetry constraints
        for j in range(half_n):
            other = n - 1 - j
            if other > j:
                constraints.append(b_var[j] == b_var[other])

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val = b_val / b_val.sum()
                sc = 2.0 * n * fast_convolve(b_val).max()
                if sc < best_score:
                    best_score = sc
                    best_a = b_val.copy()
                    a_sym = normalize(0.5 * (b_val + b_val[::-1]))
        except Exception:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent, try multi-n exploration with LP + momentum subgradient
    2. Intensive refinement on best n found
    3. Peak-peeling and symmetry polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')
    best_n = 0

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
            best_n = len(parent)
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        best_n = n_init
        parent = best_seq
        print(f"Default init: n={n_init}, score={best_score:.10f}")

    def try_update(a, label=""):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Multi-n exploration ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    n_candidates = []
    base_n = best_n
    for mult in [0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0]:
        cand_n = int(base_n * mult)
        if 200 <= cand_n <= 6000 and cand_n not in n_candidates:
            n_candidates.append(cand_n)
    # Also try some "round" numbers
    for rn in [500, 800, 1000, 1500, 2000, 2500, 3000, 4000]:
        if rn not in n_candidates and 200 <= rn <= 6000:
            n_candidates.append(rn)
    n_candidates.sort()
    print(f"  n candidates: {n_candidates}")

    # Allocate time per n-candidate
    n_total = len(n_candidates)
    time_per_n = min(rem * 0.55 / max(1, n_total), 80.0)
    print(f"  Time per n: {time_per_n:.0f}s")

    for cand_n in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_sub = min(time_per_n, rem * 0.9)
        if t_sub < 15:
            break

        print(f"  n={cand_n} (budget={t_sub:.0f}s):", end=" ")
        if cand_n == best_n:
            init_cand = best_seq.copy()
        else:
            init_cand = interpolate_sequence(parent, cand_n)

        sc_init = 2.0 * cand_n * fast_convolve(init_cand).max()
        print(f"init={sc_init:.8f}", end=" ")

        # Quick LP alternation
        t_lp = min(t_sub * 0.35, 30.0)
        if t_lp > 10:
            a_lp, sc_lp = lp_alternation(init_cand, t_lp, solver)
            try_update(a_lp, f"lp_n{cand_n}")
            init_cand = a_lp if a_lp is not None else init_cand

        rem = deadline - time.monotonic()
        if rem < 10:
            continue

        # Momentum subgradient
        t_sg = min(t_sub * 0.4, 40.0)
        if t_sg > 10:
            a_sg, sc_sg = momentum_subgradient(init_cand, t_sg, n_steps=20000)
            try_update(a_sg, f"momsg_n{cand_n}")

        rem = deadline - time.monotonic()
        if rem < 10:
            continue

        # Quick L-BFGS
        t_lb = min(t_sub * 0.25, 20.0)
        if t_lb > 8:
            a_lb, sc_lb = logsumexp_lbgfss(init_cand, t_lb)
            try_update(a_lb, f"lbgfs_n{cand_n}")

    print(f"  After multi-n: best={best_score:.10f} (n={best_n})")

    # === Phase 2: Intensive refinement on best n ===
    print(f"\n=== Phase 2: Intensive Refinement (n={best_n}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)

    if t2 > 30:
        # Longer LP alternation
        t_lp2 = min(t2 * 0.3, 120.0)
        if t_lp2 > 20:
            print(f"  LP alternation ({t_lp2:.0f}s)")
            a_lp2, _ = lp_alternation(best_seq, t_lp2, solver)
            try_update(a_lp2, "lp2")

        rem = deadline - time.monotonic()
        if rem > 20:
            # Momentum subgradient (longer)
            t_sg2 = min(t2 * 0.3, 120.0)
            if t_sg2 > 15:
                print(f"  Momentum subgradient ({t_sg2:.0f}s)")
                a_sg2, _ = momentum_subgradient(best_seq, t_sg2, n_steps=50000)
                try_update(a_sg2, "momsg2")

        rem = deadline - time.monotonic()
        if rem > 20:
            # L-BFGS (longer, lower T)
            t_lb2 = min(t2 * 0.4, 180.0)
            if t_lb2 > 20:
                print(f"  LogSumExp L-BFGS ({t_lb2:.0f}s)")
                a_lb2, _ = logsumexp_lbgfss(best_seq, t_lb2)
                try_update(a_lb2, "lbgfs2")

    # === Phase 3: Peak-peeling ===
    print(f"\n=== Phase 3: Peak Peeling (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 15:
        print(f"  Peak peeling ({t3:.0f}s)")
        a_pp, _ = peak_peeling(best_seq, t3, n_rounds=200)
        try_update(a_pp, "peel")

    # === Phase 4: Symmetry LP ===
    print(f"\n=== Phase 4: Symmetry LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 120.0)
    if t4 > 15:
        print(f"  Symmetry LP ({t4:.0f}s)")
        a_sym, _ = symmetry_lp(best_seq, solver, t4)
        try_update(a_sym, "symm")

    # === Phase 5: Final very-fine refinement ===
    print(f"\n=== Phase 5: Final Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very low T L-BFGS
        t_fin = min(rem * 0.25, 100.0)
        if t_fin > 15:
            print(f"  Final L-BFGS ({t_fin:.0f}s)")
            a_fin, _ = logsumexp_lbgfss(best_seq, t_fin)
            try_update(a_fin, "final_lb")

        rem = deadline - time.monotonic()
        if rem > 15:
            # Final momentum subgradient (very fine)
            t_fin2 = min(rem * 0.25, 60.0)
            if t_fin2 > 10:
                print(f"  Final MomSG ({t_fin2:.0f}s)")
                a_fin2, _ = momentum_subgradient(best_seq, t_fin2, n_steps=30000)
                try_update(a_fin2, "final_momsg")

        rem = deadline - time.monotonic()
        if rem > 10:
            # Final peak peeling
            t_fin3 = min(rem * 0.3, 40.0)
            if t_fin3 > 8:
                print(f"  Final peel ({t_fin3:.0f}s)")
                a_fin3, _ = peak_peeling(best_seq, t_fin3, n_rounds=100)
                try_update(a_fin3, "final_peel")

    # === Phase 6: Try a few more n values with the refined solution ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 6: Cross-n Transfer ===")
        # Interpolate the best solution to nearby n values and do quick refinement
        for mult in [0.9, 1.1, 0.8, 1.2]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            cand_n = int(best_n * mult)
            if cand_n == best_n or cand_n < 200 or cand_n > 6000:
                continue
            t_x = min(rem * 0.3, 50.0)
            if t_x < 15:
                continue
            print(f"  Cross-n to {cand_n} ({t_x:.0f}s)")
            init_x = interpolate_sequence(best_seq, cand_n)
            a_x, _ = momentum_subgradient(init_x, t_x * 0.5, n_steps=15000)
            try_update(a_x, f"cross_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_x2, _ = logsumexp_lbgfss(a_x if a_x is not None else init_x, t_x * 0.4)
                try_update(a_x2, f"cross_lb_n{cand_n}")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
