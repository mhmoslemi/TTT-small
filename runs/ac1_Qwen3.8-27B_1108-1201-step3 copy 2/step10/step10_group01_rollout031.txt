    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    b = irfft(rfft(a, fft_size) ** 2, fft_size)[:2 * n - 1]
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    
    if top_k > 0 and top_k < len(b):
        # Peak-aware: boost weight on top-k peaks
        weights = np.ones(len(b))
        top_indices = np.argsort(b)[::-1][:top_k]
        weights[top_indices] = peak_weight
        weighted_sum = (weights * exp_Tb).sum()
        lse = (max_Tb + np.log(weighted_sum)) / T
        w = weights * exp_Tb / weighted_sum
    else:
        sum_exp = exp_Tb.sum()
        lse = (max_Tb + np.log(sum_exp)) / T
        w = exp_Tb / sum_exp
    
    obj = 2.0 * n * lse
    
    # Gradient: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # This is cross-correlation, computed via FFT
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * cc
    
    # Gradient w.r.t. z (softmax Jacobian)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S) * (2.0 * n / T)
    
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=None,
                    peak_weight=2.0, top_k=10, print_progress=True):
    """
    Multi-stage LogSumExp L-BFGS with FFT-accelerated gradients.
    """
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(len(a0)) / len(a0)
        s = 1.0
    a0 = a0 / s
    n = len(a0)
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02]
    
    if maxiter_per_stage is None:
        maxiter_per_stage = max(50, 5000 // len(T_stages))
    
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Adjust top_k: use more peaks at high T, fewer at low T
        stage_top_k = min(top_k, max(1, top_k * int(T) // 10)) if T > 1 else top_k
        if T > 5:
            stage_top_k = 0  # No peak weighting at high T
        
        def func(z_local):
            return lse_objective_and_grad(z_local, n, T, peak_weight, stage_top_k)[0]
        
        def jac(z_local):
            return lse_objective_and_grad(z_local, n, T, peak_weight, stage_top_k)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        # Evaluate and track best
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        a_cur = np.maximum(a_cur, 0)
        sc = evaluate_local(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        
        if print_progress and (stage_i % 4 == 0 or stage_i == len(T_stages) - 1):
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {stage_i+1}/{len(T_stages)} (T={T:.2f}, {elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=100000, top_m=5):
    """Coordinate descent targeting top convolution peaks with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]
        
        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            # Try reducing top contributors
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-10:
                    continue
                for factor in [0.99, 0.98, 0.96, 0.94]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    # Distribute difference proportionally to sqrt(a) for better spreading
                    spread = np.sqrt(np.maximum(others, 0))
                    spread_sum = spread.sum()
                    if spread_sum < 1e-15:
                        a_trial[mask] = others + diff / len(others)
                    else:
                        a_trial[mask] = others + diff * spread / spread_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Simultaneous small reduction
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.997
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    # Random perturbation to escape
                    noise = rng.standard_normal(n) * 0.01
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
                    if step % 100 == 0:
                        break
        
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakCD {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Multi-frequency
        a = 0.5 + 0.2 * np.cos(4 * np.pi * x) + 0.1 * np.cos(7 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 9:
        # Gamma-like
        a = x ** 0.5 * (1 - x) ** 1.0 + 0.05
    elif choice == 10:
        # Step-like with smoothing
        a = 0.3 + 0.7 * (x < 0.6)
        # Smooth the step
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(3, n // 50))
    else:
        # Random smooth
        a = np.exp(-rng.uniform(2, 6) * (x - rng.uniform(0.3, 0.7)) ** 2)
        a = a + 0.1 * np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method targeting active peak set with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)
        
        if step % 10000 == 0 and step > 0:
            print(f"      SubGrad {time.monotonic()-t0:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def lp_best_response(a_fixed, max_time=5.0):
    """Simple LP: given fixed a, find b minimizing max conv(a,b) subject to b>=0, sum(b)=1."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints efficiently
    # conv(a, b)[k] = sum_j a_fixed[j] * b[k-j] for valid j
    constraints = [cp.sum(b_var) == 1]
    
    # For each k in range(2n-1), conv[k] = sum_j a[j]*b[k-j] <= M
    # This can be written as T @ b <= M where T[k,j] = a[j] if 0<=k-j<n
    # But building T explicitly is O(n^2) memory. Instead, use the structure.
    
    # For efficiency with large n, just use a subset of constraints
    # The binding constraints are near the max of conv(a, a)
    b_init = np.ones(n) / n
    b_conv = fast_convolve(a_fixed) * 0  # placeholder
    # Actually conv(a_fixed, b) for b=uniform:
    # Let's just build T for all constraints (it's a banded matrix)
    
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints.append(T @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, verbose=False, max_iter=5000)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None


def nesterov_lp_few(a0, time_limit, max_iters=500):
    """Short Nesterov LP alternation for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = lp_best_response(y, max_time=time_limit / max(1, max_iters - it))
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            print(f"      LP iter {it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. FFT-accelerated LogSumExp L-BFGS (main workhorse)
    2. Multi-length exploration
    3. Random restarts with diverse initialization
    4. Deep refinement on best candidate
    5. LP polish + coordinate descent + subgradient
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Primary LogSumExp L-BFGS (aggressive) ===
    print(f"\n=== Phase 1: LSE L-BFGS n={len(best_seq)}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02]
        a_res, _ = logsumexp_lbgfs(best_seq, t1, T_stages=T_stages,
                                    maxiter_per_stage=800, peak_weight=3.0, top_k=15)

    # === Phase 2: Second L-BFGS pass with different schedule ===

    t2 = min(rem * 0.15, 200.0)
    if t2 > 20:
        # Different temperature schedule: start lower, go very low
        T_stages2 = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.015]
        a_res2, _ = logsumexp_lbgfs(best_seq, t2, T_stages=T_stages2,
                                     maxiter_per_stage=600, peak_weight=5.0, top_k=8)
        try_update(a_res2, "lse_p2")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    candidate_lengths = []
    for factor in [0.75, 0.9, 1.1, 1.25, 1.5]:
        alt_n = int(n_best * factor)
        if 200 <= alt_n <= 5000 and alt_n != n_best:
            candidate_lengths.append(alt_n)
    candidate_lengths = list(set(candidate_lengths))[:3]
    
    for alt_n in candidate_lengths:
        if time.monotonic() > deadline - 60:
            break
        t_alt = min((deadline - time.monotonic()) / max(1, len(candidate_lengths)), 120.0)
        if t_alt < 15:
            break
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

        a_alt, _ = logsumexp_lbgfs(init_alt, t_alt, T_stages=T_stages_alt,
                                    maxiter_per_stage=400, peak_weight=3.0, top_k=10)
        try_update(a_alt, f"len_{alt_n}")

                                      maxiter_per_stage=300, peak_weight=2.0, top_k=10)
            try_update(a_r, f"restart_{ri}")
    
    # === Phase 5: Deep refinement on best ===
    print(f"\n=== Phase 5: Deep refinement, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_deep = min(rem * 0.35, 300.0)
        print(f"  Deep L-BFGS budget: {t_deep:.0f}s")
        # Very fine cooling from moderate T to very low T
        T_stages_deep = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.06, 0.04, 0.03, 0.02, 0.015, 0.01]
        a_deep, _ = logsumexp_lbgfs(best_seq, t_deep, T_stages=T_stages_deep,
                                     maxiter_per_stage=1000, peak_weight=5.0, top_k=5)
        try_update(a_deep, "deep")
    
    # === Phase 6: Peak coordinate descent ===
    print(f"\n=== Phase 6: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t_cd = min(rem * 0.12, 100.0)
    if t_cd > 10:
        a_cd, _ = peak_coordinate_descent(best_seq, t_cd, n_steps=100000, top_m=6)
        try_update(a_cd, "peak_cd")
    
    # === Phase 7: LP polish ===
    print(f"\n=== Phase 7: LP Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.20, 150.0)
    if t_lp > 20:
        a_lp, _ = nesterov_lp_few(best_seq, t_lp, max_iters=300)
        try_update(a_lp, "lp_polish")
    
    # === Phase 8: Subgradient final ===
    print(f"\n=== Phase 8: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t_sg = min(rem * 0.15, 80.0)
    if t_sg > 10:
        a_sg, _ = subgradient_refine(best_seq, t_sg, n_steps=50000)
        try_update(a_sg, "subgrad")
    
    # === Phase 9: Final micro-optimization ===
    print(f"\n=== Phase 9: Final polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.6, 30.0)
        if t_fin > 5:
            T_final = [0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.02, 0.01]
            a_fin, _ = logsumexp_lbgfs(best_seq, t_fin, T_stages=T_final,
                                        maxiter_per_stage=500, peak_weight=10.0, top_k=3)
            try_update(a_fin, "final")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
