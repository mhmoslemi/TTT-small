import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def logsumexp_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    obj = 2.0 * n * lse
    return obj, 2.0 * n * grad_z

def freq_equalize(a, n_iters=100, deadline=None):
    """
    Frequency-domain equalization: flatten the auto-convolution spectrum.
    Alternates between:
    1. Identifying peaks in time-domain auto-conv
    2. Reducing spectral power at frequencies that contribute to those peaks
    3. Projecting back to non-negative time domain
    """
    n = len(a)
    a = normalize(a)
    pad_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current spectrum
        A = np.fft.rfft(a, pad_size)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compute auto-conv
        c = fast_convolve(a)
        
        # Find peaks (top few)
        max_c = c.max()
        # Weight peaks by how much they exceed the average
        avg_c = c.sum() / len(c)
        excess = np.maximum(c - avg_c, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        weights = excess / total_excess
        
        # Water-filling: adjust P to reduce weighted peaks
        # c[k] = (1/pad) * sum_l P[l] * Re(W^{kl})
        # We want to reduce c[k] for large k, which means reducing P[l]
        # where Re(W^{kl}) > 0 and increasing where < 0
        # Use gradient of weighted sum of peaks w.r.t. P
        k_max = pad_size // 2 + 1  # number of rfft bins
        W = np.exp(2j * np.pi * np.arange(k_max)[:, None] * np.arange(k_max)[None, :] / pad_size)
        # Actually, let's use the full IFFT approach
        # c = ifft(|A|^2) for the full length pad_size
        # But c only has non-zero values for k < 2n-1
        
        # Simpler approach: adjust P proportional to -gradient of max(c)
        # Gradient of c[k] w.r.t. P[l] = W^{kl} / pad_size
        # We want to reduce the max, so move P in direction -c_max_contribution
        
        # Find the argmax of c
        k_star = np.argmax(c)
        
        # Compute gradient direction in P space
        # d(c_k*)/dP_l = Re(W^{k* * l}) / pad_size for the real part
        l = np.arange(k_max)
        # For rfft, the relationship is slightly different
        # c[k] = (1/pad) * (P[0] + 2*sum_{l=1}^{pad/2-1} P[l]*cos(2*pi*k*l/pad) + P[pad/2]*cos(2*pi*k*(pad/2)/pad))
        #      = (1/pad) * (P[0] + 2*sum_{l=1}^{pad/2-1} P[l]*cos(2*pi*k*l/pad) + P[pad/2]*(-1)^k)
        
        cos_vals = np.cos(2 * np.pi * k_star * l / pad_size)
        # For l=0: cos=1, for l=pad/2: cos=(-1)^k
        # Weight: we want to decrease P where cos > 0, increase where cos < 0
        adjustment = -cos_vals  # negative where cos > 0
        # Normalize to have zero mean (to preserve sum of P)
        adjustment -= adjustment.mean()
        # Apply small adjustment
        step_size = 0.1 * max_c / (P.sum() + 1e-30)
        P_new = P + step_size * adjustment * P  # multiplicative-like
        P_new = np.maximum(P_new, 0)
        
        # Reconstruct A with new P and same phase
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, pad_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step_size *= 0.5
            P_new = P + step_size * adjustment * P
            P_new = np.maximum(P_new, 0)
            A_new = np.sqrt(P_new) * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, pad_size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = evaluate_local(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, n_steps=50000):
    """Exact subgradient method with line search."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.1
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        
        # Find active set (within 0.1% of max)
        threshold = max_c * (1 - 1e-3)
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient (average over active set)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Multiplicative update (mirror descent)
        log_update = -eta * g / (a + 1e-30)  # in log space
        # Actually, for mirror descent with entropy regularizer:
        # a_new = a * exp(-eta * g) / sum(a * exp(-eta * g))
        log_a_new = np.log(a + 1e-30) - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            no_improve += 1
            continue
        a_new = a_new / s_new
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.01, 1.0)
        else:
            no_improve += 1
            eta *= 0.95
        
        if no_improve >= 100:
            break
    
    return best_a, best_score
