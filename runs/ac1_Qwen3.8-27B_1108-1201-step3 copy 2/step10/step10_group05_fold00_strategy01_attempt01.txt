        if target_length == n_orig and not preserve:
            return seq
        elif target_length > n_orig:
            # Uniform generation and then normalization
            new_seq = rng.uniform(0, 1, target_length) * (initial_sum / target_length)
            return new_seq
        else:
            # Linear interpolation with weighted importance
            if target_length > n_orig and preserve:
                # Interpolate and then limit by original values
                total_weight = np.sum(np.minimum(np.ones(n_orig), seq / seq.max() if seq.max() > 0 else 1))
                new_seq = np.zeros(target_length)
                # Fit the sequence within the [0, target_length] range
                new_seq = np.interp(np.linspace(0, target_length-1, n_orig, endpoint=False), 
                                    np.linspace(0, target_length-1-delta1), 
                                    seq)
                return new_seq
            else:
                # Fallback to a random resampling
                new_seq = np.zeros(target_length)
                new_seq[:] = rng.standard_cauchy(target_length)
                # Symbolic constant for better runtime scaling
                clip_threshold = 10
                new_seq = np.clip(new_seq, -clip_threshold, clip_threshold)
                new_seq = new_seq / new_seq.sum()
                if new_seq.sum() < 1e-10:
                    # Scale to typical positive range
                    new_seq *= 1e5 / clip_threshold
                return np.abs(new_seq).tolist()

def crossover(ind1, ind2, alpha_low=0.3, alpha_high=0.7):
    """
    Perform crossover between two individuals with blending.
    
    Parameters:
        ind1, ind2 (list[float]): Two individuals.
        alpha_low, alpha_high (float): Range for alpha blending.
        
    Returns:
        list[float]: New individual after crossover.
    """
    n = len(ind1)
    child = []
    # Each gene is blended
    for i in range(n):
        # Blend with a chosen alpha
        alpha = rng.uniform(alpha_low, alpha_high)
        child.append(alpha * ind1[i] + (1-alpha) * ind2[i])
    total = sum(child)
    if total > 0:
        return [x/total * 1.0 for x in child]
    else:
        # Cannot normalize, fallback to standard initialization
        return [1.0/n if np.random.random() < 0.5 else 0.0 for i in range(n)]

def mutate(ind, mutation_rate=0.1, strength=0.05):
    """
    Mutate an individual by introducing noise.
    
    Parameters:
        ind (list[float]): Individual to be mutated.
        mutation_rate (float): Probability of mutating a gene.
        strength (float): Maximum change allowed in the gene.
        
    Returns:
        list[float]: Mutated individual.
    """
    child = ind[:]  # Copy
    for i in range(len(child)):
        if rng.random() < mutation_rate:
            # Add random noise
            noise = rng.uniform(-strength, strength)
            child[i] += noise
            child[i] = max(child[i], 0)
    total = sum(child)
    if total > 0:
        return [x/total * 1.0 for x in child]
    else:
        return [1.0/len(child)] * len(child)

def full_convolve(a):
    """
    Compute the full convolution using FFT for efficiency.

    Parameters:
        a: Normalized signal.

    Returns:
        convolve_out: Convolution at max.
    """
    n = len(a)
    size = 2*n - 1
    if size <= 0:
        # Handle small sequences
        pass
    fft_size = 1 << ((size - 1) | 1).bit_length()
    A = rfft(a, fft_size)
    convolve_out = irfft(A * A, fft_size)
    # Truncate to the required size
    return np.max(convolve_out)
