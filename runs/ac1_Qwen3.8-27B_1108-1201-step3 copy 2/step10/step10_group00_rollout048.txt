    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    # This is 2n * cross_correlation(w, a)
    # cross_correlation(w, a)[j] = sum_k w[k] * a[k-j]
    # = convolve(w, flip(a))[j + n - 1] (full)
    # Or via FFT: ifft(fft(w) * conj(fft(a)))
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    cc = irfft(W * np.conj(A), fft_size)
    # cc[j] = sum_k w[k] * a[k-j] (circular), we need the linear part
    # For j in [0, n-1], we need sum over valid k
    # The first n elements of cc give us what we want (for j >= 0, k-j >= 0 means k >= j)
    # Actually cc[j] = sum_{k=0}^{N-1} w[k] * a[(k-j) mod N]
    # For linear cross-correlation: cc_linear[j] = sum_{k=j}^{2n-2} w[k] * a[k-j] for j in [0, n-1]
    # This is cc[j] for j in [0, n-1] if we zero-pad properly (which we did with fft_size >= 2n-1)
    grad_a = 2.0 * n * cc[:n]

    # Convert gradient from a-space to z-space (softmax reparameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def nesterov_peak_descent(a0, time_limit, momentum=0.9, n_top_peaks=5):
    """
    Fast Nesterov-accelerated subgradient descent targeting active convolution peaks.
    Much faster per iteration than LP-based methods.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    prev_a = a.copy()
    eta = 0.02
    eta_min = 1e-6
    patience = 0
    patience_limit = 500

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        y = a + momentum * (a - prev_a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a
            sy = 1.0
        y = y / sy

        # Compute convolution at y
        by = fast_convolve(y)
        max_by = by.max()

        # Find active peaks (top n_top or within 1% of max)
        sorted_idx = np.argsort(by)[::-1]
        active = sorted_idx[:n_top_peaks]
        # Also include all peaks within 0.5% of max
        threshold = max_by * 0.995
        active_all = np.where(by >= threshold)[0]
        active = np.unique(np.concatenate([active, active_all]))

        # Compute subgradient: average of contribution vectors for active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(active)

        # Projected gradient step on simplex
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y
        else:
            a_new = a_new / sn

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        prev_a = a
        a = a_new

        # Adaptive step size
        if patience > patience_limit:
            eta *= 0.8
            patience = 0
            if eta < eta_min:
                break

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                print(f"    NesterovPeak it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def lbgfs_multistage(a0, time_limit, T_stages=None, maxiter_total=5000):
    """
    Multi-stage LogSumExp L-BFGS with adaptive temperature annealing.
    """
    n = len(a0)
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.06, 0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.002]

    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(15, maxiter_total // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        def fun(z_vec):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_and_grad_z(z_vec, stage_T)
            return obj

        def jac(z_vec):
            if time.monotonic() > deadline:
                return np.zeros_like(z_vec)
            _, g = lse_obj_and_grad_z(z_vec, stage_T)
            return g

        try:
            res = minimize(
                fun=fun, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate current
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        if (stage_i + 1) % 4 == 0 or stage_i == n_stages - 1:
            sc = 2.0 * n * fast_convolve(a_cur).max()
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage {stage_i+1}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): score={sc:.8f}")

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def peak_shave(a0, time_limit):
    """
    Direct peak shaving: iteratively reduce the top peaks by adjusting a.
    Uses a "donor" strategy: move mass from elements contributing to peaks
    toward elements that don't contribute as much.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_peak = int(np.argmax(b))

        # Gradient at peak: g[j] = a[k_peak - j] for valid j
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        g = np.zeros(n)
        g[idx] = a[k_peak - idx]

        # Move mass: decrease where g is large, increase where g is small
        # Use a soft version
        eta = 0.01 * (max_b / b.mean())  # scale by how peaked it is
        a_new = a - eta * (g - g.mean())
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.95
            if eta < 1e-8:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Asymmetric ramp with oscillation
        a = (1 - x) * (1 + 0.2 * np.sin(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        # Multi-scale
        a = 0.3 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x + 1) + 0.1 * np.cos(14 * np.pi * x + 2)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured noise."""
    n = len(a)
    method = rng.integers(0, 4)
    x = np.linspace(0, 2 * np.pi, n)
    if method == 0:
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation = strength * np.cos(freq * x + phase)
    elif method == 1:
        # Gaussian noise
        perturbation = strength * 0.5 * rng.standard_normal(n)
    elif method == 2:
        # Sparse spikes
        perturbation = np.zeros(n)
        n_spikes = max(1, n // 50)
        spike_idx = rng.choice(n, size=n_spikes, replace=False)
        spike_vals = rng.uniform(-strength, strength, size=n_spikes)
        perturbation[spike_idx] = spike_vals
    else:
        # Multi-frequency
        perturbation = np.zeros(n)
        for _ in range(3):
            freq = rng.integers(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            perturbation += (strength / 3) * np.cos(freq * x + phase)

    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def optimize_at_length(a_init, time_budget, deadline):
    """
    Full optimization pipeline at a given length:
    1. Fast Nesterov peak descent
    2. L-BFGS LogSumExp annealing
    3. Peak shaving
    Returns best sequence found.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)

    # Stage 1: Nesterov peak descent (fast)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, time_budget * 0.35)
    if t1 > 10:
        a_nes, sc_nes = nesterov_peak_descent(a, t1, momentum=0.9, n_top_peaks=5)
        if sc_nes < best_score:
            best_score = sc_nes
            best_a = a_nes.copy()
            print(f"      Nesterov: {sc_nes:.8f}")
        a = best_a

    # Stage 2: L-BFGS LogSumExp
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.45, time_budget * 0.45)
    if t2 > 15:
        T_stages = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.006, 0.003, 0.001]
        a_lb = lbgfs_multistage(a, t2, T_stages=T_stages, maxiter_total=8000)
        sc_lb = evaluate_local(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_a = a_lb.copy()
            print(f"      L-BFGS: {sc_lb:.8f}")
        a = best_a

    # Stage 3: Peak shaving
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.2, time_budget * 0.15)
    if t3 > 5:
        a_ps, sc_ps = peak_shave(a, t3)
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps.copy()
            print(f"      PeakShave: {sc_ps:.8f}")

    # Stage 4: Quick L-BFGS polish at very low T
    rem = deadline - time.monotonic()
    if rem > 10:
        T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_fin = lbgfs_multistage(best_a, min(rem * 0.7, 30.0), T_stages=T_final, maxiter_total=3000)
        sc_fin = evaluate_local(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = a_fin.copy()
            print(f"      Final L-BFGS: {sc_fin:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast Nesterov peak descent, LogSumExp L-BFGS, peak shaving,
    multi-length exploration, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Optimize at current length ===
    print(f"\n=== Phase 1: Optimize n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc_opt = optimize_at_length(best_seq, t_alloc, deadline - (t_alloc + 5.0))
        try_update(a_opt, f"opt_n{n_best}")

    # === Phase 2: Try n=1800 ===
    print(f"\n=== Phase 2: Try n=1800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1800 = min(rem * 0.25, 350.0)
        print(f"  Budget: {t_1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800, sc_1800 = optimize_at_length(init_1800, t_1800, deadline - (t_1800 * 0.3))
        try_update(a_1800, "n1800")

    # === Phase 3: Try n=1500 ===
    print(f"\n=== Phase 3: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1500 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, sc_1500 = optimize_at_length(init_1500, t_1500, deadline - (t_1500 * 0.3))
        try_update(a_1500, "n1500")

    # === Phase 4: Try n=2000 ===
    print(f"\n=== Phase 4: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_2000 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, sc_2000 = optimize_at_length(init_2000, t_2000, deadline - (t_2000 * 0.3))
        try_update(a_2000, "n2000")

    # === Phase 5: Restart exploration at best length ===
    print(f"\n=== Phase 5: Restart exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    if rem > 50:
        n_trials = min(5, int(rem // 40))
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.02 + 0.015 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.4, 60.0)
            dl_sub = deadline - (rem_sub - t_sub)
            a_trial, sc_trial = optimize_at_length(a_pert, t_sub, dl_sub)
            try_update(a_trial, f"restart_{trial}")

    # === Phase 6: Deep optimization at best found length ===
    print(f"\n=== Phase 6: Deep optimization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_cur = len(best_seq)
        t_deep = min(rem * 0.6, 200.0)
        print(f"  Deep optimizing n={n_cur}, budget={t_deep:.0f}s")
        # Longer Nesterov
        a_nes_deep, sc_nes_deep = nesterov_peak_descent(best_seq, t_deep * 0.4, momentum=0.92, n_top_peaks=8)
        try_update(a_nes_deep, "deep_nesterov")
        # Longer L-BFGS
        rem_sub = deadline - time.monotonic()
        if rem_sub > 15:
            T_deep = [5.0, 2.5, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008,
                      0.004, 0.002, 0.001, 0.0005]
            a_deep = lbgfs_multistage(best_seq, min(rem_sub * 0.7, t_deep * 0.5),
                                      T_stages=T_deep, maxiter_total=12000)
            try_update(a_deep, "deep_lbgfs")
        # Peak shaving
        rem_sub = deadline - time.monotonic()
        if rem_sub > 10:
            a_ps_deep, sc_ps_deep = peak_shave(best_seq, min(rem_sub * 0.6, 30.0))
            try_update(a_ps_deep, "deep_peakshave")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final micro-polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        T_min = [0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.001]
        a_min = lbgfs_multistage(best_seq, min(rem * 0.7, 20.0),
                                  T_stages=T_min, maxiter_total=3000)
        try_update(a_min, "micro_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
