import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def logsumexp_lbgfs(a0, time_limit, T_stages=None, max_iter_per_stage=2000):
    """L-BFGS on LogSumExp-smoothed objective with adaptive T schedule."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = min(max_iter_per_stage, max(10, int(remaining * 15)))
        
        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.float64(np.log(np.exp(b_shift - b_max_val).sum()))
            return (2.0 * n) * lse
        
        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a: for each j, sum over k of w[k] * 2*a[k-j] (where valid)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_cur[k - idx]
            grad_a *= 2.0
            # Chain rule for softmax (without the -log(sum) normalization)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method with adaptive step size and peak averaging."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Weight peaks by their excess
        excess = np.maximum(b - M * 0.98, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            active = np.where(b >= M * 0.999)[0]
        else:
            active = np.where(excess > 0)[0]
            w = excess[active] / total_excess
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
            w = np.array([1.0])
        
        # Weighted subgradient
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * 2.0 * a[k - idx]
        
        # Projected gradient step with normalization correction
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.08)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.93
                no_improve = 0
        lr = max(1e-8, lr)
    
    return best_a, best_score


def fourier_refine(a0, time_limit, n_freq_options=None):
    """Refine in the Fourier domain by optimizing frequency coefficients."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if n_freq_options is None:
        n_freq_options = [12, 24, 48, 96]
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for n_freq in n_freq_options:
        if time.monotonic() - t0 > time_limit:
            break
        if n_freq >= n // 2:
            continue
        
        A = rfft(a, fft_size)
        n_free = 2 * n_freq + 1  # DC (real) + n_freq complex pairs
        
        x0 = np.zeros(n_free)
        x0[0] = A[0].real
        for i in range(1, n_freq + 1):
            x0[2 * i - 1] = A[i].real
            x0[2 * i] = A[i].imag
        
        A_rest = A[n_freq + 1:].copy()
        
        def fourier_obj(x):
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = x[0]
            for i in range(1, n_freq + 1):
                A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
            A_new[n_freq + 1:] = A_rest
            
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s < 1e-10:
                return 1e15
            a_candidate = a_candidate / s
            b = fast_convolve(a_candidate)
            return 2.0 * n * b.max()
        
        try:
            res = minimize(
                fourier_obj, x0, method='Nelder-Mead',
                options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14,
                         'adaptive': True}
            )
            x_opt = res.x
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = x_opt[0]
            for i in range(1, n_freq + 1):
                A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
            A_new[n_freq + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s > 1e-10:
                a_candidate = a_candidate / s
                sc = 2.0 * n * fast_convolve(a_candidate).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
        except Exception:
            pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using QP-based proximal LP alternation,
fast-gradient LogSumExp L-BFGS, subgradient refinement, Fourier-domain tuning,
iterated local search, and multi-n exploration.
Key innovations: (1) proximal QP for better-conditioned best-response,
(2) FFT-free O(n^2) gradient via BLAS dot products for speed,
(3) iterated local search with random perturbations to escape local minima,
(4) aggressive multi-n sweep with short time budgets.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP/QP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_qp_best_response(a_fixed, solver, prox_weight=0.0):
    """ Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max conv(a,b) + proximal. """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if prox_weight > 1e-10:
        obj = cp.Minimize(M_var + (prox_weight / 2.0) * cp.sum_squares(b_var - a_fixed))
    else:
        obj = cp.Minimize(M_var)
    prob = cp.Problem(obj, constraints)
    solvers_to_try = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum > 1e-15:
                    return b_val / s_sum, float(M_var.value)
        except Exception:
            continue
    return None, None


def proximal_lp_alternation(a0, time_limit, solver, max_iters=50000):
    """LP alternation with proximal QP regularization and Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    prev_best = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, it / (max_iters * 0.7))
        lam = 0.5 * (1.0 - frac) ** 3

        b_new, _ = solve_qp_best_response(y_a, solver, prox_weight=lam)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new, _ = solve_qp_best_response(y_b, solver, prox_weight=lam)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 80 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 20:
                noise_scale = 0.015 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0
            if (it + 1) % 500 == 0:
                print(f"    ProxLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, max_iter_per_stage=3000):
    """L-BFGS on LogSumExp-smoothed objective with fast BLAS-based gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [40.0, 20.0, 10.0, 5.0, 2.5, 1.2, 0.6, 0.3, 0.15, 0.07, 0.03, 0.01, 0.005, 0.002, 0.001]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = min(max_iter_per_stage, max(15, int(remaining * 12)))

        def lse_obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.float64(np.log(np.exp(b_shift - b_max_val).sum()))
            return (2.0 * n) * lse

        def lse_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            # grad_a[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j]
            # Use BLAS: for each j, dot(w[j:j+n], a)
            grad_a = np.empty(n)
            w_padded = np.zeros(2 * n - 1)
            w_padded[:len(w)] = w
            for j in range(n):
                grad_a[j] = 2.0 * np.dot(w_padded[j:j + n], a_cur)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxfun': max_iter * 250, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=60000):
    """Weighted subgradient with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.025
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        M = b.max()
        excess = np.maximum(b - M * 0.99, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            active = np.where(b >= M * 0.9995)[0]
            w = np.ones(len(active)) / len(active)
        else:
            active = np.where(excess > 0)[0]
            w = excess[active] / total_excess
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
            w = np.array([1.0])

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * 2.0 * a[k - idx]

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.003, 0.06)
        else:
            no_improve += 1
            if no_improve >= 40:
                lr *= 0.92
                no_improve = 0
        lr = max(1e-9, lr)

    return best_a, best_score


def fourier_refine(a0, time_limit, n_freq_options=None):
    """Optimize low-frequency Fourier coefficients to flatten the convolution peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if n_freq_options is None:
        n_freq_options = [8, 16, 32, 64]

    fft_size = 1 << (2 * n - 1).bit_length()

    for n_freq in n_freq_options:
        if time.monotonic() - t0 > time_limit:
            break
        if n_freq >= n // 2:
            continue
        A = rfft(a, fft_size)
        n_free = 2 * n_freq + 1
        x0 = np.zeros(n_free)
        x0[0] = A[0].real
        for i in range(1, n_freq + 1):
            x0[2 * i - 1] = A[i].real
            x0[2 * i] = A[i].imag
        A_rest = A[n_freq + 1:].copy()

        def fourier_obj(x):
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = x[0]
            for i in range(1, n_freq + 1):
                A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
            A_new[n_freq + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s < 1e-10:
                return 1e15
            a_candidate = a_candidate / s
            b = fast_convolve(a_candidate)
            return 2.0 * n * b.max()

        try:
            res = minimize(
                fourier_obj, x0, method='Nelder-Mead',
                options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True}
            )
            x_opt = res.x
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = x_opt[0]
            for i in range(1, n_freq + 1):
                A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
            A_new[n_freq + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s > 1e-10:
                a_candidate = a_candidate / s
                sc = 2.0 * n * fast_convolve(a_candidate).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
        except Exception:
            pass

    return best_a, best_score


def iterated_local_search(a0, time_limit, solver, n_restarts=30):
    """Iterated local search: perturb + short LP, keep best."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    time_per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        noise_scale = 0.03 * (1.0 + 0.5 * np.random.default_rng(r).random())
        a_pert = a + rng.standard_normal(n) * noise_scale
        a_pert = normalize(a_pert)

        # Short LP from perturbed point
        t_sub = min(time_per_restart, 15.0)
        a_new, _ = proximal_lp_alternation(a_pert, t_sub, solver, max_iters=500)
        sc = 2.0 * len(a_new) * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"    ILS restart {r}: improved to {sc:.10f}")
        if (r + 1) % 10 == 0:
            print(f"    ILS {r+1}/{n_restarts} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: Proximal QP LP alternation, fast LogSumExp L-BFGS,
    iterated local search, subgradient, Fourier refinement, and multi-n sweep.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = normalize(best_seq)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Proximal LP (main) ===
    print(f"\n=== Phase 1: Proximal LP (n={len(best_seq)}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 480.0)
    if t1 > 30:
        a_lp, _ = proximal_lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "prox_lp")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        a_lb, _ = logsumexp_lbgfs(best_seq, t2,
                                   T_stages=[30.0, 15.0, 7.0, 3.0, 1.5, 0.7, 0.35, 0.15, 0.07, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005])
        try_update(a_lb, "lbgfs")

    # === Phase 3: Subgradient ===
    print(f"\n=== Phase 3: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 90.0)
    if t3 > 10:
        a_sg, _ = subgradient_refine(best_seq, t3)
        try_update(a_sg, "subgrad")

    # === Phase 4: Iterated Local Search ===
    print(f"\n=== Phase 4: ILS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 60:
        a_ils, _ = iterated_local_search(best_seq, t4, solver, n_restarts=20)
        try_update(a_ils, "ils")

    # === Phase 5: Fourier refinement ===
    print(f"\n=== Phase 5: Fourier (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 15:
        a_four, _ = fourier_refine(best_seq, t5)
        try_update(a_four, "fourier")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        base_n = len(best_seq)
        n_candidates = sorted(set([
            int(base_n * m) for m in [0.75, 0.9, 1.1, 1.3, 1.5, 2.0]
        ]))
        n_candidates = [n for n in n_candidates if 400 <= n <= 4000]
        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.2, 70.0)
            if t_sub < 30:
                break
            print(f"  n={cand_n}, budget={t_sub:.0f}s")
            init_c = interpolate_sequence(best_seq, cand_n)
            a_c, _ = proximal_lp_alternation(init_c, t_sub * 0.5, solver, max_iters=2000)
            try_update(a_c, f"lp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 25:
                a_c2, _ = logsumexp_lbgfs(a_c, min(rem * 0.3, 40.0),
                                          T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05])
                try_update(a_c2, f"lbgfs_n{cand_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_f1 = min(rem * 0.35, 100.0)
        if t_f1 > 15:
            a_f1, _ = logsumexp_lbgfs(best_seq, t_f1,
                                       T_stages=[3.0, 1.5, 0.7, 0.35, 0.15, 0.07, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005])
            try_update(a_f1, "final_lbgfs")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_f2, _ = proximal_lp_alternation(best_seq, min(rem * 0.3, 60.0), solver)
            try_update(a_f2, "final_lp")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_f3, _ = subgradient_refine(best_seq, min(rem * 0.4, 50.0), n_steps=30000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
