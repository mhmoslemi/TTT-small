```python
        # Nesterov extrapolation for y
        y_new = a_new + (t / (t + 2)) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        t += 1
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score
```

**Key new idea: "Peak Shaving" via direct optimization**

After the LP-based approach converges to a local optimum, do a direct local search:
1. Identify the top-k peaks in the autocorrelation
2. For each peak, determine which pairs (i, j) with i+j = k (mod the convolution structure) contribute to it
3. Move mass from high-contributing positions to low-contributing positions
4. Accept the move if it reduces the max autocorrelation

This is similar to the previous "peak_equalization" but more targeted.

**Another new idea: "Spectral" initialization**

The RM algorithm iteratively:
1. Compute the autocorrelation (or power spectrum)
2. Find the peak
3. Reduce the peak by adjusting the sequence in the time domain

This is essentially what our "peak shaving" does, but the RM algorithm has a specific structure that might converge faster.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fixed Nesterov-accelerated LP best-response (Frank-Wolfe)
2. Direct subgradient mirror descent in log-domain
3. Peak-shaving local search (Robinson-McFarland style)
4. Multi-length exploration with smart interpolation
5. Diverse initialization strategies exploiting anti-palindromic structure
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Time management
_TIME_LIMIT = 1750.0  # Leave 50s margin
_START_TIME = None

def _elapsed():
    if _START_TIME is None:
        return 0.0
    return time.monotonic() - _START_TIME

def _time_remaining():
    return max(0.0, _TIME_LIMIT - _elapsed())

def fast_convolve(a):
    """Fast auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    """Fast cross-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def get_lp_solver():
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for sol in [solver, cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        if sol is None:
            continue
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    b = fast_convolve(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (sum_a * sum_a)

def compute_subgradient(a, top_frac=0.1):
    """
    Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    Uses the top (top_frac * (2n-1)) peaks for smoothing.
    Returns (gradient, max_value).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), 0.0
    
    # Select active peaks
    n_active = max(1, int(top_frac * len(b)))
    threshold = np.sort(b)[-n_active]
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Compute gradient: grad_j = (1/|active|) * sum_{k in active} sum_{i: i+(k-i)=k} a[k-i] * delta
    # Actually: d/d a[j] of conv(a,a)[k] = sum_i a[k-j] where i + (k-i) = k and i=j
    # conv(a,a)[k] = sum_i a[i] * a[k-i]
    # d/d a[j] = sum_i a[k-i] * delta(i=j) + sum_i a[i] * delta(k-i=j)
    #           = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n)
    # Wait, that's for the cross-derivative. For the self-convolution:
    # conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] = a[k-j] + (if j != k-j) a[k-j] = 2*a[k-j] (when j != k-j)
    #           = a[k-j] + a[k-j] = 2*a[k-j]
    #           = a[k-j] + a[k-j] ... 
    # Actually: sum_i a[i]*a[k-i], derivative wrt a[j]:
    # = a[k-j] (from the first a[i] term when i=j) + a[k-j] (from the second a[k-i] term when k-i=j, i.e., i=k-j)
    # If j == k-j (i.e., k=2j), we get 2*a[j]^2 and the derivative is 2*a[j]
    # Otherwise, we get a[k-j] + a[k-j] = 2*a[k-j]
    # So in all cases: d/d a[j] conv(a,a)[k] = 2*a[k-j] if 0 <= k-j < n, else 0
    
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * a[k - j]
    g /= len(active)
    return g, max_b

def mirror_descent(a_init, time_limit, top_frac=0.05):
    """Mirror descent in log-domain with backtracking line search."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-300)
    a = a / a.sum()
    z = np.log(a).copy()
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        if _time_remaining() < 5:
            break
        
        g, max_b = compute_subgradient(a, top_frac)
        if np.abs(g).max() < 1e-15:
            break
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(8):
            z_new = z - alpha * g
            # Softmax for stability
            z_shifted = z_new - z_new.max()
            a_new = np.exp(z_shifted)
            s = a_new.sum()
            if s < 1e-300:
                break
            a_new = a_new / s
            score = compute_objective(a_new)
            
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 5.0)
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            lr = max(lr * 0.85, 0.05)
            if no_improve >= 50:
                # Random perturbation
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 1e-300)
                a = a / a.sum()
                z = np.log(a)
                no_improve = 0
                lr = 0.5
    
    return best_a, best_score

def accelerated_lp(a_init, time_limit, solver=None):
    """Nesterov-accelerated LP best-response (Frank-Wolfe)."""
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_objective(a)
    
    t = 1
    t0 = time.monotonic()
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        if _time_remaining() < 10:
            break
        
        b = solve_lp_best_response(y, solver)
        if b is None:
            t += 1
            continue
        
        # Nesterov step
        alpha = 2.0 / (t + 2)
        
        # Update a
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        
        # Nesterov extrapolation
        momentum = t / (t + 2.0)
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        t += 1
        
        # Track best
        score_a = compute_objective(a)
        score_y = compute_objective(y)
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stagnation = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Stagnation handling
        if stagnation > 200 and t % 500 == 0:
            rng = np.random.default_rng(t * 7 + 3)
            noise = rng.random(n) * 0.02
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            a = y.copy()
            stagnation = 0
        
        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP t={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def peak_shaving(a_init, time_limit, n_steps=50000):
    """
    Robinson-McFarland style peak shaving.
    Directly reduces the highest autocorrelation peaks by moving mass.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit or _time_remaining() < 5:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Find the most contributing pair
        max_contrib_idx = np.argmax(contrib)
        src = idx[max_contrib_idx]
        
        # Try moving mass from src to various destinations
        improved = False
        # Try all destinations (or a subset for speed)
        if n <= 2000:
            candidates = np.arange(n)
        else:
            candidates = rng.choice(n, min(200, n), replace=False)
        
        for dst in candidates:
            if dst == src:
                continue
            # Try different fractions
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                
                new_b = fast_convolve(a_trial)
                new_max = new_b.max()
                score = 2.0 * n * new_max / (a_trial.sum() ** 2)
                
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Try random moves
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = compute_objective(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        # Accept with some probability (simulated annealing)
                        T = max(0.001, 0.1 * (1 - step / n_steps))
                        delta = score - best_score
                        if delta > 0 and rng.random() < np.exp(-delta / T):
                            a = a_trial
        
        if no_improve > 20000:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def diverse_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    if mode == 0:
        # Off-center Gaussian
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.2 * rng.random()
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 1:
        # Power law (asymmetric)
        p = 0.5 + 2.0 * rng.random()
        q = 0.5 + 2.0 * rng.random()
        x = np.linspace(0, 1, n)
        a = x ** p * (1 - x + 1e-10) ** q
    elif mode == 2:
        # Sigmoid ramp
        c = 0.3 + 0.4 * rng.random()
        width = 0.05 + 0.1 * rng.random()
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-(x - c) / width))
        a = a * (1.0 / (1.0 + np.exp((x - c - 0.3) / width)))
    elif mode == 3:
        # Cosine bump (shifted)
        shift = 0.2 + 0.6 * rng.random()
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - shift) / (2 * (1 - abs(2*shift - 1)) + 0.5)) ** 2
        a = np.maximum(a, 0)
    elif mode == 4:
        # Double bump
        c1 = 0.2 + 0.3 * rng.random()
        c2 = 0.5 + 0.3 * rng.random()
        w1 = 0.05 + 0.1 * rng.random()
        w2 = 0.05 + 0.1 * rng.random()
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + 0.5 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 5:
        # Linear ramp with oscillation
        p = 1 + 3 * rng.random()
        freq = 5 + 20 * rng.random()
        x = np.linspace(0, 1, n)
        a = x ** p * (1 + 0.3 * np.cos(2 * np.pi * freq * x))
        a = np.maximum(a, 0)
    elif mode == 6:
        # Flat top with sharp edges
        c = 0.2 + 0.6 * rng.random()
        w = 0.1 + 0.2 * rng.random()
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + ((x - c) / w) ** 4)
    else:
        # Uniform with random noise
        a = np.ones(n) + rng.random(n) * 0.5
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def optimize_at_length(a_init, time_budget, solver):
    """Run multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init, 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = compute_objective(best_a)
    t0 = time.monotonic()
    
    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = compute_objective(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")
    
    # Phase 1: Accelerated LP (main workhorse)
    rem = min(time_budget, _time_remaining())
    t_lp = rem * 0.5
    if t_lp > 20:
        print(f"    LP phase: budget={t_lp:.0f}s")
        a_out, _ = accelerated_lp(best_a, t_lp, solver)
        try_update(a_out, "nesterov_lp")
    
    # Phase 2: Mirror descent (different direction)
    rem = min(time_budget, _time_remaining())
    t_md = rem * 0.2
    if t_md > 10:
        print(f"    MD phase: budget={t_md:.0f}s")
        a_out, _ = mirror_descent(best_a, t_md, top_frac=0.05)
        try_update(a_out, "mirror_desc")
    
    # Phase 3: Peak shaving
    rem = min(time_budget, _time_remaining())
    t_ps = rem * 0.2
    if t_ps > 10:
        print(f"    PeakShave phase: budget={t_ps:.0f}s")
        a_out, _ = peak_shaving(best_a, t_ps, n_steps=30000)
        try_update(a_out, "peak_shave")
    
    # Phase 4: Another LP pass (refinement)
    rem = min(time_budget, _time_remaining())
    t_lp2 = rem * 0.1
    if t_lp2 > 10:
        print(f"    LP refine phase: budget={t_lp2:.0f}s")
        a_out, _ = accelerated_lp(best_a, t_lp2, solver)
        try_update(a_out, "lp_refine")
    
    return best_a, best_score
