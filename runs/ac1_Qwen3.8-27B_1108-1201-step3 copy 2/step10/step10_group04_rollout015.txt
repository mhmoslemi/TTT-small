def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def gradient_polish(a0, time_budget, verbose=True):
    """Polish using subgradient on active peaks with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()

    no_improve = 0

        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        # Find active peaks
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Projected gradient step (stay on simplex)
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
            st = 1.0
        a_trial = a_trial / st
        
        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        
        if score_trial < best_score - 1e-12:
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                break
    
    return best_a, best_score

def fast_convolve(a):

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
            st = 1.0
        a_trial = a_trial / st
        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.5)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200:
                break
    if verbose:
        print(f"    GradPolish done: score={best_score:.8f}")
    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, method=0):
    if method == 0:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif method == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif method == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif method == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(2 * x) + 0.1 * np.cos(4 * x)
    elif method == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 * np.exp(-3 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_state(state, a, label):
    if a is None:
        return
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def run_lp_phase(init_a, time_budget, solver, label, state, verbose=True):
    a_out, sc = lp_nesterov(init_a, time_budget, solver, verbose=verbose)
    try_update_state(state, a_out, f"lp_{label}")
    return a_out


def run_grad_phase(init_a, time_budget, label, state):
    a_out, _ = gradient_polish(init_a, time_budget, verbose=False)
    try_update_state(state, a_out, f"grad_{label}")
    return a_out


def propose_candidate(seed=42, budget_s=1800.0):
    t_start = time.monotonic()
    t_deadline = t_start + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    state = {'best_score': float('inf'), 'best_seq': None}

    def remaining():
        return t_deadline - time.monotonic()

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            state['best_seq'] = parent.copy()
            state['best_score'] = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={state['best_score']:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = structured_init(1200, 0)
        state['best_seq'] = parent.copy()
        state['best_score'] = evaluate_safe(parent)
        print(f"Default init: n=1200, score={state['best_score']:.6f}")

    n_base = len(parent)
    print(f"\n=== Phase 1: Polish at n={n_base} (score={state['best_score']:.6f}) ===")
    rem = remaining()
    t1 = min(rem * 0.18, 250.0)
    if t1 > 30:
        run_lp_phase(state['best_seq'], t1, solver, f"n{n_base}", state, verbose=True)
        rem = remaining()
        if rem > 10:
            tg = min(rem * 0.04, 40.0)
            if tg > 5:
                run_grad_phase(state['best_seq'], tg, f"n{n_base}", state)

    print(f"\n=== Phase 2: Longer lengths (best={state['best_score']:.6f}) ===")
    lengths_to_try = []
    for mult in [1.3, 1.6, 2.0, 2.5, 3.0, 3.5]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 5000:
            lengths_to_try.append(ln)
    lengths_to_try = sorted(set(lengths_to_try))[:5]

    for ln in lengths_to_try:
        rem = remaining()
        if rem < 80:
            break
        # Allocate based on length (smaller = more iterations possible)
        if ln <= 2000:
            t_alloc = min(rem * 0.22, 250.0)
        elif ln <= 3000:
            t_alloc = min(rem * 0.18, 200.0)
        else:
            t_alloc = min(rem * 0.14, 150.0)
        if t_alloc < 40:
            break
        print(f"  n={ln}, budget={t_alloc:.0f}s, remaining={rem:.0f}s")
        init_long = interpolate_seq(state['best_seq'], ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < state['best_score']:
            state['best_score'] = sc_init
            state['best_seq'] = init_long.copy()
            print(f"  Interpolation improved: {sc_init:.6f}")
        # LP phase
        run_lp_phase(init_long, t_alloc * 0.85, solver, f"n{ln}", state, verbose=True)
        # Quick grad polish
        rem = remaining()
        if rem > 15:
            tg = min(t_alloc * 0.15, 30.0)
            if tg > 5:
                run_grad_phase(state['best_seq'], tg, f"n{ln}", state)

    print(f"\n=== Phase 3: Shorter lengths (best={state['best_score']:.6f}) ===")
    for mult in [0.8, 0.65]:
        sn = int(n_base * mult)
        if 300 <= sn < n_base:
            rem = remaining()
            if rem < 50:
                break
            t_alloc = min(rem * 0.10, 80.0)
            if t_alloc < 20:
                break
            print(f"  n={sn}, budget={t_alloc:.0f}s")
            init_short = interpolate_seq(state['best_seq'], sn)
            run_lp_phase(init_short, t_alloc, solver, f"short_n{sn}", state, verbose=False)

    print(f"\n=== Phase 4: Final Polish (best={state['best_score']:.6f}, n={len(state['best_seq'])}) ===")
    rem = remaining()
    if rem > 20:
        t_lp = min(rem * 0.45, 120.0)
        if t_lp > 10:
            print(f"  Final LP: {t_lp:.0f}s")
            run_lp_phase(state['best_seq'], t_lp, solver, "final", state, verbose=True)
        rem = remaining()
        if rem > 10:
            tg = min(rem * 0.30, 50.0)
            if tg > 5:
                run_grad_phase(state['best_seq'], tg, "final", state)
        rem = remaining()
        if rem > 15:
            t_lp2 = min(rem * 0.40, 30.0)
            if t_lp2 > 5:
                run_lp_phase(state['best_seq'], t_lp2, solver, "final2", state, verbose=False)

    # Final
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs
    total_time = time.monotonic() - t_start
    print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={total_time:.0f}s")
    return [float(x) for x in state['best_seq']]
