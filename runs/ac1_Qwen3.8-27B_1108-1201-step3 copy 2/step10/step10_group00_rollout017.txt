    # Constraints:
    #   (T @ b)_k - M <= 0 for k = 0, ..., 2n-2
    #   sum(b) = 1

def fast_convolve(a):

    n = len(a)

def compute_convolution(a, b):
    """Linear convolution of a and b via np.convolve (exact)."""
    return np.convolve(np.asarray(a, dtype=np.float64), np.asarray(b, dtype=np.float64), mode='full')


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1

        je = min(n - 1, k)

def solve_lp_best_response(a, use_constraints=None, max_inner_iter=10):
    """

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, deadline):
    """Compute LogSumExp objective and gradient w.r.t. z (with sum=1 normalization)."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    # Gradient of max_k conv(a,a)_k w.r.t. a
    g_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g_a[idx] += 2.0 * w[k] * a[k - idx]

    # Gradient w.r.t. z (softmax parameterization)
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S) * 2.0 * n
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_inner):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_inner, n, stage_T, deadline)
            return obj

        def jac(z_inner):
            if time.monotonic() > deadline:
                return np.zeros_like(z_inner)
            _, g = logsumexp_obj_grad(z_inner, n, stage_T, deadline)
            return g

        try:
            maxiter = max(10, int(remaining * 20))
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter * 200, 'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def subgradient_momentum(a0, time_limit, n_steps=100000):
    """Subgradient method with momentum for the max-convolution objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    v = np.zeros(n)
    momentum = 0.9
    eta = 0.02

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Weighted average of subgradients at active peaks
        threshold = max_b * (1.0 - 1e-4)
        weights = np.maximum(b - threshold * 0.99, 0)
        if weights.sum() < 1e-15:
            weights = np.zeros_like(b)
            weights[np.argmax(b)] = 1.0

        # Subgradient direction
        g = np.zeros(n)
        active_ks = np.where(weights > 1e-15)[0]
        for k in active_ks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[k] * 2.0 * a[k - idx]
        g /= weights.sum()

        # Momentum update
        v = momentum * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        # Decay step size
        if step % 1000 == 0 and step > 0:
            eta *= 0.98

    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally (fast, assumes sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Sawtooth-like
        a = np.abs(np.sin(2 * np.pi * 5 * x)) ** 0.5
    else:
        # Random smooth
        a = np.exp(-0.5 * ((x - rng.uniform(0.2, 0.8)) / 0.15) ** 2)
        a = a + 0.1 * np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast HiGHS LP alternating with Nesterov acceleration,
    LogSumExp L-BFGS, subgradient momentum, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast Nesterov LP on parent ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.33, 580.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, max_iters=500000, use_reduced_lp=True, verbose=True)
        local_update(a_lp, "nlp_1")

    # === Phase 2: LogSumExp L-BFGS polish ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.17, 280.0)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages)
        local_update(a_lb, "lbgfs_1")

    # === Phase 3: Subgradient momentum ===
    print(f"\n=== Phase 3: SubGrad Momentum (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_momentum(best_seq, t_alloc, n_steps=200000)
        local_update(a_sg, "subgrad")

    # === Phase 4: Nesterov LP Round 2 (from updated best) ===
    print(f"\n=== Phase 4: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 50:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, max_iters=300000, use_reduced_lp=True, verbose=True)
        local_update(a_lp2, "nlp_2")

    # === Phase 5: Try n=1500 ===
    print(f"\n=== Phase 5: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150 and len(best_seq) != 1500:
        t_n1500 = min(rem * 0.18, 280.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_lp_refine(init_1500, t_n1500 * 0.7, max_iters=200000, use_reduced_lp=True, verbose=True)
        local_update(a_1500, "nlp_n1500")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 40:
            T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1500b = lbgfs_refine(best_seq, min(rem, 100.0), T_stages2)
            local_update(a_1500b, "lbgfs_n1500")

    # === Phase 6: Try n=2000 ===
    print(f"\n=== Phase 6: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120 and len(best_seq) != 2000:
        t_n2000 = min(rem * 0.15, 220.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, _ = nesterov_lp_refine(init_2000, t_n2000 * 0.75, max_iters=100000, use_reduced_lp=True, verbose=True)
        local_update(a_2000, "nlp_n2000")

    # === Phase 7: Multi-start exploration ===
    print(f"\n=== Phase 7: Multi-start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        t_multi = min(rem * 0.12, 150.0)
        n_trials = 4
        t_per_trial = t_multi / n_trials
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            if trial % 2 == 0:
                # Perturbation of current best
                a_init = perturbed_sequence(best_seq, rng, strength=0.015 + 0.005 * trial)
            else:
                # Fresh structured init
                a_init = structured_init(n_cur, rng)

            t_sub = min(rem_sub * 0.25, t_per_trial)
            if t_sub < 10:
                break
            a_opt, _ = nesterov_lp_refine(a_init, t_sub, max_iters=50000, use_reduced_lp=False, verbose=False)
            local_update(a_opt, f"multistart_{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS fine polish
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_stages3 = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages3)
            local_update(a_fin, "final_lbgfs")

        # Subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = subgradient_momentum(best_seq, min(rem * 0.3, 20.0), n_steps=50000)
            local_update(a_fin2, "final_subgrad")

        # Short LP
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin3, _ = nesterov_lp_refine(best_seq, min(rem * 0.5, 25.0), max_iters=20000, use_reduced_lp=True, verbose=False)
            local_update(a_fin3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
