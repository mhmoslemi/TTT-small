Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response (main workhorse)
2. Mirror descent with structured subgradient

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_cross_conv_matrix(a):
    """C such that C @ b = cross_conv(a, b) [full linear convolution]."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def solve_lp_resp(a, solver):
    """LP: min M s.t. cross_conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    C = build_cross_conv_matrix(a)
    prob = cp.Problem(cp.Minimize(M), [C @ b <= M, cp.sum(b) == 1])
    try:
        prob.solve(solver=solver)
    except:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    bv = np.maximum(np.asarray(b.value), 0)
    s = bv.sum()
    if s < 1e-15:
        return None, None
    bv /= s
    return bv, M.value

def compute_subgrad(a, top_frac=0.05):
    """Subgradient of max_k conv(a,a)_k using active set approximation."""
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), M
    # Active set: top fraction of peaks
    threshold = M - top_frac * M
    active = np.where(b >= threshold)[0]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # grad of g_k at position j is 2*a[k-j] (if valid)
        # For j in [lo, hi-1], grad_j = 2*a[k-j]
        js = np.arange(lo, hi)
        g[js] += 2.0 * a[k - js]
    g /= len(active)
    return g, M

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_resp, M_val = solve_lp_resp(y, solver)

        y_new = (1.0 - alpha) * y + alpha * b_resp
        y_new = np.maximum(y_new, 0)

        y = y_new

def fast_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_cross_conv_matrix(a):
    """Build matrix C (shape (2n-1, n)) such that C @ b = full_conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0.

    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    C = build_cross_conv_matrix(a)
    constraints = [C @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    bv = np.asarray(b_var.value, dtype=np.float64)
    bv = np.maximum(bv, 0)
    s = bv.sum()
    if s < 1e-15:
        return None, None
    bv = bv / s

                continue
            a_tri = a_tri / st
            sc = 2.0 * n * fast_convolve(a_tri).max()
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_tri.copy()
                a = a_tri
                no_improve = 0
                found = True
                break

        if not found:
            no_improve += 1
            if no_improve > 200:
                break
            # Random exploration
            if it % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = float(rng.uniform(0.005, 0.05))
                    a_tri = a.copy()
                    amt = a[i] * frac
                    a_tri[i] -= amt
                    a_tri[j] += amt
                    a_tri = np.maximum(a_tri, 0)
                    st = a_tri.sum()
                    if st > 1e-15:
                        a_tri = a_tri / st
                        sc = 2.0 * n * fast_convolve(a_tri).max()
                        if sc < best_score - 1e-15:
                            best_score = sc
                            best_a = a_tri.copy()
                            a = a_tri
                            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    si = fi.sum()
    if si > 1e-15:
        fi = fi / si
    return fi


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find the best available LP solver."""
    for s_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 1])
            t0 = time.monotonic()
            prob.solve(solver=s_name)
            el = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"    Solver {s_name} works (test time {el*1000:.0f}ms)")
                return s_name
        except Exception:
            continue
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-method optimization of step function coefficients."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - t_start:.0f}s")

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    best_a = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            p = p / p.sum()
            sc = evaluate_safe(p)
            best_a = p.copy()
            best_score = sc
            print(f"Parent: n={len(p)}, score={sc:.6f}")
    except Exception:
        pass

    if best_a is None:
        n = 1200
        # Try a few structured inits and pick best
        inits = [
            np.ones(n) / n,
            np.sin(np.pi * np.linspace(0, 1, n)) ** 2,
            np.cos(np.pi * (np.linspace(0, 1, n) - 0.5)) ** 2,
        ]
        for ini in inits:
            ini = np.maximum(ini, 0)
            ini = ini / ini.sum()
            sc = evaluate_safe(ini)
            if sc < best_score:
                best_score = sc
                best_a = ini.copy()
        print(f"Default init: n={n}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # === Main optimization loop ===
    # Strategy: alternate between Nesterov LP (main), mirror descent, and peak shave
    # on the current best. Periodically try other lengths.

    n_parent = len(best_a)
    # Candidate lengths to try (in order of priority)
    candidate_lengths = [n_parent]
    for mult in [1.3, 1.6, 0.75, 2.0]:
        nl = int(n_parent * mult)
        if nl != n_parent and 300 <= nl <= 6000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths))
    # Put parent first
    candidate_lengths.remove(n_parent)
    candidate_lengths = [n_parent] + candidate_lengths

    # Track best per length to avoid re-trying
    length_scores = {n_parent: best_score}

    total_time = deadline - t_start
    # Time allocation: 65% main (parent length NLP), 15% mirror descent,
    # 10% peak shave, 10% other lengths
    t_nlp_main = total_time * 0.55
    t_md = total_time * 0.15
    t_ps = total_time * 0.15
    t_other = total_time * 0.15

    # Phase 1: Heavy Nesterov LP on parent length
    print(f"\n=== Phase 1: Nesterov LP (n={n_parent}), budget={t_nlp_main:.0f}s ===")
    t1 = min(t_nlp_main, deadline - time.monotonic() - 60)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_a, t1, solver)
        try_update(a_out, "nlp_main")

    # Phase 2: Mirror descent
    rem = deadline - time.monotonic()
    t2 = min(t_md, rem * 0.3)
    if t2 > 10:
        print(f"\n=== Phase 2: Mirror Descent, budget={t2:.0f}s ===")
        a_out, _ = mirror_descent_optimize(best_a, t2)
        try_update(a_out, "md")

    # Phase 3: Peak shaving
    rem = deadline - time.monotonic()
    t3 = min(t_ps, rem * 0.3)
    if t3 > 5:
        print(f"\n=== Phase 3: Peak Shave, budget={t3:.0f}s ===")
        a_out, _ = peak_shave_local(best_a, t3)
        try_update(a_out, "peak")

    # Phase 4: Another Nesterov LP round (with updated best)
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.35, 300)
    if t4 > 20:
        print(f"\n=== Phase 4: Nesterov LP round 2, budget={t4:.0f}s ===")
        a_out, _ = nesterov_lp_optimize(best_a, t4, solver)
        try_update(a_out, "nlp_r2")

    # Phase 5: Try other lengths
    rem = deadline - time.monotonic()
    for nl in candidate_lengths[1:]:
        if rem < 60:
            break
        tl = min(rem * 0.25, 200)
        if tl < 30:
            break
        print(f"\n=== Phase 5: Trying n={nl}, budget={tl:.0f}s ===")
        init_nl = interpolate_seq(best_a, nl)
        sc_init = evaluate_safe(init_nl)
        if sc_init < best_score:
            best_score = sc_init
            best_a = init_nl.copy()
            print(f"  Interpolation improved: {sc_init:.6f}")

        # Quick NLP
        t_nlp_nl = tl * 0.7
        a_out, _ = nesterov_lp_optimize(init_nl, t_nlp_nl, solver, verbose=False)
        try_update(a_out, f"nlp_n{nl}")

        # Quick peak shave
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            t_ps_nl = min(tl * 0.2, rem2 * 0.3)
            if t_ps_nl > 5:
                a_out2, _ = peak_shave_local(best_a, t_ps_nl, verbose=False)
                try_update(a_out2, f"peak_n{nl}")

        length_scores[nl] = best_score
        rem = deadline - time.monotonic()

    # Phase 6: Final polish
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 6: Final Polish, budget={rem:.0f}s ===")
        # NLP with tight budget
        t_nlp_fin = rem * 0.5
        if t_nlp_fin > 10:
            a_out, _ = nesterov_lp_optimize(best_a, t_nlp_fin, solver, verbose=False)
            try_update(a_out, "final_nlp")

        # Peak shave
        rem = deadline - time.monotonic()
        if rem > 5:
            t_ps_fin = rem * 0.5
            if t_ps_fin > 3:
                a_out, _ = peak_shave_local(best_a, t_ps_fin, verbose=False)
                try_update(a_out, "final_peak")

    # Final verification
    if best_a is not None:
        fs = evaluate_safe(best_a)
        if np.isfinite(fs):
            best_score = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_a)}, time={elapsed_total:.0f}s)")
    print(f"{'='*60}")

    return [float(x) for x in best_a]
