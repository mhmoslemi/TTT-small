def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def weighted_subgradient(a, T=1.0, top_k=None):
    """Compute subgradient with exponential peak weighting.
    
    For small T, focuses on the single max peak.
    For large T, spreads weight across all peaks.
    Returns (gradient, max_b).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    
    if top_k is not None and top_k > 0:
        # Use top-k peaks with exponential weighting
        sorted_idx = np.argsort(b)[::-1][:top_k]
        peak_vals = b[sorted_idx]
        # Weight proportional to peak value (normalized)
        w = peak_vals / peak_vals.sum()
    else:
        # Full exponential weighting
        scale = b / max(T, max_b * 0.01)
        scale -= scale.max()
        w = np.exp(scale)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx = np.argmax(b)
            w = np.zeros(len(b))
            w[idx] = 1.0
        else:
            w /= w_sum
        # Only keep significant weights for efficiency
        threshold = w.max() * 1e-8
        significant = w > threshold
        if significant.sum() == 0:
            idx = np.argmax(b)
            w = np.zeros(len(b))
            w[idx] = 1.0
            significant = np.zeros(len(b), dtype=bool)
            significant[idx] = True
    
    g = np.zeros(n)
    for k in np.where(significant)[0] if top_k is None else sorted_idx:
        w_k = w[k]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w_k * a[k - idx]
    
    return g, max_b


def adam_logspace(a0, time_limit, T_init=0.5, T_min=0.01, top_k=5,

    best_score = 2.0 * n * fast_convolve(a0).max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)

def peak_deflation(a0, time_limit, n_rounds=100):
    """Greedy water-filling: explicitly transfer mass to reduce tallest peak.
    
    For the peak at position k*, the contribution to b[k*] from a[j] is a[j]*a[k*-j].
    We transfer mass from high-contribution positions to low-contribution ones.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        top_peaks = np.argsort(b)[::-1][:3]
        
        for k in top_peaks:
            # Compute contribution of each a[j] to b[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            
            if total_contrib < 1e-15:
                continue
            
            # Find source (highest contributor) and destination (lowest non-contributor)
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find destinations: positions with low contribution to ALL top peaks
            peak_areas = set()
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            # Score destinations by how little they contribute to peaks
            dst_scores = np.zeros(n)
            for j in range(n):
                if j == src:
                    continue
                contrib_score = 0.0
                for k2 in top_peaks:
                    kj = k2 - j
                    if 0 <= kj < n:
                        contrib_score += a[kj]
                dst_scores[j] = -contrib_score  # We want low contribution
            
            # Only consider positions not in peak areas if possible
            valid_dst = [j for j in range(n) if j != src and a[j] < a.max() * 0.5]
            if not valid_dst:
                valid_dst = [j for j in range(n) if j != src]
            
            # Pick best destination
            dst_candidates = np.array(valid_dst)
            dst = dst_candidates[np.argmax(dst_scores[dst_candidates])]
            
            # Binary search on transfer amount
            best_amt = 0.0
            best_trial_score = best_score
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_amt = amt
                    a = a_trial
            
            if best_amt > 0:
                if best_score < best_score - 1e-15:  # already updated
                    pass
                if 2.0 * n * fast_convolve(a).max() < best_score:
                    best_score = 2.0 * n * fast_convolve(a).max()
                    best_a = a.copy()
        
        # Periodic random exploration
        if rnd % 20 == 19:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score

            # Polak-Ribiere beta
            g_old = -d / alpha_prev if alpha_prev > 1e-15 else g_z
            beta_num = np.dot(g_z, g_z - g_old)
            beta_den = np.dot(g_old, g_old)
            beta = max(0.0, beta_num / (beta_den + 1e-15))
            d = -g_z + beta * d
        
        # Line search
        best_alpha = alpha
        best_trial = best_score
        for alpha in [best_alpha * 2, best_alpha, best_alpha * 0.5, best_alpha * 0.25, 0.01, 0.001]:
            z_trial = z + alpha * d
            a_trial = softmax_from_z(z_trial)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial:
                best_trial = score
                best_alpha = alpha
        
        z = z + best_alpha * d
        alpha_prev = best_alpha
        
        a_new = softmax_from_z(z)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """Initialize using spectral considerations.
    
    We want a sequence whose power spectrum is as flat as possible
    (to minimize peak autocorrelation). Start from a "chirp-like"
    sequence and project to non-negative.
    """
    choice = rng.integers(0, 6)
    if choice == 0:
        # Spatial chirp
        x = np.arange(n)
        a = np.exp(1j * np.pi * x ** 2 / n)
        a = np.abs(a)  # All ones... not useful
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n * 3)
    elif choice == 1:
        # Multi-frequency
        x = np.arange(n)
        a = 1.0
        for f in [3, 7, 13]:
            a += 0.3 * np.cos(2 * np.pi * f * x / n)
    elif choice == 2:
        # Logarithmic spacing frequencies
        x = np.arange(n)
        a = 1.0
        for f in [2, 5, 11, 23]:
            a += 0.2 * np.cos(2 * np.pi * f * x / n + rng.uniform(0, 2*np.pi))
    elif choice == 3:
        # Gaussian window × oscillation
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2) * (1 + 0.3 * np.cos(6 * np.pi * x))
    elif choice == 4:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            center = rng.uniform(0.2, 0.8)
            width = rng.uniform(0.05, 0.2)
            amp = rng.uniform(0.1, 0.4)
            x = np.linspace(0, 1, n)
            a += amp * np.exp(-((x - center) / width) ** 2)
    
    a = np.maximum(a, 0)
    return normalize(a)


def multi_temperature_adam(a0, time_limit):
    """Adam with multi-temperature schedule and adaptive top_k.
    
    Phase 1: T small, top_k=1 (focus on single peak)
    Phase 2: T medium, top_k=3 (balance top peaks)  
    Phase 3: T large, top_k=None (equalize all)
    Phase 4: T small again, top_k=1 (fine-tune)
    """
    n = len(a0)
    t0 = time.monotonic()
    
    # Phase 1: Focus on max peak
    rem = time_limit_global - t0
    t1 = min(rem * 0.30, 200.0)
    if t1 > 5:
        a1, s1 = adam_logspace(a0, t1, T_init=0.3, T_min=0.05,
                               top_k=1, lr_init=0.15)
        a0 = a1
    
    # Phase 2: Balance top peaks
    rem = time_limit_global - t0
    t2 = min(rem * 0.25, 150.0)
    if t2 > 5:
        a2, s2 = adam_logspace(a0, t2, T_init=1.0, T_min=0.3,
                               top_k=3, lr_init=0.08)
        sc2 = 2.0 * n * fast_convolve(a2).max()
        sc0 = 2.0 * n * fast_convolve(a0).max()
        if sc2 < sc0:
            a0 = a2
    
    # Phase 3: Full equalization
    rem = time_limit_global - t0
    t3 = min(rem * 0.25, 150.0)
    if t3 > 5:
        a3, s3 = adam_logspace(a0, t3, T_init=3.0, T_min=1.0,
                               top_k=5, lr_init=0.05)
        sc3 = 2.0 * n * fast_convolve(a3).max()
        sc0 = 2.0 * n * fast_convolve(a0).max()
        if sc3 < sc0:
            a0 = a3
    
    # Phase 4: Final fine-tuning
    rem = time_limit_global - t0
    t4 = min(rem * 0.20, 100.0)
    if t4 > 5:
        a4, s4 = adam_logspace(a0, t4, T_init=0.2, T_min=0.02,
                               top_k=1, lr_init=0.03)
        sc4 = 2.0 * n * fast_convolve(a4).max()
        sc0 = 2.0 * n * fast_convolve(a0).max()
        if sc4 < sc0:
            a0 = a4
    
    return a0


def optimize_at_length(a_init, n, time_budget, label=""):
    """Multi-technique optimization at fixed length."""
    best_a = normalize(a_init.copy())
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    def try_update(a, src):
        nonlocal best_a, best_score
        a = normalize(a.copy())
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  IMPROVED [{src}] n={n}: {sc:.6f} (t={elapsed:.0f}s)")
        return best_a
    
    # Technique 1: Multi-temperature Adam
    rem = time_limit_global - t0
    t1 = min(rem * 0.35, 300.0)
    if t1 > 20:
        a1 = multi_temperature_adam(best_a, t1)
        try_update(a1, "adam_multi")
    
    # Technique 2: Peak deflation
    rem = time_limit_global - t0
    t2 = min(rem * 0.25, 150.0)
    if t2 > 10:
        a2, _ = peak_deflation(best_a, t2, n_rounds=200)
        try_update(a2, "deflate")
    
    # Technique 3: Conjugate refinement
    rem = time_limit_global - t0
    t3 = min(rem * 0.20, 100.0)
    if t3 > 10:
        a3, _ = conjugate_refinement(best_a, t3, n_steps=3000)
        try_update(a3, "conj")
    
    # Technique 4: Final Adam polish
    rem = time_limit_global - t0
    t4 = min(rem * 0.20, 100.0)
    if t4 > 10:
        a4, _ = adam_logspace(best_a, t4, T_init=0.1, T_min=0.01,
                              top_k=1, lr_init=0.02)
        try_update(a4, "adam_final")
    
    return best_a, best_score

time_limit_global = 1750.0


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization techniques."""
    global time_limit_global
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 50.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_base = 960
        best_seq = spectral_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(parent)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: Deep optimization at parent length ===
    print(f"\n=== Phase 1: Optimize n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.40, 600.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, n_base, t1, label=f"n{n_base}")
        try_update(a_out, f"phase1_n{n_base}")
    
    # === Phase 2: Try nearby lengths ===
    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    
    # Generate candidate lengths
    candidate_lengths = []
    for delta in [-100, -50, -25, 25, 50, 100, 150, 200, 300]:
        ln = n_base + delta
        if 200 <= ln <= 3000:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))
    
    for ln in candidate_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 180.0)
        if t_alloc < 30:
            break
        
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        
        # Initialize by interpolation
        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolated to n={ln}: {sc_init:.6f}")
        
        a_out, _ = optimize_at_length(init_long, ln, t_alloc, label=f"n{ln}")
        try_update(a_out, f"opt_n{ln}")
    
    # === Phase 3: Diverse restarts at best length ===
    print(f"\n=== Phase 3: Diverse restarts (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    n_best = len(best_seq)
    
    for restart in range(5):
        if time.monotonic() > time_limit_global - 60:
            break
        t_restart = min((time_limit_global - time.monotonic()) * 0.12, 120.0)
        if t_restart < 20:
            break
        
        print(f"  Restart {restart+1}, budget={t_restart:.0f}s")
        
        # Create diverse initialization
        r = rng.integers(0, 4)
        if r == 0:
            # Mix with spectral init
            spec = spectral_init(n_best, rng)
            a_init = normalize((1 - 0.3) * best_seq + 0.3 * spec)
        elif r == 1:
            # Noise perturbation
            noise = rng.normal(0, 0.05, n_best)
            a_init = normalize(best_seq + noise)
        elif r == 2:
            # Slight interpolation + deinterpolation (smoothing)
            a_small = interpolate_sequence(best_seq, max(100, n_best // 2))
            a_init = interpolate_sequence(a_small, n_best)
        else:
            # Flip a random segment
            i = rng.integers(0, n_best - 50)
            j = rng.integers(i + 50, n_best)
            a_init = best_seq.copy()
            a_init[i:j] = a_init[i:j][::-1]
            a_init = normalize(a_init)
        
        a_out, _ = optimize_at_length(a_init, n_best, t_restart, label=f"r{restart}")
        try_update(a_out, f"restart{restart}")
    
    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # Intensive Adam with very low T
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 10:
            a_fin, _ = adam_logspace(best_seq, t_lp, T_init=0.05, T_min=0.005,
                                     top_k=1, lr_init=0.01)
            try_update(a_fin, "final_adam_1")
        
        # Peak deflation
        rem = time_limit_global - time.monotonic()
        t_pd = min(rem * 0.3, 80.0)
        if t_pd > 10:
            a_fin2, _ = peak_deflation(best_seq, t_pd, n_rounds=500)
            try_update(a_fin2, "final_deflate")
        
        # Conjugate
        rem = time_limit_global - time.monotonic()
        t_cr = min(rem * 0.2, 60.0)
        if t_cr > 10:
            a_fin3, _ = conjugate_refinement(best_seq, t_cr, n_steps=5000)
            try_update(a_fin3, "final_conj")
        
        # One more Adam
        rem = time_limit_global - time.monotonic()
        t_last = min(rem * 0.3, 60.0)
        if t_last > 5:
            a_fin4, _ = adam_logspace(best_seq, t_last, T_init=0.03, T_min=0.003,
                                      top_k=1, lr_init=0.005)
            try_update(a_fin4, "final_adam_2")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
