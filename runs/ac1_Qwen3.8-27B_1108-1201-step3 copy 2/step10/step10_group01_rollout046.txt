import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cross_corr(x, y):
    """Compute cross-correlation (x * y)[m] = sum_k x[k+m] * y[k] for m = 0, ..., len(y)-1."""
    len_x = len(x)
    len_y = len(y)
    N = 1 << (len_x + len_y - 1 - 1).bit_length()
    X = rfft(x, N)
    Y = rfft(y, N)
    result = irfft(np.conj(Y) * X, N)[:len_y]
    return result


def dual_value_fast(a, y):
    """Compute (T_a^T @ y)_j for all j, using FFT."""
    # (T_a^T @ y)_j = sum_k y[k] * a[k-j] for 0 <= k-j < n
    # = cross_corr(y, a)[j] where cross_corr(y, a)[j] = sum_k y[k] * a[k-j]
    # Using the formula: sum_k y[k] * a[k-j] = cross_corr(y_ext, a)[j]
    # where y_ext is y padded to length >= 2n-1
    n = len(a)
    # y has length 2n-1 (it's the autocorrelation variable)
    # a has length n
    # (T_a^T @ y)_j = sum_{k=0}^{2n-2} y[k] * a[k-j] (a[k-j]=0 if k-j outside [0,n-1])
    # = sum_{k=j}^{j+n-1} y[k] * a[k-j]
    # Let l = k-j: sum_{l=0}^{n-1} y[j+l] * a[l]
    # This is the cross-correlation of y and a at lag j
    # cross[y, a](j) = sum_l y[j+l] * a[l]
    y_pad = np.zeros(max(len(y), 2*n-1))
    y_pad[:len(y)] = y
    result = cross_corr(y_pad, a)
    return result
