"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp L-BFGS
with adaptive temperature scheduling, multiple diverse restarts, and refined
coordinate descent polishing. Key innovation: O(n log n) gradient computation
via FFT-based cross-correlation, enabling far more L-BFGS iterations within
the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(v, u):
    """
    Compute cross-correlation (v ⊛ u)[i] = sum_j v[i+j] * u[j]
    for i = 0, ..., len(v)+len(u)-2.
    v has length L1, u has length L2.
    """
    L1 = len(v)
    L2 = len(u)
    out_len = L1 + L2 - 1
    fft_size = 1 << (out_len - 1).bit_length()
    V = rfft(v, fft_size)
    U = rfft(u, fft_size)
    cross = irfft(V * np.conj(U), fft_size)[:out_len]
    return cross


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_and_grad_fft(z, T):
    """
    Compute LogSumExp objective and gradient using FFT-based cross-correlation.
    Objective: (2n/T) * logsumexp(conv(a,a)/T) where a = softmax(z)
    This is equivalent to (2n/T) * logsumexp((a*a)/T) since sum(a)=1.
    
    Actually, let me reconsider. The original objective is:
    2*n*max_k (a*a)[k] / (sum a)^2
    
    With sum(a) = 1, it's 2*n*max_k (a*a)[k].
    
    The LogSumExp approximation of max is: T * log(sum exp(x_k/T))
    
    So we minimize: 2*n*T * log(sum_k exp((a*a)[k]/T))
    
    where a = softmax(z).
    """
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # LogSumExp of b / T
    max_b = b.max()
    logsumexp_b = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    
    obj = 2.0 * n * T * logsumexp_b
    
    # Gradient computation
    w = np.exp((b - max_b) / T)
    w = w / w.sum()  # normalized weights, length 2n-1
    
    # grad of (a*a)[k] w.r.t. a[i] ≈ 2*a[k-i] (for k-i in range)
    # Sum over k: grad_a[i] = 2 * sum_k w[k] * 2 * a[k-i] * [0≤k-i<n]
    # = 4 * sum_j a[j] * w[i+j] * [i+j < 2n-1]
    # = 4 * (w ⊛ a)[i] (truncated to valid range)
    
    cross = fast_cross_corr(w, a)  # length (2n-1) + n - 1 = 3n-2
    grad_a = 4.0 * cross[:n]
    
    # Correction for diagonal terms (where k = 2i):
    # At k=2i, the derivative is a[i] not 2*a[i], so we overcounted by a[i].
    # Correction: grad_a[i] -= 2 * w[2i] * a[i] (for 2i < 2n-1)
    diag_idx = np.arange(n)
    k_diag = 2 * diag_idx
    valid = k_diag < (2 * n - 1)
    grad_a[valid] -= 2.0 * w[k_diag[valid]] * a[diag_idx[valid]]
    
    # Convert a-gradient to z-gradient (softmax reparametrization)
    # d/dz_i [f(softmax(z))] = softmax(z) * (grad_a - softmax(z) * sum(softmax(z) * grad_a))
    Sa = np.dot(grad_a, a)
    grad_z = a * (grad_a - Sa)
    
    return obj, grad_z


def logsumexp_lbgfs_fft(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """
    L-BFGS on log-sum-exp smoothed objective with decreasing temperature.
    Uses FFT-accelerated gradient computation.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)) - np.log(a0.sum() + 1e-30)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    best_z = z.copy()
    best_obj = 2.0 * n * fast_convolve(softmax_safe(z)).max()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        stage_start = time.monotonic()
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad_fft(z_inner, T)
            return obj
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_obj_and_grad_fft(z_inner, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
            z = best_z.copy()
        
        # Evaluate the true (non-smoothed) objective
        a_curr = softmax_safe(z)
        true_obj = 2.0 * n * fast_convolve(a_curr).max()
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_z = z.copy()
        
        elapsed = time.monotonic() - stage_start
        print(f"    T={T:.3f}: true_obj={true_obj:.8f}, best={best_obj:.8f}, time={elapsed:.1f}s")
    
    return softmax_safe(best_z), best_obj


def mirror_descent_fast(a0, time_limit, eta0=0.05, active_frac=0.999):
    """
    Fast mirror descent (exponential update) with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        # Update
        g_shifted = g - g.min()
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
            ys = 1.0
        y = y / ys
        
        a = a_new
        
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if min(score, score_y) < best_score - 1e-12:
            best_score = min(score, score_y)
            best_a = (a if score < score_y else y).copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 300:
            eta *= 0.95
            no_improve = 0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
    
    return best_a, best_score


def trial_and_error_local(a0, time_limit, n_trials=10000):
    """
    Random perturbation search: try small random perturbations and keep improvements.
    Different from coordinate descent - tries multi-dimensional perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Choose perturbation strategy
        strategy = rng.integers(0, 4)
        
        if strategy == 0:
            # Single coordinate decrease
            pos = rng.integers(0, n)
            if a[pos] < 1e-12:
                continue
            factor = 1.0 - rng.uniform(0.01, 0.2)
            a_trial = a.copy()
            a_trial[pos] *= factor
            diff = a[pos] - a_trial[pos]
            if diff < 1e-16:
                continue
            mask = np.ones(n, dtype=bool)
            mask[pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] = others + diff * others / others_sum
        elif strategy == 1:
            # Multi-coordinate perturbation (localized)
            center = rng.integers(0, n)
            radius = rng.integers(2, max(3, n // 50))
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            a_trial = a.copy()
            a_trial[lo:hi] *= rng.uniform(0.8, 1.2, size=hi-lo)
            a_trial[lo:hi] = np.maximum(a_trial[lo:hi], 0)
        elif strategy == 2:
            # Swap mass between two regions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue
            frac = rng.uniform(0.01, 0.1)
            transfer = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= transfer
            a_trial[j] += transfer
        else:
            # Peak-directed: decrease at positions contributing to peak
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top = idx[np.argsort(contribs)[::-1][:rng.integers(1, 4)]]
            a_trial = a.copy()
            for pos in top:
                a_trial[pos] *= (1.0 - rng.uniform(0.005, 0.05))
        
        a_trial = np.maximum(a_trial, 0)
        a_trial_sum = a_trial.sum()
        if a_trial_sum < 1e-15:
            continue
        a_trial = a_trial / a_trial_sum
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            break
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    elif choice == 5:
        # Super-Gaussian
        a = np.exp(-8 * (x - 0.5) ** 4)
    elif choice == 6:
        # Double bump
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    else:
        # Cosine window variant
        a = np.cos(np.pi * (x - 0.5)) ** 4 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_scale_optimize(a0, time_limit, n_scales=3):
    """Multi-scale: optimize at coarser resolution, then refine."""
    n_full = len(a0)
    t0 = time.monotonic()
    current = a0.copy()
    
    scales = [n_full]
    for i in range(1, n_scales):
        n_coarse = max(100, n_full // (2 ** i))
        scales.append(n_coarse)
    scales.append(n_full)
    
    for i, n_scale in enumerate(scales):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        remaining_frac = remaining / time_limit
        if remaining < 10:
            break
        
        if n_scale != n_full:
            a_scale = interpolate_sequence(current, n_scale)
        else:
            a_scale = current.copy()
        
        t_alloc = remaining * (0.4 if i < n_scales else 0.6)
        
        # Use mirror descent for coarse, L-BFGS for fine
        if n_scale < n_full:
            a_scale, _ = mirror_descent_fast(a_scale, t_alloc, eta0=0.05)
        else:
            T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
            a_scale, _ = logsumexp_lbgfs_fft(a_scale, t_alloc, T_stages=T_stages,
                                              maxiter_per_stage=3000)
        
        if n_scale != n_full:
            current = interpolate_sequence(a_scale, n_full)
        else:
            current = a_scale
    
    return current


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated LogSumExp L-BFGS (O(n log n) gradient) as the main
    optimizer, combined with mirror descent, multi-scale strategy, and
    coordinate/random perturbation polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label, verbose=True):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            if verbose:
                print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: FFT-accelerated LogSumExp L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: LogSumExp L-BFGS FFT (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb, _ = logsumexp_lbgfs_fft(best_seq, t_alloc, T_stages=T_stages,
                                       maxiter_per_stage=8000)
        try_update(a_lb, "lbgfs_fft_p1")
    
    # === Phase 2: Mirror Descent (fast, good for global structure) ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_fast(best_seq, t_alloc, eta0=0.04, active_frac=0.998)
        try_update(a_md, "mirror_descent")
    
    # === Phase 3: Multi-scale approach ===
    print(f"\n=== Phase 3: Multi-scale (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ms = multi_scale_optimize(best_seq, t_alloc, n_scales=3)
        try_update(a_ms, "multi_scale")
    
    # === Phase 4: Second L-BFGS pass at lower T ===
    print(f"\n=== Phase 4: L-BFGS refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages2 = [5.0, 3.0, 2.0, 1.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb2, _ = logsumexp_lbgfs_fft(best_seq, t_alloc, T_stages=T_stages2,
                                        maxiter_per_stage=10000)
        try_update(a_lb2, "lbgfs_fft_p2")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Random perturbation search ===
    print(f"\n=== Phase 6: Random Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_rp, _ = trial_and_error_local(best_seq, t_alloc, n_trials=50000)
        try_update(a_rp, "random_perturb")
    
    # === Phase 7: Final L-BFGS at very low T ===
    print(f"\n=== Phase 7: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_fin, _ = logsumexp_lbgfs_fft(best_seq, t_alloc, T_stages=T_stages_fin,
                                        maxiter_per_stage=10000)
        try_update(a_fin, "final_lbgfs")
    
    # === Phase 8: Final coordinate descent polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 50.0)
    if t_alloc > 5:
        a_polish, _ = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_polish, "final_polish")
    
    # === Try different n values ===
    print(f"\n=== Phase 9: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [800, 1500, 1800]:
            if n_try == n_best:
                continue
            if time.monotonic() > deadline - 30:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            t_try = min(60.0, (deadline - time.monotonic()) * 0.4)
            if t_try < 10:
                break
            T_stages_try = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp L-BFGS

Key innovation: exact O(n log n) gradient (no diagonal correction needed since

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(v, u):
    """
    Linear cross-correlation: (v ⊛ u)[m] = sum_j v[j+m] * u[j]
    v length L1, u length L2, output length L1+L2-1.
    """
    L1, L2 = len(v), len(u)
    out_len = L1 + L2 - 1
    fft_size = 1 << (out_len - 1).bit_length()
    V = rfft(v, fft_size)
    U = rfft(u, fft_size)
    return irfft(V * np.conj(U), fft_size)[:out_len]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    z = np.asarray(z, dtype=np.float64)

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage,
                                    'ftol': 1e-18, 'gtol': 1e-15,
                                    'maxfun': maxiter_per_stage * 300,
                                    'disp': False})
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
            z = best_z.copy()

        a_curr = softmax_safe(z)
        a_curr = np.maximum(a_curr, 0)
        sc = a_curr.sum()
        if sc < 1e-15:
            continue
        a_curr = a_curr / sc
        true_score = 2.0 * n * fast_convolve(a_curr).max()

        if true_score < best_score:
            best_score = true_score
            best_a = a_curr.copy()
            best_z = z.copy()

        elapsed = time.monotonic() - t_stage_start
        if elapsed > 2 or stage_idx % 3 == 0:
            print(f"    LSE T={T:.4f}: true={true_score:.8f} best={best_score:.8f} ({elapsed:.1f}s)")

    return best_a, best_score


def mirror_descent(a0, time_limit, eta0=0.05, active_frac=0.998):
    """
    Mirror descent (exponential update) with Nesterov and adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # Exponential update
        g_shifted = g - g.min()
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
            ys = 1.0
        y = y / ys

        a = a_new

        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if min(score, score_y) < best_score - 1e-12:
            best_score = min(score, score_y)
            best_a = (a if score < score_y else y).copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            eta = max(0.005, eta * 0.93)
            no_improve = 0

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def coord_descent(a0, time_limit):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 300:
                # Try peak-directed multi-coordinate
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 800:
                    break
        else:
            stall = 0

        if stall % 500 == 0 and stall > 0:
            print(f"    CD stall={stall}, score={best_score:.8f}")

    return best_a, best_score


def random_perturb_search(a0, time_limit, n_trials=100000):
    """Random perturbation search with multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        strategy = rng.integers(0, 5)

        if strategy == 0:
            # Single coordinate decrease, redistribute
            pos = rng.integers(0, n)
            if a[pos] < 1e-12:
                continue
            factor = 1.0 - rng.uniform(0.02, 0.25)
            a_trial = a.copy()
            a_trial[pos] *= factor
            diff = a[pos] - a_trial[pos]
            if diff < 1e-16:
                continue
            mask = np.ones(n, dtype=bool)
            mask[pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] = others + diff * others / others_sum

        elif strategy == 1:
            # Localized multi-coordinate perturbation
            center = rng.integers(0, n)
            radius = int(rng.integers(3, max(5, n // 40)))
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            a_trial = a.copy()
            a_trial[lo:hi] *= rng.uniform(0.75, 1.25, size=hi - lo)
            a_trial[lo:hi] = np.maximum(a_trial[lo:hi], 0)

        elif strategy == 2:
            # Mass transfer between two positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue
            frac = rng.uniform(0.01, 0.15)
            transfer = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= transfer
            a_trial[j] += transfer

        elif strategy == 3:
            # Peak-directed: reduce top contributors
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            k_top = int(rng.integers(1, 5))
            top = idx[np.argsort(contribs)[::-1][:k_top]]
            a_trial = a.copy()
            for pos in top:
                if a_trial[pos] > 1e-12:
                    a_trial[pos] *= (1.0 - rng.uniform(0.005, 0.08))

        else:
            # Global small Gaussian perturbation
            scale = rng.uniform(0.001, 0.02)
            a_trial = a + rng.standard_normal(n) * scale
            a_trial = np.maximum(a_trial, 0)

        a_trial_sum = a_trial.sum()
        if a_trial_sum < 1e-15:
            continue
        a_trial = a_trial / a_trial_sum

        score = evaluate_local(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            break

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.5) ** 4)
    elif choice == 6:
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif choice == 7:
        a = np.cos(np.pi * (x - 0.5)) ** 4 + 0.05
    elif choice == 8:
        a = np.exp(-5 * (x - 0.5) ** 2) * (1 + 0.2 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        # Flat with slight center emphasis
        a = 1.0 + 0.3 * np.exp(-20 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary engine: FFT-accelerated LogSumExp L-BFGS (O(n log n) per gradient).
    Auxiliary: mirror descent, coordinate descent, random perturbation.
    Multiple diverse restarts to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label, verbose=True):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            if verbose:
                print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Main L-BFGS with FFT gradient ===
    print(f"\n=== Phase 1: LogSumExp L-BFGS FFT (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.3,
                    1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.08,
                    0.05, 0.03, 0.015, 0.008]
        a_lb, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                                  maxiter_per_stage=15000)
        try_update(a_lb, "lbgfs_p1")

    # === Phase 2: Mirror Descent (fast global structure) ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, eta0=0.04, active_frac=0.997)
        try_update(a_md, "mirror_descent")

    # === Phase 3: Diverse restarts (explore different basins) ===
    print(f"\n=== Phase 3: Diverse Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_per_restart = min(60.0, rem * 0.08)
    n_restarts = min(4, int(rem * 0.35 / max(30, t_per_restart)))
    for r in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break
        # Generate diverse init
        a_init = best_seq.copy()
        # Perturb the parent
        perturb_scale = 0.1 + 0.05 * r
        a_init = a_init + rng.standard_normal(len(a_init)) * perturb_scale
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        # Or use structured init
        if r % 2 == 1:
            a_init = structured_init(n_best, rng)
            # Blend with current best
            a_init = 0.5 * best_seq + 0.5 * a_init
            a_init = a_init / a_init.sum()

        T_stages_r = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_r, _ = logsumexp_lbgfs(a_init, t_per_restart, T_stages=T_stages_r,
                                  maxiter_per_stage=8000)
        try_update(a_r, f"restart_{r}")

    # === Phase 4: Refine best with L-BFGS (lower T) ===
    print(f"\n=== Phase 4: L-BFGS Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages2 = [5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.8, 0.6,
                     0.45, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.015, 0.008, 0.004]
        a_lb2, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages2,
                                   maxiter_per_stage=20000)
        try_update(a_lb2, "lbgfs_p2")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 130.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coord_descent(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Random Perturbation Search ===
    print(f"\n=== Phase 6: Random Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_rp, _ = random_perturb_search(best_seq, t_alloc, n_trials=100000)
        try_update(a_rp, "random_perturb")

    # === Phase 7: Final L-BFGS at very low T ===
    print(f"\n=== Phase 7: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_fin = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25,
                        0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        a_fin, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages_fin,
                                   maxiter_per_stage=20000)
        try_update(a_fin, "final_lbgfs")

    # === Phase 8: Try different sequence lengths ===
    print(f"\n=== Phase 8: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for n_try in [900, 1000, 1400, 1600]:
            if n_try == n_best:
                continue
            if time.monotonic() > deadline - 40:
                break
            a_try = interpolate_seq(best_seq, n_try)
            t_try = min(50.0, (deadline - time.monotonic()) * 0.35)
            if t_try < 15:
                break
            T_stages_try = [10.0, 6.0, 4.0, 3.0, 2.0, 1.5, 1.0, 0.7,
                            0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_opt, _ = logsumexp_lbgfs(a_try, t_try, T_stages=T_stages_try,
                                       maxiter_per_stage=6000)
            try_update(a_opt, f"n={n_try}")

    # === Phase 9: Final coordinate polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 40.0)
    if t_alloc > 5:
        a_polish, _ = coord_descent(best_seq, t_alloc)
        try_update(a_polish, "final_polish")

    # Final verification
    if best_seq is not None:
        fs_local = evaluate_local(best_seq)
        print(f"\nFINAL (local): {fs_local:.8f} (n={len(best_seq)})")
        # Also verify with the actual evaluator
        try:
            fs_eval = evaluate_sequence([max(0.0, min(1000.0, float(x))) for x in best_seq])
            print(f"FINAL (evaluator): {fs_eval}")
        except Exception as e:
            print(f"Evaluation check error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
