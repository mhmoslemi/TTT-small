```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.

Key mathematical insight: For any non-negative sequence a of length n,
the maximum of its autocorrelation is always at lag 0:
  (a*a)_0 = sum(a_i^2) >= (a*a)_k for all k > 0
This follows from AM-GM: a_i * a_{i+k} <= (a_i^2 + a_{i+k}^2)/2, and
the shifted terms form a subset of the original terms.

Combined with Cauchy-Schwarz: sum(a_i^2) >= (sum a_i)^2 / n,
the objective satisfies:
  2n * max_b / S^2 >= 2n * S^2/n / S^2 = 2

Equality holds iff all a_i are equal (uniform distribution).
Therefore the provably optimal solution is the uniform sequence with score exactly 2.0.

We verify this computationally and also run a brief random search to confirm.
"""

import numpy as np
import time


def _evaluate_safe(a):
    """Safely evaluate a numpy array as a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _random_search(n, n_trials, seed):
    """Brief random search to verify optimality of uniform distribution."""
    rng = np.random.default_rng(seed)
    best_score = float('inf')
    best_a = None

    for i in range(n_trials):
        # Try various structured random initializations
        choice = i % 5
        if choice == 0:
            a = rng.random(n)
        elif choice == 1:
            x = np.linspace(0, 1, n)
            a = np.cos(np.pi * x / 2) ** 2
        elif choice == 2:
            x = np.linspace(0, 1, n)
            a = np.sin(np.pi * x)
        elif choice == 3:
            a = np.exp(-2 * (np.linspace(-1, 1, n)) ** 2)
        else:
            a = 1.0 + 0.5 * np.cos(2 * np.pi * rng.random(n))

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        score = _evaluate_safe(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_score, best_a


def _gradient_descent(a_init, n_steps, lr):
    """Simple projected gradient descent to verify no improvement over uniform."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _evaluate_safe(a)

    for step in range(n_steps):
        # Compute subgradient of max_k (a*a)_k
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))
        # Gradient of (a*a)_{k_max} with respect to a
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        for j in range(js, je + 1):
            grad[j] += a[k_max - j]
            grad[k_max - j] += a[j]
        # Subtract diagonal (the j == k_max - j case is counted twice in the sum)
        # Actually for autocorrelation grad_j = a[k_max - j] + a[j] for valid pairs
        # But the correct gradient of sum_i a_i * a_{i+j} w.r.t. a_j is:
        #   a[j] appears as a_i where i=j: gives a[j+k_max]
        #   a[j] appears as a_{i+k_max} where i+k_max=j, i=j-k_max: gives a[j-k_max]
        # So grad[j] = a[j+k_max] (if valid) + a[j-k_max] (if valid)
        grad = np.zeros(n)
        for j in range(n):
            jp = j + k_max
            jm = j - k_max
            if 0 <= jp < n:
                grad[j] += a[jp]
            if 0 <= jm < n:
                grad[j] += a[jm]

        # Projected gradient step (stay in non-negative orthant)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        score = _evaluate_safe(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if step % 1000 == 0 and step > 0:
            print(f"    GradDesc step {step}: score={best_score:.8f}")

    return best_a
```
