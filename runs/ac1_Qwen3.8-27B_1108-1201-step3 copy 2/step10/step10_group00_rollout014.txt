import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def convolve_at_fixed_a(a, b):
    """Compute (a * b) where a is fixed. Returns vector of length 2n-1."""
    return convolve_ab(a, b)


def get_subgradient_at_k(a, k, n):
    """Get the subgradient vector for the k-th entry of conv(a, ?).
    The k-th entry of conv(a, b) is sum_j a[j] * b[k-j].
    So the gradient w.r.t. b is a shifted version of a.
    Specifically, grad[k] = a[k-j] for j where 0 <= k-j < n.
    """
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] = a[k - j]
    return grad


def get_subgradient_at_k_fast(a, k, n):
    """Faster version using vectorized operations."""
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        # grad[j] = a[k-j] for j in [js, je]
        # Let i = k-j, then i ranges from k-je to k-js
        i_start = k - je
        i_end = k - js
        grad[js:je+1] = a[i_start:i_end+1][::-1]
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def fast_convolve_and_subgrad(a, b, top_k=1):
    """Compute conv(a,b) and return the top-k peaks with their subgradients."""
    c = convolve_ab(a, b)
    n = len(a)
    # Find top-k peaks
    if top_k == 1:
        k_max = np.argmax(c)
        max_val = c[k_max]
        grad = get_subgradient_at_k_fast(a, k_max, n)
        return max_val, k_max, grad
    else:
        # Find top-k indices
        k_indices = np.argsort(c)[::-1][:top_k]
        max_val = c[k_indices[0]]
        # Weighted subgradient
        weights = c[k_indices]
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, k_indices):
            grad += w * get_subgradient_at_k_fast(a, k, n)
        return max_val, k_indices[0], grad


def inner_subgradient(a_fixed, b_init, n_inner_steps, top_k=3):
    """Solve min_b max_k conv(a_fixed, b)[k] s.t. b in simplex using projected subgradient."""
    n = len(a_fixed)
    b = b_init.copy()
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    
    best_b = b.copy()
    best_max = convolve_ab(a_fixed, b).max()
    
    for t in range(n_inner_steps):
        max_val, k_max, grad = fast_convolve_and_subgrad(a_fixed, b, top_k=top_k)
        
        if max_val < best_max:
            best_max = max_val
            best_b = b.copy()
        
        # Polyak-like step size
        # Lower bound on optimal: 1/(2n-1) (trivial)
        # Upper bound: current max_val
        # Step size: (max_val - lower_bound) / (||grad||^2 + eps)
        lb = 1.0 / (2 * n - 1)  # trivial lower bound on max of conv for prob vector
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-15:
            eta = 1.0 / n_inner_steps
        else:
            eta = (max_val - lb) / (grad_norm_sq + 1e-15)
            eta = min(eta, 2.0 / (t + 1))  # cap the step size
        
        b = project_simplex(b - eta * grad)
    
    return best_b, best_max

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def subgrad_at_k(a, k, n):
    """Subgradient vector for k-th entry of conv(a, b) w.r.t. b."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return np.zeros(n)
    idx = np.arange(js, je + 1)
    return np.concatenate([
        np.zeros(js),
        a[k - idx],
        np.zeros(n - 1 - je)
    ])


def subgrad_at_k_vec(a, k, n):
    """Vectorized subgradient computation."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    g = np.zeros(n)
    if js <= je:
        g[js:je + 1] = a[(k - js):(k - je + 1)][::-1]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    sv = v.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    ws = w.sum()
    if ws < 1e-15:
        return np.ones(n) / n
    return w / ws


def inner_subgradient_fast(a_fixed, b_init, n_steps=30, top_k=5):
    """Fast inner problem: min_b max_k conv(a_fixed, b)[k] via projected subgradient."""

    else:
        b = b / s

        c = convolve_ab(a_fixed, b)
        max_val = c.max()

            best_max = max_val
            best_b = b.copy()

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, temp):
    """Compute LogSumExp objective and gradient w.r.t. z in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / temp)
    lse = max_b + temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

        if remaining < 3:
            break

        iters_here = max(20, total_iters // len(T_stages))
        if stage_i == len(T_stages) - 1:
            iters_here = max(iters_here, 50)

        def func(z_x):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(z_x, n, stage_T)
            return o

        def jac(z_x):
            if time.monotonic() > deadline:
                return np.zeros_like(z_x)
            _, g = logsumexp_obj_grad(z_x, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_here, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_here * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if (stage_i + 1) % 4 == 0:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"      L-BFGS stage {stage_i+1}/{len(T_stages)} (T={stage_T}): score={sc:.8f}")

    return softmax_from_z(z)


def multiplicative_subgradient_anneal(a0, time_limit, n_steps, initial_temp=1.0):
    """Mirror descent with top-k active set and simulated annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    temp = initial_temp
    cool_rate = 0.9995
    no_improve = 0
    rng = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Top-k active set
        k = 5
        sorted_idx = np.argsort(b)[::-1][:k]
        weights = b[sorted_idx]
        weights = np.exp(weights / temp - weights[0] / temp)
        weights = weights / weights.sum()

        # Weighted subgradient
        g = np.zeros(n)
        for w, k_idx in zip(weights, sorted_idx):
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k_idx - idx]

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()

        # Simulated annealing acceptance
        if score_new < score:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = score_new - score
            accept_prob = np.exp(-delta / temp) if temp > 1e-10 else 0.0
            if rng.random() < accept_prob:
                a = a_new
            no_improve += 1

        # Adapt parameters
        if no_improve >= 50:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-5:
                break

        temp *= cool_rate

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGradAnneal {elapsed:.0f}s: best={best_score:.8f} eta={eta:.6f} T={temp:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-5 * (x - 0.5)))
    else:
        a = np.ones(n) / n * (1 + 0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    sa = a.sum()
    if sa < 1e-15:
        return np.ones(n) / n
    return a / sa


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 6)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses hybrid Nesterov alternating (fast subgradient + LP), LogSumExp L-BFGS,
    annealed multiplicative subgradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Hybrid Nesterov alternating (fast + LP) ===
    print(f"\n=== Phase 1: Hybrid Nesterov n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 650.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_hybrid_alternating(best_seq, t_alloc, solver, n_max_iters=500000)
        local_update(a_lp, "hybrid_nesterov_1")

    # === Phase 2: LogSumExp L-BFGS fine ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.02, 0.01, 0.005, 0.003]
        print(f"  T_stages ({len(T_stages)} stages), budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine_multi(best_seq, t_alloc, T_stages, maxiter_per_stage=15000)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 3: Annealed multiplicative subgradient ===
    print(f"\n=== Phase 3: Annealed MultiSubGrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 130.0)
    if t_alloc > 15:
        a_ms, _ = multiplicative_subgradient_anneal(best_seq, t_alloc, n_steps=500000, initial_temp=2.0)
        local_update(a_ms, "mult_subgrad_anneal")

    # === Phase 4: Second Hybrid Nesterov (from updated best) ===
    print(f"\n=== Phase 4: Hybrid Nesterov Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_hybrid_alternating(best_seq, t_alloc, solver, n_max_iters=200000)
        local_update(a_lp2, "hybrid_nesterov_2")

    # === Phase 5: Try n=1800 ===
    print(f"\n=== Phase 5: Try n=1800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1800 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_n1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800, _ = nesterov_hybrid_alternating(init_1800, t_n1800 * 0.6, solver, n_max_iters=150000)
        local_update(a_1800, "hybrid_n1800")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1800 and rem > 40:
            T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_1800b = lbgfs_refine_multi(best_seq, min(rem, 100.0), T_stages2, maxiter_per_stage=8000)
            local_update(a_1800b, "lbgfs_n1800")

    # === Phase 6: Try n=2400 ===
    print(f"\n=== Phase 6: Try n=2400 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_n2400 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_n2400:.0f}s")
        init_2400 = interpolate_sequence(best_seq, 2400)
        a_2400, _ = nesterov_hybrid_alternating(init_2400, t_n2400 * 0.65, solver, n_max_iters=80000)
        local_update(a_2400, "hybrid_n2400")

    # === Phase 7: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 7: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.15, 120.0)
        n_cur = len(best_seq)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.015 + 0.01 * trial)
            t_sub = min(rem_sub * 0.25, 45.0)
            a_opt, _ = nesterov_hybrid_alternating(a_pert, t_sub, solver, n_max_iters=100000)
            local_update(a_opt, f"pert_trial{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.40, 60.0)
        if t_fin > 10:
            T_stages3 = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
            a_fin = lbgfs_refine_multi(best_seq, t_fin, T_stages3, maxiter_per_stage=8000)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_final_ms, _ = multiplicative_subgradient_anneal(best_seq, min(rem * 0.5, 25.0), n_steps=100000, initial_temp=0.5)
        local_update(a_final_ms, "final_mult")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_final_lp, _ = nesterov_hybrid_alternating(best_seq, min(rem * 0.6, 20.0), solver, n_max_iters=30000)
        local_update(a_final_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
