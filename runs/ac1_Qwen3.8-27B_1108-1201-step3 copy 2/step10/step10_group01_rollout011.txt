```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, active):
    """Compute subgradient of max_k (a*a)[k] averaged over active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        # g[i] += a[k-i] for i in [i_start, i_end]
        # a[k-i] for i in [i_start, i_end] = a[k-i_end : k-i_start+1][::-1]
        g[i_start:i_end+1] += a[k-i_end+1:k-i_start+1][::-1]
    g /= len(active)
    return g


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def active_set_optimize(a0, time_limit, n_steps=5000000, print_interval=50000):
    """Fast active-set subgradient optimization with simplex projection."""

    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Parameters
    eps = 0.03  # initial step size
    min_eps = 1e-6
    active_frac = 0.995  # peaks within this fraction of max are "active"
    patience_count = 0
    max_patience = 200
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b  # since sum(a) = 1
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            patience_count = 0
        else:
            patience_count += 1
        
        # Find active set
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        g = compute_subgradient(a, active)
        
        # Gradient step + projection
        a_new = a - eps * g
        a_new = project_simplex(a_new)
        
        # Check if it's an improvement
        b_new = fast_convolve(a_new)
        max_b_new = b_new.max()
        score_new = 2.0 * n * max_b_new
        
        if score_new < current_score:
            a = a_new
            # Maybe increase step size slightly
            eps = min(eps * 1.02, 0.1)
        else:
            # Rejection: decrease step size
            eps *= 0.85
            if eps < min_eps:
                eps = min_eps
        
        # Periodic restart
        if patience_count >= max_patience:
            # Small random perturbation
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = project_simplex(a)
            eps = 0.03
            patience_count = 0
            max_patience = max(100, max_patience + 50)
        
        if step % print_interval == 0:

            print(f"    AS iter={step} ({elapsed:.0f}s): score={best_score:.10f} eps={eps:.6f} |active|={len(active)}")

def solve_lp_fast(a_fixed, max_iters=200):
    """Fast LP: given fixed a (sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    # Build T: T[k,j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # Actually T @ b = convolve(a_fixed, b)
    T = np.zeros((size, n))
    idx = np.arange(n)
    for j in range(n):
        k_start = j
        k_end = j + n
        T[k_start:k_end, j] = a_fixed  # T[k,j] = a_fixed[k-j] for k in [j, j+n-1]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Try solvers
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iter=max_iters, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

    rng = np.random.default_rng(777)

            stall += 1

            continue

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:

            best_a = y.copy()

            if stall >= 15:
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, best_score
```

```python
            eps *= 0.82
            if eps < min_eps:
                eps = min_eps
            # Reset momentum on rejection
            if use_momentum and n_consecutive_reject % 10 == 0:
                momentum = max(0.3, momentum - 0.05)

        # Patience-based restart
        if patience_count % 50 == 49:
            patience_count += 1
        if patience_count >= max_patience:
            noise_scale = min(0.03, 0.01 * np.sqrt(patience_count / max_patience))
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = project_simplex(a)
            v = np.zeros(n)
            eps = 0.03
            momentum = 0.6 if use_momentum else 0.0
            patience_count = 0
            max_patience = min(max_patience + 100, 1000)

        if step > 0 and step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS iter={step} ({elapsed:.0f}s): best={best_score:.10f} "
                  f"eps={eps:.6f} mom={momentum:.2f} |act|={len(active)}")

    return best_a, best_score


def solve_lp_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver] if solver else [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for sol in solvers:
        try:
            prob.solve(solver=sol, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def find_lp_solver():
    """Find a working LP solver."""
    for sol in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sol, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return sol
        except Exception:
            continue
    return cp.ECOS


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=50000):
    """Nesterov dual-averaging LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(555)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 40:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 20:
                noise = rng.standard_normal(n) * 0.015
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
            if (it + 1) % 1500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} M={M_b:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: fast active-set subgradient (primary),
    periodic LP refinement, momentum-based acceleration, stochastic
    restarts, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.10f}")

    def do_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Primary active-set optimization ===
    print(f"\n=== Phase 1: Active-set (frac=0.995), score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_res, _ = active_set_optimize(best_seq, t1, n_steps=20000000,
                                       active_frac=0.995, use_momentum=True)
        do_update(a_res, "as1")

    # === Phase 2: Tighter active set (focus on top peaks) ===
    print(f"\n=== Phase 2: Active-set tight (frac=0.999), score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_res, _ = active_set_optimize(best_seq, t2, n_steps=20000000,
                                       active_frac=0.999, use_momentum=False)
        do_update(a_res, "as2_tight")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 350.0)
    if t3 > 40:
        print(f"  Budget: {t3:.0f}s")
        solver = find_lp_solver()
        print(f"  LP solver: {solver}")
        a_lp, _ = nesterov_lp_optimize(best_seq, t3, solver=solver, max_iters=30000)
        do_update(a_lp, "lp3")

    # === Phase 4: Active-set loose (broader averaging) ===
    print(f"\n=== Phase 4: Active-set loose (frac=0.98), score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a_res, _ = active_set_optimize(best_seq, t4, n_steps=20000000,
                                       active_frac=0.98, use_momentum=True)
        do_update(a_res, "as4_loose")

    # === Phase 5: Stochastic perturbation escape ===
    print(f"\n=== Phase 5: Stochastic, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 80.0)
        for pi in range(4):
            if time.monotonic() > deadline - 15:
                break
            if t_each < 15:
                break
            if pi == 0:
                x = np.linspace(0, 6 * np.pi, n_cur)
                pert = 0.015 * np.sin(x) + 0.008 * np.cos(3 * x)
            elif pi == 1:
                pert = 0.01 * rng.standard_normal(n_cur)
            elif pi == 2:
                # Targeted: boost low-convolution positions
                b = fast_convolve(best_seq)
                low_mask = b < np.percentile(b, 10)
                pert = np.zeros(n_cur)
                pert[low_mask] = 0.005
            else:
                # Anti-symmetric perturbation
                half = n_cur // 2
                pert = np.zeros(n_cur)
                pert[:half] = 0.01 * rng.standard_normal(half)
                pert[half:] = -pert[:half]
            a_pert = best_seq + pert
            a_pert = project_simplex(a_pert)
            a_res, _ = active_set_optimize(a_pert, t_each, n_steps=5000000,
                                           active_frac=0.995, use_momentum=True)
            do_update(a_res, f"stoch_{pi}")

    # === Phase 6: Alternative length ===
    print(f"\n=== Phase 6: Alt length, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.15)
        t_alt = min(rem * 0.4, 120.0)
        if t_alt > 20:
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = active_set_optimize(init_alt, t_alt, n_steps=10000000,
                                           active_frac=0.995, use_momentum=True)
            do_update(a_alt, f"alt_n{alt_n}")

    # === Phase 7: Final polishing (very tight active set, small steps) ===
    print(f"\n=== Phase 7: Final polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.7, 50.0)
        a_fin, _ = active_set_optimize(best_seq, t_fin, n_steps=10000000,
                                       active_frac=0.9995, use_momentum=False)
        do_update(a_fin, "final_tight")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin2, _ = active_set_optimize(best_seq, min(rem * 0.8, 30.0),
                                        n_steps=10000000, active_frac=0.998,
                                        use_momentum=True)
        do_update(a_fin2, "final2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
