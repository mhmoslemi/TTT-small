This gives us a system of equations that we can solve. The key insight is that the "gradient" of the max is a convex combination of the gradients at the active peaks, and this should be "flat" (proportional to 1) on the support of `a`.

Algorithm:
1. Find the active set S (peaks within ε of max)
2. Solve for λ_k such that the weighted sum of peak gradients is as flat as possible on the support
3. Use this to compute a direction to move

This is essentially what the LP does, but we can be more explicit.

def kkt_refinement(a, solver, n_iters=100):

        b = fast_convolve(a)
        M = b.max()
        # Active set: peaks within 0.1% of max
        active = np.where(b >= M * 0.999)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # For each active peak k, the gradient w.r.t. a[j] is:
        # d/d(a[j]) conv(a,a)[k] = 2*a[k-j] (if j≠k-j) or a[k-j] (if j=k-j)
        # More precisely: d/d(a[j]) sum_i a[i]*a[k-i] = a[k-j] + (j≠k-j)*a[k-j] = (1+jitter)*a[k-j]
        
        # Build gradient matrix G: G[j, i] = d/d(a[j]) conv(a,a)[active[i]]
        m = len(active)
        G = np.zeros((n, m))
        for i, k in enumerate(active):
            for j in range(n):
                j2 = k - j
                if 0 <= j2 < n:
                    if j == j2:
                        G[j, i] = 1.0
                    else:
                        G[j, i] = 2.0 * a[j2]
        
        # We want to find λ ≥ 0, sum(λ) = 1, such that G^T @ λ is as constant as possible
        # on the support of a (where a > 0)
        # This is equivalent to: minimize ||G^T @ λ - c * 1_support||^2 + regularization
        # subject to λ ≥ 0, sum(λ) = 1
        
        # Simpler: just set λ_k = (b[k] - M + eps) / sum(b[k] - M + eps) for active k
        # Or use the LP dual
        
        # For now, use uniform weights on active peaks
        lam = np.ones(m) / m
        
        # Gradient direction: g = G @ lam (this is the "subgradient" of the max)
        g = G @ lam
        
        # We want to move in the direction that reduces the max
        # The projected gradient (onto sum=1, a≥0) of the objective is:
        # g_proj = g - (sum(g * a) / sum(a)) * 1 (for the sum constraint)
        # Then clip to non-negative
        
        # Actually, for the problem min max_k conv s.t. sum(a)=1, a≥0:
        # The subgradient method step is:
        # a_new = a - alpha * (g - mean(g * support_mask) * support_mask)
        # where support_mask = (a > 0)
        
        # But we also need to handle the non-negativity constraint
        # Simple approach: projected subgradient
        g_centered = g - np.dot(g, a)  # center so that sum(g_centered * a) = 0
        # This ensures the step doesn't change the sum to first order
        
        alpha = 0.01
        a_new = a - alpha * g_centered
        a_new = np.maximum(a_new, 0)

            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.5
    
    return best_a, best_score

def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def build_conv_matrix(a):
    """T @ b = convolve(a, b), shape (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, solver, frac=0.95):
    """
    Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Only include constraints where conv > frac * max to reduce problem size.
    """
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    # Compute current conv with a_fixed itself to identify active constraints
    b_current = T @ a_fixed
    M_current = b_current.max()
    
    # Select constraints: include all where b_current > frac * M_current
    thresh = frac * M_current
    active_idx = np.where(b_current >= thresh)[0]
    
    T_active = T[active_idx, :]
    m = len(active_idx)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for sol in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None

    y_b = b.copy()

    rng = np.random.default_rng(12345)

    stall = 0

            break

        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            
        a_new = lp_best_response(y_b, solver, frac=0.95)

                best_a = y_a.copy()
            if sc_b < best_score:

def spectral_refinement(a0, time_limit=60, n_iters=500):
    """Gerchberg-Saxton-like spectral refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Zero-pad to length >= 2n-1
    N = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, N)
        # Current autocorrelation
        acorr = irfft(np.abs(A)**2, N)  # circular autocorrelation
        # Linear autocorrelation is acorr[0:n] (approximately, for zero-padded)
        # Actually, the linear conv is the first (2n-1) entries of the circular conv
        # when N >= 2n-1
        
        # Target: flatten |A|^2
        # Strategy: |A|^2_target = C / (1 + alpha * acorr)
        # This gives more weight to lag positions where acorr is high
        
        # Simpler: just try to make |A|^2 proportional to 1/(1+eps*acorr)
        # where acorr is the current (time-domain) autocorrelation
        
        P = np.abs(A)**2  # current power spectrum
        # Desired: reduce contribution from high-acorr lags
        # In frequency domain, this means making P flatter
        
        # Approach: P_target[k] = P[0] / N (flat spectrum)
        # But blend: P_new = (1-t)*P + t*P_flat
        t = 0.1  # blending parameter
        
        P_flat = np.full_like(P, P[0] / N)
        P_new = (1 - t) * P + t * P_flat
        
        # But P[0] = (sum a)^2 = 1 (since normalized)
        # So P_flat[0] should be 1, and P_flat[k] = 1/N for k>0
        P_flat[0] = P[0]  # keep DC
        
        # New spectrum magnitude
        mag_new = np.sqrt(P_new)
        # Keep phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse transform
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                t = 0.05
    
    return best_a, best_score

    return None


def adaptive_lp_alternation(a0, time_limit, solver, max_iters=200000):
    n = len(a0)
    a = normalize(a0)
    y_a = a.copy()
    y_b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    prev_best = best_score
    stall = 0
    frac_current = 0.95
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Progressively tighten the active set
        frac_current = max(0.99, 0.95 + 0.04 * min(1.0, it / 2000.0))
        
        b_new = lp_best_response(y_a, solver, frac=frac_current)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
        
        a_new = lp_best_response(y_b, solver, frac=frac_current)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
        
        if (it + 1) % 10 == 0:
            sc_a = 2.0 * n * fast_convolve(y_a).max()
            sc_b = 2.0 * n * fast_convolve(y_b).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = y_a.copy()
            if sc_b < best_score:
                best_score = sc_b
                best_a = y_b.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 4:
                # Escape: try several noisy restarts with short LP
                escaped = False
                for attempt in range(5):
                    scale = 0.03 + 0.02 * attempt
                    y_a_n = normalize(y_a + rng.standard_normal(n) * scale / math.sqrt(n))
                    y_b_n = normalize(y_b + rng.standard_normal(n) * scale / math.sqrt(n))
                    for sub in range(30):
                        bs = lp_best_response(y_a_n, solver, frac=0.90, max_constraints=200)
                        if bs is not None:
                            y_b_n = normalize(0.5 * y_b_n + 0.5 * bs)
                        as_ = lp_best_response(y_b_n, solver, frac=0.90, max_constraints=200)
                        if as_ is not None:
                            y_a_n = normalize(0.5 * y_a_n + 0.5 * as_)
                    sc = 2.0 * n * fast_convolve(y_a_n).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = y_a_n.copy()
                        y_a = y_a_n.copy()
                        y_b = y_b_n.copy()
                        stall = 0
                        escaped = True
                        break
                if not escaped:
                    y_a = normalize(y_a + rng.standard_normal(n) * 0.01 / math.sqrt(n))
                    y_b = normalize(y_b + rng.standard_normal(n) * 0.01 / math.sqrt(n))
                    stall = 0
            
            if (it + 1) % 2000 == 0:
                print(f"  LP alt it={it+1} ({time.monotonic()-t0:.0f}s): {best_score:.10f}")
    
    return best_a, best_score


def kkt_subgradient(a0, time_limit, n_steps=50000):
    """Subgradient with adaptive step size and active set tracking."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    rng = np.random.default_rng(99)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        
        g /= len(active)
        # Center to preserve sum to first order
        g = g - np.dot(g, a)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                lr = min(lr * 1.005, 0.05)
            else:
                no_improve += 1
                if no_improve >= 30:
                    lr *= 0.93
                    no_improve = 0
                    # Occasional noise
                    if rng.random() < 0.3:
                        a = normalize(a + rng.standard_normal(n) * 0.01 / math.sqrt(n))
        lr = max(1e-8, lr)
    
    return best_a, best_score


def spectral_gs_refinement(a0, time_limit=60):
    """Gerchberg-Saxton style spectral refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    N = 1 << (2 * n - 1).bit_length()
    
    blend_schedules = [
        (0.05, 200), (0.1, 200), (0.2, 100), (0.3, 100), (0.5, 100)
    ]
    
    for blend, max_iters in blend_schedules:
        if time.monotonic() - t0 > time_limit:
            break
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            A = rfft(a, N)
            P = np.abs(A)**2
            # Target: flatten the spectrum
            P_target = np.full_like(P, P[0] / N)
            P_target[0] = P[0]
            
            P_new = (1 - blend) * P + blend * P_target
            mag_new = np.sqrt(np.maximum(P_new, 0))
            phase = np.angle(A)
            A_new = mag_new * np.exp(1j * phase)
            
            a_new = irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = 2.0 * n * fast_convolve(a_new).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
    
    return best_a, best_score


def pairwise_descent(a0, time_limit=60):
    """Pairwise coordinate descent for final polish."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    # Precompute: for each pair (j1, j2), the contribution to conv
    # conv(a,a)[k] = sum_i a[i]*a[k-i]
    # When we change a[j1] from old to t, and a[j2] from old2 to (c-t):
    # The change in conv[k] is:
    #   (t - old)*a[k-j1] + (c-t-old2)*a[k-j2] + (t-old)*(c-t-old2) * delta terms
    # This is complex, so just do direct evaluation
    
    n_pairs = min(3000, n * 3)
    
    for rnd in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            j1 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if j1 == j2:
                continue
            
            c = a[j1] + a[j2]
            if c < 1e-15:
                continue
            
            # Store old values
            old_j1 = a[j1]
            old_j2 = a[j2]
            
            def obj(t):
                if t < -1e-15 or t > c + 1e-15:
                    return 1e15
                t = np.clip(t, 0.0, c)
                a[j1] = t
                a[j2] = c - t
                b = fast_convolve(a)
                return b.max()
            
            try:
                res = minimize_scalar(obj, bounds=(max(0, old_j1 - 0.5*c), min(c, old_j1 + 0.5*c)),
                                     method='bounded', options={'xatol': 1e-10})
                t_opt = float(np.clip(res.x, 0, c))
            except Exception:
                t_opt = old_j1
            
            a[j1] = t_opt
            a[j2] = c - t_opt
            sc = 2.0 * n * fast_convolve(a).max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                improved = True
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2)**2
    elif choice == 1:
        a = np.sin(np.pi * x)**2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x)**1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2*(x-0.3)**2) + 0.5*np.exp(-2*(x-0.7)**2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50*(x-0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1*np.cos(2*np.pi*x) + 0.05*np.cos(4*np.pi*x)
    elif choice == 8:
        # Two-level structure
        a = np.ones(n)
        a[::2] = 0.7
    else:
        # Flat with boundary effects
        a = np.ones(n)
        boundary = max(1, n // 20)
        a[:boundary] = np.linspace(0.5, 1.0, boundary)
        a[-boundary:] = np.linspace(1.0, 0.5, boundary)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = max(20, int(remaining * 15))
        
        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternation, spectral refinement,
    KKT subgradient, coordinate descent, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return False
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
            return True
        return False
    
    # === Phase 1: Adaptive LP alternation (main workhorse) ===
    print(f"\n=== Phase 1: Adaptive LP (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = adaptive_lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")
    
    # === Phase 2: KKT subgradient ===
    print(f"\n=== Phase 2: KKT Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 120.0)
    if t2 > 10:
        a_sg, _ = kkt_subgradient(best_seq, t2)
        try_update(a_sg, "kkt_sg")
    
    # === Phase 3: Spectral GS refinement ===
    print(f"\n=== Phase 3: Spectral GS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 60.0)
    if t3 > 5:
        a_sp, _ = spectral_gs_refinement(best_seq, t3)
        try_update(a_sp, "spectral_gs")
    
    # === Phase 4: Second LP round (tighter) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, 400.0)
    if t4 > 30:
        a_lp2, _ = adaptive_lp_alternation(best_seq, t4, solver)
        try_update(a_lp2, "lp2")
    
    # === Phase 5: LogSumExp L-BFGS ===
    print(f"\n=== Phase 5: L-BFGS LogSumExp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 20:
        T_st = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb, _ = lbgfs_logsumexp(best_seq, t5, T_stages=T_st)
        try_update(a_lb, "lbgfs")
    
    # === Phase 6: Pairwise coordinate descent ===
    print(f"\n=== Phase 6: Pairwise Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 60.0)
    if t6 > 5:
        a_pw, _ = pairwise_descent(best_seq, t6)
        try_update(a_pw, "pairwise")
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        base_n = n_best
        n_candidates = []
        for mult in [0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.7]:
            cand_n = int(base_n * mult)
            if 200 <= cand_n <= 8000 and cand_n != base_n and cand_n not in n_candidates:
                n_candidates.append(cand_n)
        
        # Sort by proximity to base_n
        n_candidates.sort(key=lambda x: abs(x - base_n))
        
        for cand_n in n_candidates[:5]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            
            # Quick LP
            a_cand, _ = adaptive_lp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lp_n{cand_n}")
            
            rem = deadline - time.monotonic()
            if rem > 20:
                T_sub = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                start = a_cand if a_cand is not None else init_cand
                a_cand2, _ = lbgfs_logsumexp(start, min(rem * 0.4, 50.0), T_stages=T_sub)
                try_update(a_cand2, f"lbgfs_n{cand_n}")
    
    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Short LP
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            a_fin, _ = adaptive_lp_alternation(best_seq, t_fin, solver)
            try_update(a_fin, "final_lp")
        
        # Subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = kkt_subgradient(best_seq, min(rem * 0.3, 30.0))
            try_update(a_fin2, "final_sg")
        
        # L-BFGS very low T
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = lbgfs_logsumexp(best_seq, min(rem * 0.3, 30.0),
                                        T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01])
            try_update(a_fin3, "final_lbgfs")
        
        # Final pairwise
        rem = deadline - time.monotonic()
        if rem > 5:
            a_fin4, _ = pairwise_descent(best_seq, min(rem * 0.5, 20.0))
            try_update(a_fin4, "final_pw")
    
    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
