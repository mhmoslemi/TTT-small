Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using parametric initialization,
LP-based alternating optimization with adaptive step size, and
precision polishing via coordinate descent.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

def fast_convolve(a):

        prev_g = g.copy()
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys
        
        a = a_new
        score = evaluate_local(a)
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve >= 600:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = a + noise
            a = np.maximum(a, 0)
            as_sum = a.sum()
            if as_sum > 1e-15:
                a = a / as_sum
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.10f} eta={eta:.6f}")
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, n_steps=300000):
    """Aggressive coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(77777)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find significant peaks
        threshold = max_b * 0.99995
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.99995, 0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-20:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Try simultaneous multi-coordinate adjustment
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.9999
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts > 1e-15:
                    a_trial = a_trial / ts
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    elif stall > 3000:
                        break
        else:
            stall = 0
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish {elapsed:.0f}s: score={best_score:.10f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: parametric init, LP alternating (Nesterov),
    mirror descent, second LP pass, and coordinate descent polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(seed)
    solver = find_lp_solver()
    print(f"LP solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 0: Multi-length parametric search ===
    print(f"\n=== Phase 0: Parametric Search ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        n_candidates = [800, 1000, 1200, 1400, 1600, 1800, 2000, 2500]
        t_per = min(rem * 0.08 / len(n_candidates), 3.0)
        for n_cand in n_candidates:
            if time.monotonic() > deadline - 5:
                break
            a_cand, sc_cand = parametric_search(n_cand, t_per)
            if a_cand is not None and sc_cand < best_score:
                best_score = sc_cand
                best_seq = a_cand.copy()
                print(f"  NEW BEST [param_n{n_cand}]: {sc_cand:.10f}")
        
        # Also try interpolating parent to different lengths
        if parent is not None:
            for n_cand in [1000, 1500, 2000, 2500, 3000]:
                if time.monotonic() > deadline - 5:
                    break
                a_interp = interpolate_sequence(parent, n_cand)
                sc_interp = evaluate_local(a_interp)
                if sc_interp < best_score:
                    best_score = sc_interp
                    best_seq = a_interp.copy()
                    print(f"  NEW BEST [interp_n{n_cand}]: {sc_interp:.10f}")
    
    if best_seq is None:
        a_init, _ = parametric_search(1200, 5.0)
        best_seq = a_init
        best_score = evaluate_local(best_seq)
        print(f"Default: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    print(f"\nAfter Phase 0: score={best_score:.10f}, n={n_best}")
    
    # === Phase 1: LP Alternating (main workhorse, 45% time) ===
    print(f"\n=== Phase 1: LP Alternating (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 750.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver, phase=1)
        try_update(a_lp, "lp_p1")
    
    # === Phase 2: Mirror Descent (fast refinement, 15% time) ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, eta0=0.03, active_frac=0.995)
        try_update(a_md, "mirror")
    
    # === Phase 3: Second LP pass (15% time) ===
    print(f"\n=== Phase 3: LP Alternating Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating(best_seq, t_alloc, solver, phase=2)
        try_update(a_lp2, "lp_p2")
    
    # === Phase 4: Coordinate Descent Polish (15% time) ===
    print(f"\n=== Phase 4: Coordinate Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_polish(best_seq, t_alloc)
        try_update(a_cd, "coord")
    
    # === Phase 5: Final LP micro-refinement (10% time) ===
    print(f"\n=== Phase 5: Final LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_fin, _ = lp_alternating(best_seq, t_alloc, solver, phase=3)
        try_update(a_fin, "lp_final")
    
    # Final coordinate polish with remaining time
    rem = deadline - time.monotonic()
    if rem > 5:
        print(f"\n=== Final Polish (remaining {rem:.0f}s) ===")
        a_polish, _ = coordinate_polish(best_seq, rem - 2.0)
        try_update(a_polish, "final_polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
