"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with exact max subgradient + backtracking
2. LogSumExp smoothification + L-BFGS with multi-stage temperature annealing
3. Spectrum-flatness guided initialization for diversity
4. Adaptive-length search with interpolation between lengths
5. Peak equalization local search for final polish
Key improvements over previous: cleaner gradient via FFT cross-correlation,
proper LSE gradient, better time allocation, no Nesterov bugs, and
spectrum-aware initialization exploiting the anti-concentration structure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def compute_subgrad(a, n_active=30):
    """
    Compute smoothed subgradient of max_k (a*a)_k.
    Uses exp-weighted active set for robustness.
    Returns (g, max_value, b).
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), 0.0, b

    n_use = min(n_active, len(b))
    sorted_idx = np.argpartition(b, -n_use)[-n_use:]

    # Exponential weighting based on proximity to max
    diffs = max_b - b[sorted_idx]
    scale = max(max_b * 0.15, 1e-15)
    w_sel = np.exp(-diffs / scale)
    w_sel = w_sel / w_sel.sum()

    # Build full weight vector of length 2n-1
    w = np.zeros(2 * n - 1)
    w[sorted_idx] = w_sel

    # g_j = 2 * sum_k w_k * a_{k-j} = 2 * conv(w, a_rev)[n-1+j]
    a_rev = a[::-1]
    g_full = fast_conv_cross(w, a_rev)
    g = 2.0 * g_full[n - 1:2 * n - 1]

    return g, max_b, b


def mirror_descent(a0, time_limit, n_active=30, verbose=True):
    """Mirror descent with backtracking line search on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_b = fast_conv(a)
    best_score = 2.0 * n * best_b.max()
    t0 = time.monotonic()
    lr = 1.0
    no_improve = 0

    for step in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b, b = compute_subgrad(a, n_active)

        # Backtracking line search
        alpha = lr
        improved = False
        for bt in range(25):
            z_new = z - alpha * g
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            b_new = fast_conv(a_new)
            score_new = b_new.max()

            if score_new < best_b.max() - 1e-14:
                a = a_new
                z = z_new
                best_a = a_new.copy()
                best_b = b_new
                best_score = 2.0 * n * score_new
                no_improve = 0
                lr = min(alpha * 1.03, 10.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.75, 0.005)
            if no_improve >= 80:
                # Perturbation to escape local minimum
                rng = np.random.default_rng(step * 7 + 13)
                noise = rng.random(n) * 0.03
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = np.log(np.maximum(a, 1e-300))
                no_improve = 0
                lr = 1.0

        if verbose and step % 200000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      MD: step={step}, score={best_score:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a, best_score


def lse_score(z, T):
    """Compute 2n * LSE(T * conv(softmax(z), softmax(z)))."""
    n = len(z)
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_conv(a)
    x = T * b
    x_shifted = x - x.max()
    lse = b.max() + np.log(np.exp(x_shifted).sum()) / T
    return 2.0 * n * lse


def lse_grad(z, T):
    """Gradient of LSE score w.r.t. z (log-parameterization)."""
    n = len(z)
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_conv(a)

    # Softmax weights on peaks
    x = T * b
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    w = exp_x / exp_x.sum()

    # g_a = 2 * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    g_full = fast_conv_cross(w, a_rev)
    g_a = 2.0 * g_full[n - 1:2 * n - 1]

    # Project to simplex tangent: grad_z = a * (g_a - dot(g_a, a))
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)

    return grad_z


def lse_lbfgs(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=18):
    """L-BFGS on LogSumExp smoothification with geometric temperature schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Scale max iterations with temperature (higher T = easier = fewer iters needed)
        max_iters = max(30, int(80 * (T_start / T) ** 0.2))

        def func(z_in):
            return lse_score(z_in, T)

        def jac(z_in):
            return lse_grad(z_in, T)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iters, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxfun': max_iters * 200,
                                    'maxls': 50, 'disp': False})
            z = res.x
        except Exception:
            break

        if si % 4 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = 2.0 * n * fast_conv(a_cur).max()
            print(f"      LSE stage {si}/{n_stages} (T={T:.2f}): score={sc:.8f}")

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def spectrum_flatness_init(n, rng):
    """
    Initialize by minimizing spectral variance (making |A(w)|^2 flat).
    Uses a few gradient steps on var(|A|^2) = sum(b_k^2) - (sum(a_j^2))^2.
    """
    # Start from a raised-cosine or center-heavy shape
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2
    a = a + 0.1 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    a = a / a.sum()

    # Gradient descent on spectral variance
    t0 = time.monotonic()
    lr = 0.1
    for step in range(2000):
        if time.monotonic() - t0 > 10:
            break
        b = fast_conv(a)
        # grad of sum(b_k^2) w.r.t. a_j = 4 * conv(b, a_rev)[n-1+j]
        a_rev = a[::-1]
        g1 = 4.0 * fast_conv_cross(b, a_rev)[n - 1:2 * n - 1]
        # grad of (sum a_j^2)^2 = 4 * (sum a^2) * a_j
        sum_a2 = np.dot(a, a)
        g2 = 4.0 * sum_a2 * a
        g = g1 - g2
        a = a - lr * g
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / ss
        lr *= 0.999

    # Add small random perturbation
    noise = rng.random(n) * 0.01
    a = a + noise
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def diverse_init(n, rng):
    """Generate a diverse initialization."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Center-heavy (Gaussian)
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 1:
        # Bimodal
        x = np.linspace(-1, 1, n)
        a = np.exp(-8 * (x - 0.4) ** 2) + np.exp(-8 * (x + 0.4) ** 2)
    elif choice == 2:
        # Raised cosine with oscillation
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2 * (1 + 0.3 * np.cos(10 * np.pi * x))
    elif choice == 3:
        # Spectrum flatness init
        a = spectrum_flatness_init(n, rng)
    elif choice == 4:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(2 * x) ** 2
        a = np.maximum(a, 0)
    else:
        # Uniform with small perturbation
        a = np.ones(n) / n
        a += rng.random(n) * 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_equalization(a0, time_limit, rng=None):
    """Local search: move mass from peak contributors to reduce max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(12345)

    stuck = 0
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        max_k = int(np.argmax(b))

        # Find top contributors to the peak
        j_min = max(0, max_k - (n - 1))
        j_max = min(n - 1, max_k)
        idx = np.arange(j_min, j_max + 1)
        contrib = a[idx] * a[max_k - idx][::-1]
        src_local = int(np.argmax(contrib))
        src = idx[src_local]

        # Find a destination that doesn't contribute much to any top peak
        top_peaks = np.argsort(b)[-5:]
        peak_regions = set()
        for pk in top_peaks:
            jmin = max(0, pk - (n - 1))
            jmax = min(n - 1, pk)
            peak_regions.update(range(jmin, jmax + 1))

        # Prefer destinations outside peak regions
        outside = [j for j in range(n) if j not in peak_regions and j != src]
        if outside:
            dst = int(rng.choice(outside))
        else:
            dst = int(rng.choice(np.setdiff1d(range(n), [src])))

        # Try several fractions
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
            amt = a[src] * frac
            if amt < 1e-15:

            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_conv(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stuck = 0
                improved = True
                break

        if not improved:
            stuck += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_conv(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stuck = 0

        if stuck > 2000:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length (preserves normalization)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_length(a_init, n, time_budget, label=""):
    """Multi-phase optimization at a fixed length."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()

    def try_update(new_a, src):
        nonlocal best_a, best_score
        if new_a is None:
            return
        new_a = np.maximum(new_a, 0).astype(np.float64)
        ss = new_a.sum()
        if ss < 0.01:
            return
        new_a = new_a / ss
        sc = 2.0 * len(new_a) * fast_conv(new_a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = new_a.copy()
            elapsed = time.monotonic() - t0
            print(f"    [{src}] n={len(new_a)}: {sc:.8f} (t={elapsed:.0f}s)")

    # Phase A: Mirror descent (fast, many iterations)
    rem = time_budget - (time.monotonic() - t0)
    t_md = rem * 0.45
    if t_md > 5:
        a_md, _ = mirror_descent(best_a, t_md, n_active=30, verbose=True)
        try_update(a_md, "MD")

    # Phase B: LSE L-BFGS (precise, fewer iterations)
    rem = time_budget - (time.monotonic() - t0)
    t_lse = rem * 0.35
    if t_lse > 10:
        a_lse = lse_lbfgs(best_a, t_lse, T_start=80.0, T_end=0.05, n_stages=16)
        try_update(a_lse, "LSE")

    # Phase C: Peak equalization (local polish)
    rem = time_budget - (time.monotonic() - t0)
    t_pe = rem * 0.20
    if t_pe > 3:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
        a_pe, _ = peak_equalization(best_a, t_pe, rng=rng)
        try_update(a_pe, "PE")

    # Phase D: Another MD round from the best
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        a_md2, _ = mirror_descent(best_a, rem * 0.8, n_active=20, verbose=False)
        try_update(a_md2, "MD2")

    return best_a, best_score

    best_score = float('inf')

    def global_try_update(a, label):

            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** GLOBAL BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = diverse_init(1200, rng)
        best_seq = parent.copy()
        best_score = evaluate_safe(parent)
        print(f"Generated initial: n={len(parent)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # Determine candidate lengths to try
    lengths_to_try = []
    # Base length
    lengths_to_try.append(n_base)
    # Longer lengths
    for mult in [1.2, 1.5, 2.0, 2.5]:
        ln = int(n_base * mult)
        if ln <= 4000 and ln > n_base:
            lengths_to_try.append(ln)
    # Shorter lengths
    for mult in [0.75, 0.5]:
        sn = int(n_base * mult)
        if 100 <= sn < n_base:
            lengths_to_try.append(sn)

    # Sort by priority: start with base, then longer, then shorter
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - n_base))
    # Limit to at most 5 lengths
    lengths_to_try = lengths_to_try[:5]

    print(f"\nCandidate lengths: {lengths_to_try}")
    print(f"Starting score: {best_score:.8f}\n")

    # Allocate time: give more time to base length
    # Strategy: 50% to base, 20% to next best, 10% each to others, 10% final polish
    n_lengths = len(lengths_to_try)
    if n_lengths == 1:
        time_alloc = {lengths_to_try[0]: (time_limit - time.monotonic()) * 0.9}
    elif n_lengths == 2:
        rem = time_limit - time.monotonic()
        time_alloc = {lengths_to_try[0]: rem * 0.55, lengths_to_try[1]: rem * 0.35}
    elif n_lengths == 3:
        rem = time_limit - time.monotonic()
        time_alloc = {lengths_to_try[0]: rem * 0.45, lengths_to_try[1]: rem * 0.30,
                      lengths_to_try[2]: rem * 0.20}
    else:
        rem = time_limit - time.monotonic()
        fracs = np.array([0.40, 0.25, 0.18, 0.12, 0.05])[:n_lengths]
        fracs = fracs / fracs.sum()
        time_alloc = {lengths_to_try[i]: rem * fracs[i] for i in range(n_lengths)}

    # Optimize each length
    for i, ln in enumerate(lengths_to_try):
        rem = time_limit - time.monotonic()
        if rem < 30:
            print(f"  Not enough time remaining ({rem:.0f}s), stopping length search.")
            break

        t_alloc = min(time_alloc.get(ln, rem * 0.2), rem * 0.6)
        if t_alloc < 20:
            continue

        print(f"\n{'='*50}")
        print(f"Optimizing length n={ln} (budget={t_alloc:.0f}s)")
        print(f"{'='*50}")

        # Initialize: interpolate best_seq to this length
        if ln == n_base:
            init = best_seq.copy()
        else:
            init = interpolate_seq(best_seq, ln)
            sc_init = 2.0 * ln * fast_conv(init).max()
            print(f"  Interpolated init score: {sc_init:.8f}")

        a_out, sc_out = optimize_length(init, ln, t_alloc, label=f"n{ln}")
        global_try_update(a_out, f"opt_n{ln}")

    # Final polish on the global best
    print(f"\n{'='*50}")
    print(f"Final Polish (best={best_score:.8f}, n={len(best_seq)})")
    print(f"{'='*50}")
    rem = time_limit - time.monotonic()
    if rem > 20:
        # MD polish
        t1 = rem * 0.4
        a_polish, _ = mirror_descent(best_seq, t1, n_active=15, verbose=True)
        global_try_update(a_polish, "final_MD")

        rem = time_limit - time.monotonic()
        if rem > 15:
            # LSE polish with low temperatures
            t2 = rem * 0.4
            a_polish2 = lse_lbfgs(best_seq, t2, T_start=5.0, T_end=0.02, n_stages=10)
            global_try_update(a_polish2, "final_LSE")

        rem = time_limit - time.monotonic()
        if rem > 10:
            # Peak equalization
            t3 = rem * 0.5
            rng_pe = np.random.default_rng(99999)
            a_polish3, _ = peak_equalization(best_seq, t3, rng=rng_pe)
            global_try_update(a_polish3, "final_PE")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"{'='*50}")

    return [float(x) for x in best_seq]
