import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_br(a_fixed, solver=None):
    """Solve LP best response: min v s.t. conv(a,b) <= v, sum(b)=1, b>=0.
    Returns (b_star, v_star)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    v_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= v_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(v_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(v_var), constraints)
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(v_var.value)


def spectral_nudge(a, strength=0.3, beta=0.5):
    """Apply spectral flattening nudge to help escape local minima.
    Transforms to frequency domain, partially flattens |A|^2, transforms back,
    projects to non-negative, and blends with original."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    A = np.fft.fft(a_padded)
    mag = np.abs(A)
    phase = np.angle(A)
    # Flatten: new_mag = mag^beta (beta<1 compresses dynamic range)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    # Normalize mag first
    mag_norm = mag / mag.max()
    mag_flat = mag_norm ** beta
    # Rescale to preserve energy
    mag_flat = mag_flat * (mag.sum() / mag_flat.sum()) if mag_flat.sum() > 0 else mag
    A_new = mag_flat * np.exp(1j * phase)
    a_new_padded = np.fft.ifft(A_new).real
    a_new = a_new_padded[:n].copy()
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    # Blend with original
    result = (1 - strength) * a + strength * a_new
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return a.copy()
    return result / s


def compute_subgradient(a, n_top=5):
    """Compute subgradient using top-n active lags."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    # Use top n_top peaks
    sorted_idx = np.argsort(b)[::-1]
    active = sorted_idx[:n_top]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def nesterov_lp_v2(a0, time_limit, solver=None, n_max_iters=500000,

        y = np.maximum(y, 0)

        else:

            a = y.copy()

        if score_a < best_score - 1e-12:

            else:
                stall_count += 1
            prev_best = best_score

                        y = perturbed.copy()
                        best_score = score_p

                    else:

def gradient_descent_subgrad(a0, time_limit, n_top=3, lr_init=0.1, momentum=0.9):
    """Subgradient descent with momentum on the normalized simplex."""
    n = len(a0)
    z = z_from_a(np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)).copy()
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = best_a.sum()
    best_a = best_a / s if s > 1e-15 else np.ones(n) / n
    best_score = 2.0 * n * fast_autocorr(best_a).max()
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)  # momentum
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_from_z(z)
        g, max_b = compute_subgradient(a, n_top)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Mirror descent step
        v = momentum * v + (1 - momentum) * g
        z_new = z - lr * v
        # Normalization invariance: only differences matter
        z_new = z_new - z_new.mean()
        z = z_new
        a = softmax_from_z(z)

        if no_improve > 50:
            lr = max(lr * 0.9, 0.001)
            if no_improve > 200:
                # Random restart in z-space
                rng = np.random.default_rng(step)
                z = z + rng.standard_normal(n) * 0.1
                no_improve = 0
                lr = lr_init * 0.5

        if step % 1000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter={step} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=8):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    z = z - z.mean()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_autocorr(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        lse = m + np.log(e.sum())
        return (2.0 * n / T_val) * lse

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_autocorr(a)
        Tb = T_val * b
        m = Tb.max()
        if m > 700:
            m = 700
        e = np.exp(Tb - m)
        w = e / e.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_in):
            return lse_obj(z_in, stage_T)

        def jac(z_in):
            return lse_jac(z_in, stage_T)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 200, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': 10000, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_equalize_local(a0, time_limit, max_steps=50000):
    """Local search: move mass from peak contributors to non-peak positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stuck = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        peak_k = int(np.argmax(b))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src = idx[np.argmax(contrib)]

        # Find a good destination (low contribution to peak, low value)
        peak_set = set(range(js, je + 1))
        candidates = [j for j in range(n) if j not in peak_set]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        # Pick the one with lowest value
        dst = min(candidates, key=lambda j: a[j])

        # Try several transfer fractions
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stuck += 1
            if stuck > 5000:
                break
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stuck = 0

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def raised_cosine_init(n, center=0.5, width=0.9):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - center) / width) ** 2
    a[x < center - width] = 0
    a[x > center + width] = 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_local(a):
    """Locally evaluate (same logic as evaluate_sequence for normalized sequences)."""
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        n = len(a)
        b = fast_autocorr(a)
        return float(2.0 * n * b.max() / s**2)
    except Exception:
        return float('inf')


def evaluate_safe(a):
    """Safely evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_length(a_init, n, time_budget, solver, label=""):
    """Multi-phase optimization at fixed length."""
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{src}] n={n}: {sc:.6f} (t={elapsed:.0f}s)")

    # Phase A: Quick gradient descent
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.10, 60.0)
    if t_a > 10:
        a_md, _ = gradient_descent_subgrad(best_a, t_a, n_top=3)
        try_update(a_md, f"gd_{label}")

    # Phase B: Main Nesterov LP with spectral nudging
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.60, time_budget * 0.7)
    if t_b > 30:
        a_lp, _ = nesterov_lp_v2(best_a, t_b, solver=solver,
                                  nudge_interval=60, nudge_strength=0.2, nudge_beta=0.4)
        try_update(a_lp, f"lp_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.15, 80.0)
    if t_c > 10:
        a_lb = lbgfs_logsumexp(best_a, t_c, T_start=30.0, T_end=0.05, n_stages=8)
        try_update(a_lb, f"lse_{label}")

    # Phase D: Peak equalization
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.15, 60.0)
    if t_d > 5:
        a_pe, _ = peak_equalize_local(best_a, t_d, max_steps=50000)
        try_update(a_pe, f"pe_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1200
        best_seq = raised_cosine_init(n_base)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)})")

    # ===== Phase 1: Main optimization at base length =====
    print(f"\n=== Phase 1: n={n_base} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.50, 800.0)
    if t1 > 100:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_length(best_seq, n_base, t1, solver, label=f"n{n_base}")
        try_update_global(a_out, f"main_n{n_base}")

    # ===== Phase 2: Try longer lengths =====
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer = []
    for mult in [1.2, 1.5, 2.0]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 5000:
            longer.append(ln)
    longer = sorted(set(longer))[:3]

    for ln in longer:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.25, 300.0)
        if t_alloc < 60:
            break
        print(f"  n={ln}, budget={t_alloc:.0f}s")
        init_long = interpolate_seq(best_seq, ln)
        sc_init = evaluate_local(init_long)
        cur = init_long
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
        a_out, _ = optimize_length(cur, ln, t_alloc, solver, label=f"n{ln}")
        try_update_global(a_out, f"long_n{ln}")

        # Extra LP refinement if time
        rem2 = deadline - time.monotonic()
        if rem2 > 60 and a_out is not None:
            t2 = min(rem2 * 0.15, 120.0)
            if t2 > 20:
                a_ref, _ = nesterov_lp_v2(a_out, t2, solver=solver,
                                           nudge_interval=50, nudge_strength=0.15, nudge_beta=0.5)
                try_update_global(a_ref, f"lpref_n{ln}")

    # ===== Phase 3: Alternative initializations at n_base =====
    print(f"\n=== Phase 3: Alt inits (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        # Try a few different structured inits
        n_alt = n_base
        inits = [
            raised_cosine_init(n_alt, center=0.5, width=0.85),
            raised_cosine_init(n_alt, center=0.5, width=0.95),
        ]
        # Add a "bell" init
        x = np.linspace(-1, 1, n_alt)
        bell = np.exp(-3 * x ** 2)
        bell = np.maximum(bell, 0)
        bell = bell / bell.sum()
        inits.append(bell)

        for idx_alt, init_a in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.15, 120.0)
            if t_alloc < 30:
                break
            sc = evaluate_local(init_a)
            if sc < best_score:
                # This init is already good, optimize it
                a_out, _ = nesterov_lp_v2(init_a, t_alloc, solver=solver,
                                           nudge_interval=40, nudge_strength=0.3, nudge_beta=0.3)
                try_update_global(a_out, f"alt{idx_alt}")
            else:
                # Blend with best
                blend = 0.3 * best_seq + 0.7 * init_a
                blend = blend / blend.sum()
                a_out, _ = nesterov_lp_v2(blend, t_alloc, solver=solver,
                                           nudge_interval=40, nudge_strength=0.3, nudge_beta=0.3)
                try_update_global(a_out, f"blend{idx_alt}")

    # ===== Phase 4: Final polish =====
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # More aggressive LP
        t_lp = min(rem * 0.35, 100.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_v2(best_seq, t_lp, solver=solver,
                                       nudge_interval=40, nudge_strength=0.1, nudge_beta=0.6)
            try_update_global(a_fin, "final_lp")

        # LSE fine-tune
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.25, 60.0)
            if t_lse > 10:
                a_fin2 = lbgfs_logsumexp(best_seq, t_lse, T_start=10.0, T_end=0.03, n_stages=8)
                try_update_global(a_fin2, "final_lse")

        # Peak equalization
        rem = deadline - time.monotonic()
        if rem > 15:
            t_pe = min(rem * 0.25, 50.0)
            if t_pe > 5:
                a_fin3, _ = peak_equalize_local(best_seq, t_pe, max_steps=30000)
                try_update_global(a_fin3, "final_pe")

        # One more short LP
        rem = deadline - time.monotonic()
        if rem > 10:
            t_lp2 = min(rem * 0.3, 30.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_lp_v2(best_seq, t_lp2, solver=solver,
                                            nudge_interval=30, nudge_strength=0.05, nudge_beta=0.7)
                try_update_global(a_fin4, "final_lp2")

    # Final verification with actual evaluate_sequence
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            else:
                # Re-check with local
                best_score = evaluate_safe(best_seq)

    print(f"\nFINAL: score={best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
