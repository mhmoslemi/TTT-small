"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response alternating minimization with multiple sequence lengths
2. Peak-driven coordinate descent with optimized 1D steps
3. Pair-move local search
4. Fine-grained polishing with adaptive step sizes
5. Multi-start with diverse initializations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.concatenate([b_val, np.zeros(2*n-1 - len(b_val))]) if len(b_val) < 2*n-1 else b_val[:n]
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def coord_descent_peak(a0, time_limit, verbose=False):
    """
    Peak-driven coordinate descent.
    Focuses on coordinates that affect the current peaks.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    total_2n = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Find active peaks (top few)
        n_peaks = min(20, total_2n)
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_set = set(peak_idx.tolist())

        # Process coordinates that affect peaks
        # For peak at k, affected coords are i in [max(0,k-n+1), min(n-1,k)]
        affected_coords = set()
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                affected_coords.add(i)

        # Process affected coords in order of their contribution to peaks
        coord_list = list(affected_coords)
        # Sort by a[i] descending (larger coords have more impact)
        coord_list.sort(key=lambda i: -a[i])

        for i in coord_list:
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]

            # Try to find optimal delta
            # Since a_shifted >= 0, increasing delta increases all affected values
            # So we want delta <= 0 to reduce peaks
            val = a[i]

            best_M = M
            best_delta = 0.0

            # Try various reduction amounts
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                delta = -val * frac
                new_aff = conv_aff + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * val + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_M - 1e-14:
                    best_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=False):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Focus on coordinates affecting the current top peak
        k_star = int(np.argmax(conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)

        for i in range(i_start, i_end + 1):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]
            val = a[i]

            best_M = M
            best_delta = 0.0

            # Very fine steps
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.008, 0.015]:
                delta = -val * frac
                new_aff = conv_aff + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * val + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Also try increasing some small coordinates (to spread mass)
        # Find coordinates with small a[i] that are near the peak
        for i in range(i_start, i_end + 1):
            if a[i] > 0.5 * a[i_start:i_end+1].max():
                continue
            if a[i] > 0.01:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]
            val = a[i]

            best_M = M
            best_delta = 0.0

            # Try small increases
            base = max(val, M * 0.01 / max(n, 1))
            for frac in [0.1, 0.5, 1.0, 2.0, 5.0]:
                delta = base * frac
                new_aff = conv_aff + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * val + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_M < M - 1e-15:
                a[i] = a[i] + best_delta
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=False):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Damped update
        if it <= 5:
            alpha = 1.0
        elif it <= 20:
            alpha = 0.7
        else:
            alpha = max(0.3, 8.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if abs(score_a - best_score) / best_score > 0.01:
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise = np.random.standard_normal(n) * (0.01 / np.sqrt(n))
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.001:
                a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** 2
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * 0.3
        c2 = n * 0.7
        w = n * 0.15
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / 7)
    else:
        a = np.ones(n) / n
        noise = np.random.standard_normal(n) * (0.1 / np.sqrt(n))
        a = a + noise
        a = np.maximum(a, 0)

    return normalize(a)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)

    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS

def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def pair_move_search(a0, time_limit, verbose=False):

    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find the peak
        k_star = int(np.argmax(conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)

        # Find the largest contributor at the peak
        contributions = []
        for i in range(i_start, i_end + 1):
            j = k_star - i
            if j != i:
                contributions.append((2 * a[i] * a[j], i, j))
            else:
                contributions.append((a[i] * a[j], i, j))
        contributions.sort(reverse=True)

        # Try moving mass from the largest contributor to a "quiet" region
        for (contrib, i_src, j_src) in contributions[:5]:
            if contrib < 1e-10:
                continue

            # Find a target coordinate where adding mass would be "cheap"
            # (i.e., where the convolution values are low)
            # Try coordinates near the edges of the sequence
            target_candidates = []
            for j_dst in range(n):
                if j_dst == i_src or j_dst == j_src:
                    continue

                k_lo = j_dst
                k_hi = min(j_dst + n - 1, 2 * n - 2)
                local_max = float(conv[k_lo:k_hi + 1].max())

                    target_candidates.append((local_max, j_dst))

                # Try transferring small amount from max(a[i_src], a[j_src]) to j_dst

                    src = i_src
                else:
                    src = j_src

                    if 2 * src >= k1s and 2 * src <= k1e:
                        new_conv[2 * src] = conv[2 * src] + 2.0 * (-delta) * a[src] + delta * delta

                    # Add contribution of a[j_dst]
                    k2s = j_dst
                    k2e = min(j_dst + n - 1, 2 * n - 2)
                    new_conv[k2s:k2e + 1] += 2.0 * delta * a[k2s - j_dst:k2e - j_dst + 1]
                    if 2 * j_dst >= k2s and 2 * j_dst <= k2e:
                        # Need to use the already-modified value at 2*j_dst
                        base = new_conv[2 * j_dst] - 2.0 * delta * a[j_dst] - delta * delta
                        if 2 * j_dst >= k1s and 2 * j_dst <= k1e:
                            base = conv[2 * j_dst] + 2.0 * (-delta) * a[k1s - src:k1e - src + 1][2 * j_dst - k1s]
                            if 2 * src == 2 * j_dst:
                                base += delta * delta
                        base += 2.0 * delta * a[j_dst] + delta * delta
                        new_conv[2 * j_dst] = base

                    new_M = float(new_conv.max())
                    if new_M < M - 1e-13:
                        a = normalize(new_conv * 0 + np.maximum(
                            np.array(a, copy=True) + np.array(
                                [(-delta if i == src else delta if i == j_dst else 0) for i in range(n)]
                            ), 0))
                        # Recompute conv properly
                        conv = fast_convolve(a)
                        M = float(conv.max())
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and rounds % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairMove round={rounds} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** 2
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * 0.3
        c2 = n * 0.7
        w = n * 0.12
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / 7)
    elif init_type == 5:
        # Smoothed random
        t = np.arange(n)
        a = np.ones(n) / n
        for freq in [3, 5, 8, 13]:
            a += 0.05 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n
    return normalize(a)


def propose_candidate():
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP from parent ===
    if parent is not None:
        n_p = len(parent)
        print(f"\n=== Phase 1: LP from parent (n={n_p}) ===")
        t_alloc = min(120.0, (deadline - time.monotonic()) * 0.1)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, s_lp = lp_alternating(parent, t_alloc, solver, verbose=True)
        try_update(a_lp, "lp_parent")

        # Quick CD after LP
        if best_seq is not None and time.monotonic() < deadline - 200:
            t_alloc2 = min(40.0, (deadline - time.monotonic()) * 0.03)
            a_cd, s_cd = coord_descent_peak(best_seq, t_alloc2, verbose=False)
            try_update(a_cd, "cd_after_lp")

    # === Phase 2: Try n=2048 ===
    print(f"\n=== Phase 2: Try n=2048 (best={best_score:.8f}) ===")
    n_large = 2048
    t_start = time.monotonic()
    t_budget = min(250.0, (deadline - t_start) * 0.15)

    inits_large = [
        diverse_init(n_large, rng, 0),
        diverse_init(n_large, rng, 1),
        diverse_init(n_large, rng, 5),
    ]

    for idx, init in enumerate(inits_large):
        if time.monotonic() > t_start + t_budget:
            break
        t_alloc = min(t_start + t_budget - time.monotonic(), 80.0)
        if t_alloc < 20:
            break
        print(f"  LP n={n_large} init {idx+1}/{len(inits_large)}, budget={t_alloc:.0f}s")
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_n2048_{idx+1}")

        # CD on the best so far
        if best_seq is not None and len(best_seq) == n_large:
            t_alloc2 = min(20.0, (deadline - time.monotonic()) * 0.02)
            if t_alloc2 > 10:
                a_cd, s_cd = coord_descent_peak(best_seq, t_alloc2, verbose=False)
                try_update(a_cd, f"cd_n2048_{idx+1}")

    # === Phase 3: Intensive CD on best ===
    if best_seq is None:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback: n={n}, score={best_score:.8f}")

    print(f"\n=== Phase 3: Intensive CD (best={best_score:.8f}, n={len(best_seq)}) ===")
    t_start = time.monotonic()
    t_budget = min(300.0, (deadline - t_start) * 0.2)

    # Multiple rounds of CD with increasing refinement
    n_rounds = 3
    for r in range(n_rounds):
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.3, t_budget / (n_rounds - r))
        if t_alloc < 15:
            break
        print(f"  CD round {r+1}/{n_rounds}, budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent_peak(best_seq, t_alloc, verbose=(r == 0))
        try_update(a_cd, f"cd_r{r+1}")

    # === Phase 4: LP refinement ===
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 4: LP refinement (best={best_score:.8f}) ===")
        t_alloc = min(150.0, (deadline - time.monotonic()) * 0.15)
        if t_alloc > 30:
            a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
            try_update(a_lp, "lp_refine")

    # === Phase 5: Pair moves ===
    if time.monotonic() < deadline - 80:
        print(f"\n=== Phase 5: Pair moves (best={best_score:.8f}) ===")
        t_alloc = min(80.0, (deadline - time.monotonic()) * 0.1)
        if t_alloc > 15 and len(best_seq) <= 2048:
            a_pm, s_pm = pair_move_search(best_seq, t_alloc, verbose=False)
            try_update(a_pm, "pair_move")

    # === Phase 6: Fine polishing ===
    print(f"\n=== Phase 6: Fine polishing (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 20:
        t_alloc = min(80.0, deadline - time.monotonic() - 20)
        if t_alloc > 15:
            a_fp, s_fp = fine_polish(best_seq, t_alloc, verbose=True)
            try_update(a_fp, "fine_polish")

    # === Phase 7: Final squeeze ===
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final squeeze (best={best_score:.8f}) ===")
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            # Alternate between CD and fine polish
            half = remaining / 2
            a_cd, s_cd = coord_descent_peak(best_seq, half * 0.8, verbose=False)
            try_update(a_cd, "final_cd")
            if time.monotonic() < deadline - 10:
                a_fp, s_fp = fine_polish(best_seq, min(30.0, deadline - time.monotonic() - 10), verbose=False)
                try_update(a_fp, "final_fine")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        return [float(x) for x in best_seq]

    return [1.0]
