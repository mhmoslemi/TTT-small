    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def spectral_init(n, rng, flatness=0.5):
    """Initialize with a flat-ish power spectrum, then project to non-negative."""
    fft_size = 1 << (2 * n - 2).bit_length()
    n_freq = fft_size // 2 + 1
    # Start with flat magnitude, random phase
    mag = np.ones(n_freq)
    mag[0] = np.sqrt(n)  # DC component
    # Add some structure
    for m in range(1, n_freq):
        mag[m] = 1.0 + flatness * rng.uniform(-1, 1)
    phase = rng.uniform(0, 2 * np.pi, n_freq)
    phase[0] = 0
    A = mag * np.exp(1j * phase)
    a = irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multiplicative_weights(a0, time_limit, verbose=False, label=""):
    """
    Multiplicative weights update to minimize max conv.
    a[i] *= exp(-eta * grad[i]) where grad[i] = sum over active k of a[k-i]
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.1
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Find active lags (those near the max)
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # d/d a[j] = sum over i where k-i = j or i = j of a[k-i] + a[i]
            # = a[k-j] + a[j] if both are valid, or just one term
            # Actually d(a*a)[k]/da[j] = a[k-j] + a[j] (if k != 2j, else 2*a[j])
            # Wait: (a*a)[k] = sum_i a[i]*a[k-i]
            # d/d a[j]: terms where i=j gives a[k-j], terms where k-i=j (i=k-j) gives a[j]
            # If j != k-j: derivative = a[k-j] + a[j]
            # If j == k-j (i.e., k=2j): derivative = 2*a[j]
            
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            # grad[j] += a[k-j] for j in valid range
            # This is: grad[0:n] += reverse of a[max(0,k-n+1):min(n,k+1)]
            # Plus grad[j] += a[j] (shifted)
            
            # More simply: grad += a reversed and shifted appropriately
            # (a*a)[k] = <a, rev(a) shifted by k>
            # d/d a[j] (a*a)[k] = a[k-j] (from i=j term) + a[j] (from k-i=j term if k-j is valid)
            
            j_lo = max(0, int(k) - n + 1)
            j_hi = min(n - 1, int(k))
            
            # First part: a[k-j] for j in [j_lo, j_hi]
            # a[k-j] where j goes from j_lo to j_hi means a[k-j_hi] to a[k-j_lo]
            idx1 = int(k) - np.arange(j_hi, j_lo - 1, -1)
            grad[j_lo:j_hi+1] += a[idx1]
            
            # Second part: a[j] for j in [j_lo, j_hi] (from the k-i=j term)
            # But we need to avoid double-counting when j = k-j, i.e., k = 2j
            # Actually, the derivative formula handles this automatically:
            # d/d a[j] sum_i a[i]*a[k-i] = a[k-j] + a[j] ( counting both terms)
            # When k = 2j, both terms give a[j], so derivative = 2*a[j]
            # This is already captured by adding both parts
            
            grad[j_lo:j_hi+1] += a[j_lo:j_hi+1]
        
        # Normalize gradient
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad = grad / grad_max
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-15:
            a_new = a
        else:
            a_new = a_new / cs
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Try smaller step
            a_half = a * np.exp(-eta * 0.5 * grad)
            a_half = np.maximum(a_half, 0)
            cs = a_half.sum()
            if cs > 1e-15:
                a_half = a_half / cs
                sc_half = compute_score(a_half)
                if sc_half < compute_score(a) - 1e-12:
                    a = a_half
                    if sc_half < best_score:
                        best_score = sc_half
                        best_a = a_half.copy()
                        no_improve = 0
                
        eta *= 0.995
        if no_improve > 200:
            eta *= 0.5
            no_improve = 0
        if no_improve > 1000:
            break
        
        if verbose and iters % 50 == 0:
            print(f"    MW {label} it={iters}: best={best_score:.8f}, eta={eta:.5f}")
    
    return best_a, best_score

def pair_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Pair coordinate descent: optimize pairs of coordinates simultaneously.
    For each pair (i, j), find (a[i], a[j]) that minimizes the score.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Process coordinates in random order, grouping into pairs
        order = rng.permutation(n)
        
        for idx in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            j = order[idx + 1]
            
            vi = a[i]
            vj = a[j]
            
            if vi < 1e-14 and vj < 1e-14:
                continue
            
            # Affected lags for coordinate i: [i, i+n-1]
            # Affected lags for coordinate j: [j, j+n-1]
            k_lo = min(i, j)
            k_hi = max(i + n - 1, j + n - 1)
            k_hi = min(k_hi, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            # For the affected region, compute how conv changes with (x, y) = (a[i], a[j])
            # conv[k] changes by:
            #   2*x*a[k-i] (for i-part) + 2*y*a[k-j] (for j-part)
            #   + x^2 if 2i = k (self-term for i)
            #   + y^2 if 2j = k (self-term for j)
            #   + 2*x*y if k-i = j and k-j = i, i.e., k = i+j (cross term)
            
            ki_lo = i
            ki_hi = min(i + n - 1, 2 * n - 2)
            kj_lo = j
            kj_hi = min(j + n - 1, 2 * n - 2)
            
            # For efficiency, use a grid search over (x, y)
            # x in [0, 3*vi], y in [0, 3*vj]
            # Try a few candidate values
            
            best_x, best_y = vi, vj
            best_local = 2.0 * n * max(unaff_max, float(conv[k_lo:k_hi+1].max())) / (current_sum * current_sum)
            
            # Candidate values for x
            x_candidates = [0.0, vi * 0.5, vi * 0.8, vi, vi * 1.2, vi * 1.5, vi * 2.0]
            x_candidates = [x for x in x_candidates if 0 <= x <= 3 * max(vi, 1.0/n)]
            
            # Candidate values for y
            y_candidates = [0.0, vj * 0.5, vj * 0.8, vj, vj * 1.2, vj * 1.5, vj * 2.0]
            y_candidates = [y for y in y_candidates if 0 <= y <= 3 * max(vj, 1.0/n)]
            
            for x in x_candidates:
                for y in y_candidates:
                    if abs(x - vi) < 1e-14 and abs(y - vj) < 1e-14:
                        continue
                    # Compute new conv in affected region
                    delta_i = x - vi
                    delta_j = y - vj
                    
                    # conv[k] += 2*delta_i*a[k-i] + delta_i*(x+vi) if 2i=k
                    #          + 2*delta_j*a[k-j] + delta_j*(y+vj) if 2j=k
                    #          + 2*delta_i*delta_j if k=i+j (cross term, approximate)
                    
                    new_conv = conv[k_lo:k_hi+1].copy()
                    
                    # i-part
                    for k in range(ki_lo, ki_hi + 1):
                        if k < k_lo or k > k_hi:
                            continue
                        idx_in_region = k - k_lo
                        partner_idx = k - i
                        if 0 <= partner_idx < n and partner_idx != i:
                            new_conv[idx_in_region] += 2.0 * delta_i * a[partner_idx]
                        if k == 2 * i:
                            new_conv[idx_in_region] += delta_i * (x + vi)  # = delta_i * (2*vi + delta_i)
                    
                    # j-part
                    for k in range(kj_lo, kj_hi + 1):
                        if k < k_lo or k > k_hi:
                            continue
                        idx_in_region = k - k_lo
                        partner_idx = k - j
                        if 0 <= partner_idx < n and partner_idx != j:
                            new_conv[idx_in_region] += 2.0 * delta_j * a[partner_idx]
                        if k == 2 * j:
                            new_conv[idx_in_region] += delta_j * (y + vj)
                    
                    # Cross term: if k = i + j, then a[i]*a[j] appears in conv[i+j]
                    # When we change a[i] by delta_i and a[j] by delta_j:
                    # conv[i+j] gets an additional delta_i * a[j] + a[i] * delta_j + delta_i * delta_j
                    # But wait, this is already captured above if i != j
                    # (the i-part gives 2*delta_i*a[k-i] which includes a[j] when k-i=j, i.e., k=i+j)
                    # (the j-part gives 2*delta_j*a[k-j] which includes a[i] when k-j=i, i.e., k=i+j)
                    # So we'd be double-counting. Let me reconsider.
                    
                    # Actually, conv[k] = sum_p a[p]*a[k-p]
                    # d conv[k] / da[i] = a[k-i] (from p=i) + a[i] (from k-p=i, p=k-i)
                    #   but if k-i = i (i.e., k=2i), then it's 2*a[i]
                    # d conv[k] / da[j] = a[k-j] (from p=j) + a[j] (from k-p=j, p=k-j)
                    #   but if k-j = j (i.e., k=2j), then it's 2*a[j]
                    
                    # For the change (vi,vj) -> (x,y):
                    # conv[k] new = conv[k] old - vi*a[k-i] - a[vi]*... 
                    # This is getting complex. Let me just recompute directly for the affected lags.
                    
                    # Simpler approach: just recompute conv[k] for affected k using the new values
                    # Since we only change 2 coordinates, this is O(n) per pair
                    pass  # Skip this complex approach for now
                    
                    # Use the linear approximation instead (faster)
                    # conv[k] ~= conv[k] + 2*delta_i*a[k-i] + 2*delta_j*a[k-j] (first order)
                    # Plus second order corrections for self terms
                    
                    pass
        
        # This pair approach is getting too complex and slow. Let me use a simpler version.
        # Actually, let me just use single-coordinate descent but with better step sizes.
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_a, best_score

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact single-coordinate descent with golden section search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            
            if has_quad:
                quad_idx = 2 * i - k_lo
            
            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum, _n=n):
                ns = _cs + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)
            
            upper = max(3.0 * v, 5.0 / n)
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = _score(c)
            fd = _score(d)
            for _ in range(35):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _score(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _score(d)
            best_x = (lo + hi) * 0.5
            
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")
    
    return best_a, best_score, sweeps

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val
