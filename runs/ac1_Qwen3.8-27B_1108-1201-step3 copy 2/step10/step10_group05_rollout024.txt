def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def get_lp_solver():

            prob.solve(solver=solver_name)

        except Exception:

    return cp.ECOS

def build_conv_matrix(a):

def peak_shaving_convolution(a0, time_limit, label=""):
    """Directly shave peaks in the convolution domain by adjusting source coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Find current peaks
        peak_threshold = M * 0.85
        peak_idx = np.where(conv >= peak_threshold)[0]

        if len(peak_idx) == 0:
            break

        # Compute "pressure" on each coordinate from peaks
        pressure = np.zeros(n)
        for k in peak_idx:
            w = (conv[k] - M * 0.85) / max(M * 0.15, 1e-15)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[i] * a[j]

        p_max = pressure.max()
        if p_max < 1e-15:
            break

        # Reduce high-pressure coordinates, boost low-pressure
        target_reduction = 0.1
        a_new = a.copy()
        high_pressure = pressure > 0.5 * p_max
        low_pressure = pressure < 0.2 * p_max

        total_reduce = 0.0
        for i in np.where(high_pressure)[0]:
            reduction = min(a[i] * target_reduction, a[i] * 0.5)
            a_new[i] -= reduction
            total_reduce += reduction

        # Distribute to low-pressure coordinates
        if total_reduce > 1e-15 and low_pressure.any():
            lp_indices = np.where(low_pressure)[0]
            # Distribute proportionally to (1 - pressure[i])
            weights = 1.0 - pressure[lp_indices] / max(p_max, 1e-15)
            weights = np.maximum(weights, 0.1)
            weights = weights / weights.sum()
            a_new[lp_indices] += total_reduce * weights

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        new_conv = fast_convolve(a_new)
        new_M = float(new_conv.max())

        if new_M < M - 1e-14:
            a = a_new
            conv = new_conv
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
        else:
            # Revert and try smaller step
            a_new2 = 0.9 * a + 0.1 * a_new
            a_new2 = np.maximum(a_new2, 0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 / s2
            new_M2 = fast_convolve(a_new2).max()
            if new_M2 < M - 1e-14:
                a = a_new2
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
            else:
                break

    return best_a, best_score

def smoothed_gradient_descent(a0, time_limit, label="", beta_init=50.0, lr_init=0.05):

            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 50 == 0:
            beta *= 1.005

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}")

    return best_a, best_score
