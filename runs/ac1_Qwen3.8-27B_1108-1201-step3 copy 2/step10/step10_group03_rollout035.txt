def fast_convolve(a):

"""

L-BFGS with FFT-accelerated gradients, top-m peak robustness, multi-n exploration,
Fourier basis optimization, structured random restarts, and peak redistribution.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_grad_lse(a, w):
    """Compute gradient of LogSumExp-smoothed objective using FFT.
    grad[j] = 2 * sum_{d=0}^{n-1} a[d] * w[j+d]
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, N)
    W = rfft(w, N)
    conv = irfft(A * W, N)
    idx = np.arange(n) + n - 1
    return 2.0 * conv[idx]


def lse_objective_and_grad(z, T, n):
    """Compute LSE-smoothed objective and gradient in log-parametrization."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    lse = b_max + np.log(np.exp(b_shift - b_max).sum())
    val = (2.0 * n) * lse
    # Gradient w.r.t. a
    grad_a = compute_grad_lse(a, w)
    # Chain rule for softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def adaptive_lbgfs(a0, time_limit, T_init=50.0, T_min=0.003):
    """L-BFGS with adaptive temperature decreasing based on progress."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    T = T_init
    no_improve_consecutive = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj(z_in):
            val, _ = lse_objective_and_grad(z_in, T, n)
            return val

        def grad(z_in):
            _, g = lse_objective_and_grad(z_in, T, n)
            return g

        max_iter = max(30, int(min(remaining * 5, 500)))
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-13,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
                no_improve_consecutive = 0
                # Reduce temperature
                T = max(T * 0.65, T_min)
            else:
                no_improve_consecutive += 1
                if no_improve_consecutive >= 3:
                    T = max(T * 0.4, T_min)
                    no_improve_consecutive = 0
                    # Small random perturbation to escape
                    z = z + 0.01 * np.random.default_rng(int(time.monotonic()*1000)).standard_normal(n)
        except Exception:
            T = max(T * 0.5, T_min)

    # Final evaluation
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc = 2.0 * n * fast_convolve(a_final).max()
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_a, best_score


def top_m_lbgfs(a0, time_limit, m=5, T_init=20.0):
    """L-BFGS using top-m peaks for more robust optimization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    T = T_init

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj(z_in):
            z_max = z_in.max()
            exp_z = np.exp(z_in - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z_in):
            z_max = z_in.max()
            exp_z = np.exp(z_in - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Focus weights on top-m peaks
            top_idx = np.argsort(b)[-m:]
            w_top = np.zeros_like(w)
            w_top[top_idx] = w[top_idx]
            w_top_sum = w_top.sum()
            if w_top_sum > 1e-30:
                w_top = w_top / w_top_sum
                w = w_top
            grad_a = compute_grad_lse(a_cur, w)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        max_iter = max(20, int(min(remaining * 5, 300)))
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-13,
                         'maxfun': max_iter * 80, 'disp': False}
            )
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
                T = max(T * 0.7, 0.01)
            else:
                T = max(T * 0.5, 0.01)
        except Exception:
            T = max(T * 0.5, 0.01)

    return best_a, best_score


def fourier_basis_optimize(a0, time_limit, n_modes=30):
    """Optimize using a low-dimensional Fourier basis parametrization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Base shape: raised cosine
    x = np.linspace(0, 1, n)
    base = 0.5 * (1 + np.cos(np.pi * x))
    base = np.maximum(base, 0.01)
    base = base / base.sum()

    # Fourier basis: a[j] = base[j] * (1 + sum_k c_k * cos(2*pi*k*j/n) + d_k * sin(2*pi*k*j/n))
    # Parameters: c_1..c_K, d_1..d_K (2K parameters)
    K = n_modes
    n_params = 2 * K

    def params_to_a(params):
        c = params[:K]
        d = params[K:]
        j = np.arange(n)
        a_candidate = base.copy()
        for k in range(1, K + 1):
            a_candidate = a_candidate * (1.0 + c[k-1] * np.cos(2 * np.pi * k * j / n) +
                                          d[k-1] * np.sin(2 * np.pi * k * j / n))
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-15:
            return base.copy()
        return a_candidate / s

    def obj(params):
        a_cand = params_to_a(params)
        return local_eval(a_cand)

    # Start with zero parameters (i.e., base shape)
    x0 = np.zeros(n_params)

    # Try Nelder-Mead first (derivative-free, good for low-dim)
    if n_params <= 60:
        try:
            res = minimize(
                obj, x0, method='Nelder-Mead',
                options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14,
                         'adaptive': True}
            )
            a_try = params_to_a(res.x)
            sc = local_eval(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
        except Exception:
            pass

    # Try L-BFGS with numerical gradient
    if time.monotonic() - t0 < time_limit - 5:
        try:
            res2 = minimize(
                obj, res.x if 'res' in dir() else x0, method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': 50000}
            )
            a_try = params_to_a(res2.x)
            sc = local_eval(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_redistribute(a0, time_limit, n_steps=20000):
    """Direct peak redistribution: move mass from peak-contributing positions
    to less-contributing positions using a multiplicative update."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find active peaks (within 1% of max)
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute gradient (sum over active peaks)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        # Multiplicative update (preserves non-negativity)
        a_trial = a * np.exp(-lr * grad / max(grad.max(), 1e-15))
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
                # Occasional random perturbation
                if rng.random() < 0.1:
                    a = a + 0.005 * rng.standard_normal(n)
                    a = normalize(a)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a = np.exp(-3 * (x - 0.5) ** 2)
    else:
        a = 0.3 + 0.5 * np.exp(-5 * (x - 0.5) ** 2) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive-temperature L-BFGS, top-m peaks, Fourier basis optimization,
    peak redistribution, multi-n exploration, and structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Main adaptive L-BFGS on parent ===
    print(f"\n=== Phase 1: Adaptive L-BFGS (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = adaptive_lbgfs(best_seq, t1, T_init=50.0, T_min=0.003)
        try_update(a_lp, "adaptive_lbgfs")

    # === Phase 2: Peak redistribution ===
    print(f"\n=== Phase 2: Peak Redistribution (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 120.0)
    if t2 > 15:
        a_pr, sc_pr = peak_redistribute(best_seq, t2, n_steps=30000)
        try_update(a_pr, "peak_redist")

    # === Phase 3: Top-m L-BFGS ===
    print(f"\n=== Phase 3: Top-m L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 15:
        for m in [3, 5, 10]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t3_sub = min(rem * 0.3, 40.0)
            a_tm, sc_tm = top_m_lbgfs(best_seq, t3_sub, m=m, T_init=10.0)
            try_update(a_tm, f"topm_{m}")

    # === Phase 4: Fourier basis optimization ===
    print(f"\n=== Phase 4: Fourier Basis (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        for K in [15, 25, 40]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t4_sub = min(rem * 0.3, 60.0)
            a_fb, sc_fb = fourier_basis_optimize(best_seq, t4_sub, n_modes=K)
            try_update(a_fb, f"fourier_K{K}")

    # === Phase 5: Second adaptive L-BFGS pass (lower T) ===
    print(f"\n=== Phase 5: L-BFGS Pass 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 300.0)
    if t5 > 30:
        a_lp2, sc_lp2 = adaptive_lbgfs(best_seq, t5, T_init=5.0, T_min=0.002)
        try_update(a_lp2, "lbgfs_pass2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = len(best_seq)
        for cand_n in [400, 500, 600, 800, 1000, 1200, 1500, 1800, 2000, 2500, 3000]:
            if cand_n not in n_candidates and cand_n != base_n:
                n_candidates.append(cand_n)
        # Sort by distance from current best n
        n_candidates.sort(key=lambda x: abs(x - base_n))

        for cand_n in n_candidates[:6]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 25:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick L-BFGS
            a_cand, _ = adaptive_lbgfs(init_cand, t_sub * 0.6, T_init=20.0, T_min=0.01)
            try_update(a_cand, f"lbgfs_n{cand_n}")
            # Peak redistribution
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cand2, _ = peak_redistribute(a_cand, min(rem * 0.3, 30.0), n_steps=10000)
                try_update(a_cand2, f"redist_n{cand_n}")

    # === Phase 7: Structured random restarts ===
    print(f"\n=== Phase 7: Random Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(5, int(rem // 30))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_sub = min(rem * 0.3, 60.0)

            # Different perturbation strategies
            strategy = r % 4
            if strategy == 0:
                # Gaussian noise
                a_restart = best_seq + 0.05 * rng.standard_normal(len(best_seq))
                a_restart = normalize(a_restart)
            elif strategy == 1:
                # Smooth noise (low-frequency perturbation)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                x = np.linspace(0, 1, len(best_seq))
                perturbation = 0.1 * np.cos(2 * np.pi * freq * x + phase)
                a_restart = best_seq * (1 + perturbation)
                a_restart = normalize(a_restart)
            elif strategy == 2:
                # Interpolate to different n and back
                new_n = len(best_seq) + rng.integers(-200, 200)
                new_n = max(200, new_n)
                a_interp = interpolate_sequence(best_seq, new_n)
                a_restart = interpolate_sequence(a_interp, len(best_seq))
            else:
                # Blend with a structured shape
                a_struct = structured_init(len(best_seq), rng)
                alpha = rng.uniform(0.2, 0.5)
                a_restart = (1 - alpha) * best_seq + alpha * a_struct
                a_restart = normalize(a_restart)

            a_cand, _ = adaptive_lbgfs(a_restart, t_sub, T_init=10.0, T_min=0.01)
            try_update(a_cand, f"restart_{r}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low temperature L-BFGS
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            a_fin, _ = adaptive_lbgfs(best_seq, t_fin, T_init=1.0, T_min=0.001)
            try_update(a_fin, "final_lbgfs")

        # Quick peak redistribution
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin2, _ = peak_redistribute(best_seq, min(rem * 0.3, 20.0), n_steps=5000)
            try_update(a_fin2, "final_redist")

        # Very top-m focused
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = top_m_lbgfs(best_seq, min(rem * 0.3, 20.0), m=2, T_init=5.0)
            try_update(a_fin3, "final_topm2")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
