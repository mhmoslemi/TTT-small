"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast gradient-based optimization
with top-k soft-max smoothing, diverse Fourier-based initialization, and 
multi-scale refinement. No LP solves - pure gradient descent with adaptive 
step sizes and active-set tracking for speed.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_eval(a):
    """Fast local evaluation of the objective (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def topk_soft_max(b, k, T):
    """Compute top-k soft-max approximation to max."""
    if k >= len(b):
        return np.max(b)
    # Get top-k values
    idx = np.argpartition(b, -k)[-k:]
    top_vals = b[idx]
    T_shift = top_vals / T
    T_max = T_shift.max()
    exp_ws = np.exp(T_shift - T_max)
    w = exp_ws / exp_ws.sum()
    return T_max + np.log(exp_ws.sum())

def gradient_max_conv(a, b, top_k=None, T=1.0):
    """Compute gradient of max(conv(a,a)) w.r.t. a.
    Uses weighted average of active peaks."""
    n = len(a)
    size = 2 * n - 1
    
    if top_k is None or top_k >= size:
        # Use full softmax
        T_shift = b / T
        T_max = T_shift.max()
        exp_ws = np.exp(T_shift - T_max)
        w = exp_ws / exp_ws.sum()
    else:
        # Top-k softmax
        idx = np.argpartition(b, -top_k)[-top_k:]
        w = np.zeros(size)
        top_vals = b[idx] / T
        top_max = top_vals.max()
        exp_ws = np.exp(top_vals - top_max)
        w_vals = exp_ws / exp_ws.sum()
        w[idx] = w_vals
    
    # Gradient: d/db[a] * conv(a,a)[k] / da[j] = 2*a[k-j] (for valid k-j)
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # This is 2 * conv(w, a)[j] (with appropriate truncation)
    
    # Use FFT: grad_a = 2 * irfft(rfft(w, fft_size) * rfft(a, fft_size))[:n]
    # But w has length 2n-1 and a has length n, so we need to be careful
    
    # Actually, grad_a[j] = 2 * sum_k w[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
    # = 2 * sum_{k-j=0}^{n-1} w[k] * a[k-j]
    # = 2 * (w * a)[j] in the correlation sense
    
    # Let's compute it directly using FFT
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We need: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # This is the cross-correlation of w and a
    # = sum_k w[k] * a[k-j] = cross-correlation
    # In frequency: IFFT(conj(W) * A) gives cross-correlation
    # Actually: cross_correlation(w,a)[j] = sum_k w[k] * a[k-j]
    # = IFFT(conj(FFT(w)) * FFT(a))[j]
    
    grad_a = 2.0 * np.real(irfft(np.conj(W) * A, fft_size))[:n]
    
    return grad_a

def subgradient_optimize(a0, time_limit, top_k=20, T_start=5.0, T_end=0.01):
    """Subgradient/gradient method with adaptive step size and annealing T."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adaptive parameters
    lr = 0.01
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal T
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_convolve(a)
        M = b.max()
        
        # Compute gradient
        grad_a = gradient_max_conv(a, b, top_k=top_k, T=T)
        
        # Projected gradient step (stay in simplex)
        a_trial = a - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        total_iters += 1
        
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.8
                no_improve = 0
            lr = max(1e-8, lr)
        
        # Increase top_k as we converge
        if progress > 0.5:
            top_k = max(5, top_k // 2)
        
        if total_iters % 5000 == 0:
            print(f"    Subgrad iter={total_iters} ({elapsed:.0f}s): score={best_score:.10f}, lr={lr:.6f}, T={T:.4f}")
    
    return best_a, best_score

def lsm_optimize(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        max_iter = max(50, int(remaining * 10))
        
        def obj_func(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return 2.0 * n * lse
        
        def grad_func(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # grad_a = 2 * cross_correlation(w, a)
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A_cur = rfft(a_cur, fft_size)
            grad_a = 2.0 * np.real(irfft(np.conj(W) * A_cur, fft_size))[:n]
            # Chain rule for log-softmax: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': max_iter * 50, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def fourier_init(n, rng, method='random'):
    """Generate initial sequence in Fourier domain."""
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if method == 'random':
        # Random Fourier coefficients
        A = np.zeros(fft_size // 2 + 1, dtype=complex)
        A[0] = 1.0  # DC = 1 (sum = 1)
        # Random complex amplitudes with decay
        freqs = np.arange(1, fft_size // 2 + 1)
        decay = np.exp(-0.5 * freqs / n)
        A[1:] = (rng.standard_normal(fft_size // 2) + 1j * rng.standard_normal(fft_size // 2)) * decay * 0.1
        a = np.real(irfft(A, fft_size))[:n]
    elif method == 'lowpass':
        A = np.zeros(fft_size // 2 + 1, dtype=complex)
        A[0] = 1.0
        cutoff = rng.integers(3, n // 2)
        for i in range(1, cutoff + 1):
            A[i] = (rng.standard_normal() + 1j * rng.standard_normal()) * 0.05
        a = np.real(irfft(A, fft_size))[:n]
    elif method == 'cosine':
        x = np.linspace(0, 1, n)
        k = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * math.pi)
        a = 1.0 + 0.3 * np.cos(2 * math.pi * k * x + phase)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def active_peak_optimize(a0, time_limit):
    """Optimize by focusing on the active (peak) constraints.
    At each step, identify the top peaks and solve a small

    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        size = 2 * n - 1
        
        # Find active peaks (within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # For diversity, sometimes use a slightly wider set
        if len(active) < 5 and rng.random() < 0.3:
            threshold = M * 0.99
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
        
        # Compute gradient as average over active peaks
        fft_size = 1 << (size - 1).bit_length()
        
        # Weight: uniform over active peaks (or use soft-max)
        if len(active) <= 3:
            w = np.zeros(size)
            w[active] = 1.0 / len(active)
        else:
            # Soft-max weighting
            active_vals = b[active]
            T_w = 0.01 * M
            exp_w = np.exp((active_vals - active_vals.max()) / T_w)
            w = np.zeros(size)
            w[active] = exp_w / exp_w.sum()
        
        # Gradient
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * np.real(irfft(np.conj(W) * A, fft_size))[:n]
        
        # Step
        a_trial = a - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.03)
        else:
            no_improve += 1
            if no_improve >= 100:
                # Add perturbation
                noise = rng.standard_normal(n) * 0.01
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc_trial = 2.0 * n * fast_convolve(a_trial).max()
                if sc_trial < best_score:
                    a = a_trial
                lr *= 0.7
                no_improve = 0
            lr = max(1e-9, lr)
        
        if (int((time.monotonic() - t0) * 10) % 50) == 0:
            pass  # rate-limited print
    
    return best_a, best_score

def gausssian_process_style(a0, time_limit, n_restarts=5):
    """Try multiple perturbations and keep the best."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 30:
            break
        remaining = (time_limit - (time.monotonic() - t0)) / (n_restarts - restart)
        remaining = min(remaining, 120.0)
        
        # Perturb the current best
        scale = 0.05 / (restart + 1)
        a_pert = best_a + rng.standard_normal(n) * scale
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        
        # Quick optimization
        a_opt, sc_opt = subgradient_optimize(a_pert, remaining, top_k=10, T_start=3.0, T_end=0.01)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"    GPR restart {restart+1}: new best {best_score:.10f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast gradient-based optimization
with momentum, top-k soft-max peak tracking, diverse Fourier-based initialization,
and multi-scale refinement. Key improvements: (1) no LP solves - pure FFT-based
gradient methods for speed, (2) Nesterov-style momentum for faster convergence,
(3) adaptive top-k soft-max for smooth peak tracking, (4) diverse initialization
from low-autocorrelation Fourier patterns, (5) multi-n exploration with
interpolation-based transfer.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient(a, b, top_k, T):
    """Compute gradient of soft-max of top-k peaks of conv(a,a) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1

    if top_k >= size:
        T_shift = b / T
        T_max = T_shift.max()
        exp_ws = np.exp(T_shift - T_max)
        w = exp_ws / exp_ws.sum()
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        w = np.zeros(size)
        top_vals = b[idx] / T
        top_max = top_vals.max()
        exp_ws = np.exp(top_vals - top_max)
        w_vals = exp_ws / exp_ws.sum()
        w[idx] = w_vals

    # grad_a[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # = 2 * cross_correlation(w, a)[m]
    # Use FFT: IFFT(conj(W) * A)
    fft_size = 1 << (3 * n - 3).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * np.real(irfft(np.conj(W) * A, fft_size))[:n]

    return grad_a


def momentum_subgrad(a0, time_limit, top_k=20, T_start=10.0, T_end=0.01,
                     beta=0.9, lr_init=0.01):
    """Momentum-based subgradient with annealed soft-max."""
    n = len(a0)
    a = normalize(a0.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        T = T_start * (T_end / T_start) ** progress

        b = fast_convolve(a)
        M = b.max()

        grad_a = compute_gradient(a, b, top_k=top_k, T=T)

        # Momentum update
        v = beta * v + (1.0 - beta) * grad_a
        a_trial = a - lr * v
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        total_iters += 1

        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.01, 0.04)
        else:
            no_improve += 1
            if no_improve >= 100:
                lr *= 0.7
                no_improve = 0
            lr = max(1e-8, lr)

        # Adapt beta: increase momentum as we converge
        beta_eff = min(0.98, beta + 0.05 * progress)
        # Re-compute v with new beta
        v = beta_eff * v

        if total_iters % 20000 == 0:
            print(f"    MomSubgrad iter={total_iters} ({elapsed:.0f}s): "
                  f"score={best_score:.10f}, lr={lr:.6f}, T={T:.4f}")

    return best_a, best_score

    """Plain subgradient without momentum (for diversity)."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_convolve(a)
        M = b.max()
        grad_a = compute_gradient(a, b, top_k=top_k, T=T)

        a_trial = a - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()

        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.008, 0.03)
        else:
            no_improve += 1
            if no_improve >= 80:
                # Perturb to escape local min
                noise = rng.standard_normal(n) * 0.02
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sc_pert = 2.0 * n * fast_convolve(a_pert).max()
                if sc_pert < best_score:
                    a = a_pert
                lr *= 0.75
                no_improve = 0
            lr = max(1e-9, lr)

    return best_a, best_score


def lsm_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        max_iter = max(50, int(remaining * 8))

        def obj_func(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
            return 2.0 * n * lse

        def grad_func(z_arr):
            z_max = z_arr.max()
            exp_z = np.exp(z_arr - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            W = rfft(w, fft_size)
            A_cur = rfft(a_cur, fft_size)
            grad_a = 2.0 * np.real(irfft(np.conj(W) * A_cur, fft_size))[:n]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': max_iter * 30, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def fourier_init(n, rng, method='random'):
    """Generate initial sequence in Fourier domain."""
    fft_size = 1 << (2 * n - 1).bit_length()

    if method == 'random':
        A = np.zeros(fft_size // 2 + 1, dtype=complex)
        A[0] = 1.0
        freqs = np.arange(1, fft_size // 2 + 1)
        decay = np.exp(-0.5 * freqs / n)
        A[1:] = (rng.standard_normal(fft_size // 2) + 1j * rng.standard_normal(fft_size // 2)) * decay * 0.1
        a = np.real(irfft(A, fft_size))[:n]
    elif method == 'lowpass':
        A = np.zeros(fft_size // 2 + 1, dtype=complex)
        A[0] = 1.0
        cutoff = rng.integers(3, max(4, n // 3))
        for i in range(1, cutoff + 1):
            A[i] = (rng.standard_normal() + 1j * rng.standard_normal()) * 0.05
        a = np.real(irfft(A, fft_size))[:n]
    elif method == 'cosine':
        x = np.linspace(0, 1, n)
        k = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * math.pi)
        a = 1.0 + 0.3 * np.cos(2 * math.pi * k * x + phase)
    elif method == 'barker_like':
        # Alternating pattern with smooth falloff
        x = np.linspace(0, 1, n)
        a = np.abs(np.cos(math.pi * x * 3)) * np.exp(-2 * x * (1 - x))
        a = a + 0.1
    elif method == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init_search(n, time_limit):
    """Try multiple diverse initializations and return the best after quick opt."""
    rng = np.random.default_rng(999)
    methods = ['random', 'lowpass', 'cosine', 'barker_like', 'triangle',
               'random', 'lowpass', 'cosine']
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()

    per_method_time = time_limit / len(methods)

    for i, method in enumerate(methods):
        if time.monotonic() - t0 > time_limit - 5:
            break
        a_init = fourier_init(n, rng, method=method)
        t_sub = min(per_method_time * 0.8, (time_limit - (time.monotonic() - t0)) * 0.5)
        if t_sub < 5:
            break
        a_opt, sc = plain_subgrad(a_init, t_sub, top_k=15, T_start=8.0, T_end=0.5)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbation_restart(a0, time_limit, n_restarts=5):
    """Try multiple perturbations of current best and optimize each."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 20:
            break
        remaining = min((time_limit - (time.monotonic() - t0)) / (n_restarts - restart), 120.0)

        scale = 0.04 / (restart + 1)
        a_pert = best_a + rng.standard_normal(n) * scale
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a_opt, sc_opt = momentum_subgrad(a_pert, remaining, top_k=15,
                                         T_start=5.0, T_end=0.01, beta=0.9, lr_init=0.008)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"    PerturbRestart {restart+1}: new best {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast FFT-based gradient methods with momentum, annealed top-k soft-max,
    L-BFGS LogSumExp refinement, diverse Fourier initialization, perturbation
    restarts, and multi-n exploration. No LP solves - pure gradient-based for speed.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = fourier_init(n_init, rng, method='random')
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 0: Diverse initialization (quick) ===
    print(f"\n=== Phase 0: Diverse Init (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t0_budget = min(rem * 0.06, 60.0)
        a_div, sc_div = diverse_init_search(n_best, t0_budget)
        try_update(a_div, "diverse_init")

    # === Phase 1: Momentum subgradient (main workhorse) ===
    print(f"\n=== Phase 1: Momentum Subgrad (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 350.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_mom, sc_mom = momentum_subgrad(best_seq, t1, top_k=20,
                                        T_start=10.0, T_end=0.01,
                                        beta=0.9, lr_init=0.01)
        try_update(a_mom, "momentum")

    # === Phase 2: Plain subgradient with different params ===
    print(f"\n=== Phase 2: Plain Subgrad (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 180.0)
    if t2 > 20:
        a_plain, sc_plain = plain_subgrad(best_seq, t2, top_k=10,
                                         T_start=5.0, T_end=0.005)
        try_update(a_plain, "plain_subgrad")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 300.0)
    if t3 > 30:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb, sc_lb = lsm_lbgfs(best_seq, t3, T_stages=T_stages)
        try_update(a_lb, "lbgfs_lse")

    # === Phase 4: Perturbation restarts ===
    print(f"\n=== Phase 4: Perturbation Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 30:
        a_pert, sc_pert = perturbation_restart(best_seq, t4, n_restarts=5)
        try_update(a_pert, "perturb_restart")

    # === Phase 5: Second momentum pass ===
    print(f"\n=== Phase 5: Momentum Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 20:
        a_mom2, sc_mom2 = momentum_subgrad(best_seq, t5, top_k=10,
                                          T_start=5.0, T_end=0.005,
                                          beta=0.95, lr_init=0.005)
        try_update(a_mom2, "momentum2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = n_best
        for mult in [0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 300 <= cand_n <= 6000 and cand_n != n_best:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.2, 70.0)
            if t_sub < 15:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            a_cand, _ = momentum_subgrad(init_cand, t_sub * 0.6, top_k=15,
                                         T_start=5.0, T_end=0.01)
            try_update(a_cand, f"mom_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cand2, _ = lsm_lbgfs(a_cand, min(rem * 0.35, 50.0),
                                       T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.01])
                try_update(a_cand2, f"lse_n{cand_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # High-momentum subgradient
        t_fin1 = min(rem * 0.35, 50.0)
        if t_fin1 > 10:
            a_fin, _ = momentum_subgrad(best_seq, t_fin1, top_k=5,
                                       T_start=2.0, T_end=0.005,
                                       beta=0.98, lr_init=0.003)
            try_update(a_fin, "final_mom")

        # Very low T L-BFGS
        rem = deadline - time.monotonic()
        if rem > 15:
            t_fin2 = min(rem * 0.35, 50.0)
            a_fin2, _ = lsm_lbgfs(best_seq, t_fin2,
                                  T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
            try_update(a_fin2, "final_lbgfs")

        # Last momentum burst
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = momentum_subgrad(best_seq, min(rem * 0.4, 25.0),
                                        top_k=3, T_start=1.0, T_end=0.003,
                                        beta=0.99, lr_init=0.002)
            try_update(a_fin3, "final_mom2")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
