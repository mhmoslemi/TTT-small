import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(fixed_seq):
    """Build (2n-1) x n matrix T such that T @ b = convolve(fixed_seq, b)."""
    n = len(fixed_seq)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        js = max(0, j)
        je = min(size, j + n)
        idx = np.arange(js, je)
        T[idx, j] = fixed_seq[idx - j]
    return T


def local_eval(a):
    """Fast local evaluation of the objective (assumes a is normalized, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_simplex(a):
    """Project onto the probability simplex {a >= 0, sum(a) = 1}."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] >= 0.1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_response_lp(fixed_a, solver, warm_start=None):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(fixed_a)
    if warm_start is not None:
        warm_start = np.maximum(warm_start, 0)
        ws = warm_start / warm_start.sum()
        b0 = cp.Value(ws)
    else:
        b0 = None

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed_a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1.0]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for attempt_solver in [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            if b0 is not None:
                b_var.value = ws
            prob.solve(solver=attempt_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def asymmetric_init(n, mode, rng):
    """Create asymmetric initial sequences of various types."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if mode == 0:
        # Linear ramp (strongly asymmetric)
        a = (x + 0.1) ** 1.0
    elif mode == 1:
        # Exponential ramp
        a = np.exp(2.0 * x)
    elif mode == 2:
        # Quadratic ramp
        a = (x + 0.05) ** 2.0
    elif mode == 3:
        # Cubic with floor
        a = (x + 0.1) ** 3.0 + 0.05
    elif mode == 4:
        # Super-exponential
        a = np.exp(4.0 * x)
    elif mode == 5:
        # Power law
        a = (x + 0.05) ** 0.5
    elif mode == 6:
        # Gaussian bump (off-center)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.1 * np.exp(-8.0 * (x - 0.8) ** 2)
    elif mode == 7:
        # Random asymmetric
        a = np.ones(n) * 0.1
        a += rng.random(n) * x  # bias toward end
        a = np.maximum(a, 0.01)
    elif mode == 8:
        # Double exponential (decaying from both ends, peak in middle-left)
        a = np.exp(-3.0 * np.abs(x - 0.4)) + 0.3 * np.exp(-5.0 * np.abs(x - 0.7))
    else:
        # Smooth asymmetric polynomial
        a = 1.0 + 0.5 * x + 0.3 * x ** 2 - 0.2 * x ** 3
        a = np.maximum(a, 0.01)
    return normalize_simplex(a)


def lp_alternation(a0, solver, time_limit, n_iters_max=50000, verbose=True):
    """Bidirectional LP alternation with Nesterov averaging and perturbation."""
    n = len(a0)
    a = normalize_simplex(a0)
    b = a.copy()
    y_a = a.copy()  # Nesterow averages
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    # Pre-build the convolution matrix T_a (rebuild when a changes significantly)
    T_a = build_conv_matrix(a)
    T_b = build_conv_matrix(b)

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given y_a
        T_ya = build_conv_matrix(y_a)
        b_new, M1 = solve_response_lp(y_a, solver, warm_start=b)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize_simplex(y_b)
            b = (it * b + (it + 2) * alpha * b_new) / (it + (it + 2) * alpha) if it > 0 else b_new
            b = normalize_simplex(b)

        # Step 2: Best a given y_b
        T_yb = build_conv_matrix(y_b)
        a_new, M2 = solve_response_lp(y_b, solver, warm_start=a)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize_simplex(y_a)
            a = (it * a + (it + 3) * alpha * a_new) / (it + (it + 3) * alpha) if it > 0 else a_new
            a = normalize_simplex(a)

        # Evaluate and track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        # Check for stalling
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                # Perturb to escape local optimum
                noise_scale = 0.03 / np.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize_simplex(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize_simplex(y_b)
                a = a + rng.standard_normal(n) * noise_scale * 0.5
                a = normalize_simplex(a)
                b = b + rng.standard_normal(n) * noise_scale * 0.5
                b = normalize_simplex(b)
                stall = 0

            if verbose and (it + 1) % 1000 == 0:
                print(f"      LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, n_steps=500000):
    """Nesterov-averaged projected subgradient method on the non-smooth objective."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Nesterov averaging
    avg_a = np.zeros(n)
    total_weight = 0.0
    best_avg_score = float('inf')
    best_avg_a = a.copy()

    # Step size parameters
    # Use diminishing step size: lr_t = c / sqrt(t)
    # Choose c based on initial "spread"
    b0 = fast_convolve(a)
    M0 = b0.max()
    c_lr = 0.5 * np.sqrt(M0)  # heuristic

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M

        # Find active set: positions where b[k] is within 1% of max
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient (average of gradients of active quadratics)
        g = np.zeros(n)
        for k in active:
            # dq_k/da[m] = 2*a[k-m] (when valid)
            # More carefully: for each m, the contribution is a[k-m] if 0<=k-m<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Step size
        t = step + 1
        lr = c_lr / np.sqrt(t)

        # Gradient step
        a_trial = a - lr * g

        # Project to simplex
        a_new = normalize_simplex(a_trial)

        # Update Nesterov average
        weight = 2.0 / (t + 1.0)
        avg_a = (1.0 - weight) * avg_a + weight * a_new
        total_weight += 1.0

        # Evaluate
        sc_new = 2.0 * n * fast_convolve(a_new).max()
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Also check the average
        if step % 100 == 0 and total_weight > 10:
            sc_avg = 2.0 * n * fast_convolve(avg_a).max()
            if sc_avg < best_avg_score:
                best_avg_score = sc_avg
                best_avg_a = avg_a.copy()
            if best_avg_score < best_score:
                best_score = best_avg_score
                best_a = best_avg_a.copy()

        a = a_new

        # Periodic verbose
        if (step + 1) % 50000 == 0:
            print(f"      SubGrad step={step+1} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_list=None):
    """LogSumExp-smoothed L-BFGS optimization with log-parametrization."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_list is None:
        T_list = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    a_safe = np.maximum(a, 1e-300)
    z = np.log(a_safe) - np.log(a_safe.sum())

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 15))

        # Use closure-free approach: pass T via a global-like mechanism
        # Actually, we need to define the objective inline. Let's use a different approach.
        T_current = T

        def obj_func(z_param):
            z_max = z_param.max()
            exp_z = np.exp(z_param - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_scaled = b / T_current
            b_max = b_scaled.max()
            lse = b_max + np.log(np.sum(np.exp(b_scaled - b_max)))
            return 2.0 * n * lse

        def grad_func(z_param):
            z_max = z_param.max()
            exp_z = np.exp(z_param - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_scaled = b / T_current
            b_max = b_scaled.max()
            exp_ws = np.exp(b_scaled - b_max)
            w = exp_ws / exp_ws.sum()

            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]

            # Chain rule for softmax parametrization
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def interpolate_to_n(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def active_set_polish(a0, time_limit, n_rounds=200):
    """Polish by only perturbing elements that affect the active peaks."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Active peaks (within 0.1% of max)
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Find which a[j] values affect the active peaks
        affected = set()
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected.update(range(js, je + 1))
        affected = list(affected)

        if not affected:
            break

        # Compute subgradient only on affected elements
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Small step
        lr = 0.01 * M / (np.linalg.norm(g) + 1e-15)
        a_trial = a - lr * g
        a_trial = normalize_simplex(a_trial)
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try random perturbation on affected elements
            n_perturb = max(3, len(affected) // 5)
            perturb_idx = rng.choice(affected, size=min(n_perturb, len(affected)), replace=False)
            a_trial = a.copy()
            a_trial[perturb_idx] *= (1.0 + rng.standard_normal(len(perturb_idx)) * 0.01)
            a_trial = normalize_simplex(a_trial)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial

    return best_a, best_score

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Multi-n, multi-init quick search (25% time) ===
    print(f"\n=== Phase 1: Multi-n Quick Search ===")
    t1_budget = min((deadline - time.monotonic()) * 0.25, 400.0)
    n_trials = []

    # Determine which n values to try
    if parent is not None:
        base_n = len(parent)
        n_trials = [base_n]
        for mult in [0.8, 0.9, 1.1, 1.25, 1.5]:
            cn = int(base_n * mult)
            if 400 <= cn <= 5000 and cn not in n_trials:
                n_trials.append(cn)
    else:
        n_trials = [800, 1000, 1200, 1500, 2000]

    # Also always try some asymmetric inits at the best n
    if parent is not None:
        n_trials.insert(0, len(parent))

    seen_n = set()
    for n_try in n_trials:
        if n_try in seen_n:
            continue
        seen_n.add(n_try)
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(rem * 0.25, 60.0)

        # Try a few asymmetric inits for this n
        init_score = float('inf')
        init_best = None
        for mode in [0, 1, 3, 5]:
            a_init = asymmetric_init(n_try, mode, rng)
            sc = 2.0 * n_try * fast_convolve(a_init).max()
            if sc < init_score:
                init_score = sc
                init_best = a_init

        if init_best is None:
            continue

        # Quick LP (short time)
        a_lp, sc_lp = lp_alternation(init_best, solver, time_limit=t_sub * 0.7,
                                      n_iters_max=5000, verbose=False)
        try_update(a_lp, f"lp_n{n_try}_m{mode}")

        # Quick subgradient
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sg, sc_sg = subgradient_nesterov(a_lp, min(rem * 0.3, 15.0), n_steps=50000)
            try_update(a_sg, f"sg_n{n_try}")

    # === Phase 2: Intensive LP on best (35% time) ===
    print(f"\n=== Phase 2: Intensive LP (score={best_score:.10f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 600.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_int, sc_int = lp_alternation(best_seq, solver, time_limit=t2,
                                        n_iters_max=100000, verbose=True)
        try_update(a_int, "intensive_lp")

    # === Phase 3: Subgradient with Nesterov averaging (20% time) ===
    print(f"\n=== Phase 3: Nesterov Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        a_ns, sc_ns = subgradient_nesterov(best_seq, t3, n_steps=2000000)
        try_update(a_ns, "nesterov_sg")

    # === Phase 4: LogSumExp L-BFGS (15% time) ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        T_list = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_lse, sc_lse = logsumexp_lbgfs(best_seq, t4, T_list=T_list)
        try_update(a_lse, "logsumexp")

    # === Phase 5: Active-set polishing (remaining time) ===
    print(f"\n=== Phase 5: Active Set Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.5, 200.0)
    if t5 > 10:
        a_pol, sc_pol = active_set_polish(best_seq, t5, n_rounds=50000)
        try_update(a_pol, "active_polish")

    # === Phase 6: Final intensive round ===
    print(f"\n=== Phase 6: Final Round (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick LP restart with perturbation
        t6a = min(rem * 0.3, 60.0)
        if t6a > 15:
            # Perturb best solution
            n_best = len(best_seq)
            a_pert = best_seq + rng.standard_normal(n_best) * 0.02
            a_pert = normalize_simplex(a_pert)
            a_r, sc_r = lp_alternation(a_pert, solver, time_limit=t6a,
                                        n_iters_max=5000, verbose=False)
            try_update(a_r, "restart_lp")

        # Final subgradient
        rem = deadline - time.monotonic()
        t6b = min(rem * 0.4, 80.0)
        if t6b > 10:
            a_fsg, _ = subgradient_nesterov(best_seq, t6b, n_steps=500000)
            try_update(a_fsg, "final_sg")

        # Final LogSumExp
        rem = deadline - time.monotonic()
        t6c = min(rem * 0.3, 60.0)
        if t6c > 15:
            a_flse, _ = logsumexp_lbgfs(best_seq, t6c,
                                         T_list=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01])
            try_update(a_flse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
