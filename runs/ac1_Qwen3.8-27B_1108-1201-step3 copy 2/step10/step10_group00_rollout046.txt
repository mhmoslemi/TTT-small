The problem is: minimize f(a) = max_k (a*a)[k] subject to a >= 0, sum(a) = 1.

The subgradient at the argmax k* is: 2 * a[k* - i] (for i such that 0 <= k*-i < n), i.e., the vector `v` where `v[i] = 2*a[k*-i]` (with a[j] = 0 for j outside [0,n-1]).

where Proj_simplex is the projection onto {a >= 0, sum(a) = 1}.

This is very fast per iteration (the projection onto the simplex is O(n log n)).

But the subgradient method converges slowly (O(1/sqrt(t))). We can accelerate it with Nesterov acceleration.

- v_t = extrapolate(a_{t-1}, a_{t-2})
- g_t = subgradient of f at v_t
- a_t = Proj_simplex(v_t - eta_t * g_t)

With eta_t = eta_0 / sqrt(t), this gives O(1/t) convergence.

This could be much faster per iteration than the LP method, and we might be able to do many more iterations in the same time.

Now let me think about the accelerated subgradient method more carefully.

For minimizing f(a) = max_k (a*a)[k] over the simplex:

The Nesterov method (from Nesterov 1983, as formulated in Beck & Teboulle 2009):
- Initialize: x_0 = a0, y_0 = a0
- For t = 1, 2, ...:
  - alpha_t = 2 / (t + 1) (or 2/(t+2) depending on convention)
  - g_t = subgradient of f at y_{t-1}
  - x_t = Proj(y_{t-1} - alpha_t * g_t / c) where c is a bound on the Lipschitz constant (or use adaptive step)
  - y_t = (1 - alpha_{t+1}) * x_t + alpha_{t+1} * y_{t-1} (or the other convention)

Actually, there are different conventions. Let me use the one from the "Nesterov accelerated gradient" paper:

Method (restart when going backwards):
- x_0, t = 1
- For each iteration:
  - y = x + (t-1)/(t+2) * (x - x_old)  [extrapolation]
  - g = subgrad(f, y)
  - x_new = Proj(y - (1/(t*L)) * g)  [step]
  - if f(x_new) > f(x): x_new = x  [restart]
  - x_old = x, x = x_new, t = t + 1

Where L is the Lipschitz constant of the gradient (if f were smooth). For a nonsmooth function, we use the subgradient norm as a proxy.

Actually, for the subgradient method, the convergence rate is O(1/sqrt(t)) without acceleration and O(1/t) with Nesterov acceleration (under appropriate conditions). The step size for the accelerated method is eta_t = eta_0 / (t * sqrt(f(x_0) - f*)) or something like that.

import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size, n=size)


def proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.nonzero(u * idx > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, max_attempts=3):
    """Given fixed a (sum=1), LP to find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    
    for attempt in range(max_attempts):
        for s in solvers_to_try:
            if s is None:
                continue
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                    b_val = np.asarray(b_var.value, dtype=np.float64)
                    b_val = np.maximum(b_val, 0)
                    sv = b_val.sum()
                    if sv < 1e-15:
                        continue
                    b_val = b_val / sv
                    return b_val
            except Exception:
                continue
        # If all failed, try with relaxed constraint
        if attempt < max_attempts - 1:
            time.sleep(0.1)
    
    return None


def subgradient_at(a, k_star):
    """Compute subgradient of max_k (a*a)[k] at the specific k_star."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    g[idx] = 2.0 * a[k_star - idx][::-1]
    return g


def weighted_subgradient(a, b_conv, weights):
    """Compute weighted subgradient using active set weights."""
    n = len(a)
    g = np.zeros(n)
    for k in range(len(b_conv)):
        if weights[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += weights[k] * 2.0 * a[k - idx][::-1]
    return g


def accel_subgradient(a0, time_limit, n_max_iters=5000000):
    """Nesterov-accelerated subgradient method for min max_k (a*a)[k] s.t. a in simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    x_prev = a.copy()
    x_curr = a.copy()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    
    t_start = time.monotonic()
    t_iter = 1
    eta0 = 1.0
    momentum = 0.0
    
    for it in range(n_max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Extrapolation (Nesterov)
        if t_iter > 1:
            momentum = (t_iter - 1) / (t_iter + 2.0)
            y = x_curr + momentum * (x_curr - x_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = x_curr.copy()
        else:
            y = x_curr.copy()
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_star = np.argmax(conv_y)
        g = subgradient_at(y, k_star)
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            g_norm = 1e-15
        eta = eta0 / (t_iter * g_norm)
        
        # Projected gradient step
        x_new = y - eta * g
        x_new = proj_simplex(x_new)
        
        # Restart check
        conv_new = fast_convolve(x_new)
        val_new = conv_new.max()
        
        if val_new > best_val * 1.001:  # Got worse, restart
            x_prev = x_curr.copy()
            # Don't update x_curr, keep it
            eta0 = max(eta0 * 0.9, 0.01)
        else:
            x_prev = x_curr.copy()
            x_curr = x_new
            if val_new < best_val:
                best_val = val_new
                best_a = x_new.copy()
        
        t_iter += 1
        
        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    AccelSubGrad iter={t_iter} ({elapsed:.0f}s): val={best_val:.8f}")
    
    return best_a, best_val


def logsumexp_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s_exp = exp_z.sum()
    if s_exp < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s_exp
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx][::-1]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_val):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_val, n, stage_T)
            return obj
        
        def jac(z_val):
            if time.monotonic() > deadline:
                return np.zeros_like(z_val)
            _, g = logsumexp_obj_grad(z_val, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s_exp = exp_z.sum()
    if s_exp < 1e-300:
        return np.ones(n) / n
    return exp_z / s_exp

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
