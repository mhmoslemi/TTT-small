"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import scipy.sparse
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_T(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k - j] if 0 <= k - j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    T = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def solve_lp(a, solver):
    """Solve LP: min M s.t. convolve(a, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    return None, None

def gs_initialize(n, n_iter=100, rng=None):
    """Gerchberg-Saxton inspired initialization for flat auto-convolution."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.ones(n) / n
    for _ in range(n_iter):
        # FFT
        A = rfft(a, 2 * n)
        # Make magnitude as flat as possible (keep DC = 1)
        mag = np.abs(A)
        phase = np.angle(A)
        # Set all non-DC magnitudes to the mean
        target_mag = np.mean(mag[1:])
        new_mag = mag.copy()
        new_mag[1:] = target_mag
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.allclose(a, a_new, atol=1e-10):
            a = a_new
            break
        a = a_new
    return a

def lp_alternation(a_init, n_iters, solver, use_nesterov=True, stall_restart=20):
    """LP alternation with optional Nesterov acceleration."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    if use_nesterov:
        y = a.copy()
    
    t0 = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > 3600:
            break
        
        # Solve LP for b given current a (or y if Nesterov)
        a_use = y if use_nesterov else a
        b_new, M_val = solve_lp(a_use, solver)
        if b_new is None:
            stall += 1
            if stall > stall_restart:
                break
            continue
        stall = 0
        
        if use_nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * b_new
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
            # Primal iterated average
            a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa
        else:
            a = b_new
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
    
    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func_loc(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse
        
        def jac_loc(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            # grad of sum_k w_k * (a*a)_k w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def peak_coordinate_descent(a0, time_limit, top_m=8):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

        a = np.ones(n) / n

    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]
        
        improved = False
        for peak_k in top_peaks[:5]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:6]
            
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
            # Try simultaneous reduction
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.998
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = evaluate_local(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

def subgradient_refine(a0, time_limit):
    """Subgradient method targeting active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-7, lr)
    
    return best_a, best_score

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1:
        # Already in the simplex if all non-negative
        v = np.maximum(v, 0)
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return v / s
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def projected_subgradient(a_init, time_limit, lr0=0.01):
    """Projected subgradient method for min max_k (a*a)_k s.t. sum(a)=1, a>=0."""
    n = len(a_init)
    a = simplex_project(a_init.astype(np.float64))
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = lr0
    no_improve = 0
    avg_grad = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of gradients of active constraints
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2 * a[k - idx]
        g /= len(active)
        
        # Average gradient (Polyak averaging)
        avg_grad = (avg_grad * no_improve + g) / (no_improve + 1)
        
        # Step
        a_trial = a - lr * avg_grad
        a_trial = simplex_project(a_trial)
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive lr
        if no_improve >= 20:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)
    
    return best_a, best_score

    data_T = np.tile(a, n)

    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)

    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:

        b_val = b_val / s
        return b_val, M_val
    return None, None

```python
def precompute_indices(n):
    """Precompute row and column indices for the convolution matrix T."""
    size = 2 * n - 1
    # For column j, nonzeros are at rows j, j+1, ..., j+n-1 with values a[0], ..., a[n-1]
    # Total nonzeros: n * n
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    for j in range(n):
        start = j * n
        rows[start:start+n] = j + np.arange(n, dtype=np.int32)
        cols[start:start+n] = j
    return rows, cols, size
```

This is O(n^2) to compute but only needs to be done once per n value.

```python
def solve_lp_fast(a, rows, cols, size):
    """Solve LP: min M s.t. convolve(a, b)_k <= M, sum(b)=1, b>=0 using scipy linprog."""
    n = len(a)
    # Build T as sparse
    data_T = np.tile(a, n)
    T = scipy.sparse.csr_matrix((data_T, (rows, cols)), shape=(size, n))
    
    # A_ub = [T, -1] (for T @ b - M <= 0)
    ones_col = scipy.sparse.csc_matrix(np.ones((size, 1)))
    A_ub = scipy.sparse.hstack([T, -ones_col], format='csc')
    b_ub = np.zeros(size)
    
    # A_eq = [1, 1, ..., 1, 0] (for sum(b) = 1)
    A_eq_data = np.ones(n)
    A_eq = scipy.sparse.csc_matrix((A_eq_data, (np.zeros(n, dtype=np.int32), np.arange(n, dtype=np.int32))), shape=(1, n+1))
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # Minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        b_val = res.x[:n].copy()
        M_val = float(res.x[n])
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, M_val
    return None, None
```

def lp_alternation(a_init, time_limit, idx_data, use_nesterov=True, max_iters=100000):
    n = len(a_init)
    rows, cols, size = idx_data
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    if use_nesterov:
        y = a.copy()
    
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_use = y if use_nesterov else a
        b_new, M_val = solve_lp_fast(a_use, rows, cols, size)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        stall = 0
        
        if use_nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * b_new
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
            a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa
        else:
            a = b_new
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        
        if (it + 1) % 100 == 0:
            score_a = evaluate_local(a)
            if use_nesterov:
                score_y = evaluate_local(y)
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
    
    # Final evaluation
    final_score = evaluate_local(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if use_nesterov:
        final_score_y = evaluate_local(y)
        if final_score_y < best_score:
            best_score = final_score_y
            best_a = y.copy()
    
    return best_a, best_score
