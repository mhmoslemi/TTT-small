Let me think about the multiplicative refinement more carefully:

The idea is:
- We have a non-negative sequence a with sum=1
- We want to minimize max(conv(a,a))
- At each step, identify the "active" peaks (positions where conv is near max)
- Compute the gradient of the smoothed max
- Apply a multiplicative update

The multiplicative update (also known as "Exponentiated Gradient" or "Hedge" algorithm) is:
a_new[i] = a[i] * exp(-eta * g[i]) / Z

where Z = sum_j a[j] * exp(-eta * g[j]) and g[i] is the gradient.

This is equivalent to mirror descent with the negative entropy regularizer, which is the natural choice for the simplex constraint (a >= 0, sum(a) = 1).

The convergence rate of mirror descent on the simplex is O(sqrt(log(n)/T)) for convex objectives, which is decent for moderate n.

For the gradient, I'll use the soft-max gradient:
g[i] = sum_k softmax_w[k] * 2 * a[k-i] / |active|

where softmax_w[k] = exp(c[k]/T) / sum_j exp(c[j]/T) for a small temperature T.

For very small T, this approaches the subgradient of the max.

```python
    """Solve LP with constraint generation."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Start with top constraints
    # First, compute a quick estimate of where the peaks will be
    # (using the current a as a proxy)
    c = fast_convolve(a_fixed)
    top_k = np.argsort(c)[-n_constraints:]

    T = build_conv_matrix(a_fixed)  # Full matrix
    T_sub = T[top_k, :]

        cp.sum(b_var) == 1
    ])
    prob.solve(solver=solver)
    
    b = np.asarray(b_var.value)
    
    # Check all constraints

        ])
```

When we change a[i] → a[i] - δ and a[j] → a[j] + δ, the change in conv(a,a) is:

Δconv[k] = conv(a',a')[k] - conv(a,a)[k]

where a' = a + δ(e_j - e_i).

conv(a', a') = conv(a, a) + δ(conv(e_j, a) + conv(a, e_j)) - δ(conv(e_i, a) + conv(a, e_i)) + δ²(conv(e_j, e_j) - conv(e_j, e_i) - conv(e_i, e_j) + conv(e_i, e_i))

Now:
- conv(e_j, a)[k] = a[k-j] (if 0 ≤ k-j ≤ n-1, else 0)
- conv(a, e_j)[k] = a[k-j] (same thing by symmetry of convolution with a delta... wait, no)

Actually, conv(a, e_j)[k] = sum_i a[i] * e_j[k-i] = a[k-j] (if 0 ≤ k-j ≤ n-1)

And conv(e_j, a)[k] = sum_i e_j[i] * a[k-i] = a[k-j] (if 0 ≤ k-j ≤ n-1)

So conv(e_j, a) = conv(a, e_j) = a shifted by j.

Therefore:
Δconv[k] = 2δ * (a[k-j] - a[k-i]) + δ² * (δ_{k, 2j} - δ_{k, i+j} - δ_{k, i+j} + δ_{k, 2i})
= 2δ * (a[k-j] - a[k-i]) + δ² * (δ_{k, 2j} - 2δ_{k, i+j} + δ_{k, 2i})

For small δ, the first-order term dominates. So:
Δconv[k] ≈ 2δ * (a[k-j] - a[k-i])

The new max is:
max_k (conv_a[k] + 2δ * (a[k-j] - a[k-i]))

To compute this efficiently:
- v[k] = a[k-j] - a[k-i] (with zero-padding where indices are out of range)
- new_max = max(conv_a + 2δ * v)

This is O(n) to compute v and O(n) to find the max. Total: O(n) per move.

```python
def evaluate_move(conv_a, a, i, j, delta):
    """Evaluate the change in max when moving delta from position i to position j."""
    n = len(a)
    # v[k] = a[k-j] - a[k-i] (zero-padded)
    v = np.zeros(2*n - 1)
    # a[k-j] for k in [j, j+n-1]
    v[j:j+n] = a
    # -a[k-i] for k in [i, i+n-1]
    v[i:i+n] -= a
    # New max
    new_conv = conv_a + 2 * delta * v
    return new_conv.max()
```

"""

2. Intensive simulated annealing with O(n) incremental convolution updates

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def local_obj(a, conv_a=None):
    """Compute 2n*max(conv) given sum=1 normalization assumption."""
    n = len(a)
    if conv_a is None:
        conv_a = fast_convolve(a)
    return 2.0 * n * conv_a.max()


def local_eval(a):
    """Full evaluation matching evaluate_sequence objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def solve_best_response(a_fixed, max_cg_iters=30, n_init=400):
    """
    Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Uses constraint generation with scipy HiGHS LP.
    """
    n = len(a_fixed)
    size = 2 * n - 1

    # Initial constraint set: top positions of conv(a, uniform)
    b_guess = np.ones(n) / n
    c = convolve_ab(a_fixed, b_guess)
    active_k = np.argsort(c)[-n_init:]

    best_b = b_guess.copy()
    best_M = float(c.max())

    for cg_it in range(max_cg_iters):
        m_act = len(active_k)
        # Build A_ub: m_act x (n+1)
        A_ub = np.zeros((m_act, n + 1))
        for idx, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[idx, js:je+1] = a_fixed[k - js:k - js + (je - js + 1)]
            A_ub[idx, n] = -1.0
        b_ub = np.zeros(m_act)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0.0, None)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success:
                b_val = np.maximum(res.x[:n], 0)
                sv = b_val.sum()
                if sv < 1e-10:
                    break
                b_val /= sv
                M_val = res.x[n]
                c_new = convolve_ab(a_fixed, b_val)
                new_max = float(c_new.max())
                if new_max < best_M:
                    best_M = new_max
                    best_b = b_val.copy()
                # Check violations
                violated = np.where(c_new > M_val + 1e-9)[0]
                if len(violated) == 0:
                    break
                active_k = np.unique(np.concatenate([active_k, violated]))
                if len(active_k) > 2000:
                    active_k = np.argsort(c_new)[-2000:]
        except Exception:
            break

    return best_b, best_M


def lp_alternation(a0, time_limit, max_iters=5000):
    """Bidirectional LP alternation with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0.copy())
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = local_obj(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve best b given y_a
        b_new, _ = solve_best_response(y_a, max_cg_iters=20, n_init=300)
        if b_new is not None and np.isfinite(b_new).all():
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
            b = normalize(b)

        # Solve best a given y_b
        a_new, _ = solve_best_response(y_b, max_cg_iters=20, n_init=300)
        if a_new is not None and np.isfinite(a_new).all():
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + alpha * (it + 3) * a_new) / ((it + 1) + alpha * (it + 3))
            a = normalize(a)

        sc_a = local_obj(a)
        sc_b = local_obj(b)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
            stall = 0

        if (it + 1) % 20 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                rng = np.random.default_rng(it)
                y_a = normalize(y_a + rng.standard_normal(n) * 0.03)
                y_b = normalize(y_b + rng.standard_normal(n) * 0.03)
                stall = 0
            if (it + 1) % 100 == 0:
                print(f"    LP alt iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def simulated_annealing_refine(a0, time_limit, seed=42, verbose=True):
    """
    High-throughput simulated annealing with O(n) incremental convolution updates.
    Uses single-point moves (90%) and two-point moves (10%).
    """
    n = len(a0)
    a = normalize(a0.copy())
    conv_a = fast_convolve(a).copy()
    current_obj = local_obj(a, conv_a)
    best_a = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    T = 0.03
    T_min = 5e-7
    base_delta = 0.3 / math.sqrt(n)  # ~0.0087 for n=1200
    n_steps = 0
    n_accept = 0
    print_idx = 0

    # Preallocate
    v = np.zeros(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        n_steps += 1
        r = rng.random()

        if r < 0.85:
            # Single position move
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = base_delta * (0.3 + 0.7 * rng.random())
            if delta > a[i] * 0.6:
                delta = a[i] * 0.6
            if delta < 1e-12:
                continue

            # Fast evaluation using incremental update
            v[:] = 0
            v[j:j+n] += a
            v[i:i+n] -= a
            new_max = (conv_a + 2.0 * delta * v).max()
            new_obj = 2.0 * n * new_max
        elif r < 0.95:
            # Two-point move: decrease a[i] and a[i2], increase a[j] and a[j2]
            i = rng.integers(n)
            i2 = rng.integers(n)
            if i2 == i:
                continue
            j = rng.integers(n)
            j2 = rng.integers(n)
            if j == i or j == i2 or j2 == i or j2 == i2:
                continue
            delta = base_delta * 0.5 * (0.3 + 0.7 * rng.random())
            if delta > a[i] * 0.5 or delta > a[i2] * 0.5:
                delta = min(a[i], a[i2]) * 0.5
            if delta < 1e-12:
                continue

            v[:] = 0
            v[j:j+n] += a
            v[j2:j2+n] += a
            v[i:i+n] -= a
            v[i2:i2+n] -= a
            new_max = (conv_a + 2.0 * delta * v).max()
            new_obj = 2.0 * n * new_max
        else:
            # Block move via FFT (expensive but powerful)
            w = int(rng.integers(2, min(8, max(2, n // 20))))
            i = rng.integers(0, n - w)
            j = rng.integers(0, n - w)
            if j >= i and j < i + w:
                continue
            src = a[i:i+w].sum()
            if src < 1e-12:
                continue
            frac = 0.1 + 0.3 * rng.random()
            amt = frac * src / w
            if amt * w > a[i:i+w].min() * 0.5:
                continue
            a_trial = a.copy()
            a_trial[i:i+w] -= amt
            a_trial[j:j+w] += amt
            if a_trial.min() < -1e-12:
                continue
            new_conv = fast_convolve(a_trial)
            new_obj = 2.0 * n * new_conv.max()

        # Metropolis
        delta_obj = new_obj - current_obj
        if delta_obj <= 0 or (T > T_min and rng.random() < math.exp(-delta_obj / T)):
            # Accept
            n_accept += 1
            if r < 0.85:
                conv_a += 2.0 * delta * v
                # Second-order correction
                d2 = delta * delta
                pos_2j = 2 * j
                if pos_2j < 2 * n - 1:
                    conv_a[pos_2j] += d2
                pos_ij = i + j
                if pos_ij < 2 * n - 1:
                    conv_a[pos_ij] -= 2.0 * d2
                pos_2i = 2 * i
                if pos_2i < 2 * n - 1:
                    conv_a[pos_2i] += d2
                a[i] -= delta
                a[j] += delta
                current_obj = new_obj
            elif r < 0.95:
                conv_a += 2.0 * delta * v
                d2 = delta * delta
                for p in (2*j, i+j, 2*i, 2*j2, i2+j2, 2*i2):
                    if p < 2*n-1:
                        pass
                # Second order for two-point: more complex, just recompute if needed
                # For simplicity, recompute conv_a periodically
                if n_steps % 50 == 0:
                    conv_a = fast_convolve(a)
                a[i] -= delta
                a[i2] -= delta
                a[j] += delta
                a[j2] += delta
                current_obj = new_obj
            else:
                a = a_trial / a_trial.sum()
                conv_a = new_conv
                current_obj = new_obj

            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()

        # Adaptive temperature
        if n_steps % 10000 == 0:
            rate = n_accept / float(n_steps)
            n_accept = 0
            # Adjust T based on acceptance rate
            if rate > 0.7:
                T *= 0.85
            elif rate < 0.15:
                T *= 1.3
            T = max(T_min, min(T, 0.1))
            # Occasional re-normalization to drift
            s = a.sum()
            if abs(s - 1.0) > 1e-6:
                a = a / s
                conv_a = fast_convolve(a)
                current_obj = local_obj(a, conv_a)

        if verbose and n_steps % 500000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA step={n_steps} ({elapsed:.0f}s): best={best_obj:.10f}, T={T:.2e}")

    return best_a, best_obj


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clip = np.maximum(a, 1e-300)
    z = np.log(a_clip)
    best_a = a.copy()
    best_score = local_obj(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 15))

        def lse_obj(zv):
            z_max = zv.max()
            exp_z = np.exp(zv - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return 2.0 * n * lse

        def lse_grad(zv):
            z_max = zv.max()
            exp_z = np.exp(zv - z_max)
            S = exp_z.sum()
            a_cur = exp_z / S
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # grad w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a_cur[k - idx]
            # Chain rule for softmax
            gS = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - gS)

        try:
            res = minimize(lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'disp': False})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = local_obj(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method with averaging for final polish."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_obj(a)
    t0 = time.monotonic()
    lr = 0.02
    avg_a = a.copy()
    count = 0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fast_convolve(a)
        M = conv_a.max()
        threshold = M * 0.999
        active = np.where(conv_a >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        count += 1
        avg_a = avg_a + (a_trial - avg_a) / count

        sc = local_obj(a_trial)
        sc_avg = local_obj(avg_a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        elif sc_avg < best_score - 1e-12:
            best_score = sc_avg
            best_a = avg_a.copy()
            a = a_trial
            no_improve = 0
        else:
            a = a_trial
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def structured_init(n, seed=0):
    """Create diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4.0 * x * (1.0 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x) + 0.1 * np.cos(2.0 * np.pi * 7.0 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1.0 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-30.0 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * x) + 0.05 * np.cos(4.0 * np.pi * x)
    elif choice == 8:
        a = np.maximum(0, 1.0 - 2.0 * np.abs(x - 0.5)) ** 0.7
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP alternation (300s)
    Phase 2: Simulated annealing (500s)
    Phase 3: Multi-n exploration (300s)
    Phase 4: L-BFGS + subgradient polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, seed)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)
    print(f"\nStarting optimization. n={n_best}, score={best_score:.10f}")

    # === Phase 1: LP Alternation ===
    print(f"\n=== Phase 1: LP Alternation (budget: 280s) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 280.0)
    if t1 > 20:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, max_iters=2000)
        try_update(a_lp, "lp_alt")

    # === Phase 2: Simulated Annealing (main workhorse) ===
    print(f"\n=== Phase 2: Simulated Annealing (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 500.0)
    if t2 > 30:
        print(f"  Time budget: {t2:.0f}s")
        a_sa, sc_sa = simulated_annealing_refine(best_seq, t2, seed=seed + 1)
        try_update(a_sa, "sim_anneal")

        # Second SA round with different seed if time permits
        rem = deadline - time.monotonic()
        t2b = min(rem * 0.15, 100.0)
        if t2b > 20:
            a_sa2, sc_sa2 = simulated_annealing_refine(best_seq, t2b, seed=seed + 100)
            try_update(a_sa2, "sim_anneal_2")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_candidates = []
        base_n = n_best
        for mult in [1.25, 1.5, 1.75, 0.75, 2.0]:
            cand_n = int(base_n * mult)
            if 400 <= cand_n <= 5000 and cand_n != base_n:
                n_candidates.append(cand_n)

        for cand_n in n_candidates[:4]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP + SA
            t_lp = t_sub * 0.35
            t_sa = t_sub * 0.60
            a_cand, _ = lp_alternation(init_cand, t_lp, max_iters=500)
            try_update(a_cand, f"lp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cand2, _ = simulated_annealing_refine(
                    best_seq if len(best_seq) == cand_n else a_cand,
                    min(rem * 0.3, t_sa), seed=seed + cand_n % 1000)
                try_update(a_cand2, f"sa_n{cand_n}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS on log-sum-exp
        t_lbg = min(rem * 0.4, 80.0)
        if t_lbg > 10:
            a_lb, sc_lb = lbgfs_logsumexp(best_seq, t_lbg,
                                         T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
            try_update(a_lb, "lbgfs_final")

        # Subgradient
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.3, 50.0)
        if t_sg > 10:
            a_sg, sc_sg = subgradient_refine(best_seq, t_sg, n_steps=20000)
            try_update(a_sg, "subgrad_final")

        # Final quick SA round
        rem = deadline - time.monotonic()
        t_sa_final = min(rem * 0.5, 60.0)
        if t_sa_final > 10:
            a_sf, sc_sf = simulated_annealing_refine(best_seq, t_sa_final, seed=seed + 999, verbose=False)
            try_update(a_sf, "sa_final")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
