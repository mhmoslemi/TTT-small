```python
def proj_simplex(v, s=1.0):
    """Project v onto the simplex {x : sum(x) = s, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast FISTA (Nesterov-accelerated projected subgradient) for quick refinement
2. PDLP-accelerated LP alternating minimization with Nesterov dual averaging
3. LogSumExp L-BFGS with multi-stage temperature annealing
4. Multi-length exploration (n=1200, 1500, 1800)
5. Active-set-guided structured perturbations for local minimum escape
Key innovations: PDLP for fast LP solving, FISTA as solver-free fast phase,
symmetric initialization, and active-set analysis for perturbation guidance.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_active(a, active):
    """Compute subgradient of max(conv(a,a)) at given active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
    g /= float(len(active))
    return g


def get_active_set(b, rel_tol=1e-6):
    """Get indices within rel_tol of the maximum."""
    max_b = b.max()
    if max_b <= 0:
        return np.array([len(b) // 2])
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    return active


def fista_refine(a0, time_limit=60.0, max_iter=200000):
    """FISTA with subgradient of max(conv(a,a)), projected onto simplex."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    y = a.copy()
    t = 1.0
    eta = 0.5
    eta_scale = 1.0
    no_improve = 0
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        active = get_active_set(b, rel_tol=1e-5)
        g = subgradient_at_active(y, active)
        
        step = eta * eta_scale
        a_new = proj_simplex(y - step * g, target_sum=1.0)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        t = t_new
        a = a_new
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 2000:
            eta_scale *= 0.7
            no_improve = 0
            if eta_scale < 0.005:
                break
            # Restart momentum
            t = 1.0
            y = best_a.copy()
        
        if it % 50000 == 0 and it > 0:
            print(f"    FISTA it={it}: score={best_score:.8f} eta_s={eta_scale:.4f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_fast_lp_solver():
    """Find the fastest available LP solver."""
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            c = np.ones(10)
            prob = cp.Problem(cp.Minimize(c @ x), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.PDLP


def lp_best_response(a_fixed, solver, max_time=5.0):
    """Solve LP to find b minimizing max conv(a_fixed, b) s.t. sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    seen = set()
    for s in solvers:
        if s in seen:
            continue
        seen.add(s)
        try:
            if s == cp.PDLP:
                prob.solve(solver=s, verbose=False, max_iter=5000)
            else:
                prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb < 1e-15:
                    continue
                b_val /= sb
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue
    return None, None

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)

        if sy < 1e-15:

def logsumexp_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_autocorr(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    z = np.log(a)
    z -= z.max()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    
    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        iters = max(10, maxiter_per_stage // n_stages)
        
        def func(xx):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(xx, n, T_val)
            return o
        
        def jac(xx):
            if time.monotonic() > deadline:
                return np.zeros_like(xx)
            _, g = logsumexp_obj_grad(xx, n, T_val)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a_out = exp_z / exp_z.sum()
    return a_out


def multiplicative_subgrad(a0, time_limit, n_steps=100000):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    for step in range(n_steps):

        b = fast_autocorr(a)
        max_b = b.max()
        active = get_active_set(b, rel_tol=1e-4)
        g = subgradient_at_active(a, active)
        
        eta_eff = eta * scale
        log_mult = -eta_eff * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()

def structured_init(n, rng):
    """Generate structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x) * 4)
    elif choice == 3:
        # Symmetric tent
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif choice == 4:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 5:
        # Two bumps
        a = np.exp(-10 * (x - 0.3) ** 2) + 0.7 * np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 6:
        # Power law decay
        a = (1.0 - x) ** 0.5 + 0.01
    else:
        #Smooth bump with modulation
        base = np.exp(-4 * (x - 0.5) ** 2)
        a = base * (1.0 + 0.1 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interp_to_n(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_init(n, base_func):
    """Create a symmetric sequence from a base function defined on [0, 0.5]."""
    half = n // 2
    x_half = np.linspace(0, 0.5, half + 1)
    a_half = base_func(x_half)
    a = np.zeros(n)
    a[:half + 1] = a_half
    a[half:] = a_half[:max(0, half + 1 - (n - half))][::-1]
    # Simpler: just mirror
    a = np.zeros(n)
    a[:n//2 + 1] = a_half[:n//2 + 1]
    a[n//2 + 1:] = a_half[:n - n//2 - 1][::-1]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast FISTA (Nesterov-accelerated projected subgradient) for solver-free refinement
2. PDLP-accelerated LP alternating minimization with Nesterov dual averaging

4. Multi-length exploration (n=1200, 1500, 1800)
5. Active-set-guided structured perturbations for local minimum escape
Key innovations: PDLP for fast LP, FFT-based O(n log n) LogSumExp gradient,
FISTA as instant no-solver phase, and adaptive time budgeting.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x)=target_sum, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_active_set(b, rel_tol=1e-6):
    """Get indices within rel_tol fraction of the maximum."""
    max_b = b.max()
    if max_b <= 0:
        return np.array([len(b) // 2])
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    return active


def subgradient_at_active(a, active):
    """Subgradient of max(conv(a,a)) averaged over active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
    g /= float(len(active))
    return g

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / (it + 1.0 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_autocorr(a).max()
        score_y = 2.0 * n * fast_autocorr(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 5 else 0
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp={avg_lp*1000:.0f}ms")

        if len(lp_times) > 50 and np.mean(lp_times[-30:]) > 5.0:
            print(f"    LP too slow ({np.mean(lp_times[-30:]):.1f}s), stopping at it={it+1}")
            break

    return best_a, best_score


def logsumexp_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient w.r.t. log-domain z, using FFT for gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_autocorr(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w_sum = exp_terms.sum()
    lse = max_b + T_temp * np.log(w_sum)
    obj = 2.0 * n * lse

    w = exp_terms / w_sum

    # FFT-based gradient: g[j] = sum_k w[k] * a[k-j]
    # Need N >= (2n-1) + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad_a = corr[:n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing using FFT gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    z = np.log(a)
    z -= z.max()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        iters = max(10, maxiter_per_stage // n_stages)
        # Scale iterations by remaining time fraction
        time_per_stage = time_limit / n_stages
        frac = min(1.0, remaining / time_per_stage)
        iters = max(10, int(iters * frac))

        def func(xx):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(xx, n, T_val)
            return o

        def jac(xx):
            if time.monotonic() > deadline:
                return np.zeros_like(xx)
            _, g = logsumexp_obj_grad(xx, n, T_val)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a_out = exp_z / exp_z.sum()
    return a_out


def multiplicative_subgrad(a0, time_limit, n_steps=100000):
    """Multiplicative weights subgradient descent targeting active peaks."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.02
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        active = get_active_set(b, rel_tol=1e-4)
        g = subgradient_at_active(a, active)

        eta_eff = eta * scale
        log_mult = -eta_eff * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x) * 4)
    elif choice == 3:
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        a = (1.0 - x) ** 0.4 + 0.01
    elif choice == 7:
        base = np.exp(-3 * (x - 0.5) ** 2)
        a = base * (1.0 + 0.15 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        a = (1.0 - x) ** 0.8 * (1.0 + 0.5 * x)
    else:
        a = np.exp(-5 * (x - 0.4) ** 2) + 0.3 * np.exp(-5 * (x - 0.8) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interp_to_n(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate():
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FISTA, LP alternating (PDLP), FFT-accelerated L-BFGS, multi-length search.
    """
    rng = np.random.default_rng(2024)
    deadline = time.monotonic() + 1750.0

    best_seq = None
    best_score = float('inf')

    # Find LP solver
    solver = find_fast_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: FISTA quick refinement (no solver needed) ===
    print(f"\n=== Phase 1: FISTA n={n_best} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.06, 50.0)
    if t1 > 10:
        print(f"  Budget: {t1:.0f}s")
        a_f, _ = fista_refine(best_seq, t1, max_iter=200000)
        local_update(a_f, "fista_1")

    # === Phase 2: Nesterov LP Round 1 ===
    print(f"\n=== Phase 2: Nesterov LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.28, 480.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, max_iters=1000000)
        local_update(a_lp, "lp_1")

    # === Phase 3: L-BFGS LogSumExp (FFT-accelerated) ===
    print(f"\n=== Phase 3: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 350.0)
    if t3 > 30:
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        print(f"  Budget: {t3:.0f}s, {len(T_stages)} stages")
        a_lb = lbgfs_refine(best_seq, t3, T_stages, maxiter_per_stage=10000)
        local_update(a_lb, "lbgfs_1")

    # === Phase 4: LP Round 2 (from improved best) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t4, solver, max_iters=500000)
        local_update(a_lp2, "lp_2")

    # === Phase 5: Multi-n exploration ===
    for n_try in [1500, 1800]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        print(f"\n=== Phase 5: n={n_try} (best score={best_score:.8f}) ===")
        t_total = min(rem * 0.22, 250.0)
        if t_total < 30:
            continue

        init_n = interp_to_n(best_seq, n_try)

        # Quick FISTA at new n
        rem_sub = deadline - time.monotonic()
        t_f = min(rem_sub * 0.15, t_total * 0.2, 40.0)
        if t_f > 10:
            a_fn, _ = fista_refine(init_n, t_f, max_iter=100000)
            local_update(a_fn, f"fista_n{n_try}")

        # LP at new n
        rem_sub = deadline - time.monotonic()
        t_lp_n = min(rem_sub * 0.35, t_total * 0.5)
        if t_lp_n > 30:
            a_lp_n, _ = nesterov_lp_refine(best_seq, t_lp_n, solver, max_iters=300000)
            local_update(a_lp_n, f"lp_n{n_try}")

        # L-BFGS if this n became best
        rem_sub = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem_sub > 40:
            t_lb_n = min(rem_sub * 0.3, 150.0)
            T_stages2 = [5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
            a_lb_n = lbgfs_refine(best_seq, t_lb_n, T_stages2, maxiter_per_stage=6000)
            local_update(a_lb_n, f"lbgfs_n{n_try}")

    # === Phase 6: Perturbation escape with diverse strategies ===
    print(f"\n=== Phase 6: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        perturbations = []

        # Generate diverse perturbations
        x = np.linspace(0, 2 * np.pi, n_cur)
        perturbations.append(0.02 * np.cos(2 * x + 0.3))
        perturbations.append(0.015 * np.cos(5 * x + 1.0))
        perturbations.append(0.01 * (np.cos(3 * x) + 0.5 * np.cos(7 * x + 2.0)))
        x_lin = np.linspace(0, 1, n_cur)
        perturbations.append(0.02 * np.exp(-15 * (x_lin - 0.5) ** 2))
        perturbations.append(0.015 * (x_lin ** 2 - 0.25) * 4)

        for trial, pert in enumerate(perturbations):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break

            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p

            # Quick optimization on perturbed
            t_sub = min(rem_sub * 0.25, 60.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, max_iters=150000)
                local_update(a_opt, f"pert{trial}")

            # If still time, try L-BFGS
            rem_sub2 = deadline - time.monotonic()
            if best_score < 1.505 and rem_sub2 > 25:
                T_stages3 = [3.0, 1.5, 0.7, 0.3, 0.1, 0.05, 0.02]
                t_lb_p = min(rem_sub2 * 0.3, 35.0)
                if t_lb_p > 10:
                    a_lb_p = lbgfs_refine(a_pert if s_p > 1e-15 else best_seq, t_lb_p,
                                          T_stages3, maxiter_per_stage=3000)
                    local_update(a_lb_p, f"pert{trial}_lb")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 45.0)
        if t_fin > 8:
            T_stages4 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages4, maxiter_per_stage=4000)
            local_update(a_fin, "final_lb")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgrad(best_seq, min(rem * 0.3, 20.0), n_steps=80000)
        local_update(a_ms, "final_ms")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_lp_f, _ = nesterov_lp_refine(best_seq, min(rem * 0.4, 15.0), solver, max_iters=50000)
        local_update(a_lp_f, "final_lp")

    # Final FISTA polish
    rem = deadline - time.monotonic()
    if rem > 8:
        a_f_fin, _ = fista_refine(best_seq, min(rem * 0.5, 10.0), max_iter=50000)
        local_update(a_f_fin, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
