def fast_convolve(a):
    n = len(a)

    """Iteratively whiten the spectrum of a while maintaining nonnegativity."""
    n = len(a)

    s = a.sum()
    if s < 1e-15:
        return a

            alpha = 1.0

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def softmax_gradient_step(a, T, lr):
    """One step of soft-subgradient descent."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    # Soft weights
    w = np.exp((conv - max_c) / T)
    w = w / w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        m = np.arange(0, k_hi - k_lo + 1)
        grad[j] = 2.0 * np.dot(w[k_lo:k_hi+1], a[m])
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = a
    else:
        a_new = a_new / s
    return a_new, conv, w

def soft_subgradient_descent(a0, n_iters=5000, T0=0.01, T1=0.001, lr0=0.1, time_limit=60):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (3 * n).bit_length()  # for the gradient computation
    a_rev = a[::-1].copy()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature and learning rate
        frac = it / max(n_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        lr = lr0 * (1.0 - frac * 0.8)
        
        # Compute convolution
        conv = fast_convolve(a)
        
        # Compute soft weights
        max_c = conv.max()
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient via FFT
        a_rev = a[::-1]
        A_rev = rfft(a_rev, fft_size)
        W = rfft(w, fft_size)
        grad_conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * grad_conv[n-1 : 2*n-1]
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score
