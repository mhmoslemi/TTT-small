"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with adaptive multi-point step search (primary)
2. Subgradient descent with peak-weighted projection for global exploration
3. Multi-initialization strategy (geometric, parent, random)
4. Fine-grained polishing with very small steps
Key insight: optimal sequences are strongly asymmetric, reducing center cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_auto_conv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_auto_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def coord_descent(a0, time_limit, n_coarse_sweeps=0, verbose=True, label=""):
    """
    Fast coordinate descent with adaptive step sizes.
    Uses incremental convolution updates for O(n) per coordinate.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_auto_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute step factors (asymmetric: more resolution near 0)
    factors = np.array([
        -0.49, -0.44, -0.39, -0.34, -0.29, -0.24, -0.19, -0.15,
        -0.12, -0.09, -0.07, -0.05, -0.03, -0.02, -0.01,
        0.01, 0.02, 0.03, 0.05, 0.07, 0.09, 0.12, 0.15,
        0.19, 0.24, 0.29, 0.34, 0.39, 0.44, 0.49
    ])

    conv_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            k_start = i
            k_end = min(i + n - 1, conv_len - 1)
            affected_len = k_end - k_start + 1
            quad_idx = 2 * i - k_start  # position of quadratic term within affected range

            # Unaffected max
            if k_start > 0:
                unaff_l = conv[:k_start].max()
            else:
                unaff_l = 0.0
            if k_end < conv_len - 1:
                unaff_r = conv[k_end + 1:].max()
            else:
                unaff_r = 0.0
            unaff_max = max(unaff_l, unaff_r)

            if ai < 1e-14:
                # Try small increments from zero
                a_shifted = a[k_start - i:k_end - i + 1]
                best_delta = 0.0
                best_nm = M
                for trial in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    new_aff = conv[k_start:k_end + 1] + 2.0 * trial * a_shifted
                    if 0 <= quad_idx < affected_len:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] += trial * trial
                    nm = max(unaff_max, float(new_aff.max()))
                    if nm < best_nm - 1e-15:
                        best_nm = nm
                        best_delta = trial
                if best_delta > 0:
                    a[i] = best_delta
                    changes = 2.0 * best_delta * a_shifted
                    if 0 <= quad_idx < affected_len:
                        changes = changes.copy()
                        changes[quad_idx] += best_delta * best_delta
                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cs = 2.0 * n * M
                    if cs < best_score:
                        best_score = cs
                        best_a = a.copy()
                    improved = True
                continue

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_nm = M

            # Try all factors
            for f in factors:
                delta = f * ai
                nv = ai + delta
                if nv < 0:
                    continue
                new_aff = conv_aff + 2.0 * delta * a_shifted
                if 0 <= quad_idx < affected_len:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] += delta * delta
                nm = max(unaff_max, float(new_aff.max()))
                if nm < best_nm - 1e-15:
                    best_nm = nm
                    best_delta = delta

            # If best step is at the boundary, try finer steps near it
            if abs(best_delta) >= 0.44 * ai - 1e-16 and best_delta != 0:
                # Try finer steps
                fine_start = best_delta / ai
                fine_factors = np.linspace(fine_start * 0.9, fine_start * 1.01, 8)
                for f in fine_factors:
                    delta = f * ai
                    nv = ai + delta
                    if nv < 0:
                        continue
                    new_aff = conv_aff + 2.0 * delta * a_shifted
                    if 0 <= quad_idx < affected_len:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] += delta * delta
                    nm = max(unaff_max, float(new_aff.max()))
                    if nm < best_nm - 1e-15:
                        best_nm = nm
                        best_delta = delta

            if best_delta != 0.0 and best_nm < M - 1e-15:
                a[i] = ai + best_delta
                changes = 2.0 * best_delta * a_shifted
                if 0 <= quad_idx < affected_len:
                    changes = changes.copy()
                    changes[quad_idx] += 2.0 * best_delta * ai + best_delta * best_delta - 2.0 * best_delta * a_shifted[quad_idx]
                    # Actually: the change at quad_idx should be (a[i]+d)^2 - a[i]^2 = 2*a[i]*d + d^2
                    # But 2*delta*a_shifted[quad_idx] = 2*delta*a[i] (since a_shifted[quad_idx] = a[2i-i] = a[i])
                    # So we just need to add delta^2 additionally
                    # Let me redo: changes[j] = 2*delta*a_shifted[j] for all j
                    # At quad_idx, we need: 2*delta*a[i] + delta^2
                    # changes[quad_idx] = 2*delta*a_shifted[quad_idx] = 2*delta*a[i]
                    # So we add delta^2 to changes[quad_idx]:
                    changes[quad_idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cs = 2.0 * n * M
                if cs < best_score:
                    best_score = cs
                    best_a = a.copy()
                improved = True

        if not improved:

        else:

        if verbose and sweeps % 50 == 0:

def subgradient_descent(a0, time_limit, lr_init=0.005, top_k=50, verbose=True):
    """Subgradient descent with peak-weighting and simplex projection."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())

    best_a = a.copy()
    conv = fast_auto_conv(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    conv_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        # Multi-peak subgradient with softmax weighting
        k_act = min(top_k, conv_len)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # conv[k] = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] (if k != 2j)
            # or a[k-j] + a[k-j] but if k-j == j then it's 2*a[j]
            # Actually: d conv[k]/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] for k-j != j
            # and = 2*a[j] for k-j == j (same thing)
            # So it's always 2*a[k-j]
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_auto_conv(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def geometric_init(n, r, normalize_to_sum_one=True):
    """Create geometric sequence with ratio r."""
    a = r ** np.arange(n)
    if normalize_to_sum_one:
        a = a / a.sum()
    return a


def exp_init(n, decay):
    """Create exponential decay sequence."""
    a = np.exp(-np.arange(n) / decay)
    a = a / a.sum()
    return a


def power_init(n, p):
    """Create power-law decay sequence."""
    a = (1.0 + np.arange(n)) ** (-p)
    a = a / a.sum()
    return a


def raised_cosine_init(n, asym=0.5):
    """Raised cosine with asymmetry parameter."""
    t = np.arange(n) / max(n - 1, 1)
    # Asymmetric raised cosine
    a = 0.5 * (1 + np.cos(np.pi * (2 * t - 2 * asym)))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def random_smooth_init(n, rng, n_freq=5):
    """Random smooth sequence from low-frequency cosine components."""
    freqs = rng.integers(1, 20, size=n_freq)
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    amps = rng.uniform(0.3, 1.0, size=n_freq)
    t = np.arange(n)
    a = np.ones(n)
    for i in range(n_freq):
        a += amps[i] * np.cos(2 * np.pi * freqs[i] * t / n + phases[i])
    a = np.maximum(a, 0.01)
    return a / a.sum()

        remaining = t5_end - time.monotonic()
        if remaining < 20:
            break
        # Perturb best sequence
        noise_scale = rng.uniform(0.01, 0.05) / np.sqrt(n)
        pert = best_seq + rng.standard_normal(n) * noise_scale
        pert = normalize(pert)
        t_alloc = min(remaining * 0.5, 50.0)
        a_r, s_r = coord_descent(pert, t_alloc, verbose=False, label=f"restart{r}")
        try_update(a_r, f"restart{r}")

    # === Phase 6: Fine polishing ===
    print(f"\n=== Phase 6: Fine polish (best={best_score:.8f}) ===")
    t6_end = deadline - 30
    if time.monotonic() < t6_end - 10:
        t_alloc = min(t6_end - time.monotonic() - 10, 120.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a6, s6 = fine_polish(best_seq, t_alloc, verbose=True)
        try_update(a6, "fine_polish")

    # === Phase 7: Final CD pass ===
    if time.monotonic() < deadline - 10:
        t_alloc = min(deadline - time.monotonic() - 10, 60.0)
        if t_alloc > 10:
            print(f"\n=== Phase 7: Final CD ({t_alloc:.0f}s) ===")
            a7, s7 = coord_descent(best_seq, t_alloc, verbose=False, label="final")
            try_update(a7, "final_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
