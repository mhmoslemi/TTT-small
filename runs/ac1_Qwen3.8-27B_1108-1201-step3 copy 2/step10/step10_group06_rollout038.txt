Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp gradient optimization with Adam
2. Peak-shaving projected gradient  
3. Exact coordinate descent with golden section search
4. LP Fictitious Play with cvxpy
5. Multi-size continuation and exploration

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def cross_corr_sliding(w, a):
    """Compute h[i] = sum_j w[i+j] * a[j] for i=0..n-1, where w has length 2n-1."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    w_rev = w[::-1]  # w_rev[k] = w[2n-2-k]
    W = rfft(w_rev, N)
    A = rfft(a, N)
    c = irfft(W * A, N)
    # h[i] = c[2n-2-i]
    h = c[n - 1:2 * n - 1][::-1]
    return h

def smooth_max_gradient(a, T):
    """Compute gradient of 2n * LSE(T, conv(a,a)) where a is normalized (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    # Compute weights w_k = exp(T * conv_k) / sum(exp(T * conv_k))
    # For numerical stability, subtract max
    tc = T * conv
    tc_shifted = tc - tc.max()
    exp_tc = np.exp(tc_shifted)
    w = exp_tc / exp_tc.sum()
    # Gradient = 4n * h where h[i] = sum_j w[i+j] * a[j]
    h = cross_corr_sliding(w, a)
    return 4.0 * n * h

def adam_smooth_max(a0, time_limit, T_start=200.0, T_end=5000.0, lr=0.01):
    """Adam optimizer for smooth max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        grad = smooth_max_gradient(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if step % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SM step={step}, T={T:.0f}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
        
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    return best_a, best_score

import numpy as np

_LP_SOLVER = None


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def cross_corr_w_a(w, a):
    """Compute h[i] = sum_j w[i+j] * a[j] for i=0..n-1."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    h = np.real(np.fft.ifft(W * np.conj(A)))[:n]
    return h


def smooth_max_grad(a, T):
    """Gradient of 2n * LSE(T, conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    tc = T * conv
    tc = tc - tc.max()
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    h = cross_corr_w_a(w, a)
    return 4.0 * n * h


def adam_smooth_max(a0, time_limit, T_start=500.0, T_end=8000.0, lr0=0.02):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        lr = lr0 * (1.0 - 0.5 * progress)

        grad = smooth_max_grad(a, T)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 200:
            break
        if step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SM-Adam step={step}, T={T:.0f}, score={best_score:.8f}, el={elapsed:.0f}s")

    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

    return T

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

    except Exception:

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val = b_val / b_val.sum()

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def make_coord_score_fn(a, conv, i, current_sum, n):
    """Precompute data needed for coordinate i scoring."""
    v = a[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    n_aff = k_hi - k_lo + 1

    unaff_max = 0.0
    if k_lo > 0:
        unaff_max = max(unaff_max, float(conv[:k_lo].max()))
    if k_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

    partners = np.arange(k_lo, k_hi + 1) - i
    a_partners = a[partners]
    has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

    c_vals = np.empty(n_aff)
    for idx in range(n_aff):
        k = k_lo + idx
        if k == 2 * i:
            c_vals[idx] = conv[k] - v * v
        else:
            c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

    if has_quad:
        quad_idx = 2 * i - k_lo
    else:
        quad_idx = -1

    return (k_lo, k_hi, n_aff, unaff_max, partners, a_partners,
            has_quad, c_vals, quad_idx, v, current_sum, n)


def coord_score(x, precomp):
    (k_lo, k_hi, n_aff, unaff_max, partners, a_partners,
     has_quad, c_vals, quad_idx, v, current_sum, n) = precomp
    ns = current_sum + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hoping(a0, time_limit, rng, verbose=False):

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.01 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Peak-focused reduction
            conv_p = fast_convolve(a)

            excess = np.maximum(conv_p - target, 0)
            w = excess / max(excess.sum(), 1e-15)

            perturbed = a - 0.05 * h / max(np.abs(h).max(), 1e-15)
        elif pert_type == 2:

            blen = max(2, rng.integers(n // 10, n // 4))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            # Frequency-domain perturbation
            N_fft = 1 << (n - 1).bit_length()
            A = np.fft.rfft(a, N_fft)
            phase_noise = rng.standard_normal(len(A)) * 0.1
            A_pert = A * np.exp(1j * phase_noise)
            perturbed = np.real(np.fft.irfft(A_pert, N_fft))[:n]
        else:

            n_zero = rng.integers(3, max(4, n // 50))
            idx = rng.choice(n, size=n_zero, replace=False)
            mass = perturbed[idx].sum()
            perturbed[idx] = 0
            add_idx = rng.choice(n, size=n_zero * 2, replace=False)
            perturbed[add_idx] += mass / len(add_idx)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
