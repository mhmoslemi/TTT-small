"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact line-search coordinate descent with intersection-point candidate generation
2. Pair updates (mass transfer between coordinated peaks)
3. Multi-scale: tries n=1024, n=2048, n=512

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_conv(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def coord_descent_exact(a0, time_limit, verbose=True, label=""):
    """Coordinate descent with exact line search using intersection-point candidates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    best_M = M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            n_aff = k_end - k_start + 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end + 1]
            conv_aff = conv[k_start:k_end + 1]

            min_delta = -a[i] * 0.999
            # Upper bound
            max_shift = float(a_shifted.max())
            if max_shift < 1e-15:
                max_delta = a[i]
            else:
                max_delta = a[i] * 2.0 + (M + unaff_max) * 0.1 / max_shift

            # Get top peaks in affected region for intersection computation
            top_k = min(20, n_aff)
            if n_aff > top_k:
                top_idx = np.argpartition(conv_aff, -top_k)[-top_k:]
            else:
                top_idx = np.arange(n_aff)
            top_idx = top_idx[np.argsort(conv_aff[top_idx])[::-1]]

            # Generate candidates
            candidates = [0.0, min_delta]

            # Intersection points of top pairs
            for p in range(len(top_idx)):
                for q in range(p + 1, len(top_idx)):
                    k1 = int(top_idx[p])
                    k2 = int(top_idx[q])
                    a1 = a_shifted[k1]
                    a2 = a_shifted[k2]
                    if abs(a1 - a2) < 1e-15:
                        continue
                    d = (conv_aff[k2] - conv_aff[k1]) / (2.0 * (a1 - a2))
                    if min_delta * 0.9 < d < max_delta:
                        candidates.append(d)

            # Quadratic term (k=2i) intersection with unaff_max
            if 2 * i >= k_start and 2 * i <= k_end:
                c_val = float(conv[2 * i]) - unaff_max
                disc = a[i] * a[i] - c_val
                if disc >= 0:
                    d1 = -a[i] + np.sqrt(disc)
                    d2 = -a[i] - np.sqrt(disc)
                    if min_delta * 0.9 < d1 < max_delta:
                        candidates.append(d1)
                    if min_delta * 0.9 < d2 < max_delta:
                        candidates.append(d2)

            # Evaluate best candidate
            best_delta = 0.0
            best_new_M = M

            for delta in candidates:
                delta = max(min_delta, min(delta, max_delta))
                new_M = unaff_max
                # Check top peaks (covers >99.9% of cases)
                for k_idx in top_idx:
                    k = k_start + int(k_idx)
                    if k == 2 * i:
                        val = float(conv[k]) + 2.0 * delta * a[i] + delta * delta
                    else:
                        val = float(conv[k]) + 2.0 * delta * a_shifted[int(k_idx)]
                    if val > new_M:
                        new_M = val
                # Also check unaff_max (already included)
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                old_val = a[i]
                a[i] = max(old_val + best_delta, 0.0)
                actual_delta = a[i] - old_val
                # Update convolution
                for k_idx in range(n_aff):
                    k = k_start + k_idx
                    if k == 2 * i:
                        conv[k] += 2.0 * actual_delta * old_val + actual_delta * actual_delta
                    else:
                        conv[k] += 2.0 * actual_delta * a_shifted[k_idx]
                # Verify M
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                improved = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()

        if time.monotonic() - t0 > time_limit:
            break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweep % 20 == 0:
            print(f"    CD[{label}] sweep={sweep} ({time.monotonic()-t0:.0f}s): M={M:.10f} best={best_M:.10f}")

    return best_a, best_M


def pair_updates(a0, time_limit, verbose=True, label=""):
    """Transfer mass between coordinate pairs that jointly contribute to a peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    best_M = M
    best_a = a.copy()
    t0 = time.monotonic()
    rnd = 0

    while time.monotonic() - t0 < time_limit:
        rnd += 1
        improved = False

        # Find top peaks
        K = min(8, 2 * n - 1)
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.99:
                continue

            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Find top contributing pairs
            best_pair = None
            best_contrib = 0
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contrib = a[i] * a[j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_pair = (i, j)

            if best_pair is None or best_contrib < 1e-12:
                continue

            i, j = best_pair
            if i == j:
                # Single coordinate contributes: try reducing it
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = -a[i] * frac
                    # Check if reducing helps
                    k_s, k_e = i, min(i + n - 1, 2 * n - 2)
                    unaff = 0.0
                    if k_s > 0:
                        unaff = max(unaff, float(conv[:k_s].max()))
                    if k_e < 2 * n - 2:
                        unaff = max(unaff, float(conv[k_e+1:].max()))
                    new_M = unaff
                    sh = a[k_s - i:k_e - i + 1]
                    for kk in range(k_s, k_e + 1):
                        if kk == 2 * i:
                            val = conv[kk] + 2 * delta * a[i] + delta * delta
                        else:
                            val = conv[kk] + 2 * delta * sh[kk - k_s]
                        if val > new_M:
                            new_M = val
                    if new_M < M - 1e-12:
                        a[i] = max(a[i] + delta, 0.0)
                        for kk in range(k_s, k_e + 1):
                            if kk == 2 * i:
                                conv[kk] += 2 * delta * (a[i] - delta) + delta * delta
                            else:
                                conv[kk] += 2 * delta * sh[kk - k_s]
                        M = float(conv.max())
                        improved = True
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        break
            else:
                # Transfer mass from the larger to a third location
                if a[i] >= a[j]:
                    src = i
                else:
                    src = j

                # Try reducing src and see if it helps
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    delta = -a[src] * frac
                    k_s, k_e = src, min(src + n - 1, 2 * n - 2)
                    unaff = 0.0
                    if k_s > 0:
                        unaff = max(unaff, float(conv[:k_s].max()))
                    if k_e < 2 * n - 2:
                        unaff = max(unaff, float(conv[k_e+1:].max()))
                    sh = a[k_s - src:k_e - src + 1]
                    new_M = unaff
                    for kk in range(k_s, k_e + 1):
                        if kk == 2 * src:
                            val = conv[kk] + 2 * delta * a[src] + delta * delta
                        else:
                            val = conv[kk] + 2 * delta * sh[kk - k_s]
                        if val > new_M:
                            new_M = val
                    if new_M < M - 1e-12:
                        a[src] = max(a[src] + delta, 0.0)
                        for kk in range(k_s, k_e + 1):
                            if kk == 2 * src:
                                conv[kk] += 2 * delta * (a[src] - delta) + delta * delta
                            else:
                                conv[kk] += 2 * delta * sh[kk - k_s]
                        M = float(conv.max())
                        improved = True
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        break

        if not improved and rnd > 30:
            break

        if verbose and rnd % 10 == 0:
            print(f"    Pair[{label}] round={rnd} ({time.monotonic()-t0:.0f}s): M={M:.10f} best={best_M:.10f}")

    return best_a, best_M


def fourier_perturb(a0, n_freqs, rng, scale):
    """Add a smooth Fourier perturbation to escape local minima."""
    n = len(a0)
    a = a0.copy()
    t = np.arange(n)
    for k in range(1, n_freqs + 1):
        amp = rng.uniform(0, scale / k)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return normalize(a)


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    best_M = M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    factors = [0.999, 0.998, 0.997, 0.9985, 1.0005, 1.001, 1.002, 1.0015]

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            n_aff = k_end - k_start + 1

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_M = unaff_max
                for k_idx in range(n_aff):
                    k = k_start + k_idx
                    if k == 2 * i:
                        val = float(conv[k]) + 2.0 * delta * a[i] + delta * delta
                    else:
                        val = float(conv[k]) + 2.0 * delta * a_shifted[k_idx]
                    if val > new_M:
                        new_M = val
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                old_val = a[i]
                a[i] = max(old_val + best_delta, 0.0)
                actual_delta = a[i] - old_val
                for k_idx in range(n_aff):
                    k = k_start + k_idx
                    if k == 2 * i:
                        conv[k] += 2.0 * actual_delta * old_val + actual_delta * actual_delta
                    else:
                        conv[k] += 2.0 * actual_delta * a_shifted[k_idx]
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                improved = True
                if M < best_M:
                    best_M = M

def upscale_sequence(a, new_n):
    """Upscale a sequence to a new length using interpolation."""
    old_n = len(a)
    t_old = np.arange(old_n)
    t_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    return normalize(a_new)


def downscale_sequence(a, new_n):
    """Downscale a sequence to a new length."""
    old_n = len(a)
    t_old = np.arange(old_n)
    t_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    return normalize(a_new)

    # === Phase 5: Fine polish on best ===
    print(f"\n=== Phase 5: Fine polish (best={best_score:.8f}) ===")
    t5 = min(deadline - time.monotonic() - 20, 100.0)
    if t5 > 10:
        a_fp, _ = fine_polish(best_seq, t5, verbose=True)
        try_update(a_fp, "fine")

    # Final CD pass
    t6 = min(deadline - time.monotonic() - 5, 40.0)
    if t6 > 5:
        a_fin, _ = coord_descent_exact(best_seq, t6, verbose=False, label="final")
        try_update(a_fin, "final_cd")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
