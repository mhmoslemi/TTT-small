def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_dual_matrix(a):
    """Build sparse matrix for dual LP: (A^T y)_j = sum_k y_k * a_{k-j}."""
    n = len(a)

    cols = []

            if 0 <= idx < n:

                cols.append(k)
                vals.append(a[idx])

    else:
        w = w / w_sum

    # Gradient: sg_j = 2 * sum_k w_k * a_{k-j}
    # = 2 * conv(w, a_flipped)[j + n - 1]
    a_flipped = a[::-1]
    corr = convolve_ab(w, a_flipped)
    sg = 2.0 * corr[n - 1:n - 1 + n]
    return sg, conv.max()


def nesterov_lse_optimize(a0, time_limit, beta_init=50.0, beta_final=500.0,
                          lr_init=0.01, verbose=True, label=""):
    """
    Nesterov-accelerated gradient descent on log-sum-exp approximation.
    Minimize f(p) = (1/beta)*log(sum exp(beta*conv_k(p))) on the simplex.
    """
    n = len(a0)
    p = np.maximum(a0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    v = p.copy()  # Nesterov momentum variable
    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        beta = beta_init * (beta_final / beta_init) ** progress

        # Nesterov extrapolation
        mu = 0.5 / (0.5 + np.sqrt(1.0 + 2.0 * it))
        p_new = simplex_project(v + mu * (v - p))

        # Gradient step with backtracking
        grad, M = logsumexp_gradient(p_new, beta)
        # Normalize gradient for projection
        gnorm = np.sqrt(np.sum((grad - grad.mean()) ** 2))
        if gnorm < 1e-15:
            break

        step = lr / (1.0 + it * 0.001)
        trial = p_new - step * grad
        trial = simplex_project(trial)

        # Check improvement
        trial_M = fast_convolve(trial).max()
        if trial_M > M * 1.01:
            step *= 0.5
            trial = simplex_project(p_new - step * grad)

        v = p.copy()
        p = trial

        # Track best
        M_p = fast_convolve(p).max()
        sc = 2.0 * n * M_p
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

        if verbose and it % 100 == 0:
            print(f"    LSE[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f} lr={step:.6f}")

        if step < 1e-12:
            break

    return best_p, best_score


def adaptive_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with adaptive step sizes and incremental convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_left = conv[:k_start].max() if k_start > 0 else -1e30
            unaff_right = conv[k_end + 1:].max() if k_end < 2 * n - 2 else -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            base = max(a[i], 1e-8)
            factors = [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5,
                       -0.02, 0.02, -0.01, 0.01]

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        a = np.ones(n) / n

        a = a / s

    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_left = conv[:k_start].max() if k_start > 0 else -1e30

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.999, 0.997, 0.995, 0.998, 1.002, 1.004, 1.006, 1.001, 0.9995, 1.0005]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_alternating_improved(a0, time_limit, verbose=True, label=""):
    """
    Improved LP-based alternating minimization with:
    - Cosine damping schedule
    - Periodic random restarts
    - Dual-based subgradient for direction
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Prebuild the dual matrix (depends on a, so we rebuild periodically)
    A_dual = build_dual_matrix(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        progress = (time.monotonic() - t0) / time_limit

        # Rebuild dual matrix every 50 iterations (a has changed)
        if it % 50 == 1:
            A_dual = build_dual_matrix(a)

        # Solve dual LP
        y_star = lp_best_response(a, A_dual)

        # Compute subgradient direction: sg_j = sum_k y_k * a_{k-j}
        sg = subgradient_from_dual_fast(a, y_star)

        # Damping schedule: cosine from 1.0 to 0.1
        alpha = 0.1 + 0.9 * (1.0 + np.cos(np.pi * min(progress, 1.0))) / 2.0

        # Update: move towards the "anti-subgradient" direction
        # We want to minimize max_k conv(a,a)_k, so we move opposite to sg
        # But we need to stay on the simplex
        lr = 0.1 * (1.0 - 0.5 * progress)
        a_new = a - lr * (sg - sg.mean())
        a_new = simplex_project(a_new)

        # Damped update
        a_damped = (1 - alpha) * a + alpha * a_new
        a_damped = np.maximum(a_damped, 0)
        a_damped = a_damped / a_damped.sum()

        # Evaluate
        conv_new = fast_convolve(a_damped)
        score_new = 2.0 * n * conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_damped.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_damped

        # Random perturbation if stuck
        if stagnation > 40:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = simplex_project(a_pert)
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} alpha={alpha:.3f}")

    return best_a, best_score


def spectral_perturbation(a0, rng, strength=0.1):
    """
    Perturb a sequence in the frequency domain to create diverse candidates.
    """
    n = len(a0)
    a = a0.copy()
    # FFT
    A = np.fft.rfft(a)
    # Random phase perturbation
    phases = rng.uniform(-np.pi * strength, np.pi * strength, size=len(A))
    A_pert = A * np.exp(1j * phases)
    # IFFT
    a_pert = np.fft.irfft(A_pert, n=n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return simplex_project(a0 + rng.standard_normal(n) * 0.01)
    return a_pert / s


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0) ** 2
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** 1.5
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        w = n * 0.12
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(5, 15))
        t = np.arange(n)
        a = 0.5 + 0.45 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.4 * np.cos(6 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Asymmetric: mass concentrated on left
        t = np.arange(n)
        a = np.exp(-t / (n * 0.3))
    elif init_type == 7:
        # Power law
        t = np.arange(1, n + 1)
        a = 1.0 / np.power(t, rng.uniform(0.3, 1.0))
    else:
        freqs = rng.integers(1, 10, size=3)
        t = np.arange(n)
        a = np.ones(n)
        for f in freqs:
            a += 0.3 * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_coordinate_exchange(a0, time_limit, verbose=True):
    """
    Try swapping mass between pairs of coordinates to reduce the max convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Find the top peak
        k_peak = int(np.argmax(conv))

        # Find pairs that contribute most to this peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)

        # Try moving mass from high-contribution coordinates to low ones
        contributions = []
        for i in range(i_start, i_end + 1):
            j = k_peak - i
            if j >= 0 and j < n:
                contributions.append((i, j, a[i] * a[j]))

        contributions.sort(key=lambda x: x[2], reverse=True)

        # Try reducing the top contributor
        for (i, j, contrib) in contributions[:5]:
            if contrib < 1e-14:
                continue
            # Try moving some mass from i to a random other coordinate
            for _ in range(3):
                target = int(rng_global.integers(0, n))
                if target == i:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    delta = -a[i] * frac
                    a_test = a.copy()
                    a_test[i] = max(a_test[i] + delta, 0)
                    a_test[target] += delta
                    if a_test[target] < 0:
                        continue
                    s_test = a_test.sum()
                    a_test = a_test / s_test

                    conv_test = conv.copy()
                    # Update convolution for change at i
                    k_s = i
                    k_e = min(i + n - 1, 2 * n - 2)
                    old_ai = a[i]
                    new_ai = a_test[i] * s  # account for renormalization... this is getting complex
                    # Just recompute for simplicity in this local search
                    conv_test = fast_convolve(a_test)
                    M_test = float(conv_test.max())
                    if M_test < M - 1e-13:
                        a = a_test
                        conv = conv_test
                        M = M_test
                        improved = True
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        break

        if not improved and it > 100:
            break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2CE it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            if t_quick < 5:
                break
            a_q, s_q = adaptive_coord_descent(a_pert, t_quick, verbose=False, label=f"pert{i}")
            try_update(a_q, f"pert_cd{i}")

    # Final fine polish
    if time.monotonic() < t4_end - 5:
        t_final = min(t4_end - time.monotonic() - 5, 30.0)
        if t_final > 5:
            a_fin, s_fin = fine_polish(best_seq, t_final, verbose=False)
            try_update(a_fin, "final_polish")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
