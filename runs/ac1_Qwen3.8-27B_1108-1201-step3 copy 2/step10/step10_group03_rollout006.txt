import cvxpy as cp

from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_eval(a):
    """Fast local evaluation of the objective (assumes a is normalized, non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_T(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a_fixed, solver, fallback_solvers=None):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    if fallback_solvers:
        solvers_to_try.extend([s for s in fallback_solvers if s != solver])
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val > 1e-15:
                    return b_val / s_val, float(M_var.value)
        except Exception:
            continue
    return None, None

def lp_alternation(a0, time_limit, solver, fallback_solvers, max_iters=100000):
    """Bidirectional LP alternation with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Best b given y_a
        b_new, M1 = lp_best_response(y_a, solver, fallback_solvers)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = b_new.copy()
        
        # Step 2: Best a given y_b
        a_new, M2 = lp_best_response(y_b, solver, fallback_solvers)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = a_new.copy()
        
        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 15:
                noise_scale = 0.01 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise_scale)
                y_b = normalize(y_b + rng.standard_normal(n) * noise_scale)
                stall = 0
            
            if (it + 1) % 2000 == 0:
                print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score

def spectral_refine(a0, time_limit, n_iters=2000, alpha=0.1):
    """Spectral (Gerchberg-Saxton style) refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        M = best_score / (2.0 * n)
        |A2 = np.abs(A) ** 2
        # Reduce magnitudes that exceed M
        exceed = |A2 > M
        if np.any(exceed):
            scale = np.ones(fft_size // 2 + 1)
            scale[exceed] = np.sqrt(M / |A2[exceed])
            A = A * scale
            a_new = irfft(A, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = 2.0 * n * fast_convolve(a_new).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                else:
                    # Try with blending
                    a_blend = (1-alpha)*a + alpha*a_new
                    a_blend = normalize(a_blend)
                    sc = 2.0 * n * fast_convolve(a_blend).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_blend.copy()
                        a = a_blend
    
    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    # Center z for numerical stability
    z = z - z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 30))
        
        def obj(z):
            exp_z = np.exp(z - z.max())
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z):
            exp_z = np.exp(z - z.max())
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            # Chain rule for softmax (log-param)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 100}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def peak_target_qp(a0, solver, time_limit, n_steps=200):
    """Peak-targeted quadratic programming: minimize sum of squared peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Find top peaks (within 5% of max)
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # QP: minimize (1/2) * sum_{k in active} (linearized_b[k])^2
        #     s.t. sum(delta) = 0, a + delta >= 0
        # Linearized b[k] ≈ b[k] + 2 * sum_j a[k-j] * delta_j
        # We want to minimize: sum_k (b[k] + 2 * sum_j a[k-j] * delta_j)^2
        # which is quadratic in delta
        
        m = len(active)
        # Build Hessian: H.p[j1, j2] = 4 * sum_k a[k-j1] * a[k-j2] for k in active
        # This is 4 * (P^T P) where P has rows P[k_idx, j] = a[k - j]
        P = np.zeros((m, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            P[i, idx] = a[k - idx]
        
        b_active = b[active]
        # Quadratic: minimize ||P @ delta + b_active / 2||^2 + regularization
        # H = 4 * P^T P, g = 4 * P^T @ (b_active / 2) = 2 * P^T @ b_active
        H = 4.0 * (P.T @ P)
        g = 2.0 * (P.T @ b_active)
        
        # Add small regularization for numerical stability
        H += 1e-10 * np.eye(n)
        
        # Solve: minimize (1/2) delta^T H delta + g^T delta + lambda * sum(delta)^2
        # s.t. a + delta >= 0, sum(delta) = 0
        # Using Lagrange: H @ delta + g + nu * 1 + mu = 0, a + delta >= 0
        # For now, use cvxpy
        delta_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Minimize(cp.quad_form(delta_var, cp.psd_wrap(H)) + g @ delta_var),
            [cp.sum(delta_var) == 0, a + delta_var >= 0]
        )
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                delta_val = np.asarray(delta_var.value, dtype=np.float64)
                # Try different step sizes
                for t in [0.5, 0.3, 0.1, 0.05]:
                    a_trial = normalize(a + t * delta_val)
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        break
        except Exception:
            pass
    
    return best_a, best_score

def cos_window(n, power=2):
    """Create a cosine window of length n."""
    x = np.linspace(0, 1, n)
    a = (np.cos(np.pi * x / 2.0)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def gaussian_window(n, sigma=0.3):
    """Create a Gaussian window centered in [0, n-1]."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    fallback = [s for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP] if s != solver]
    print(f"Using LP solver: {solver}, fallback: {fallback}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = cos_window(n_init, power=2)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: LP alternation on parent ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, solver, fallback)
        try_update(a_lp, "lp_parent")
    
    # === Phase 2: Spectral refinement ===
    print(f"\n=== Phase 2: Spectral Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 150.0)
    if t2 > 15:
        a_spec, sc_spec = spectral_refine(best_seq, t2, n_iters=3000, alpha=0.05)
        try_update(a_spec, "spectral")
    
    # === Phase 3: Peak-targeted QP ===
    print(f"\n=== Phase 3: Peak QP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 15:
        a_qp, sc_qp = peak_target_qp(best_seq, solver, t3, n_steps=200)
        try_update(a_qp, "peak_qp")
    
    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, 350.0)
    if t4 > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  T_stages: {T_stages}, budget={t4:.0f}s")
        a_lb, sc_lb = logsumexp_lbgfs(best_seq, t4, T_stages=T_stages)
        try_update(a_lb, "lbgfs")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        # Try different n values
        n_options = [600, 800, 1000, 1200, 1400, 1600, 2000, 2500, 3000]
        n_options = [n for n in n_options if 200 <= n <= 5000 and n != n_best]
        
        for cand_n in n_options:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.15, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            
            # Start from interpolated parent + cosine blend
            init_cand = interpolate_sequence(best_seq, cand_n)
            
            # Quick LP alternation
            a_cand, _ = lp_alternation(init_cand, t_sub * 0.5, solver, fallback)
            try_update(a_cand, f"lp_n{cand_n}")
            
            rem = deadline - time.monotonic()
            if rem > 30:
                T_st4 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                a_cand2, _ = logsumexp_lbgfs(a_cand if a_cand is not None else init_cand,
                                             min(rem * 0.3, 50.0), T_stages=T_st4)
                try_update(a_cand2, f"lbgfs_n{cand_n}")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Another round of LP
        t_lp = min(rem * 0.2, 60.0)
        if t_lp > 10:
            a_fin_lp, _ = lp_alternation(best_seq, t_lp, solver, fallback)
            try_update(a_fin
