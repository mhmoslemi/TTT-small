def solve_best_response_lp_direct(a, active_indices):

    c[n] = 1  # minimize M

    # sum_j a[k-j] * b_j - M <= 0
    A_ub_rows = []
    for k in active_indices:
        row = np.zeros(n + 1)
        row[n] = -1  # coefficient of M
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        A_ub_rows.append(row)

        A_ub = np.array(A_ub_rows)
        b_ub = np.zeros(len(active_indices))

        A_ub = np.empty((0, n+1))
        b_ub = np.zeros(0)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-12:

    Solve: min M s.t. conv(a,b)[k] <= M for k in active_indices,
           sum(b)=1, b>=0
    Returns (b, M) or (None, None).
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    active_indices = sorted(set(active_indices))
    n_constraints = len(active_indices)
    
    if n_constraints == 0:
        # No constraints: b = uniform minimizes trivially
        return np.ones(n) / n, 0.0
    
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for row_idx, k in enumerate(active_indices):
        A_ub[row_idx, n] = -1.0  # -M
        # Coefficients of b: a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[row_idx, j] = a[k - j]
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0  # M doesn't appear in equality
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return None, None
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
            if not res.success:
                return None, None
        except Exception:
            return None, None
    
    b_val = res.x[:n]
    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-12:
        return None, None
    b_val = b_val / s
    M_val = max(M_val, 0)
    return b_val, float(M_val)


def cutting_plane_best_response(a, max_cp_iters=40, init_half_width=50):
    """
    Best response via cutting plane: min_M s.t. conv(a,b)[k]<=M all k.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Initial active set: center ± init_half_width
    active = set()
    k_center = n - 1
    for k in range(max(0, k_center - init_half_width), 
                   min(size, k_center + init_half_width + 1)):
        active.add(k)
    
    b_val = np.ones(n) / n
    M_val = np.inf
    
    for cp_it in range(max_cp_iters):
        result = solve_best_response_lp(a, sorted(active))
        if result[0] is None:
            break
        b_new, M_new = result
        
        # Check full feasibility
        conv_full = fast_convolve(a, b_new)
        max_viol = conv_full.max()
        
        if max_viol <= M_new * (1 + 1e-7) + 1e-14:
            return b_new, M_new
        
        # Add violated constraints
        threshold = M_new * (1 + 1e-7) + 1e-14
        violated = np.where(conv_full > threshold)[0]
        if len(violated) == 0:
            return b_new, M_new
        
        # Add top 15 most violated
        top = violated[np.argsort(conv_full[violated])[::-1]][:15]
        for k in top:
            active.add(int(k))
        
        b_val = b_new
        M_val = M_new
    
    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, n, max_iters=100000):
    """
    Nesterov-accelerated LP alternating.
    f(a) = max_k conv(a,a)[k]. Best response: b = argmin_{sum=1,b>=0} max_k conv(a,b)[k].
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Nesterov parameters
    t_param = 2.0
    y = a.copy()  # extrapolated point
    
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve best response at extrapolated point y
        b_new, M_val = cutting_plane_best_response(y, max_cp_iters=30, 
                                                    init_half_width=40)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / ((it + 2) ** 2)
        a_old = a.copy()
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = a_old.copy()
        else:
            a = a / s_a
        
        # Extrapolation
        t_param_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_param * t_param))
        beta = (t_param - 1.0) / t_param_new
        y = b_new + beta * (b_new - a_old)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        t_param = t_param_new
        
        # Evaluate
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Periodic diversification
        if stall_count > 200:
            noise_scale = 0.01 * math.sqrt(stall_count / 200)
            noise = rng.standard_normal(n) * noise_scale
            y = y + noise
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            a = a + noise * 0.5
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            stall_count = 0
            t_param = 2.0  # Reset Nesterov momentum
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLPM iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_param:.2f} stall={stall_count}")
    
    elapsed = time.monotonic() - t0
    print(f"    NLPM done after {elapsed:.0f}s: best_score={best_score:.8f}")
    return best_a, best_score


def mirror_descent_refine(a0, time_limit, eta0=0.05):
    """
    Mirror descent with subgradient for refinement.
    Used when LP is too slow or for final polish.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    y = a.copy()
    stall = 0
    rng = np.random.default_rng(54321)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        b_y = fast_convolve(y)
        max_b = b_y.max()
        threshold = max_b * 0.999
        active = np.where(b_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_y))])
        
        # Subgradient: g_i = (1/|S|) sum_{k in S} a[k-i] (for valid indices)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(active)
        
        # Mirror descent update (exponential weighting)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shift = g - g.min()
        log_a_new = log_a - eta * g_shift
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y / s_y
        
        a = a_new
        
        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            y = a.copy()
            stall = 0
            eta = max(eta * 0.5, 0.001)
    
    return best_a, best_score
