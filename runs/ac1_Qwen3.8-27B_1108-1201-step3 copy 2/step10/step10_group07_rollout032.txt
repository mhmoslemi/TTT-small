One more consideration: the LP gives the best response, which minimizes max_k (a*b)[k]. But the actual objective is max_k (b*b)[k]. These are different! The LP best response minimizes the cross-convolution, not the auto-convolution.

The alternating scheme works because: if we alternate a → b*(a) → a*(b*) → ..., we get a sequence where each step is optimal in the other variable. At a fixed point, a = b*(a) means that a is optimal given a, which is a first-order optimality condition for the minimax problem.

But there's a subtlety: the LP best response minimizes max_k (a*b)[k], while the objective is max_k (b*b)[k]. At the fixed point, both are equal (since a = b), so it's consistent. But during the iterations, the LP objective and the actual objective might diverge, leading to suboptimal convergence.

An alternative: use the auto-convolution directly in the optimization. For the "b" step, minimize max_k (b*b)[k] subject to sum(b) = 1, b ≥ 0. This is a QP (quadratic constraints), not an LP. We can solve it via:
- Bisection on M: for a given M, check feasibility of (b*b)[k] ≤ M, sum(b) = 1, b ≥ 0. This is a feasibility problem with quadratic constraints, solvable by other means.
- Or use the LP as a surrogate (it's a linearization of the quadratic).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic LP best-response alternating minimization
2. Coordinate descent with exact line search on palindromic pairs
3. Smoothed optimization via log-sum-exp + L-BFGS
4. Random restarts and multi-n exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    n = len(a)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def make_palindromic(a):
    """Average a with its reverse to enforce palindromic symmetry."""
    return (a + a[::-1]) / 2.0


def palindromic_free(a):
    """Get the free variables of a palindromic sequence (first half)."""
    n = len(a)
    d = (n + 1) // 2
    return a[:d]


def palindromic_reconstruct(free):
    """Reconstruct a palindromic sequence from its free variables."""
    d = len(free)
    n = 2 * d - 1  # for odd
    # Actually, for even n: d = n/2, free = a[0..d-1], a[d..2d-1] = reverse(free)
    # For odd n: d = (n+1)/2, free = a[0..d-1], a[d..2d-2] = reverse(free)[:-1]
    n = 2 * d
    a = np.zeros(n)
    a[:d] = free
    a[d:] = free[::-1]
    return a


def palindromic_reconstruct_general(free):
    """Reconstruct from free vars. n = 2*len(free) for even, 2*len(free)-1 for odd."""
    d = len(free)
    # We'll use even n = 2*d
    n = 2 * d
    a = np.zeros(n)
    a[:d] = free
    a[d:] = free[::-1]
    return a


def build_pal_cross_matrix(a):
    """
    Build matrix T such that T @ x = (a * b)[:n] where b is palindromic
    with free variables x = b[0..d-1] and d = n//2.
    Returns T of shape (n, d).
    """
    n = len(a)
    d = n // 2
    T = np.zeros((n, d))
    for j in range(d):
        # b[j] = x[j], b[n-1-j] = x[j]
        # b[j] contributes to (a*b)[k] at k = i+j for all valid i
        # b[n-1-j] contributes to (a*b)[k] at k = i+(n-1-j) for all valid i
        for i in range(n):
            k1 = i + j
            if k1 < n:
                T[k1, j] += a[i]
            k2 = i + (n - 1 - j)
            if k2 < n:
                T[k2, j] += a[i]
    return T


def build_pal_cross_matrix_fast(a):
    """
    More efficient construction of the palindromic cross-convolution matrix.
    T[k, j] = a[k-j] (if 0<=k-j<n) + a[k-(n-1-j)] (if 0<=k-(n-1-j)<n)
    """
    n = len(a)
    d = n // 2
    T = np.zeros((n, d))
    for j in range(d):
        # First term: a[k-j] for k in [j, j+n-1], but we only want k in [0, n-1]
        # So k in [j, n-1], and a[k-j] for k-j in [0, n-1-j]
        idx = np.arange(j, n)
        T[idx, j] += a[:n-j]
        # Second term: a[k-(n-1-j)] for k-(n-1-j) in [0, n-1]
        # k in [n-1-j, n-1-j+n-1] = [n-1-j, 2n-2-j], intersect [0, n-1]
        # k in [n-1-j, n-1]
        start = n - 1 - j

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic LP best-response alternating minimization (primary)
2. Coordinate descent with exact 1D line search on palindromic pairs
3. Smoothed log-sum-exp optimization with L-BFGS
4. Multi-n exploration and random restarts

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def make_palindromic(a):
    return (a + a[::-1].copy()) / 2.0


def reconstruct_pal(free, n):
    """Reconstruct palindromic sequence of length n from first n//2 values."""
    d = n // 2
    a = np.zeros(n)
    a[:d] = free
    a[d:] = free[::-1].copy()
    return a


def build_pal_cross_matrix(a):
    """Build T (n x d) such that T @ x gives first n values of cross-conv(a, b_pal(x))."""
    n = len(a)
    d = n // 2
    T = np.zeros((n, d))
    for j in range(d):
        # First term: a[k-j] for k in [j, n-1]
        idx = np.arange(j, n)
        T[idx, j] += a[:n - j]
        # Second term: a[k-(n-1-j)] for k in [n-1-j, n-1]
        start = n - 1 - j
        if start < n:
            idx2 = np.arange(start, n)
            T[idx2, j] += a[:n - start]
    return T


def find_solver():

    for name in [cp.PDLP, cp.ECOS, cp.CLARABEL, cp.SCS]:

                return name
        except Exception:

    find_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response_pal(a, solver):
    """LP: min M s.t. cross_conv(a,b)_k <= M, sum(b)=1, b>=0, b palindromic."""
    n = len(a)
    d = n // 2
    T = build_pal_cross_matrix(a)
    x = cp.Variable(d, nonneg=True)
    M = cp.Variable()
    constraints = [
        T @ x <= M,
        cp.sum(x) == 0.5
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None, float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, float('inf')
    x_val = np.asarray(x.value, dtype=np.float64)
    x_val = np.maximum(x_val, 0)
    s = x_val.sum()
    if s < 1e-15:
        return None, float('inf')
    x_val = x_val / s
    b = reconstruct_pal(x_val, n)
    M_val = float(M.value) if M.value is not None else float('inf')
    return b, M_val


def lp_alternating_pal(a0, time_limit, solver, verbose=True):
    """Alternating LP minimization with palindromic constraint."""
    n = len(a0)
    a = make_palindromic(np.maximum(a0, 0).astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = make_palindromic(a)

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response_pal(a, solver)
        if b_new is None:

        b_new = make_palindromic(b_new)
        s = b_new.sum()
        if s < 1e-15:

        b_new = b_new / s

        # Damped update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 8.0 / (it - 40.0))

        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        a_new = make_palindromic(a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score_new = score_fast(a_new)
        score_b = score_fast(b_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stagnation = 0
        elif score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        if abs(score_new - prev_score) < 1e-12:
            stagnation += 1
        prev_score = score_new

        a = a_new

        # Random perturbation to escape stagnation
        if stagnation > 15:
            rng = np.random.default_rng(it * 7919)
            for _ in range(5):
                noise = rng.standard_normal(n) * (0.02 / np.sqrt(n))
                a_p = a + noise
                a_p = np.maximum(a_p, 0)

                if a_p.sum() < 1e-15:

                a_p = a_p / a_p.sum()
                sc_p = score_fast(a_p)
                if sc_p < best_score * 1.001:
                    a = a_p
                    break
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.10f} M_lp={M_lp:.10f}")

def coord_descent_pal(a0, time_limit, verbose=True, fine=False):
    """Coordinate descent on palindromic pairs with binary search line search."""
    n = len(a0)
    d = n // 2
    a = make_palindromic(np.maximum(a0, 0).astype(np.float64)).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = make_palindromic(a)

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(d):
            if a[i] < 1e-18:
                continue

            # Affected range: [i, 2*n-2-i]
            k_start = i
            k_end = min(2 * n - 2 - i, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Change in conv when a[i] += delta and a[n-1-i] += delta
            # delta_c[k] = 2*delta*(a[k-i] + a[k-(n-1-i)]) + delta^2*(1_{k=2i} + 2*1_{k=n-1} + 1_{k=2(n-1-i)})
            # Only keep linear part for efficiency (delta^2 is tiny)
            k_range = np.arange(k_start, k_end + 1)
            # a[k-i] for k in range, valid when 0 <= k-i < n
            valid1 = (k_range >= i) & (k_range - i < n)
            a1 = np.zeros(len(k_range))
            a1[valid1] = a[k_range[valid1] - i]
            # a[k-(n-1-i)] for k in range, valid when 0 <= k-(n-1-i) < n
            valid2 = (k_range >= n - 1 - i) & (k_range - (n - 1 - i) < n)
            a2 = np.zeros(len(k_range))
            a2[valid2] = a[k_range[valid2] - (n - 1 - i)]

            slope = 2.0 * (a1 + a2)  # linear coefficient of delta
            conv_aff = conv[k_start:k_end + 1].copy()

            def eval_delta(delta):
                if delta == 0:
                    return max(unaff_max, float(conv_aff.max()))
                new_aff = conv_aff + delta * slope
                # Add quadratic correction at specific points
                for k_special in [2 * i, n - 1, 2 * (n - 1 - i)]:
                    if k_start <= k_special <= k_end:
                        idx = k_special - k_start
                        new_aff = new_aff.copy()
                        new_aff[idx] += delta * delta
                return max(unaff_max, float(new_aff.max()))

            # Binary search for optimal delta in [-a[i], a_max_increase]
            # The function is convex in delta
            lo = -a[i]
            hi = a[i] * 2.0  # reasonable upper bound

            # Check if decreasing helps
            obj_lo = eval_delta(lo)
            obj_0 = max(unaff_max, float(conv_aff.max()))
            obj_hi = eval_delta(hi)

            if obj_lo >= obj_0 and obj_hi >= obj_0:
                # delta=0 is optimal (or close)
                continue

            # Golden section search
            if fine:
                n_iter = 30
            else:
                n_iter = 40

            gr = (np.sqrt(5) - 1) / 2
            c = lo + (1 - gr) * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = eval_delta(c)
            fd = eval_delta(d_pt)

            for _ in range(n_iter):
                if fc < fd:
                    hi = d_pt
                    d_pt = c
                    fd = fc
                    c = lo + (1 - gr) * (hi - lo)
                    fc = eval_delta(c)
                else:
                    lo = c
                    c = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = eval_delta(d_pt)

            best_delta = (lo + hi) / 2.0
            best_val = min(fc, fd)

            if best_val < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0)
                a[n - 1 - i] = a[i]
                conv[k_start:k_end + 1] += best_delta * slope
                # Quadratic correction
                for k_special in [2 * i, n - 1, 2 * (n - 1 - i)]:
                    if k_start <= k_special <= k_end:
                        conv[k_special] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > (10 if fine else 30):
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f} M={M:.10f}")

    return best_a, best_score


def smoothed_lbfgs(a0, time_limit, verbose=True):
    """Optimize using log-sum-exp smoothed max with L-BFGS."""
    n = len(a0)
    a = make_palindromic(np.maximum(a0, 0).astype(np.float64)).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = make_palindromic(a)

    d = n // 2
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    fd = a[:d].copy()

    def objective_and_grad(x):
        a_full = reconstruct_pal(np.maximum(x, 0), n)
        s = a_full.sum()
        if s < 1e-15:
            return 1e10, np.zeros(d)
        a_n = a_full / s
        conv = fast_convolve(a_n)
        # Log-sum-exp approximation
        t = 0.01
        shifted = (conv - conv.max()) / t
        exp_vals = np.exp(np.minimum(shifted, 50))
        lse = conv.max() + t * np.log(exp_vals.sum())
        # Constraint: sum(a) = 1 is already enforced by normalization
        # We optimize in the space of free variables, normalizing after
        grad_conv = np.exp(shifted) / exp_vals.sum() * (1.0 / t) * t
        # grad_conv[k] = exp((conv[k]-max)/t) / sum(exp(...)) -- this is the softmax weight
        # d(conv[k])/d(a[j]) = 2*a[k-j] (for auto-conv)
        w = np.exp(shifted) / exp_vals.sum()
        grad_a = np.zeros(n)
        for k_idx in range(len(w)):
            if w[k_idx] > 1e-10:
                k = k_idx
                for j in range(max(0, k - n + 1), min(n, k + 1)):
                    grad_a[j] += 2.0 * w[k_idx] * a_n[k - j]
        # Normalization correction (approximately)
        grad_a -= grad_a.sum() * a_n
        grad_x = grad_a[:d].copy()
        return lse, grad_x

    # First try with moderate smoothing
    result = minimize(objective_and_grad, fd, method='L-BFGS-B',
                      options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12,
                               'maxfun': 500})

    a_new = reconstruct_pal(np.maximum(result.x, 0), n)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
        a_new = make_palindromic(a_new)
        sc = score_fast(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            fd = a_new[:d].copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS ({elapsed:.0f}s): best={best_score:.10f}")

    # Try with tighter smoothing
    if time.monotonic() - t0 < time_limit * 0.7:
        def objective_tight(x):
            a_full = reconstruct_pal(np.maximum(x, 0), n)
            s = a_full.sum()
            if s < 1e-15:
                return 1e10, np.zeros(d)
            a_n = a_full / s
            conv = fast_convolve(a_n)
            t = 0.001
            shifted = (conv - conv.max()) / t
            exp_vals = np.exp(np.minimum(shifted, 80))
            lse = conv.max() + t * np.log(exp_vals.sum())
            w = np.exp(shifted) / exp_vals.sum()
            grad_a = np.zeros(n)
            for k_idx in range(len(w)):
                if w[k_idx] > 1e-10:
                    k = k_idx
                    j_start = max(0, k - n + 1)
                    j_end = min(n - 1, k)
                    grad_a[j_start:j_end+1] += 2.0 * w[k_idx] * a_n[k - j_end:k - j_start + 1][::-1]
            grad_a -= grad_a.sum() * a_n
            grad_x = grad_a[:d].copy()
            return lse, grad_x

        result2 = minimize(objective_tight, fd, method='L-BFGS-B',
                           options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxfun': 500})
        a_new2 = reconstruct_pal(np.maximum(result2.x, 0), n)
        s2 = a_new2.sum()
        if s2 > 1e-15:
            a_new2 = a_new2 / s2
            a_new2 = make_palindromic(a_new2)
            sc2 = score_fast(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS tight ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 12))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    else:
        a = np.ones(n) * (0.5 + 0.3 * np.random.random(n))
    a = make_palindromic(np.maximum(a, 0.01))
    s = a.sum()
    return a / s

        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        candidate = make_palindromic(candidate)
        candidate = candidate / candidate.sum()
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = score_fast(candidate)
        if np.isfinite(sc) and sc < best_score:
