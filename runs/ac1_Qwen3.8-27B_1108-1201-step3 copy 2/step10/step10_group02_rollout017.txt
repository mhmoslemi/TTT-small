def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None, warm_start=None):

def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver, warm_start=True)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def mirror_descent_optimize(a0, time_limit, eta0=0.06, active_frac=0.999,
                            nesterov=True, bb_adapt=True):
    """Mirror descent with dual BB step sizes and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient(y, active)

        if bb_adapt and it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-5, min(eta, 1.0))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * (0.003 + 0.003 * np.sqrt(no_improve / 500))
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov LP alternating for refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42424)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 15:
                noise_scale = 0.012 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_and_grad(z, T, top_k=None):
    """Compute LogSumExp objective and gradient, optionally using top-k peaks."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)

    if top_k is not None and top_k < len(b):
        sorted_idx = np.argsort(b)[::-1][:top_k]
        b_top = b[sorted_idx]
        max_b = b_top.max()
        exp_b = np.exp(b_top - max_b)
        w = exp_b / exp_b.sum()
        w_full = np.zeros(len(b))
        w_full[sorted_idx] = w
    else:
        max_b = b.max()
        exp_b = np.exp(b - max_b)
        w = exp_b / exp_b.sum()
        w_full = w

    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if w_full[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w_full[k] * a[k - idx]

    logsumexp_val = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp_val

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z_inner, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_obj_and_grad(z_inner, T)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def coordinate_descent_refine(a0, time_limit, n_steps=50000):
    """Aggressive coordinate descent targeting peak-contributing elements with multi-coordinate moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.99998
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9999, 0.9998, 0.9997, 0.9995, 0.999, 0.998, 0.997, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Multi-coordinate simultaneous moves
        if not improved:
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_positions = idx[np.argsort(contribs)[::-1][:5]]

            a_trial = a.copy()
            for pos in top_positions:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.999
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                score = evaluate_local(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 500:
                # Random perturbation to escape
                n_perturb = max(3, n // 200)
                positions = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for pos in positions:
                    factor = 1.0 + rng.standard_normal() * 0.005
                    a_trial[pos] *= max(0.5, factor)
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                if stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.10f} stall={stall}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.3)
    else:
        a = 1.0 / (1.0 + np.exp(-5.0 * (x - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase iterative optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses iterated cycles of mirror descent, LP alternating, LogSumExp L-BFGS,
    and coordinate descent with increasing precision in later cycles.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Main iterative loop with 3 cycles of increasing precision ===
    for cycle in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        cycle_label = f"Cycle {cycle+1}"
        print(f"\n=== {cycle_label} (remaining={remaining:.0f}s, score={best_score:.10f}) ===")

        # Phase A: Mirror Descent
        frac_md = [0.30, 0.25, 0.20][cycle]
        t_alloc = remaining * frac_md
        if t_alloc > 20:
            print(f"  MD: budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_optimize(best_seq, t_alloc,
                                              eta0=[0.05, 0.03, 0.015][cycle],
                                              active_frac=[0.999, 0.9995, 0.9999][cycle],
                                              nesterov=True, bb_adapt=True)
            try_update(a_md, f"md_c{cycle+1}")

        # Phase B: LP Alternating
        remaining = deadline - time.monotonic()
        frac_lp = [0.25, 0.22, 0.18][cycle]
        t_alloc = remaining * frac_lp
        if t_alloc > 20:
            print(f"  LP: budget={t_alloc:.0f}s")
            a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                          n_max_iters=50000)
            try_update(a_lp, f"lp_c{cycle+1}")

        # Phase C: LogSumExp L-BFGS
        remaining = deadline - time.monotonic()
        frac_lb = [0.25, 0.28, 0.30][cycle]
        t_alloc = remaining * frac_lb
        if t_alloc > 15:
            T_stages = [
                [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01],
                [8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005],
                [5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            ][cycle]
            print(f"  L-BFGS: budget={t_alloc:.0f}s")
            a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                                    maxiter_per_stage=6000)
            try_update(a_lb, f"lbgfs_c{cycle+1}")

        # Phase D: Coordinate Descent
        remaining = deadline - time.monotonic()
        frac_cd = [0.15, 0.20, 0.25][cycle]
        t_alloc = remaining * frac_cd
        if t_alloc > 10:
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent_refine(best_seq, t_alloc,
                                                  n_steps=100000)
            try_update(a_cd, f"cd_c{cycle+1}")

    # === Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Final Polish (remaining={remaining:.0f}s) ===")

        # Very low temperature L-BFGS
        t_lb = remaining * 0.4
        if t_lb > 5:
            T_fin = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_fin = logsumexp_lbgfs(best_seq, t_lb, T_stages=T_fin,
                                     maxiter_per_stage=4000)
            try_update(a_fin, "final_lbgfs")

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.5
        if t_cd > 5:
            a_polish, _ = coordinate_descent_refine(best_seq, t_cd, n_steps=200000)
            try_update(a_polish, "final_cd")

        # One more very brief MD pass
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_md_final, _ = mirror_descent_optimize(best_seq, remaining,
                                                     eta0=0.005, active_frac=0.99999,
                                                     nesterov=True, bb_adapt=True)
            try_update(a_md_final, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
