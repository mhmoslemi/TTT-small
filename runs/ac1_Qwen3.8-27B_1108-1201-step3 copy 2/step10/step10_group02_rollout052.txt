def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (sum-normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix_rows(a, k_indices):
    """Build rows of conv matrix for specific lag indices."""
    n = len(a)
    m = len(k_indices)
    T = np.zeros((m, n))
    for i, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[i, js:je+1] = a[je:js-1:-1]
    return T


def solve_lp_best_response(a_fixed, active_indices, solver):

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

            b_val = np.asarray(b_var.value, dtype=np.float64)

            if s < 1e-15:

    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

            if prob.status in ('optimal', 'optimal_inaccurate'):

def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def logsumexp_objective(z, T, n):
    """Compute log-sum-exp smoothed objective and gradient in z-space."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    log_sum_exp = max_b + np.log(np.exp(b - max_b).sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient wrt a
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    size = len(b)
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Convert to z-gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Multi-n exploration ===
    print(f"\n=== Phase 1: Multi-n exploration (score={best_score:.8f}) ===")
    n_current = len(best_seq)
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.11, 200.0)
    if t_phase1 > 30:
        print(f"  Budget: {t_phase1:.0f}s")
        n_options = [n_current]
        for delta in [-200, -100, 100, 200, 300]:
            n_try = n_current + delta
            if 400 <= n_try <= 2500:
                n_options.append(n_try)
        
        t_per_n = t_phase1 / len(n_options)
        for n_try in n_options:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_current:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_alloc = min(t_per_n, deadline - time.monotonic() - 5)
            if t_alloc < 10:
                continue
            print(f"  Trying n={n_try} ({t_alloc:.0f}s)...")
            a_test, sc_test = lp_alternating(a_interp, t_alloc, solver, top_k=100, nesterov_frac=0.8)
            try_update(a_test, f"n_{n_try}")
    
    # === Phase 2: Main LP alternating ===
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main LP alternating (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.39, 700.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s")
        a_lp, sc_lp = lp_alternating(best_seq, t_phase2, solver, top_k=300, nesterov_frac=0.65)
        try_update(a_lp, "lp_main")
    
    # === Phase 3: Perturbation search ===
    print(f"\n=== Phase 3: Perturbation search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.28, 500.0)
    if t_phase3 > 30:
        print(f"  Budget: {t_phase3:.0f}s")
        a_pert, sc_pert = perturb_and_refine(best_seq, t_phase3, solver, n_perturb=60)
        try_update(a_pert, "perturb")
    
    # === Phase 4: Log-sum-exp polishing ===
    print(f"\n=== Phase 4: LogSumExp polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.14, 250.0)
    if t_phase4 > 20:
        print(f"  Budget: {t_phase4:.0f}s")
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_ls = logsumexp_optimize(best_seq, t_phase4, T_stages=T_stages)
        try_update(a_ls, "logsumexp")
    
    # === Phase 5: Final LP refinement (full constraints) ===
    print(f"\n=== Phase 5: Final LP refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.17, 300.0)
    if t_phase5 > 20:
        print(f"  Budget: {t_phase5:.0f}s, using full constraints")
        n_best = len(best_seq)
        a_fin, sc_fin = lp_alternating(best_seq, t_phase5, solver, top_k=2 * n_best - 1, nesterov_frac=0.3)
        try_update(a_fin, "final_lp")
    
    # === Phase 6: Very final polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase6 = min(rem * 0.06, 100.0)
    if t_phase6 > 10:
        # Quick gradient descent at very low temperature
        T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_polish = logsumexp_optimize(best_seq, t_phase6, T_stages=T_stages_fin)
        try_update(a_polish, "polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
