        if time.monotonic() - t0 > time_limit:

            break

        frac = stage / max(1, n_stages - 1)

            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_s = b / T
            b_max = b_s.max()
            lse = b_max + np.log(np.exp(b_s - b_max).sum())
            return 2.0 * n * lse

            exp_z = np.exp(z_m)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_s = b / T
            b_max = b_s.max()
            exp_ws = np.exp(b_s - b_max)
            w = exp_ws / exp_ws.sum()

            # This is 2 * (w * a_cur) (convolution)
            # Use FFT for efficiency
            grad_a = fast_convolve_sym(a_cur, w)
            
            # Chain rule for softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

            """Compute sum_k w[k] * a[|k-j|] type convolution (actually w*a linear conv, symmetric part)."""

            # This is the cross-correlation of w and a, which is the same as convolution of w and a[::-1]
            # Actually: sum_k w[k]*a[k-j] = (w * a_rev)[n-1-j] where a_rev = a[::-1]
            # Or more simply: it's the cross-correlation, which can be done via FFT

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # cross-correlation: ifft(conj(W) * A) gives sum_k w[k] * a[k+j]
            # We want sum_k w[k] * a[k-j] = ifft(W * conj(A))[j]... 
            # Let me just use the direct formula via convolution
            # sum_k w[k]*a[k-j] = (w * a_flip)[n-1-j] where a_flip = a[::-1]
            # Actually: conv(w, a)[k] = sum_j w[j]*a[k-j]
            # So sum_k w[k]*a[k-j] = conv(w, a)[2*k-j]... no this is getting confusing
            # Let me just do it the straightforward way:
            # grad_a[j] = sum_{k=max(j,0)}^{min(n-1+k, 2n-2)} w[k] * a[k-j] (with bounds)

            corr = irfft(W * np.conj(A), fft_size)[:n]

            return 2.0 * corr

                z = z - z.max()
        except Exception as e:
            if verbose:
                print(f"  LSE refine error at stage {stage}: {e}")
    
    return best_a, best_score

def spectral_init(n, mode=0, rng=None):
    """Create structured initial sequence with spectral properties."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    modes = [
        lambda: np.cos(np.pi * x / 2) ** 2,
        lambda: np.sin(np.pi * x) ** 2,
        lambda: np.exp(-3 * x * (1 - x)),
        lambda: np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * 2 * x), 0),
        lambda: np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2)),
        lambda: 0.5 * np.exp(-((x - 0.35) ** 2) / 0.02) + 0.5 * np.exp(-((x - 0.65) ** 2) / 0.02),
        lambda: 1.0 / (1.0 + np.exp(-30 * (x - 0.5))),
        lambda: np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0),
        lambda: np.sin(np.pi * x) ** 0.7,
        lambda: np.exp(-2 * x * (1 - x)) + 0.3 * np.exp(-8 * (x - 0.5) ** 2),
    ]
    a = modes[mode % len(modes)]()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def lp_game_optimize(a0, time_limit, solver, verbose=True):
    """
    LP game-theoretic alternation with Nesterov acceleration.
    Key: monitors dual gap (M_auto - M_lp) for convergence detection.
    """
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LP Game start: n={n}, score={best_score:.10f}")
    
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min M s.t. (y*b)[k] <= M, sum(b)=1, b>=0
        T = build_conv_matrix_sparse(y)
        b_new, M_lp = solve_best_response(y, solver, T)
        if b_new is None:
            break
        
        # Nesterov step with step size 2/(t+2)
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)
        
        # Evaluate
        conv_y = fast_convolve(y)
        M_auto = conv_y.max()
        score_y = 2.0 * n * M_auto
        
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Also check b_new directly
        conv_b = fast_convolve(b_new)
        score_b = 2.0 * n * conv_b.max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Gap = M_auto - M_lp (should decrease to 0 at equilibrium)
        gap = M_auto - M_lp
        
        if (it + 1) % 200 == 0:
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}, "
                      f"M_auto={M_auto:.6e}, M_lp={M_lp:.6e}, gap={gap:.6e}")
            
            if best_score < prev_best - 1e-14:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            # Rejuvenation when stuck
            if stall >= 6:
                noise_scale = 0.03 / math.sqrt(stall)
                y = normalize(y + rng.standard_normal(n) * noise_scale)
                stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  LP Game done: {elapsed:.0f}s, best={best_score:.10f}")
    
    return best_a, best_score


def logsumexp_refine(a0, time_limit, verbose=True):
    """
    FFT-accelerated log-sum-exp refinement with multi-stage temperature annealing.
    Uses softmax parameterization for non-negativity.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LSE refine start: n={n}, score={best_score:.10f}")
    
    # Multi-stage temperature: from moderate to very low
    n_stages = 12
    # Target: at lowest T, the LSE approximates the max very well
    T_values = np.geomspace(1.0, 0.001, n_stages)
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        T = T_values[stage]
        
        # Set up softmax parameterization
        a_safe = np.maximum(a, 1e-20)
        z = np.log(a_safe) - np.log(a_safe).max()
        
        max_iter = max(30, int(remaining * 8 / (n_stages - stage)))
        
        def obj_func(z_vec):
            z_m = z_vec - z_vec.max()
            exp_z = np.exp(z_m)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_s = b / T
            b_max = b_s.max()
            lse = b_max + np.log(np.exp(b_s - b_max).sum())
            return 2.0 * n * lse
        
        def grad_func(z_vec):
            z_m = z_vec - z_vec.max()
            exp_z = np.exp(z_m)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_s = b / T
            b_max = b_s.max()
            exp_ws = np.exp(b_s - b_max)
            w = exp_ws / exp_ws.sum()
            
            # grad_a[j] = 2 * sum_k w[k] * a_cur[k-j] (cross-correlation)
            # Using FFT: ifft(FFT(w) * conj(FFT(a)))[:n]
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a_cur, fft_size)
            grad_a = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
            
            # Softmax chain rule: grad_z = a * (grad_a - a^T grad_a)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj_func, z, jac=grad_func, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-12,
                         'maxfun': max_iter * 200}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  LSE refine done: {elapsed:.0f}s, best={best_score:.10f}")
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, verbose=True):
    """
    Active-set subgradient with adaptive step size.
    Focuses on reducing the top few peaks simultaneously.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  Subgrad start: n={n}, score={best_score:.10f}")
    
    lr = 0.02
    no_improve = 0
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Active set: all positions within 0.1% of max
        threshold = M * (1.0 - 0.001)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Limit active set size for efficiency
        if len(active) > 50:
            # Take the top 50
            top_idx = np.argsort(b)[::-1][:50]
            active = top_idx
        
        # Compute subgradient: g[j] = (1/|S|) * sum_{k in S} 2*a[k-j]
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr = max(lr * 0.9, 1e-6)
                no_improve = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  Subgrad done: {elapsed:.0f}s, best={best_score:.10f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using cubic spline-like approach."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    # Use numpy interp with linear (simple and effective)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_n_lp_search(parent, time_limit, solver, n_values, verbose=True):
    """
    For each n in n_values, run LP alternation + LSE refinement.
    Interpolates parent to each n as initialization.
    """
    best_a = parent.copy()
    best_score = 2.0 * len(parent) * fast_convolve(parent).max()
    t0 = time.monotonic()
    
    if verbose:
        print(f"  Multi-n search: trying n_values={n_values}")
    
    for cand_n in n_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 60:
            break
        
        # Allocate time proportionally to n (larger n gets more time)
        n_frac = cand_n / max(n_values)
        t_budget = min(remaining * 0.7, max(60, 300 * n_frac))
        
        if verbose:
            print(f"  --- n={cand_n}, budget={t_budget:.0f}s ---")
        
        # Initialize from parent
        if len(parent) == cand_n:
            init = parent.copy()
        else:
            init = interpolate_sequence(parent, cand_n)
        
        # Run LP alternation
        t_lp = t_budget * 0.6
        a_lp, sc_lp = lp_game_optimize(init, t_lp, solver, verbose=verbose)
        
        # Run LSE refinement on the best so far
        remaining2 = time_limit - (time.monotonic() - t0)
        t_lse = min(remaining2 * 0.3, t_budget * 0.3)
        if t_lse > 10:
            a_lse, sc_lse = logsumexp_refine(a_lp, t_lse, verbose=verbose)
            if sc_lse < sc_lp:
                a_lp = a_lse
                sc_lp = sc_lse
        
        # Run subgradient polish
        remaining3 = time_limit - (time.monotonic() - t0)
        t_sg = min(remaining3 * 0.15, 30.0)
        if t_sg > 5:
            a_sg, sc_sg = subgradient_refine(a_lp, t_sg, verbose=False)
            if sc_sg < sc_lp:
                a_lp = a_sg
                sc_lp = sc_sg
        
        # Update global best
        full_score = local_eval(a_lp)
        if full_score < best_score:
            best_score = full_score
            best_a = a_lp.copy()
            if verbose:
                print(f"  NEW BEST: n={cand_n}, score={full_score:.10f}")
    
    return best_a, best_score


def peak_equalize_lp(a0, solver, time_limit):
    """
    Iteratively solve LP to equalize the top peaks of the auto-convolution.
    At each step, identify peaks and solve an LP that reduces them.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Find peaks (within 1% of max)
        threshold = M * 0.99
        peak_positions = np.where(b >= threshold)[0]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([int(np.argmax(b))])
        
        # Solve LP: minimize max conv(a, b) with emphasis on peak positions
        # Weight the peak constraints more heavily
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix_sparse(a)
        
        constraints = [cp.sum(b_var) == 1]
        
        # Add weighted constraints: peaks get weight 1, others get small weight
        for k in range(2 * n - 1):
            if k in peak_positions:
                constraints.append((T @ b_var)[k] <= M_var)
            # Don't add constraints for non-peak positions (they have slack)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    # Blend with current a
                    t = 0.3
                    a_new = (1 - t) * a + t * b_val
                    a_new = normalize(a_new)
                    sc = 2.0 * n * fast_convolve(a_new).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        a = a_new
        except Exception:
            pass
    
    return best_a, best_score
