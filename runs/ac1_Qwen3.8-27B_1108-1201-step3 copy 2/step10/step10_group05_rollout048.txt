def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def fft_conv_same(a):

def fft_gradient(w, a):
    """
    Compute grad[i] = sum_j w[i+j] * a[j] for the objective sum_k w[k]*conv(a,a)[k].
    The full gradient is 2*grad (accounting for both factors in the product).
    Uses FFT for O(n log n) computation.
    """

    w_len = len(w)

    # = conv(a_reversed, w)[i + n - 1]
    flipped = a[::-1]
    size = n + w_len - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(flipped, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(A * W, fft_size)[:size]
    return conv_result[n - 1:n - 1 + n]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                ul = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
            else:
                ur = -1e30
            unaff_max = max(ul, ur)

            affected = conv[k_start:k_end + 1]
            # Coefficients for delta: conv[k] changes by delta*(a[k-i] + a[i]) for k != 2i
            # and delta*(2*a[i]) + delta^2 for k = 2i
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Generate candidates
            candidates = [0.0, -ai * 0.5, -ai * 0.8, -ai * 0.95]
            
            # Where lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-14:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.2 < d < ai * 0.5:
                        candidates.append(d)

            # Top affected crossings
            top_k = min(4, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-14:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.2 < d < ai * 0.5:
                            candidates.append(d)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fft_conv_same(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score

def fine_polish(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fft_conv_same(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
               1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                ul = float(conv[:k_start].max())
            else:
                ul = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
            else:
                ur = -1e30
            unaff_max = max(ul, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fft_conv_same(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        pressure[j] += a0[j] * a0[k - j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.15):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(5, n // 8):
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def multiscale_optimize(a0, time_limit, label=""):
    """Optimize on a coarser grid, then refine back to full resolution."""
    n = len(a0)
    if n < 64:
        return a0, 2.0 * n * fft_conv_same(a0).max()

    # Downsample
    coarse_n = max(32, n // 4)
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, coarse_n)
    a_norm = a0 / a0.sum()
    a_coarse = np.interp(new_x, old_x, a_norm)
    a_coarse = np.maximum(a_coarse, 0)
    a_coarse = a_coarse / a_coarse.sum()

    # Optimize coarse
    t_coarse = time_limit * 0.4
    a_c_best, s_c = topk_nesterov_md(a_coarse, t_coarse, label=label + "_coarse", seed=7)
    
    # Upsample back
    a_up = np.interp(old_x, new_x, a_c_best)
    a_up = np.maximum(a_up, 0)
    a_up = a_up / a_up.sum()

    # Blend with original
    a_blend = 0.5 * a_norm + 0.5 * a_up
    a_blend = a_blend / a_blend.sum()

    # Refine
    t_refine = time_limit * 0.6 - (time.monotonic() - (time_limit - time_limit))  # just use remaining
    remaining = time_limit - (time.monotonic() - (time.monotonic() - t_refine))
    # Simpler: just do a quick CD
    a_final, s_final = smart_coord_descent(a_blend, time_limit * 0.5, label=label + "_refine")
    return a_final, s_final

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  *** NEW BEST [{label}]: {score:.7f} (n={len(a)}) ***")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization combining FFT-accelerated methods:
    1. Quick CD polish of parent
    2. Top-K Nesterov mirror descent (main workhorse)
    3. Dual-guided iterative best response
    4. Multi-restart with structured perturbations
    5. Fine polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(99)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = np.ones(n_init) / n_init
        print("Using uniform initialization, n=1024")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Starting: n={len(best_seq)}, score={best_score:.7f}")

    n = len(best_seq)

    # === Phase 1: Quick smart CD on parent (60s) ===
    print(f"\n=== Phase 1: Smart CD (best={best_score:.7f}) ===")
    t1 = min(60.0, (deadline - time.monotonic()) * 0.04)
    a1, s1 = smart_coord_descent(best_seq, t1, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_cd")
    elapsed = time.monotonic() - (deadline - 1785.0)
    print(f"  Phase 1 done ({elapsed:.0f}s): best={best_score:.7f}")

    # === Phase 2: Top-K Nesterov MD (main, ~700s) ===
    print(f"\n=== Phase 2: Top-K Nesterov MD (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t2 = rem * 0.40
    a2, s2 = topk_nesterov_md(best_seq, t2, label="p2", K_init=3, K_max=60, seed=42)
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_topk")
    print(f"  Phase 2 done: best={best_score:.7f}")

    # === Phase 3: Dual-guided iteration (~300s) ===
    print(f"\n=== Phase 3: Dual-Guided (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t3 = rem * 0.20
    a3, s3 = dual_guided_iteration(best_seq, t3, label="p3", seed=1)
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_dual")
    print(f"  Phase 3 done: best={best_score:.7f}")

    # === Phase 4: Another round of Top-K MD with different params (~250s) ===
    print(f"\n=== Phase 4: Top-K MD round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t4 = rem * 0.25
    a4, s4 = topk_nesterov_md(best_seq, t4, label="p4", K_init=10, K_max=100, seed=123)
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_topk2")
    print(f"  Phase 4 done: best={best_score:.7f}")

    # === Phase 5: Multi-restart with perturbation (~remaining) ===
    print(f"\n=== Phase 5: Multi-Restart (best={best_score:.7f}) ===")
    for rnd in range(5):
        rem = deadline - time.monotonic() - 120
        if rem < 60:
            break

        # Perturb
        if rnd % 3 == 0:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.03 + 0.01 * rnd)
        elif rnd % 3 == 1:
            perturbed = random_restructure(best_seq, rng, frac=0.08 + 0.03 * rnd)
        else:
            # Asymmetric perturbation: shift mass
            a_sh = best_seq.copy()
            shift = rng.integers(max(1, n // 32), max(2, n // 16))
            a_sh = np.roll(a_sh, shift)
            a_sh = simplex_project(np.maximum(a_sh, 0))
            perturbed = 0.7 * best_seq + 0.3 * a_sh
            perturbed = perturbed / perturbed.sum()

        # Quick optimize the perturbed version
        t5 = min(rem * 0.3, 80.0)
        a5, s5 = topk_nesterov_md(perturbed, t5, label=f"p5r{rnd}", K_init=5, K_max=40, seed=100 + rnd)
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_r{rnd}")

        # Quick CD on it
        t5b = min(rem * 0.15, 30.0)
        a5b, s5b = smart_coord_descent(a5, t5b, label=f"p5cd{rnd}")
        best_seq, best_score = check_and_update(a5b, best_seq, best_score, f"p5cd_r{rnd}")

    # === Phase 6: CD + perturb micro-loop ===
    print(f"\n=== Phase 6: CD + Perturb Loop (best={best_score:.7f}) ===")
    for rnd in range(4):
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        t6 = min(rem * 0.3, 60.0)
        a6, s6 = smart_coord_descent(best_seq, t6, label=f"p6r{rnd}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_cd{rnd}")

        if rnd < 3 and rem > 40:
            pert = peak_targeted_perturb(best_seq, rng, strength=0.02)
            t6b = min(rem * 0.15, 25.0)
            a6b, s6b = smart_coord_descent(pert, t6b, label=f"p6p{rnd}")
            best_seq, best_score = check_and_update(a6b, best_seq, best_score, f"p6p_{rnd}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t7 = min(rem * 0.5, 100.0)
        a7, s7 = fine_polish(best_seq, t7, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_fine")

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a7b, s7b = smart_coord_descent(best_seq, min(rem * 0.6, 40.0), label="p7b")
        best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b_cd")

    # === Try different lengths ===
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 10
    alt_lengths = []
    for alt in [768, 1536, 2048]:
        if alt != n and rem > 60:
            alt_lengths.append(alt)
            rem -= 60

    for n_alt in alt_lengths[:2]:
        rem = deadline - time.monotonic() - 10
        if rem < 30:
            break
        # Interpolate
        old_x = np.linspace(0, 1, len(best_seq))
        new_x = np.linspace(0, 1, n_alt)
        a_interp = np.interp(new_x, old_x, best_seq / best_seq.sum())
        a_interp = np.maximum(a_interp, 0)
        a_interp = a_interp / a_interp.sum()

        t8 = min(rem * 0.6, 50.0)
        a8, s8 = topk_nesterov_md(a_interp, t8, label=f"p8_n{n_alt}", K_init=5, K_max=40, seed=200)
        a8b, s8b = smart_coord_descent(a8, min(rem * 0.2, 20.0), label=f"p8cd_n{n_alt}")
        best_seq, best_score = check_and_update(a8b, best_seq, best_score, f"p8_n{n_alt}")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
