Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_T_sub(a_fixed, peak_indices):
    """Build submatrix of convolution matrix for given peak indices."""
    n = len(a_fixed)
    K = len(peak_indices)
    T = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        # T[i, j] = a_fixed[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # For j in [js, je], k-j is in [k-je, k-js] which is in [0, n-1]
        # T[i, j] = a_fixed[k-j]
        # When j=js: k-js = k - max(0, k-n+1)
        # When j=je: k-je = k - min(n-1, k)
        # So k-j ranges from k-je to k-js as j goes from je to js
        # i.e., as j increases from js to je, k-j decreases from k-js to k-je
        T[i, js:je+1] = a_fixed[k-je:k-js+1][::-1]
    return T


def solve_lp_reduced(a_fixed, peak_indices, solver, max_iter=10000):
    """Solve LP with only the given peak constraints active."""
    n = len(a_fixed)
    K = len(peak_indices)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T_sub = build_T_sub(a_fixed, peak_indices)
    constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, max_iter=max_iter, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None, None
            b_val = b_val / s
            return b_val, float(M_var.value), prob
    except Exception:
        pass
    
    # Try alternative solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None, None
                b_val = b_val / s
                return b_val, float(M_var.value), prob
        except Exception:
            continue
    return None, None, None


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000,
                          K_peaks=300, stall_patience=15, line_search_steps=8):

    Nesterov LP alternation with:

    - Exact line search for step size
    - Dual-guided active set
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        top_k = np.argsort(c_y)[::-1][:K_peaks]

        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        
        # Line search for step size
        best_t = 0.0
        best_trial_score = best_score
        
        # Try several step sizes

        for t in ts:
            a_trial = (1.0 - t) * a + t * b_new
            a_trial = np.maximum(a_trial, 0)

            a = np.maximum(a, 0)
            a = a / a.sum()

        if sy < 1e-15:

def gradient_peak_elimination(a0, time_limit, max_steps=50000, K_peaks=10):
    """
    Direct gradient-based peak elimination with:
    - Exact gradient (2*a[k-j])
    - Projected gradient with line search
    - Momentum
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.005
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        
        # Active set: top K peaks
        top_k = np.argsort(c)[::-1][:K_peaks]
        
        # Compute gradient: g[j] = sum_{k in active} 2*a[k-j]
        g = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # g[j] += 2 * a[k-j] for j in [js, je]
            # a[k-j] for j=js to je is a[k-js] to a[k-je], reversed
            g[js:je+1] += 2.0 * a[k-je:k-js+1][::-1]
        
        # Project gradient: subtract mean to handle sum constraint
        g = g - g.mean()
        
        # Momentum
        momentum = beta * momentum + (1.0 - beta) * g
        
        # Try different step sizes
        best_lr = 0
        best_trial_score = best_score
        for lr_trial in [lr * 0.5, lr, lr * 1.5, lr * 2]:
            a_trial = a - lr_trial * momentum
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_lr = lr_trial
        
        if best_lr > 0:
            a = a - best_lr * momentum
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200:
                # Restart momentum with different K
                K_peaks = max(3, K_peaks // 2)
                momentum = np.zeros(n)
                lr = 0.01
                no_improve = 0
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradPE {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f} K={K_peaks}")
    
    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    z = np.log(a0 / a0.sum())
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]
    
    n_stages = len(T_stages)
    iters_per = max(10, maxiter_per_stage // n_stages)
    
    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time.limit - (time.monotonic() - t0) if hasattr(time, 'limit') else time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_loc(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse
        
        def jac_loc(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                grad_a[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.002
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        top_k = np.argsort(c)[::-1][:K_peaks]

        g = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * a[k - je:k - js + 1][::-1]

        g = g - g.mean()
        momentum = beta * momentum + (1.0 - beta) * g

        best_lr_t = 0.0
        best_trial = best_score
        for lr_t in [lr * 0.5, lr, lr * 1.5]:
            a_trial = a - lr_t * momentum
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial - 1e-14:
                best_trial = score
                best_lr_t = lr_t

        if best_lr_t > 0:
            a = a - best_lr_t * momentum
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_trial
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
            if no_improve > 300:
                momentum = np.zeros(n)
                lr = 0.005
                no_improve = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradPE {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def lbgfs_lse(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    z = np.log(a0 / a0.sum())
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per = max(10, maxiter_per_stage // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 3:
            break

        def func_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(b_in - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse

        def jac_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(b_in - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                grad_a[js:je + 1] += w[k] * a_in[k - je:k - js + 1][::-1]
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase: NLP-CG (main) -> GradPE -> LP2 -> L-BFGS -> Diversity -> Polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_local(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 1: Main NLP-CG
    print(f"\n=== P1: NLP-CG n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 600.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, _ = nesterov_lp_cg(best_seq, t1, solver=solver,
                                max_iters=80000, max_active=300,
                                initial_k=120, stall_patience=12)
        try_update(a1, "nlp_cg_1")

    # Phase 2: Gradient PE
    print(f"\n=== P2: GradPE, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 250.0)
    if t2 > 10:
        a2, _ = gradient_peak_elim(best_seq, t2, max_steps=40000, K_peaks=5)
        try_update(a2, "grad_pe_1")

    # Phase 3: Second LP pass
    print(f"\n=== P3: NLP-CG pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 300.0)
    if t3 > 30:
        a3, _ = nesterov_lp_cg(best_seq, t3, solver=solver,
                                max_iters=50000, max_active=250,
                                initial_k=100, stall_patience=10)
        try_update(a3, "nlp_cg_2")

    # Phase 4: Gradient PE again
    print(f"\n=== P4: GradPE 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 8:
        a4, _ = gradient_peak_elim(best_seq, t4, max_steps=25000, K_peaks=3)
        try_update(a4, "grad_pe_2")

    # Phase 5: L-BFGS
    print(f"\n=== P5: L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 150.0)
    if t5 > 10:
        a5 = lbgfs_lse(best_seq, t5,
                       T_stages=[5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                       maxiter_per_stage=5000)
        try_update(a5, "lbgfs")

    # Phase 6: Diversity
    print(f"\n=== P6: Diversity, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 50.0)
        for pi in range(3):
            if time.monotonic() > deadline - 15:
                break
            if t_each < 10:
                break
            x = np.linspace(0, 1, n_cur)
            if pi == 0:
                pert = 0.015 * np.cos(2 * np.pi * 3.7 * x)
            elif pi == 1:
                pert = 0.012 * rng.standard_normal(n_cur)
            else:
                pert = 0.01 * np.sin(2 * np.pi * 5.3 * x)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_res, _ = nesterov_lp_cg(a_pert, min(t_each * 0.7, 35.0), solver=solver,
                                       max_iters=10000, max_active=200,
                                       initial_k=80, stall_patience=6)
            try_update(a_res, f"div_{pi}_lp")
            rem_sub = deadline - time.monotonic()
            if rem_sub > 10:
                a_res2, _ = gradient_peak_elim(a_res, min(rem_sub * 0.4, 20.0),
                                                max_steps=10000, K_peaks=5)
                try_update(a_res2, f"div_{pi}_gp")

    # Final polish
    print(f"\n=== Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 8:
        a_fin = lbgfs_lse(best_seq, min(rem * 0.5, 25.0),
                          T_stages=[0.3, 0.15, 0.08, 0.04, 0.02, 0.01],
                          maxiter_per_stage=2000)
        try_update(a_fin, "final")

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
