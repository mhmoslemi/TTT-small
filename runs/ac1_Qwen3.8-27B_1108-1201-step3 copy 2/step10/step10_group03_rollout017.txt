```python
        grad_a[m_start:m_end+1] += 2.0 * w[k] * a[k - np.arange(m_start, m_end + 1)]
    
    return grad_a
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LogSumExp L-BFGS with FFT-based gradient and very low temperature
2. Coordinate descent with multi-candidate line search
3. Linearized peak-flattening LP for intermediate refinement
4. Multi-n exploration with interpolation and re-optimization
5. Stochastic perturbations to escape local minima
Key insight: asymmetry in the sequence is crucial for score < 2.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def compute_grad_a_ffT(w, a):
    """Compute grad_a[m] = 2 * sum_k w[k] * a[k-m] using FFT cross-correlation."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    corr = irfft(W * np.conj(A), L)
    return 2.0 * corr[:n]


def lse_lbgfs_optimize(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        max_iter = max(100, int(remaining * 40))
        time_per_stage = remaining / max(1, len(T_stages) - T_stages.index(T)) if T in T_stages else remaining

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            grad_a = compute_grad_a_ffT(w, a_cur)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if (time.monotonic() - t0) % 60 < time_per_stage:
            print(f"    LSE-LBFGS T={T}: score={best_score:.10f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_passes=5, n_candidates=7):
    """Coordinate descent with multi-candidate evaluation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = 0
        for m in order:
            if time.monotonic() - t0 > time_limit:
                break
            t0_val = a[m]
            if t0_val < 1e-10:
                candidates = [0.5 / n, 1.0 / n, 2.0 / n]
            else:
                candidates = [t0_val * f for f in [0.5, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5]]
                # Also try absolute perturbations
                candidates.append(max(0, t0_val - 0.005))
                candidates.append(t0_val + 0.005)

            best_t = t0_val
            best_sc = best_score
            for t in candidates:
                if t < 0:
                    continue
                remaining_sum = 1.0 - t
                if remaining_sum < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[m] = t
                others_sum = a_trial.sum() - t
                if others_sum < 1e-10:
                    continue
                a_trial[np.arange(n) != m] *= remaining_sum / others_sum
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_sc:
                    best_sc = sc
                    best_t = t

            if best_t != t0_val:
                a[m] = best_t
                remaining_sum = 1.0 - best_t
                others_sum = a.sum() - best_t
                if others_sum > 1e-10:
                    a[np.arange(n) != m] *= remaining_sum / others_sum
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved += 1

        print(f"    CoordDescent pass {p+1}: score={best_score:.10f}, improved={improved} ({time.monotonic()-t0:.0f}s)")
        if improved == 0:
            break

    return best_a, best_score


def peak_flattening_lp(a0, solver, time_limit, n_steps=200):
    """LP-based peak flattening."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.95
        active_k = np.where(b >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b))])

        m_active = len(active_k)
        G = np.zeros((m_active, n))
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            G[i, idx] = 2.0 * a[k - idx]

        delta_var = cp.Variable(n)
        eps_var = cp.Variable()
        obj = cp.Minimize(cp.sum(G @ delta_var) / m_active + 10.0 * eps_var)
        constraints = [
            cp.sum(delta_var) == 0,
            a + delta_var >= 0,
            G @ delta_var <= m_active * eps_var,
        ]

        near_inactive = np.where((b < threshold) & (b >= M * 0.7))[0]
        if len(near_inactive) > 0 and len(near_inactive) < 300:
            G_2 = np.zeros((len(near_inactive), n))
            for i, k in enumerate(near_inactive):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                G_2[i, idx] = 2.0 * a[k - idx]
            constraints.append(G_2 @ delta_var <= 0)

        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                delta_val = np.asarray(delta_var.value, dtype=np.float64)
                for step_frac in [0.7, 0.4, 0.2]:
                    a_trial = a + step_frac * delta_val
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        break
        except Exception:
            pass

    return best_a, best_score
