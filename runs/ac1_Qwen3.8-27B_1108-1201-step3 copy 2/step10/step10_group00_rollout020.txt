Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration (main workhorse)
2. Active-set QCQP polish (new: exact optimization on the active constraint set)
3. Low-temperature LogSumExp L-BFGS for final numerical precision

Key: The problem is convex (max of PSD quadratics over simplex), so global optimality

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def solve_active_qcqp(a, solver, tol=1e-6, max_active=50):
    """
    Solve the active-set QCQP:
    min t subject to a^T Q_k a <= t for k in active set, sum(a)=1, a>=0
    where active set = positions where conv is within tol of max.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Find active positions (within tol of max)
    threshold = max_b * (1.0 - tol)
    active = np.where(b >= threshold)[0]
    
    # Limit to top max_active positions
    if len(active) > max_active:
        # Keep the top max_active by value
        active = active[np.argsort(b[active])[-max_active:]]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    print(f"    Active set size: {len(active)}, values: {b[active][:5]}...")
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        # conv(a,a)[k] = sum_{i+j=k} a[i]*a[j]
        # Build the quadratic form
        # For k, the pairs are (j, k-j) for j = max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        indices = np.arange(js, je + 1)
        partners = k - indices
        
        # Build the matrix Q_k (sparse)
        # Q_k[i,j] = 1/2 if i+j=k and i!=j, Q_k[i,i] = 1 if 2i=k
        # a^T Q_k a = sum_{i+j=k} a[i]*a[j] (counting each pair once)
        
        # For CVXPY, we can express this as:
        # sum_j a[j] * a[k-j] = a[indices]^T * a[partners]
        # But this is not a standard quadratic form in CVXPY unless we use quad_form
        
        # Let's build the full matrix (might be large but n=1200 is OK)
        Q = np.zeros((n, n))
        for idx, partner in zip(indices, partners):
            if idx == partner:
                Q[idx, idx] = 1.0
            else:
                Q[idx, partner] = 0.5
                Q[partner, idx] = 0.5
        
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                a_val = a_val / s_sum
                return a_val
        except Exception as e:
            print(f"    QCQP solver {s} failed: {e}")
            continue
    
    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Structured perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp:.1f}ms")

        # If LP is too slow, break early
        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 3.0:
            print(f"    LP too slow, stopping after {it+1} iters")
            break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):

    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_inner):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_inner, n, stage_T)
            return obj

        def jac(z_inner):

                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, float(max(0, M_var.value))


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=30):
    """
    Solve the active-set QCQP:
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]

    if len(active) == 0:
        active = np.array([np.argmax(b)])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception as e:
            print(f"    QCQP solver {s} failed: {e}")
            continue

    return None, None


def solve_single_qp(a, k_active, solver):
    """Solve min a^T Q_k a s.t. sum(a)=1, a>=0 for a single active position."""
    n = len(a)
    Q = build_q_sparse(int(k_active), n)
    a_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.quad_form(a_var, Q, assume_PSD=True)),
                      [cp.sum(a_var) == 1])
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                return a_val / s_sum
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 5.0:
            print(f"    LP too slow ({np.mean(lp_times[-100:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating with Nesterov acceleration, active-set QCQP polish,
    low-temp L-BFGS, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.42, 680.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Active-set QCQP polish ===
    print(f"\n=== Phase 2: Active QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.12, 180.0)
    if t_qcqp > 15:
        q_start = time.monotonic()
        a_qcqp, t_val = solve_active_qcqp(best_seq, solver, rel_tol=1e-4, max_active=20)
        update_best(a_qcqp, "qcqp_wide")
        # Try tighter active set
        a_qcqp2, t_val2 = solve_active_qcqp(best_seq, solver, rel_tol=1e-6, max_active=8)
        update_best(a_qcqp2, "qcqp_tight")
        # Try single QP on argmax
        b_cur = fast_convolve(best_seq)
        k_max = int(np.argmax(b_cur))
        a_qp = solve_single_qp(best_seq, k_max, solver)
        update_best(a_qp, "single_qp")
        print(f"  QCQP time: {time.monotonic() - q_start:.1f}s")

    # === Phase 3: LP alternating round 2 (from polished) ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: Low-temp L-BFGS ===
    print(f"\n=== Phase 4: L-BFGS Low-Temp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.10, 150.0)
    if t_lbgfs > 20:
        T_stages = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb = lbgfs_refine(best_seq, t_lbgfs, T_stages, maxiter_per_stage=8000)
        update_best(a_lb, "lbgfs_lowtemp")

    # === Phase 5: QCQP polish again ===
    print(f"\n=== Phase 5: QCQP Polish Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_qcqp3, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-6, max_active=5)
        update_best(a_qcqp3, "qcqp_final")

    # === Phase 6: Try n=1199 ===
    print(f"\n=== Phase 6: Try n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1199 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_refine(init_1199, t_1199 * 0.7, solver, n_max_iters=200000)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1199 and rem > 10:
            a_q, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-5, max_active=10)
            update_best(a_q, "qcqp_n1199")

    # === Phase 7: Try n=1201 ===
    print(f"\n=== Phase 7: Try n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1201 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Re-interpolate from the n=1200 solution
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, _ = nesterov_lp_refine(init_1201, t_1201 * 0.7, solver, n_max_iters=200000)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1201 and rem > 10:
            a_q, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-5, max_active=10)
            update_best(a_q, "qcqp_n1201")

    # === Phase 8: Final multiplicative subgradient + QCQP ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.3, 20.0), n_steps=50000)
        update_best(a_ms, "final_mult")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_qf, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-7, max_active=3)
            update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
