    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Fast cross-correlation/convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Convert log-space to probability simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def compute_score(a):
    """Compute the objective: 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


_lp_solver = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response optimization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step with adaptive momentum
        alpha = 2.0 / ((it + 2.0) ** 0.5 + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Also track a weighted average (like in Nesterov's method)
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 1000 == 0 and it > 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                rng = np.random.default_rng(it * 13 + 37)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=15):
    """L-BFGS on LogSumExp-smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_func(z_in, T):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700.0
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * lse

    def lse_grad(z_in, T):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700.0
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad wrt a: 2 * sum_k w[k] * a[k-j] for each j
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        try:
            res = minimize(
                fun=lambda z_in: lse_func(z_in, stage_T),
                x0=z,
                jac=lambda z_in: lse_grad(z_in, stage_T),
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': 50000, 'disp': False}
            )
            z = res.x
            sc = compute_score(softmax_from_z(z))
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): score={sc:.8f}")
        except Exception as e:
            print(f"    LSE stage {si+1} failed: {e}")

    return softmax_from_z(z)


def mirror_descent(a0, time_limit, n_active_frac=0.05):
    """Mirror descent with subgradient and backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.5

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        alpha = lr
        improved = False
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                lr = min(alpha * 1.05, 2.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr = max(lr * 0.9, 0.01)
            if step % 5000 == 0 and step > 10000:
                rng = np.random.default_rng(step)
                noise = rng.random(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                lr = 0.5

    return best_a, best_score


def spectral_init(n, rng):
    """Initialize using spectral flatness considerations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
    elif choice == 2:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x * x)
    elif choice == 3:
        # Sinc-squared
        x = np.linspace(-3, 3, n)
        x = np.maximum(np.abs(x), 1e-10)
        a = np.sinc(x / np.pi) ** 2
    elif choice == 4:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(4 * np.pi * x) + 0.05 * np.cos(8 * np.pi * x)
    else:
        # Random smooth
        x = np.linspace(0, 1, n)
        a = 1.0 + rng.random() * 0.1 * np.sin(2 * np.pi * rng.random() * 5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, time_budget, solver):
    """Run multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s
    best_score = compute_score(best_a)
    t0 = time.monotonic()

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    NEW [{label}] n={n}: {sc:.8f} ({elapsed:.0f}s)")

    # Phase 1: Mirror descent (fast, good for initial improvement)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.2, 60.0)
    if t1 > 5:
        a_md, _ = mirror_descent(best_a, t1)
        try_update(a_md, "MD")

    # Phase 2: LP Nesterov (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.5, 300.0)
    if t2 > 20:
        a_lp, _ = nesterov_lp_optimize(best_a, t2, solver=solver)
        try_update(a_lp, "LP")

    # Phase 3: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.2, 120.0)
    if t3 > 10:
        a_lse = logsumexp_lbgfs(best_a, t3, T_start=20.0, T_end=0.02, n_stages=12)
        try_update(a_lse, "LSE")

    # Phase 4: Final LP polish
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.1, 60.0)
    if t4 > 5:
        a_lp2, _ = nesterov_lp_optimize(best_a, t4, solver=solver)
        try_update(a_lp2, "LP2")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds the best non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.8f} (n={len(a)})")

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    # Also try uniform sequences as baseline
    for n_try in [960, 1200, 1500]:
        u = np.ones(n_try) / n_try
        sc_u = evaluate_safe(u)
        print(f"Uniform n={n_try}: score={sc_u:.8f}")
        try_update(u, f"uniform_{n_try}")

    # Determine base length
    n_base = len(best_seq) if best_seq is not None else 1200

    # Phase 1: Optimize at base length (main effort)
    print(f"\n=== Phase 1: Optimize n={n_base} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s, current best: {best_score:.8f}")
        a_out, sc_out = optimize_at_length(best_seq, t1, solver)
        try_update(a_out, f"opt_n{n_base}")

    # Phase 2: Try nearby lengths
    print(f"\n=== Phase 2: Length exploration (best={best_score:.8f}) ===")
    lengths_to_try = []
    for mult in [0.8, 0.9, 1.1, 1.25, 1.5]:
        ln = int(n_base * mult)
        if 200 <= ln <= 5000 and ln != n_base:
            lengths_to_try.append(ln)
    lengths_to_try = sorted(set(lengths_to_try))

    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.15, 150.0)
        if t_alloc < 30:
            break

        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        init = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init)
        try_update(init, f"interp_n{ln}")

        a_out, _ = optimize_at_length(init, t_alloc, solver)
        try_update(a_out, f"opt_n{ln}")

    # Phase 3: Try diverse initializations at base length
    print(f"\n=== Phase 3: Diverse restarts (best={best_score:.8f}) ===")
    for init_type in range(4):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.12, 100.0)
        if t_alloc < 20:
            break

        a_init = spectral_init(n_base, rng)
        sc_init = compute_score(a_init)
        print(f"  Restart {init_type}: init_score={sc_init:.8f}")
        a_out, _ = optimize_at_length(a_init, t_alloc, solver)
        try_update(a_out, f"restart_{init_type}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_optimize(best_seq, t_lp, solver=solver)
            try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lse = min(rem * 0.3, 80.0)
        if t_lse > 10:
            a_lse = logsumexp_lbgfs(best_seq, t_lse, T_start=5.0, T_end=0.01, n_stages=10)
            try_update(a_lse, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_md = min(rem * 0.3, 40.0)
        if t_md > 5:
            a_md, _ = mirror_descent(best_seq, t_md, n_active_frac=0.02)
            try_update(a_md, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL VERIFIED SCORE: {best_score:.8f} (n={len(best_seq)})")
        else:
            print("WARNING: Final sequence failed evaluation, falling back to uniform")
            best_seq = np.ones(1200) / 1200
            best_score = evaluate_safe(best_seq)

    print(f"FINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
