def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):

    s = exp_z.sum()

    return exp_z / s

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.

    g_m = sum_k y_star[k] * (a[k-m] + a[m-k]) with bounds."""

    # Note: a[m-k] is valid when 0 <= m-k < n, i.e., k < m, and a[m-k] means index m-k in a.

    # g[m] = sum_k y_star[k] * a[k-m] * [0<=k-m<n] + sum_k y_star[k] * a[m-k] * [0<=m-k<n]

    # Part 1: sum_k y_star[k] * a[k-m] where k goes from m to min(m+n-1, size-1)
    # Let i = k-m: sum_i y_star[i+m] * a[i] for i=0..n-1 (y_star out of range = 0)
    # This is the inner product of a with y_star shifted by m (reversed appropriately)

    # Part 2: sum_k y_star[k] * a[m-k] where k goes from max(0, m-n+1) to m
    # Let i = m-k: sum_i y_star[m-i] * a[i] for i=0..n-1 (y_star out of range = 0)
    # This is the inner product of a with y_star reversed and shifted

    Y = rfft(y_star, fft_size)

    # In FFT: ifft(fft(a) * conj(fft(y_star))) gives correlation

        wk = y_star[k]

def lse_objective_and_grad(z, T_val, t0, time_limit):
    """Compute LogSumExp objective and gradient."""
    if time.monotonic() - t0 > time_limit:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient
    w = exp_Tb / exp_Tb.sum()
    g = 2.0 * compute_dual_subgradient(a, w)
    # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(g, a)
    grad_z = a * (g - S)

    return obj, grad_z

def mirror_descent_lp(a0, time_limit, solver=None):
    """Mirror descent using LP dual subgradients with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP to get best response and dual
        b_resp, M_val, y_dual, z_dual = solve_lp_best_response(a, solver)
        if b_resp is None:
            # Fallback: use active set subgradient
            b = fast_convolve(a)
            max_b = b.max()
            threshold = max_b - 0.05 * max_b
            active = np.where(b >= threshold)[0]
            y_dual = np.zeros(len(b))
            y_dual[active] = 1.0 / len(active)

        # Compute subgradient from dual
        g = compute_dual_subgradient(a, y_dual)

        # Mirror descent step
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.05, 2.0)
        else:
            no_improve += 1
            lr = max(lr * 0.95, 0.01)
            if no_improve >= 50:
                # Also update towards the LP best response
                a = 0.7 * a + 0.3 * b_resp
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.5

        if step % 200 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-LP step={step} ({elapsed:.0f}s): score={best_score:.6f} M_lp={M_val:.6f}")

    return best_a, best_score


def game_iteration(a0, time_limit, solver=None):
    """Game-theoretic iteration: alternate between best response and averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Aurora-style: maintain running average
    a_avg = a.copy()
    b_curr = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()

    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # b = best_response(a_avg)
        b_resp, M_val, y_dual, z_dual = solve_lp_best_response(a_avg, solver)
        if b_resp is None:
            b_resp = a_avg.copy()

        # a = best_response(b_resp)
        a_resp, M_val2, _, _ = solve_lp_best_response(b_resp, solver)
        if a_resp is None:
            a_resp = b_resp.copy()

        # Update running average (Aurora)
        alpha = 2.0 / (t + 2.0)
        a_avg = (1.0 - alpha) * a_avg + alpha * a_resp
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa > 1e-15:
            a_avg = a_avg / sa

        # Track best
        score_avg = 2.0 * n * fast_convolve(a_avg).max()
        score_resp = 2.0 * n * fast_convolve(a_resp).max()

        if score_avg < best_score - 1e-12:
            best_score = score_avg
            best_a = a_avg.copy()
        if score_resp < best_score - 1e-12:
            best_score = score_resp
            best_a = a_resp.copy()

        if t % 100 == 0 and t > 0:
            elapsed = time.monotonic() - t0
            print(f"    GameIter t={t} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def local_polish(a0, time_limit, rng, n_steps=100000):
    """Local polish: try small perturbations targeting the max autocorrelation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Try moving mass from high contributors to random positions
        n_tries = 5
        improved = False
        for _ in range(n_tries):
            # Pick source weighted by contribution
            src_local = rng.choice(len(idx), p=contrib / total_contrib)
            src = idx[src_local]

            # Pick destination: prefer positions that don't contribute to top peaks
            top_peaks = np.argsort(b)[::-1][:5]
            in_peak = np.zeros(n, dtype=bool)
            for kp in top_peaks:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                in_peak[js2:je2+1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.array([j for j in range(n) if j != src])
            dst = candidates[rng.integers(len(candidates))]

            frac = rng.uniform(0.005, 0.05)
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Random small perturbation
            k = rng.integers(max(2, n // 50), max(3, n // 10))
            idx_p = rng.choice(n, k, replace=False)
            a_trial = a.copy()
            a_trial[idx_p] += rng.normal(0, 0.01, k)
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

    return best_a, best_score


def lbgfs_manual(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=12):
    """Manual L-BFGS-like optimization using LogSumExp, no closures."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Simple heavy-ball gradient descent in log-space
        z_prev = z.copy()
        momentum = 0.7
        lr = 1.0

        n_iters = max(10, int(100 * remaining / max(time_limit, 1)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_objective_and_grad(z, stage_T, t0, time_limit)
            if not np.isfinite(obj) or not np.isfinite(grad).all():
                break
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            v = momentum * (z - z_prev) - lr * grad / grad_norm
            z_new = z + v
            # Check if objective improved
            obj_new, _ = lse_objective_and_grad(z_new, stage_T, t0, time_limit)
            if np.isfinite(obj_new) and obj_new < obj:
                z = z_new
                lr = min(lr * 1.1, 5.0)
            else:
                lr = max(lr * 0.7, 0.01)
            z_prev = z_new if np.isfinite(obj_new) and obj_new <= obj else z

    return softmax_from_z(z)


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src_label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{src_label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase A: Game iteration (LP-based)
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.40, 400.0)
    if t_a > 30:
        print(f"  Phase A: Game iteration ({t_a:.0f}s)")
        a_out, _ = game_iteration(best_a, t_a, solver=solver)
        try_update(a_out, f"game_{label}")

    # Phase B: Mirror descent with LP dual
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.30, 300.0)
    if t_b > 15:
        print(f"  Phase B: MD-LP ({t_b:.0f}s)")
        a_out, _ = mirror_descent_lp(best_a, t_b, solver=solver)
        try_update(a_out, f"mdlp_{label}")

    # Phase C: LogSumExp
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.20, 150.0)
    if t_c > 10:
        print(f"  Phase C: LogSumExp ({t_c:.0f}s)")
        a_out = lbgfs_manual(best_a, t_c, T_start=20.0, T_end=0.02, n_stages=12)
        try_update(a_out, f"lse_{label}")

    # Phase D: Local polish
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.10, 60.0)
    if t_d > 5:
        print(f"  Phase D: Local polish ({t_d:.0f}s)")
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        a_out, _ = local_polish(best_a, t_d, rng, n_steps=50000)
        try_update(a_out, f"polish_{label}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function that minimizes the autocorrelation objective."""
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1000
        best_seq = structured_init(n_base, rng)
        best_score = 2.0 * n_base * fast_convolve(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Optimize at base length
    print(f"\n=== Phase 1: Optimize at n={n_base} (best={best_score:.6f}) ===")
    rem = _global_deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t1, solver, label=f"n{n_base}")
        try_update(a_out, f"opt_n{n_base}")

    # Phase 2: Try other lengths
    print(f"\n=== Phase 2: Other lengths (best={best_score:.6f}) ===")
    candidate_lengths = []
    for mult in [0.8, 1.25, 1.5, 2.0, 0.6]:
        ln = int(n_base * mult)
        if 200 <= ln <= 4000 and ln != n_base:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))[:4]

    for ln in candidate_lengths:
        rem = _global_deadline - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.25, 300.0)
        if t_alloc < 60:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = 2.0 * ln * fast_convolve(init_long).max()
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"  Interpolation improved: {sc_init:.6f}")

        a_out, _ = optimize_at_length(init_long, t_alloc, solver, label=f"n{ln}")
        try_update(a_out, f"opt_long_n{ln}")

    # Phase 3: Final polish on the best
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = _global_deadline - time.monotonic()
    if rem > 60:
        # More LP game iterations
        t_lp = min(rem * 0.4, 200.0)
        if t_lp > 20:
            a_fin, _ = game_iteration(best_seq, t_lp, solver=solver)
            try_update(a_fin, "final_game")

    rem = _global_deadline - time.monotonic()
    if rem > 30:
        # MD-LP
        t_mdlp = min(rem * 0.3, 100.0)
        if t_mdlp > 10:
            a_fin2, _ = mirror_descent_lp(best_seq, t_mdlp, solver=solver)
            try_update(a_fin2, "final_mdlp")

    rem = _global_deadline - time.monotonic()
    if rem > 20:
        # LogSumExp fine-tune
        t_lse = min(rem * 0.25, 60.0)
        if t_lse > 5:
            a_fin3 = lbgfs_manual(best_seq, t_lse, T_start=5.0, T_end=0.01, n_stages=10)
            try_update(a_fin3, "final_lse")

    rem = _global_deadline - time.monotonic()
    if rem > 15:
        # Local search
        t_local = min(rem * 0.5, 40.0)
        if t_local > 5:
            rng2 = np.random.default_rng(12345)
            a_fin4, _ = local_polish(best_seq, t_local, rng2, n_steps=30000)
            try_update(a_fin4, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
