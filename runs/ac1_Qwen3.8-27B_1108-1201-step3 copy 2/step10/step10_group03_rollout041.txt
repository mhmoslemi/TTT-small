Actually, one more thought. Let me reconsider the "dual-guided" approach.

When we solve the LP "given a, find b minimizing max conv(a,b)", the optimal solution b* has the property that:
- conv(a, b*)[k] = M* for k in S (the active set)
- conv(a, b*)[k] < M* for k not in S

The dual variables λ_k (k in S) satisfy:
- sum_{k in S} λ_k * a[k-j] = c (constant) for all j in support of b*

This means that b* is "orthogonal" (in a weighted sense) to the "fluctuations" of a.

Now, if we want to update a to reduce the max of conv(a,a), we should move a in a direction that:
1. Reduces conv(a,a)[k] for k in S.
2. Doesn't increase conv(a,a)[k] for k not in S (too much).

The gradient of conv(a,a)[k] w.r.t. a[j] is a[k-j] (for valid indices). So the gradient of the "active set objective" is:
g[j] = sum_{k in S} λ_k * a[k-j]

This is exactly the "weighted sum" that's constant on the support of b*! So the gradient is "flat" on the support of b*, meaning the LP is at a "saddle point" in that direction.

This is consistent with the minimax interpretation: at the LP fixed point, a is a "best response" to b, and b is a "best response" to a. The gradient of the max constraint is "balanced" in the sense that it's constant on the support.

To escape this saddle point, we need to move in a direction that's NOT captured by the active set. This is where the smooth optimization (LogSumExp or power mean) comes in: it considers ALL convolution entries, not just the active ones, and can find a direction that reduces the overall "height" of the convolution.

So the algorithm should be:
1. LP alternation to get close to the fixed point.
2. Smooth optimization to escape the saddle point and find a better solution.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiple structured initializations (polynomial densities, parent)
2. Bidirectional LP alternation with Polyak averaging and adaptive restarts
3. Power-mean smooth optimization with L-BFGS (log-parametrization)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return a * a
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_a(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver] + [s for s in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL] if s != solver]
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_alternation(a0, time_limit, solver, max_iters=500000):
    """Bidirectional LP alternation with Polyak-Ruppert averaging."""
    n = len(a0)
    a = normalize_a(a0)
    b = a.copy()
    # Running averages (Polyak-Ruppert)
    a_avg = a.copy()
    b_avg = b.copy()
    count = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given current a_avg
        b_new, M1 = solve_lp_best_response(a_avg, solver)
        if b_new is not None:
            count += 1
            b_avg = b_avg + (b_new - b_avg) / (count + 1)
            b = b_avg.copy()

        # Step 2: Best a given current b_avg
        a_new, M2 = solve_lp_best_response(b_avg, solver)
        if a_new is not None:
            count += 1
            a_avg = a_avg + (a_new - a_avg) / (count + 1)
            a = a_avg.copy()

        # Track best among a, b, a_avg, b_avg
        for cand in [a, b, a_avg, b_avg]:
            b_conv = fast_convolve(cand)
            sc = 2.0 * n * b_conv.max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        # Stall detection and restart
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Perturb the running averages
                noise = 0.03 / math.sqrt(stall_count)
                a_avg = a_avg + rng.standard_normal(n) * noise
                a_avg = normalize_a(a_avg)
                b_avg = b_avg + rng.standard_normal(n) * noise
                b_avg = normalize_a(b_avg)
                stall_count = 0

            if (it + 1) % 5000 == 0:
                el = time.monotonic() - t0
                print(f"    LP alt iter={it+1} ({el:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def power_mean_lbgfs(a0, time_limit, p_stages=None):
    """L-BFGS on power-mean smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize_a(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if p_stages is None:
        p_stages = [8, 16, 32, 64, 128, 256, 512, 1024]

    for p in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 10))

        def pm_obj(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_pos = np.maximum(b, 1e-30)
            log_sum = np.log(np.sum(b_pos ** p))
            val = np.exp(log_sum / p)
            return 2.0 * n * val

        def pm_grad(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_pos = np.maximum(b, 1e-30)
            weights = b_pos ** (p - 1)
            w_sum = weights.sum()
            if w_sum < 1e-30:
                return np.zeros(n)
            weights = weights / w_sum
            # Gradient of sum_k w[k]*conv(k) w.r.t. a[j] = 2 * sum_k w[k]*a[k-j]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if weights[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * weights[k] * a_cur[k - idx]
            # Power mean scaling factor
            b_pos_p = b_pos ** p
            pm_val = b_pos_p.sum() ** (1.0 / p)
            scale = (p - 1) / (p * pm_val) if pm_val > 0 else 1.0
            grad_a *= scale
            # Chain rule for log-parametrization
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                pm_obj, z, jac=pm_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score

    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Use all entries above 95% of max as active
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Weighted subgradient (weight by how close to max)
        w = (b[active] - threshold) / (M - threshold + 1e-15)
        w = w / w.sum()

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def fourier_refine(a0, time_limit, n_freq=32):
    """Refine in Fourier domain using Nelder-Mead."""
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-13, 'fatol': 1e-13, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, kind, rng=None):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = np.linspace(0.01, 1.0, n)
    elif kind == 'quadratic':
        a = x ** 2 + 0.01
    elif kind == 'cosine':
        a = np.cos(np.pi * x / 2) ** 2 + 0.01
    elif kind == 'sin':
        a = np.sin(np.pi * x) ** 2 + 0.01
    elif kind == 'bump':
        a = np.exp(-4 * x * (1 - x))
    elif kind == 'bimodal':
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    elif kind == 'polynomial3':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternation, power-mean and LogSumExp L-BFGS, subgradient,
    Fourier refinement, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Try multiple initializations
    init_kinds = ['parent', 'linear', 'quadratic', 'cosine', 'bimodal', 'polynomial3', 'bump', 'sin']
    n_base = 1200

    # If we have a parent, use its length; otherwise use 1200
    if parent is not None:
        n_base = len(parent)

    print(f"\n=== Phase 0: Multiple Initializations (n={n_base}) ===")
    init_candidates = []
    if parent is not None:
        init_candidates.append(('parent', parent))
    for kind in ['linear', 'quadratic', 'cosine', 'bimodal', 'polynomial3', 'bump', 'sin', 'uniform']:
        init_candidates.append((kind, make_init(n_base, kind)))

    # Quick evaluation of each init
    for label, a_init in init_candidates:
        sc = local_eval(a_init)
        print(f"  Init [{label}]: score={sc:.10f}")
        try_update(a_init, f"init_{label}")

    # === Phase 1: LP alternation on the best init ===
    print(f"\n=== Phase 1: LP Alternation (n={n_base}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")

    # === Phase 2: Power-mean L-BFGS ===
    print(f"\n=== Phase 2: Power-Mean L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 20:
        p_stages = [8, 16, 32, 64, 128, 256, 512, 1024, 2048]
        a_pm, sc_pm = power_mean_lbgfs(best_seq, t2, p_stages=p_stages)
        try_update(a_pm, "power_mean")

    # === Phase 3: LogSumExp L-BFGS (very low T) ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        T_stages = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lse, sc_lse = logsumexp_lbgfs(best_seq, t3, T_stages=T_stages)
        try_update(a_lse, "lgebfs_lowT")

    # === Phase 4: Second LP round (finer) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        a_lp2, sc_lp2 = lp_alternation(best_seq, t4, solver)
        try_update(a_lp2, "lp_round2")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 10:
        a_sg, sc_sg = subgradient_refine(best_seq, t5, n_steps=30000)
        try_update(a_sg, "subgrad")

    # === Phase 6: Fourier refinement ===
    print(f"\n=== Phase 6: Fourier (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.07, 60.0)
    if t6 > 10:
        for nf in [16, 32]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_fr, sc_fr = fourier_refine(best_seq, min(rem * 0.4, 30.0), n_freq=nf)
            try_update(a_fr, f"fourier_{nf}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_best = len(best_seq)
        n_candidates = []
        for mult in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.5]:
            cand_n = int(n_best * mult)
            if 200 <= cand_n <= 6000 and cand_n != n_best and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP
            a_cand, _ = lp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cand2, _ = power_mean_lbgfs(a_cand, min(rem * 0.4, 50.0),
                                               p_stages=[16, 64, 256, 1024])
                try_update(a_cand2, f"pm_n{cand_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low T LogSumExp
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            a_fin, _ = logsumexp_lbgfs(best_seq, t_fin,
                                       T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_fin, "final_lse")

        # Final subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            t_sg = min(rem * 0.3, 30.0)
            if t_sg > 5:
                a_fin_sg, _ = subgradient_refine(best_seq, t_sg, n_steps=20000)
                try_update(a_fin_sg, "final_sg")

        # Final LP
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin_lp, _ = lp_alternation(best_seq, min(rem * 0.4, 40.0), solver)
            try_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
