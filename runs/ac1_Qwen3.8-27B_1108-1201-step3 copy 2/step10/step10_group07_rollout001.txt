"""

2. Exact 1D coordinate descent (solving the 1D max-of-linears problem)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    v = v.copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def smoothed_obj_and_grad(a, beta):
    """
    Compute the smoothed objective and its gradient.
    Objective: (1/beta) * log(sum_k exp(beta * conv(a,a)[k]))
    Gradient: 2 * cross_correlation(weights, a)
    where weights = exp(beta*conv) / sum(exp(beta*conv))
    """
    n = len(a)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    max_val = beta_conv.max()
    exp_vals = np.exp(beta_conv - max_val)
    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp
    obj = (max_val + np.log(sum_exp)) / beta
    
    # Gradient: grad[j] = 2 * sum_m weights[j+m] * a[m]
    # = 2 * (weights * a_flipped)[j + n - 1]
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(weights, fft_size)
    Af = rfft(a_flip, fft_size)
    conv_result = irfft(W * Af, fft_size)[:conv_len]
    grad = 2.0 * conv_result[(n-1):(2*n-1)]
    
    return obj, grad


def softmax_lbgfs(a0, beta_levels, time_per_level, verbose=True):
    """
    Optimize using L-BFGS with softmax parameterization.
    a = softmax(z), so a >= 0 and sum(a) = 1 automatically.
    
    Gradient w.r.t. z:
    dz_j = a_j * (da_j - sum_i a_i * da_i)
    where da = gradient of smoothed objective w.r.t. a
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize z from a (log of a, with handling of zeros)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.max()  # for numerical stability
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    def softmax(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        return exp_z / exp_z.sum()
    
    def obj_z(z):
        a_tmp = softmax(z)
        conv = fast_convolve(a_tmp)
        beta_conv = beta * conv
        max_val = beta_conv.max()
        log_sum_exp = max_val + np.log(sum(np.exp(beta_conv - max_val)))
        return 2.0 * n * log_sum_exp / beta
    
    def grad_z(z):
        a_tmp = softmax(z)
        conv = fast_convolve(a_tmp)
        beta_conv = beta * conv
        max_val = beta_conv.max()
        exp_vals = np.exp(beta_conv - max_val)
        sum_exp = exp_vals.sum()
        weights = exp_vals / sum_exp
        
        # da = gradient w.r.t. a
        a_flip = a_tmp[::-1]
        conv_len = (2 * n - 1) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        W = rfft(weights, fft_size)
        Af = rfft(a_flip, fft_size)
        conv_result = irfft(W * Af, fft_size)[:conv_len]
        da = 2.0 * conv_result[(n-1):(2*n-1)]
        
        # dz_j = a_j * (da_j - sum_i a_i * da_i)
        mean_grad = np.sum(a_tmp * da)
        dz = a_tmp * (da - mean_grad)
        return 2.0 * n * dz / beta
    
    for beta in beta_levels:
        if time.monotonic() - t0 > time_per_level * len(beta_levels):
            break
        level_deadline = time.monotonic() + time_per_level
        
        def func(z_inner):
            return obj_z(z_inner)
        
        def jac(z_inner):
            return grad_z(z_inner)
        
        result = scipy_minimize(func, z, method='L-BFGS-B', jac=jac,
                                options={'maxiter': 200, 'gtol': 1e-8,
                                         'max_fun': 500})
        
        z = result.x
        a_new = softmax(z)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS beta={beta}: smoothed_obj={result.fun:.10f}, score={score:.8f}, best={best_score:.8f} ({elapsed:.0f}s)")
        
        # Check time
        if time.monotonic() > level_deadline:
            break
    
    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=True):

                pass
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            # For the affected part:
            # conv_new[k] = conv[k] + 2*delta*a[k-i] + delta^2 * (1 if k==2i else 0)
            # where delta = new_a[i] - old_a[i], so new_a[i] = a[i] + delta
            # Let v = a[i] (current value), and we're choosing new value u = v + delta
            # conv_new[k] = conv[k] + 2*(u-v)*a[k-i] + (u-v)^2 * 1[k=2i]
            
            a_shifted = a[k_start - i:k_end + 1]  # a[k-i] for k in range
            conv_aff = conv[k_start:k_end + 1].copy()
            has_quad = (2 * i >= k_start and 2 * i <= k_end)
            
            # We want to minimize: max(unaff_max, max_k (conv_aff[k] + 2*(u-v)*a_shifted[k-ks] + (u-v)^2*1[k=2i-ks]))
            # subject to u >= 0
            
            # This is a 1D problem. The optimal u is where the "active" constraints are balanced.
            # For simplicity, use a golden section search on u in [0, 2*v + 1e-10]
            
            v = a[i]
            # Search range: [0, max(2*v, v + (M - conv[k=2i])/max(1, a[i]) + 1e-10)]
            # To be safe, search [0, 3*v + 1]

            hi = max(3.0 * v + 1e-6, 1e-6)

                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * v + delta * delta
                return max(unaff_max, float(new_aff.max()))
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d_ = lo + gr * (hi - lo)
            fc = f_of_u(c)

            for _ in range(30):

                    d_ = c
                    fd = fc

                    fc = f_of_u(c)
                else:
                    lo = c
                    c = d_
                    fc = fd
                    d_ = lo + gr * (hi - lo)
                    fd = f_of_u(d_)
            
            u_opt = (lo + hi) / 2.0
            f_opt = f_of_u(u_opt)
            
            # Also check u=0
            f_zero = f_of_u(0.0)
            if f_zero < f_opt:
                u_opt = 0.0
                f_opt = f_zero
            
            current_best = max(unaff_max, float(conv_aff.max()))
            if f_opt < current_best - 1e-14:
                delta = u_opt - v
                a[i] = u_opt
                changes = 2.0 * delta * a_shifted
                if has_quad:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * v + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
