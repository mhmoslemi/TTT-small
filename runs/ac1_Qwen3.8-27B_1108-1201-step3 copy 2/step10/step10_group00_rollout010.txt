import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.APPS, cp.GLPK, cp.GUROBI]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS


def solve_active_lp(a_fixed, active_indices, solver):
    """Solve reduced LP: min M s.t. (conv(a,b))[k] <= M for k in active_indices, sum(b)=1, b>=0."""

    M_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # (conv(a,b))[k] = sum_{j=js}^{je} a[j] * b[k-j]
        # Coefficient of b[i] is a[k-i] for i in [js, je]... wait
        # Let's redo: (conv(a,b))[k] = sum_j a[j]*b[k-j]
        # For j in [js, je], k-j is in [k-je, k-js] which should be in [0, n-1]
        # Coefficient of b[i] (i in [0,n-1]) is a[k-i] if k-i in [js, je], i.e., i in [k-je, k-js]
        # So: coeff of b[i] = a[k-i] for i in [max(0,k-je), min(n-1,k-js)]
        i_lo = max(0, k - je)
        i_hi = min(n - 1, k - js)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        coeff = a_fixed[k - idx]
        constraints.append(coeff @ b_var[i_lo:i_hi + 1] <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:

            continue

            prob.solve(solver=s)

                break
        except Exception:
            continue

    if b_val is None or not np.all(np.isfinite(b_val)):
        return None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lse_obj_grad(z, n, T):
    """LogSumExp objective and gradient w.r.t. z (log-domain parameter)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s

    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return 0.0, np.zeros(n)

    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse / (1.0)  # sum(a) = 1 in this parameterization

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def nesterov_active_lp(a0, time_limit, solver, n_max_iters=1000000, M_start=10, M_max=60):
    """Nesterov-accelerated active-set LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_conv = fast_convolve(a)
    best_score = 2.0 * n * best_conv.max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine active set size (grow gradually)
        progress = min(1.0, (it + 1) / 200.0)
        M = int(M_start + (M_max - M_start) * progress)

        # Find top M peaks of conv(y, y)
        conv_y = fast_convolve(y)
        top_k = np.argsort(conv_y)[-M:]

        # Solve active LP
        b_new = solve_active_lp(y, top_k, solver)
        if b_new is None:
            # Fallback: use subgradient step
            k_star = int(np.argmax(conv_y))
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            g = np.zeros(n)
            idx = np.arange(js, je + 1)
            g[idx] = y[k_star - idx]
            eta = 1.0 / (np.sqrt(it + 1) * n)
            b_new = y - eta * g
            b_new = np.maximum(b_new, 0)
            b_new = b_new / b_new.sum()

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # FISTA-style running average
        a = ((it + 1) * a + 2.0 * b_new) / (it + 3)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate
        conv_a = fast_convolve(a)
        score_a = 2.0 * n * conv_a.max()
        score_y = 2.0 * n * conv_y.max()  # reuse

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Structured perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"  ActiveLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M={M}")

    return best_a, best_score


def lse_lbgfs(a0, time_limit, T_stages):
    """LogSumExp L-BFGS with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300) * n)  # center at zero
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        maxiter = max(30, int(200 * remaining / max(1, len(T_stages) - stage_i)))

        def func(z_val):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad(z_val, n, stage_T)
            # Check current best
            a_cur = softmax_from_z(z_val)
            sc = 2.0 * n * fast_convolve(a_cur).max()
            if sc < lse_lbgfs._best_score:
                lse_lbgfs._best_score = sc
                lse_lbgfs._best_a = a_cur.copy()
            return obj

        def jac(z_val):
            if time.monotonic() > deadline:
                return np.zeros_like(z_val)
            _, g = lse_obj_grad(z_val, n, stage_T)
            return g

        lse_lbgfs._best_score = best_score
        lse_lbgfs._best_a = best_a.copy()

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        best_score = lse_lbgfs._best_score
        best_a = lse_lbgfs._best_a.copy()

        elapsed = time.monotonic() - t0
        print(f"  LSE-LBFGS stage {stage_i+1}/{len(T_stages)} T={stage_T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def multiplicative_weights(a0, time_limit, n_steps=100000):
    """Multiplicative weights (mirror descent) on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    eta = 0.02

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.8
            no_improve = 0
            if eta < 0.001:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"  MultiW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def structured_init(n, rng, seed_type=0):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if seed_type == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif seed_type == 1:
        a = np.sin(np.pi * x) ** 2
    elif seed_type == 2:
        a = np.exp(-6 * x * (1 - x))
    elif seed_type == 3:
        a = 0.7 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif seed_type == 4:
        a = (1 - x) ** 2 + 0.05
    elif seed_type == 5:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2)
    elif seed_type == 6:
        a = np.sinc(3 * x - 1.5) ** 2
        a = np.maximum(a, 0)
    elif seed_type == 7:
        # Tent-like
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        a = a ** 1.5 + 0.02
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)

def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    num_harmonics = rng.integers(1, 4)
    pert = np.zeros(n)
    for _ in range(num_harmonics):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength / num_harmonics
        pert += amp * np.cos(freq * x + phase)
    a_pert = a + pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

                continue
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                trial = a.copy()
                trial[i] = old_val * mult
                s_trial = trial.sum()
                if s_trial < 0.01:
                    continue
                trial = trial / s_trial
                sc = 2.0 * n * fast_convolve(trial).max()
                if sc < best_score - 1e-12:
                    a = trial
                    best_score = sc
                    best_a = a.copy()

        elapsed = time.monotonic() - t0
        if (p + 1) % 2 == 0:
            print(f"  CoordRefine pass {p+1}/{n_passes} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses active-set LP with Nesterov acceleration, LogSumExp L-BFGS,
    multiplicative weights, coordinate refinement, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, 0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    # Helper: try to update best
    state = {'best_seq': best_seq, 'best_score': best_score}

    def local_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < state['best_score']:
            state['best_score'] = sc
            state['best_seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_cur = len(best_seq)

    # === Phase 1: Active-set Nesterov LP on parent ===
    print(f"\n=== Phase 1: ActiveSetLP n={n_cur} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_active_lp(best_seq, t_alloc, solver, n_max_iters=1000000, M_start=8, M_max=50)
        local_update(a_lp, "active_lp_1")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: LSE-LBFGS (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 380.0)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
        print(f"  {len(T_stages)} stages, budget={t_alloc:.0f}s")
        a_lb, _ = lse_lbgfs(state['best_seq'], t_alloc, T_stages)
        local_update(a_lb, "lse_lbgfs")

    # === Phase 3: Active-set LP round 2 ===
    print(f"\n=== Phase 3: ActiveSetLP Round 2 (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 40:
        a_lp2, _ = nesterov_active_lp(state['best_seq'], t_alloc, solver, n_max_iters=500000, M_start=15, M_max=60)
        local_update(a_lp2, "active_lp_2")

    # === Phase 4: Multiplicative weights ===
    print(f"\n=== Phase 4: MultiW (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_mw, _ = multiplicative_weights(state['best_seq'], t_alloc, n_steps=200000)
        local_update(a_mw, "multiw")

    # === Phase 5: Coordinate refinement ===
    print(f"\n=== Phase 5: CoordinateRefine (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_cr, _ = coordinate_refine(state['best_seq'], t_alloc, n_passes=3)
        local_update(a_cr, "coord_refine")

    # === Phase 6: Try n=1400 ===
    print(f"\n=== Phase 6: Try n=1400 (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_n1400 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_n1400:.0f}s")
        init_1400 = interpolate_sequence(state['best_seq'], 1400)
        a_1400, _ = nesterov_active_lp(init_1400, t_n1400 * 0.7, solver, n_max_iters=300000, M_start=10, M_max=40)
        local_update(a_1400, "lp_n1400")

        rem = deadline - time.monotonic()
        if state['best_seq'] is not None and len(state['best_seq']) == 1400 and rem > 40:
            T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1400b, _ = lse_lbgfs(state['best_seq'], min(rem, 100.0), T_stages2)
            local_update(a_1400b, "lse_n1400")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation Escape (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_pert = len(state['best_seq'])
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            a_pert = perturbed_sequence(state['best_seq'], rng, strength=0.015 + 0.01 * trial)
            t_sub = min(rem_sub * 0.3, 50.0)
            a_opt, _ = nesterov_active_lp(a_pert, t_sub, solver, n_max_iters=200000, M_start=10, M_max=40)
            local_update(a_opt, f"pert{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={state['best_score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_stages3 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_fin, _ = lse_lbgfs(state['best_seq'], min(rem * 0.5, 40.0), T_stages3)
        local_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin_lp, _ = nesterov_active_lp(state['best_seq'], min(rem * 0.6, 25.0), solver, n_max_iters=100000, M_start=20, M_max=50)
        local_update(a_fin_lp, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin_mw, _ = multiplicative_weights(state['best_seq'], min(rem * 0.5, 15.0), n_steps=50000)
        local_update(a_fin_mw, "final_mw")

    # Final verification
    if state['best_seq'] is not None:
        fs = evaluate_local(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = min(state['best_score'], fs)
        print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")

    result = state['best_seq']
    if result is None:
        result = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in result]
