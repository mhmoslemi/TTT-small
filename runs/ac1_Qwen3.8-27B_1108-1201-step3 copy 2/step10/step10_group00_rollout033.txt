    fft_size = 1 << (size - 1).bit_length()

At the optimum, for all i with a_i > 0:

where lambda is a probability distribution on the peaks.

If I assume lambda is uniform over the top m peaks, then:
(1/m) * sum_{k in top_m} 2 * a_{k-i} = mu for all i in supp(a)

This is a system of linear equations in a! Specifically:
For each i in supp(a): sum_{k in top_m} a_{k-i} = mu * m / 2

The left side is: sum_{k in top_m} a_{k-i} = (a * indicator_top_m)(i) (cross-correlation)

Wait, this is a convolution! sum_{k in top_m} a_{k-i} = sum_j a_j * [count of k in top_m with k-i=j] = (a * w)(i) where w_j = |{k in top_m : k-i=j}|... no, this isn't a standard convolution.

Let me think again. sum_{k in top_m} a_{k-i} for a fixed i:

Hmm, this makes the linear system less clean. But it's still linear in a!

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Pair-swapping coordinate descent (exact 1D optimization per coordinate pair)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def parametric_init(n, rng):
    """Try multiple parametric families, return best initialization."""
    best_a = None
    best_score = float('inf')
    x = np.arange(n, dtype=np.float64)

    candidates = []
    # Exponential decay
    for beta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        candidates.append(np.exp(-beta * x / n))
    # Power law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append((x + 1) ** (-alpha))
    # Sine shapes
    for p in [0.5, 1, 1.5, 2, 3, 4]:
        candidates.append(np.sin(np.pi * (x + 0.5) / n) ** p)
    # Front-loaded Gaussian
    for mu in [0.15, 0.2, 0.25, 0.3, 0.4]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x / n - mu) ** 2) / (2 * sigma ** 2)))
    # Bimodal
    for mu1 in [0.2, 0.3]:
        for mu2 in [0.6, 0.7]:
            candidates.append(np.exp(-5 * (x / n - mu1) ** 2) + 0.5 * np.exp(-5 * (x / n - mu2) ** 2))
    # Linear decay
    for slope in [0.5, 1.0, 2.0, 3.0]:
        candidates.append(np.maximum(1.0 - slope * x / n, 0.01))
    # Geometric (front-heavy)
    for r in [0.95, 0.9, 0.85, 0.8]:
        candidates.append(r ** x)
    # Cosine window variants
    for p in [1, 2, 3]:
        candidates.append(np.cos(np.pi * x / (2 * n)) ** (2 * p))

    for cand in candidates:
        a = normalize(cand)
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def kkt_peak_balance(a_init, m_peaks=8, max_iter=200, time_limit=60.0, tol=1e-10):
    """
    Solve KKT linear system: at optimum, for all i with a_i>0:
        sum_{k in active} (1/m) * 2*a_{k-i} = mu
    This gives a linear system in a (and mu) that we solve iteratively.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()

        # Find active peaks (top m)
        top_idx = np.argsort(c)[::-1][:m_peaks]
        top_set = set(int(k) for k in top_idx)

        # Build linear system
        # For each i in [0, n-1]: sum_{k in top_set} a[(k-i) % nothing, just k-i if valid] = C
        # Plus: sum_i a_i = 1
        # Total: n+1 equations, n+1 unknowns (a_0..a_{n-1}, C)

        A_mat = np.zeros((n + 1, n + 1))
        b_vec = np.zeros(n + 1)

        for i in range(n):
            row = i
            for k in top_set:
                j = k - i
                if 0 <= j < n:
                    A_mat[row, j] = 1.0
            A_mat[row, n] = -1.0  # -C term
            b_vec[row] = 0.0

        # Normalization constraint
        A_mat[n, :n] = 1.0
        b_vec[n] = 1.0

        # Solve (use lstsq for robustness)
        try:
            # Add small regularization for stability
            reg = 1e-12
            A_reg = A_mat.copy()
            A_reg[np.diag_indices_from(A_reg)] += reg
            sol = np.linalg.solve(A_reg, b_vec)
        except np.linalg.LinAlgError:
            sol, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=1e-14)

        a_new = sol[:n]
        C_val = sol[n]

        # Check quality
        if np.all(a_new >= -1e-6):
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
            # Check convergence
            if abs(C_val - 0) < tol and sc < best_score + 1e-12:
                break
        else:
            # Some a_i negative: blend with current
            frac = 0.3
            a_blend = (1 - frac) * a + frac * np.maximum(a_new, 0)
            a_blend = normalize(a_blend)
            sc = evaluate_local(a_blend)
            if sc < best_score:
                best_score = sc
                best_a = a_blend.copy()
            a = a_blend

    return best_a, best_score


def pair_swap_cd(a_init, n_pairs=20000, time_limit=120.0, rng=None):
    """
    Coordinate descent via pair swaps: change a[i] by +d and a[j] by -d.
    For each pair, find optimal d by 1D minimization.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a_init)
    a = normalize(a_init.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    c = fast_convolve(a)

    def score_with_swap(a, i, j, d):
        """Compute score after swapping d from j to i."""
        a_new = a.copy()
        a_new[i] += d
        a_new[j] -= d
        if a_new[i] < 0 or a_new[j] < 0:
            return float('inf')
        b = fast_convolve(a_new)
        s = a_new.sum()
        if s < 0.01:
            return float('inf')
        return 2.0 * n * b.max() / (s * s)

    for step in range(n_pairs):
        if time.monotonic() - t0 > time_limit:
            break

        # Choose random pair, bias toward high-convolution indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Determine valid range of d
        d_min = max(-a[i], -0.5)  # a[i] + d >= 0
        d_max = min(a[j], 0.5)    # a[j] - d >= 0
        if d_min >= d_max:
            continue

        # 1D minimization
        def obj_1d(logit_d):
            # Map logit to (-1, 1) then scale to valid range
            t = 1.0 / (1.0 + np.exp(-logit_d))
            d = d_min + t * (d_max - d_min)
            d = np.clip(d, d_min + 1e-15, d_max - 1e-15)
            return score_with_swap(a, i, j, float(d))

        res = minimize_scalar(obj_1d, bounds=(-50.0, 50.0), method='bounded',
                              options={'xatol': 1e-10, 'maxiter': 50})

        if res.fun < best_score - 1e-15:
            t_opt = 1.0 / (1.0 + np.exp(-res.x))
            d_opt = d_min + t_opt * (d_max - d_min)
            d_opt = np.clip(d_opt, d_min + 1e-15, d_max - 1e-15)
            a[i] += d_opt
            a[j] -= d_opt
            a = normalize(a)
            best_score = res.fun
            best_a = a.copy()

        # Periodic reporting
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PairSwap {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def accelerated_mirror_descent(a_init, time_limit=120.0, n_steps=500000, rng=None):
    """
    Nesterov-accelerated mirror descent (multiplicative weights)
    with adaptive step size and active-set subgradient.
    """
    if rng is None:
        rng = np.random.default_rng(777)
    n = len(a_init)
    y = normalize(a_init.copy())
    a = y.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    eta = 0.1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution and subgradient
        c = fast_convolve(y)
        M = c.max()

        # Active set: peaks within 0.1% of max
        threshold = M * (1.0 - 1e-3)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Limit active set size
        if len(active) > 20:
            top = np.argsort(c[active])[-20:]
            active = active[top]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2.0

        # Multiplicative update on y
        log_update = -eta * g
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = 1.0
        y_new = y_new / s_y
        y = y_new

        # Nesterov extrapolation
        alpha = 2.0 / (step + 2)
        a_new = (1.0 - alpha) * a + alpha * y
        a_new = np.maximum(a_new, 0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_a
        a = a_new

        # Evaluate
        sc = evaluate_local(a)
        sc_y = evaluate_local(y)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break
        elif no_improve == 0 and eta < 0.3:
            eta *= 1.1

    return best_a, best_score


def simulated_annealing(a_init, time_limit=60.0, n_trials=5000, rng=None):
    """Random perturbations with simulated annealing acceptance."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a_init)
    a = normalize(a_init.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)

    T_start = 0.01
    T_end = 0.0001

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        progress = trial / n_trials
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation
        strength = 0.05 * (1 - progress) + 0.01
        noise = rng.normal(0, strength, n)
        a_new = a + noise
        a_new = normalize(a_new)
        sc_new = evaluate_local(a_new)

        # Acceptance criterion
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        elif sc_new < best_score + T:
            if rng.random() < np.exp(-(sc_new - best_score) / max(T, 1e-10)):
                a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if sc_md < best_score:
            best_score = sc_md
            best_a = a_md.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses KKT linear-system solving, pair-swap coordinate descent,
    accelerated mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq, best_score = parametric_init(n_init, rng)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting score: {best_score:.8f} at n={n_best}")

    # === Phase 1: KKT peak balancing on parent ===
    print(f"\n=== Phase 1: KKT Balance n={n_best} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_kkt, _ = kkt_peak_balance(best_seq, m_peaks=10, max_iter=500,
                                    time_limit=t1 * 0.6)
        local_update(a_kkt, "kkt1")
        # Follow up with more KKT iterations
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            a_kkt2, _ = kkt_peak_balance(best_seq, m_peaks=15, max_iter=300,
                                         time_limit=min(rem * 0.5, 100.0))
            local_update(a_kkt2, "kkt2")

    # === Phase 2: Accelerated mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 280.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_md, _ = accelerated_mirror_descent(best_seq, time_limit=t2,
                                             n_steps=300000, rng=rng)
        local_update(a_md, "mirror_descent")

    # === Phase 3: KKT again with updated best ===
    print(f"\n=== Phase 3: KKT Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        a_kkt3, _ = kkt_peak_balance(best_seq, m_peaks=12, max_iter=400,
                                     time_limit=t3)
        local_update(a_kkt3, "kkt3")

    # === Phase 4: Pair-swap coordinate descent ===
    print(f"\n=== Phase 4: Pair Swap CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s")
        a_ps, _ = pair_swap_cd(best_seq, n_pairs=15000, time_limit=t4, rng=rng)
        local_update(a_ps, "pair_swap")

    # === Phase 5: Try different lengths ===
    lengths_to_try = [1000, 1100, 1300, 1400, 1500]
    for n_try in lengths_to_try:
        if n_try == n_best:
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_len = min(rem * 0.12, 120.0)
        if t_len < 15:
            break
        print(f"\n=== Trying n={n_try} (budget={t_len:.0f}s) ===")

        # Interpolate
        a_interp = interpolate_sequence(best_seq, n_try)
        sc_interp = evaluate_local(a_interp)
        local_update(a_interp, f"interp_{n_try}")

        # KKT on this length
        a_kkt_n, _ = kkt_peak_balance(a_interp, m_peaks=10, max_iter=200,
                                      time_limit=t_len * 0.5)
        local_update(a_kkt_n, f"kkt_{n_try}")

        # Mirror descent
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == n_try:
            a_md_n, _ = accelerated_mirror_descent(best_seq, time_limit=min(rem * 0.4, 40.0),
                                                   n_steps=50000, rng=rng)
            local_update(a_md_n, f"md_{n_try}")

    # === Phase 6: Simulated annealing ===
    print(f"\n=== Phase 6: Simulated Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 80.0)
    if t6 > 10:
        a_sa, _ = simulated_annealing(best_seq, time_limit=t6, n_trials=3000, rng=rng)
        local_update(a_sa, "sim_anneal")

    # === Phase 7: Combined refinement on best ===
    print(f"\n=== Phase 7: Combined Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t7 = min(rem * 0.25, 180.0)
        # KKT with more peaks
        a_r1, _ = kkt_peak_balance(best_seq, m_peaks=20, max_iter=300,
                                   time_limit=t7 * 0.4)
        local_update(a_r1, "refine_kkt20")
        # Mirror descent
        rem = deadline - time.monotonic()
        if rem > 15:
            a_r2, _ = accelerated_mirror_descent(best_seq, time_limit=t7 * 0.3,
                                                 n_steps=100000, rng=rng)
            local_update(a_r2, "refine_md")
        # Pair swap
        rem = deadline - time.monotonic()
        if rem > 10:
            a_r3, _ = pair_swap_cd(best_seq, n_pairs=5000,
                                   time_limit=min(rem * 0.5, 30.0), rng=rng)
            local_update(a_r3, "refine_ps")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.3, 60.0)
        a_final, _ = fine_polish(best_seq, time_limit=t8)
        local_update(a_final, "final_polish")

    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        a_final2, _ = kkt_peak_balance(best_seq, m_peaks=15, max_iter=100,
                                       time_limit=min(rem * 0.5, 20.0))
        local_update(a_final2, "final_kkt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
