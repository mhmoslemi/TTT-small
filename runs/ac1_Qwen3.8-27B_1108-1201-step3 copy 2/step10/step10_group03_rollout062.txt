Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
exact 1D subproblem solving, LP-based alternating minimization,
spectral smoothing for escaping local minima, and L-BFGS polish.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def coordinate_descent(a0, n_passes=50, verbose=False):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with exact
1D subproblem solving, LP-based alternating minimization, spectral smoothing
for escaping local minima, and multi-n exploration with L-BFGS polish.
Key novel component: exact coordinate descent exploiting the structure of the
auto-convolution to solve each 1D subproblem via golden section in O(n) per
evaluation, enabling hundreds of full passes within the time budget.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(fixed, solver):
    """Given fixed (sum=1), find other >= 0, sum=1 minimizing max conv(fixed, other)."""
    n = len(fixed)
    var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ var <= M_var, cp.sum(var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = np.asarray(var.value, dtype=np.float64)
                val = np.maximum(val, 0)
                sv = val.sum()
                if sv > 1e-15:
                    return val / sv
        except Exception:
            continue
    return None


def lp_alternation(a0, time_limit, solver, max_iters=500):
    """Alternating LP minimization with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = y_b

        a_new = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = y_a

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

    return best_a, best_score


def coordinate_descent(a0, n_passes=50, verbose=False, time_limit=None):
    """
    Coordinate descent with exact 1D subproblem solving via golden section.
    Key insight: for coordinate j, changing a[j] to x changes the convolution
    as c_new[k] = base[k] + x*inf[k] + x^2*delta_{k,2j} where
    inf[k] = 2*a[k-j] for k in [j, j+n-1], k != 2j, else 0.
    """
    n = len(a0)
    a = normalize(a0).copy()
    c = fast_convolve(a)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    size = 2 * n - 1
    t0 = time.monotonic()
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for p in range(n_passes):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        for j in range(n):
            # Build inf array: inf[k] = 2*a[k-j] for k in [j, j+n-1], zero out k=2j
            inf = np.zeros(size)
            s_idx = j
            e_idx = min(j + n - 1, size - 1)
            if s_idx <= e_idx:
                inf[s_idx:e_idx + 1] = 2.0 * a[0:e_idx - s_idx + 1]
            two_j = 2 * j
            if two_j < size:
                inf[two_j] = 0.0

            # Compute base: base[k] = c[k] - a[j]*inf[k] - a[j]^2*delta_{k,2j}
            aj = a[j]
            base = c - aj * inf
            if two_j < size:
                base[two_j] -= aj * aj

            # Evaluate current objective value (for comparison)
            S = a.sum()
            S_minus = S - aj

            # Golden section search for optimal x in [0, x_max]
            x_max = max(5.0 * aj, 2.0 / n)
            if S_minus < 1e-15:
                x_max = max(x_max, 1.0)

            # Objective: max( max_k(base[k] + x*inf[k]), base[2j] + x^2 ) / (S_minus + x)^2
            # Pre-compute parts that don't depend on x
            base_max_no_inf = np.max(base)  # lower bound on numerator at x=0

            def eval_1d(x):
                vals = base + x * inf
                v = np.max(vals)
                if two_j < size:
                    vq = base[two_j] + x * x
                    if vq > v:
                        v = vq
                denom = S_minus + x
                if denom < 1e-15:
                    return 1e30
                return v / (denom * denom)

            # Get current value
            f_cur = eval_1d(aj)

            # Golden section
            lo = 0.0
            hi = x_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_1d(c1)
            f2 = eval_1d(c2)

            for _ in range(45):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_1d(c2)

            x_opt = (lo + hi) / 2.0
            f_opt = eval_1d(x_opt)

            # Also check boundaries
            f_at_0 = eval_1d(0.0)
            if f_at_0 < f_opt:
                x_opt = 0.0
                f_opt = f_at_0

            if f_opt < f_cur * (1.0 - 1e-11):
                d = x_opt - aj
                a[j] = x_opt
                # Update c incrementally
                c += d * inf
                if two_j < size:
                    c[two_j] += x_opt * x_opt - aj * aj

        # Renormalize
        a = a / a.sum()
        c = fast_convolve(a)
        sc = 2.0 * n * c.max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and (p + 1) % 10 == 0:
            print(f"      CD pass {p+1}: score={sc:.10f}")

    return best_a, best_score


def spectral_smooth(a0, alpha=0.7, n_iter=3):
    """
    Spectral smoothing: modify the magnitude of the DFT to flatten the
    convolution spectrum, then invert. Used to escape local minima.
    """
    n = len(a0)
    a = normalize(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    for _ in range(n_iter):
        C = np.abs(A) ** 2
        mean_C = C.mean()
        # Blend towards uniform spectrum
        C_new = alpha * C + (1.0 - alpha) * mean_C
        # Preserve phase, update magnitude
        phase = np.angle(A)
        A = np.sqrt(C_new) * np.exp(1j * phase)
        # Invert
        a_new = irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Check if improvement
        sc = 2.0 * n * fast_convolve(a_new).max()
        sc_old = 2.0 * n * fast_convolve(a).max()
        if sc < sc_old:
            a = a_new
            A = rfft(a, fft_size)
        else:
            alpha = min(0.95, alpha * 1.3)
            if alpha > 0.94:
                break

    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S_dot = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S_dot)

        try:
            res = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def phase_pipeline(a0, n, time_budget, solver, label=""):
    """Run the full optimization pipeline on a sequence of length n."""
    t0 = time.monotonic()
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    def try_update(arr):
        nonlocal best_a, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 1e-10:
            return
        arr = arr / s
        sc = 2.0 * n * fast_convolve(arr).max()
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()
            print(f"    [{label}] NEW: {sc:.10f}")

    # Step 1: LP alternation (if time allows)
    t_lp = min(time_budget * 0.2, 120.0)
    if t_lp > 20:
        rem = time_budget - (time.monotonic() - t0)
        if rem > 20:
            a_lp, _ = lp_alternation(best_a, t_lp, solver)
            try_update(a_lp)

    # Step 2: Coordinate descent (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 30:
        t_cd = rem * 0.35
        a_cd, _ = coordinate_descent(best_a, n_passes=200, time_limit=t_cd)
        try_update(a_cd)

    # Step 3: Spectral smoothing + CD (escape local minima)
    for trial in range(3):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 30:
            break
        alpha_val = 0.5 + 0.15 * trial
        a_spec = spectral_smooth(best_a, alpha=alpha_val, n_iter=5)
        rem = time_budget - (time.monotonic() - t0)
        if rem < 20:
            break
        t_cd2 = rem * 0.25
        a_cd2, _ = coordinate_descent(a_spec, n_passes=100, time_limit=t_cd2)
        try_update(a_cd2)
        try_update(a_spec)

    # Step 4: More coordinate descent
    rem = time_budget - (time.monotonic() - t0)
    if rem > 30:
        a_cd3, _ = coordinate_descent(best_a, n_passes=200, time_limit=rem * 0.4)
        try_update(a_cd3)

    # Step 5: L-BFGS polish
    rem = time_budget - (time.monotonic() - t0)
    if rem > 20:
        a_lb, _ = lbgfs_logsumexp(best_a, rem * 0.5,
                                  T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02])
        try_update(a_lb)

    # Step 6: Final CD polish
    rem = time_budget - (time.monotonic() - t0)
    if rem > 15:
        a_fin, _ = coordinate_descent(best_a, n_passes=100, time_limit=rem * 0.6)
        try_update(a_fin)

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    else:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses coordinate descent (exact 1D subproblems), LP alternation,
    spectral smoothing, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_global_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Optimize at parent n ===
    print(f"\n=== Phase 1: Optimize n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.5, 900.0)
    if t1 > 60:
        a1, s1 = phase_pipeline(best_seq, n_best, t1, solver, label="n_parent")
        try_global_update(a1, f"phase1_n{n_best}")

    # === Phase 2: Try other n values ===
    print(f"\n=== Phase 2: Multi-n exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_candidates = []
        for mult in [0.75, 0.85, 0.9, 1.1, 1.25, 1.5]:
            cand_n = int(n_best * mult)
            if 300 <= cand_n <= 4000 and cand_n != n_best and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        # Sort by proximity to n_best
        n_candidates.sort(key=lambda x: abs(x - n_best))
        n_candidates = n_candidates[:4]  # Limit to 4 candidates

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 200.0)
            if t_sub < 40:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            a_cand, _ = phase_pipeline(init_cand, cand_n, t_sub, solver, label=f"n{cand_n}")
            try_global_update(a_cand, f"phase2_n{cand_n}")

    # === Phase 3: Random restarts ===
    print(f"\n=== Phase 3: Random restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for restart in range(3):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_r = min(rem * 0.2, 120.0)
            if t_r < 30:
                break
            n_r = n_best
            a_r = structured_init(n_r, rng)
            # Quick CD
            a_r2, _ = coordinate_descent(a_r, n_passes=100, time_limit=t_r * 0.7)
            try_global_update(a_r2, f"restart{restart}")

    # === Phase 4: Final intensive polish on best ===
    print(f"\n=== Phase 4: Final polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_fin = len(best_seq)
        # Long CD run
        t_cd_fin = rem * 0.4
        a_fin1, _ = coordinate_descent(best_seq, n_passes=500, time_limit=t_cd_fin)
        try_global_update(a_fin1, "final_cd")

        rem = deadline - time.monotonic()
        if rem > 20:
            # Spectral + CD
            a_spec = spectral_smooth(best_seq, alpha=0.6, n_iter=5)
            a_fin2, _ = coordinate_descent(a_spec, n_passes=100, time_limit=rem * 0.3)
            try_global_update(a_fin2, "final_spec_cd")
            try_global_update(a_spec, "final_spec")

        rem = deadline - time.monotonic()
        if rem > 15:
            # L-BFGS very low T
            a_fin3, _ = lbgfs_logsumexp(best_seq, rem * 0.5,
                                        T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01])
            try_global_update(a_fin3, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 10:
            # Final CD
            a_fin4, _ = coordinate_descent(best_seq, n_passes=200, time_limit=rem * 0.7)
            try_global_update(a_fin4, "final_cd2")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
