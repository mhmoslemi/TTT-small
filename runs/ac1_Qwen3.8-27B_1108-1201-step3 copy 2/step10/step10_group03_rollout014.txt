In signal processing, the optimal flat-top window is the one that minimizes the sidelobe level of its frequency response. But our problem is slightly different: we want to minimize the peak of the time-domain autocorrelation, not the frequency response.

However, there's a duality: the peak of the time-domain autocorrelation is related to the average of the squared frequency response. Specifically:
max_k (a*a)[k] = (1/N) max_k |sum_freq A(freq) e^{2πi freq k/N}|^2

This is exactly the PAPR of the signal A in the frequency domain! And it's a well-studied quantity.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (assumes sum=1, non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_min_peak(a_fixed, active_fractions, solver, warm_b=None):
    """
    Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max over active set of conv(a,b).
    Active set = top `active_fractions` fraction of convolution positions.
    Returns (b, M) or (None, None).
    """
    n = len(a_fixed)
    b = fast_convolve(a_fixed)
    M_current = b.max()
    
    # Select active constraints: top fraction of peaks
    n_active = max(5, int(len(b) * active_fractions))
    top_idx = np.argsort(b)[-n_active:]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints for active positions only
    T = build_conv_matrix(a_fixed)
    T_active = T[top_idx, :]
    
    constraints = [
        T_active @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        if warm_b is not None:
            b_var.value = warm_b
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s, float(M_var.value)
    except Exception:
        pass
    
    # Fallback solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def peak_equalization_bisection(a0, solver, time_limit):
    """
    Bisection on M: find the smallest M such that there exists b >= 0, sum(b)=1

            break
        
        # Current peak
        b_conv = fast_convolve(a)
        M_current = b_conv.max()
        
        # Lower bound: average = 1/(2n-1) (since sum of conv = 1)
        M_lower = 1.0 / (2 * n - 1)
        M_upper = M_current
        
        # Bisection: find smallest M achievable
        for bisect in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            M_mid = 0.5 * (M_lower + M_upper)
            
            # Check feasibility: exists b >= 0, sum(b)=1 with T_a @ b <= M_mid?
            b_var = cp.Variable(n, nonneg=True)
            T = build_conv_matrix(a)
            constraints = [
                T @ b_var <= M_mid,
                cp.sum(b_var) == 1
            ]
            prob = cp.Problem(cp.Minimize(cp.sum(b_var)), constraints)  # just feasibility
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    M_upper = M_mid
                    b_val = np.asarray(b_var.value, dtype=np.float64)

def logsumexp_lbgfs_v2(a0, time_limit):
    """
    L-BFGS on LogSumExp-smoothed objective with very aggressive temperature annealing.
    Uses log-parametrization for non-negativity.
    """
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Very aggressive temperature schedule
    T_stages = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 5))
        
        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a_j: 2 * sum_k w[k] * a[k-j] (where valid)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            # Chain rule for softmax
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def subgradient_v2(a0, time_limit, n_steps=50000):

        M = b.max()
        
        # Use top-k peaks for gradient (smooths out noise)

        threshold = b[top_k_idx[0]]
        active = np.where(b >= threshold)[0]

        g = np.zeros(n)

        if st < 1e-15:

            if no_improve >= 50:

                no_improve = 0
        lr = max(1e-8, lr)
        
        if (step + 1) % 10000 == 0:
            print(f"    SubGrad step={step+1}: score={best_score:.10f}, lr={lr:.2e}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=100):
    """
    Coordinate descent: for each coordinate j, find the optimal value of a[j]
    given all other coordinates fixed.
    
    For fixed a[-j], the convolution entries are:
    conv[k] = conv_base[k] + a[j]*(a[k-j] + (a[j-k] if j-k != j else 0))
    Actually, conv[k] = sum_{i != j} a[i]*a[k-i] + a[j]*a[k-j] + a[k-j]*a[j] - a[j]*a[k-j]
    (the last term subtracts the double-count when i=j and k-i=j, i.e., k=2j)
    
    More carefully: conv[k] = sum_i a[i]*a[k-i]
    = sum_{i != j, k-i != j} a[i]*a[k-i] + a[j]*a[k-j] + a[k-j]*a[j] - (a[j]*a[j] if k==2j)
    = conv_excl[k] + a[j]*(a[k-j] + a[k-j]) - (a[j]^2 if k==2j)
    where conv_excl[k] = sum_{i != j, k-i != j} a[i]*a[k-i]
    
    This is quadratic in a[j] for each k, making it complex. Let's use a simpler approach:
    line search along each coordinate.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for round_idx in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random coordinate order
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try a few values around current
            # The optimal a[j] should balance the peaks it contributes to
            # Try: (0, 0.25*old, 0.5*old, 0.75*old, old, 1.5*old, 2*old, 3*old)
            candidates = [0.0, 0.25 * old_val, 0.5 * old_val, 0.75 * old_val,
                         old_val, 1.5 * old_val, 2.0 * old_val, 4.0 * old_val]
            
            best_local = old_val
            best_local_sc = best_score
            
            for cv in candidates:
                a[j] = cv
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    continue
                # Quick eval
                b = fast_convolve(a)
                sc = 2.0 * n * b.max() / (s * s)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = cv
            
            a[j] = best_local
            s = a.sum()
            if s > 1e-15:
                a = a / s
            
            sc = 2.0 * n * fast_convolve(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                improved = True
        
        if not improved and round_idx > 5:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: LP Alternation (35% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Alternation (n={n_best}, score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation_v2(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")
    
    # === Phase 2: Peak Equalization Bisection (20% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak Equalization (score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        print(f"  Time budget: {t2:.0f}s")
        a_pe, sc_pe = peak_equalization_bisection(best_seq, solver, t2)
        try_update(a_pe, "peak_equiv")
    
    # === Phase 3: L-BFGS LogSumExp (25% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS LogSumExp (score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 350.0)
    if t3 > 30:
        print(f"  Time budget: {t3:.0f}s")
        a_lb, sc_lb = logsumexp_lbgfs_v2(best_seq, t3)
        try_update(a_lb, "lbgfs")
    
    # === Phase 4: Subgradient (10% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Subgradient (score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 10:
        a_sg, sc_sg = subgradient_v2(best_seq, t4, n_steps=50000)
        try_update(a_sg, "subgrad")
    
    # === Phase 5: Perturbation + Refine (5% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Perturbation Refine (score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.05, 60.0)
    if t5 > 10:
        a_pert, sc_pert = perturb_and_refine(best_seq, solver, t5, n_perturbs=4)
        try_update(a_pert, "perturb")
    
    # === Phase 6: Final intensive L-BFGS polish (5% time) ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish (score={best_score:.10f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.05, 50.0)
    if t6 > 10:
        # Very low T L-BFGS
        n = len(best_seq)
        a = normalize(best_seq)
        a_clipped = np.maximum(a, 1e-300)
        z = np.log(a_clipped) - np.log(a_clipped.sum())
        
        for T in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            max_iter = max(20, int(remaining * 3))
            
            def obj(z):
                z_max = z.max()
                exp_z = np.exp(z - z_max)
                a_cur = exp_z / exp_z.sum()
                b = fast_convolve(a_cur)
                b_shift = b / T
                b_max = b_shift.max()
                lse = b_max + np.log(np.exp(b_shift - b_max).sum())
                return (2.0 * n) * lse
            
            def grad(z):
                z_max = z.max()
                exp_z = np.exp(z - z_max)
                a_cur = exp_z / exp_z.sum()
                b = fast_convolve(a_cur)
                b_shift = b / T
                b_max = b_shift.max()
                exp_ws = np.exp(b_shift - b_max)
                w = exp_ws / exp_ws.sum()
                grad_a = np.zeros(n)
                for k in range(len(b)):
                    if w[k] < 1e-14:
                        continue
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
                S = np.dot(grad_a, a_cur)
                return a_cur * (grad_a - S)
            
            try:
                res = minimize(
                    obj, z, jac=grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                             'maxfun': max_iter * 300, 'disp': False}
                )
                z = res.x
                a_new = np.exp(z - z.max())
                a_new = a_new / a_new.sum()
                try_update(a_new, f"polish_T{T}")
            except Exception:
                pass
    
    # === Phase 7: Try nearby n values if time remains ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n{'='*60}")
        print(f"Phase 7: Multi-n Exploration (remaining {rem:.0f}s)")
        print(f"{'='*60}")
        for mult in [1.1, 1.2, 0.9]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            cand_n = int(n_best * mult)
            if cand_n == n_best or cand_n < 400 or cand_n > 3000:
                continue
            t_sub = min(rem * 0.3, 40.0)
            if t_sub < 15:
                break
            print(f"  Trying n={cand_n} (budget {t_sub:.0f}s)")
            init_cand = interpolate_sequence(best_seq, cand_n)
            a_c, sc_c = subgradient_v2(init_cand, t_sub, n_steps=10000)
            try_update(a_c, f"n{cand_n}")
    
    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        print(f"{'='*60}")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
