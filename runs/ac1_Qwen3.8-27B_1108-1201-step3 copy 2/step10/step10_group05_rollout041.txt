**Key mathematical insight**: The objective `2*n*max(conv) / (sum a)^2` with sum(a) = 1 is `2*n*max(conv)`.

The maximum of conv is achieved at some position k*. At the optimum, we expect:
- Multiple positions to be at the maximum (otherwise we could reduce the peak further)
- The gradient conditions to hold

For a KKT analysis:
- If a[i] > 0 (interior), then the subgradient of max(conv) dotted with the direction that increases a[i] (and decreases others to maintain sum=1) must be >= 0.
- If a[i] = 0 (boundary), then the subgradient condition is one-sided.

This suggests an approach: **identify the active peak positions, then solve the KKT system**.

If the active positions are k1, k2, ..., kP (where conv[k_j] = M for all j), and the active variables are i1, i2, ..., iq (where a[i_j] > 0), then:
- For each active position k_j: conv[k_j] = M (P equations)
- For each active variable i_j: sum of subgradient components = 0 (q equations)
- sum(a) = 1 (1 equation)
- Total: P + q + 1 equations in q + 1 unknowns (q values of a[i_j] plus M)

If P = q (which is typical at an optimal solution of a minimax problem), we have a square system that we can solve.

This is essentially the "active set" approach, which is what the LP does implicitly.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def multiplicative_weights_fp(a0, time_limit, eta_init=0.1, verbose=False, label=""):

        a = np.ones(n) / n

        a = a / s

    t0 = time.monotonic()
    it = 0
    eta = eta_init
    decay = 0.999
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        # Use average of top few peak positions for smoother gradient
        sorted_idx = np.argsort(conv)[::-1]
        top_m = min(5, 2*n-1)
        avg_peak = conv[sorted_idx[:top_m]].mean()
        
        # Subgradient: for each i, g[i] = average over top peaks of a[k-i]
        g = np.zeros(n)
        for k in sorted_idx[:top_m]:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                g[i] += a[i] * a[k-i]
        g /= top_m
        
        # Multiplicative weights update
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new /= s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eta *= decay
        
        if it % 20 == 0:
            print(f"    MW {label} it={it}: best={best_score:.6f}, eta={eta:.4f}")
    
    return best_a, best_score

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            offsets = np.arange(k_start - i, k_end - i + 1)
            coeffs = a[offsets] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 2.0 < d < ai * 2.0:
                        candidates.append(d)

            top_k = min(10, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 2.0 < d < ai * 2.0:
                            candidates.append(d)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            print(f"    SCD {label} sweep={sweeps}: best={best_score:.6f}")

    return best_a, best_score

def peak_subgradient_method(a0, time_limit, label=""):
    """Subgradient method focusing only on peak positions with adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    eta = 0.5
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        M = conv.max()

        # Find top peak positions
        threshold = M - 0.05 * M
        peak_idx = np.where(conv >= threshold)[0]

        if len(peak_idx) == 0:
            peak_idx = [int(np.argmax(conv))]

        # Compute subgradient (average over peaks)
        g = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                g[i] += a[i] * a[k - i]
        g /= len(peak_idx)
        g *= 2.0  # Factor of 2 from the objective

        # Step
        a_new = a - eta * g
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive step size
        if no_improve > 30:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-6:
                break
        elif it % 50 == 0:
            eta *= 0.98

        if it % 20 == 0:
            print(f"    PSG {label} it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, eta={eta:.5f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng, init_type=0):
    """Generate structured initializations."""
    if init_type == 0:
        return np.ones(n) / n
    elif init_type == 1:
        # Sine-based
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.sin(x) + 0.2 * np.sin(2 * x + 0.5)
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 2:
        # Cosine bell
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 3:
        # Random with smoothing
        a = rng.random(n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 4:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2)
        s = a.sum()
        return a / s
    else:
        # Weighted random
        w = rng.random(n)
        w = w ** 2
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n

def peak_targeted_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a0[i] * a0[k - i]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
