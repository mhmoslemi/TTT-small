```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a_fixed):
    """Build matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def solve_lp(a_fixed):
    """Solve the best-response LP."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    # LP: min M
    # s.t. T @ b - M <= 0  (size constraints)
    #      sum(b) = 1
    #      b >= 0
    #      M >= 0 (add for numerical stability)
    
    # A_ub: (size x n)  -M part -> [T, -1]
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # A_eq: sum(b) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: b >= 0, M >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        b_val = np.maximum(result.x[:n], 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val = b_val / s
        return b_val
    return None

def nesterov_lp(a0, time_limit, n_max_iters):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp(y)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        
        # Update y (dual variable)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Update a (primal variable) - Nesterov extrapolation
        a_new = y + (alpha / (it + 2)) * (y - a)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating (scipy HiGHS) with Nesterov acceleration
2. LogSumExp L-BFGS refinement with correct gradient
3. Multi-length exploration (n=800, 1200, 1500, 2000)

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_T(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def lp_best_response(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_T(a_fixed)
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b_val = np.maximum(res.x[:n], 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val /= s
        return b_val
    return None

def nesterov_lp_optimize(a0, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    prev_best = best_score
    stall_count = 0

        if b_new is None:

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)

            y = a.copy()
        else:
            y = y / sy

            momentum = (it - 1) / (it + 1)
        else:
            momentum = 0.0

        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

            else:

def logsumexp_gradient(a, T):
    """Compute LogSumExp objective and gradient (in a-space, projected to simplex)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp
    lse = max_b + T * np.log(sum_exp)
    obj = 2.0 * n * lse
    
    # Gradient: grad_a[j] = 2 * sum_k w_k * a_{k-j} * I(j in V_k) - 2 * (sum_k w_k * c_k) * ... 
    # Actually, for the simplex gradient (directly in a-space):
    # df/da_j = 2n * sum_k w_k * (2 * a_{k-j} * I(j in V_k))  [from dc_k/da_j = 2*a_{k-j} if j in V_k]
    # Wait, dc_k/da_j = a_{k-j} (if j in V_k) + a_j (if k-j in V_k)
    # If j in V_k, then k-j in V_k, so dc_k/da_j = 2 * a_{k-j}
    # So df/da_j = 2n * sum_k w_k * 2 * a_{k-j} * I(j in V_k)
    #            = 4n * sum_k w_k * a_{k-j} * I(j in V_k)
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Project to simplex: subtract mean (weighted by a)
    # For softmax parameterization: grad_z = a * (grad_a - sum(a*grad_a))
    # We return the "effective" gradient for use in L-BFGS on z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def lse_lbgfs_refine(a0, time_limit, T_stages):
    n = len(a0)
    # Work in log domain
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    for T_stage in T_stages:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros(n)
            a = np.exp(z - z.max())
            a = a / a.sum()
            obj, grad_z = logsumexp_gradient(a, T_stage)
            return obj, grad_z
        
        try:
            res = minimize(
                fun=func, x0=z0, jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-25, 'gtol': 1e-20,
                         'maxfun': 10000, 'disp': False}
            )
            z0 = res.x
        except Exception:
            pass
    
    z_max = z0.max()
    a_final = np.exp(z0 - z_max)
    a_final = a_final / a_final.sum()
    return a_final

def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating (scipy HiGHS) with Nesterov acceleration and adaptive restarts
2. LogSumExp L-BFGS refinement with temperature annealing
3. Multi-length exploration (n=800, 1200, 1500, 2000)
4. Adaptive time allocation with convergence detection
Key improvements: direct scipy linprog (vs cvxpy overhead), correct gradient,
finer temperature schedule, and structured restarts for escape from local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast, no external call)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T(a_fixed):
    """Build (2n-1) x n matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def lp_best_response(a_fixed):
    """Solve LP: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_T(a_fixed)

    # A_ub: [T | -1] @ [b; M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)

    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Bounds: all variables >= 0
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val
    except Exception:
        pass
    return None


def nesterov_lp_optimize(a0, time_limit):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    it = 0
    prev_best = best_score
    no_improve_iters = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b_new = lp_best_response(y)
        if b_new is None:
            break

        it += 1
        alpha = 2.0 / (it + 1)

        # Nesterov dual averaging
        y_prev = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Nesterov primal extrapolation
        if it > 1:
            momentum = (it - 1) / (it + 1)
        else:
            momentum = 0.0
        a_new = y + momentum * (y - y_prev)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve_iters = 0
        elif score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # Stall detection and structured restart
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it} ({elapsed:.0f}s): score={best_score:.8f}")
            if no_improve_iters >= 20000:
                # Perturb y with low-frequency noise
                x_lin = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 4)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x_lin + phase)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                no_improve_iters = 0
                # Also perturb a
                a = a + 0.02 * np.cos(2 * x_lin + phase * 0.7)
                a = np.maximum(a, 0)
                sa2 = a.sum()
                if sa2 > 1e-15:
                    a = a / sa2

    return best_a, best_score


def logsumexp_obj_and_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient in z-space (softmax param)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp
    lse = max_b + T * np.log(sum_exp)
    obj = 2.0 * n * lse

    # Gradient computation:
    # dc_k/da_j = 2*a_{k-j} if j in V_k else 0
    # df/da_j = 2n * sum_k w_k * dc_k/da_j = 4n * sum_k w_k * a_{k-j} * I(j in V_k)
    # df/dz_j = a_j * (df/da_j - sum_i a_i * df/da_i)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]

    # Scale: 4n * grad_a is df/da, but for L-BFGS the constant doesn't matter much
    S = np.dot(grad_a, a)
    grad_z = 4.0 * n * a * (grad_a - S)
    return obj, grad_z


def lse_refine(a0, time_limit, T_stages, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing (z-space)."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros(n)
            z_shift = z_val - z_val.max()
            a_val = np.exp(z_shift)
            ss = a_val.sum()
            if ss < 1e-300:
                a_val = np.ones(n) / n
            else:
                a_val = a_val / ss
            obj, g = logsumexp_obj_and_grad(a_val, T_stage)
            return obj, g

        try:
            res = minimize(
                fun=func, x0=z, jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_max = z.max()
    a_final = np.exp(z - z_max)
    ss = a_final.sum()
    if ss < 1e-300:
        a_final = np.ones(n) / n
    else:
        a_final = a_final / ss
    return a_final


def interpolated_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multiplicative_descent(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) on active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score
```
