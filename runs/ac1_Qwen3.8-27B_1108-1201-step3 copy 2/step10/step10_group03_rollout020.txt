def pgy_optimize(a0, time_limit):
    """
    Bilinear game solving with exponentiated gradient + self-play polish.
    """
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    
    # Nesterov averaging
    a_av = a.copy()
    b_av = b.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Exponentiated gradient parameters
    # We'll use a temperature schedule and adaptive learning rate
    
    # Log parameters for a and b (for multiplicative updates)
    log_a = np.log(a + 1e-20)
    log_b = np.log(b + 1e-20)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / time_limit
        
        # Temperature: from n (smooth) to 1 (sharp)
        T = max(1.0, n * (1 - progress) ** 1.5)
        
        # Learning rate (decaying)
        eta = 5.0 / (it + 10)
        
        # Current strategies
        a_cur = np.exp(log_a - log_a.max())
        a_cur /= a_cur.sum()
        b_cur = np.exp(log_b - log_b.max())
        b_cur /= b_cur.sum()
        
        # Cross-convolution
        c = cross_convolve(a_cur, b_cur)  # length 2n-1
        c_max = c.max()
        
        # Softmax weights
        p = np.exp((c - c_max) / T)
        p /= p.sum()
        
        # Gradient for a: g_a[j] = sum_k p[k] * b_cur[k-j]
        g_a = compute_grad(p, b_cur)
        # Gradient for b: g_b[j] = sum_k p[k] * a_cur[k-j]
        g_b = compute_grad(p, a_cur)
        
        # Multiplicative update (exponentiated gradient)
        log_a -= eta * g_a
        log_b -= eta * g_b
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        a_new = np.exp(log_a - log_a.max())
        a_new /= a_new.sum()
        b_new = np.exp(log_b - log_b.max())
        b_new /= b_new.sum()
        
        a_av = (1 - alpha) * a_av + alpha * a_new
        b_av = (1 - alpha) * b_av + alpha * b_new
        
        # Evaluate
        sc_a = 2.0 * n * fast_convolve(a_av).max()
        sc_b = 2.0 * n * fast_convolve(b_av).max()
        sc_a_new = 2.0 * n * fast_convolve(a_new).max()
        sc_b_new = 2.0 * n * fast_convolve(b_new).max()
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_av.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b_av.copy()
        if sc_a_new < best_score:
            best_score = sc_a_new
            best_a = a_new.copy()
        if sc_b_new < best_score:
            best_score = sc_b_new
            best_a = b_new.copy()
    
    return best_a, best_score

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def cross_corr(p, a):
    """cross_corr[j] = sum_k p[k] * a[k-j] for j=0,...,n-1"""
    n = len(a)
    p_len = len(p)
    size = p_len + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a[::-1], fft_size)
    c = irfft(P * A, fft_size)
    return c[n-1:2*n-1]
