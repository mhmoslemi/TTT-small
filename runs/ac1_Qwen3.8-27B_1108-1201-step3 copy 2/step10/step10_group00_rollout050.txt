"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated methods:
1. Nesterov-accelerated exact subgradient (O(n log n) per iter via FFT)
2. LogSumExp L-BFGS with fine temperature annealing (FFT-based gradient)
3. LP best-response for final polish
4. Multi-length exploration
Key improvement: O(n log n) gradient via FFT instead of O(n^2) loop,
enabling 10-100x more optimizer iterations per second.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Linear convolution of two arrays via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, n, T):
    """LogSumExp objective and gradient w.r.t. z using FFT. O(n log n)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    
    smooth_max = max_b + T * np.log(w_sum)
    obj = 2.0 * n * smooth_max
    
    # grad_a[i] = 2 * sum_j w[j] * a[j-i]
    # = 2 * conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1]
    c = convolve_pair(w_rev, a)
    # c has length (2n-1) + n - 1 = 3n-2
    # We need c[2n-2-i] for i=0,...,n-1
    # That's c[n-1 : 2n-1] reversed
    grad_a = 2.0 * c[n-1:2*n-1][::-1]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z


def exact_subgrad_z(z, n, active_tol=1e-6):
    """Exact subgradient of 2n*max(conv(a,a)) w.r.t. z using FFT."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Active set: indices where b[k] is within active_tol of max
    threshold = max_b - active_tol * max_b
    active = np.where(b >= threshold)[0]
    
    # Weight: uniform over active set
    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    
    # grad_a[i] = 2n * 2 * sum_j w[j] * a[j-i]
    # = 4n * conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1]
    c = convolve_pair(w_rev, a)
    grad_a = 4.0 * n * c[n-1:2*n-1][::-1]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return 2.0 * n * max_b, 2.0 * n * grad_z


def nesterov_subgrad(a0, time_limit, n_max_iters=5000000):
    """Nesterov-accelerated subgradient method on the probability simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    
    t0 = time.monotonic()
    a_prev = a.copy()
    
    # Step size parameters
    f_lb = 2.0 * n / (2 * n - 1)  # Theoretical lower bound
    eta_base = 0.1
    eta_scale = 1.0
    no_improve = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        if it == 0:
            y = a.copy()
            beta = 0.0
        else:
            beta = (it - 1) / (it + 1)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y / s_y
        
        a_prev = a.copy()
        
        # Compute subgradient at y
        b_y = fast_convolve(y)
        max_b_y = b_y.max()
        threshold = max_b_y - 1e-6 * max_b_y
        active = np.where(b_y >= threshold)[0]
        
        # grad_a[i] = 2 * sum_j w[j] * a[j-i] where w = 1/|S| on S
        w = np.zeros(len(b_y))
        w[active] = 1.0 / len(active)

        c = convolve_pair(w_rev, y)
        grad_a = 2.0 * c[n-1:2*n-1][::-1]
        
        # Project gradient onto tangent space of simplex
        grad_a = grad_a - np.mean(grad_a)
        
        # Step size (Polyak-like)
        current_val = 2.0 * n * max_b_y
        eta = eta_base * eta_scale / np.sqrt(it + 1)
        
        # Update
        a_new = y - eta * grad_a
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt step size
        if no_improve == 50:

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta_scale={eta_scale:.4f}")
    
    return best_a, best_score


def lbgfs_lse(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing, FFT-based."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    best_z = z.copy()
    best_obj = float('inf')
    
    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        
        def func_wrapper(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj
        
        def jac_wrapper(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func_wrapper, x0=z, jac=jac_wrapper, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
            obj_check, _ = lse_obj_grad_z(z, n, stage_T)
            # Track best in terms of the actual max objective
            a_check = softmax_from_z(z)
            actual_score = 2.0 * n * fast_convolve(a_check).max()
            if actual_score < best_obj:
                best_obj = actual_score
                best_z = z.copy()
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} (T={stage_T}) failed: {e}")
            z = best_z.copy()
    
    a_result = softmax_from_z(best_z)
    return a_result


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T
