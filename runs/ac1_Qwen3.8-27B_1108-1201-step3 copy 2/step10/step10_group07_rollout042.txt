def top_weights(c, top_k=20, beta=5000.0):
    m=2*c.size-1
    k=min(top_k,m)
    idx=np.argpartition(c, -k)[-k:]
    w=np.zeros(m)
    vals=c[idx]
    z=beta*(vals - vals.max())
    ez=np.exp(z)
    w[idx]=ez/ez.sum()
    return w

Strategy:

def normalize_sequence(a):
    """Clip to nonnegative, remove nonfinite values, and make sum = 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return None
    return a / s


def convolve_fft(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    if b is a:
        B = A
    else:
        B = rfft(b, fft_size)
    out = irfft(A * B, fft_size)
    return out[:size]


def make_conv(a):
    """Convolution used by local search; direct for moderate n, FFT for large n."""
    n = len(a)
    if n <= 1536:
        return np.convolve(a, a, mode="full")
    return convolve_fft(a)


def score_fast(a):
    """Fast FFT-based score.  Used inside LP/smooth searches."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    c = convolve_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def score_best(a):
    """More reliable score: direct for n <= 4096, FFT otherwise."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    if n <= 4096:
        c = np.convolve(a, a, mode="full")
    else:
        c = convolve_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def top_weights(c, top_k, beta):
    """Soft weights on the top convolution peaks."""
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    k = min(max(1, int(top_k)), m)
    w = np.zeros(m, dtype=np.float64)

    if k == m:
        idx = np.arange(m)
        vals = c
    else:
        idx = np.argpartition(c, -k)[-k:]
        vals = c[idx]

    if vals.size == 0:
        w[:] = 1.0 / m
        return w

    vmax = float(np.max(vals))
    z = beta * (vals - vmax)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        w[idx] = 1.0 / len(idx)
    else:
        w[idx] = ez / s
    return w


def gradient_from_weights(a, w):
    """
    Gradient of sum_k w_k conv(a,a)_k with respect to a.

    For c_k = sum_i a_i a_{k-i}, dc_k/da_i = 2 a_{k-i} when valid.
    """
    n = len(a)
    ar = a[::-1]
    h = convolve_fft(ar, w)
    return 2.0 * h[n - 1:n - 1 + n]


def make_smooth_noise(rng, n, length_scale=5.0):
    """Smooth zero-mean Gaussian noise, circular spectral low-pass."""
    x = rng.standard_normal(n)
    X = np.fft.rfft(x)
    f = np.fft.rfftfreq(n) * n
    mask = np.exp(-(f / max(1.0, length_scale)) ** 2)
    y = np.fft.irfft(X * mask, n)
    y -= float(y.mean())
    return y


def gaussian_smooth(a, sigma):
    """Simple Gaussian smoothing."""
    a = np.asarray(a, dtype=np.float64)
    radius = max(1, int(6.0 * sigma))
    k = np.arange(-radius, radius + 1, dtype=np.float64)
    kern = np.exp(-(k * k) / (2.0 * sigma * sigma))
    kern /= kern.sum()
    return np.convolve(a, kern, mode="same")


def resample_sequence(a, new_n):
    """Linear resampling on [0,1]."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def eval_pair_move(a, conv, prefix, suffix, d, r, fracs, current_M):
    """
    Try moving mass delta from d to r:
        a_d' = a_d - delta, a_r' = a_r + delta.
    Sum remains 1.  Return (new_M, delta, new_affected_conv) or None.
    """
    a_d = float(a[d])
    if a_d <= 1e-18:
        return None

    n = len(a)
    k_start = min(d, r)
    k_end = max(d, r) + n - 1
    if k_end > 2 * n - 2:
        k_end = 2 * n - 2
    L = k_end - k_start + 1

    conv_aff = conv[k_start:k_end + 1].copy()
    unaff = max(float(prefix[k_start]), float(suffix[k_end + 1]))

    Lvec = np.zeros(L, dtype=np.float64)
    Qvec = np.zeros(L, dtype=np.float64)

    # Donor d: linear term -2 delta a_{k-d}
    ds = max(k_start, d)
    de = min(k_end, d + n - 1)
    if ds <= de:
        Lvec[ds - k_start:de - k_start + 1] -= 2.0 * a[ds - d:de - d + 1]

    # Receiver r: linear term +2 delta a_{k-r}
    rs = max(k_start, r)
    re = min(k_end, r + n - 1)
    if rs <= re:
        Lvec[rs - k_start:re - k_start + 1] += 2.0 * a[rs - r:re - r + 1]

    # Quadratic terms.
    if k_start <= 2 * d <= k_end:
        Qvec[2 * d - k_start] += 1.0
    if k_start <= 2 * r <= k_end:
        Qvec[2 * r - k_start] += 1.0
    kk = d + r
    if d != r and k_start <= kk <= k_end:
        Qvec[kk - k_start] -= 2.0

    deltas = a_d * fracs
    mask = (deltas > 1e-18) & (deltas < a_d * (1.0 - 1e-12))
    deltas = deltas[mask]
    if deltas.size == 0:
        return None

    d2 = deltas * deltas
    vals = conv_aff[np.newaxis, :] + Lvec[np.newaxis, :] * deltas[:, np.newaxis] \
        + Qvec[np.newaxis, :] * d2[:, np.newaxis]
    vals = np.maximum(vals, 0.0, out=vals)

    aff_max = vals.max(axis=1)
    new_M_arr = np.maximum(unaff, aff_max)
    idx = int(np.argmin(new_M_arr))
    best_M = float(new_M_arr[idx])

    if not np.isfinite(best_M) or best_M >= current_M - 1e-15:
        return None

    return best_M, float(deltas[idx]), vals[idx]


def apply_pair_move(a, conv, d, r, delta, new_aff):
    n = len(a)
    a[d] = max(float(a[d]) - float(delta), 0.0)
    a[r] = float(a[r]) + float(delta)

    k_start = min(d, r)
    k_end = max(d, r) + n - 1
    if k_end > 2 * n - 2:
        k_end = 2 * n - 2

    conv[k_start:k_end + 1] = new_aff

    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a[:] = 1.0 / n
        conv[:] = 0.0
        return

    if abs(s - 1.0) > 1e-10:
        a /= s
        conv /= (s * s)

    conv = np.maximum(conv, 0.0, out=conv)


def eval_coord_move(a, conv, prefix, suffix, i, fracs, current_M):
    """
    Try changing coordinate i and then normalizing:
        b_i = a_i + delta, others unchanged, a' = b / (1 + delta).
    Return (new_M, delta, new_affected_raw_conv, s2) or None.
    """
    a_i = float(a[i])
    if a_i <= 1e-18:
        # Still allow a tiny creation if delta positive?  fracs are relative to a_i, so no.
        return None

    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    L = k_end - k_start + 1

    conv_aff = conv[k_start:k_end + 1].copy()
    unaff = max(float(prefix[k_start]), float(suffix[k_end + 1]))

    a_seg = a[0:L]
    Qvec = np.zeros(L, dtype=np.float64)
    if k_start <= 2 * i <= k_end:
        Qvec[2 * i - k_start] = 1.0

    deltas = a_i * fracs
    new_raw_sum = 1.0 + deltas
    mask = (new_raw_sum > 1e-12) & ((a_i + deltas) > -1e-15)
    deltas = deltas[mask]
    new_raw_sum = new_raw_sum[mask]
    if deltas.size == 0:
        return None

    d2 = deltas * deltas
    vals = conv_aff[np.newaxis, :] \
        + (2.0 * deltas)[:, np.newaxis] * a_seg[np.newaxis, :] \
        + Qvec[np.newaxis, :] * d2[:, np.newaxis]
    vals = np.maximum(vals, 0.0, out=vals)

    s2 = new_raw_sum * new_raw_sum
    aff_max = vals.max(axis=1)
    new_M_arr = np.maximum(unaff, aff_max) / s2
    idx = int(np.argmin(new_M_arr))
    best_M = float(new_M_arr[idx])

    if not np.isfinite(best_M) or best_M >= current_M - 1e-15:
        return None

    return best_M, float(deltas[idx]), vals[idx], float(s2[idx])


def apply_coord_move(a, conv, i, delta, new_aff_raw, s2):
    n = len(a)
    old_i = float(a[i])
    new_i = max(old_i + float(delta), 0.0)
    new_raw_sum = 1.0 + float(delta)

    if new_raw_sum <= 1e-12:
        return

    a[i] = new_i
    a /= new_raw_sum

    scale = 1.0 / (new_raw_sum * new_raw_sum)

    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)

    conv[k_start:k_end + 1] = new_aff_raw * scale
    if k_start > 0:
        conv[:k_start] *= scale
    if k_end + 1 < conv.size:
        conv[k_end + 1:] *= scale

    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a[:] = 1.0 / n
        conv[:] = 0.0
        return

    if abs(s - 1.0) > 1e-10:
        a /= s
        conv /= (s * s)

    conv = np.maximum(conv, 0.0, out=conv)


def run_local(a0, time_limit, verbose=True, label="", rng=None, fine=False,
              top_k=25, beta=3000.0):
    """
    Sum-preserving best-first local search.

    Each round:
    - compute peak weights from the current max convolution,
    - compute the gradient of the weighted peak objective,
    - evaluate a set of pair transfers and normalized single-coordinate moves,
    - apply the best improving exact move.

    The returned score is the fast internal score 2*n*max(conv) for sum=1.
    """
    if rng is None:
        rng = np.random.default_rng(123456789)

    a = normalize_sequence(a0)
    if a is None:
        return None, float("inf")

    t0 = time.monotonic()
    n = len(a)
    conv = make_conv(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = float(2.0 * n * M)

    rounds = 0
    no_improve = 0
    beta_cur = float(beta)
    top_cur = int(top_k)

    if fine:
        pair_fr = FINE_PAIR_FRACS.copy()
        coord_fr = FINE_COORD_FRACS.copy()
        coord_count = min(80, max(16, 120000 // max(1, n)))
        pair_count = min(8, max(2, 12000 // max(1, n)))
    else:
        pair_fr = PAIR_FRACS.copy()
        coord_fr = COORD_FRACS.copy()
        coord_count = min(55, max(18, 55000 // max(1, n)))
        pair_count = min(10, max(3, 11000 // max(1, n)))

    if coord_count > n:
        coord_count = n
    if pair_count > n:
        pair_count = n

    while time.monotonic() - t0 < time_limit:
        rounds += 1

        # Periodic exact cleanup.
        if rounds % 25 == 0:
            a = normalize_sequence(a)
            if a is None:
                break
            conv = make_conv(a)
            M = float(conv.max())
            sc = float(2.0 * n * M)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        m = conv.size
        prefix = np.empty(m + 1, dtype=np.float64)
        prefix[0] = -1e30
        prefix[1:] = np.maximum.accumulate(conv)

        suffix = np.empty(m + 1, dtype=np.float64)
        suffix[m] = -1e30
        suffix[m - 1::-1] = np.maximum.accumulate(conv[::-1])[::-1]

        w = top_weights(conv, top_cur, beta_cur)
        g = gradient_from_weights(a, w)

        if not np.all(np.isfinite(g)):
            a = best_a.copy()
            conv = make_conv(a)
            M = float(conv.max())
            continue

        base = float(np.dot(g, a))
        abs_dev = np.abs(g - base)

        if coord_count < n:
            coord_idx = np.argpartition(abs_dev, coord_count - 1)[-coord_count:]
        else:
            coord_idx = np.arange(n)

        order = np.argsort(g)
        recipients = order[:pair_count]
        donors = order[::-1][:pair_count]

        best_m = M
        best_move = None

        # Single-coordinate moves with renormalization.
        for ii in coord_idx:
            i = int(ii)
            res = eval_coord_move(a, conv, prefix, suffix, i, coord_fr, M)
            if res is not None and res[0] < best_m:
                best_m = res[0]
                best_move = ("c", i, res[1], res[2], res[3])

        # Pair transfer moves.
        pair_rec_count = min(3, pair_count)
        for dd in donors:
            d = int(dd)
            gd = float(g[d])
            for rr in recipients[:pair_rec_count]:
                r = int(rr)
                if d == r:
                    continue
                if g[r] > gd - 1e-14:
                    continue
                res = eval_pair_move(a, conv, prefix, suffix, d, r, pair_fr, M)
                if res is not None and res[0] < best_m:
                    best_m = res[0]
                    best_move = ("p", d, r, res[1], res[2])

        if best_move is not None:
            if best_move[0] == "c":
                apply_coord_move(a, conv, best_move[1], best_move[2], best_move[3], best_move[4])
            else:
                apply_pair_move(a, conv, best_move[1], best_move[2], best_move[3], best_move[4])

            M = float(conv.max())
            sc = float(2.0 * n * M)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            if no_improve == 4:
                beta_cur = min(60000.0, beta_cur * 1.7)
                top_cur = max(8, top_cur // 2)
            elif no_improve == 10:
                beta_cur = max(800.0, beta_cur * 0.5)
                top_cur = min(120, top_cur * 2)
                if fine:
                    pair_fr = PAIR_FRACS.copy()
                    coord_fr = COORD_FRACS.copy()
                else:
                    pair_fr = FINE_PAIR_FRACS.copy()
                    coord_fr = FINE_COORD_FRACS.copy()
            elif no_improve in (20, 30, 40):
                eps = 0.002 * (1.0 + 0.2 * float(no_improve))
                noise = make_smooth_noise(rng, n, max(2.0, float(n) / 64.0))
                a_try = a + eps * noise
                a_try = normalize_sequence(a_try)
                if a_try is not None:
                    conv_try = make_conv(a_try)
                    M_try = float(conv_try.max())
                    best_M = best_score / (2.0 * n) if best_score > 0 else M
                    if M_try < max(M * 1.05, best_M * 1.02):
                        a = a_try
                        conv = conv_try
                        M = M_try
                    else:
                        a = best_a.copy()
                        conv = make_conv(a)
                        M = float(conv.max())
                    no_improve = 0

            if no_improve > 45:
                break

        if verbose and rounds % 25 == 0:
            print(f"    local[{label}] round={rounds} t={time.monotonic()-t0:.1f}s "
                  f"score={best_score:.8f} M={M:.10f}")

    a_final = normalize_sequence(best_a)
    if a_final is None:
        return None, float("inf")
    cv = make_conv(a_final)
    return a_final, float(2.0 * len(a_final) * cv.max())

def analytic_init(n, kind, rng):
    t = np.linspace(0.0, 1.0, n)

    if kind == 0:
        a = np.ones(n, dtype=np.float64)
    elif kind == 1:
        a = np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.0)
    elif kind == 2:
        a = 0.2 + 0.8 * t
    elif kind == 3:
        a = np.exp(2.0 * (t - 0.5))
    elif kind == 4:
        shift = float(rng.uniform(0.0, 0.5))
        a = 0.1 + 0.9 * (0.5 + 0.5 * np.cos(2.0 * np.pi * (t - shift))) ** 2
    elif kind == 5:
        c1 = float(rng.uniform(0.2, 0.4))
        c2 = float(rng.uniform(0.6, 0.8))
        w = float(rng.uniform(0.05, 0.15))
        a = np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 0.05
    elif kind == 6:
        a = 0.3 + 0.7 * np.sin(np.pi * t) ** 3
    else:
        a = np.sin(np.pi * t) ** 2 + 0.2 * t + 0.05

    return normalize_sequence(a)


# ---------------------------------------------------------------------
# LP-based alternating minimization.
# ---------------------------------------------------------------------

def find_lp_solver():
    cached = getattr(find_lp_solver, "_cached", None)
    if cached is not None:
        return cached

    for name in ["PDLP", "ECOS", "SCS", "CLARABEL", "ECOS_BB", "GLPK"]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ("optimal", "optimal_inaccurate"):
                find_lp_solver._cached = name
                return name
        except Exception:
            continue

    find_lp_solver._cached = "ECOS"
    return "ECOS"


def lp_alternating(a0, time_limit, verbose=True):
    a = normalize_sequence(a0)
    if a is None:
        return None, float("inf")

    t0 = time.monotonic()
    n = len(a)
    best_a = a.copy()
    best_score = score_fast(a)

    solver = find_lp_solver()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        c = np.concatenate((a, np.zeros(n - 1, dtype=np.float64)))
        r = np.zeros(n, dtype=np.float64)
        T = toeplitz(c, r)

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable(nonneg=True)
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver="ECOS", verbose=False)
            except Exception:
                break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64).ravel()
        b_val = np.maximum(b_val, 0.0)
        s = float(b_val.sum())
        if not np.isfinite(s) or s <= 1e-12:
            break
        b_val /= s

        sc_b = score_fast(b_val)
        if np.isfinite(sc_b) and sc_b < best_score:
            best_score = sc_b
            best_a = b_val.copy()

        if it <= 12:
            alpha = 0.85
        else:
            alpha = max(0.25, 8.0 / float(it + 10.0))

        a_new = (1.0 - alpha) * a + alpha * b_val
        a = normalize_sequence(a_new)
        if a is None:
            break

        sc_a = score_fast(a)
        if np.isfinite(sc_a) and sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and it % 5 == 0:
            print(f"    LP it={it} t={time.monotonic()-t0:.1f}s score={best_score:.8f}")

    return best_a, best_score


# ---------------------------------------------------------------------
# Smooth L-BFGS on softmax parameterization.
# ---------------------------------------------------------------------

_SMOOTH = {
    "beta": 1000.0,
    "cache_valid": False,
    "value": 0.0,
    "grad": None
}

_CALLBACK = {
    "sub_deadline": 0.0,
    "total_deadline": 0.0
}


def smooth_obj_grad(y):
    beta = float(_SMOOTH["beta"])
    y = np.asarray(y, dtype=np.float64)
    z = y - float(np.max(y))
    e = np.exp(z)
    a = e / float(e.sum())

    n = len(a)
    c = convolve_fft(a)
    cmax = float(c.max())
    zc = beta * (c - cmax)
    ez = np.exp(zc)
    zsum = float(ez.sum())
    if zsum <= 0.0 or not np.isfinite(zsum):
        zsum = 1.0
        ez = np.ones_like(ez)

    value = float(cmax + np.log(zsum) / beta)
    w = ez / zsum

    ar = a[::-1]
    h = convolve_fft(ar, w)
    g = 2.0 * h[n - 1:n - 1 + n]
    avg = float(np.dot(a, g))
    grad = a * (g - avg)

    return value, grad


def smooth_fun(y):
    f, g = smooth_obj_grad(y)
    _SMOOTH["cache_valid"] = True
    _SMOOTH["value"] = f
    _SMOOTH["grad"] = g
    return f


def smooth_jac(y):
    if _SMOOTH["cache_valid"]:
        _SMOOTH["cache_valid"] = False
        return _SMOOTH["grad"]
    return smooth_obj_grad(y)[1]


def smooth_callback(pk):
    now = time.monotonic()
    return now < min(_CALLBACK["sub_deadline"], _CALLBACK["total_deadline"])


def run_smooth(a0, time_budget):
    a = normalize_sequence(a0)
    if a is None:
        return None, float("inf")

    t0 = time.monotonic()
    total_deadline = t0 + max(10.0, float(time_budget))
    n = len(a)

    y0 = np.log(np.maximum(a, 1e-300))
    y0 -= float(y0.mean())

    bounds = [(-60.0, 60.0)] * n
    best_a = a.copy()
    best_score = score_fast(a)

    betas = [2000.0, 8000.0, 25000.0]

    for bi, beta in enumerate(betas):
        if time.monotonic() >= total_deadline - 20.0:
            break

        remaining_betas = len(betas) - bi
        alloc = (total_deadline - time.monotonic()) / float(remaining_betas)
        sub_deadline = min(total_deadline, time.monotonic() + max(15.0, alloc * 1.4))

        _SMOOTH["beta"] = beta
        _SMOOTH["cache_valid"] = False
        _CALLBACK["sub_deadline"] = sub_deadline
        _CALLBACK["total_deadline"] = total_deadline

        try:
            res = minimize(
                smooth_fun,
                y0,
                method="L-BFGS-B",
                jac=smooth_jac,
                bounds=bounds,
                callback=smooth_callback,
                options={
                    "maxiter": 3000,
                    "maxfun": 8000,
                    "ftol": 1e-14,
                    "gtol": 1e-9,
                    "maxls": 25
                }
            )
            y = np.asarray(res.x, dtype=np.float64)
        except Exception:
            continue

        z = y - float(np.max(y))
        e = np.exp(z)
        a_new = e / float(e.sum())
        a_new = normalize_sequence(a_new)
        if a_new is None:
            continue

        sc = score_fast(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            y0 = y0.copy()  # keep last y for next beta

        y0 = y

    return best_a, best_score


# ---------------------------------------------------------------------
# Fourier-parameter diversity search.
# ---------------------------------------------------------------------

_FOURIER = {
    "B": None,
    "base": None,
    "n": 0
}


def fourier_obj(p):
    B = _FOURIER["B"]
    base = _FOURIER["base"]
    v = base + B @ np.asarray(p, dtype=np.float64)
    z = v - float(np.max(v))
    e = np.exp(z)
    a = e / float(e.sum())
    return score_fast(a)


def run_fourier(a0, time_limit, rng):
    a = normalize_sequence(a0)
    if a is None:
        return None, float("inf")

    t0 = time.monotonic()
    n = len(a)

    if n <= 1200:
        K = 5
        maxiter = 45
    elif n <= 2400:
        K = 4
        maxiter = 32
    else:
        K = 3
        maxiter = 24

    if maxiter > 40:
        maxiter = 40

    i = np.arange(n, dtype=np.float64)
    cols = []
    for m in range(1, K + 1):
        cols.append(np.cos(2.0 * np.pi * m * i / float(n)))
        cols.append(np.sin(2.0 * np.pi * m * i / float(n)))

    if not cols:
        return a, score_fast(a)

    B = np.column_stack(cols)
    base = np.log(np.maximum(a, 1e-300))
    base -= float(base.mean())

    _FOURIER["B"] = B
    _FOURIER["base"] = base
    _FOURIER["n"] = n

    bounds = [(-3.0, 3.0)] * B.shape[1]
    best_a = a.copy()
    best_score = score_fast(a)

    try:
        seed = int(rng.integers(1, 2**31 - 1))
        res = differential_evolution(
            fourier_obj,
            bounds,
            maxiter=maxiter,
            popsize=10,
            tol=1e-5,
            seed=seed,
            polish=False
        )

        p = np.asarray(res.x, dtype=np.float64)
        v = base + B @ p
        z = v - float(np.max(v))
        e = np.exp(z)
        a_new = e / float(e.sum())
        a_new = normalize_sequence(a_new)
        if a_new is not None:
            sc = score_fast(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    except Exception:
        pass

    # If we still have a little time, do a quick local pass on the best Fourier result.
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20.0:
        try:
            a_loc, sc_loc = run_local(best_a, min(remaining * 0.7, 60.0),
                                      verbose=False, label="fourier_local", rng=rng)
            if a_loc is not None and np.isfinite(sc_loc) and sc_loc < best_score:
                best_a = a_loc
                best_score = sc_loc
        except Exception:
            pass

    return best_a, best_score

            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and float(p.sum()) > 0.01:
                return [float(x) for x in p]
        except Exception:
            pass
