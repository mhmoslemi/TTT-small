    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Anneal t from t_start to t_end
    n_iters = 200
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t = t_start * (t_end / t_start) ** (it / n_iters)
        
        # Compute softmax weights
        conv = fast_convolve(a)
        # Shift to avoid overflow
        conv_shifted = conv - conv.max()
        exp_vals = np.exp(t * conv_shifted)
        p = exp_vals / exp_vals.sum()
        
        # Gradient: grad_j = 2 * sum_k p_k * a_{k-j}
        grad = np.zeros(n)
        for k in range(2*n-1):
            if p[k] < 1e-15:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n-1, k)
            partners = k - np.arange(i_start, i_end+1)
            grad[i_start:i_end+1] += 2.0 * p[k] * a[partners]

        lr = 1.0 / (t * n)
        for step in range(50):
            a_new = a - lr * grad
            a_new = simplex_project(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            # Recompute gradient
            conv = fast_convolve(a)
            conv_shifted = conv - conv.max()

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None
_LP_SOLVER_TRIED = False

def get_lp_solver(force_retry=False):
    global _LP_SOLVER, _LP_SOLVER_TRIED
    if _LP_SOLVER is not None and not force_retry:
        return _LP_SOLVER
    if _LP_SOLVER_TRIED and not force_retry:
        return _LP_SOLVER
    import cvxpy as cp
    candidates = [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CLARABEL]
    for name in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            c = np.ones(10)
            prob = cp.Problem(cp.Minimize(c @ x), [cp.sum(x) == 1, x <= 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER_TRIED = True
    # Fallback to ECOS which is always available
    return cp.ECOS

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix_rows(a, lags):
    """Build rows of the convolution matrix for specific lags only."""
    n = len(a)
    rows = np.zeros((len(lags), n))
    for row_idx, k in enumerate(lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows[row_idx, i] = a[k - i]
    return rows

def solve_lp_cutting_plane(a_fixed, solver, max_iters=20, top_k=50):
    """
    Solve min M s.t. (T_a @ b)_k <= M, sum(b)=1, b>=0
    using cutting plane (start with top-k constraints, add violated ones).
    """
    import cvxpy as cp
    n = len(a_fixed)
    
    # Initial active set: top-k lags of a*a plus edges
    conv_a = fast_convolve(a_fixed)
    top_lags = np.argpartition(conv_a, -top_k)[-top_k:]
    # Always include edges
    for edge in [0, 1, 2, 2*n-3, 2*n-2, 2*n-4]:
        if edge not in top_lags:
            top_lags = np.append(top_lags, edge)
    active_lags = set(int(x) for x in top_lags)
    
    b_val = None
    for cp_iter in range(max_iters):
        lags_sorted = sorted(active_lags)
        T_rows = build_conv_matrix_rows(a_fixed, lags_sorted)
        
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_rows @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None, None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        
        b_candidate = np.asarray(b_var.value, dtype=np.float64)
        b_candidate = np.maximum(b_candidate, 0)
        s = b_candidate.sum()
        if s < 1e-15:
            return None, None
        b_candidate = b_candidate / s
        
        # Check all constraints
        conv_ab = T_all = None  # full convolution
        # Compute a * b
        full_conv = fast_convolve(a_fixed)  # this is a*a, not a*b!
        # We need a*b, not a*a
        # Let me compute it properly
        fft_size = 1 << (2*n - 1 - 1).bit_length()
        A = rfft(a_fixed, fft_size)
        B = rfft(b_candidate, fft_size)
        conv_ab = irfft(A * B, fft_size)[:2*n-1]
        
        M_candidate = float(M_var.value)
        max_conv = float(conv_ab.max())
        
        if max_conv <= M_candidate + 1e-9:
            # All constraints satisfied, we have the optimal solution
            return b_candidate, max_conv
        
        # Add violated constraints
        violated = np.where(conv_ab > M_candidate + 1e-10)[0]
        for v in violated:
            active_lags.add(int(v))
        
        b_val = b_candidate
    
    return b_val, float(fast_convolve(a_fixed).max())  # fallback

def softmax_gradient_convolve(a, p, n):
    """Compute gradient: grad_j = 2 * sum_k p_k * a_{k-j} using FFT."""
    # This is the cross-correlation of p and a, times 2
    # cross_corr[j] = sum_k p[k] * a[k-j] (with a[m]=0 for invalid m)
    # = (p * flip(a))[j + offset]  ... let me think
    # Actually: sum_k p[k] * a[k-j] for k-j in [0,n-1]
    # Let's define q[m] = a[m] for m in [0,n-1], 0 else
    # Then sum_k p[k] * q[k-j] = (p * flip(q))[j + (n-1)] ... 
    # Simpler: use the convolution directly
    # (p * a)[j] = sum_k p[k] * a[j-k]
    # We want sum_k p[k] * a[k-j] = (p * flip(a))[j]
    # where flip(a)[m] = a[n-1-m]
    # Hmm, let me just use direct FFT correlation
    
    fft_size = 1 << (2*n - 1 - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(conj(P) * A) gives sum_k p[k-j] * a[k]
    # We want sum_k p[k] * a[k-j]
    # = ifft(conj(A) * P) ... let me verify
    # (f * g)[n] = sum_k f[k] * g[n-k]
    # ifft(FFT(f) * FFT(g))[n] = sum_k f[k] * g[(n-k) mod N]
    # For linear: zero-pad to size >= len(f)+len(g)-1
    # sum_k p[k] * a[k-j] = (p * flip_a)[j'] where flip_a[m] = a[-m]
    # Actually the simplest: just compute (p * a) and then flip
    # (p*a)[j] = sum_k p[k] * a[j-k]
    # We want c[j] = sum_k p[k] * a[k-j]
    # c[j] = (p * a)[n-1-j] ... no that's not right either
    
    # Let me just do it directly with the definition
    # c[j] = sum_{k: k-j >= 0, k-j < n} p[k] * a[k-j]
    #      = sum_{m >= 0, m < n} p[m+j] * a[m]  (substituting m = k-j)
    # This is the cross-correlation of p and a at lag j
    
    # Using FFT: cross_corr = ifft(conj(FFT(a)) * FFT(p))
    # (with zero padding)
    result = irfft(np.conj(A) * P, fft_size)[:2*n-1]
    # result[j] should give sum_k conj(a[k])*p[k+j]... 
    # Let me just verify numerically for small cases in the code
    # For now, let's use a direct approach
    return result * 2.0

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = (np.arange(k_lo, k_hi + 1) - i)
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = conv[k_lo:k_hi + 1] - 2.0 * v * a_partners
            if has_quad:
                c_vals = c_vals.copy()
                quad_idx = 2 * i - k_lo
                c_vals[quad_idx] = conv[2*i] - v * v

            # Golden section search for best x
            def _score_func(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, max(3.0 * v, 5.0 / n)
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = _score_func(c_pt)
            fd = _score_func(d_pt)
            for _ in range(30):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d_pt; d_pt = c_pt; fd = fc
                    c_pt = hi - gr * (hi - lo); fc = _score_func(c_pt)
                else:
                    lo = c_pt; c_pt = d_pt; fc = fd
                    d_pt = lo + gr * (hi - lo); fd = _score_func(d_pt)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver, verbose=False):
    """LP Fictitious Play with cutting-plane LP."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        b, M_val = solve_lp_cutting_plane(a, solver, max_iters=10, top_k=min(80, 2*n-1))
        if b is None:
            # Fallback: use gradient step
            conv_a = fast_convolve(a)
            top_k = min(20, 2*n-1)
            peaks = np.argpartition(conv_a, -top_k)[-top_k:]
            grad = np.zeros(n)
            for k in peaks:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                partners = k - np.arange(i_start, i_end+1)
                grad[i_start:i_end+1] += 2.0 * a[partners]
            grad /= np.linalg.norm(grad) + 1e-15
            b = simplex_project(a - 0.05 * grad)
        
        # Line search for step size
        best_t = 0.0
        best_ls = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
        else:
            alpha = 3.0 / (iters + 3.0)
            a_new = (1.0 - alpha) * a + alpha * b
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

        score_new = compute_score(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        a_avg = (iters * a_avg + a) / (iters + 1)

        # Also check a_avg
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP it={iters}: best={best_score:.8f}")

        if no_improve > 40:
            break

    return best_a, best_score

def softmax_optimize(a0, time_limit, verbose=False):
    """Softmax-approximated gradient descent with annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Anneal temperature
    n_phases = 10
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        t = 50.0 * (10.0 ** (phase / max(1, n_phases - 1)))
        phase_time = time_limit / n_phases - (time.monotonic() - t0)
        if phase_time < 1:
            break
        
        phase_start = time.monotonic()
        lr = 0.01
        while time.monotonic() - phase_start < phase_time:
            conv = fast_convolve(a)
            max_v = float(conv.max())
            shifted = conv - max_v
            exp_vals = np.exp(t * shifted)
            p = exp_vals / exp_vals.sum()
            
            # Gradient: grad_j = 2 * sum_k p_k * a_{k-j}
            # Compute using direct sum over significant p_k
            k_significant = np.where(p > 1e-10 / len(p))[0]
            grad = np.zeros(n)
            for k in k_significant:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                partners = k - np.arange(i_start, i_end + 1)
                grad[i_start:i_end+1] += 2.0 * p[k] * a[partners]
            
            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            grad = grad / grad_norm
            
            # Multiple inner steps
            for _ in range(10):
                a_new = a - lr * grad
                a_new = simplex_project(a_new)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
                lr *= 0.95
            
            # Recompute gradient
            conv = fast_convolve(a)
            max_v = float(conv.max())
            shifted = conv - max_v
            exp_vals = np.exp(t * shifted)
            p = exp_vals / exp_vals.sum()
            k_significant = np.where(p > 1e-10 / len(p))[0]
            grad = np.zeros(n)
            for k in k_significant:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                partners = k - np.arange(i_start, i_end + 1)
                grad[i_start:i_end+1] += 2.0 * p[k] * a[partners]

        if verbose:
            print(f"    Softmax phase {phase}: t={t:.1f}, best={best_score:.8f}")

    return best_a, best_score

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 100.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Block swap
            blen = max(5, rng.integers(n // 10, n // 4))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        elif pert_type == 2:
            # Peak reduction: reduce values at positions contributing to peaks
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                n_reduce = max(2, (i_end - i_start + 1) // 5)
                idx = np.arange(i_start, i_end + 1)
                picks = rng.choice(idx, size=min(n_reduce, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        elif pert_type == 3:
            # Shift mixing
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, shift)
            w = 0.3 + 0.4 * rng.random()
            perturbed = (1 - w) * a + w * shifted
        else:
            # Random restart with asymmetric init
            center = n * rng.uniform(0.2, 0.8)
            width = n * rng.uniform(0.15, 0.4)
            t_arr = np.arange(n, dtype=np.float64)
            new_a = np.exp(-0.5 * ((t_arr - center) / width) ** 2)
            w = 0.3 + 0.5 * rng.random()
            perturbed = (1 - w) * a + w * new_a / new_a.sum()

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = exact_coord_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
