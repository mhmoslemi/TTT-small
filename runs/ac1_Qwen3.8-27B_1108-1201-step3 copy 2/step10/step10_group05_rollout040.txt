"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with log-sum-exp smoothing (FFT-accelerated)
2. Improved coordinate descent with analytical candidate generation
3. Dual fictitious play (LP-free, using soft best-response)
4. Peak-targeted stochastic perturbation with basin hopping
5. Multi-length exploration with structured initializations

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        s = v.sum()
        if s > 0:
            return v / s
        return np.ones(n) / n

def smoothed_grad_fft(a, beta):
    """FFT-accelerated gradient of (1/beta)*log(sum_k exp(beta*(a*a)[k])."""
    n = len(a)
    r = fast_convolve(a)
    r_scaled = beta * (r - r.max())
    r_scaled = np.clip(r_scaled, -50, 0)
    w = np.exp(r_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(r)
        w[r.argmax()] = 1.0
    else:
        w /= w_sum
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g, r.max()

def nesterov_smoothed_opt(a0, time_limit, beta_init=15.0, lr_init=0.02, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        grad, _ = smoothed_grad_fft(y_proj, beta)

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        conv_a = fast_convolve(a)
        score = 2.0 * n * conv_a.max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            lr *= 0.92
            beta *= 1.15
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            beta *= 1.005

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 200 == 0 and verbose_nesterov():
            elapsed = time.monotonic() - t0
            print(f"    NMD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_score

def smart_cd(a0, time_limit, label="", max_sweeps=5000):
    """Improved smart coordinate descent with analytical candidates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and sweeps < max_sweeps:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(np.max(conv[k_end + 1:])))

            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai * 0.999]

            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, -0.02, -0.05, -0.1, -0.2, -0.3, -0.5]:
                candidates.append(frac * ai)

            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j in range(len(top_idx)):
                c = coeffs[top_idx[j]]
                if c > 1e-12:
                    d = (unaff_max - affected[top_idx[j]]) / c
                    if -ai * 1.2 < d < ai * 0.5:
                        candidates.append(d)

            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, min(j1 + 3, len(top_idx))):
                    c1, c2 = coeffs[top_idx[j1]], coeffs[top_idx[j2]]
                    denom = c1 - c2
                    if abs(denom) > 1e-12:
                        d = (affected[top_idx[j2]] - affected[top_idx[j1]]) / denom
                        if -ai * 1.2 < d < ai * 0.5:
                            candidates.append(d)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12 or delta > ai * 0.5:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-16:
                    continue

                new_aff = affected + delta * coeffs
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] += delta * delta

                new_max = max(unaff_max, float(np.max(new_aff)))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

    return best_a, best_score

def dual_fictitious_play(a0, time_limit, label=""):
    """Dual fictitious play with soft best-response (no LP needed)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        conv_a = fast_convolve(a)
        M = conv_a.max()

        # Soft best response: b[j] proportional to 1/(max_k a[k-j] + eps)
        # This approximates the LP solution
        difficulty = np.zeros(n)
        for j in range(n):
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            difficulty[j] = np.max(a[max(0, k_lo - (n-1)):k_hi - (n-1) + 1]) if k_hi - (n-1) >= max(0, k_lo - (n-1)) else 0

        # Simpler: difficulty[j] = max of a over valid k-j
        # (a*b)[k] = sum_j a[k-j] * b[j], so the "cost" of b[j] is max_k a[k-j]
        difficulty = np.zeros(n)
        for j in range(n):
            ks = np.arange(max(0, j), min(2*n-1, j+n))
            js = ks - j
            difficulty[j] = np.max(a[js])

        eps = 1e-6
        b = 1.0 / (difficulty + eps)
        b = np.maximum(b, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / b_sum

        alpha = min(1.0, 8.0 / (iters + 3))
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b.copy()

        a = a_new

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DIP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

        if iters > 500 and (time.monotonic() - t0) > 30:
            break

    return best_a, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                pressure[i] += a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def block_cd(a0, time_limit, block_size=4, label=""):
    """Block coordinate descent: optimize small blocks simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit and sweeps < 200:
        sweeps += 1
        improved_this_sweep = False

        for start in range(0, n - block_size + 1, block_size):
            if time.monotonic() - t0 > time_limit:
                break

            idx = np.arange(start, start + block_size)
            block = a[idx].copy()
            if block.max() < 1e-15:
                continue

            # Try random perturbation of the block
            for trial in range(5):
                delta = (rng := np.random.default_rng(hash((sweeps, start, trial)) % 2**32)).standard_normal(block_size) * block * 0.1
                new_block = np.maximum(block + delta, 0)
                if new_block.sum() < 1e-15:
                    continue
                # Normalize block to maintain total sum
                scale = block.sum() / new_block.sum() if new_block.sum() > 0 else 1
                new_block *= scale

                a_trial = a.copy()
                a_trial[idx] = new_block

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved_this_sweep = True
                    break

        if not improved_this_sweep and sweeps > 50:
            break

    return best_a, best_score
