def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:

    import cvxpy as cp
    for name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.CBC]:

def build_conv_matrix(a):

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

        solver = get_lp_solver()

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        a = a / s
    conv = fast_convolve(a).copy()

    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        # Use random ordering for diversity after first sweep
        if sweeps > 1:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score_func(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                           _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                           _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = _qm
                nc = nv.max()
                if nc > nm:
                    nm = nc
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score_func, 0.0, upper, max_iter=25)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

    a_avg = a.copy()
    decay = 5.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        # Polyak-like step size based on duality gap
        # Current value = max(a*a), LP value = M_val
        cur_max = float(fast_convolve(a).max())
        duality_gap = cur_max - M_val
        if duality_gap > 1e-10:
            polyak_t = duality_gap / (cur_max + 1e-15)
            polyak_t = np.clip(polyak_t, 0.05, 0.8)
        else:
            polyak_t = 0.1

        # Try multiple step sizes
        best_t = 0.0
        best_ls = best_score
        trial_ts = [polyak_t * 0.5, polyak_t, polyak_t * 1.5, polyak_t * 2.0,
                    0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        trial_ts = sorted(set(np.clip(trial_ts, 0.01, 1.0)))
        for trial_t in trial_ts:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        # Check all candidates
        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f}, gap={duality_gap:.6f}")

        if no_improve > 40:
            break

    return best_a, best_score, iters

def smoothed_gradient_descent(a0, time_limit, verbose=False):
    """Use log-sum-exp smoothing of max to get a smooth gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.002
    T_smooth = 50.0  # smoothing parameter (larger = closer to true max)
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        conv = fast_convolve(a)
        max_val = float(conv.max())

        # Log-sum-exp gradient
        shifted = T_smooth * (conv - max_val)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-15:
            break
        weights = weights / ws

        # Gradient of (1/T)log(sum(exp(T*conv_k))) w.r.t. a_i
        grad = np.zeros(n)
        # Only use top contributing lags for efficiency
        top_k = min(50, 2 * n - 1)
        active = np.argsort(weights)[::-1][:top_k]
        for k in active:
            if weights[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += weights[k] * (a[k - j])

        grad /= T_smooth

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            a_new = a
        else:
            a_new = a_new / cs

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.997
        if no_improve > 50:
            lr *= 0.8
            T_smooth = max(20.0, T_smooth * 0.9)
            no_improve = 0
        if no_improve > 200:
            break

        if verbose and iters % 20 == 0:
            print(f"    SGD it={iters}: best={best_score:.8f}, lr={lr:.5f}")

    return best_a, best_score

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_geometric(n, rng):
    r = rng.uniform(0.7, 0.98)
    a = r ** np.arange(n)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.6, 1.4, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        perturb_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 40:
            break
        t_alloc = min(rem * 0.35, 100.0)

        pert_type = perturb_count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Peak-focusing: reduce values that contribute to top convolution peaks
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        elif pert_type == 1:
            # Gaussian noise
            sigma = 0.01 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            # Block shuffle
            n_blocks = rng.integers(3, 8)
            block_size = n // n_blocks
            indices = np.arange(n_blocks)
            rng.shuffle(indices)
            new_a = np.zeros(n)
            for orig, new in enumerate(indices):
                s1 = orig * block_size
                s2 = s1 + (block_size if orig < n_blocks - 1 else n - s1)
                new_a[new * block_size:new * block_size + (s2 - s1)] = perturbed[s1:s2]
            perturbed = new_a
        elif pert_type == 3:
            # Spectral perturbation: modify random frequency components
            n_fft = 1 << (n - 1).bit_length()
            A = rfft(a, n_fft)
            n_modify = max(3, n // 10)
            freqs = rng.choice(1, len(A) - 1, size=n_modify, replace=False)
            for fi in freqs:
                A[fi] *= (1.0 + rng.standard_normal() * 0.15)
                if fi < len(A) - 1:
                    A[-fi] = np.conj(A[fi])
            perturbed = irfft(A, n_fft)[:n]
        else:
            # Mix with shifted version
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.concatenate([a[-shift:], a[:-shift]])
            perturbed = (1.0 - 0.3) * a + 0.3 * shifted

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_{perturb_count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {perturb_count} (type={pert_type}): improved to {best_score:.8f}")

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization for minimizing 2*n*max(conv)/(sum)^2:
    1. Load parent and evaluate
    2. Quick diverse initialization + coordinate descent screening
    3. LP Fictitious Play (main driver with Polyak steps)
    4. Smoothed gradient descent (log-sum-exp)
    5. Exact coordinate descent polishing
    6. Multi-n exploration via interpolation
    7. Structured basin hopping
    8. Final fine polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Using default parent: n={len(parent)}, score={best_score:.8f}")

    # Phase 1: Diverse initialization + quick screening
    print("\n=== Phase 1: Diverse Init ===")
    t1_start = time.monotonic()
    n_parent = len(parent)

    n_values = [n_parent]
    for alt in [1024, 1280, 1536, 2048]:
        if alt != n_parent and 256 <= alt <= 2048:
            n_values.append(alt)

    candidates = []
    for n in n_values[:5]:
        if time.monotonic() - t1_start > 150:
            break
        inits = []
        inits.append(interpolate_seq(parent, n))
        inits.append(init_asymmetric(n, rng, 0.25))
        inits.append(init_asymmetric(n, rng, 0.4))
        inits.append(init_geometric(n, rng))
        inits.append(init_flat_spectrum(n, rng))

        for j, cand in enumerate(inits[:4]):
            t_left = 150.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(2.0, min(15.0, t_left / 5.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_{j}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best={candidates[0][0]:.8f}")
    for s_c, a_c, n_c in candidates[:3]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    n_work = len(best_seq)
    print(f"\nWorking with n={n_work}, current best={best_score:.8f}")

    # Phase 2: LP Fictitious Play (main optimizer)
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.55, 350.0)
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, s_fp, iters = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

    # Phase 3: Smoothed gradient descent
    print(f"\n=== Phase 3: Smoothed GD (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t_sg = min(rem * 0.4, 120.0)
        a_sg, s_sg = smoothed_gradient_descent(best_seq, t_sg, verbose=True)
        sc = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sg.copy()
            print(f"  NEW BEST SGD: {sc:.8f}")

    # Phase 4: Exact Coordinate Descent Polishing
    print(f"\n=== Phase 4: ECD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(2):
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        t_cd = min(rem * 0.3, 150.0)
        print(f"  ECD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph4_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for alt_n in [n_work + 256, n_work + 512, max(1024, n_work - 256)]:
        if alt_n < 512 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        a_interp = interpolate_seq(best_seq, alt_n)
        # Quick coordinate descent
        t_alloc = min(rem * 0.3, 80.0)
        a_opt, s_opt, _ = exact_coord_descent(a_interp, t_alloc, label=f"multi_n{alt_n}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST n={alt_n}: {sc:.8f}")
        # Try LP on this length too
        rem2 = deadline - time.monotonic() - 200
        if rem2 > 30 and sc < best_score + 0.01:
            t_fp2 = min(rem2 * 0.3, 60.0)
            a_fp2, s_fp2, _ = lp_fictitious_play(a_opt, t_fp2, solver=solver)
            sc2 = float(evaluate_sequence(a_fp2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_fp2.copy()
                print(f"  NEW BEST FP n={alt_n}: {sc2:.8f}")

    # Phase 6: Basin Hopping
    print(f"\n=== Phase 6: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_bh = min(rem * 0.7, 200.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 7: Second LP round on best
    print(f"\n=== Phase 7: Second LP Round (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        t_fp3 = min(rem * 0.6, 150.0)
        a_fp3, s_fp3, _ = lp_fictitious_play(best_seq, t_fp3, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp3.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp3.copy()
            print(f"  NEW BEST LP2: {sc:.8f}")

    # Phase 8: Final polishing
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_final = min(rem * 0.5, 60.0)
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_final, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    rem = deadline - time.monotonic() - 10
    if rem > 5:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
