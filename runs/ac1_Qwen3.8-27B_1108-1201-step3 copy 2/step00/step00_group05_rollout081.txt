def _active_gradient(a, conv, kmax=32, margin=1e-3):
    n=len(a)
    maxc=np.max(conv)
    if maxc <=0: return None, 0
    thresh = maxc * (1.0 - margin)
    idx = np.nonzero(conv >= thresh)[0]
    if idx.size > kmax:
       # choose top kmax
       part = np.argpartition(conv, -kmax)[-kmax:]
       idx = part
    if idx.size == 0:
       idx = np.array([int(np.argmax(conv))])
    # optional weights uniform
    # Compute gradient
    g=np.zeros(n)
    inv=1.0/len(idx)
    for kk in idx:
       k=int(kk)
       jm = k - (n-1)
       if jm < 0: jm = 0
       jx = k
       if jx > n-1: jx = n-1
       if jm <= jx:
          # indices jm..jx correspond a[k - idx], segment a[k-jx : k-jm + 1][::-1]
          start = k - jx
          stop = k - jm + 1
          seg = a[start:stop][::-1]
          g[jm:jx+1] += inv * 2.0 * seg
          if (k % 2 == 0):
              jd = k//2
              if jm <= jd <= jx:
                  g[jd] -= inv * a[jd]
    return g, len(idx)

def _active_gradient(a, conv, kmax=16):
    n = len(a)
    maxc = float(np.max(conv))
    if maxc <= 0.0 or not np.isfinite(maxc):
        return None, 0
    nconv = len(conv)
    thresh = maxc * (1.0 - 1e-3)
    idx = np.nonzero(conv >= thresh)[0]
    if idx.size > kmax:
        # If too many, take top kmax. For nconv maybe <=kmax? no.
        if nconv <= kmax:
            idx = np.arange(nconv)
        else:
            idx = np.argpartition(conv, kmax-1)[-kmax:]
    if idx.size == 0:
        idx = np.array([int(np.argmax(conv))])
    g = np.zeros(n, dtype=np.float64)
    inv = 1.0 / float(idx.size)
    for kk in idx:
        k = int(kk)
        jmin = k - (n - 1)
        if jmin < 0: jmin = 0
        jmax = k
        if jmax >= n: jmax = n - 1
        if jmin <= jmax:
            start = k - jmax
            stop = k - jmin + 1
            # segment length jmax - jmin + 1
            seg = a[start:stop][::-1]
            g[jmin:jmax+1] += inv * 2.0 * seg
            if (k & 1) == 0:
                jd = k >> 1
                if jmin <= jd <= jmax:
                    g[jd] -= inv * a[jd]
    return g, int(idx.size)

def _normalize(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(arr, 0.0, 1000.0, out=arr)
    s = float(arr.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _sanitize_logits(loga):
    arr = np.asarray(loga, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=-35.0)
    m = float(np.max(arr))
    if np.isfinite(m):
        arr = arr - m
    np.clip(arr, -35.0, 0.0, out=arr)
    return arr


def _softmax_from_log(loga):
    loga = _sanitize_logits(loga)
    y = np.exp(loga)
    s = float(y.sum())
    n = y.size
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return y / s


def _fft_next_size(m):
    n = max(2, int(m))
    r = 1
    while r < n:
        r <<= 1
    return r


def _self_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    N = _fft_next_size(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:size]
    np.maximum(c, 0.0, out=c)
    return c

def _active_gradient(a, conv, kmax=16):
    n = len(a)
    maxc = float(np.max(conv))
    if maxc <= 0.0 or not np.isfinite(maxc):
        return None, 0
    thresh = maxc * (1.0 - 1e-3)
    idx = np.nonzero(conv >= thresh)[0]
    nconv = len(conv)
    if idx.size > kmax:
        if nconv <= kmax:
            idx = np.arange(nconv)
        else:
            idx = np.argpartition(conv, kmax - 1)[-kmax:]
    if idx.size == 0:
        idx = np.array([int(np.argmax(conv))], dtype=np.int64)
    g = np.zeros(n, dtype=np.float64)
    inv = 1.0 / float(idx.size)
    for kk in idx:
        k = int(kk)
        jmin = k - (n - 1)
        if jmin < 0:
            jmin = 0
        jmax = k
        if jmax >= n:
            jmax = n - 1
        if jmin <= jmax:
            start = k - jmax
            stop = k - jmin + 1

            g[jmin:jmax + 1] += inv * 2.0 * seg
            if (k & 1) == 0:
                jd = k >> 1
                if jmin <= jd <= jmax:
                    g[jd] -= inv * a[jd]
    return g, int(idx.size)

                    cur_conv = _self_conv(cur_a)
                    cur_score = _fast_score(cur_a)
                    if cur_score > best_local_score * 1.05:
                        loga = _sanitize_logits(best_local_log)
                        cur_a = _softmax_from_log(loga)
                        cur_conv = _self_conv(cur_a)
                        cur_score = best_local_score
                stall = 0
                alpha = alpha0

    _update_best(best_local_a)
    return best_local_a


def _polish(start_a, deadline, rng):
    global BEST_SCORE, BEST_SEQ
    a = _normalize(start_a)
    n = int(a.size)
    if n == 0:
        a = np.ones(1024, dtype=np.float64)
        n = 1024
    loga = np.log(a + 1e-300)
    loga -= float(loga.mean())
    loga = _sanitize_logits(loga)
    cur_score = _fast_score(a)
    _update_best(a)

    sigma = 0.035
    stall = 0

    while time.monotonic() < deadline:
        old_log = loga.copy()
        kind = int(rng.integers(0, 4))
        if kind == 0:
            width = int(rng.integers(1, max(2, n // 50)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            loga[start:stop] = old_log[start:stop] + rng.normal(0.0, sigma, stop - start)
        elif kind == 1:
            m = int(rng.integers(max(1, n // 2000), max(2, n // 100)))
            idx = rng.choice(n, size=m, replace=False)
            old_vals = old_log[idx].copy()
            loga[idx] = old_vals + rng.normal(0.0, sigma, m)
        elif kind == 2:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            t = float(rng.normal(0.0, 2.0 * sigma))
            loga[i] = old_log[i] + t
            loga[j] = old_log[j] - t
        else:
            p = _antisym_normal(n, rng)
            loga = old_log + sigma * p

        loga = _sanitize_logits(loga)
        new_a = _softmax_from_log(loga)
        s = _fast_score(new_a)
        if s < cur_score:
            a = new_a
            loga = _sanitize_logits(np.log(a + 1e-300))
            cur_score = s
            _update_best(a)
            stall = 0
            sigma = min(1.0, sigma * 1.05)
        else:
            loga = old_log
            stall += 1
            if stall > 700:
                sigma = max(0.01, min(1.0, sigma * 1.25))
                stall = 0

        rem = deadline - time.monotonic()
        if rem < 25.0:
            sigma = min(sigma, 0.008)

    _update_best(a)
    return a


def propose_candidate(seed=1234, budget_s=1800.0, **kwargs):
    global BEST_SCORE, BEST_SEQ
    BEST_SCORE = float("inf")
    BEST_SEQ = None

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1.0
    # Leave a little room for final verification.
    deadline = t0 + max(1.0, budget - 30.0)

    parent = None
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.sum() > 0.0 and 64 <= arr.size <= 20000:
                parent = arr / float(arr.sum())
    except Exception:
        parent = None

    base_n = int(parent.size) if parent is not None else 4096
    if base_n < 64:
        base_n = 1024
    if base_n > 20000:
        base_n = 8192

    raw_lengths = [
        base_n,
        4096,
        2048,
        1024,
        8192,
        int(round(base_n * 0.5)),
        int(round(base_n * 1.5)),
    ]
    lengths = []
    for m in raw_lengths:
        try:
            mm = int(m)
        except Exception:
            continue
        if 128 <= mm <= 20000 and mm not in lengths:
            lengths.append(mm)
        if len(lengths) == 3:
            break
    if not lengths:
        lengths = [2048]

    polish_time = max(30.0, 0.18 * float(budget))
    main_deadline = deadline - polish_time
    if main_deadline < t0 + 60.0:
        main_deadline = deadline - 10.0

    main_total = max(60.0, main_deadline - t0)
    if len(lengths) == 1:
        ends = [min(deadline, t0 + main_total)]
    elif len(lengths) == 2:
        bounds = [0.0, 0.65, 1.0]
        ends = [t0 + main_total * b for b in bounds[1:]]
    else:
        bounds = [0.0, 0.5, 0.8, 1.0]
        ends = [t0 + main_total * b for b in bounds[1:]]

    for i, L in enumerate(lengths):
        if time.monotonic() >= main_deadline:
            break
        start_seg = time.monotonic()
        end_seg = float(ends[i])
        if end_seg < start_seg + 25.0:
            end_seg = min(deadline, start_seg + 25.0)
        if end_seg > main_deadline:
            end_seg = main_deadline

        parent_arg = parent if parent is not None and parent.size == L else None
        init_a = _choose_initial(L, parent_arg, rng)

        if end_seg - time.monotonic() > 55.0:
            coarse_deadline = min(end_seg - 20.0, time.monotonic() + (20.0 if i == 0 else 8.0))
            coarse_a = _coarse_shape(L, coarse_deadline, rng)
            if coarse_a is not None:
                cs = _fast_score(coarse_a)
                is_ = _fast_score(init_a)
                if cs < is_:
                    init_a = coarse_a

        if end_seg - time.monotonic() > 130.0:
            first_deadline = time.monotonic() + 0.6 * (end_seg - time.monotonic())
            if first_deadline > end_seg - 10.0:
                first_deadline = end_seg - 10.0
            _optimize(init_a, first_deadline, rng,
                      kmax=16, blend=0.005, noise=0.01,
                      alpha0=0.25, stall_limit=1200)

            if end_seg - time.monotonic() > 25.0:
                init2 = _choose_initial(L, parent_arg, rng)
                if end_seg - time.monotonic() > 65.0:
                    cap = 18.0 if i == 0 else 8.0
                    coarse_deadline2 = min(end_seg - 15.0, time.monotonic() + cap)
                    coarse2 = _coarse_shape(L, coarse_deadline2, rng)
                    if coarse2 is not None:
                        cs2 = _fast_score(coarse2)
                        is2 = _fast_score(init2)
                        if cs2 < is2:
                            init2 = coarse2
                _optimize(init2, end_seg, rng,
                          kmax=8, blend=0.002, noise=0.005,
                          alpha0=0.45, stall_limit=700)
        else:
            _optimize(init_a, end_seg, rng)

    best_a = BEST_SEQ
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64)
        best_a = best_a / float(best_a.sum())
        BEST_SEQ = best_a.copy()
        BEST_SCORE = _fast_score(best_a)

    if time.monotonic() < deadline:
        best_a = _polish(best_a, deadline, rng)

    best_a = _normalize(best_a)
    BEST_SEQ = best_a.copy()
    _update_best(best_a)

    # Optional final authoritative check for moderate lengths, if enough time remains.
    try:
        if len(best_a) <= 6000 and (time.monotonic() < deadline + 18.0):
            val = evaluate_sequence(best_a.tolist())
            if not np.isfinite(val):
                fb = np.ones(1024, dtype=np.float64)
                fb = fb / float(fb.sum())
                best_a = fb
                BEST_SEQ = fb.copy()
                BEST_SCORE = 2.0
    except Exception:
        pass

    return [float(x) for x in best_a.tolist()]
