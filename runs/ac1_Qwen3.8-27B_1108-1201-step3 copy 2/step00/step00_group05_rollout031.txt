    b = np.convolve(a, a)
    # Soft-max weights
    w = np.exp(b / tau)
    w /= w.sum()
    # Gradient: grad[j] = sum_k w[k] * db[k]/da[j] = sum_k w[k] * a[k-j] (if valid)
    # This is a convolution-like operation
    grad = np.zeros(n)
    for j in range(n):
        # db[k]/da[j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(2*n-1, j+n)
        # a[k-j] for k in [k_start, k_end) is a[0], a[1], ..., a[min(n-1, k_end-1-j)]
        # But we need w[k] for k in [k_start, k_end)

    # Renormalize to keep sum constant
