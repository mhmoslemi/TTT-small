For a real non-negative sequence, can we achieve a flat spectrum? The DFT of a real sequence is conjugate-symmetric, so A(ω) = conj(A(-ω)). And |A(ω)|^2 = A(ω)*conj(A(ω)) is real and even. For it to be constant, we need |A(ω)| = c for all ω. This means the sequence is a "flat-spectrum" sequence.

A symmetric sequence with flat magnitude spectrum... For a symmetric sequence a[k] = a[n-1-k], the DFT is real (up to a linear phase factor), and |A(ω)| = |A(ω)| (trivially). import numpy as np
import time

def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} approximately."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s > 0:
        v = v * S / s
    return v

def subgradient_step(a, S, lr):
    """One subgradient step to minimize max(conv(a,a)) with sum constraint."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    k_star = np.argmax(c)
    
    # Subgradient: v[p] = 2*a[k_star - p] if 0 <= k_star - p < n
    # This is 2 * reverse(a) shifted appropriately
    # v[p] = 2*a[k_star - p] for p in [max(0, k_star-n+1), min(k_star, n-1)]
    v = np.zeros(n)
    p_start = max(0, k_star - n + 1)
    p_end = min(k_star, n - 1)
    for p in range(p_start, p_end + 1):
        v[p] = 2 * a[k_star - p]
    
    # Update
    a_new = a - lr * v
    a_new = project_onto_simplex(a_new, S)
    return a_new, c.max()

    # Build sparse Toeplitz matrix
    first_col = np.zeros(2*n - 1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]

    A_ub = A  # (2n-1) x n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        return result.x
