"""
Search for non-negative step-function heights minimising
    2 * n * max_k conv(a,a)_k  /  (sum a)^2 .
The objective is scale-invariant, so we fix sum(a)=1 in the inner loop and
minimise M = max_k (a*a)_k directly (final value = 2*n*M).  The peak of the
autocorrelation is reduced by making the sequence ASYMMETRIC / monotone (a
linear ramp already gives ~1.33 analytically, well under the 1.5027 target);
a Nesterov accelerated subgradient method on M, with FFT-based autocorrelation,
backtracking line search, non-negativity projection and renormalisation, is then
used to push M further down.  Multiple analytic monotone starts (different
exponents, both orientations) plus the supplied reference sequence are explored
in parallel time-sliced, and the best is returned.
"""

import time
import math
import numpy as np


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fast_autocorr(a):
    """Full (length 2n-1) autocorrelation of a via real FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def grad_peak(a, k, n):
    """
    d C[k] / d a[i] where C = conv(a,a).
    dC[k]/da[i] = a[k-i] * 1[i in J] + a[k-i] * 1[k-i in J],
    J = [max(0,k-n+1), min(k,n-1)].
    """
    grad = np.zeros(n)
    j_min = k - n + 1
    if j_min < 0:
        j_min = 0
    j_max = k
    if j_max > n - 1:
        j_max = n - 1
    # case 1: i in [j_min, j_max]
    idx1 = np.arange(j_min, j_max + 1)
    grad[idx1] += a[k - idx1]
    # case 2: k-i in [j_min, j_max]  =>  i in [k-j_max, k-j_min]
    i_lo = k - j_max
    if i_lo < 0:
        i_lo = 0
    i_hi = k - j_min
    if i_hi > n - 1:
        i_hi = n - 1
    idx2 = np.arange(i_lo, i_hi + 1)
    grad[idx2] += a[k - idx2]
    return grad


def normalise(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-9:
        return np.ones_like(a)
    return a / s


def objective_local(a):
    n = a.size
    C = fast_autocorr(a)
    return float(C.max())


def optimise_start(a0, n, rng, deadline, label):
    """Nesterov subgradient descent on M=max autocorrelation, sum pinned to 1."""
    a = normalise(np.asarray(a0, dtype=np.float64).copy())
    C = fast_autocorr(a)
    M = float(C.max())
    best = a.copy()
    bestM = M

    lr = 40.0
    beta = 0.85
    v = np.zeros(n)
    no_imp = 0
    max_no_imp = int(3 * max(50, n // 40))

    while time.monotonic() < deadline:
        C = fast_autocorr(a)
        M = float(C.max())
        if M < bestM - 1e-12:
            bestM = M
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > max_no_imp:
            r = rng.random()
            if r < 0.35:
                # gently re-mix with a fresh monotone ramp
                t = 0.3 + 0.4 * rng.random()
                ramp = np.arange(1, n + 1, dtype=np.float64)
                ramp **= float(rng.uniform(0.7, 2.0))
                a = (1.0 - t) * a + t * normalise(ramp)
            else:
                # multiplicative block kick
                w = int(rng.integers(max(2, n // 200), max(3, n // 12)))
                s0 = int(rng.integers(0, n - w + 1))
                a[s0:s0 + w] *= np.exp(rng.normal(0.0, 0.35, size=w))
                if rng.random() < 0.4:
                    a = 0.5 * (a + a[::-1])
            a = normalise(a)
            v = np.zeros(n)
            no_imp = 0
            if no_imp > 5 * max_no_imp:
                break
            continue

        k = int(C.argmax())
        g = grad_peak(a, k, n)
        g = g - g.mean()          # remove the sum-preserving gauge component
        v = beta * v + g

        accepted = False
        for _ in range(6):
            a_try = a - lr * v
            a_try = normalise(a_try)
            M_try = objective_local(a_try)
            if M_try < M + 1e-11:
                a, M = a_try, M_try
                accepted = True
                if M_try < bestM - 1e-12:
                    lr *= 1.03
                break
            else:
                lr *= 0.5
        if not accepted:
            # tiny exploratory nudge forward
            a = normalise(a + rng.normal(0.0, 0.4 / n, size=n))
            v = 0.5 * v
        else:
            lr = min(lr, 2000.0)
        if (int(no_imp) % 300) == 0:
            lr *= 0.97

    return best, float(2.0 * n * bestM)
