        return time.monotonic() - start_time

        return total_budget - elapsed()

    try:

        n = len(init)
    except:
        n = 2535
        init = np.ones(n)

    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}, n = {n}")

        """Compute the true objective (scale-invariant, so normalize first)."""
        s = a.sum()
        if s < 0.01:
            return np.inf
        c = np.convolve(a, a)
        return 2 * n * c.max() / (s ** 2)
    
    def optimize_from(a0, max_iters=5000, verbose=False):

        a = a0.copy()
        s = a.sum()
        if s <= 0:
            return None, np.inf
        a = a / s * n  # normalize to sum = n
        
        T = 5.0  # temperature
        T_min = 0.5
        lr = 0.05
        best_local = a.copy()
        best_local_score = true_objective(a)
        
        for it in range(max_iters):
            if remaining() < 5:
                break
            
            c = np.convolve(a, a)
            
            # Periodic evaluation
            if it % 100 == 0:
                score = 2.0 * c.max() / n  # since sum(a) = n, objective = 2n*c.max()/n² = 2*c.max()/n
                if score < best_local_score:
                    best_local = a.copy()
                    best_local_score = score
                    if verbose:
                        print(f"  iter {it}: score = {best_local_score:.6f}")

            c_scaled = c / T
            c_max = c_scaled.max()
            w = np.exp(c_scaled - c_max)
            w_sum = w.sum()

Minimizes 2n * max(convolve(a,a)) / (sum(a))^2 using a combination of:
1. Spectral initialization (phase-retrieval style) to get flat autocorrelation
2. Log-sum-exp gradient descent for refinement
3. Multiple initializations (windows, spectral, random)
The key insight is the objective is scale-invariant, so we fix sum(a)=n
and minimize the peak of the autocorrelation relative to its mean.
"""

import numpy as np

def _normalize_sum(a, target_sum):
    """Normalize array to have the given sum, maintaining non-negativity."""
    s = a.sum()
    if s <= 1e-30:
        return a
    return a / s * target_sum


def _true_objective(a, n):
    """Compute the true (unsmoothed) objective."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s ** 2))


def _spectral_init(n, rng, target_flatness=1.0):
    """
    Use phase-retrieval style iteration to find a non-negative sequence
    whose autocorrelation is as flat as possible.
    """
    # Work with zero-padded sequence of length N (power of 2 for FFT speed)
    N = 1
    while N < 2 * n - 1:
        N *= 2

    # Target: flat autocorrelation (all lags equal)
    c_target = np.ones(N)
    C_target = np.fft.rfft(c_target)
    # |A|^2 should equal C_target, so |A| = sqrt(C_target)
    log_C_target = np.log(np.maximum(C_target, 1e-30))

    # Initialize with random complex spectrum
    A = np.exp(rng.uniform(0, 2 * np.pi, N // 2 + 1) * 1j) * np.sqrt(np.abs(C_target)) + 0j
    A = A.astype(np.complex128)

    best_a = None
    best_score = np.inf

    for it in range(200):
        # Match magnitude to target spectrum
        target_mag = np.sqrt(np.maximum(C_target, 0).real)
        A = target_mag * np.exp(1j * np.angle(A))

        # Inverse FFT to get time domain
        a_padded = np.fft.irfft(A, N)

        # Extract and project to non-negative

    return best_a if best_a is not None else np.ones(n)


def _gradient_step(a, c, T, lr):
    """
    One multiplicative gradient step on logsumexp(c/T).
    Returns updated a (normalized to sum=n).
    """
    n = len(a)
    c_scaled = c / T
    c_max = c_scaled.max()
    w = np.exp(c_scaled - c_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return a
    w = w / w_sum

    # grad[i] = sum_j w[i+j] * a[j]  (cross-correlation)
    grad = np.correlate(w, a, mode='valid')

    # Project gradient (zero-mean for sum constraint)
    grad = grad - grad.mean()

    # Scale gradient for stability
    grad_max = np.abs(grad).max()
    if grad_max < 1e-30:
        grad_max = 1.0

    # Multiplicative update (maintains non-negativity)
    a_new = a * np.exp(-lr * grad / grad_max)
    a_new = np.maximum(a_new, 0.0)

    if a_new.sum() > 1e-30:
        a_new = _normalize_sum(a_new, n)
    else:
        a_new = a.copy()

    return a_new

    """
    Refine initial sequence a0 using log-sum-exp gradient descent.
    Returns (best_sequence, best_score).
    """

    s = a.sum()

    # Scale T relative to the data
    c_init = np.convolve(a, a)
    T_scale = c_init.max()
    T = 0.1 * T_scale
    T_min = 0.005 * T_scale

    stall_counter = 0
    last_improvement = best_local_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)

        # Periodic true evaluation
        if it % 50 == 0:
            score = 2.0 * c.max() / n  # since sum(a) = n
            if score < best_local_score:
                best_local_score = score
                best_local = a.copy()
                stall_counter = 0
            else:
                stall_counter += 1

            # Reheat if stuck
            if stall_counter > 10:
                T = 0.1 * T_scale
                lr = 0.15
                stall_counter = 0
                # Small perturbation

                a = np.maximum(a, 0)
                a = _normalize_sum(a, n)

        lr = max(0.001, lr * lr_decay)

        best_local_score = score
