def simplex_project(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(u)+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(a, M, max_time=None):
  n=a.size
  if n>700: return None
  L=2*n-1
  idx=np.arange(n)
  rows=np.tile(idx,n)+np.repeat(np.arange(n),n)
  cols=np.repeat(np.arange(n),n)
  data=np.tile(a,n)
  A=scipy.sparse.csr_matrix((data,(rows,cols)),shape=(L,n))
  c=-np.ones(n)
  b_ub=np.full(L,M*(1+1e-9)+1e-15)
  res=scipy.optimize.linprog(c,A_ub=A,b_ub=b_ub,bounds=[(0,None)]*n,method='highs')
  if not res.success: return None
  b=np.clip(res.x,0,None); B=b.sum()
  if B <= 1.0: return None
  return b/B

def _simplex_project(v, s=1.0):
    v=np.asarray(v,float)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u * np.arange(1,v.size+1) > (cssv - s))[0]
    if rho.size==0: return np.full(v.size, s/v.size)
    rho=rho[-1]
    theta=(cssv[rho]-s)/(rho+1.0)
    return np.maximum(v-theta,0.0)

import time
import numpy as np
import scipy.sparse
import scipy.optimize

LP_MAX_N = 512
_LP_INDEX_CACHE = {}


def _pow2_at_least(x):
    n = 1
    while n < x:
        n <<= 1
    return max(n, 2)


def _normalize_nonng(v):
    v = np.asarray(v, dtype=np.float64).reshape(-1).copy()
    if v.size == 0:
        return v
    v[~np.isfinite(v)] = 0.0
    np.clip(v, 0.0, None, out=v)
    s = float(np.sum(v))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(v.size, 1.0 / max(1, v.size), dtype=np.float64)
    v /= s
    return v


def _simplex_project(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, v.size + 1)
    ok = np.nonzero(u * idx > (cssv - s))[0]
    if ok.size == 0:
        return np.full(v.size, s / float(v.size), dtype=np.float64)
    rho = int(ok[-1])
    theta = float(cssv[rho] - s) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    ss = float(out.sum())
    if ss > 0.0 and np.isfinite(ss):
        out /= (ss / s)
    return out


def _fft_conv(a, N):
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = c[:L]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _score_and_max(a, N=None):
    a = _normalize_nonng(a)
    n = a.size
    if n == 0:
        return float("inf"), float("inf"), 0
    if N is None:
        N = _pow2_at_least(2 * n - 1)
    c = _fft_conv(a, N)
    m = float(np.max(c)) if c.size else float("inf")
    return float(2.0 * n * m), m, int(N)


def _resample_seq(a, n):
    a = _normalize_nonng(a)
    old = a.size
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if old == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if old == 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if old == n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old)
    new_x = np.linspace(0.0, 1.0, n)
    b = np.interp(new_x, old_x, a)
    return _normalize_nonng(b)


def _add_init_if_valid(out, v):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    if v.size == 0:
        return
    v[~np.isfinite(v)] = 0.0
    np.clip(v, 0.0, None, out=v)
    s = float(v.sum())
    if np.isfinite(s) and s > 0.0:
        out.append(v / s)

        return out

    for _ in range(2):

        for mode in (1, 2, 3):
            w = 2.0 * np.pi * mode * ts
            v += amp * rng.uniform(-1.0, 1.0) * np.cos(w)
            v += amp * rng.uniform(-1.0, 1.0) * np.sin(w)
            amp *= 0.55
        _add_init_if_valid(out, v)

    for conc in (0.7, 1.5, 4.0):
        _add_init_if_valid(out, rng.dirichlet(np.full(n, conc, dtype=np.float64)))

    if len(out) > 18:
        out = out[:18]
    return out


def _sort_pool(pool):
    if not pool:
        return pool
    pairs = [(float(pool[i][0]), i) for i in range(len(pool))]
    pairs.sort()
    pool[:] = [pool[i] for _, i in pairs]
    return pool


def _trim_pool(pool, max_size=40):
    _sort_pool(pool)
    if len(pool) > max_size:
        del pool[max_size:]
    return pool


def _add_pool(pool, a, score):
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size == 0 or not np.isfinite(score):
        return
    score = float(score)
    pool.append([score, a, int(a.size)])
    if len(pool) > 80:
        _trim_pool(pool, 60)


def _soft_pg_optimize(a, time_limit, rng=None):
    if rng is None:
        rng = np.random.default_rng()

    a = _normalize_nonng(a)
    n = a.size
    if n == 0:
        return a, float("inf"), float("inf")
    if time_limit <= 0.05:
        score, m, N = _score_and_max(a)
        return a, score, m

    N = _pow2_at_least(2 * n - 1)
    L = 2 * n - 1
    score, bestM, _ = _score_and_max(a, N)
    best = a.copy()

    t0 = time.time()
    no_improve = 0
    it = 0
    base_step = 1.0 / float(n)
    steps = base_step * np.array([0.02, 0.08, 0.25, 0.9, 2.5], dtype=np.float64)

    while time.time() - t0 < time_limit:
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        c = np.where(c > 0.0, c, 0.0)
        M = float(np.max(c))
        curr_score = float(2.0 * n * M)

        if M < bestM - 1e-15:
            best = a.copy()
            bestM = M
            score = curr_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2500:
            stag = no_improve / 3000.0
            sigma = 0.03 + 0.12 * min(3.0, stag)
            sigma = min(sigma, 0.45)
            kick = np.exp(rng.normal(0.0, sigma, n))
            a = _simplex_project(a * kick, 1.0)
            no_improve = 0
            it += 1
            continue

        elapsed_frac = min(1.0, (time.time() - t0) / max(0.5, float(time_limit)))
        ratio = 5.0 + 25.0 * elapsed_frac
        gamma = ratio / max(M, 1e-300)
        gamma = min(gamma, 200.0 / max(M, 1e-300))

        if gamma > 0.0:
            ex = np.exp(gamma * (c - M))
            s_ex = float(ex.sum())
            if np.isfinite(s_ex) and s_ex > 0.0:
                w = ex / s_ex
            else:
                top = np.argpartition(c, -16)[-16:]
                w = np.zeros(L, dtype=np.float64)
                w[top] = 1.0 / float(top.size)
        else:
            w = np.ones(L, dtype=np.float64) / float(L)

        Y = np.fft.rfft(w, N)
        g = 2.0 * np.fft.irfft(np.conj(A) * Y, N)[:n]
        g = np.asarray(g, dtype=np.float64)
        g -= g.mean()
        maxg = float(np.max(np.abs(g)))

        if maxg <= 1e-14 * max(1.0, float(np.max(a))):
            break

        d = -g / maxg

        best_step_a = None
        best_step_M = M
        for st in steps:
            v = a + st * d
            a2 = _simplex_project(v, 1.0)
            c2 = _fft_conv(a2, N)
            M2 = float(np.max(c2)) if c2.size else float("inf")
            if M2 < best_step_M:
                best_step_M = M2
                best_step_a = a2

        if best_step_a is not None and best_step_M < M - 1e-15:
            a = best_step_a
            best = a.copy()
            bestM = best_step_M
            score = float(2.0 * n * bestM)
            no_improve = 0
        else:
            if no_improve > 800 and rng.random() < 0.12:
                sigma = 0.02 + 0.05 * min(1.0, no_improve / 2000.0)
                kick = np.exp(rng.normal(0.0, sigma, n))
                a = _simplex_project(a * kick, 1.0)
                no_improve = 0

        it += 1

    return best, score, bestM


def _lp_indices(n):
    key = int(n)
    if key in _LP_INDEX_CACHE:
        return _LP_INDEX_CACHE[key]
    idx = np.arange(n, dtype=np.int64)
    rows = np.tile(idx, n) + np.repeat(idx, n)
    cols = np.repeat(idx, n)
    _LP_INDEX_CACHE[key] = (rows, cols)
    return _LP_INDEX_CACHE[key]


def _lp_direction(a, M):
    n = a.size
    if n == 0 or n > LP_MAX_N:
        return None
    if not np.isfinite(M) or M <= 0.0:
        return None

    rows, cols = _lp_indices(n)
    Mw = float(M) * float(n)
    data = np.tile(a * float(n), n)
    shape = (2 * n - 1, n)

    try:
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=shape)
        c = -np.ones(n, dtype=np.float64)
        slack = 1e-9 + 1e-8 * max(1.0, Mw)
        b_ub = np.full(shape[0], Mw + slack, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if not getattr(res, "success", False):
        return None

    b = np.asarray(res.x, dtype=np.float64)
    b[~np.isfinite(b)] = 0.0
    np.clip(b, 0.0, None, out=b)
    B = float(b.sum())
    if not np.isfinite(B) or B <= 1.0 + 1e-12:
        return None
    return b / B


def _lp_update(a, time_limit, rng=None):
    a = _normalize_nonng(a)
    n = a.size
    if n == 0 or time_limit <= 0.05 or n > LP_MAX_N:
        score, m, _ = _score_and_max(a)
        return a, score, m

    N = _pow2_at_least(2 * n - 1)
    score, bestM, _ = _score_and_max(a, N)
    best = a.copy()

    t0 = time.time()
    attempts = 0

    while time.time() - t0 < time_limit and attempts < 10:
        g = _lp_direction(best, bestM)
        if g is None:
            break
        attempts += 1

        improved = False
        for t in (0.02, 0.05, 0.10, 0.20, 0.35, 0.55, 0.85):
            a2 = _simplex_project((1.0 - t) * best + t * g, 1.0)
            _, M2, _ = _score_and_max(a2, N)
            if M2 < bestM - 1e-15:
                best = a2
                bestM = M2
                improved = True
                break

        if not improved:
            break

        if time.time() - t0 < time_limit - 2.0:
            a3, s3, m3 = _soft_pg_optimize(
                best, min(2.5, max(0.2, time_limit - (time.time() - t0))), rng
            )
            if np.isfinite(m3) and m3 < bestM:
                best = a3
                bestM = m3

    return best, float(2.0 * n * bestM), bestM


def _block_search(a, time_limit, rng):
    a = _normalize_nonng(a)
    n = a.size
    if n == 0 or time_limit <= 0.05:
        score, m, _ = _score_and_max(a)
        return a, score, m

    N = _pow2_at_least(2 * n - 1)
    score, bestM, _ = _score_and_max(a, N)
    best = a.copy()

    t0 = time.time()
    scale = 0.05
    idx = np.arange(n, dtype=np.float64)

    while time.time() - t0 < time_limit:
        cand = best.copy()
        nblocks = int(rng.integers(1, 4))

        for _ in range(nblocks):
            maxw = max(2, n // 10)
            width = int(rng.integers(1, maxw + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand[start:stop] *= np.exp(rng.normal(0.0, scale, stop - start))

            if rng.random() < 0.25:
                center = float(rng.integers(0, n))
                sigma = float(max(3, n // 40))
                amp = 0.05 * float(max(np.mean(cand), 1.0 / n))
                cand += amp * np.exp(-((idx - center) ** 2) / (2.0 * sigma * sigma))

        if rng.random() < 0.20:
            cand = 0.5 * (cand + cand[::-1])

        cand = _normalize_nonng(cand)
        _, M2, _ = _score_and_max(cand, N)

        if M2 < bestM:
            best = cand
            bestM = M2
            score = float(2.0 * n * bestM)
            scale = max(0.015, scale * 0.88)
        else:
            scale = min(0.35, scale * 1.03)

    return best, score, bestM


def _exact_score(a):
    a = _normalize_nonng(a)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    m = float(np.max(c)) if c.size else float("inf")
    return float(2.0 * n * m / (s * s))


def propose_candidate(seed=0, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if budget_s is None:
        total = 1800.0
    else:
        total = float(budget_s)
    total = max(30.0, min(total, 1795.0))

    reserve = min(180.0, max(30.0, 0.10 * total))
    deadline = t_start + max(5.0, total - reserve)

    base = None
    try:
        base = _normalize_nonng(height_sequence_1)
        if base.size < 2:
            base = None
    except Exception:
        base = None

    pool = []

    if total > 600.0:
        ns = [128, 256, 512, 1024, 2048]
    elif total > 300.0:
        ns = [128, 256, 512, 1024]
    else:
        ns = [128, 256]

    survey_budget = min(120.0, max(15.0, 0.07 * total))
    survey_end = min(deadline - max(30.0, 0.25 * total), t_start + survey_budget)

    for n in ns:
        for a in _make_inits(n, rng, base):
            if time.time() > survey_end:
                break
            quick = max(0.2, min(2.0, survey_budget / 24.0))
            a2, sc2, _ = _soft_pg_optimize(a, quick, rng)
            _add_pool(pool, a2, sc2)

        if time.time() > survey_end:
            break

    _trim_pool(pool, 45)
    if not pool:
        a = np.ones(256, dtype=np.float64) / 256.0
        score, _, _ = _score_and_max(a)
        pool.append([score, a, 256])

    _sort_pool(pool)
    best_score = float(pool[0][0])
    best_a = np.asarray(pool[0][1], dtype=np.float64).copy()
    best_n = int(pool[0][2])
    print("initial best score", best_score, "n", best_n)

    long_budget = min(360.0, max(40.0, 0.22 * total))
    min_remain_after_long = max(90.0, block_budget if False else reserve)
    if time.time() + long_budget < deadline - 60.0:
        b = min(long_budget, max(1.0, deadline - 60.0 - time.time()))
        a2, sc2, _ = _soft_pg_optimize(best_a, b, rng)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_a = a2
            best_n = int(best_a.size)
            _add_pool(pool, best_a, best_score)
            _trim_pool(pool, 45)
            print("long soft improved", best_score, "n", best_n)

    _sort_pool(pool)
    best_a = np.asarray(pool[0][1], dtype=np.float64).copy()
    best_n = int(pool[0][2])
    best_score = float(pool[0][0])

    lp_budget = min(220.0, max(30.0, 0.12 * total))
    lp_end = min(deadline - 120.0, t_start + 1795.0 - reserve + lp_budget)
    lp_candidates = []

    for item in pool[:8]:
        if item[2] <= LP_MAX_N:
            lp_candidates.append(np.asarray(item[1], dtype=np.float64).copy())

    if best_n > LP_MAX_N:
        lp_candidates.append(_resample_seq(best_a, min(LP_MAX_N, 256)))
        lp_candidates.append(_resample_seq(best_a, LP_MAX_N))

    for cand in lp_candidates:
        if time.time() > lp_end:
            break
        t_left_lp = max(1.0, min(45.0, lp_end - time.time()))
        a2, sc2, _ = _lp_update(cand, t_left_lp, rng)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_a = a2
            best_n = int(best_a.size)
            _add_pool(pool, best_a, best_score)
            _trim_pool(pool, 45)
            print("LP improved", best_score, "n", best_n)

    block_budget = min(180.0, max(30.0, 0.10 * total))
    block_start = max(deadline, t_start + total - block_budget)
    exploration_end = max(time.time(), deadline - block_budget - 10.0)

    while time.time() < exploration_end:
        slice_s = min(20.0, max(1.0, exploration_end - time.time()))
        if slice_s <= 0.0:
            break

        if rng.random() < 0.65:
            a = best_a.copy()
        elif len(pool) > 0:
            k = int(rng.integers(0, min(len(pool), 8)))
            a = np.asarray(pool[k][1], dtype=np.float64).copy()
            if base is not None and rng.random() < 0.25:
                a = _resample_seq(_resample_seq(base, a.size), a.size)
        else:
            a = best_a.copy()

        if rng.random() < 0.15:
            a *= np.exp(rng.normal(0.0, 0.04, a.size))
            a = _normalize_nonng(a)

        a2, sc2, _ = _soft_pg_optimize(a, slice_s, rng)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_a = a2
            best_n = int(best_a.size)
            _add_pool(pool, best_a, best_score)
            _trim_pool(pool, 45)

    if time.time() < deadline - 2.0:
        b = max(1.0, deadline - 3.0 - time.time())
        a2, sc2, _ = _block_search(best_a, b, rng)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_a = a2
            best_n = int(best_a.size)

    _sort_pool(pool)
    best_a = _normalize_nonng(pool[0][1])
    best_score = float(pool[0][0])

    final_candidates = []
    seen = set()
    final_candidates.append((best_a, best_score))
    seen.add(best_a.size)

    for item in pool[:8]:
        n = int(item[2])
        a = np.asarray(item[1], dtype=np.float64)
        if n == 0 or n > 2600:
            continue
        key = (n, tuple(round(float(x) * 1000.0) for x in a[::max(1, n // 32)]))
        if key in seen:
            continue
        seen.add(key)
        final_candidates.append((a, float(item[0])))

    verified_best_a = best_a
    verified_best_score = best_score

    for a, sc in final_candidates[:8]:
        if time.time() > deadline - 3.0:
            break
        a = _normalize_nonng(a)
        if a.size == 0 or a.size > 2600:
            continue
        try:
            esc = _exact_score(a)
        except Exception:
            esc = float("inf")
        if np.isfinite(esc) and esc < verified_best_score:
            verified_best_score = float(esc)
            verified_best_a = a

    verified_best_a = _normalize_nonng(verified_best_a)
    if verified_best_a.size == 0:
        verified_best_a = _normalize_nonng(np.ones(256, dtype=np.float64))

    print("final best score", verified_best_score, "n", verified_best_a.size)
    return [float(x) for x in verified_best_a]
