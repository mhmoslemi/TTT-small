**Key insight:** For non-negative sequences, the "DC component" (the average) always contributes a constant to the autocorrelation, and the AC components add oscillating terms. The total autocorrelation is `DC^2 + 2*DC*AC + AC^2` (in the frequency domain, it's `|A(k)|^2 = |DC + AC(k)|^2`). To minimize the peak, we want `|A(k)|` to be as flat as possible.

For a non-negative sequence, the DC component is `sum(a)/n > 0`. The AC components must cancel the DC at some frequencies to make `|A(k)|` small there, but they add to the DC at other frequencies.

The optimal solution would have `|A(k)|` constant for all k, which would mean `conv(a,a)` is flat (a delta in the frequency domain, or a constant in the time domain). But for a non-negative real sequence, this is impossible (it would require the sequence to be constant, giving a triangular autocorrelation).

So the best we can do is make the autocorrelation "as flat as possible" while keeping the sequence non-negative.

```python
def build_convolution_matrix(a):
    """Build the sparse convolution matrix T where (T@g)[k] = sum_i a[i]*g[k-i]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
```

```python
def coordinate_descent_step(a, K=20):
    """Do one step of coordinate descent to reduce peak autocorrelation."""
    n = len(a)
    b = np.convolve(a, a)
    peak_lag = b.argmax()
    peak_val = b[peak_lag]
    
    # Find coordinates to decrease (high contribution to peak)
    # Contribution of a[i] to b[peak_lag] is a[i] * a[peak_lag - i]
    contrib = np.zeros(n)
    lo = max(0, peak_lag - n + 1)
    hi = min(n - 1, peak_lag)
    for i in range(lo, hi + 1):
        contrib[i] = a[i] * a[peak_lag - i]
    
    # Top K contributors to decrease
    top_decrease = np.argsort(contrib)[-K:]
    
    # Coordinates to increase (low contribution, and not in the peak)
    # We want to increase a[j] where a[peak_lag - j] is small
    # (to not increase the peak)
    remaining = [j for j in range(n) if j not in set(top_decrease.tolist())]
    # Score: how much increasing a[j] hurts (should be small)
    scores = np.array([a[peak_lag - j] if 0 <= peak_lag - j < n else 0 for j in remaining])
    # Also prefer to increase coordinates that help reduce other high lags
    top_increase_idx = np.argsort(scores)[:K]
    top_increase = np.array(remaining)[top_increase_idx]
    
    # Perform swap: move mass from top_decrease to top_increase
    mass_to_move = min(a[top_decrease].min(), 0.1 * a[top_decrease].min())
    # Actually, let's move a fixed fraction
    frac = 0.1
    for i in top_decrease:
        delta = a[i] * frac
        a[i] -= delta
        # Distribute to top_increase
        for j in top_increase:
            a[j] += delta / len(top_increase)
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a
```

```python
"""
LP-based conditional gradient method for minimizing 2n*max(conv(a,a))/sum(a)^2
with non-negative sequences. Uses active-set LP for efficiency, with fallback
to random/bump perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def build_conv_matrix(a):
    """Build sparse convolution matrix T where (T@g)[k] = sum_i a[i]*g[k-i]."""
    n = len(a)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()

def solve_lp(a, M, timeout=25):
    """Solve: max 1^T g s.t. T_a @ g <= M*1, g >= 0."""
    n = len(a)
    try:
        T = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'output_flag': 0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception as e:
        print(f"LP error: {e}")
    return None

def solve_lp_active(a, M, frac=0.9, timeout=20):
    """Solve LP with only near-active constraints."""
    n = len(a)
    b = np.convolve(a, a)
    active = np.where(b >= frac * M)[0]
    
    rows = []
    cols = []
    data = []
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            data.extend((a[k - hi:k - lo + 1][::-1]).tolist())
    
    if not rows:
        return None
    
    T = coo_matrix((data, (rows, cols)), shape=(len(active), n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'output_flag': 0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            # Verify against full constraints
            check = np.convolve(a, g)
            if check.max() <= M * (1 + 1e-4):
                return g
            else:
                scale = M / check.max()
                return g * scale
    except Exception:
        pass
    return None
```
