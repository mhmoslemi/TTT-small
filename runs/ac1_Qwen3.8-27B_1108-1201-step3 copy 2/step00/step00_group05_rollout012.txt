```python
import numpy as np
import scipy.sparse
import scipy.optimize
import time

def solve_lp(a, M):
    """Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    # Build the constraint matrix A where A[k,i] = a[k-i] if 0 <= k-i < n
    # Rows: k = 0, ..., 2n-2 (2n-1 rows)
    # Cols: i = 0, ..., n-1 (n cols)
    
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # b_i contributes to (a*b)_k for k = i, i+1, ..., i+n-1
        # (a*b)_k = sum_j a_{k-j} * b_j, so b_i contributes a_{k-i} to row k
        # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
        ks = range(i, i + n)
        for k in ks:
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(b) (linprog minimizes)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    return None
```

```python
def build_convolution_matrix(a, n):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    # Each column i has non-zeros in rows i to i+n-1

    for i in range(n):
        k_start = i
        k_end = i + n
        rows.extend(range(k_start, k_end))
        cols.extend([i] * n)
        vals.extend(a)
```

```python
def build_A(a, n):
    """Build sparse matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    # Column i has non-zeros in rows i to i+n-1 (clamped to [0, 2n-2])
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i

        # a[k-i] for k = k_start to k_end-1

        rows.extend(range(k_start, k_end))
        cols.extend([i] * (k_end - k_start))
        vals.extend(a[idx])
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
def build_A(a, n):
    """Build sparse matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    # Use the fact that each column is a shifted version of a
    # Column i: rows i to i+n-1, values a[0] to a[n-1]
    # But we need to clip to [0, 2n-2]
    
    # Total rows: 2n-1, Total cols: n
    # For col i: rows max(0, i) to min(2n-2, i+n-1)
    # Values: a[max(0, row-i)] for each row
    
    # Vectorized construction:
    col_idx = np.repeat(np.arange(n), n)  # Each column index repeated n times
    # For column i, the rows are i, i+1, ..., i+n-1
    # But we need to clip: row = i + j for j = 0,...,n-1, clipped to [0, 2n-2]
    row_offsets = np.tile(np.arange(n), n)  # 0,1,...,n-1, 0,1,...,n-1, ...
    rows = col_idx + row_offsets  # i + j for each (i,j)
    # Clip to valid range
    valid = (rows >= 0) & (rows <= 2*n-2)
    rows = rows[valid]
    cols = col_idx[valid]
    vals = np.tile(a, n)[valid]  # a[j] for each valid (i,j)
    
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
def subgradient_step(a, lr):
    """One subgradient step to reduce max convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    S = np.where(conv >= M - 1e-15)[0]  # Indices where conv is at max
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in S:
        # (B_k * a)[i] = a[k-i] if 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # (B_k * a)[i] = a[k-i] for i in [i_start, i_end]
            # Let j = k-i, so j ranges from k-i_end to k-i_start
            j_start = k - i_end
            j_end = k - i_start
            grad[i_start:i_end+1] += a[j_start:j_end+1][::-1]
    
    grad = 2.0 * grad / len(S)
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / a_new_sum
    
    return a_new
```
