"""
Minimizer for step-function convolution objective.

Key insight: monotone (power-law) ramps give dramatically low scores far below
the uniform baseline (score 2).  A sequence a[i] = (i + c)^p, scaled so its
maximum is well within the evaluator's [0, 1000] box (so no value is clamped
and the shape is preserved) yields a score that decreases as the power p grows.
Because the objective 2*n*max(conv)/sum^2 is scale-invariant, we are free to
scale every candidate to max = 500 without changing its score.

The program sweeps (length, power, offset) on this family with a time guard,
keeps the exact evaluator's verdict (via evaluate_sequence), then refines the
winner with a fine continuous (power, offset) re-sweep plus a safe block
multiplicative local search that only ever accepts improvements.  It always
returns the best scale-valid sequence it found before the 1800 s budget.
"""

import numpy as np
import time


def make_ramp(n, p, c):
    """Return a nonnegative power-ramp of length n scaled to max = 500.0."""
    i = np.arange(n, dtype=np.float64)
    base = i + max(1e-6, float(c))
    f = np.power(base, float(p))
    if not np.all(np.isfinite(f)):
        f = np.power(base, 1.0)
    m = float(f.max())
    if not np.isfinite(m) or m <= 0.0:
        return np.ones(n, dtype=np.float64)
    f = f / m * 500.0
    return np.clip(f, 0.0, 1000.0)


def safe_eval(seq):
    """Call the (externally provided) evaluator, never raising."""
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float("inf")


def block_perturb(base, rng):
    """One nonnegative multiplicative block perturbation, kept in [0, 1000]."""
    cand = np.array(base, dtype=np.float64, copy=True)
    n = cand.size
    width = int(rng.integers(1, max(2, n // 16)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.2)))
    # Occasionally apply a light monotone re-sharpening to explore steeper ramps.
    if rng.random() < 0.25:
        idx = np.arange(n, dtype=np.float64)
        w = 0.5 + 0.5 * idx / max(1.0, n - 1.0)
        cand = cand * np.exp(rng.normal(0.0, 0.15) * w)
    return np.clip(cand, 0.0, 1000.0)


def rescale_max(seq, target=500.0):
    """Scale a nonnegative vector so its maximum equals ``target`` (safe)."""
    a = np.array(seq, dtype=np.float64)
    m = float(a.max())
    if not np.isfinite(m) or m <= 0.0:
        return np.ones(a.size, dtype=np.float64) * target
    a = a / m * target
    return np.clip(a, 0.0, 1000.0)

    # ---- Phase 1: broad discrete sweep over the power-ramp family ----

    if best is None:
