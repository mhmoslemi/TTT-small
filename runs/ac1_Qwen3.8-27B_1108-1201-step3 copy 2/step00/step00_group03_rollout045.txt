This makes intuitive sense: to reduce the center autocorrelation `sum a[j]*a[n-1-j]`, we should reduce the product `a[j]*a[n-1-j]`, which we do by reducing whichever is larger (or both).

For off-center lags, the gradient is a window of a (not the full sequence), so the step is more localized.

            p_score = _score_fast(a_pert, size)
            delta = p_score - score
            if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-max(delta, 0) / T)):
                a = a_pert
                score = p_score
                v = np.zeros(n)
            stuck = 0
            T = max(T * 0.92, 1e-9)

        # Block perturbation every 5000 iters
        if total_iters % 5000 == 0:
            a_block = _block_perturb(a, rng, n)
            b_score = _score_fast(a_block, size)
            delta = b_score - score
            if delta < 0 or (T > 1e-9 and rng.random() < math.exp(-max(delta, 0) / T)):
                a = a_block
                score = b_score
                v = np.zeros(n)

        # Reduce LR when stuck for long
        if stuck > 10000:
            lr = max(0.001, lr * 0.5)
            stuck = 0

        # Progress logging
        if time.monotonic() - last_print > 30:
            elapsed = time.monotonic() - t0
            rate = total_iters / elapsed
            print(f"  iter={total_iters}, n={n}, best={best_score:.6f}, "
                  f"cur={score:.6f}, rate={rate:.0f}/s, "
                  f"elapsed={elapsed:.0f}s, lr={lr:.4f}, T={T:.1e}")
            last_print = time.monotonic()

    # Final: scale and return
    target_sum = math.sqrt(2.0 * best_a.size)
    result = best_a * target_sum
    result = np.clip(result, 0.0, 1000.0)

    # Verify with actual evaluator
    final_eval = evaluate_sequence([float(x) for x in result])
    print(f"FINAL: n={result.size}, evaluate_sequence={final_eval:.6f}, "
          f"elapsed={time.monotonic() - t0:.0f}s, total_iters={total_iters}")

    return [float(x) for x in result]
