    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)

    # dc/da[j] = a[m_star - j] + a[j] (with appropriate index handling)
    grad_peak = np.zeros(n)
    for j in range(n):

        if 0 <= idx < n:

For a non-negative sequence, making the spectrum flat is constrained. But there are sequences (like Barker sequences, Zadoff-Chu sequences) that have nearly flat spectra.

The key insight: we want a non-negative sequence whose DTFT has approximately constant magnitude. This is the "minimum peak-to-average power ratio" (PAPR) problem for the autocorrelation, or equivalently, the "flat minimum" problem.

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:

    return res.x

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a combination of gradient descent and LP-based refinement (inspired by the
minimum peak autocorrelation approach). Multiple starting points and adaptive
step sizes are used to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-10:
        return a
    return a * (target / S)


def _grad_step(a, lr):
    """One projected gradient step on the peak autocorrelation."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    
    # Gradient of c[m_star] w.r.t. a[j] = 2*a[m_star-j] if valid
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    
    # Project out mean (sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_t)
    return a


def _lp_step(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix A where A[m,j] = f[m-j] if 0<=m-j<n
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            # A[m, lo:hi+1] = f[m-hi:m-lo:-1]
            A[m, lo:hi + 1] = f[m - hi:m - lo:-1]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None
