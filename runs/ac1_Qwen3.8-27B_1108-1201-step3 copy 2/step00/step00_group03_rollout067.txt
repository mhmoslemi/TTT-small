"""
LP-projection + subgradient hybrid search for the minimum-peak self-convolution
of a non-negative step function.

Score = 2*n*max(convolve(a,a)) / (sum a)^2  (scale-invariant), so we fix
sum(a) = sqrt(2n) during the search and only the shape matters.

Two complementary moves:
  * LP step   : maximise sum(g)  s.t.  (f*g)_k <= M, g >= 0  (M = max(f*f)),
                then blend f <- (1-t)f + t*g with a line search on t.
                This is the state-of-the-art projection and escapes the flat
                (uniform) plateau.
  * subgradient: fast O(n) descent on max_k (f*f)_k to make quick local gains
                between (and as a fallback for) the more expensive LP calls.

Multiple asymmetric initialisations (ramps, tents, the provided sequence) are
tried because asymmetric shapes push the convolution peak off the centre and
can reach scores < 2, which the LP then smooths out.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse


def _build_conv_matrix(f, n):
    """Sparse banded matrix A with A[k, i] = f[i] when 0 <= k-i <= n-1."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi < lo:
            continue
        # i in [lo, hi]  ->  rows k, cols lo..hi, values f[lo..hi]
        idx = np.arange(lo, hi + 1)
        rows.extend([k] * (hi - lo + 1))
        cols.extend(idx.tolist())
        data.extend(f[lo:hi + 1].tolist())
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(f, M, n, time_limit):
    """Solve max sum(g) s.t. f*g <= M, g >= 0.  Returns g or None."""
    if M <= 0.0:
        return None
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)},
        )
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if not np.isfinite(g).all() or np.sum(g) < 1e-12:
        return None
    return g


def _subgrad_step(f, target_sum, alpha, rng):
    """One projected subgradient step on max_k (f*f)_k with sum fixed."""
    n = len(f)
    c = np.convolve(f, f)
    k = int(np.argmax(c))
    sg = np.zeros(n)
    lo = k - n + 1
    if lo < 0:
        lo = 0
    hi = k
    if hi > n - 1:
        hi = n - 1
    if hi >= lo:
        sg[lo:hi + 1] = f[(k - lo):(k - hi - 1):-1]
    # gradient of a constant-sum normalisation of (f - alpha*sg)
    step = alpha * (sg - sg.mean())
    fnew = f + step
    fnew = np.maximum(fnew, 0.0)
    s = fnew.sum()
    if s < 1e-12:
        fnew = f + rng.normal(0.0, 1e-3, n)
        fnew = np.maximum(fnew, 0.0)
        s = fnew.sum()
        if s < 1e-12:
            return f
    fnew = fnew * (target_sum / s)
    return np.clip(fnew, 0.0, 1000.0)

                inits.append(h.copy())
        except Exception:
            pass
    base = np.arange(1, n + 1, dtype=np.float64)
    inits.append(base)
    inits.append(base[::-1].copy())
    # asymmetric tent: steep rise, gentle fall
    mid = int(0.35 * (n - 1))
    tent = np.concatenate([
        np.arange(1, mid + 2, dtype=np.float64),
        np.linspace(mid + 1, 0.2, n - 1 - mid, dtype=np.float64),
    ])
    inits.append(np.clip(tent, 0.0, None))
    # gentle bell (Hann) as a smooth symmetric reference
    xx = np.linspace(0.0, np.pi, n, dtype=np.float64)
    inits.append(0.5 * (1.0 - np.cos(xx)))
    # random smooth positive sequence
    ramp_smooth = np.cumsum(np.abs(rng.normal(0.0, 1.0, n))) + 1.0
    inits.append(ramp_smooth)
    return inits
