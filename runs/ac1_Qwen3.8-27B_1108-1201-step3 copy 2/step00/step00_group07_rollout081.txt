```python
"""
Multi-start optimizer for nonnegative coefficient sequences.

The main search uses a log-domain multiplicative descent direction derived from
the self-convolution.  For several smoothed weightings of hot convolution
entries, it computes the log-gradient against a variable log(a), applies
multiplicative updates of the form a <- a * exp(-shift * shape), normalizes
the sum, and keeps the best FFT-approximated score.  Periodic random
perturbations and multiple starting points are used to avoid local minima.
The final answer is exactly checked with evaluate_sequence on a small pool of
the best candidates.
"""

import numpy as np
import time


def _next_power_of_two(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(a, S):
    """Return a nonnegative finite vector with sum S (unless impossible)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
        S = np.sqrt(2.0)

    a = a.copy()
    a[~np.isfinite(a)] = 0.0
    np.clip(a, 0.0, 1000.0, out=a)

    n = a.size
    floor = 1e-12 * max(1.0, float(S)) / max(1, n)
    a += floor

    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a.fill(1.0)
        s = float(a.size)

    a *= float(S) / s
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _fast_conv(a, L):
    """Linear self-convolution using FFT, clipped to be nonnegative."""
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)
    n = a.size
    out = out[:2 * n - 1]
    out = np.maximum(out, 0.0)
    return out


def _conv_max(a, L):
    """Max entry of linear self-convolution, computed with FFT."""
    b = _fast_conv(a, L)
    return float(np.max(b))


def _cross_corr_with_A(A, w, L, n):
    """
    H[i] = sum_l a[l] * w[i+l], for i=0..n-1.

    A must be rfft(a, L).  w has length 2n-1.  L must be large enough to avoid
    circular aliasing for this cross-correlation.
    """
    W = np.fft.rfft(w, L)
    out = np.fft.irfft(W * np.conj(A), L)
    out = out[:n]
    out[~np.isfinite(out)] = 0.0
    np.maximum(out, 0.0, out=out)
    return out


def _make_shape(a, H):
    """Normalize a positive log-gradient signal into a bounded shape in [0,1]."""
    rel = a * H
    rel[~np.isfinite(rel)] = 0.0
    mn = float(np.min(rel))
    mx = float(np.max(rel))
    if mx - mn <= 1e-12:
        return None
    shape = (rel - mn) / (mx - mn)
    shape[~np.isfinite(shape)] = 0.0
    return shape


def _softmax_w(b, beta):
    """Softmax weight over convolution entries; returns normalized weights."""
    m = float(np.max(b))
    if not np.isfinite(m):
        return None
    e = np.exp(beta * (b - m))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return e / s


def _excess_w(b, target, gamma=1.0):
    """Weights proportional to (b-target)_+^gamma; returns normalized weights."""
    target = float(target)
    over = b > target
    if not np.any(over):
        return None
    vals = b[over] - target
    if gamma != 1.0:
        vals = np.power(vals, gamma)
    s = float(np.sum(vals))
    if s <= 0.0 or not np.isfinite(s):
        return None
    w = np.zeros_like(b, dtype=np.float64)
    w[over] = vals / s
    return w


def _unique_shifts(shifts):
    """Clean and deduplicate a small tuple/list of positive shift sizes."""
    out = []
    for x in shifts:
        try:
            s = float(x)
        except Exception:
            continue
        if not np.isfinite(s) or s <= 1e-4:
            continue
        if s > 12.0:
            s = 12.0
        close = False
        for t in out:
            if abs(s - t) <= 0.08 * max(1.0, t):
                close = True
                break
        if not close:
            out.append(s)
    out.sort()
    return out


def _perturb_sequence(base, rng, sigma, S):
    """Multiplicative log-normal perturbation, keeping sequence valid."""
    base = np.asarray(base, dtype=np.float64).copy()
    if base.size == 0:
        return np.ones(1, dtype=np.float64)
    sigma = float(max(1e-6, sigma))
    noise = np.exp(rng.normal(0.0, sigma, size=base.size))
    if rng.random() < 0.25:
        noise = 0.5 * (noise + noise[::-1])
    return _normalize(base * noise, S)


def _make_start(n, variant, rng, original=None):
    """Create a valid initial sequence of length n. variant selects the family."""
    S = np.sqrt(2.0 * float(n))
    variant = int(variant)
    a = None

    if variant == 0:
        a = np.ones(n, dtype=np.float64)

    elif variant == 1:
        # Anti-palindromic random two-level-ish start.
        a = np.ones(n, dtype=np.float64)
        half = n // 2
        t = rng.random(half)
        for i in range(half):
            a[i] = t[i]
            a[n - 1 - i] = 1.0 - t[i]
        if n % 2 == 1:
            a[half] = 0.5
        a = np.maximum(a, 0.0)

    elif variant == 2:
        # Smooth odd-harmonic anti-palindromic-ish start.
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n, dtype=np.float64)
        for k in range(1, 5):
            amp = rng.uniform(0.05, 0.20)
            a += amp * np.cos(np.pi * (2 * k - 1) * x)
        a = np.maximum(a, 0.0)

    elif variant == 3:
        # Blocky anti-symmetric random start.
        K = max(16, n // 128)
        chunk = max(1, n // K)
        a = np.zeros(n, dtype=np.float64)
        for b_idx in range(K):
            level = rng.uniform(0.1, 1.0)
            start = b_idx * chunk
            stop = min(n, start + chunk)
            a[start:stop] = level
        a = 0.5 * (a + (1.0 - a[::-1]))
        a = np.maximum(a, 0.0)

    elif variant == 4 and original is not None:
        orig = np.asarray(original, dtype=np.float64).ravel()
        if orig.size > 1:
            xo = np.linspace(0.0, 1.0, orig.size)
            xn = np.linspace(0.0, 1.0, n)
            a = np.interp(xn, xo, orig)
        else:
            a = np.ones(n, dtype=np.float64)
        a = np.maximum(a, 0.0)

    else:
        a = rng.normal(1.0, 0.45, size=n)
        a = np.maximum(a, 0.0)

    if a is None or not np.all(np.isfinite(a)):
        a = np.ones(n, dtype=np.float64)
    return _normalize(a, S)


def _pool_push(pool, seq, score):
    """Maintain a small pool keyed by FFT score."""
    if seq is None:
        return False
    if not np.isfinite(score):
        return False
    score = float(score)

    arr = np.asarray(seq, dtype=np.float64).copy()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return False

    max_pool = 50
    if len(pool) >= max_pool:
        worst_idx = 0
        worst_val = -np.inf
        for i, item in enumerate(pool):
            if item[0] > worst_val:
                worst_val = item[0]
                worst_idx = i
        if score >= worst_val:
            return False
        pool.pop(worst_idx)

    pool.append((score, arr))
    return True


def _fft_key(item):
    return item[0]


def _optimize(a, deadline, rng, tag, pool, original=None):
    """
    Optimize one sequence.  Returns (best_sequence, best_fft_score).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(256, dtype=np.float64)

    n = a.size
    S = np.sqrt(2.0 * float(n))
    L = _next_power_of_two(2 * n - 1)
    Lgrad = _next_power_of_two(3 * n - 2)

    a = _normalize(a, S)
    b = _fast_conv(a, L)
    M = float(np.max(b))

    best = a.copy()
    best_M = M
    _pool_push(pool, best, best_M)

    shift_ref = 0.5
    no_impr = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        if M <= 0.0 or not np.isfinite(M):
            a = best.copy()
            b = _fast_conv(a, L)
            M = float(np.max(b))

        if M < best_M - 1e-9:
            best_M = M
            best = a.copy()
            _pool_push(pool, best, best_M)
            no_impr = 0
        else:
            no_impr += 1

        if best_M <= 1.000005:
            break

        # If the current point has wandered far from the best, recenter it.
        if no_impr > 10 and M > best_M + 0.20:
            a = _perturb_sequence(best, rng, 0.05, S)
            b = _fast_conv(a, L)
            M = float(np.max(b))
            no_impr = 0
            continue

        avg = float(np.sum(b)) / float(2 * n - 1)
        if avg <= 0.0 or not np.isfinite(avg):
            avg = 1.0

        # Build several smoothed descent weightings.
        weights = []

        ws8 = _softmax_w(b, 8.0)
        if ws8 is not None:
            weights.append(ws8)

        wavg = _excess_w(b, avg, 1.0)
        if wavg is not None:
            weights.append(wavg)

        if no_impr > 10:
            ws20 = _softmax_w(b, 20.0)
            if ws20 is not None:
                weights.append(ws20)

        mid = avg + 0.5 * (M - avg)
        wmid = _excess_w(b, mid, 1.0)
        if wmid is not None:
            weights.append(wmid)

        if no_impr > 25:
            wtop = _excess_w(b, M - 0.05 * max(1e-6, M - avg), 1.0)
            if wtop is not None:
                weights.append(wtop)

            wavg05 = _excess_w(b, avg, 0.5)
            if wavg05 is not None:
                weights.append(wavg05)

        A = np.fft.rfft(a, Lgrad)

        best_cand = None
        best_cand_M = M
        best_s = shift_ref

        shift_list = _unique_shifts((0.05, 0.20, 0.50, shift_ref, shift_ref * 2.0, shift_ref * 4.0))
        max_shapes = 5 if no_impr > 25 else 3
        if len(weights) > max_shapes:
            weights = weights[:max_shapes]

        for w in weights:
            if time.monotonic() + 1.0 > deadline:
                break
            H = _cross_corr_with_A(A, w, Lgrad, n)
            shape = _make_shape(a, H)
            if shape is None:
                continue

            for s in shift_list:
                if time.monotonic() + 0.2 > deadline:
                    break
                cand = a * np.exp(-float(s) * shape)
                cand = _normalize(cand, S)
                b2 = _fast_conv(cand, L)
                m2 = float(np.max(b2))
                if np.isfinite(m2) and m2 < best_cand_M - 1e-12:
                    best_cand_M = m2
                    best_cand = cand
                    best_s = float(s)

        if best_cand is not None:
            a = best_cand
            if best_cand_M < M:
                no_impr = 0
                shift_ref = max(0.05, min(10.0, 0.7 * shift_ref + 0.3 * best_s))
            else:
                shift_ref = max(0.05, min(10.0, shift_ref * 0.8))
            M = best_cand_M
        else:
            sigma = min(1.0, 0.05 * (1.0 + float(no_impr) / 20.0))
            a = _perturb_sequence(best, rng, sigma, S)
            b = _fast_conv(a, L)
            M = float(np.max(b))
            shift_ref = 0.5
            no_impr += 1

        if no_impr > 500:
            a = _perturb_sequence(best, rng, 1.0, S)
            b = _fast_conv(a, L)
            M = float(np.max(b))
            no_impr = 0

        if time.monotonic() - last_print > 30.0:
            print(tag + " cur=%.6f best=%.6f no_impr=%d" % (M, best_M, no_impr))
            last_print = time.monotonic()

    return best, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best sequence within the given time budget.
    """
    rng = np.random.default_rng(seed)
    total = float(max(1.0, float(budget_s)))
    horizon = time.monotonic() + max(0.0, total - 35.0)

    pool = []

    original = globals().get("height_sequence_1", None)
    orig_arr = None
    if original is not None:
        tmp = np.asarray(original, dtype=np.float64).ravel()
        tmp = tmp[np.isfinite(tmp)]
        if tmp.size > 0:
            tmp = np.clip(tmp, 0.0, 1000.0)
            # Seed the pool with the provided construction, using an FFT proxy.
            S0 = np.sqrt(2.0 * float(tmp.size))
            tnorm = _normalize(tmp, S0)
            L0 = _next_power_of_two(2 * tmp.size - 1)
            m0 = _conv_max(tnorm, L0)
            _pool_push(pool, tnorm, m0)
            orig_arr = tmp

    best_seq = None
    best_fft = np.inf

    # If nothing is available yet, add a safe uniform fallback.
    if len(pool) == 0:
        fallback = np.ones(256, dtype=np.float64)
        fallback = _normalize(fallback, np.sqrt(2.0 * 256.0))
        pool.append((2.0, fallback.copy()))

    # Update global best from the initial pool.
    for item in pool:
        if item[0] < best_fft:
            best_fft = item[0]
            best_seq = item[1].copy()

    # Multi-start plan.  This is intentionally biased toward longer sequences,
    # but also explores shorter ones because they can be optimized very quickly.
    plan = [
        (2048, 0, 45),
        (2048, 1, 45),
        (4096, 0, 80),
        (4096, 1, 80),
        (4096, 2, 80),
        (8192, 0, 260),
        (8192, 1, 260),
        (8192, 2, 260),
        (8192, 3, 200),
    ]

    if orig_arr is not None:
        on = orig_arr.size
        if 2048 <= on <= 16384:
            # Interpolate original into a safe length if it is not already near a power of two.
            on_handle = on
            if on > 8192:
                on_handle = 8192
            plan.append((on_handle, 4, 120))

    for n, variant, secs in plan:
        if time.monotonic() >= horizon - 45.0:
            break
        start = _make_start(n, variant, rng, orig_arr)
        limit = min(horizon, time.monotonic() + float(secs))
        tag = "n=%d v=%d" % (n, variant)
        loc_best, loc_M = _optimize(start, limit, rng, tag, pool, orig_arr)

        if np.isfinite(loc_M) and loc_M < best_fft:
            best_fft = loc_M
            best_seq = loc_best.copy()
            print("global FFT improved to %.6f from %s" % (best_fft, tag))

    # Refine the best sequence with remaining time.
    if best_seq is None:
        best_seq = _normalize(np.ones(256, dtype=np.float64), np.sqrt(2.0 * 256.0))

    refine_start = time.monotonic()
    refine_deadline = horizon - 12.0
    if refine_deadline > refine_start + 20.0:
        tag = "refine"
        loc_best, loc_M = _optimize(best_seq.copy(), refine_deadline, rng, tag, pool, orig_arr)
        if np.isfinite(loc_M) and loc_M < best_fft:
            best_fft = loc_M
            best_seq = loc_best.copy()

    # Exact polish: evaluate a small set of the best FFT-score candidates.
    candidates = []

    # Include the current best.
    if best_seq is not None:
        candidates.append(best_seq)

    # Include the original construction if available and valid.
    if orig_arr is not None and orig_arr.size > 0:
        candidates.append(orig_arr)

    # Include top pools.
    if len(pool) > 0:
        pool_sorted = sorted(pool, key=_fft_key)
        for item in pool_sorted:
            candidates.append(item[1])
            if len(candidates) >= 8:
                break

    # Trim to a small number of unique-ish candidates.
    deduped = []
    for seq in candidates:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        if arr.size == 0 or not np.all(np.isfinite(arr)):
            continue
        arr = np.clip(arr, 0.0, 1000.0)
        repeat = False
        for old in deduped:
            if old.size == arr.size:
                # Cheap fingerprint: first/last and a few sample values.
                if (
                    abs(arr[0] - old[0]) <= 1e-10
                    and abs(arr[-1] - old[-1]) <= 1e-10
                    and abs(arr[arr.size // 2] - old[old.size // 2]) <= 1e-10
                    and abs(arr[arr.size // 3] - old[old.size // 3]) <= 1e-10
                    and abs(arr[2 * arr.size // 3] - old[2 * old.size // 3]) <= 1e-10
                ):
                    repeat = True
                    break
            if repeat:
                break
        if not repeat:
            deduped.append(arr)
        if len(deduped) >= 8:
            break

    best_exact = np.inf
    best_exact_seq = None

    for arr in deduped:
        if arr.size == 0:
            continue
        try:
            sc = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_exact_seq = arr.copy()
            print("exact candidate score=%.6f" % best_exact)

    # Safe fallback if exact eval somehow fails.
    if best_exact_seq is None:
        if best_seq is not None:
            best_exact_seq = np.asarray(best_seq, dtype=np.float64).copy()
        else:
            best_exact_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_exact_seq]
```
