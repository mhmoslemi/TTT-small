def solve_lp_direction(a, M):
    """Solve the LP to find the best direction b.
    
    Maximize sum(b) subject to:
    - (a * b)_m <= M for all m
    - b >= 0
    """
    n = len(a)
    # Build the constraint matrix A where (A @ b)_m = (a * b)_m
    # (a * b)_m = sum_i a_i * b_{m-i} for valid i
    # So A[m][j] = a[m-j] if 0 <= m-j < n, else 0
    
    # Actually, let's be more careful:
    # (a * b)_m = sum_{i: 0<=i<n, 0<=m-i<n} a[i] * b[m-i]
    # Let j = m-i, then i = m-j, and we need 0 <= m-j < n, i.e., m-n+1 <= j <= m
    # Also 0 <= j < n, so max(0, m-n+1) <= j <= min(m, n-1)
    # (a * b)_m = sum_{j=max(0,m-n+1)}^{min(m,n-1)} a[m-j] * b[j]
    
    # So A[m, j] = a[m-j] if max(0, m-n+1) <= j <= min(m, n-1), else 0
    
    # This matrix is banded. Let's build it as a sparse matrix.
    # Row m has non-zero entries in columns max(0, m-n+1) to min(m, n-1).
    
    rows = []
    cols = []
    vals = []
    
    for m in range(2*n - 1):
        j_start = max(0, m - n + 1)
        j_end = min(m, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(m)
            cols.append(j)
            vals.append(a[m - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: maximize c^T b subject to A @ b <= M (upper bound), b >= 0
    # linprog minimizes, so: minimize -sum(b) subject to A @ b <= M, b >= 0
    
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
