        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
    else:
        rows = np.array([], dtype=int)
        cols = np.array([], dtype=int)

LP-based alternating minimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Uses sparse LP (CLARABEL) with bowl-shape

"""

def build_conv_matrix(n, f):
    """Build sparse matrix A where (A @ g)[k] = convolve(f, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            m = np.arange(lo, hi + 1)
            rows.append(np.full(len(m), k, dtype=np.int32))
            cols.append(m.astype(np.int32))
            vals.append(f[k - m])
    if rows:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
    else:
        rows = np.array([], dtype=np.int32)
        cols = np.array([], dtype=np.int32)
        vals = np.array([], dtype=np.float64)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_step(f, M):
    """Solve max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    g = cp.Variable(n, non_neg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue
    val = g.value
    if val is None:
        return None
    val = np.asarray(val, dtype=np.float64)
    val = np.clip(val, 0, None)
    if val.sum() < 1e-10:
        return None
    return val

    block *= np.exp(rng.normal(0.0, sigma, size=width))
    candidate[start:start + width] = block
    return np.clip(candidate, 0, None)

    return 2.0 * n * max_b / (s * s)
