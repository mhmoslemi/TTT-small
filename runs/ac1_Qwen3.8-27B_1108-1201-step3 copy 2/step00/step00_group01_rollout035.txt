    # Main optimization: mirror descent with adaptive controls
    lr = 1.5
    patience = 0
    max_patience = 150
    step_count = 0
    last_validate = time.monotonic()
    lr_schedule = 0.997

    while time.monotonic() < deadline - 60:
        conv = compute_convolution_fft(best)

        # Subgradient mirror descent step
        cand = subgradient_step(best, conv, lr, top_k=3)
        s = fast_score(cand)

        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
            patience = 0
            lr = max(0.4, lr * lr_schedule)
            if best_score < global_best_score:
                global_best = best.copy()
                global_best_score = best_score
                if step_count % 200 == 0:
                    print(f"  Step {step_count}: fast={best_score:.6f}")
        else:
            patience += 1
            lr = max(0.4, lr * 0.99)

        step_count += 1

        # Periodic validation with exact evaluator
        if time.monotonic() - last_validate > 8.0:
            vs = float(evaluate_sequence(best.tolist()))
            if np.isfinite(vs):
                if vs < global_best_score:
                    global_best = best.copy()
                    global_best_score = vs
                best_score = min(best_score, vs)
            last_validate = time.monotonic()

        # Escape strategies when stuck
        if patience >= max_patience:
            escaped = False
            # Strategy 1: Multiple random block perturbations
            for _ in range(20):
                cand = perturb_block(best, rng, (0.02, 0.4), (0.3, 1.5))
                if cand.sum() > 0:
                    cand = cand / cand.sum() * best.sum()
                s = fast_score(cand)
                if np.isfinite(s) and s < best_score:
                    best = cand
                    best_score = s
                    patience = 0
                    lr = 1.5
                    escaped = True
                    break

            # Strategy 2: Global perturbation
            if not escaped:
                for _ in range(5):
                    cand = perturb_global(best, rng, strength=rng.uniform(0.2, 0.8))
                    if cand.sum() > 0:
                        cand = cand / cand.sum() * best.sum()
                    s = fast_score(cand)
                    if np.isfinite(s) and s < best_score:
                        best = cand
                        best_score = s
                        patience = 0
                        lr = 1.5
                        escaped = True
                        break

            # Strategy 3: New structured initialization
            if not escaped:
                choices = [
                    chirp_sequence(n, rng.uniform(0.1, 0.6), rng.uniform(0, 2 * np.pi)),
                    sinusoid_sequence(n, rng.uniform(1, 15), rng.uniform(0.1, 0.5),
                                      rng.uniform(0, 2 * np.pi)),
                    chirp_sequence(n, rng.uniform(0.1, 0.4), 0.0) *
                    sinusoid_sequence(n, rng.uniform(1, 5), 0.3, rng.uniform(0, 2 * np.pi)),
                ]
                for cand in choices:
                    if cand.sum() > 0:
                        cand = cand / cand.sum() * best.sum()
                    s = fast_score(cand)
                    if np.isfinite(s) and s < best_score:
                        best = cand
                        best_score = s
                        patience = 0
                        lr = 1.5
                        escaped = True
                        print(f"  Escape: {s:.6f}")
                        break

            if not escaped:
                patience = 0
                lr = 1.5

    # Final local search polish
    while time.monotonic() < deadline - 10:
        best, best_score = local_search(best, rng, 50, fast_score)
        if best_score < global_best_score:
            global_best = best.copy()
            global_best_score = best_score
        best = global_best.copy()
        best_score = global_best_score

    # Final exact validation
    final = float(evaluate_sequence(global_best.tolist()))
    if np.isfinite(final) and final < global_best_score:
        global_best_score = final

    # If the fast score and exact differ, use the exact one
    print(f"Final global best: {global_best_score:.6f}")
    print(f"Final exact check: {final:.6f}")

    return [float(x) for x in global_best]
