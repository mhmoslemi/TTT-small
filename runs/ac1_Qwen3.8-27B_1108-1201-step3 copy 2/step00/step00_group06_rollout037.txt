    size = 2 * n

    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

    n = len(h)
    conv = fast_convolve(h, h)
    max_b = conv.max()
    return 2.0 * n * max_b  # since sum=1, (sum)^2 = 1

def lp_step(h, M, time_limit=20):
    """Solve the LP to find the best descent direction."""
    n = len(h)
    
    # Build convolution matrix A where A[k,i] = h[k-i] (0 if out of range)
    # Shape: (2n-1) x n
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(h[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n - 1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

                     options={'time_limit': time_limit, 'presolve': True})

        g = np.maximum(result.x, 0)
        return g
    return None

    n = len(h)
