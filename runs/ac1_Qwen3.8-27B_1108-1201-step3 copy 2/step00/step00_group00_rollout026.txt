from scipy.optimize import linprog
from scipy import sparse

def compute_convolution(a):
    """Compute linear convolution using FFT for speed."""
    n = len(a)
    # Use FFT-based convolution
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a, fsize)
    result = np.fft.irfft(A * A, fsize)[:size]
    return result

def solve_lp(a, M, active_indices):
    """
    Solve: max sum(g) s.t. (a*g)_k <= M for k in active, g >= 0

    n = len(a)
    m = len(active_indices)

    rows = []
    cols = []
    data = []

        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(idx)
                cols.append(i)
                data.append(a[j])

import time
from scipy.optimize import linprog
from scipy import sparse

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def build_convolution_matrix(a, active_indices):
    """
    Build sparse matrix A where A[k, i] = a[k-i] if 0 <= k-i < len(a).
    Rows correspond to active_indices.
    """
    n = len(a)
    m = len(active_indices)
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_indices):
        # For constraint k: sum_i A[k,i] * g_i = sum_i a[k-i] * g_i
        # A[k,i] = a[k-i] if 0 <= k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(idx)
            cols.append(i)
            data.append(a[k - i])
    return sparse.csc_matrix((data, (rows, cols)), shape=(m, n))

def solve_lp(a, M, active_indices):
    """Solve max sum(g) s.t. Ag <= M*1, g >= 0."""
    n = len(a)
    A = build_convolution_matrix(a, active_indices)
    m = len(active_indices)
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

max $\sum g_j$
s.t. $(a*g)_k \leq M$ for all $k$ in active set
$g \geq 0$

The constraint $(a*g)_k \leq M$ is:
$\sum_{j} a_j g_{k-j} \leq M$

This is a LINEAR constraint in $g$ (since $a$ is fixed). Good.

The objective $\sum g_j$ is also linear. So this is a standard LP.

The key question is: does the LP always have a feasible solution? Yes, $g = 0$ is always feasible. And the objective is bounded above (since $g \geq 0$ and the constraints limit $g$).

The optimal solution $g^*$ will have $\sum g^*_j > 0$ (since $g = 0$ gives objective 0, and there's always a positive feasible solution as long as $M > 0$).

Now, the direction $d = g^* - a$ (after rescaling) should point towards a sequence with lower peak convolution. But it might not always improve the objective (the LP only ensures that $a * g^* \leq M$, not that $g^* * g^*$ has a lower peak).

The line search handles this: we try different step sizes $t$ and pick the best one.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def build_sparse_constraint(a, active):
    n = len(a)
    m = len(active)
    rows = np.zeros(m * n, dtype=np.int32)  # overestimate
    cols = np.zeros(m * n, dtype=np.int32)
    data = np.zeros(m * n, dtype=np.float64)
    count = 0
    for idx, k in enumerate(active):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows[count] = idx
            cols[count] = i
            data[count] = a[k - i]
            count += 1
    return sparse.csc_matrix((data[:count], (rows[:count], cols[:count])), shape=(m, n))
