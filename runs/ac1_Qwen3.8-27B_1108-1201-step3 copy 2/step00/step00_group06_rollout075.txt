def build_B2K_set(n, K, max_size=500):
    """Greedily build a B_2[K] set in [0, n-1]."""
    B = []
    B_set = set()
    cnt = {}  # sum -> count of unordered pairs (b1,b2) with b1<=b2
    
    for step in range(max_size):
        best_b = -1
        best_new_max = float('inf')
        
        for b in range(n):
            if b in B_set:
                continue
            
            # Compute new multiplicity if we add b
            # New sums: b+b (diagonal, count 1) and b+b2 for b2 in B (count 1 each)
            new_max = 0
            
            # Check the new sums
            # Diagonal: b+b
            v = cnt.get(2*b, 0) + 1
            if v > K:
                continue  # Can't add b
            new_max = max(new_max, v)
            
            # Off-diagonal: b + b2 for b2 in B
            valid = True
            for b2 in B:
                s = b + b2
                v = cnt.get(s, 0) + 1
                if v > K:
                    valid = False
                    break
                new_max = max(new_max, v)
            
            if not valid:
                continue
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_b = b
        
        if best_b == -1:
            break  # Can't add any more elements
        
        B.append(best_b)
        B_set.add(best_b)
        cnt[2*best_b] = cnt.get(2*best_b, 0) + 1
        for b2 in B[:-1]:
            s = best_b + b2
            cnt[s] = cnt.get(s, 0) + 1
    
    return B, cnt

def compute_score_from_set(B, n):
    """Compute the actual score for a 0-1 sequence with 1s at positions in B."""
    m = len(B)
    if m == 0:
        return float('inf')
    # Build the convolution
    # c[k] = number of ordered pairs (b1,b2) in B×B with b1+b2=k
    # For efficiency, use the unordered pair counts
    # c[k] = 2 * (unordered pairs with sum k, b1<b2) + (1 if k=2b for some b, else 0)
    cnt_ordered = {}
    for b1 in B:
        for b2 in B:
            s = b1 + b2
            cnt_ordered[s] = cnt_ordered.get(s, 0) + 1
    M = max(cnt_ordered.values())
    return 2.0 * n * M / (m * m)
