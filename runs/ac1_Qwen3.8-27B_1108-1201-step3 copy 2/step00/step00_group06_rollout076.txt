def project_simplex(v):
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(p, alpha):
    n = len(p)
    conv = np.convolve(p, p)
    max_c = conv.max()
    shifted = alpha * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    value = max_c + np.log(Z) / alpha
    w = exp_vals / Z
    
    # Gradient computation
    conv_wp = np.convolve(w, p)  # length 3n-2
    conv_wrev_p = np.convolve(w[::-1], p)  # length 3n-2
    grad = np.zeros(n)
    grad += conv_wp[:n]
    grad += conv_wrev_p[n-1:2*n-1][::-1]
    
    return value, grad

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
Multi-phase optimization for minimizing the peak auto-convolution of a
non-negative sequence. Uses smoothed projected gradient descent with
momentum (log-sum-exp approximation of max), multiple random restarts
with diverse initializations, and optional LP-based polishing.
"""


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_and_grad(p, alpha):
    """Compute smoothed objective and gradient. Objective approximates max(conv(p,p))."""
    n = len(p)
    conv = np.convolve(p, p)
    max_c = conv.max()
    shifted = alpha * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    value = max_c + np.log(Z) / alpha
    w = exp_vals / Z
    conv_wp = np.convolve(w, p)
    conv_wrev_p = np.convolve(w[::-1], p)
    grad = np.zeros(n)
    grad += conv_wp[:n]
    grad += conv_wrev_p[n - 1:2 * n - 1][::-1]
    return value, grad


def pgd_optimize(p, n_iters, alpha_start, alpha_end, lr_scale, momentum, deadline):
    """Projected gradient descent with momentum and alpha annealing."""
    p = p.copy()
    s = p.sum()
    if s < 1e-10:
        p = np.ones_like(p) / len(p)
    else:
        p = p / s
    n = len(p)
    v = np.zeros(n)
    best_p = p.copy()
    best_val = smooth_obj_and_grad(p, min(alpha_end, 200.0))[0]
    patience = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_scale / (1.0 + 0.03 * it)

        value, grad = smooth_obj_and_grad(p, alpha)
        v = momentum * v + (1.0 - momentum) * grad
        p_new = p - lr * v
        p_new = project_simplex(p_new)

        check_val = smooth_obj_and_grad(p_new, min(alpha, 200.0))[0]
        if check_val < best_val:
            best_val = check_val
            best_p = p_new.copy()
            patience = 0
        else:
            patience += 1

        p = p_new
        if patience > 80:
            break

    return best_p, best_val


def mirror_descent_optimize(p, n_iters, alpha_start, alpha_end, lr_scale, deadline):
    """Mirror descent (exponential weights) with alpha annealing."""
    p = p.copy()
    s = p.sum()
    if s < 1e-10:
        p = np.ones_like(p) / len(p)
    else:
        p = p / s
    n = len(p)
    best_p = p.copy()
    best_val = smooth_obj_and_grad(p, min(alpha_end, 200.0))[0]
    patience = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_scale / (1.0 + 0.05 * it)

        value, grad = smooth_obj_and_grad(p, alpha)
        p_new = p * np.exp(-lr * grad)
        p_new = project_simplex(p_new)

        check_val = smooth_obj_and_grad(p_new, min(alpha, 200.0))[0]
        if check_val < best_val:
            best_val = check_val
            best_p = p_new.copy()
            patience = 0
        else:
            patience += 1

        p = p_new
        if patience > 60:
            break

    return best_p, best_val


def lp_polish(p, max_iters=8, deadline=None):
    """LP-based polish: iteratively find spread-out direction via linear program."""
    n = len(p)
    if n > 1200:
        return p
    p = p.copy()
    s = p.sum()
    if s < 1e-10:
        return p
    p = p / s
    best_p = p.copy()
    best_peak = np.max(np.convolve(p, p))

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = np.convolve(p, p)
        M = conv.max()

        c_obj = -np.ones(n)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                for j in range(j_lo, j_hi + 1):
                    rows.append(k)
                    cols.append(j)
                    vals.append(p[k - j])

        if len(rows) == 0:
            break
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs', maxiter=2000)
        except Exception:
            break
        if res.status != 0:
            break

        q = res.x
        S = q.sum()
        if S < 1e-10:
            break
        q_norm = q / S

        improved = False
        for t in [0.2, 0.4, 0.6, 0.8]:
            p_try = (1.0 - t) * p + t * q_norm
            p_try = project_simplex(p_try)
            peak = np.max(np.convolve(p_try, p_try))
            if peak < best_peak:
                best_peak = peak
                best_p = p_try.copy()
                p = p_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_p


def make_initialization(init_type, n, rng):
    """Create an asymmetric non-negative initial sequence of length n."""
    if init_type == 'exp':
        rate = 1.0 + 2.0 * rng.random()
        p = np.exp(-np.linspace(0, rate, n))
    elif init_type == 'twoblock':
        ratio = 0.2 + 0.6 * rng.random()
        amp = 1.0 + 2.0 * rng.random()
        p = np.ones(n)
        p[:int(n * ratio)] *= amp
    elif init_type == 'random':
        p = rng.exponential(1.0, size=n)
    elif init_type == 'power':
        gamma = 0.3 + 0.7 * rng.random()
        p = (np.arange(n) + 1.0) ** (-gamma)
    elif init_type == 'bell':
        center = 0.2 + 0.6 * rng.random()
        width = 0.1 + 0.4 * rng.random()
        x = np.linspace(0, 1, n)
        p = np.exp(-((x - center) / width) ** 2)
    elif init_type == 'ramp':
        slope = 0.5 + 2.0 * rng.random()
        p = 1.0 + slope * np.linspace(0, 1, n)
    elif init_type == 'bimodal':
        c1 = 0.2 + 0.2 * rng.random()
        c2 = 0.6 + 0.2 * rng.random()
        w1 = 0.1 + 0.2 * rng.random()
        w2 = 0.1 + 0.2 * rng.random()
        x = np.linspace(0, 1, n)
        p = np.exp(-((x - c1) / w1) ** 2) + np.exp(-((x - c2) / w2) ** 2)
    else:
        p = np.ones(n)
    return np.maximum(p, 0.0)
