```python
Anti-symmetric projected gradient descent to minimize the peak of the
autocorrelation of a non-negative step function.

Key insight: For non-negative sequences, anti-symmetric perturbations
(a[i] + a[n-1-i] = constant) reduce the central peak of the convolution
while keeping the sum constant. We optimize in this subspace using
projected gradient descent on the active lag.

import numpy as np

def _compute_conv(a, Nfft):
    """Compute linear convolution a*a via FFT."""
    N = len(a)
    conv = np.fft.irfft(np.fft.rfft(a, Nfft) ** 2, Nfft)[:2*N-1]
    return conv

def _make_antisymmetric(v, n):
    """Enforce anti-symmetry: v[n-1-i] = -v[i]."""
    return 0.5 * (v - v[::-1])
```

Looking at this problem, I need to minimize $2n \cdot \max(f*f) / (\sum a_j)^2$ for non-negative sequences. The all-ones sequence gives exactly 2.0. 

**Key insight:** For non-negative sequences, anti-symmetric perturbations (where $a[i] + a[n-1-i] = 2$) reduce the central peak of the convolution since $\text{conv}[n-1] = n - \sum v[i]^2$. I'll optimize in this anti-symmetric subspace using projected gradient descent on the active lag, with multiple random initializations and frequency-based starting points.

```python
"""
Anti-symmetric projected gradient descent to minimize 2n*max(f*f)/(sum f)^2.
Uses the insight that anti-symmetric perturbations (a[i]+a[n-1-i]=2) reduce
the central autocorrelation peak. Multiple restarts with diverse initializations.
"""
import numpy as np
import time


def _fft_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    N = len(a)
    Nfft = 1 << (2 * N - 1).bit_length()
    A = np.fft.rfft(a, Nfft)
    return np.fft.irfft(A * A, Nfft)[:2 * N - 1]


def _antisym_proj(v):
    """Project vector to anti-symmetric subspace: v[n-1-i] = -v[i]."""
    return 0.5 * (v - v[::-1])


def _project_v(v):
    """Project to anti-symmetric subspace with v >= -1 constraint."""
    v = _antisym_proj(v)
    v = np.clip(v, -1.0, 1.0)
    v = _antisym_proj(v)
    v = np.clip(v, -1.0, 1.0)
    return v


def _compute_gradient(a, k_star, n):
    """Compute gradient of conv[k_star] w.r.t. a, projected to anti-sym subspace."""
    g = np.zeros(n)
    ks = k_star
    # g[j] = 2*a[ks-j] - 2*a[ks-n+1+j] (a[m]=0 for m outside [0,n-1])
    j1lo = max(0, ks - (n - 1))
    j1hi = min(n - 1, ks)
    if j1lo <= j1hi:
        g[j1lo:j1hi + 1] += 2.0 * a[ks - j1hi:ks - j1lo - 1:-1]
    j2lo = max(0, n - 1 - ks)
    j2hi = min(n - 1, 2 * n - 2 - ks)
    if j2lo <= j2hi:
        g[j2lo:j2hi + 1] -= 2.0 * a[ks - n + 1 + j2lo:ks - n + 1 + j2hi + 1]
    return _antisym_proj(g)


def _optimize_from_start(a, n, Nfft, rng, deadline, best_holder):
    """Run projected gradient descent from initial a. Returns best found."""
    best_local_score = np.inf
    best_local_a = a.copy()
    lr = 1.0 / n
    patience = 0
    best_local_max = np.inf
    
    for step in range(50000):
        if time.monotonic() > deadline - 5.0:
            break
        
        conv = np.fft.irfft(np.fft.rfft(a, Nfft) ** 2, Nfft)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        current_max = float(conv[k_star])
        s = float(np.sum(a))
        score = 2.0 * n * current_max / (s * s)
        
        if score < best_local_score:
            best_local_score = score
            best_local_a = a.copy()
            best_local_max = current_max
            patience = 0
            if score < best_holder[0]:
                best_holder[0] = score
                best_holder[1] = a.copy()
        else:
            patience += 1
        
        if patience > 3000 and step > 100:
            # Perturbative restart to escape local minimum
            v = a - 1.0
            scale = 0.05 + 0.1 * (patience / 5000.0)
            pert = rng.standard_normal(n) * scale
            v = _project_v(v + pert)
            a = 1.0 + v
            patience = 0
            continue
        
        # Cubic interpolation restart for smoother escape
        if patience > 500 and step > 50 and step % 100 == 0:
            v = a - 1.0
            # Try a frequency-based nudge
            idx = np.arange(n)
            k_freq = rng.integers(1, max(2, n // 8))
            nudge = 0.02 * np.sin(2 * np.pi * k_freq * (idx + 0.5) / n)
            v = _project_v(v + nudge)
            a = 1.0 + v
            continue
        
        g = _compute_gradient(a, k_star, n)
        
        # Backtracking line search
        v = a - 1.0
        for _ in range(10):
            v_new = _project_v(v - lr * g)
            a_trial = 1.0 + v_new
            conv_trial = np.fft.irfft(np.fft.rfft(a_trial, Nfft) ** 2, Nfft)[:2 * n - 1]
            max_trial = float(np.max(conv_trial))
            if max_trial < current_max * (1 - 1e-8):
                a = a_trial
                v = v_new
                lr = min(lr * 1.05, 10.0 / n)
                break
            else:
                lr = max(lr * 0.5, 0.01 / n)
        else:
            # Couldn't make progress, try smaller step
            v_new = _project_v(v - 0.001 * g)
            a = 1.0 + v_new
    
    return best_local_score, best_local_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    
    Algorithm: Anti-symmetric parameterization (a[i]+a[n-1-i]=2) with
    projected gradient descent on the active lag. Multiple diverse
    initializations and sequence lengths.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)
    
    best_holder = [np.inf, None]
    
    # First, get a quick baseline from the provided sequence
    if 'height_sequence_1' in dir() or True:
        try:
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
                sc = float(evaluate_sequence(parent.tolist()))
                print(f"Parent sequence: n={len(parent)}, score={sc:.6f}")
                if sc < best_holder[0]:
                    best_holder[0] = sc
                    best_holder[1] = parent.copy()
        except Exception as e:
            print(f"Could not load parent: {e}")
    
    # All-ones baseline
    for n in [1024, 4096]:
        a = np.ones(n)
        conv = _fft_conv(a)
        sc = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
        print(f"All-ones n={n}: score={sc:.6f}")
        
    # Main optimization loop
    sequence_lengths = [2048, 4096, 8192]
    
    for n in sequence_lengths:
        if time.monotonic() > deadline - 60:
            break
        print(f"\n=== Optimizing n={n} ===")
        Nfft = 1 << (2 * n - 1).bit_length()
        idx = np.arange(n)
        
        # Generate diverse initializations
        initializations = []
        
        # Fundamental sine
        for amp in [0.3, 0.5, 0.7, 0.9]:
            v = amp * np.sin(2 * np.pi * (idx + 0.5) / n)
            v = _project_v(v)
            initializations.append(1.0 + v)
        
        # Two harmonics
        for amp1, amp2 in [(0.5, 0.3), (0.4, 0.5), (0.6, 0.4)]:
            v = (amp1 * np.sin(2 * np.pi * (idx + 0.5) / n) +
                 amp2 * np.sin(4 * np.pi * (idx + 0.5) / n))
            v = _project_v(v)
            initializations.append(1.0 + v)
        
        # Random smooth (sum of random sines)
        for _ in range(4):
            v = np.zeros(n)
            num_freqs = rng.integers(3, 12)
            for _ in range(num_freqs):
                f = rng.uniform(1, n // 4)
                p = rng.uniform(0, 2 * np.pi)
                a_ = rng.uniform(0.2, 0.6) / num_freqs
                v += a_ * np.sin(2 * np.pi * f * (idx + 0.5) / n + p)
            v = _project_v(np.clip(v, -1, 1))
            initializations.append(1.0 + v)
        
        # Binary-like
        for amp in [0.6, 0.8, 0.95]:
            v = amp * rng.choice(np.array([-1.0, 1.0]), size=n)
            v = _project_v(v)
            initializations.append(1.0 + v)
        
        # Ramp
        v = 0.7 * (2.0 * idx / (n - 1) - 1.0)
        v = _project_v(v)
        initializations.append(1.0 + v)
        
        # Cosine
        v = 0.8 * np.cos(2 * np.pi * (idx + 0.5) / n)
        v = _project_v(v)
        initializations.append(1.0 + v)
        
        # Sparse
        v = np.zeros(n)
        positions = rng.choice(n, size=n // 8, replace=False)
        v[positions] = rng.uniform(-1, 1, size=len(positions))
        v = _project_v(v)
        initializations.append(1.0 + v)
        
        # Pure random small
        v = 0.3 * rng.standard_normal(n)
        v = _project_v(v)
        initializations.append(1.0 + v)
        
        # Track per-init best
        init_best_score = np.inf
        init_best_a = None
        
        for init_idx, a0 in enumerate(initializations):
            if time.monotonic() > deadline - 30:
                break
            
            # Quick evaluation of initial point
            conv0 = _fft_conv(a0)
            max0 = float(np.max(conv0))
            s0 = float(np.sum(a0))
            sc0 = 2.0 * n * max0 / (s0 * s0)
            
            if sc0 < best_holder[0]:
                best_holder[0] = sc0
                best_holder[1] = a0.copy()
            if sc0 < init_best_score:
                init_best_score = sc0
                init_best_a = a0.copy()
            
            print(f"  n={n}, init {init_idx:2d}: initial score={sc0:.6f}")
            
            # Run optimization
            local_score, local_a = _optimize_from_start(
                a0.copy(), n, Nfft, rng, deadline, best_holder
            )
            
            if local_score < init_best_score:
                init_best_score = local_score
                init_best_a = local_a
            if local_score < best_holder[0]:
                best_holder[0] = local_score
                best_holder[1] = local_a.copy()
        
        print(f"  n={n} best: {best_holder[0]:.6f}")
    
    # Final: try to improve best found using its structure
    if best_holder[1] is not None:
        a_best = best_holder[1].copy()
        n = len(a_best)
        Nfft = 1 << (2 * n - 1).bit_length()
        
        # Refine with more careful gradient steps using softmax objective
        tau = 30.0
        lr = 0.5 / n
        
        for step in range(5000):
            if time.monotonic() > deadline - 2.0:
                break
            
            conv = np.fft.irfft(np.fft.rfft(a_best, Nfft) ** 2, Nfft)[:2 * n - 1]
            # Smooth max via softmax
            max_c = np.max(conv)
            shifted = tau * (conv - max_c)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            
            # Weighted gradient
            g = np.zeros(n)
            for k in np.where(weights > 1e-8)[0]:
                wk = weights[k]
                gk = _compute_gradient(a_best, int(k), n)
                g += wk * gk
            
            v = a_best - 1.0
            for _ in range(5):
                v_new = _project_v(v - lr * g)
                a_trial = 1.0 + v_new
                conv_trial = np.fft.irfft(np.fft.rfft(a_trial, Nfft) ** 2, Nfft)[:2 * n - 1]
                max_trial = float(np.max(conv_trial))
                current_max = float(max_c)
                if max_trial < current_max:
                    a_best = a_trial
                    v = v_new
                    lr = min(lr * 1.02, 5.0 / n)
                    s = float(np.sum(a_best))
                    sc = 2.0 * n * max_trial / (s * s)
                    if sc < best_holder[0]:
                        best_holder[0] = sc
                        best_holder[1] = a_best.copy()
                    break
                else:
                    lr = max(lr * 0.5, 0.01 / n)
    
    if best_holder[1] is None:
        best_seq = np.ones(2048)
    else:
        best_seq = best_holder[1]
    
    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
    
    return [float(x) for x in best_seq]
```
