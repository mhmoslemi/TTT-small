"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using projected gradient descent
on the non-negative simplex. Key insight: non-palindromic sequences can achieve
max(convolve(a,a)) lower than the palindromic Cauchy-Schwarz bound, breaking
below 2.0. We explore multiple lengths and non-uniform initializations.

import numpy as np
import time


def compute_gradient_at_k(a, k_star, n):
    """Compute subgradient of convolve(a,a)[k_star] w.r.t. a.
    d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n, else 0."""
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad


def run_pgd(a_init, n, deadline, rng, lr=0.05, decay=0.9999, max_patience=5000):
    """Run projected gradient descent minimizing max(convolve(a,a)) s.t. sum(a)=1, a>=0."""
    a = np.maximum(a_init.copy(), 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    c = np.convolve(a, a)
    best_a = a.copy()
    best_max_c = float(np.max(c))

    patience = 0
    n_iter = 0

    while time.monotonic() < deadline and patience < max_patience:
        n_iter += 1
        if n_iter % 5000 == 0:
            print(f"      iter={n_iter}, max_c={best_max_c:.8f}")

        c = np.convolve(a, a)
        max_c = float(np.max(c))

        if max_c < best_max_c - 1e-12:
            best_max_c = max_c
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Pick a random argmax for the subgradient
        max_indices = np.where(c >= max_c - 1e-12)[0]
        k_star = int(max_indices[rng.integers(len(max_indices))])

        grad = compute_gradient_at_k(a, k_star, n)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s

        lr *= decay

    return best_a


def make_inits(n, rng):
    """Generate diverse non-constant initializations of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    # Linearly decreasing
    inits.append(np.linspace(1.0, 0.01, n))
    # Linearly increasing
    inits.append(np.linspace(0.01, 1.0, n))
    # Off-center Gaussian
    inits.append(np.exp(-0.5 * ((x - n * 0.3) / (n * 0.15)) ** 2))
    # Off-center Gaussian (other side)
    inits.append(np.exp(-0.5 * ((x - n * 0.7) / (n * 0.15)) ** 2))
    # Very narrow peak
    inits.append(np.exp(-0.5 * ((x - n * 0.5) / (n * 0.05)) ** 2))
    # Half-ones
    h = np.zeros(n)
    h[:n // 2] = 1.0
    inits.append(h)
    # Bimodal
    inits.append(np.exp(-0.5 * ((x - n * 0.2) / (n * 0.08)) ** 2) +
                 np.exp(-0.5 * ((x - n * 0.8) / (n * 0.08)) ** 2))
    # Random
    inits.append(np.maximum(rng.random(n), 0.01))
    # Structured random (sparse high values)
    a_sparse = np.full(n, 0.05)
    idx = rng.integers(0, n, size=max(1, n // 4))
    a_sparse[idx] = 1.0
    inits.append(a_sparse)
    # Sine wave (shifted to be positive)
    inits.append(0.5 + 0.5 * np.sin(2 * np.pi * x / n))
    return inits
