```python
import numpy as np
import time

def self_convolve(f):
    """Compute linear self-convolution using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
    return conv

def conv_gradient(f, peak_indices):
    """Compute subgradient of max convolution value."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_indices:
        # d(f*f)_k / d(f_j) = f_j + f[k-j] (if 0 <= k-j < n)
        g = np.zeros(n)
        g[:] = f  # f_j term
        # Add f[k-j] for valid j
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            g[j] += f[k - j]
        grad += g
    grad /= len(peak_indices)
    return grad
```

The "linearization" at point $f$ gives the LP constraints $(f*g)_k \le M$ (treating $f$ as fixed in the bilinear term).

```python
import numpy as np
import time

def self_convolve(f):
    n = len(f)
    size = 2*n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F*F, fft_size)[:size]

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    
    # Start from given sequence
    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    n = len(f)
    
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    
    # Normalize
    S = np.sum(f)
    f = f / S
    
    # Phase 1: LP-based iteration
    for it in range(500):
        if time.monotonic() > deadline - 30:
            break
        
        conv = self_convolve(f)
        M = np.max(conv)
        current_score = 2 * n * M  # since S=1
        
        if current_score < best_score:
            best_score = current_score
            best_seq = f.copy()
        
        # Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0
        # Build constraint matrix C: C[k,j] = f[k-j] if 0<=k-j<n else 0
        # This is a (2n-1) x n matrix
        
        # Use scipy linprog
        from scipy.optimize import linprog
        
        # c = -ones (minimize -sum(g) = maximize sum(g))
        c = -np.ones(n)
        
        # A_ub: (2n-1) x n, A_ub @ g <= M * ones(2n-1)
        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A_ub[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
        
        b_ub = np.full(2*n-1, M)
        
        # Bounds: g >= 0
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if result.success:
                g = result.x
                S_g = np.sum(g)
                if S_g > 0:
                    g = g / S_g  # normalize
                    # Blend
                    t = 0.1
                    new_f = (1-t)*f + t*g
                    new_f = np.maximum(0, new_f)
                    new_S = np.sum(new_f)
                    if new_S > 0:
                        new_f = new_f / new_S
                        new_conv = self_convolve(new_f)
                        new_M = np.max(new_conv)
                        new_score = 2 * n * new_M
                        if new_score < current_score:
                            f = new_f
                        else:
                            f = g  # just move to g
                    else:
                        f = g
            else:
                break
        except Exception:
            break
        
        print(f"Iter {it}: score={current_score:.6f}, best={best_score:.6f}, time={time.monotonic():.0f}")
    
    return [float(x) for x in best_seq]
```
