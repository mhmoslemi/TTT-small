Multi-stage optimization for minimizing 2*n*max(a*a)/(sum(a))^2.
Uses subgradient descent on the peak of the self-convolution,
with adaptive step size and structured initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute objective value directly."""
    n = a.size
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def subgradient_descent_step(a, alpha, rng):
    """One step of subgradient descent targeting the convolution peak."""
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    
    # Find near-peak positions
    threshold = 0.85 * M
    peak_mask = c >= threshold
    peak_indices = np.where(peak_mask)[0]
    
    # Compute combined subgradient
    # g[i] = 2 * sum_{k in peaks} a[k-i] (where valid)
    g = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # a[k-i] for i in [lo, hi] means a[j] for j in [k-hi, k-lo] = [lo, hi]... 
        # Actually: for i in [lo, hi], k-i is in [k-hi, k-lo] = [k-min(k,n-1), k-max(0,k-n+1)]
        # If k < n: i in [0, k], k-i in [0, k], so g[i] += a[k-i] for i in [0,k]
        # If k >= n: i in [k-n+1, n-1], k-i in [1, n-1-(k-n+1)] = [1, 2n-2-k]
        
        # Simpler: for each valid i, g[i] += a[k-i]
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # Clamped to [0, n-1]
        
        # Vectorized: g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        # k-hi to k-lo has hi-lo+1 elements
        # a[k-hi], a[k-hi+1], ..., a[k-lo]
        # Reversed: a[k-lo], a[k-lo-1], ..., a[k-hi]
        # g[lo] += a[k-lo], g[lo+1] += a[k-lo-1], ..., g[hi] += a[k-hi]
        # So g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        
        seg = a[k-hi:k-lo+1][::-1]
        g[lo:hi+1] += seg
    
    g *= 2.0

    g_mean = g.mean()
    g -= g_mean
    
    # Scale the gradient
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return a.copy()
    g = g / g_norm
    
    # Determine step size
    # We want a - alpha * g to stay non-negative
    # alpha_max = min(a[i] / g[i] for i where g[i] > 0)
    positive_g = g > 0
    if positive_g.any():

    else:
        alpha_max = 1.0
    
    # Try a fraction of alpha_max
    step = alpha * min(alpha_max * 0.9, 1.0)

    a_new = np.maximum(a_new, 0)
    S_new = a_new.sum()
    if S_new < 1e-10:
        return a.copy()
    a_new = a_new / S_new  # Re-normalize to sum = 1
    
    return a_new
