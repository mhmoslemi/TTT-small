    else:

    g = _conv(w, a[::-1])
    return 2.0 * g[n - 1:2 * n - 1]


def _md_up(a, g, lr):
    S = a.sum()
    a2 = a * np.exp(-lr * g)
    s = a2.sum()
    if s < 1e-30:
        return np.full(len(a), S / len(a))
    return a2 / s * S


def _pg_up(a, g, lr):
    return _proj(a - lr * g, a.sum())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    # Build diverse asymmetric initializations
    x = np.arange(n, dtype=np.float64)
    xm = x / max(n - 1, 1)
    inits = [
        parent * np.exp(rng.normal(0, 0.5, n)),
        parent * np.exp(rng.normal(0, 1.0, n)),
        (x + 1.0) / n,
        np.exp(-2.0 * (xm - 0.3) ** 2) + 0.1,
        np.exp(-2.0 * (xm - 0.7) ** 2) + 0.1,
        1.0 + 0.5 * np.sin(2.0 * np.pi * xm),
        1.0 + 0.5 * np.sin(2.0 * np.pi * xm + 0.7),
        1.0 + 0.4 * np.sin(4.0 * np.pi * xm) + 0.2 * np.sin(6.0 * np.pi * xm),
        rng.random(n) + 0.1,
        np.exp(-x / (0.25 * n)) + 0.05,
        np.exp(-x / (0.15 * n)) + 0.05,
        np.exp(-x / (0.4 * n)) + 0.1,
        1.0 + 0.3 * np.cos(2.0 * np.pi * xm + 0.5),
    ]

    best_a = None
    best_obj = np.inf
    for c in inits:
        c = np.clip(c, 0.0, 1000.0)
        o = _obj(c)
        if o < best_obj:
            best_obj = o
            best_a = c.copy()

    print(f"[init] best={best_obj:.8f} n={n}", flush=True)

    a = best_a.copy()

    # Main optimization loop
    it = 0
    no_impr = 0
    base_lr = 0.6
    use_md = True

    while time.monotonic() < deadline:
        it += 1
        T = min(3.0 + 0.008 * it, 400.0)
        lr = base_lr / np.sqrt(1.0 + it)

        g = _grad(a, T)

        if use_md:
            a_new = _md_up(a, g, lr)
        else:
            a_new = _pg_up(a, g, lr)

        o = _obj(a_new)
        if o < best_obj:
            best_obj = o
            best_a = a_new.copy()
            a = a_new
            no_impr = 0
        else:
            no_impr += 1

        if it % 30000 == 0:
            print(f"  it={it} T={T:.1f} obj={best_obj:.8f} impr={no_impr}", flush=True)

        # Switch method periodically
        if it % 100000 == 0:
            use_md = not use_md

        # Restart with block perturbation
        if no_impr > 60000:
            a = best_a.copy()
            nb = int(rng.integers(5, 40))
            edges = np.sort(rng.integers(0, n + 1, nb + 1))
            for i in range(nb):
                s0, s1 = int(edges[i]), int(edges[i + 1])
                if s1 > s0:
                    a[s0:s1] *= float(np.exp(rng.normal(0.0, 0.6)))
            a = np.clip(a, 0.0, 1000.0)
            sa = a.sum()
            if sa > 1e-10:
                a = a / sa * best_a.sum()
            no_impr = 0
            it = 0
            print(f"  restart obj={best_obj:.8f}", flush=True)

        # Occasionally try a fresh random init
        if no_impr == 30000:
            trial = np.clip(rng.random(n) * 5.0 + 0.01, 0, 1000)
            to = _obj(trial)
            if to < best_obj:
                best_obj = to
                best_a = trial.copy()
                a = trial.copy()
                it = 0
                no_impr = 0
                print(f"  fresh random init obj={best_obj:.8f}", flush=True)

    # Final polish
    a = best_a.copy()
    polished = 0
    for p in range(10000):
        if time.monotonic() > deadline:
            break
        g = _grad(a, 800.0)
        a_new = _md_up(a, g, 0.08 / np.sqrt(1 + p))
        o = _obj(a_new)
        if o < best_obj:
            best_obj = o
            best_a = a_new.copy()
            a = a_new
            polished += 1

    print(f"[polish] improved {polished} times", flush=True)

    # Second polish with even smaller steps
    a = best_a.copy()
    for p in range(5000):
        if time.monotonic() > deadline:
            break
        g = _grad(a, 2000.0)
        a_new = _md_up(a, g, 0.02 / np.sqrt(1 + p))
        o = _obj(a_new)
        if o < best_obj:
            best_obj = o
            best_a = a_new.copy()
            a = a_new

    final = evaluate_sequence([float(v) for v in best_a])
    print(f"[final] evaluator={final:.8f} local={best_obj:.8f}", flush=True)
    return [float(v) for v in best_a]
