def fast_autocorr(a):
    """Compute full linear autocorrelation a*a using FFT. Matches np.convolve(a,a,'full')."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(a, size)
    conv = np.fft.irfft(F * F, size)[:2 * n - 1]
    return conv


def compute_J(a):
    """Compute J(a) = 2n * max(a*a) / (sum a)^2 using FFT."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_autocorr(a)
    max_val = np.max(conv)
    return 2.0 * n * max_val / (S * S)


def compute_grad(a):
    """Compute a subgradient of J at a. Returns (J, grad) or (inf, None)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, None
    conv = fast_autocorr(a)
    max_val = np.max(conv)

    # Find all (near-)maximizing indices
    tol = 1e-9 * max_val + 1e-15
    tied = np.where(np.abs(conv - max_val) < tol)[0]

    # dM/da_i = 2*a_{k-i} for valid k, i
    grad_M = np.zeros(n)
    for k in tied:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            grad_M[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad_M /= len(tied)

    # dJ/da_i = (2n/S^2) * dM/da_i - 4n*max_val/S^3
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * max_val) / (S * S * S)
    J = 2.0 * n * max_val / (S * S)
    return J, grad

    return a / S

    """
    Partial coordinate descent: optimize n_coords random coordinates.
