import numpy as np

from scipy.fft import fft, ifft, next_fast_len


def _fast_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = fft(a, size)
    return ifft(A * A).real[:2 * n - 1]


def _fast_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 using FFT (fast, no verification)."""
    n = len(a)
    c = _fast_convolve(a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def _initialize(n, init_type, rng):

    eta_max = 5.0 / n

    it = 0
