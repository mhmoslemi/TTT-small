6. Renormalize

import numpy as np
import time

def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()

def _soft_gradient(x, T):
    n = len(x)
    c = np.convolve(x, x)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    soft_max = c_max + T * np.log(sum_exp)
    S = x.sum()
    # grad_corr[i] = sum_j p[i+j] * x[j]
    L = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, L)
    X = np.fft.rfft(x, L)
    grad_corr = np.fft.irfft(P * np.conj(X), L)[:n]
    grad = 2.0 * grad_corr - 2.0 * soft_max / S
    return grad, c, soft_max

import time

def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

    """Returns (gradient, soft_max_value, c) for the soft objective."""
    n = len(x)
    c = np.convolve(x, x)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    soft_max = c_max + T * np.log(sum_exp)
    S = x.sum()
    L = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, L)
    X = np.fft.rfft(x, L)
    grad_corr = np.fft.irfft(P * np.conj(X), L)[:n]
    grad = 2.0 * (grad_corr - soft_max / S)
    return grad, soft_max, c

```python
def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
