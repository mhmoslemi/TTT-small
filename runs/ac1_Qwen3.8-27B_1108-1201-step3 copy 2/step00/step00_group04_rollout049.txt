def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)

def logsumexp_gradient_step(a, alpha, eta):
    """One gradient step using log-sum-exp smoothing."""
    n = len(a)
    # Compute convolution
    c = np.convolve(a, a)
    
    # Compute weights using log-sum-exp (for numerical stability)
    max_c = np.max(c)
    w = np.exp(alpha * (c - max_c))
    w_sum = np.sum(w)
    if w_sum == 0:
        return a, np.inf
    w = w / w_sum
    
    # Compute gradient: g[i] = 2 * sum_k a[k-i] * w[k]
    # Using convolve formulation: g[i] = 2 * convolve(a[::-1], w, mode='full')[n-1+i]
    # But this is O(n^2) for each i. Use FFT instead.
    
    # FFT approach:
    N = 1
    while N < 3 * n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # g[i] = 2 * sum_k a[k-i] * w[k]
    # = 2 * IFFT(conj(FFT(w)) * FFT(a))[i] (circular correlation)
    # Wait, I need to be more careful with the indexing.
    # 
    # Let me use the convolve formula: g[i] = 2 * conv(a[::-1], w, 'full')[n-1+i]
    # In FFT: conv(a[::-1], w) = IFFT(FFT(a[::-1]) * FFT(w))
    fa_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(fa_rev * fw, N)
    g = 2.0 * conv[n-1 : 2*n-1]
    
    # Gradient step with non-negativity constraint
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize

    if s > 0:

    return a_new, obj
