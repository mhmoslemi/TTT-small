import numpy as np
import time

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    # Compute cumsum
    cssv = np.cumsum(u)
    # Find rho
    cond = u - (cssv - 1) / np.arange(1, n + 1) > 0
    rho = np.max(np.where(cond)[0]) + 1
    tau = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - tau, 0)

def compute_convolution(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def score_sequence(a):
    """Compute the score without calling evaluate_sequence (for speed)."""
    n = len(a)
    c = compute_convolution(a)

```python
"""
Minimizes 2n*max(a*a)/(sum(a))^2 via mirror descent with smoothed max,
multiple restarts, and adaptive parameters.
"""

import numpy as np
import time

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / rho) > 0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    idx = np.where(cond)[0]
    max_rho = int(idx[-1]) + 1
    tau = (cssv[max_rho - 1] - 1.0) / max_rho
    return np.maximum(v - tau, 0.0)

def _fft_convolve(a):
    """Self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _compute_score(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smoothed_gradient(a, c, temp):
    """Compute gradient of temp*log(sum(exp(c/temp)))."""
    n = len(a)
    conv_len = c.shape[0]
    c_shift = c - c.max()
    e = np.exp(c_shift / temp)
    total = e.sum()
    if total < 1e-300:
        w = np.ones(conv_len) / conv_len
    else:
        w = e / total
    # Gradient via FFT correlation: grad = 2 * correlate(w, a)[:n]
    total_len = conv_len + n - 1
    size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    return 2.0 * corr[:n]

def _mirror_step(a, grad, lr):
    """Mirror descent (exponential) step on the simplex."""
    # a > 0 guaranteed if we start positive
    log_ratio = -lr * grad
    # Clip to avoid overflow
    log_ratio = np.clip(log_ratio, -50, 50)
    a_new = a * np.exp(log_ratio)
    s = a_new.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a_new / s

def _optimize(a_init, time_budget=60.0, seed=0):
    """Run optimization from a_init, return (best_a, best_score)."""
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    n = len(a)
    
    # Ensure on simplex
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    # Ensure strictly positive for mirror descent
    a = a + 1e-10
    a = a / a.sum()
    
    deadline = time.monotonic() + time_budget
    best_a = a.copy()
    best_score = _compute_score(a)
    
    temp = 0.1  # temperature for smoothing
    lr = 0.01  # learning rate
    patience = 500
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        grad = _smoothed_gradient(a, c, temp)
        
        a_new = _mirror_step(a, grad, lr)
        a = a_new
        
        score = _compute_score(a)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameter updates
        if no_improve == 100:
            lr *= 0.9
            temp *= 0.95
        elif no_improve == patience:
            # Restart with different parameters
            temp = max(0.01, temp * 0.5)
            lr = lr * 2.0
            no_improve = 0
        
        if temp < 0.005:
            temp = 0.005
    
    return best_a, best_score

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
