    candidates.append(("ramp", t + 0.1))

    # Two-block
    two_block = np.ones(n)
    two_block[:n // 2] = 1.5
    two_block[n // 2:] = 0.5
    candidates.append(("two_block", two_block))

    # Periodic (period 2: alternating 1.5 and 0.5)
    periodic = np.ones(n)
    periodic[::2] = 1.5
    periodic[1::2] = 0.5
    candidates.append(("periodic2", periodic))

    # Periodic (period 3)
    per3 = np.ones(n)
    per3[::3] = 2.0
    per3[1::3] = 0.5
    per3[2::3] = 0.833
    candidates.append(("periodic3", per3))

    # Evaluate all structured candidates
    for name, cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        score = compute_score_fast(cand)
        verified = evaluate_sequence(cand.tolist())
        if np.isfinite(verified) and verified < best_score:
            best = cand
            best_score = verified
            print(f"  {name}: score = {verified:.6f} (IMPROVEMENT!)")
        else:
            print(f"  {name}: score = {verified:.6f}")

    # Random search with various perturbation strategies
    iters = 0
    while time.monotonic() < deadline:
        iters += 1
        strat = iters % 5
        cand = np.ones(n)

        if strat == 0:
            # Single perturbation
            idx = rng.integers(0, n)
            cand[idx] = rng.uniform(0.0, 3.0)
        elif strat == 1:
            # Block perturbation
            w = rng.integers(max(1, n // 100), max(2, n // 10))
            s = rng.integers(0, max(1, n - w))
            cand[s:s + w] *= np.exp(rng.normal(0.0, 0.5))
        elif strat == 2:
            # Random noise
            cand += rng.normal(0, 0.3, n)
            cand = np.maximum(cand, 0.0)
        elif strat == 3:
            # Sparse random peaks
            for _ in range(rng.integers(1, 6)):
                idx = rng.integers(0, n)
                cand[idx] = rng.uniform(0.5, 3.0)
        else:
            # Symmetric random
            half = n // 2
            first = np.ones(half)
            first[rng.integers(0, half)] = rng.uniform(0.0, 3.0)
            cand[:half] = first
            cand[half:] = first[::-1] if n % 2 == 0 else np.concatenate([first[::-1], [1.0]])

        cand = np.clip(cand, 0.0, 1000.0)
        score = compute_score_fast(cand)
        if score < best_score + 1e-10:  # Only verify if fast score suggests improvement
            verified = evaluate_sequence(cand.tolist())
            if np.isfinite(verified) and verified < best_score:
                best = cand
                best_score = verified
                print(f"  Random iter {iters}: score = {verified:.6f} (IMPROVEMENT!)")
                break

        if iters % 5000 == 0:
            print(f"  Random search iter {iters}: best = {best_score:.6f}")
            if time.monotonic() > deadline - 10:
                break

    return [float(x) for x in best]
