import numpy as np
import time

def project_onto_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, target_sum / len(v))
    # Binary search for tau
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, len(v) + 1) > (cumsum[-1] - target_sum))[0]
    if len(rho) == 0:
        rho = len(v) - 1
    else:
        rho = rho[0]
    tau = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(v - tau, 0)

def compute_score_fast(a):
    """Compute 2*n*max(conv)/(sum)^2 efficiently."""
    conv = np.convolve(a, a)
    return 2 * len(a) * conv.max() / (a.sum() ** 2)

def soft_max_gradient(a, beta):
    """Compute gradient of soft-max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    # Soft-max weights

    r = np.convolve(a[::-1].copy(), w)[n-1:2*n-1]
    return 2 * r, conv
