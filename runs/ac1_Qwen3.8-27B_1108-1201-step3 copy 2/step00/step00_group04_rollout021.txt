import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def objective_from_C(C, n, S):
    M = C.max()
    return 2.0 * n * M / (S * S)

def compute_grad(a, k_star):
    n = len(a)
    j_min = max(0, k_star - n + 1)
    j_max = min(k_star, n - 1)
    
    grad = np.zeros(n)
    
    # Case 1: i in [j_min, j_max]
    idx1 = np.arange(j_min, j_max + 1)
    grad[idx1] += a[k_star - idx1]
    
    # Case 2: k_star - i in [j_min, j_max] => i in [k_star - j_max, k_star - j_min]
    i_lo = max(0, k_star - j_max)
    i_hi = min(n - 1, k_star - j_min)
    idx2 = np.arange(i_lo, i_hi + 1)
    grad[idx2] += a[k_star - idx2]
    
    return grad

def optimize_from_start(a, n, rng, deadline, patience=500):
    """Run gradient descent from a starting point."""
    S_target = np.sqrt(2.0 * n)
    S = a.sum()
    if S > 0.01:
        a = a * (S_target / S)
    
    best_a = a.copy()
    best_score = objective_from_C(fast_autocorr(a), n, a.sum())
    
    lr = 0.01
    v = np.zeros(n)  # momentum
    beta = 0.9
    no_improve = 0
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        S = a.sum()
        if S < 0.01:
            a = np.abs(a) + 0.01
            S = a.sum()
            a = a * (S_target / S)
            continue
        
        C = fast_autocorr(a)
        M = C.max()
        obj = 2.0 * n * M / (S * S)
        
        if obj < best_score - 1e-9:
            best_score = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Random restart
            if rng.random() < 0.5:
                # Small perturbation
                a = np.clip(a + rng.normal(0, S_target/n * 0.1, n), 0, 1000)
            else:
                # Larger perturbation
                idx = rng.integers(0, n, size=n//10)
                a[idx] *= np.exp(rng.normal(0, 0.3, size=len(idx)))
            a = np.clip(a, 0, 1000)
            S = a.sum()
            a = a * (S_target / S)
            no_improve = 0
            if no_improve > patience * 5:
                break
            continue
        
        k_star = C.argmax()
        grad_C = compute_grad(a, k_star)
        
        # Gradient of objective
        grad_obj = (2.0 * n / (S * S)) * grad_C - (4.0 * n * M / (S ** 3))
        
        # Nesterov momentum
        v = beta * v + grad_obj
        a = a - lr * v
        a = np.clip(a, 0, 1000)
        
        # Re-normalize
        S = a.sum()
        if S > 0.01:
            a = a * (S_target / S)
        
        # Decay learning rate
        if it % 500 == 0:
            lr *= 0.9
        
        # Also try to reduce M by directly lowering the peak coordinate
        # (heuristic: reduce a at positions with high sensitivity to C[k_star])
        if it % 1000 == 500:

            sens = grad_C / max(S, 1e-10)

            a = np.clip(a, 0, 1000)
            S = a.sum()
            a = a * (S_target / S)
    
    return best_a, best_score
