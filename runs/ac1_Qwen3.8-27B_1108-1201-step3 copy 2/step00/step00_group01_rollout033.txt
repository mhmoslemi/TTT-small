def solve_lp(f, M):
    """Solve max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # Build A matrix where A[k, i] = f[i] if 0 <= k-i < n, else 0

    # g >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

def build_conv_matrix(f):
    """Build the sparse convolution matrix A where A @ g = convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # Column i: A[k, i] = f[i] for k in [i, i+n-1] (clipped to [0, 2n-2])
        k_start = i
        k_end = min(2*n-2, i + n - 1) + 1
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M):
    """Solve max sum(g) s.t. g >= 0, (f*g)[k] <= M."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_conv_matrix(f)
    b = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except:
        pass
    return None

def peak_gradient(f, k_star):
    """Gradient of (f*f)[k_star] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    for j in range(n):
        k_minus_j = k_star - j
        if 0 <= k_minus_j < n:
            grad[j] = 2.0 * f[k_minus_j]
    return grad

def project_simplex(v):

    theta = (cssv[rho] - 1) / (rho + 1.0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _build_conv_matrix(f):
    """Build sparse matrix A where A @ g = np.convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        rows.extend(range(k_start, k_end))
        cols.extend([i] * (k_end - k_start))
        vals.extend([f[i]] * (k_end - k_start))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(f, M):
    """Solve max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_obj = -np.ones(n)
    A = _build_conv_matrix(f)
    b = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _peak_grad(f, k_star):
    """Gradient of (f*f)[k_star] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for j in range(lo, hi):
        kmj = k_star - j
        if 0 <= kmj < n:
            grad[j] = 2.0 * f[kmj]
    return grad

    cool = 0.999

    while time.monotonic() - start < time_limit:
        # Random block perturbation
        width = max(1, int(rng.integers(1, max(2, n // 10))))
        pos = int(rng.integers(0, n - width + 1))
        f_new = f.copy()
        f_new[pos:pos + width] *= np.exp(rng.normal(0.0, 0.3, width))

        # Occasionally do a global rescale perturbation
        if rng.random() < 0.2:
            f_new *= np.exp(rng.normal(0.0, 0.1))

        f_new = np.maximum(f_new, 0.0)
        f_new = _project_simplex(f_new)

        c_new = np.convolve(f_new, f_new)
        M_new = float(np.max(c_new))

        # Metropolis acceptance
        delta = M_new - M
        if delta < 0 or rng.random() < np.exp(-max(0.0, delta) / T):
            f = f_new
            M = M_new
            c = c_new
            if M < best_M:
                best_f = f.copy()
                best_M = M
        T = max(T_min, T * cool)

    return best_f


def _make_initial(n, rng, mode=0):
    """Create initial sequence of length n."""
    if mode == 0:
        # Uniform
        f = np.ones(n)
    elif mode == 1:
        # Slight asymmetry (linear ramp)
        f = np.linspace(0.5, 1.5, n)
    elif mode == 2:
        # Cosine modulation
        f = 1.0 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) / n)
    elif mode == 3:
        # Alternating
        f = np.where(np.arange(n) % 2 == 0, 1.5, 0.5)
    elif mode == 4:
        # Two bumps
        centers = [n // 3, 2 * n // 3]
        f = np.zeros(n)
        for c in centers:
            for i in range(n):
                f[i] += np.exp(-((i - c) ** 2) / (2 * (n / 8.0) ** 2))
    elif mode == 5:
        # Random
        f = rng.random(n) + 0.1
    else:
        # Smooth random
        f = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 8, n // 3))
            amp = rng.uniform(0.2, 0.8) * rng.choice([-1.0, 1.0])
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                f[i] += amp * np.exp(-d ** 2 / (2 * width ** 2))
        f = np.maximum(f, 0.1)

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
    else:
        f = f / s
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/(sum(f))^2.
    Uses projected gradient, LP refinement, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Start from the provided sequence if valid
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0 and np.isfinite(parent).all():
            best = parent.copy()
        else:
            best = np.ones(100)
    except NameError:
        best = np.ones(100)

    best = np.maximum(best, 0.0)
    bs = best.sum()
    if bs > 1e-10:
        best = best / bs
    best_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Initial score: {best_score:.6f}, n={len(best)}")

    # Phase 1: Try PGD for various n and starting points
    n_list = [50, 100, 200, 400, 6121]
    modes = [0, 1, 2, 3, 4, 5, 6]

    for n in n_list:
        if time.monotonic() > deadline - 120:
            break
        if n > 5000:
            time_alloc = 60.0
        elif n > 200:
            time_alloc = 45.0
        elif n > 100:
            time_alloc = 40.0
        else:
            time_alloc = 35.0

        for mode in modes:
            if time.monotonic() > deadline - 60:
                break
            f_init = _make_initial(n, rng, mode)
            # PGD refine
            f = _pgd_refine(f_init, time_limit=time_alloc, rng=rng)
            c = np.convolve(f, f)
            M = float(np.max(c))
            score = 2.0 * n * M
            if score < best_score:
                best = f.copy()
                best_score = score
                print(f"PGD n={n} mode={mode}: score={score:.6f}")

    # Phase 2: LP refinement on best candidates
    if time.monotonic() < deadline - 60:
        n_best = len(best)
        if n_best <= 1500:
            f = _lp_refine(best, time_limit=min(120.0, deadline - time.monotonic() - 30))
            if f is not None and len(f) == n_best:
                c = np.convolve(f, f)
                M = float(np.max(c))
                score = 2.0 * n_best * M
                if score < best_score:
                    best = f.copy()
                    best_score = score
                    print(f"LP refine: score={score:.6f}")

    # Phase 3: Local search / simulated annealing
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 10
        f = _local_search(best, time_limit=remaining, rng=rng)
        n_best = len(f)
        c = np.convolve(f, f)
        M = float(np.max(c))
        score = 2.0 * n_best * M
        if score < best_score:
            best = f.copy()
            best_score = score
            print(f"Local search: score={score:.6f}")

    # Phase 4: Final PGD polish
    if time.monotonic() < deadline - 5:
        n_best = len(best)
        if n_best <= 5000:
            time_left = deadline - time.monotonic() - 2
            f = _pgd_refine(best, time_limit=time_left, rng=rng)
            c = np.convolve(f, f)
            M = float(np.max(c))
            score = 2.0 * n_best * M
            if score < best_score:
                best = f.copy()
                best_score = score
                print(f"Final PGD: score={score:.6f}")

    # Verify with actual evaluator
    final_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Final score (evaluated): {final_score:.6f}, n={len(best)}")

    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback to the better of our estimates
        best_score = min(best_score, final_score) if np.isfinite(final_score) else best_score

    return [float(x) for x in best]
