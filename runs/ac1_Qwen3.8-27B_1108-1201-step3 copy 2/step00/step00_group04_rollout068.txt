def targeted_move(a, k_star, rng):
    """Make a move that targets reducing c[k_star]."""
    n = len(a)
    a_new = a.copy()
    
    # Find positions that contribute to c[k_star]
    # c[k_star] = sum_i a[i] * a[k_star - i]
    # Valid i: max(0, k_star - (n-1)) <= i <= min(n-1, k_star)
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(n - 1, k_star)
    
    if i_lo > i_hi:
        return a_new  # shouldn't happen
    
    # Get contributions
    contribs = np.zeros(n)
    for i in range(i_lo, i_hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contribs[i] += a[i] * a[j]
    
    # Find the position with highest contribution
    # But we need to be careful: reducing a[i] reduces c[k_star] by ~a[j]*delta
    # but also affects other lags
    
    # Pick a random high-contributor
    total_contrib = contribs.sum()
    if total_contrib < 1e-10:
        return a_new
    
    # Weight-probability selection
    probs = contribs / total_contrib
    i_src = rng.choice(n, p=probs)
    
    # Pick a destination that doesn't contribute to c[k_star]
    # Position j contributes to c[k_star] if 0 <= k_star - j < n
    # i.e., k_star - (n-1) <= j <= k_star
    # So positions OUTSIDE this range don't contribute
    
    # Find positions that don't contribute
    j_lo_bad = max(0, k_star - (n - 1))
    j_hi_bad = min(n - 1, k_star)
    
    # Good destinations: 0..j_lo_bad-1 or j_hi_bad+1..n-1
    good_dests = []
    if j_lo_bad > 0:
        good_dests.extend(range(0, j_lo_bad))
    if j_hi_bad < n - 1:
        good_dests.extend(range(j_hi_bad + 1, n))
    
    if not good_dests:
        # All positions contribute (this happens for center lags when n is small)
        # Just pick a random position
        j_dst = rng.integers(0, n)
    else:
        j_dst = good_dests[rng.integers(0, len(good_dests))]
    
    if i_src == j_dst:
        return a_new
    
    # Transfer mass
    delta = a[i_src] * rng.uniform(0.01, 0.1)
    a_new[i_src] -= delta
    a_new[j_dst] += delta
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new

import numpy as np

def fast_eval(a):
    """Fast evaluation without the overhead of list conversion."""
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = c.max()
    return float(2.0 * n * max_c / (S * S))

def make_ramp(n, rng, direction='up'):
    """Create a ramp-like initialization."""
    if direction == 'up':
        base = np.linspace(1.0, 2.0, n)
    elif direction == 'down':
        base = np.linspace(2.0, 1.0, n)
    else:  # two-level
        mid = n // 2 + int(rng.integers(-n//10, n//10))
        base = np.ones(n)
        base[mid:] = 1.5 + rng.uniform(0, 0.5)
    # Add some noise
    base += rng.normal(0, 0.05, n)
    return np.clip(base, 0.0, 1000.0)

def make_random_nonpal(n, rng):
    """Random initialization that's definitely non-palindromic."""
    a = rng.uniform(0.5, 2.0, n)
    # Add a linear trend to ensure non-palindromic
    trend = rng.uniform(-0.5, 0.5)
    a += trend * np.linspace(-1, 1, n)
    return np.clip(a, 0.0, 1000.0)

def make_block_perturbation(a, rng, width_range=None):
    """Perturb a random block."""
    n = a.shape[0]
    a_new = a.copy()
    if width_range is None:
        width_range = (max(1, n//200), max(2, n//10))
    width = int(rng.integers(width_range[0], width_range[1] + 1))
    start = int(rng.integers(0, n - width + 1))
    # Multiply by random factor
    factor = float(np.exp(rng.normal(0.0, 0.15)))
    a_new[start:start+width] *= factor
    # Or add random offset
    if rng.random() < 0.3:
        a_new[start:start+width] += rng.uniform(-0.1, 0.1) * max(a[start:start+width].max(), 0.1)
    return np.clip(a_new, 0.0, 1000.0)

def make_targeted_move(a, rng):
    """Make a move targeting the peak of the convolution."""
    n = a.shape[0]
    a_new = a.copy()
    
    # Compute conv and find peak
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    
    # Find contributing range for lag k_star
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(n - 1, k_star)
    
    if i_lo > i_hi:
        return a_new  # shouldn't happen
    
    # Compute contributions a[i] * a[k_star - i]
    indices_src = np.arange(i_lo, i_hi + 1)
    indices_dst = k_star - indices_src
    contribs = a[indices_src] * a[indices_dst]
    
    total = contribs.sum()
    if total < 1e-10:
        return a_new
    
    # Pick source position (weighted by contribution)
    probs = contribs / total
    idx = int(rng.choice(len(indices_src), p=probs))
    i_src = int(indices_src[idx])
    
    # Pick destination that doesn't contribute to c[k_star]
    # Positions j where j < i_lo or j > i_hi (in range [0, n-1])
    # Actually, position j contributes to c[k_star] iff i_lo <= j <= i_hi
    # (where i_lo = max(0, k_star-(n-1)), i_hi = min(n-1, k_star))
    
    # Good destinations: [0, i_lo-1] ∪ [i_hi+1, n-1]
    n_good_left = i_lo  # positions 0..i_lo-1
    n_good_right = n - 1 - i_hi  # positions i_hi+1..n-1
    n_good = n_good_left + n_good_right
    
    if n_good == 0:
        # Can't avoid (small n or center lag)
        j_dst = int(rng.integers(0, n))
    else:
        if rng.random() < n_good_left / n_good:
            j_dst = int(rng.integers(0, i_lo))
        else:
            j_dst = int(rng.integers(i_hi + 1, n))
    
    if i_src == j_dst:
        return a_new
    
    # Transfer a fraction of mass
    frac = float(rng.uniform(0.05, 0.25))
    delta = a[i_src] * frac
    if delta < 1e-10:
        return a_new
    a_new[i_src] -= delta
    a_new[j_dst] += delta
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new

def local_search(a_init, rng, time_budget, best_score):
    """Run local search from a starting point."""
    start_time = time.monotonic()
    a = a_init.copy()
    S = a.sum()
    # Normalize to avoid numerical issues
    if S > 0:
        a = a / S * 1.0  # normalize sum to 1
    
    score = fast_eval(a)
    best_a = a.copy()
    best_sc = score
    
    # Simulated annealing
    temp = 0.01
    cool_rate = 0.9999
    min_temp = 1e-6
    
    iters = 0
    while time.monotonic() - start_time < time_budget:
        # Choose move type
        r = rng.random()
        if r < 0.5:

        elif r < 0.8:

        else:

        if S_new < 0.01:
            iters += 1
            continue
        a_new = a_new / S_new
        
        score_new = fast_eval(a_new)

            best_sc = score_new
            a = a_new
            # Reset temperature on improvement

                score = score_new

"""
Multi-start local search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: non-palindromic sequences can beat the palindromic bound of 2.0.
Uses convolution-aware targeted moves plus random exploration with simulated annealing.

import numpy as np
import time


def fast_eval(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is finite, non-negative."""
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = c.max()
    return float(2.0 * n * max_c / (S * S))


def fast_eval_given_conv(a, c):
    """Compute objective given pre-computed convolution."""
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def init_ramp_up(n, rng):
    """Linear ramp increasing."""
    a = np.linspace(0.5, 2.0, n)
    a += rng.normal(0, 0.05, n)
    return np.clip(a, 0.0, 1000.0)


def init_ramp_down(n, rng):
    """Linear ramp decreasing."""
    a = np.linspace(2.0, 0.5, n)
    a += rng.normal(0, 0.05, n)
    return np.clip(a, 0.0, 1000.0)


def init_two_level(n, rng):
    """Two-level: first part low, second part high (asymmetric split)."""
    split = int(n * rng.uniform(0.3, 0.7))
    a = np.ones(n)
    a[split:] = rng.uniform(1.5, 3.0)
    a += rng.normal(0, 0.05, n)
    return np.clip(a, 0.0, 1000.0)


def init_exponential(n, rng):
    """Exponential decay or growth (asymmetric)."""
    direction = rng.choice([1, -1])
    x = np.linspace(0, 3.0, n)
    a = np.exp(direction * x / 1.5)
    a = a / a.max() * 2.0
    a += rng.normal(0, 0.03, n)
    return np.clip(a, 0.0, 1000.0)


def init_random_trend(n, rng):
    """Random values with a linear trend component."""
    a = rng.uniform(0.5, 2.0, n)
    trend_slope = rng.uniform(-0.8, 0.8)
    a += trend_slope * np.linspace(-0.5, 0.5, n)
    return np.clip(a, 0.0, 1000.0)


def init_gauss_offset(n, rng):
    """Gaussian bump offset from center."""
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.1, 0.4)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = a / a.max() * 2.0
    a += rng.normal(0, 0.05, n)
    return np.clip(a, 0.0, 1000.0)


def init_two_bumps(n, rng):
    """Two Gaussian bumps at asymmetric positions."""
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.8 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    a = a / a.max() * 2.0
    a += rng.normal(0, 0.03, n)
    return np.clip(a, 0.0, 1000.0)


def init_sinusoidal(n, rng):
    """Sinusoidal modulation (non-palindromic phase)."""
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    a = 1.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    a += rng.normal(0, 0.05, n)
    return np.clip(a, 0.0, 1000.0)


def targeted_move(a, c, k_star, n, rng):
    """
    Move that targets reducing c[k_star].
    Transfers mass from a high-contributing position to one that doesn't
    contribute (or contributes less) to the peak lag.
    """
    a_new = a.copy()

    # Contributing range for lag k_star
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(n - 1, k_star)

    if i_lo > i_hi:
        return a_new

    idx_src = np.arange(i_lo, i_hi + 1)
    idx_dst = k_star - idx_src
    contribs = a[idx_src] * a[idx_dst]

    total = contribs.sum()
    if total < 1e-12:
        return a_new

    # Pick source weighted by contribution
    probs = contribs / total
    sel = rng.choice(len(idx_src), p=probs)
    i_from = int(idx_src[sel])

    # Pick destination that does NOT contribute to c[k_star]
    # j contributes iff i_lo <= j <= i_hi
    good_left = i_lo  # positions 0..i_lo-1
    good_right = n - 1 - i_hi  # positions i_hi+1..n-1
    good_total = good_left + good_right

    if good_total == 0:
        # All positions contribute (rare, only for small n)
        i_to = int(rng.integers(0, n))
    else:
        if rng.random() < good_left / good_total:
            i_to = int(rng.integers(0, i_lo))
        else:
            i_to = int(rng.integers(i_hi + 1, n))

    if i_from == i_to:
        return a_new

    # Transfer fraction of mass
    frac = float(rng.uniform(0.05, 0.30))
    delta = a[i_from] * frac
    if delta < 1e-12:
        return a_new
    a_new[i_from] -= delta
    a_new[i_to] += delta
    np.clip(a_new, 0.0, 1000.0, out=a_new)

    return a_new


def targeted_move_multi(a, c, n, rng):
    """Target multiple peak lags simultaneously."""
    a_new = a.copy()
    max_val = c.max()

    # Find all lags within 5% of max
    threshold = 0.95 * max_val
    peak_lags = np.where(c >= threshold)[0]

    for k_star in peak_lags[:3]:  # Handle top 3 peaks
        k_star = int(k_star)
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        if i_lo > i_hi:
            continue

        idx_src = np.arange(i_lo, i_hi + 1)
        idx_dst = k_star - idx_src
        contribs = a_new[idx_src] * a_new[idx_dst]
        total = contribs.sum()
        if total < 1e-12:
            continue

        probs = contribs / total
        sel = rng.choice(len(idx_src), p=probs)
        i_from = int(idx_src[sel])

        good_left = i_lo
        good_right = n - 1 - i_hi
        good_total = good_left + good_right

        if good_total == 0:
            i_to = int(rng.integers(0, n))
        else:
            if rng.random() < good_left / good_total:
                i_to = int(rng.integers(0, i_lo))
            else:
                i_to = int(rng.integers(i_hi + 1, n))

        if i_from != i_to:
            frac = float(rng.uniform(0.03, 0.15))
            delta = a_new[i_from] * frac
            if delta > 1e-12:
                a_new[i_from] -= delta
                a_new[i_to] += delta

    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


def block_perturb(a, rng):
    """Random block perturbation (no symmetrization!)."""
    n = a.shape[0]
    a_new = a.copy()
    # 1-3 blocks
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        width = int(rng.integers(max(1, n // 500), max(2, n // 8)))
        start = int(rng.integers(0, n - width + 1))
        stop = min(n, start + width)
        r = rng.random()
        if r < 0.5:
            a_new[start:stop] *= float(np.exp(rng.normal(0, 0.12)))
        elif r < 0.8:
            a_new[start:stop] += float(rng.normal(0, 0.05)) * max(a[start:stop].max(), 0.5)
        else:
            # Swap with another block
            w2 = int(rng.integers(max(1, n // 500), max(2, n // 8)))
            s2 = int(rng.integers(0, n - w2 + 1))
            e2 = min(n, s2 + w2)
            min_w = min(stop - start, e2 - s2)
            if min_w > 0:
                a_new[start:start + min_w], a_new[s2:s2 + min_w] = a_new[s2:s2 + min_w].copy(), a_new[start:start + min_w].copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


def local_search(a_init, rng, time_budget):
    """
    Simulated annealing local search with convolution-aware moves.
    Returns (best_a, best_score, n_iters).
    """
    start_time = time.monotonic()
    n = a_init.shape[0]
    a = a_init.copy()
    S = a.sum()
    if S < 0.01:
        return a, np.inf, 0
    a = a / S  # normalize sum to 1
    score = fast_eval(a)
    best_a = a.copy()
    best_sc = score
    temp = 0.005
    iters = 0
    no_improve = 0

    while time.monotonic() - start_time < time_budget:
        # Compute convolution
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        # Choose move
        r = rng.random()
        if r < 0.45:
            a_new = targeted_move(a, c, k_star, n, rng)
        elif r < 0.70:
            a_new = targeted_move_multi(a, c, n, rng)
        elif r < 0.90:
            a_new = block_perturb(a, rng)
        else:
            # Small Gaussian perturbation
            a_new = a + rng.normal(0, 0.03, n)
            np.clip(a_new, 0.0, 1000.0, out=a_new)

        S_new = a_new.sum()
        if S_new < 0.01:
            iters += 1
            continue
        a_new = a_new / S_new

        # Evaluate
        c_new = np.convolve(a_new, a_new)
        score_new = fast_eval_given_conv(a_new, c_new)

        if not np.isfinite(score_new):
            iters += 1
            continue

        if score_new < best_sc:
            best_a = a_new.copy()
            best_sc = score_new
            a = a_new
            score = score_new
            temp = max(temp, 0.005)
            no_improve = 0
        else:
            delta = score_new - best_sc
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                a = a_new
                score = score_new
            no_improve += 1

        # Adaptive cooling
        if no_improve > 200:
            temp *= 0.9
            if temp < 1e-6:
                # Kick: large perturbation to escape local minimum
                a = a * np.exp(rng.normal(0, 0.3, n))
                S_k = a.sum()
                if S_k > 0.01:
                    a = a / S_k
                score = fast_eval(a)
                temp = 0.01
                no_improve = 0

        temp *= 0.99995
        iters += 1

    return best_a, best_sc, iters
