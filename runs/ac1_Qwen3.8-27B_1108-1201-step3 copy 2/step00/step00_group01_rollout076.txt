Uses centered subgradient descent with FFT convolution,

def fast_self_convolve(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    result_len = 2 * n - 1
    N = 1 << (result_len - 1).bit_length()
    fa = np.fft.rfft(a, N)
    real = np.fft.irfft(fa * fa, N)
    return real[:result_len]


def compute_centered_gradient(a, peaks, n):
    """Compute centered subgradient of max convolution value.
    
    For peak at position k, dM/da_j = a[k-j] for valid j.
    We center the gradient to preserve the sum (scale invariance).
    """
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            grad[idx] += a[k - idx]
    grad /= len(peaks)
    grad_mean = np.mean(grad)
    return grad - grad_mean
