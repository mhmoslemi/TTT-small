"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a)^2)
using FFT-based smooth-max exponentiated gradient, multiple sequence lengths,
structured/random restarts, and a final exact hill-climb polish.
"""

import numpy as np
import time

try:
    from scipy.fft import fft, ifft, next_fast_len
except Exception:
    from numpy.fft import fft, ifft

    def next_fast_len(n):
        n = int(max(1, n))
        return 1 << (n - 1).bit_length()


def _normalize(a):
    """Convert to finite nonnegative float array normalized to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _normalize_positive(v):
    """Normalize a possibly messy vector to a positive-sum-1 vector."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return v / s


def _conv_self(a):
    """Full linear self-convolution of a, using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_fast_len(2 * n - 1)
    fa = fft(a, n=size)
    c = np.real(ifft(fa * fa, n=size))
    return c[:2 * n - 1]


def _score_sum1(a):
    """Exact objective for a normalized sequence with sum(a) == 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    c = _conv_self(a)
    m = float(np.max(c)) if c.size else 0.0
    if m < 0.0:
        m = 0.0
    return 2.0 * float(n) * m


def _smooth_val(c, alpha):
    """Smooth log-sum-exp approximation to max(c)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    if alpha <= 1.0:
        return m
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        s = float(np.sum(np.exp(alpha * (c - m))))
    if not np.isfinite(s) or s <= 1e-300:
        return m
    return m + float(np.log(s)) / float(alpha)


def _smooth_weights(c, alpha):
    """Weights proportional to exp(alpha*c), normalized."""
    c = np.asarray(c, dtype=np.float64)
    k = c.size
    if k == 0:
        return np.empty(0, dtype=np.float64)
    if alpha <= 1.0:
        return np.full(k, 1.0 / float(k), dtype=np.float64)
    m = float(np.max(c))
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        w = np.exp(alpha * (c - m))
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        ix = int(np.argmax(c))
        w = np.zeros(k, dtype=np.float64)
        w[ix] = 1.0
        return w
    return w / s


def _conv_self_grad(a, weights):
    """
    Gradient of sum_k weights[k] * c_k with respect to a,
    where c = full linear convolution of a with itself.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(weights, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_fast_len(2 * n - 1)
    b = a[::-1]
    conv = np.real(ifft(fft(w, n=size) * fft(b, n=size), n=size))
    return 2.0 * conv[n - 1:n - 1 + n]


def _exp_grad_optimize(a0, alpha, eta, max_iter, deadline):
    """
    Exponentiated-gradient method on the probability simplex for the
    smooth-max self-convolution objective.
    """
    n = len(a0)
    if n == 0:
        return a0, float("inf")

    a = _normalize(a0)
    best_a = a.copy()
    best_score = _score_sum1(best_a)
    eta = max(0.1, float(eta))
    no_improve = 0

    for it in range(int(max_iter)):
        if time.monotonic() >= float(deadline):
            break

        c = _conv_self(a)
        val = _smooth_val(c, float(alpha))
        maxc = float(np.max(c)) if c.size else 0.0
        if maxc < 0.0:
            maxc = 0.0
        score = 2.0 * float(n) * maxc

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 4000:
                break

        w = _smooth_weights(c, float(alpha))
        g = _conv_self_grad(a, w)
        if not np.all(np.isfinite(g)):
            break
        g = np.asarray(g, dtype=np.float64)
        g -= float(np.mean(g))
        if not np.any(g):
            break

        accepted = False
        for _attempt in range(6):
            expts = -eta * g
            expts = np.clip(expts, -30.0, 30.0)
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                z = a * np.exp(expts)
            z = np.where(np.isfinite(z), z, 0.0)
            z = np.maximum(z, 0.0)
            s = float(np.sum(z))
            if not np.isfinite(s) or s <= 1e-300:
                z = np.full(n, 1.0 / float(n), dtype=np.float64)
            z /= s

            c_new = _conv_self(z)
            val_new = _smooth_val(c_new, float(alpha))
            maxc_new = float(np.max(c_new)) if c_new.size else 0.0
            if maxc_new < 0.0:
                maxc_new = 0.0
            score_new = 2.0 * float(n) * maxc_new

            if val_new <= val * (1.0 + 1e-10) + 1e-14:
                a = z
                val = val_new
                if score_new < best_score:
                    best_score = score_new
                    best_a = z.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if val_new < val:
                    eta = min(eta * 1.12, 5.0 * float(n))
                else:
                    eta = min(eta * 1.02, 5.0 * float(n))
                accepted = True
                break

            eta *= 0.5

        if not accepted:
            eta = max(0.1, eta * 0.75)
            no_improve += 10

        if it % 200 == 199:
            a = 0.5 * (a + a[::-1])
            a = _normalize(a)

        if it % 1000 == 999:
            a = 0.95 * a + 0.05 * (1.0 / float(n))
            a = _normalize(a)

    return best_a, best_score


def _make_starts(n, rng, max_starts=9):
    """Create several normalized nonnegative starting sequences."""
    n = int(n)
    if n < 1:
        return []

    starts = []

    starts.append(_normalize_positive(np.ones(n, dtype=np.float64)))

    r = rng.random(n) + 1e-12
    starts.append(_normalize_positive(r))

    try:
        phases = np.exp(1j * 2.0 * np.pi * rng.random(n))
        m = np.abs(ifft(phases, n=n))
        starts.append(_normalize_positive(m + 1e-12))
    except Exception:
        pass

    try:
        x = (rng.random(n) > 0.5).astype(np.float64)
        m = np.abs(fft(x, n=n))
        starts.append(_normalize_positive(m + 1e-12))
    except Exception:
        pass

    base = np.ones(n, dtype=np.float64) / float(n)
    for _ in range(2):
        z = base.copy()
        w = max(1, n // 128)
        nblocks = max(3, min(128, n // 64))
        for _b in range(nblocks):
            st = int(rng.integers(0, max(1, n - w + 1)))
            stop = min(n, st + w)
            z[st:stop] *= float(np.exp(rng.normal(0.0, 0.5)))
        starts.append(_normalize_positive(z))

    x = np.arange(n, dtype=np.float64)
    sigma = max(1e-9, 0.2 * float(n))
    g = np.exp(-0.5 * ((x - 0.5 * float(n)) / sigma) ** 2)
    starts.append(_normalize_positive(g + 1e-12))

    inv = np.exp(0.5 * ((x - 0.5 * float(n)) / sigma) ** 2)
    starts.append(_normalize_positive(inv + 1e-12))

    s = np.abs(np.sin(np.pi * (x + 0.5) / float(n)))
    starts.append(_normalize_positive(s + 1e-12))

    return starts[:int(max_starts)]


def _exact_hill_climb(a, deadline, rng):
    """Exact FFT-score multiplicative block hill-climbing polish."""
    a = _normalize(a)
    n = len(a)
    if n == 0:
        return a, float("inf")

    best = a.copy()
    best_score = _score_sum1(best)
    temp = 0.35
    no_improve = 0
    it = 0

    while time.monotonic() < float(deadline):
        cand = best.copy()

        if n < 512:
            blocks = 1
        elif n < 4096:
            blocks = 2
        else:
            blocks = 3

        for _ in range(blocks):
            min_w = max(1, n // 4000)
            max_w = max(min_w + 1, n // 16)
            w = int(rng.integers(min_w, max_w + 1))
            st = int(rng.integers(0, max(1, n - w + 1)))
            stop = min(n, st + w)
            f = float(np.exp(rng.normal(0.0, temp)))
            if not np.isfinite(f):
                f = 1.0
            f = min(50.0, max(0.02, f))
            cand[st:stop] *= f

        if rng.random() < 0.05:
            cand = 0.5 * (cand + best[::-1])

        if rng.random() < 0.03:
            w = max(1, n // 64)
            st = int(rng.integers(0, max(1, n - w + 1)))
            stop = min(n, st + w)
            cand[st:stop] = float(np.mean(cand))

        cand = _normalize(cand)
        sc = _score_sum1(cand)

        if sc < best_score:
            best = cand
            best_score = sc
            no_improve = 0
            temp = max(0.02, temp * 0.99)
            if it % 2000 == 0:
                print(f"hill-climb n={n} score={best_score:.6f} t={time.monotonic():.1f}", flush=True)
        else:
            no_improve += 1
            if no_improve > 1000:
                temp = max(0.02, temp * 0.8)
                no_improve = 0

        it += 1

    return best, best_score


def _optimize_length(n, budget_s, rng):
    """Optimize for one fixed sequence length within a time budget."""
    n = int(n)
    if n < 2:
        return None, float("inf")

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    if deadline - t0 < 3.0:
        return None, float("inf")

    max_starts = 9 if n <= 512 else 7 if n <= 2048 else 5
    starts = _make_starts(n, rng, max_starts)
    if not starts:
        starts = [_normalize_positive(np.ones(n, dtype=np.float64))]

    if n <= 512:
        alphas = [max(1.0, 0.5 * n), max(1.0, 5.0 * n), max(1.0, 20.0 * n)]
    elif n <= 4096:
        alphas = [max(1.0, 1.0 * n), max(1.0, 8.0 * n), max(1.0, 40.0 * n)]
    else:
        alphas = [max(1.0, 2.0 * n), max(1.0, 15.0 * n)]

    a_for_start = [s.copy() for s in starts]
    best_a = starts[0].copy()
    best_score = _score_sum1(best_a)

    total_tasks = len(starts) * len(alphas)
    idx = 0

    for i, _s in enumerate(starts):
        for alpha in alphas:
            now = time.monotonic()
            remaining = deadline - now
            if remaining < 2.0:
                break

            tasks_left = max(1, total_tasks - idx)
            task_deadline = min(deadline, now + max(2.0, remaining * 1.2 / float(tasks_left)))
            idx += 1

            eta = max(1.0, float(n) / 40.0)
            a_for_start[i], sc = _exp_grad_optimize(
                a_for_start[i],
                float(alpha),
                eta,
                200000,
                task_deadline
            )

            if sc < best_score:
                best_score = sc
                best_a = a_for_start[i].copy()
                print(
                    f"optimize n={n} start={i} alpha={alpha:.0f} "
                    f"score={best_score:.6f} elapsed={time.monotonic()-t0:.1f}s",
                    flush=True
                )

        if time.monotonic() >= deadline:
            break

    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        a2, sc2 = _exact_hill_climb(best_a, time.monotonic() + 0.4 * remaining, rng)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            print(f"hill-polish n={n} score={best_score:.6f}", flush=True)

    return best_a, best_score


def _scale_for_return(a):
    """
    Scale a normalized sequence to a safe evaluator-friendly scale.
    The objective is scale-invariant, so return a version well within
    evaluator bounds [0, 1000] and with sum > 0.01.
    """
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return [1.0, 1.0]

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size

    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    else:
        a = a / s

    target = float(np.sqrt(2.0 * float(n)))
    mx = float(np.max(a))
    if mx > 1e-300:
        scale = min(target, 999.0 / mx)
    else:
        scale = target

    out = a * scale
    out_sum = float(np.sum(out))
    if not np.isfinite(out_sum) or out_sum < 0.01:
        if out_sum > 0.0:
            out = (out / out_sum) * 0.1
        else:
            out = np.full(n, 0.1, dtype=np.float64)

    return [float(x) for x in out]


def propose_candidate(budget_s=1800, **kwargs):
    """
    Main entry point. Searches over several sequence lengths and returns
    the best nonnegative sequence found within the time budget.
    """
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1800.0

    t0 = time.monotonic()
    final_reserve = min(220.0, 0.12 * budget_s)
    final_deadline = t0 + max(60.0, budget_s - 15.0)
    work_deadline = t0 + max(60.0, budget_s - final_reserve - 10.0)

    rng = np.random.default_rng(777)

    best_a = None
    best_score = float("inf")

    # Use the known previous construction if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        ps = float(np.sum(p))
        if p.size > 0 and np.isfinite(ps) and ps > 1e-12:
            pn = p / ps
            sc = _score_sum1(pn)
            best_a = pn.copy()
            best_score = sc
            parent_n = int(p.size)
            print(f"initial parent n={parent_n} internal score={best_score:.6f}", flush=True)
        else:
            parent_n = 0
    except Exception:
        parent_n = 0

    if best_a is None:
        best_a = _normalize_positive(np.ones(1024, dtype=np.float64))
        best_score = _score_sum1(best_a)
        parent_n = 0

    # Task fractions are proportional allocations before the final polish.
    task_pairs = []
    primary = parent_n if 64 <= parent_n <= 16384 else 2048
    task_pairs.append((int(primary), 0.25))
    task_pairs.append((256, 0.10))
    task_pairs.append((1024, 0.20))
    task_pairs.append((4096, 0.30))
    task_pairs.append((8192, 0.15))

    merged = {}
    order = []
    for n, f in task_pairs:
        if n < 2:
            continue
        if n in merged:
            merged[n] = max(merged[n], f)
        else:
            merged[n] = f
            order.append(n)

    total_frac = float(sum(merged.values()))
    rem_frac = total_frac

    for n in order:
        if time.monotonic() >= work_deadline:
            break

        f = merged[n]
        remaining = work_deadline - time.monotonic()
        if remaining <= 10.0:
            break

        alloc = remaining * (float(f) / max(1e-12, rem_frac))
        rem_frac -= float(f)
        alloc = max(5.0, min(alloc, remaining))

        a, sc = _optimize_length(n, alloc, rng)
        if a is not None and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"global update n={n} score={best_score:.6f} elapsed={time.monotonic()-t0:.1f}s", flush=True)

    # Final polish with all remaining time.
    if best_a is None:
        best_a = _normalize_positive(np.ones(1024, dtype=np.float64))
        best_score = _score_sum1(best_a)

    if time.monotonic() < final_deadline:
        if len(best_a) <= 12000:
            a2, sc2 = _exact_hill_climb(best_a, final_deadline, rng)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
                print(f"final polish score={best_score:.6f} n={len(best_a)}", flush=True)

    # Optional sanity check with the supplied evaluator if it is cheap enough.
    if best_a is not None and len(best_a) <= 5000:
        out = _scale_for_return(best_a)
        if "evaluate_sequence" in globals():
            try:
                official = float(evaluate_sequence(out))
                print(f"official final check n={len(best_a)} score={official:.6f}", flush=True)
            except Exception:
                pass

    return _scale_for_return(best_a)
