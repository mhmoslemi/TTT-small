import numpy as np
import time


def fft_convolve(a):
    """Full auto-convolution using FFT. O(n log n)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def fast_score(a):
    """Compute J = 2n*max(conv)/sum^2 using FFT."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)


def lP_gradient(a, p):
    """Compute gradient of ||a*a||_p^p w.r.t. a. O(n log n)."""
    n = len(a)
    c = fft_convolve(a)
    c = np.maximum(c, 0.0)
    if p <= 1:
        weight = np.ones_like(c)
    else:
        weight = c ** (p - 1)
    b = a[::-1]
    N2 = 1
    while N2 < n + len(weight):
        N2 <<= 1
    full = np.fft.irfft(np.fft.rfft(b, N2) * np.fft.rfft(weight, N2), N2)
    grad = full[n - 1:2 * n - 1]
    return grad, c


def spectral_flatten(a, alpha, n_iter):
    """Iterative spectral flattening to reduce convolution peak."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        new_mag = mag ** alpha
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        ns = np.sum(a_new)
        if ns < 1e-10:
            break
        a = a_new / ns
    return a


def lp_descent(a, p_start, p_end, n_iter, lr):
    """Lp gradient descent with adaptive learning rate."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    for it in range(n_iter):
        p = p_start + (p_end - p_start) * it / max(1, n_iter - 1)
        grad, c = lP_gradient(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        new_a = a - lr * grad
        new_a = np.maximum(new_a, 0.0)
        ns = np.sum(new_a)
        if ns < 1e-10:
            lr *= 0.5
            continue
        new_a = new_a / ns
        new_c = fft_convolve(new_a)
        new_obj = np.max(new_c)
        old_obj = np.max(c)
        if new_obj < old_obj:
            a = new_a
            lr = min(lr * 1.03, 0.06)
        else:
            lr *= 0.7
    return a

    print(f"Best initial: {best_score:.6f} (n={len(best_seq)})", flush=True)

    # Split time budget among different initial candidates
    # Give most time to the best, some to runners-up
    scored_inits = []
    for name, seq in inits:
        try:
            sc = fast_score(seq)
            if np.isfinite(sc):
                scored_inits.append((sc, name, seq.copy()))
        except Exception:
            pass
    scored_inits.sort(key=lambda x: x[0])

    # Allocate time: top 3 get 40%, 25%, 15%, rest share 20%
    top_n = min(3, len(scored_inits))
    if top_n > 0:
        allocations = [0.40, 0.25, 0.15][:top_n]
        remaining = 1.0 - sum(allocations)
        rest = len(scored_inits) - top_n
        if rest > 0:
            per_rest = remaining / rest
            allocations.extend([per_rest] * rest)
        else:
            allocations.append(remaining)

    total_available = deadline - time.monotonic()
    t0 = time.monotonic()

    for idx, (sc_i, name, seq) in enumerate(scored_inits):
        if time.monotonic() > deadline:
            break
        frac = allocations[idx] if idx < len(allocations) else 0.02
        sub_deadline = min(deadline, t0 + total_available * sum(allocations[:idx + 1]))
        # Also enforce minimum time per candidate
        sub_deadline = max(sub_deadline, time.monotonic() + 5.0)

        print(f"Optimizing {name} (init score={sc_i:.4f}), time allocated until {sub_deadline - time.monotonic():.1f}s", flush=True)
        result_seq, result_score = try_optimize(seq, sub_deadline, rng, label=f"[{name}]")

        if result_score < best_score:
            best_score = result_score
            best_seq = result_seq.copy()
            print(f"  New global best: {best_score:.6f}", flush=True)

        # Also try optimizing from current global best
        if time.monotonic() < deadline and idx < 2:
            result2_seq, result2_score = try_optimize(best_seq.copy(), deadline, rng, label=f"[global_{idx}]")
            if result2_score < best_score:
                best_score = result2_score
                best_seq = result2_seq.copy()
                print(f"  New global best (refine): {best_score:.6f}", flush=True)

    # Final local search polish
    if time.monotonic() < deadline:
        print("Final polish phase...", flush=True)
        best_seq, best_score = try_optimize(best_seq, deadline, rng, label="[polish]")

    # Verify with evaluate_sequence
    final_score = evaluate_sequence([float(x) for x in best_seq])
    if np.isfinite(final_score):
        print(f"Final verified score: {final_score:.6f} (n={len(best_seq)})", flush=True)
    else:
        print(f"WARNING: final score not finite: {final_score}", flush=True)

    return [float(x) for x in best_seq]
