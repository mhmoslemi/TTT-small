```python
def build_conv_matrix(f):

    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)

            rows.extend(np.full(len(js), k))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
```

```python
    b_var = cp.Variable(n, nonneg=True)
    active_constraints = []
    
    # Start with the constraint at the current max position
    k_max = int(np.argmax(np.convolve(f, f)))

    b_val = None
    for cg_iter in range(max_constraints):
        # Build constraint matrix with current active constraints
        A_rows = []
        for k in active_constraints:
            j_lo = max(0, k - n + 1)
```

```python
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

def build_conv_matrix_dense(f):
    """Build dense matrix A such that (A @ b)[k] = np.convolve(f, b)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = f[k - j]
    return A

def build_conv_matrix_sparse(f):
    """Build sparse matrix A such that (A @ b)[k] = np.convolve(f, b)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        rows.extend(np.full(len(js), k).tolist())
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_solve(f, M, n, use_sparse=True):
    """Solve: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    if use_sparse:
        A = build_conv_matrix_sparse(f)
    else:
        A = build_conv_matrix_dense(f)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0.0)

def objective_normalized(a, n):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    return 2.0 * n * np.max(np.convolve(a, a))

    """Run LP-based iterations. Returns best sequence found."""

        a_new = (1 - t) * a + t * b_n
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = objective_normalized(a_new, n)
        a = a_new
        
        if score < best_score - 1e-10:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

            print(f"  Iter {it}: score={best_score:.8f}, M={np.max(np.convolve(a,a)):.10f}, t={t:.4f}")
```

**Projected gradient method:**
1. Compute g = subgradient of max_k conv(a,a)[k]
2. a = a - lr * g
3. Project onto {a >= 0, sum(a) = 1}

The projection onto {a >= 0, sum(a) = 1} can be done as follows:
- Find τ such that sum(max(a - τ, 0)) = 1
- Return max(a - τ, 0)

This can be done in O(n) time using a selection algorithm.

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 1 and np.all(v >= 0):
        # Already in the simplex (on the boundary)
        # Actually we need sum = 1, not sum <= 1
        pass
    
    # Use the sorting-based method
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
from scipy.optimize import linprog

def lp_solve_scipy(f, M, n):
    """Solve: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = build_conv_matrix_sparse(f)  # (f*b)[k] <= M
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None
```
