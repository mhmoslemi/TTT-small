"""
Iterative LP-based optimization of non-negative sequences to minimize
the peak of self-convolution, targeting score < 2.

Algorithm:
1. Initialize with several candidate sequences (uniform, given, structured).
2. Iteratively solve an LP to find a direction g that reduces the peak.
3. Blend and project, repeat.
4. Return the best sequence found within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def convolve_self(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def score_normalized(a):
    """Compute score assuming sum_a = 1."""
    c = convolve_self(a)
    return 2.0 * len(a) * np.max(c)


def solve_lp_direction(a, active_indices):
    """
    Solve LP: min t s.t. (a*g)[k] <= t for k in active_indices,
    g >= 0, sum(g) = 1.
    Returns g or None.
    """
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n
    
    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # A_ub[idx, i] = a[k-i] for i in [i_lo, i_hi]
        for i in range(i_lo, i_hi + 1):
            A_ub[idx, i] = a[k - i]
        A_ub[idx, n] = -1.0
    
    b_ub = np.ones(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-12:
                g /= s
            return g
    except Exception:
        pass
    return None


def run_lp_iteration(a, n_active=100, step_size=0.1):
    """
    Perform one LP-based iteration.
    Returns (new_a, new_score, old_score).
    """
    n = len(a)
    c = convolve_self(a)
    old_score = 2.0 * n * np.max(c)  # assuming sum_a = 1
    
    # Find active set
    k = min(n_active, 2 * n - 1)
    active = np.argsort(c)[-k:]
    
    # Solve LP
    g = solve_lp_direction(a, active)
    if g is None:
        return a, old_score, old_score
    
    # Blend
    new_a = (1.0 - step_size) * a + step_size * g
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-12:
        return a, old_score, old_score
    new_a /= s
    
    new_score = score_normalized(new_a)
    return new_a, new_score, old_score

"""
Iterative LP-based optimization to minimize peak self-convolution of non-negative sequences.
Uses active-set LP to find descent directions, multiple initializations, and adaptive steps.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def convolve_self(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_fast_len(2 * n - 1)
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def quick_score(a):
    """Fast score computation assuming sum_a=1."""
    n = len(a)
    c = convolve_self(a)
    return 2.0 * n * float(np.max(c))


def solve_lp_direction(a, active_indices):
    """
    Solve LP: min t s.t. (a*g)[k] <= t for k in active_indices, g>=0, sum(g)=1.
    Returns normalized g or None.
    """
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n

    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[idx, i] = a[k - i]
        A_ub[idx, n] = -1.0

    b_ub = np.ones(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-15:
                g /= s
            return g
        elif 'status' in res and hasattr(res, 'status') and res.status == 7:
            # Time limit exceeded, use incumbent if available
            if res.x is not None:
                g = np.maximum(res.x[:n], 0.0)
                s = g.sum()
                if s > 1e-15:
                    g /= s
                    return g
    except Exception:
        pass
    return None


def lp_iteration(a, n_active=80):
    """Perform one LP-based iteration. Returns (g, t_opt) or (None, None)."""
    n = len(a)
    c = convolve_self(a)
    k = min(n_active, 2 * n - 1)
    active = np.argsort(c)[-k:]
    g = solve_lp_direction(a, active)
    if g is None:
        return None, None
    # Compute optimal t for reference
    ag = convolve_self(a)  # not quite right, need cross-conv
    return g, None


def cross_convolve(a, g):
    """Compute cross-convolution a * g."""
    na, ng = len(a), len(g)
    size = next_fast_len(na + ng - 1)
    fa = fft(a, size)
    fg = fft(g, size)
    return ifft(fa * fg).real[:na + ng - 1]


def run_optimization(a, deadline, n_active=80, max_stagnant=30):
    """Run LP-based optimization from initial sequence a. Returns best found."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a, quick_score(a)
    a /= s

    cur_score = quick_score(a)
    best_a = a.copy()
    best_score = cur_score
    step = 0.15
    stagnant = 0
    iters = 0

    while time.monotonic() < deadline - 1.0:
        # Find active set
        c = convolve_self(a)
        k = min(n_active, 2 * len(a) - 1)
        active = np.argsort(c)[-k:]

        # Solve LP
        g = solve_lp_direction(a, active)
        if g is None:
            stagnant += 5
            if stagnant > max_stagnant:
                break
            step *= 0.5
            continue

        iters += 1

        # Try multiple step sizes, pick best
        best_step_a = None
        best_step_score = cur_score
        for t in [step, step * 0.5, step * 0.2, 0.01]:
            new_a = (1.0 - t) * a + t * g
            new_a = np.maximum(new_a, 0.0)
            ns = new_a.sum()
            if ns < 1e-12:
                continue
            new_a /= ns
            ns_score = quick_score(new_a)
            if ns_score < best_step_score:
                best_step_score = ns_score
                best_step_a = new_a

        if best_step_a is not None and best_step_score < cur_score - 1e-12:
            a = best_step_a
            cur_score = best_step_score
            stagnant = 0
            # Gradually increase step if making good progress
            step = min(0.3, step * 1.05)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            stagnant += 1
            step *= 0.7
            if stagnant > max_stagnant:
                break

    return best_a, best_score


def make_initialization(kind, n, rng):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine_dip':
        # Dip in the middle
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x / n)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == 'bimodal':
        # Two bumps
        x = np.arange(n)
        center1 = n * 0.25
        center2 = n * 0.75
        width = n * 0.15
        a = np.exp(-0.5 * ((x - center1) / width) ** 2) + np.exp(-0.5 * ((x - center2) / width) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'random_exp':
        a = rng.exponential(1.0, size=n)
        return a / a.sum()
    elif kind == 'random_gaussian_clipped':
        a = np.abs(rng.normal(1.0, 0.5, size=n))
        return a / a.sum()
    elif kind == 'sawtooth':
        x = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * x / n)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == 'center_hole':
        a = np.ones(n)
        hole_size = max(1, n // 10)
        mid = n // 2
        a[mid - hole_size // 2: mid + hole_size // 2] = 0.1
        a = np.maximum(a, 0.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequences minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses LP-based iterative optimization with multiple initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    best_seq = None
    best_score = np.inf
    all_results = []

    # Define what to try: (initialization_type, sequence_length, fraction_of_time)
    search_plan = [
        ('cosine_dip', 1024, 0.12),
        ('bimodal', 1024, 0.12),
        ('center_hole', 1024, 0.10),
        ('cosine_dip', 2048, 0.12),
        ('bimodal', 2048, 0.12),
        ('center_hole', 2048, 0.10),
        ('random_exp', 1024, 0.08),
        ('random_gaussian_clipped', 1024, 0.08),
        ('sawtooth', 2048, 0.08),
        ('cosine_dip', 4096, 0.08),
    ]

    # Also try the given sequence
    given_seq = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01 and len(parent) > 0:
            given_seq = parent / parent.sum()
            search_plan.append(('given', len(parent), 0.08))
    except NameError:
        pass

    total_frac = sum(f for _, _, f in search_plan)
    # Normalize fractions
    search_plan = [(k, n, f / total_frac * (budget_s - 5.0)) for k, n, f in search_plan]

    for init_kind, n, time_alloc in search_plan:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            break
        time_alloc = min(time_alloc, remaining - 2.0)
        if time_alloc < 2.0:
            break

        # Create initialization
        if init_kind == 'given' and given_seq is not None:
            a = given_seq.copy()
        else:
            a = make_initialization(init_kind, n, rng)

        initial_score = quick_score(a)
        print(f"Starting: {init_kind} n={n}, initial score={initial_score:.6f}", flush=True)

        if initial_score < best_score:
            best_score = initial_score
            best_seq = a.copy()

        # Run optimization
        sub_deadline = time.monotonic() + time_alloc
        result_a, result_score = run_optimization(a, sub_deadline, n_active=80, max_stagnant=40)

        if result_score < best_score:
            best_score = result_score
            best_seq = result_a.copy()
            print(f"  New best: {best_score:.6f} ({init_kind}, n={n})", flush=True)
        else:
            print(f"  Score: {result_score:.6f} (best: {best_score:.6f})", flush=True)

        # Second pass with larger active set if time remains
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5.0 and result_score < best_score + 0.5:
            sub2_deadline = time.monotonic() + min(remaining2 - 2.0, time_alloc * 0.5)
            result_a2, result_score2 = run_optimization(result_a, sub2_deadline, n_active=150, max_stagnant=50)
            if result_score2 < best_score:
                best_score = result_score2
                best_seq = result_a2.copy()
                print(f"  Refine new best: {best_score:.6f}", flush=True)

    # Final: try to improve the best sequence further with all remaining time
    remaining = deadline - time.monotonic()
    if remaining > 5.0 and best_seq is not None:
        print(f"Final refinement with {remaining:.1f}s remaining...", flush=True)
        final_a, final_score = run_optimization(best_seq, deadline - 1.0, n_active=120, max_stagnant=60)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_a.copy()

    # Verify with the actual evaluator
    if best_seq is not None:
        final_verified = float(evaluate_sequence([float(x) for x in np.maximum(best_seq, 0.0)]))
        print(f"Final verified score: {final_verified:.6f}", flush=True)
        return [float(max(0.0, x)) for x in best_seq]
    else:
        # Fallback: uniform
        a = np.ones(256) / 256
        return [float(x) for x in a]
