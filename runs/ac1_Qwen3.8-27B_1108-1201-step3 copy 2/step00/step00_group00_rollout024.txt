Gradient-based optimization with soft-max smoothing to minimize
2n * max(a*a) / (sum(a))^2. Uses L-BFGS-B with continuation in smoothing parameter.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global parameters for the objective function (to avoid closures)
_G_TAU = 100.0
_G_LAM = 1e-4

def _compute_obj_grad(a, tau, lam):
    """Compute objective and gradient for the soft-smoothed problem."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return 1e10, np.ones(n)
    
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    tau_c = tau * (c - c_max)
    # Clip for numerical stability
    tau_c = np.clip(tau_c, -500, 500)
    exp_c = np.exp(tau_c)
    sum_exp = exp_c.sum()
    if sum_exp == 0:
        return 1e10, np.ones(n)
    soft_max = (c_max + np.log(sum_exp)) / tau
    w = exp_c / sum_exp
    
    # Gradient of soft_max w.r.t. a
    grad_sm = np.correlate(w, a, mode='valid')
    
    # Objective: 2n * soft_max / S^2 + lam * (S - 1)^2
    obj = 2.0 * n * soft_max / (S * S) + lam * (S - 1.0) ** 2

    # d(lam*(S-1)^2)/d(a_j) = 2*lam*(S-1)
    grad = (2.0 * n / (S ** 3)) * (grad_sm * S - 2.0 * soft_max) + 2.0 * lam * (S - 1.0)
    
    return obj, grad
