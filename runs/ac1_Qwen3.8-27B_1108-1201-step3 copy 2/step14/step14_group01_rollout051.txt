```python
#           = ifft(conj(FFT(w)) * FFT(a))[i] ... wait, need to be careful
#           Actually: cross-correlation of w and a at lag i = sum_k w[k] * a[k+i]
#           But we want sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] (sub j=k-i)
#           = cross-correlation of a and w at lag i = sum_j a[j] * w[j+i]
#           = ifft(FFT(a) * conj(FFT(w)))[i] -- NO
#           Let me use the standard formula:
#           (a ⋆ w)[i] = sum_j a[j] * conj(w[j-i]) = for real: sum_j a[j] * w[j-i]
#           This is what we want for the first term! (with k=j, i fixed)
#           Wait: sum_k w[k] * a[k-i], let j = k-i, = sum_j w[j+i] * a[j] = (a ⋆ w)[i] (conjugate of cross-correlation)
#           For real signals: (a ⋆ w)[i] = ifft(conj(FFT(a)) * FFT(w))[i]
#           Actually, the standard definition: (a ⋆ w)[i] = sum_j a[j] * w[j+i]
#           And ifft(conj(FFT(a)) * FFT(w))[i] = sum_j a[j] * w[j+i] (circular)
#           Yes! So first term = ifft(conj(FFT(a)) * FFT(w))[:n]

# Second term: sum_k w[k] * a[i-k] = sum_j a[j] * w[i-j] (sub j=i-k)
#           = (a * w)[i] where * is convolution
#           = ifft(FFT(a) * FFT(w))[i]
```

```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = name
                print(f"  Using solver: {name}")
                return name
        except Exception:
            continue
    _solver = cp.SCS
    print(f"  Using fallback solver: SCS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)

def normalize_prob(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for m in range(n):
        r = np.arange(m, m + n)
        rows.append(r)
        cols.append(np.full(n, m))
        vals.append(a)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def lp_best_response_full(y, solver):
    n = len(y)
    size = 2 * n - 1
    conv_y = fast_conv_self(y)
    M = conv_y.max()
    if M < 1e-15:
        return np.ones(n) / n
    T = build_T_sparse(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    if val.sum() < 1e-15:
        return None
    return val

def lse_gd(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 10
    temps = np.geomspace(40.0, 0.3, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, int(remaining * 4))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            term1 = irfft(np.conj(A) * W, fft_size)[:n]
            term2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (term1 + term2)
            grad = grad - np.mean(grad)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-14:
                continue
            step = 1.0 / (grad_norm * np.sqrt(n) * 2.0)
            
            accepted = False
            for trial in range(12):
                a_new = a - step * grad
                a_new = normalize_prob(a_new)
                sc_new = fast_conv_self(a_new).max()
                if sc_new <= max_c + 0.01 * T_val * max_c:
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                continue
            
            a = a - step * grad
            a = normalize_prob(a)
            sc = fast_conv_self(a).max()
            score = 2.0 * n * sc
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and (si % 3 == 0 or si == n_stages - 1):
            print(f"    LSE s{si+1}/{n_stages} T={T_val:.2f}: {best_score:.8f}")
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    x = a.copy()
    avg = np.zeros(n)
    n_avg = 0
    stagnant = 0
    last_best = best_score
    no_improve_count = 0

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        b = lp_best_response_full(y, solver)
        if b is None:
            stagnant += 1
            if stagnant > 50:
                break
            continue
        stagnant = 0
        b = b / b.sum()
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        if t == 1:
            x_new = b.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = x_new / x_new.sum()
        
        sc_y = 2.0 * n * fast_conv_self(y_new).max() / y_new.sum()**2
        if sc_y < best_score - 1e-12:

            best_a = y_new.copy()
            no_improve_count = 0
        else:

        sc_x = 2.0 * n * fast_conv_self(x_new).max() / x_new.sum()**2
        if sc_x < best_score - 1e-12:
            best_score = sc_x
            best_a = x_new.copy()
            no_improve_count = 0
        
        y_prev = y
        y = y_new
        x = x_new
        
        avg += x_new
        n_avg += 1
        
        if n_avg % 20 == 0:
            avg_check = normalize_prob(avg)
            sc_avg = 2.0 * n * fast_conv_self(avg_check).max() / avg_check.sum()**2
            if sc_avg < best_score - 1e-12:
                best_score = sc_avg
                best_a = avg_check.copy()
                no_improve_count = 0
            avg = np.zeros(n)
            n_avg = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f} stag={stagnant}")
        
        if no_improve_count > 2000 and t > 2000:
            break
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, rng, verbose=True):
    n = len(a0)
    a = normalize_prob(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    step_size = 0.1
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv_self(a)
        max_c = conv.max()
        
        T_val = max(max_c * 0.05, 0.01)
        shifted = (conv - max_c) / T_val
        shifted = np.maximum(shifted, -50.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        term1 = irfft(np.conj(A) * W, fft_size)[:n]
        term2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (term1 + term2)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        direction = -grad / grad_norm
        best_sc = best_score
        best_a_new = a.copy()
        
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_try = a + frac * step_size * direction
            a_try = normalize_prob(a_try)
            sc_try = 2.0 * n * fast_conv_self(a_try).max()
            if sc_try < best_sc:
                best_sc = sc_try
                best_a_new = a_try
        
        if best_sc < best_score:
            a = best_a_new
            best_score = best_sc
            best_a = a.copy()
            step_size = min(step_size * 1.1, 1.0)
        else:
            step_size *= 0.9
            if step_size < 0.001:
                break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_prob(f_interp)

        try_update(a_out, "lp2")
    
    # Phase 5: Multi-start exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P5: Multi-start (budget={remaining:.0f}s) ===")
    if remaining > 100:
        n_starts = min(5, int(remaining // 60))
        for s in range(n_starts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_s = min(remaining * 0.3, 150.0)
            # Different perturbation strategies
            if s == 0:
                init_s = soft_perturb(best_seq, rng, strength=0.1)
            elif s == 1:
                init_s = soft_perturb(best_seq, rng, strength=0.2)
            elif s == 2:
                init_s = structured_init(n_base, rng)
            elif s == 3:
                # Antisymmetric perturbation
                r = rng.standard_normal(n_base)
                asym = r - r[::-1]
                init_s = normalize_prob(best_seq + 0.1 * asym)
            else:
                init_s = structured_init(n_base, rng)
            
            a_out, _ = nesterov_lp_optimize(init_s, t_s, solver, verbose=False)
            try_update(a_out, f"ms_{s}")
    
    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.3, 200.0)
    print(f"\n=== P6: Pair CD (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_polish(best_seq, t6, rng, verbose=False)
        try_update(a_out, "pcd")
    
    # Phase 7: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P7: Length exploration (budget={remaining:.0f}s) ===")
    if remaining > 120:
        for n_try in [n_base - 200, n_base + 200, n_base + 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            if n_try < 100:
                continue
            t_l = min(remaining * 0.35, 200.0)
            init_l = interpolate_seq(best_seq, n_try)
            a_out, _ = nesterov_lp_optimize(init_l, t_l, solver, verbose=False)
            try_update(a_out, f"len_{n_try}")
    
    # Phase 8: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.6, 300.0)
    print(f"\n=== P8: Final LP polish (budget={t8:.0f}s) ===")
    if t8 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t8, solver, verbose=True)
        try_update(a_out, "final_lp")
    
    # Phase 9: Final subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_out, _ = subgradient_polish(best_seq, remaining * 0.7, rng, verbose=False)
        try_update(a_out, "final_sg")
    
    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
```
