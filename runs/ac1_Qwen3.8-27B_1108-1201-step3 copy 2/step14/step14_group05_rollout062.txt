   f. Normalize: a_i = exp(log_a_i) / sum(exp(log_a))

    """Compute gradient of the k-th autocorrelation coefficient w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # c_k = sum_j a[j] * a[j+k] (for valid j)
    # d(c_k)/d(a[i]) = a[i+k] (if i+k < n) + a[i-k] (if i >= k)
    if k == 0:
        g = 2.0 * a
    else:
        # First term: a[i+k] for i = 0, ..., n-1-k
        g[:n-k] += a[k:n]
        # Second term: a[i-k] for i = k, ..., n-1
        g[k:] += a[:n-k]
    return g

def smooth_gradient(a, T, n_top=20):

    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    
    # Softmax weights
    log_w = (c - max_c) / T
    # Clip to avoid underflow
    log_w = np.maximum(log_w, -50.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback to hard max
        k_star = int(np.argmax(c))
        return compute_autocorr_gradient(a, k_star)
    w = w / w_sum
    
    # Only use top n_top weights for efficiency
    if n_top < len(w):
        top_idx = np.argsort(w)[::-1][:n_top]
        w = w.copy()
        w_keep = w[top_idx].sum()
        w = np.zeros_like(w)
        w[top_idx] = w[top_idx] / w_keep

    grad = np.zeros(n)
    for k_idx in np.where(w > 1e-8)[0]:
        grad += w[k_idx] * compute_autocorr_gradient(a, k_idx)
    
    return grad

def seeg_optimize(a0, time_limit, verbose=True, T_init=None, T_end=None):
    """Smoothed Exponentiated Gradient optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    if T_init is None:
        T_init = max(0.01, best_score * 0.1)
    if T_end is None:
        T_end = max(0.001, best_score * 0.005)
    
    t0 = time.monotonic()
    steps = 0
    eta = 0.1
    log_a = np.log(np.maximum(a, 1e-30))
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature
        T = T_init * (T_end / T_init) ** progress
        # Anneal step size
        eta = 0.1 / (1.0 + 0.5 * steps / n)
        
        c = fast_autocorr(a)
        max_c = c.max()
        
        # Compute smoothed gradient
        n_top = min(50, 2*n-1)
        # Use top peaks
        log_w = (c - max_c) / T
        log_w = np.maximum(log_w, -30.0)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            k_star = int(np.argmax(c))
            grad = compute_autocorr_gradient(a, k_star)
        else:
            w = w / w_sum
            # Only significant weights
            sig = w > 1e-6
            if sig.sum() < n_top:
                top_idx = np.argsort(w)[::-1][:n_top]
                sig = np.zeros_like(w, dtype=bool)
                sig[top_idx] = True
            else:
                top_idx = np.where(sig)[0]
            
            grad = np.zeros(n)
            for k_idx in top_idx:
                if w[k_idx] < 1e-8:
                    continue
                grad += w[k_idx] * compute_autocorr_gradient(a, k_idx)
        
        # Center gradient (maintain sum in log-space)
        grad -= np.mean(grad)
        
        # Exponentiated gradient update in log-space
        log_a = log_a - eta * grad
        # Periodic renormalization to prevent overflow
        log_a -= np.max(log_a)
        
        # Convert back and normalize
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        steps += 1
        if verbose and steps % 10000 == 0:
            print(f"    SEEG step={steps} ({elapsed:.0f}s): T={T:.4f}, eta={eta:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()

    best_M = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    steps = 0
    
    # Hyperparameters
    T_init = best_M * 0.05
    T_min = best_M * 0.001
    eta_init = 0.5
    
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= np.max(log_a)  # numerical stability
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal
        T = T_init * (T_min / T_init) ** progress
        eta = eta_init * 0.99999 ** steps  # slow decay
        
        # Compute autocorrelation
        c = fast_autocorr(a)
        max_c = c.max()
        
        # Softmax weights
        log_w = (c - max_c) / T
        log_w = np.maximum(log_w, -50)
        w = np.exp(log_w)
        w /= w.sum()
        
        # Compute gradient via FFT
        grad = weighted_autocorr_gradient(a, w)
        
        # Center gradient for simplex constraint
        grad -= np.mean(grad)
        
        # Update in log-space
        log_a -= eta * grad
        log_a -= np.max(log_a)  # rescale
        
        # Convert to a
        a = np.exp(log_a)
        a /= a.sum()
        
        # Track best
        if max_c < best_M:
            best_M = max_c
            best_a = a.copy()
        
        steps += 1
        if verbose and steps % 5000 == 0:
            print(f"    SEEG step={steps} ({elapsed:.0f}s): T={T:.4f}, eta={eta:.4f}, best_M={best_M:.8f}")
    
    return best_a, best_M

    Solve max sum(b) s.t. T@b <= M, b >= 0
    using active set / cutting plane method.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Start with the top constraint
    c = fast_autocorr(a)
    active = np.zeros(size, dtype=bool)
    p_star = int(np.argmax(c))
    active[p_star] = True
    
    b = None
    for it in range(max_iters):
        # Solve LP with current active set
        # max sum(b) s.t. T[active,:] @ b <= M[active], b >= 0
        # Use cvxpy
        b_var = cp.Variable(n, nonneg=True)
        constraints = [build_conv_row(a, k) @ b_var <= M for k in np.where(active)[0]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        
        # Check for violated constraints
        c_b = fast_autocorr_conv(a, b)  # T @ b

            break  # Feasible!

        new_k = int(np.argmax(c_b))
        active[new_k] = True
    
    if b is None or b.sum() < 1e-15:
        return None
    return b

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def weighted_autocorr_gradient(a, w):
    """
    Compute grad_i = 2 * sum_j a[j] * w[i+j] * 1(i+j < 2n-1)
    using FFT cross-correlation. This is the gradient of the weighted sum
    of autocorrelation peaks with weights w.
    """
    n = len(a)
    w_len = len(w)
    fft_size = 1 << (n + w_len - 1).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    corr = irfft(np.conj(W) * A, fft_size)
    return 2.0 * corr[0:n]

def seeg_optimize(a0, time_limit, verbose=True, T_factor_init=0.05, T_factor_end=0.003):
    """
    Smoothed Exponentiated Gradient with Nesterov acceleration in log-space.
    Uses FFT-based O(n log n) gradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    t0 = time.monotonic()
    steps = 0

    T_init = best_M * T_factor_init
    T_end = best_M * T_factor_end
    eta_base = 1.0

    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= np.max(log_a)
    prev_log_a = log_a.copy()
    t_nest = 1

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        T = T_init * (T_end / T_init) ** progress
        eta = eta_base / (1.0 + 0.1 * steps / max(n, 1))

        # Compute autocorrelation and softmax weights
        c = fast_autocorr(a)
        max_c = c.max()

        log_w = (c - max_c) / T
        log_w = np.maximum(log_w, -40.0)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            k_star = int(np.argmax(c))
            w = np.zeros(len(c))
            w[k_star] = 1.0
        else:
            w = w / w_sum

        # Compute gradient via FFT
        grad = weighted_autocorr_gradient(a, w)

        # Work in log-space with Nesterov
        t_nest += 1
        momentum = (t_nest - 1.0) / (t_nest + 1.5)
        log_a_new = log_a + momentum * (log_a - prev_log_a)
        prev_log_a = log_a.copy()

        # Center gradient (maintains sum constraint)
        grad_centered = grad - np.mean(grad)

        # Step in log-space
        log_a_new = log_a_new - eta * grad_centered
        log_a_new -= np.max(log_a_new)

        # Convert back
        a = np.exp(log_a_new)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            log_a = np.log(a)
        else:
            a = a / s
            log_a = np.log(np.maximum(a, 1e-30))
            log_a -= np.max(log_a)

        steps += 1

        if max_c < best_M:
            best_M = max_c
            best_a = a.copy()

        if verbose and steps % 20000 == 0:
            print(f"    SEEG step={steps} ({elapsed:.0f}s): T={T:.4f}, eta={eta:.4f}, best_M={best_M:.8f}")

    return best_a, best_M


def lp_best_response_cutting(a, M, solver, max_constr=30, verbose=False):
    """
    Solve max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    using cutting plane / active set method.
    Exploits Toeplitz structure: T@b = conv(a,b) computed via FFT.
    """
    n = len(a)
    size = 2 * n - 1

    # Start with top few constraints
    c = fast_autocorr(a)
    top_k = min(3, size)
    active_set = set(np.argsort(c)[::-1][:top_k].tolist())

    b = None
    for it in range(50):
        active_list = sorted(active_set)
        if len(active_list) > max_constr:
            # Keep only top max_constr
            active_vals = c[active_list]
            keep = np.argsort(active_vals)[::-1][:max_constr]
            active_set = set(active_list[i] for i in keep)
            active_list = sorted(active_set)

        # Build constraint matrix rows
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_list:
            # Row k of T: T[k, m] = a[k-m] if 0<=k-m<n, else 0
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                row = np.zeros(n)
                row[m_start:m_end + 1] = a[k - np.arange(m_start, m_end + 1)]
                constraints.append(row @ b_var <= M)
            else:
                continue

        if not constraints:
            b = np.ones(n) * M
            break

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break

        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

        # Check feasibility via FFT
        conv_ab_val = conv_ab(a, b)
        violated = conv_ab_val > M * (1 + 1e-7)
        if not violated.any():
            break

        # Add most violated constraint
        if len(active_set) < max_constr:
            new_k = int(np.argmax(conv_ab_val))
            active_set.add(new_k)
        else:
            break

    if b is None or b.sum() < 1e-15:
        return None
    return b


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_constr=25):
    """Fessler-Parks accelerated LP Nesterov with cutting-plane LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.7

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()

        b = lp_best_response_cutting(y, M_curr, solver, max_constr=max_constr)
        if b is None:
            n_iters += 1
            continue
        if b.sum() < 1e-15:
            n_iters += 1
            continue

        g = b * norm / b.sum()
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 100:
            gamma = max(1.0, gamma - 0.1)
            if gamma <= 1.0:
                gamma = 1.5
                stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}, stall={stall}")

        # Restart if stuck
        if stall > 200:
            rng = np.random.default_rng(n_iters * 7 + 3)
            pert = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.7

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
