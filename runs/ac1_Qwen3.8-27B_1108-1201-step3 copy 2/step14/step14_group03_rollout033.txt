- Then, scale to have the right sum.
(These two operations don't commute perfectly, but for small steps it's approximately fine.)

Alternatively, we can parameterize a = softmax(z) * sqrt(2n) and do gradient descent in z-space. This automatically satisfies the non-negativity and sum constraints.

Let me use the softmax parameterization:
- a_i = sqrt(2n) * exp(z_i) / sum_j exp(z_j)
- da_i/dz_j = a_i * (delta_{ij} - a_j / sum(a)) = a_i * delta_{ij} - a_i * a_j / sqrt(2n)

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_gradient_descent(a0, time_limit, beta_schedule, lr=None, verbose=True):
    """
    Minimize max(conv(a,a)) using LSE-smoothed gradient descent with 
    softmax parameterization.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize z = log(a) for softmax parameterization
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe / a_safe.sum())  # z = log(softmax params)
    
    n_iters = 0
    stuck_count = 0
    prev_score = best_score
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Get current beta
        if t < len(beta_schedule):
            beta = beta_schedule[t]
        else:
            beta = beta_schedule[-1]
        
        # Compute convolution
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # Compute softmax weights p
        conv_shifted = conv - max_conv  # numerical stability
        p = np.exp(beta * conv_shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(size)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum
        
        # Compute gradient: grad[i] = sum_k p[k] * (a[k-i] + a[i-k])
        # g1[i] = sum_k p[k] * a[k-i] (for valid k-i)
        # g2[i] = sum_k p[k] * a[i-k] (for valid i-k)
        
        # Using FFT:
        # g1 = ifft(fft(p, fft_size) * conj(fft(a, fft_size)))[:n]
        P = rfft(p, fft_size)
        A = rfft(a, fft_size)
        g1 = np.zeros(n)
        g1_full = irfft(P * np.conj(A), fft_size)
        g1[:n] = g1_full[:n]
        
        # g2[i] = sum_j a[j] * p[i-j] = conv(a, p)[i]
        g2_full = irfft(A * P, fft_size)
        g2 = g2_full[:n]
        
        grad = g1 + g2
        
        # Gradient step in z-space (softmax parameterization)
        # dF/dz_i = a_i * (grad[i] - <grad, a>/norm)
        grad_dot_a = np.dot(grad, a)
        z_grad = a * (grad - grad_dot_a / norm)
        
        # Adaptive learning rate
        if lr is None:
            lr_t = norm / (beta * (1.0 + 0.1 * t) * (1.0 + t / 100.0))
        else:
            lr_t = lr
        
        # Update z
        z = z - lr_t * z_grad
        
        # Recover a from z
        z_max = z.max()
        a_new = np.exp(z - z_max)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new * norm / a_new_sum
        
        a = a_new
        n_iters += 1
        
        # Check score
        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter={n_iters} ({elapsed:.0f}s): beta={beta:.1f}, best={best_score:.8f}")
    
    return best_a, best_score

def lse_optimize(a0, time_limit, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    
    # Beta schedule: start low, increase over time
    n_iters = 0
    warmup = 200
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        beta = min(50.0, 0.5 * t)  # increase beta over time
        
        # Compute convolution
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # Softmax weights
        shifted = conv - max_conv
        p = np.exp(beta * shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:

        A = rfft(a, fft_size)
        g1 = irfft(P * np.conj(A), fft_size)[:n]

        # Multiplicative update
        # Scale gradient for numerical stability
        max_grad = grad.max()
        if max_grad < 1e-30:
            break

        lr = norm / (max_grad * (1.0 + 0.5 * t / 1000.0))

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE Mirror Descent with Nesterov (primary, 45% time) - O(n log n) per iter
2. LP Best-Response with Nesterov (refinement, 30% time)
3. Random-pair coordinate descent (polish, 15% time)
4. Final LP refinement (10% time)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_mirror_descent(a0, time_limit, verbose=True):
    """LSE-smoothed mirror descent with adaptive parameters."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    
    n_iters = 0
    no_improve = 0
    
    # Nesterov momentum
    y = a.copy()  # momentum variable
    a_mom = a.copy()  # Nesterov lookahead
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: start moderate, increase
        beta = 5.0 + 3.0 * t  # linear increase
        beta = min(beta, 100.0)
        
        # Compute convolution at lookahead point
        conv = fast_autocorr(y)
        max_conv = conv.max()
        
        # Softmax weights (temperature-controlled)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        p_sum = exp_vals.sum()
        if p_sum < 1e-30:
            p = np.zeros(size)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / p_sum
        
        # Gradient: grad = 2 * cross_correlation(p, y)[:n]
        P = rfft(p, fft_size)
        Y = rfft(y, fft_size)
        g1 = irfft(P * np.conj(Y), fft_size)[:n]
        grad = 2.0 * g1
        
        # Center the gradient (remove mean for simplex constraint)
        # The "effective" gradient in the mirror descent sense
        max_grad = grad.max()
        if max_grad < 1e-30:
            break
        
        # Step size: eta such that eta * max_grad ~ 1 (one "full" step)
        # Decrease over time for stability
        eta = 1.0 / (1.0 + 0.3 * t / 500.0)
        eta = min(eta, 1.0)
        
        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-eta * grad / max_grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        
        # Nesterov update for y
        alpha = 2.0 / (t + 2.0)
        y = (1.0 - alpha) * y + alpha * a_new
        
        a = a_new
        n_iters += 1
        
        # Check improvement (every 20 iterations to save time)
        if n_iters % 20 == 0:
            score = fast_autocorr(a).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if verbose and n_iters % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-MD iter={n_iters} ({elapsed:.0f}s): beta={beta:.0f}, best={best_score:.10f}")
    
    # Final check
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    
    # Build the constraint matrix T where (T @ b)[k] = conv(a,b)[k]
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    n_iters = 0
    stuck_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue
        
        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check both y and a
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def random_pair_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent with random pair (i,j) moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    conv = fast_autocorr(a)
    
    no_improve = 0
    n_moves = 0
    
    while time.monotonic() - t0 < time_limit:
        # Pick random pair (i, j)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Compute the delta vector for moving mass from i to j
        # delta_vec[k] = change in conv[k] per unit mass moved from i to j
        # Moving d from i to j: a[i] -= d, a[j] += d
        # Change in conv[k]:
        #   -d * (a[k-i] + a[i-k]) + d * (a[k-j] + a[j-k]) + second order
