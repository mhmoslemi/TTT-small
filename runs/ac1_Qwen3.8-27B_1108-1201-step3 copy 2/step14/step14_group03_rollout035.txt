def build_conv_matrix(a):
    """Build the matrix T such that (T @ b)_k = (a*b)_k for full convolution."""
    n = len(a)

    T = np.zeros((size, n))

def lp_best_response(a, M, solver):

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a, n):
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = min(size, j + n)
        T[lo:hi, j] = a[:hi - lo]
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, n)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    restart_count = 0
    last_restart_score = best_score
    no_improve = 0
    t = 1

    while True:
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            no_improve += 1
            if no_improve > 100:
                break
            t += 1
            continue

        no_improve = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            t += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        t += 1

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    no_improve = 0

        if best_score < last_restart_score * 0.995 or (t > 300 and no_improve > 150):
            y = best_a.copy()
            a = best_a.copy()
            t = 1
            no_improve = 0
            restart_count += 1
            last_restart_score = best_score
            if verbose and restart_count <= 15:
                elapsed = time.monotonic() - t0
                print(f"    RL restart #{restart_count} ({elapsed:.0f}s): score={best_score:.8f}")

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    RL iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n).astype(np.float64)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n).astype(np.float64)
    return f_interp / ss


def lse_grad_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, n)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_stages = 10
    temps = np.geomspace(8.0, 0.05, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(50, min(800, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            grad = np.zeros(n)
            for i in range(n):
                idx = np.arange(i, i + n)
                valid = idx < size
                if valid.all():
                    grad[i] = 2.0 * np.dot(w[:n], a)
                    # Actually need w[i:i+n]
                    grad[i] = 2.0 * np.dot(w[i:i + n], a)
                else:
                    grad[i] = 2.0 * np.dot(w[idx[valid]], a[:valid.sum()])

            max_grad = np.max(np.abs(grad))
            if max_grad < 1e-15:
                break
            step = 0.3 / max_grad
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, n)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        top_peaks = np.argsort(conv)[-8:]
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            for k_star in top_peaks:
                k_star = int(k_star)
                if a[i] < norm * 1e-12:
                    break

                ki = k_star - i
                if ki < 0 or ki >= n:
                    continue
                a_ki = a[ki]
                if a_ki < 1e-15:
                    continue

                j_lo = max(0, k_star - n + 1)
                j_hi = min(n - 1, k_star)
                if j_lo > j_hi:
                    continue

                j_range = np.arange(j_lo, j_hi + 1)
                j_range = j_range[j_range != i]
                if len(j_range) == 0:
                    continue

                k_j_vals = k_star - j_range
                a_kj = a[k_j_vals]
                best_j_idx = np.argmin(a_kj)
                j = int(j_range[best_j_idx])

                max_d = a[i] * 0.8
                if max_d < 1e-15:
                    continue

                n_samples = 12
                deltas = np.linspace(0.0, max_d, n_samples)
                best_d = 0.0
                best_local_max = conv.max()

                for d in deltas:
                    if d <= 0:
                        continue
                    new_max_est = best_local_max
                    for k_check in [k_star, min(size - 1, k_star + j - i), min(size - 1, 2 * i), min(size - 1, 2 * j), min(size - 1, i + j)]:
                        if k_check < 0:
                            continue
                        # First-order: 2*d*(a[k-j] - a[k-i])
                        val = conv[k_check]
                        if 0 <= k_check - i < n:
                            val += 2.0 * d * (-a[k_check - i])
                        if 0 <= k_check - j < n:
                            val += 2.0 * d * a[k_check - j]
                        if 2 * i == k_check:
                            val += d * d
                        if 2 * j == k_check:
                            val -= d * d
                        if i + j == k_check:
                            val += 2.0 * d * d - 2.0 * d * d
                        new_max_est = max(new_max_est, val)

                    if new_max_est < best_local_max - 1e-14:
                        best_local_max = new_max_est
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgrad_nesterov(a0, time_limit, verbose=True):

            if s_a < 1e-15:

        score_y = fast_autocorr(y).max()

        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: LP Nesterov on n=2000 (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_2 = 2000
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: LP Nesterov n={n_2} (budget={t2:.0f}s) ===")
    if t2 > 40:
        init_2 = interpolate_sequence(best_seq, n_2)
        a_norm2 = init_2 * np.sqrt(2.0 * n_2) / init_2.sum()
        a_out2, _ = nesterov_lp(a_norm2, t2, solver, verbose=True)
        do_try_update(a_out2, f"lp_n{n_2}")

    # Phase 3: LP Nesterov on n=2400 (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_3 = 2400
    t3 = remaining * 0.12
    print(f"\n=== Phase 3: LP Nesterov n={n_3} (budget={t3:.0f}s) ===")
    if t3 > 40:
        init_3 = interpolate_sequence(best_seq, n_3)
        a_norm3 = init_3 * np.sqrt(2.0 * n_3) / init_3.sum()
        a_out3, _ = nesterov_lp(a_norm3, t3, solver, verbose=True)
        do_try_update(a_out3, f"lp_n{n_3}")

    # Phase 4: LSE gradient descent on best (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LSE GD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm4 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out4, _ = lse_grad_descent(a_norm4, t4, verbose=True)
        do_try_update(a_out4, f"lse_n{n_best}")

    # Phase 5: Subgradient Nesterov on best (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Subgrad Nesterov n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm5 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out5, _ = subgrad_nesterov(a_norm5, t5, verbose=True)
        do_try_update(a_out5, f"sg_n{n_best}")

    # Phase 6: Peak-focused CD on best (25% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.50
    print(f"\n=== Phase 6: Peak CD n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm6 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out6, _ = peak_cd(a_norm6, t6, rng, verbose=True)
        do_try_update(a_out6, f"pcd_n{n_best}")

    # Phase 7: Final LP Nesterov polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = remaining * 0.70
    print(f"\n=== Phase 7: Final LP Nesterov n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm7 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out7, _ = nesterov_lp(a_norm7, t7, solver, verbose=True)
        do_try_update(a_out7, f"lp_final_n{n_best}")

    # Phase 8: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = remaining * 0.80
    print(f"\n=== Phase 8: Final CD n={n_best} (budget={t8:.0f}s) ===")
    if t8 > 5:
        a_norm8 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out8, _ = peak_cd(a_norm8, t8, rng, verbose=False)
        do_try_update(a_out8, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
