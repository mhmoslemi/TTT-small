_global_solver = None

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_sum(a, target_sum):
    """Normalize a to have the target sum, keeping non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s


def lse_nesterov(a0, time_limit, verbose=True):
    """Nesterov-accelerated LogSumExp gradient descent."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sum(a0, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.time()
    step = 0
    
    # Temperature schedule
    b = fast_autocorr(a)
    T_start = max(0.1, best_score * 0.3)
    T_end = max(0.005, best_score * 0.01)
    
    # Nesterov state
    y = a.copy()
    prev_y = a.copy()
    
    # Adaptive step size
    lr = 1.0 / n
    
    while time.time() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.time() - t0) / time_limit)
        
        # Current temperature
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        b = fast_autocorr(y)
        max_b = b.max()
        
        if max_b < 1e-15:
            break
        
        # LSE weights
        scaled_b = (b - max_b) / T
        # Clip to avoid overflow
        scaled_b = np.clip(scaled_b, -50, 50)
        w = np.exp(scaled_b)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # This is 2 * corr(w, a) at appropriate offsets
        # conv(w, a_rev)[n-1 : 2n-1] gives what we want
        a_rev = a[::-1]
        fft_size = 1 << (2 * n).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_result = irfft(W * AR, fft_size)
        grad = 2.0 * conv_result[n - 1:2 * n - 1]
        
        # Project gradient to tangent space (sum a = norm_val)
        grad = grad - np.mean(grad)
        
        # Gradient norm for step size adaptation
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        # Use backtracking if needed
        alpha = lr
        for _ in range(5):
            y_new = y - alpha * grad
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-15:
                alpha *= 0.5
                continue
            y_new = y_new * norm_val / s_new
            
            b_new = fast_autocorr(y_new)
            if b_new.max() <= max_b * (1 + 1e-8) + 1e-10:
                break
            alpha *= 0.5
        
        if alpha < 1e-15:
            alpha = lr * 0.5
        
        # Nesterov momentum
        t = step + 1
        momentum = (t - 1 + np.sqrt(1 + 4 * (t - 1)**2)) / (2 * t)
        # Standard Nesterov: momentum = (t-1+0.5)/(t+0.5)
        momentum = (t - 0.5) / (t + 0.5)
        
        a_new = y_new + momentum * (y_new - a)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm_val / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        step += 1
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Adaptive learning rate
        if step % 100 == 0:
            recent = score_a if score_a < score_y else score_y
            if recent < best_score * 1.001:
                lr = min(lr * 1.1, 10.0 / n)
            else:
                lr = max(lr * 0.8, 0.01 / n)
        
        # Restart if stuck
        if step % 500 == 0 and score_y > best_score * 1.01:
            rng = np.random.default_rng(step)
            y = np.maximum(y + rng.normal(0, 0.02 * norm_val / n, n), 0)
            y = y * norm_val / y.sum()
            a = y.copy()
            prev_y = y.copy()
            momentum = 1.0
        
        if verbose and step % 2000 == 0:
            elapsed = time.time() - t0
            print(f"    LSE-Nest step={step} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4f}, lr={lr:.4f}")
    
    return best_a, best_score


def fast_lp_best_response(a, M, solver, max_constraints=200):
    """
    Solve the LP best-response: max sum(b) s.t. T@a <= M, b >= 0.
    Uses constraint reduction based on violation magnitude.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Compute current constraint values
    # T @ a = autocorrelation of a (which we already have)
    # Actually T @ b for the NEW variable b, with a FIXED
    # (T b)[k] = sum_j a[k-j] * b[j] (valid indices)
    
    # Build the convolution matrix implicitly via FFT
    # For the LP, we need T as a matrix. For large n, this is expensive.
    # Instead, use a different approach: solve the LP with a custom constraint evaluation.
    
    # For practical purposes, build the matrix for the active constraints only.
    
    # First, find which constraints are most likely active
    # The constraint k is: sum_j a[k-j] * b[j] <= M
    # If a is concentrated, only constraints near the "center" of a's support matter
    
    n_var = cp.Variable(n, nonneg=True)
    
    # Start with all constraints (might be slow for large n)
    # Build T matrix
    T = np.zeros((size, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(size - 1, j + n - 1)
        # T[k, j] = a[k - j] if 0 <= k-j < n
        ks = range(max(0, j), min(size, j + n))
        for k in ks:
            T[k, j] = a[k - j]
    
    # Select top constraints by violation potential
    # Use the current best response direction to identify active set
    # Initial: use all constraints if size <= 4000, else reduce
    
    if size <= max_constraints * 3:
        constraints = [T @ n_var <= M, n_var >= 0]
    else:
        # Heuristic: keep constraints where a has significant energy nearby
        a_energy = np.zeros(size)
        for j in range(n):
            for k in range(max(0, j), min(size, j + n)):
                a_energy[k] += a[j] ** 2
        
        top_idx = np.argsort(a_energy)[::-1][:max_constraints]
        constraints = [T[top_idx, :] @ n_var <= M[top_idx], n_var >= 0]
    
    prob = cp.Problem(cp.Maximize(cp.sum(n_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if n_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(n_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    # Verify and refine: check if any removed constraints are violated
    # Compute T @ b_val for all k
    check = np.zeros(size)
    for j in range(n):
        if b_val[j] > 1e-10:
            check[j:j+n] += b_val[j] * a[j: j + (size - j)]  # This is wrong
    
    # Actually let's compute the convolution properly
    # (a * b)[k] = sum_j a[j] * b[k-j]
    # This is the same as convolve(a, b)
    full_conv = np.convolve(a, b_val)  # This is O(n^2) but n=1600 so ~2.5M ops
    
    if full_conv.max() > M * 1.01:
        # Need to add violated constraints and re-solve
        violated = full_conv > M * 0.99
        if violated.sum() < max_constraints:
            top_v = np.where(violated)[0]
            constraints = [T[top_v, :] @ n_var <= M[top_v], n_var >= 0]
            prob = cp.Problem(cp.Maximize(cp.sum(n_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate') and n_var.value is not None:
                    b_val = np.maximum(np.asarray(n_var.value, dtype=np.float64), 0.0)
            except Exception:
                pass
    
    return b_val
