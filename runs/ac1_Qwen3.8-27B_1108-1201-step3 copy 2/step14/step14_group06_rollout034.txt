            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        if use_tight_M:
            progress = min(1.0, t / 500.0)
            eps = eps_start + (eps_max - eps_start) * progress
            eps = min(eps, 0.002 * np.sqrt(t + 1))
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            M_fallback = M_curr * (1.0 - eps * 0.5)
            b = lp_best_response(y, M_fallback, solver, T)
            if b is None:
                b = lp_best_response(y, M_curr, solver, T)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.1), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 90.0:
            stall_count += 1
            if stall_count > 3:
                if restart_count < 3:
                    restart_count += 1
                    rng_r = np.random.default_rng(t)
                    perturb = rng_r.normal(0, 0.05 * norm_val / n, n)
                    y = y + perturb
                    y = np.maximum(y, 0.0)
                    s_y = y.sum()
                    if s_y > 1e-15:
                        y = y * norm_val / s_y
                    a = y.copy()
                    stall_count = 0
                    last_improve = time.monotonic()
                    if verbose:
                        print(f"    Restart #{restart_count} at iter={n_iters}")
                else:
                    if verbose:
                        print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                    break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eps={eps:.4f}")

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=6, lr=0.05):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 200.0)
        m = np.zeros(n)
        v = np.zeros(n)

        steps_in_stage = 0
        while time.monotonic() - t0 < t0 + time_limit:
            if time.monotonic() - t0 > time_limit:
                break
            stage_elapsed = time.monotonic() - (t0 + sum(temps[:si] * 0) + si * 0)
            if si > 0:
                pass

            obj_val, grad = lse_objective_and_grad(a, n, T)
            adam_t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** adam_t)
            v_hat = v / (1 - beta2 ** adam_t)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new * norm_val / s_new
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
            steps_in_stage += 1

            if steps_in_stage % 500 == 0:
                stage_t = time.monotonic() - t0
                if stage_t > stage_time:
                    break

        if time.monotonic() - t0 > time_limit:
            break
        print(f"    LSE-Adam T={T:.1f}: score={best_score:.8f} ({steps_in_stage} steps)")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False

        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            for _ in range(50):
                i = rng.integers(n)
                if a[i] < 1e-10:
                    continue
                j1 = rng.integers(n)
                j2 = rng.integers(n)
                if j1 == i or j2 == i or j1 == j2:
                    continue
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j1] += amt * 0.6
                a_trial[j2] += amt * 0.4
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            i = rng.integers(1, n - 1)
            vals = a[i-1:i+2].copy()
            s = vals.sum()
            if s > 1e-10:
                smoothed = np.array([0.3, 0.4, 0.3]) * s
                a_trial = a.copy()
                a_trial[i-1:i+2] = smoothed
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score
