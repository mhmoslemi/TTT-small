    candidates = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Fallback LP solver: SCS")
    return _solver_cache


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.maximum(np.asarray(a, dtype=np.float64), 0.0)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_active_constraints(a, active_idx):
    n = len(a)
    size = 2 * n - 1
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_max_sum(a, M, solver, active_idx=None):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_min_max(a, solver, active_idx=None, normalize_sum=1.0):
    n = len(a)
    size = 2 * n - 1
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if active_idx is None or len(active_idx) >= size:
        T = build_active_constraints(a, np.arange(size))
    else:
        T = build_active_constraints(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, None
    return val, t_val


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, alpha_min=0.1, beta_blend=0.3):
    """Nesterov-accelerated LP iteration with dual LP blending."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_reset = 80

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 50 == 0:
            tol = 0.03 + 0.02 * np.exp(-t / 300.0)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 8:
                active_set = np.where(conv_y >= M_curr * 0.7)[0]
            if len(active_set) > 0.9 * size:
                active_set = np.where(conv_y >= M_curr * 0.9)[0]

        # Max-sum LP
        b_max = lp_max_sum(y, M_curr, solver, active_set)

        # Min-max LP (for blending)
        b_min = None
        if t % 3 == 0:
            b_min, _ = lp_min_max(y, solver, active_set, normalize_sum=norm)

        if b_max is None:
            if t % 20 == 0:
                active_set = None
                b_max = lp_max_sum(y, M_curr, solver, None)
            if b_max is None:
                continue

        b_sum = b_max.sum()
        g = b_max * norm / b_sum

        if b_min is not None and b_min.sum() > 1e-10:
            g_min_norm = b_min * norm / b_min.sum()
            g = (1.0 - beta_blend) * g + beta_blend * g_min_norm

        # Verify no constraint violations (expand active set if needed)
        if len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_check = build_active_constraints(y, inactive)
                viol = T_check @ g
                viol_idx = np.where(viol > M_curr * (1.0 + 1e-6))[0]
                if len(viol_idx) > 0:
                    active_set = np.union1d(active_set, inactive[viol_idx])

        # Nesterov update
        alpha_t = max(2.0 / (t + 2.0), alpha_min)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.93)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_reset:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=True):
    """Individual coordinate descent with golden section search."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    phi = 0.6180339887498949

    def obj_val(a_vec):
        c = fast_autocorr(a_vec)
        s = a_vec.sum()
        return 2.0 * n * c.max() / (s * s)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            # Golden section search for optimal a[i]
            # Keep sum fixed: when we change a[i] to a[i]+delta, we renormalize
            # So effectively we're optimizing a[i] relative to others
            lo = 0.0
            hi = ai * 3.0 + norm * 0.1

            # Sample to bracket
            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)

            a_test1 = a.copy()
            a_test1[i] = c1
            a_test1 = a_test1 * norm / a_test1.sum()
            f1 = evaluate_local(a_test1)

            a_test2 = a.copy()
            a_test2[i] = c2
            a_test2 = a_test2 * norm / a_test2.sum()
            f2 = evaluate_local(a_test2)

            for _ in range(30):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_test1 = a.copy()
                    a_test1[i] = c1
                    a_test1 = a_test1 * norm / a_test1.sum()
                    f1 = evaluate_local(a_test1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a_test2 = a.copy()
                    a_test2[i] = c2
                    a_test2 = a_test2 * norm / a_test2.sum()
                    f2 = evaluate_local(a_test2)

            optimal_val = min(f1, f2)
            a_test_opt = a.copy()
            a_test_opt[i] = c1 if f1 < f2 else c2
            a_test_opt = a_test_opt * norm / a_test_opt.sum()
            sc = evaluate_local(a_test_opt)

            if sc < best_score - 1e-12:
                a = a_test_opt
                conv = fast_autocorr(a)
                best_score = conv.max()
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_swap_cd(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(0, n, 2):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx % n]
            j = perm[(idx + 1) % n]
            if i == j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 * norm and aj < 1e-12 * norm:
                continue

            # Try moving mass from i to j and vice versa
            best_delta = 0.0
            best_val = eval_local_score = fast_autocorr(a).max()

            # Sample delta values
            max_d = max(ai, 0)
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                d = max_d * frac
                if d <= 0 or d > ai:
                    continue
                a_try = a.copy()
                a_try[i] = ai - d
                a_try[j] = aj + d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_delta = d

            # Also try reverse direction
            max_d2 = max(aj, 0)
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                d = max_d2 * frac
                if d <= 0 or d > aj:
                    continue
                a_try = a.copy()
                a_try[i] = ai + d
                a_try[j] = aj - d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_delta = -d

            if abs(best_delta) > 1e-14:
                a_try = a.copy()
                if best_delta > 0:
                    a_try[i] -= best_delta
                    a_try[j] += best_delta
                else:
                    a_try[i] += abs(best_delta)
                    a_try[j] -= abs(best_delta)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * norm / s_try
                    sc = fast_autocorr(a_try).max()
                    if sc < best_score - 1e-13:
                        a = a_try
                        conv = fast_autocorr(a)
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Final CD n={n_best} ===")
        t8 = min(remaining * 0.6, 120.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_descent(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final pair CD ===
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 9: Final PCD n={n_best} ===")
        t9 = min(remaining * 0.7, 80.0)
        if t9 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_swap_cd(a_norm, t9, rng, verbose=False)
            try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
