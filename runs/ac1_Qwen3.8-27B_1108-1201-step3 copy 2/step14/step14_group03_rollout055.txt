def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Use the fact that T @ b = conv(a, b)
    # conv(a,b)[k] = sum_j a[j] * b[k-j] for valid indices
    # Instead of building T explicitly, use a different formulation:
    # max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0
    # We build T as (2n-1) x n matrix where T[k,j] = a[j] if 0<=k-j<n
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        if a[j] > 0:
            T[j:min(j + n, size), j] = a[j]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = None
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    # Anderson acceleration memory
    aa_m = 3
    aa_hist_y = []
    aa_hist_x = []

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Adaptive M: slightly looser early on
        M_factor = 1.0 + 0.05 * np.exp(-t / 200.0)
        M_use = M_curr * M_factor

        b = lp_best_response(y, M_use, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov with periodic Anderson-like restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            # Use Anderson-like extrapolation every few steps
            if t % 5 == 0 and prev_y is not None:
                # Simple extrapolation: use direction from last 2 steps
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y_new + momentum * (y_new - y)
            else:
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Periodic random perturbation to escape local minima
        if n_iters % 100 == 0 and n_iters > 50:
            if np.random.random() < 0.3:
                noise = np.random.default_rng(n_iters).normal(0, 0.001 * norm / n, n)
                y_pert = y + noise
                y_pert = np.maximum(y_pert, 0)
                sp = y_pert.sum()
                if sp > 1e-10:
                    y_pert = y_pert * norm / sp
                    sp_score = fast_autocorr(y_pert).max()
                    if sp_score < best_score:

                        best_a = y_pert.copy()
                        y = y_pert.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def exact_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Exact coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            x_old = a[i]
            if x_old < norm * 1e-12:
                continue

            # Compute base: conv without contribution of a[i]
            # conv[k] = base[k] + 2*x*a[k-i]*(valid, k!=2i) + x^2*(k==2i)
            base = conv.copy()
            # Subtract contribution of x_old
            for k in range(i, min(i + n, size)):
                if k == 2 * i:
                    base[k] -= x_old * x_old
                else:
                    base[k] -= 2.0 * x_old * a[k - i]

            # Search for optimal x in [x_lo, x_hi]
            # Objective: max_k (base[k] + 2*x*a[k-i]*(valid,k!=2i) + x^2*(k==2i)) / (sum-a[i]+x)^2
            # sum = norm, so S(x) = norm - x_old + x
            s_base = norm - x_old

            x_lo = max(0.0, x_old * 0.3)
            x_hi = min(norm * 0.1, x_old * 3.0 + norm * 0.01)

            n_samples = 30
            candidates = np.linspace(x_lo, x_hi, n_samples)

            best_x = x_old
            best_obj = conv.max()

            # Precompute the "linear coefficients" for each k
            # For k in [i, i+n-1], k != 2i: coefficient is 2*a[k-i]
            # For k = 2i: coefficient is 1 (quadratic)
            lin_coeffs = np.zeros(size)
            for k in range(i, min(i + n, size)):
                if k != 2 * i:
                    lin_coeffs[k] = 2.0 * a[k - i]
            quad_k = 2 * i if 2 * i < size else -1

            for x in candidates:
                if x < 0:
                    continue
                S = s_base + x
                if S < 1e-15:
                    continue
                # New conv = base + x * lin_coeffs + x^2 * (1 at quad_k)
                new_conv_max = 0.0
                # Compute max efficiently
                vals = base + x * lin_coeffs
                if quad_k >= 0:
                    vals[quad_k] += x * x
                new_conv_max = vals.max()
                obj = new_conv_max / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if best_x != x_old and abs(best_x - x_old) > 1e-15:
                # Update
                a[i] = best_x
                S = s_base + best_x
                a = a * norm / S
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                # else: keep the change anyway (it improved locally even if global max didn't drop)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def gradient_polish(a0, time_limit, verbose=True):
    """Polish using subgradient of the max-convolution function with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    step_size = norm / (n * 10.0)
    n_steps = 0
    patience = 0

    while time.monotonic() - t0 < time_limit and patience < 200:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]

        # Subgradient: derivative of conv(a,a)[max_idx] w.r.t. a
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/dc: 2 * a[k-j] summed over valid j -> this is 2 * (a * delta_k)[j] where delta_k is shifted
        # More precisely: d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (with a[out-of-range] = 0)
        grad = np.zeros(n)
        for j in range(n):
            # a[k-j] valid if 0 <= k-j < n
            if 0 <= max_idx - j < n:
                grad[j] += a[max_idx - j]
            # a[j-k] valid if 0 <= j-k < n, i.e., k <= j < k+n
            if 0 <= j - max_idx < n:
                grad[j] += a[j - max_idx]

        # Project gradient descent with normalization
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s

        score = fast_autocorr(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            patience = 0
            # Reduce step size slowly
            step_size *= 0.995
        else:
            patience += 1
            step_size *= 0.95

        a = a_new
        n_steps += 1

        if verbose and n_steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GP step={n_steps} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on base length (50% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining * 0.50:.0f}s) ===")
    t1 = remaining * 0.50
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Exact coordinate descent (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Exact CD n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = exact_coordinate_descent(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"ecd_n{n_best}")

    # Phase 3: Gradient polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Gradient polish n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t3 = remaining * 0.10
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_polish(a_norm, t3, verbose=True)
        do_try_update(a_out, f"gp_n{n_best}")

    # Phase 4: Second LP Nesterov (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Pair CD polish (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_best} (budget={remaining * 0.05:.0f}s) ===")
    t5 = remaining * 0.05
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t5, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 6: Try larger length (interpolation + LP)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base + 200
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={remaining * 0.7:.0f}s) ===")
        t6 = remaining * 0.7
        if t6 > 30:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            t6a = min(t6 * 0.6, remaining * 0.4)
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t6a, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 20 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = exact_coordinate_descent(a_norm_alt2, min(remaining2 * 0.5, 30.0), rng, verbose=False)
                do_try_update(a_out_alt2, f"ecd_n{n_alt}")

    # Phase 7: Try smaller length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_small = max(800, n_base - 200)
        print(f"\n=== Phase 7: Small length n={n_small} (budget={remaining * 0.5:.0f}s) ===")
        t7 = remaining * 0.5
        if t7 > 20:
            init_small = interpolate_sequence(best_seq, n_small)
            a_norm_small = init_small * np.sqrt(2.0 * n_small) / init_small.sum()
            a_out_small, _ = nesterov_lp_optimize(a_norm_small, min(t7, remaining * 0.4), solver, verbose=False)
            do_try_update(a_out_small, f"lp_n{n_small}")

    # Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        n_fin = len(best_seq)
        print(f"\n=== Final Polish: n={n_fin} (budget={remaining:.0f}s) ===")
        a_norm_fin = best_seq * np.sqrt(2.0 * n_fin) / best_seq.sum()
        a_out_fin, _ = exact_coordinate_descent(a_norm_fin, remaining * 0.5, rng, verbose=False)
        do_try_update(a_out_fin, f"final_cd_n{n_fin}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            a_norm_fin2 = best_seq * np.sqrt(2.0 * n_fin) / best_seq.sum()
            a_out_fin2, _ = gradient_polish(a_norm_fin2, remaining2 * 0.7, verbose=False)
            do_try_update(a_out_fin2, f"final_gp_n{n_fin}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
