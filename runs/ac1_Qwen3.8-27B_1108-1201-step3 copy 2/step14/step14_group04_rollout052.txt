_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def spectral_flat_shaping(a0, time_limit, seed=42):
    """
    Novel spectral domain method: iteratively flatten the power spectrum
    of the zero-padded sequence to minimize peak autocorrelation.
    Based on Gerchberg-Saxton style iterative projection.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L = 1 << (2 * n - 1).bit_length()
    stall = 0
    n_iters = 0

    # Adaptive relaxation
    relax = 0.15

    while time.monotonic() - t0 < time_limit:
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = rfft(a_pad)

        # Target: make |A| as flat as possible (excluding DC)
        # Parseval: sum|A[k]|^2 = L * sum(a[j]^2)
        energy = L * np.dot(a, a)
        dc = A[0].real  # = sum(a) = 1
        # Target magnitude for non-DC bins
        remaining_energy = energy - dc * dc
        if remaining_energy < 0:
            remaining_energy = 0
        n_bins = L // 2  # number of rfft bins
        target_mag = np.sqrt(remaining_energy / max(n_bins - 1, 1))

        # Blend current and target magnitudes
        cur_mag = np.abs(A)
        new_mag = (1.0 - relax) * cur_mag + relax * target_mag
        new_mag[0] = dc  # keep DC fixed

        # Reconstruct with new magnitudes, preserve phases
        phases = np.angle(A)
        A_new = new_mag * np.exp(1j * phases)

        # Inverse FFT
        a_new_full = irfft(A_new, L)
        a_new = a_new_full[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = fast_autocorr(a_new).max()
        n_iters += 1

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly increase relaxation as we progress
            relax = min(0.3, relax * 1.001)
        else:
            a = a_new  # still accept to continue iterating
            stall += 1

            if stall % 100 == 0:
                # Adaptive relaxation: decrease if stuck
                relax = max(0.03, relax * 0.9)

            if stall > 2000:
                # Try random perturbation
                relax = 0.15
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            sc_t = fast_autocorr(a_trial).max()
                            if sc_t < best_score:
                                a = a_trial
                                best_score = sc_t
                                best_a = a.copy()
                                stall = 0
                            else:
                                a = a_trial  # accept anyway for exploration
                if stall > 5000:
                    break

        if n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final normalize to standard norm
    best_a = best_a * norm / best_a.sum()
    return best_a, best_score

        mag[valley_idx] += delta
        mag[peak_idx] = max(mag[peak_idx], 0.0)

        # Reconstruct
        phases = np.angle(A)
        A_new = mag * np.exp(1j * phases)
        a_new_full = irfft(A_new, L)
        a_new = a_new_full[:n]

            if stall > 5000:
                break
            continue

        # Move mass from the largest contributor to a low-position
        src = idx[np.argmax(contrib)]

        # Find a good destination: positions not contributing to top peaks
        top_peaks = np.argsort(b)[::-1][:5]
        peak_mask = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            jp = max(0, kp - n + 1)
            jep = min(n - 1, kp)
            peak_mask[jp:jep + 1] = True

        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]
        # Pick from low-mass positions
        candidates = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
        dst = candidates[rng.integers(len(candidates))]

        # Try several step sizes
        improved = False
        for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                b = fast_autocorr(a)
                peak_idx = np.argmax(b)
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                if stall > 5000:
                    break

    best_a = best_a * norm / max(best_a.sum(), 1e-15)
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LP, spectral shaping,
    and coordinate descent. Novel spectral flat-shaping complements LP.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [20.0, 5.0, 1.5, 0.5, 0.2]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Spectral flat-shaping (NOVEL) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Spectral flat-shaping n={n_best} (budget={t3:.0f}s) ===")
        a_in = best_seq / best_seq.sum()
        a_out, _ = spectral_flat_shaping(a_in, t3, seed=42)
        try_update(a_out, f"spectral_n{n_best}")

        # Also try water-filling variant
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 80.0)
        if t3b > 15:
            print(f"\n=== Phase 3b: Spectral water-filling n={n_best} (budget={t3b:.0f}s) ===")
            a_in = best_seq / best_seq.sum()
            a_out, _ = spectral_water_filling(a_in, t3b, seed=99)
            try_update(a_out, f"waterfill_n{n_best}")

    # === Phase 4: LP polish after spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Coord descent n={n_best} (budget={t5:.0f}s) ===")
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_descent(a_in, t5, seed=555)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP at n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov(a_norm, t6, solver, verbose=True)
        try_update(a_out, f"lp3_n{n2}")

        # Spectral on longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.05, 50.0)
        if t6b > 10 and best_seq is not None:
            n_best = len(best_seq)
            if n_best == n2:
                print(f"\n=== Phase 6b: Spectral at n={n2} (budget={t6b:.0f}s) ===")
                a_in = best_seq / best_seq.sum()
                a_out, _ = spectral_flat_shaping(a_in, t6b, seed=314)
                try_update(a_out, f"spectral_n{n2}")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 15:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=False)
            try_update(a_out, "lp_final")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.3, 40.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final coord descent n={n_best} (budget={t8:.0f}s) ===")
            a_in = best_seq / best_seq.sum()
            a_out, _ = coordinate_descent(a_in, t8, seed=4242)
            try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
