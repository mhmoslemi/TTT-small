def build_constraint_matrix(a, n):
    """Build Toeplitz matrix T such that conv(a, b) = T @ b."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lp_best_response_dual(a, M, solver, T=None):
    """Solve the dual LP to get the lag distribution p."""
    n = len(a)
    size = 2 * n - 1
    if T is None:
        T = build_constraint_matrix(a, n)
    # Dual: min M * sum(p) s.t. T^T @ p >= 1, p >= 0
    p_var = cp.Variable(size, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(p_var)), [T.T @ p_var >= 1])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    p_val = np.asarray(p_var.value, dtype=np.float64)
    if p_val is None:
        return None
    p_val = np.maximum(p_val, 0.0)
    s = p_val.sum()
    if s < 1e-15:
        return None
    return p_val / s  # Normalize to probability distribution

def subgradient_of_max_conv(a, k_peak):
    """
    Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    At the peak lag k_peak, the subgradient is:
    g[j] = 2 * a[k_peak - j] if max(0, k_peak-n+1) <= j <= min(n-1, k_peak), else 0
    """
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    for j in range(js, je + 1):
        g[j] = 2.0 * a[k_peak - j]
    return g

def analytical_coord_descent(a0, time_limit, seed=42):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Triangular: high at edges, low in middle
        x = np.linspace(0.0, 1.0, n)
        a = np.abs(2.0 * x - 1.0) ** 1.5

        # Two peaks

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):

    g = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    for j in range(js, je + 1):
        g[j] = 2.0 * a[k_peak - j]
    return g


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, strategy="adaptive"):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    constraint_avg = a.copy()
    constraint_avg_count = 1
    last_constraint_update = 0

    for t in range(1, 80000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive M schedule based on strategy
        if strategy == "adaptive":
            frac = min(1.0, t / 3000.0)
            # Start with small reduction, grow to 4-5%
            eps = 0.005 + 0.045 * frac
        elif strategy == "aggressive":
            frac = min(1.0, t / 2000.0)
            eps = 0.01 + 0.06 * frac
        else:
            frac = min(1.0, t / 5000.0)
            eps = 0.003 + 0.03 * frac
        M_used = M_curr * (1.0 - eps)

        # Update constraint average
        if t - last_constraint_update >= 30:
            constraint_avg = (constraint_avg * constraint_avg_count + y) / (constraint_avg_count + 1)
            constraint_avg_count += 1
            last_constraint_update = t

        # Solve LP with constraint average
        T = build_constraint_matrix(constraint_avg, n)
        b = lp_best_response(constraint_avg, M_used, solver, T)
        if b is None:
            T2 = build_constraint_matrix(y, n)
            b = lp_best_response(y, M_curr * 0.995, solver, T2)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.5)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        # Nesterov restart every 800 iterations to avoid oscillation
        if n_iters % 800 == 0:
            y = best_a.copy()
            # Recompute momentum from scratch
            a_new = g * norm_val / g.sum()
            y = (1.0 - 0.5) * y + 0.5 * a_new
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm_val / s_y
            a = y.copy()
            continue

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1 or n_iters % 800 == 0:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 80 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 50.0:
            stall_count += 1
            if stall_count > 6:
                if verbose:
                    print(f"    LP Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def exp_gradient_optimize(a0, time_limit):
    """
    Exponentiated gradient (mirror descent) in log-space.
    a = softmax(u). The gradient of f in u-space is:
    grad_u[j] = a[j] * (g[j] - sum_i g[i]*a[i])
    where g is the subgradient of f w.r.t. a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    u = np.log(a / a.sum())
    u = u - u.max()
    t0 = time.monotonic()

    best_a = (a / a.sum()).copy()
    best_score = fast_autocorr(best_a).max()

    y = u.copy()
    avg_u = np.zeros(n)
    n_avg = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute a from y
        a_y = np.exp(y - y.max())
        a_y = a_y / a_y.sum()

        b = fast_autocorr(a_y)
        M_curr = b.max()
        k_peak = int(np.argmax(b))

        # Subgradient w.r.t. a
        g = subgradient_of_max_conv(a_y, k_peak)

        # Gradient in u-space: grad_u = a * (g - <g, a>)
        g_dot_a = np.dot(g, a_y)
        grad_u = a_y * (g - g_dot_a)

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = y - alpha * grad_u

        if t == 1:
            u_new = y_new
        else:
            momentum = (t - 1.0) / (t + 2.0)
            u_new = y_new + momentum * (y_new - y)

        # Re-center to prevent overflow
        if t % 200 == 0:
            u_new = u_new - u_new.max()
            y_new = y_new - y_new.max()

        y = y_new
        u = u_new

        # Compute a and score
        a_u = np.exp(u - u.max())
        a_u = a_u / a_u.sum()
        score_u = fast_autocorr(a_u).max()

        if score_u < best_score:
            best_score = score_u
            best_a = a_u.copy()
            last_improve = time.monotonic()

        score_y = M_curr
        if score_y < best_score:
            best_score = score_y
            best_a = a_y.copy()
            last_improve = time.monotonic()

        # Averaging
        avg_u += u
        n_avg += 1
        if n_avg % 500 == 0:
            avg_a = np.exp(avg_u / n_avg - (avg_u / n_avg).max())
            avg_a = avg_a / avg_a.sum()
            score_avg = fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()
                last_improve = time.monotonic()

        if time.monotonic() - last_improve > 40.0:
            stall_count += 1
            if stall_count > 15:
                break
        else:
            stall_count = 0

        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExpGrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_optimize(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Top contributing positions for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:12]

        # Positions not contributing to top 5 peaks
        top_peaks = np.argsort(b)[::-1][:5]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        non_peak = np.where(~peak_mask)[0]

        # Low-value positions
        sorted_by_val = np.argsort(a)
        low_val = sorted_by_val[:15]

        improved = False

        # Strategy 1: Move from peak contributors to non-peak/low positions
        for src in top_contrib:
            if a[src] < 1e-8:
                continue
            candidates = list(set(list(non_peak[:12]) + list(low_val[:12])))
            candidates = [c for c in candidates if c != src]
            for dst in candidates:
                for frac in [0.1, 0.2, 0.35]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # Strategy 2: Try swaps
        if not improved:
            for _ in range(80):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a[j], a[i]
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        # Strategy 3: Random transfer
        if not improved:
            for _ in range(150):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.12)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if
