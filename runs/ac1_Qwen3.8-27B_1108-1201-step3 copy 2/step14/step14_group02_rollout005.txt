**Algorithm: Smoothed gradient descent with backtracking line search on the simplex.**

1. Start with `a` on the simplex (sum = 1, a >= 0).
2. Compute `C = conv(a, a)`.
3. Compute weights `w_k = softplus(C, beta)` (softmax-like).
4. Compute gradient `g = sum_k w_k * 2 * shift(a, k)`.
5. Take a step: `a_new = a - step * g`.
6. Project onto the simplex (or use a mirror map).
7. Adjust step size (backtracking).

The challenge is the projection onto the simplex, which is O(n log n).

Alternatively, use the **Fenchel-Young** approach or **mirror descent** with the negative entropy (which gives the multiplicative update / "exponentiated gradient"):
`a_new[j] = a[j] * exp(-step * g[j]) / Z`

This automatically stays in the positive orthant, and we just need to normalize for the simplex constraint.

**Another key idea: Use the "active set" structure.**

At the optimum, only a few lags are "active" (achieve the max). We can identify these and use them for optimization. The active set changes as we iterate.

**Let me also consider a completely different approach: spectral methods.**

We want `conv(a, a)` to be as flat as possible. In the frequency domain:
`FFT(conv(a,a)) = |FFT(a)|^2` (for circular convolution; for linear, we zero-pad).

For the linear convolution to be flat, we need `|FFT(a_padded)|^2` to be flat (in the frequency domain), where `a_padded` is `a` zero-padded to length `>= 2n-1`.

A flat `|FFT(a)|^2` means `a` is a "flat spectrum" sequence, which is related to Zadoff-Chu sequences and other sequences with good correlation properties.

For a complex sequence, we can achieve a perfectly flat spectrum (all `|FFT[k]|^2` equal). For a real, non-negative sequence with fixed length, this is harder.

def smoothed_md_step(a, beta, step_size):
    """One step of mirror descent with smoothed max objective."""
    n = len(a)
    # Compute convolution
    C = fast_autocorr(a)  # length 2n-1
    
    # Compute softmax weights
    log_w = beta * (C - C.max())
    w = np.exp(log_w)
    w /= w.sum()
    
    # Compute gradient: grad[m] = 2 * sum_j a[j] * w[j+m]
    N = 1 << (3 * n - 2).bit_length()
    F_a = np.fft.rfft(a, N)
    F_w = np.fft.rfft(w, N)
    grad = 2.0 * np.fft.irfft(F_w * np.conj(F_a), N)[:n]
    
    # Mirror descent step (multiplicative update)
    log_a = np.log(a + 1e-30)
    log_a_new = log_a - step_size * grad
    log_a_new -= log_a_new.max()  # numerical stability
    a_new = np.exp(log_a_new)
    a_new /= a_new.sum()
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    
    return a_new, C.max(), w

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent with log-sum-exp annealing (primary)
2. LP-based Frank-Wolfe with Nesterov acceleration
3. Coordinate push with peak-aware mass transfer

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smooth_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*C[k])) w.r.t. a."""
    n = len(a)
    C = fast_autocorr(a)
    # Softmax weights
    C_shifted = C - C.max()
    w = np.exp(beta * C_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(C)) / len(C)
    else:
        w = w / w_sum
    # grad[m] = 2 * sum_j a[j] * w[j+m] for j=0..n-1
    # Use FFT: this is the cross-correlation of w and a
    N = 1 << (3 * n - 2).bit_length()
    F_a = rfft(a, N)
    F_w = rfft(w, N)
    grad = 2.0 * irfft(F_w * np.conj(F_a), N)[:n]
    return grad, C.max(), w

def smoothed_mirror_descent(a0, time_limit, beta_start=5.0, beta_end=30.0, 
                             max_iters=500000, verbose=True):
    """Mirror descent with log-sum-exp smoothing and annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Average for CVX tracking
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        frac = min(1.0, t / (max_iters * 0.7))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Step size: decrease over time
        step = 1.0 / (np.sqrt(t) + 1.0) * np.sqrt(n) * 0.5
        
        grad, M_curr, w = compute_smooth_gradient(a, beta)
        
        # Mirror descent step (multiplicative / exponentiated gradient)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Track best
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        # Running average (Polyak averaging)
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = norm_sum1(avg_a / n_avg)
            M_avg = fast_autocorr(avg_check).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=None):
    """Nesterov-accelerated Frank-Wolfe with LP best-response."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    iter_count = 0
    
    for t in range(1, max_iters if max_iters else 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build T matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta_fw = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_fw * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_push(a0, time_limit, verbose=True, max_stall=200):

        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]
