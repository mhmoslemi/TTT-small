def fast_conv_self(a):

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(30.0, 0.2, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(300, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(8):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= max_c + 0.05 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_constraints=200):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    T_mat = build_T(a)

    active_set = np.arange(size)
    n_active = size

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening: start small, grow if needed
        if n_active > max_constraints:
            threshold = M * (0.7 + 0.25 * (n_active / size))
            active_set = np.where(conv_y >= threshold)[0]
            if len(active_set) < 30:
                active_set = np.where(conv_y >= M * 0.4)[0]
            n_active = len(active_set)
            # Rebuild T for current a (T changes as a changes)
            T_mat = build_T(y)
        else:
            T_mat = build_T(y)

        b = lp_solve(y, M, solver, T_mat, active_set)
        if b is None:
            b = lp_solve(y, M, solver, T_mat, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Check if dropped constraints are violated
        if n_active < size:
            check_conv = fast_conv_cross(y, g)
            violated = check_conv > M * (1 + 1e-3)
            if np.any(violated):
                n_violated = np.sum(violated)
                if n_violated > 5:
                    # Expand active set
                    new_threshold = M * 0.6
                    active_set = np.where(conv_y >= new_threshold)[0]
                    n_active = len(active_set)
                    T_mat = build_T(y)
                    # Re-solve with expanded set
                    b = lp_solve(y, M, solver, T_mat, active_set)
                    if b is not None:
                        b_sum = b.sum()
                        if b_sum > 1e-15:
                            g = b * norm / b_sum

        # Nesterov momentum update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, active={n_active}")

    return best_a, best_score

def subgradient_descent(a0, time_limit, verbose=True):
    """Sharp subgradient descent on max(conv(a,a)) with momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    v = np.zeros(n)
    momentum = 0.85
    eta = 1.0 / (3.0 * best_score * np.sqrt(n))

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        M = conv.max()

        # Active set: lags where conv is close to max
        threshold = M * 0.98
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.where(conv == M)[0]

        # Compute subgradient: for each active k, d conv[k]/d a[i] = a[k-i] + a[i-k]
        subgrad = np.zeros(n)
        for k in active:
            # a[k-i] for i where 0 <= k-i < n => k-n+1 <= i <= k
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            if hi1 > lo1:
                idx = np.arange(lo1, hi1)
                subgrad[idx] += a[k - idx]
            # a[i-k] for i where 0 <= i-k < n => k <= i <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if hi2 > lo2:
                idx2 = np.arange(lo2, hi2)
                subgrad[idx2] += a[idx2 - k]

        subgrad /= len(active)

        # Center for simplex constraint
        sg_centered = subgrad - subgrad.mean()

        # Momentum
        v = momentum * v + (1.0 - momentum) * sg_centered

        # Step with backtracking
        a_old = a.copy()
        trial_eta = eta
        for trial in range(6):
            a_new = a - trial_eta * v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                trial_eta *= 0.5
                continue
            a_new = a_new * norm / s_new
            new_max = fast_conv_self(a_new).max()
            if new_max <= M + 0.01:
                break
            trial_eta *= 0.5

        a = a_new
        M_new = fast_conv_self(a).max()

        if M_new < best_score:
            best_score = M_new
            best_a = a.copy()

        # Decay step size slowly
        eta *= 0.9995

        if verbose and it % 100 == 0:
            print(f"    SubGrad it={it}: max={M:.8f}, best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Select j based on peak lags + random
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            peak_threshold = cur_max * 0.95
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:4]:
                j_struct = (kl - i) % n
                if j_struct != i:
                    j_candidates.add(j_struct)
                j_struct2 = (kl + i) % n
                if j_struct2 != i:
                    j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: change in conv when transferring delta from i to j
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Binary search for optimal delta
                lo_d, hi_d = 0.0, ai
                best_d = 0.0
                best_v = cur_max
                for _ in range(12):
                    mid = (lo_d + hi_d) / 2.0
                    if mid < 1e-15:
                        lo_d = mid
                        continue
                    new_conv = conv + mid * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * mid + mid * mid
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = mid
                    # Check if max is decreasing or increasing at mid
                    # Find which position achieves max
                    max_pos = np.argmax(new_conv)
                    if g[max_pos] < 0:
                        lo_d = mid  # increasing delta helps
                    else:
                        hi_d = mid  # increasing delta hurts

                if best_d > 1e-10 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        eta = max(eta, 0.01 / best_score)

        if verbose and it % 200 == 0:
            print(f"    MD it={it}: max={M:.8f}, best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    else:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse, long run)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True, max_constraints=150)
        try_update(a_out, f"lp1_n{n2}")

    # Phase 3: Subgradient descent (different landscape exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: Subgradient n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = subgradient_descent(best_seq, t3, verbose=True)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # Phase 4: LP Nesterov round 2 (continue from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True, max_constraints=120)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Mirror descent (multiplicative weights)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 5: Mirror Descent n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = mirror_descent(best_seq, t5, verbose=True)
        try_update(a_out, f"mirror_n{len(best_seq)}")

    # Phase 6: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Pair CD n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(best_seq)}")

    # Phase 7: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t7 = min(remaining * 0.30, 200.0)
    print(f"\n=== Phase 7: Longer length n={n_long} (budget={t7:.0f}s) ===")
    if t7 > 60 and remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        sc_long_init = eval_fn(init_long)
        print(f"  Long init score: {sc_long_init:.6f}")
        t7a = min(t7 * 0.5, 100.0)
        a_out, _ = nesterov_lp_optimize(init_long, t7a, solver, verbose=False, max_constraints=100)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.5, 80.0)
            a_out2, _ = pair_cd_optimize(interpolate_seq(best_seq, n_long), t7b, rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 8: Final polish - alternate LSE + CD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 50.0), rng, verbose=False)
            try_update(a_out, f"pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
