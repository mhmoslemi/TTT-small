import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build the convolution matrix T where T[k,j] = a_{k-j} if valid."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_simple(T, M, n, max_constraints=None):
    """
    Solve the LP: maximize sum(b) subject to T @ b <= M, b >= 0.
    Uses only the most violated / most binding constraints.
    Returns best response vector b or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    size = T.shape[0]
    if max_constraints is None:
        max_constraints = min(size, 200)

    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))

    # Start with all constraints, then reduce
    constraints = [T @ b_var <= M]
    prob = cp.Problem(objective, constraints)

    # Try solving
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try with reduced constraints
        circ = T @ np.ones(n)
        active = circ >= M * 0.9
        if active.sum() == 0:
            active = np.zeros(size, dtype=bool)
            active[np.argsort(circ)[::-1][:max_constraints]] = True
        elif active.sum() > max_constraints:
            idx = np.where(active)[0]
            top = idx[np.argsort(circ[idx])[::-1][:max_constraints]]
            active = np.zeros(size, dtype=bool)
            active[top] = True

        constraints = [T[active, :] @ b_var <= M[active]]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None

    if prob.status not in ('optimal', 'optimal_inaccurate') or b_var.value is None:
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, verbose=True, max_c=150):
    """LP-based Nesterov accelerated best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    T = build_conv_matrix(a)

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8

    while time.monotonic() - t0 < time_limit:
        t_iter = n_iters + 1
        M_curr = fast_autocorr(y).max()

        # Solve LP best response
        b = solve_lp_simple(T, M_curr, n, max_constraints=max_c)
        if b is None:
            stall += 5
            if stall > 500:
                _restart_nesterov(y, a, prev_y, n, norm, seed=n_iters)
                y = np.maximum(y + np.random.default_rng(n_iters).normal(0, 0.02 * y.mean(), n), 0)
                y = y * norm / y.sum()
                a = y.copy()
                prev_y = y.copy()
                T = build_conv_matrix(a)
                stall = 0
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks Nesterov step
        alpha = 2.0 / (t_iter + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t_iter == 1:
            a_new = g.copy()
        else:
            momentum = (t_iter + 1.0 - gamma) / (t_iter + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Update T (since a changed, the convolution matrix changes)
        # But T depends on a, not y... Actually T depends on the current a for the convolution.
        # We rebuild T periodically to save time
        if n_iters % 100 == 0:
            T = build_conv_matrix(y)

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 200:
            gamma = max(1.0, gamma - 0.15)
        elif stall == 0 and best_M < score_y:
            gamma = min(2.0, gamma + 0.05)

        # Restart if stuck
        if stall > 400:
            rng = np.random.default_rng(n_iters * 31 + 7)
            y = np.maximum(y + rng.normal(0, 0.03 * y.mean(), n), 0)
            y = y * norm / y.sum()
            a = y.copy()
            prev_y = y.copy()
            T = build_conv_matrix(y)
            stall = 0
            gamma = 1.8
            if verbose:
                print(f"    [LP Restart at iter {n_iters}]")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, gamma={gamma:.2f}")

    return best_a, best_M


def nesterov_subgradient(a0, time_limit, verbose=True):
    """
    Direct Nesterov subgradient method with peak cycling.
    The subgradient of max_k (Ta)_k is T[k*, :] where k* is an argmax.
    We cycle through top peaks with weighting.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    gamma = 1.8

    while time.monotonic() - t0 < time_limit:
        t_iter = iters + 1
        b = fast_autocorr(y)
        M_curr = b.max()

        # Compute weighted subgradient over top K peaks
        K = min(8, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:K]
        peak_vals = b[sorted_idx]
        max_peak = peak_vals[0]
        spread = max_peak - peak_vals[-1] + 1e-10

        # Temperature-decayed weighting
        temp = max(0.01, 0.1 * spread * (1.0 - min(1.0, iters / 5000.0)))
        w = np.exp((peak_vals - max_peak) / temp)
        w = w / w.sum()

        # Compute subgradient: weighted sum of T rows
        grad = np.zeros(n)
        for m, p in enumerate(sorted_idx):
            if w[m] < 1e-7:
                continue
            # Row p of T is: a[max(0,p-n+1):min(n,p+1)] placed at positions [max(0,p-n+1):min(n,p+1)]
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js <= je:
                grad[js:je + 1] += 2.0 * w[m] * y[p - np.arange(js, je + 1)]

        # Project onto tangent of simplex {sum = norm}
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Polyak step size
        polyak = (M_curr - best_M * 0.999) / (grad_norm * grad_norm + 1e-20)
        step = min(1.0 / n, max(polyak, 1e-8 / n))

        # Nesterov step
        alpha = 2.0 / (t_iter + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y - alpha * step * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t_iter == 1:
            a_new = y_new.copy()
        else:
            momentum = (t_iter + 1.0 - gamma) / (t_iter + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new
        iters += 1

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma and restart
        if stall > 500:
            rng = np.random.default_rng(iters * 17 + 3)
            perturb = 0.02 * y.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            y = y * norm / y.sum()
            a = y.copy()
            prev_y = y.copy()
            stall = 0
            gamma = 1.8

        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={iters} ({elapsed:.0f}s): best_M={best_M:.8f}, stall={stall}")

    return best_a, best_M


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """
    Fast peak shaving: identify the dominant peak, find the largest contributor,
    and redistribute to a non-contributing position.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing pair with max product
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = int(idx[np.argmax(contrib)])

        # Find best destination: non-contributing, low value
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            non_contrib_idx = np.array([i for i in range(n) if i != src])

        # Try several destinations and fractions
        n_try = min(5, len(non_contrib_idx))
        candidates = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_try]

        best_trial_M = max_b
        best_move = None
        for dst in candidates:
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[src] -= amt
                a_try[dst] += amt
                a_try = np.maximum(a_try, 0)
                s_t = a_try.sum()
                if s_t < 1e-15:
                    continue
                a_try = a_try * norm / s_t
                sc = fast_autocorr(a_try).max()
                if sc < best_trial_M:
                    best_trial_M = sc
                    best_move = (src, dst, frac)

        if best_move is not None:
            src, dst, frac = best_move
            a[src] -= a[src] * frac
            a[dst] += a[src] * frac / (1 - frac) if frac < 1 else 0
            # Redo properly
            a = best_a.copy()  # reset
            src, dst, frac = best_move
            amt = a[src] * frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a * norm / s_t
            sc = fast_autocorr(a).max()
            if sc < best_M:
                best_M = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            n_stall += 1
            # Random perturbation to escape
            if n_stall % 50 == 0:
                idx_p = rng.choice(n, size=max(3, n // 100), replace=False)
                for i in idx_p:
                    j = rng.integers(n)
                    if i != j:
                        amt = a[i] * rng.uniform(0.05, 0.3)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a * norm / s_t
                sc = fast_autocorr(a).max()
                if sc < best_M:
                    best_M = sc
                    best_a = a.copy()
                    n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best_M={best_M:.8f}")

    return best_a, best_M


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, verbose=True, max_c=150)
        do_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Direct Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SubGrad Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_subgradient(a_norm, t2, verbose=True)
        do_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (refined start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, verbose=True, max_c=120)
        do_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base + 80
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_long = interpolate_to_length(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t4, verbose=True, max_c=100)
        do_update(a_out, f"lp_long_n{n2}")

    # === Phase 5: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t5, seed=777, verbose=True)
        do_update(a_out, f"peak_n{n_cur}")

    # === Phase 6: Subgradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: SubGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_subgradient(a_norm, t6, verbose=True)
        do_update(a_out, f"subgrad2_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, verbose=True, max_c=100)
        do_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 100.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=31415, verbose=True)
        do_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
