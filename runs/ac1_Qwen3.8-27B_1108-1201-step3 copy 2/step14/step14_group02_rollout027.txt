    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    return a / s

def interpolate_sequence(f, new_n):

    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)

def spectral_gradient(a, top_k=30, beta=5.0):
    """Compute subgradient of max(conv) using top-k active constraints."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()

    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    
    # Weight by beta-power
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum
    
    # Compute gradient: grad[j] = sum_k w_k * 2 * a[k-j] (for valid indices)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):

            continue
        idx = np.arange(i_start, i_end + 1)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid multi-phase approach:
1. LP-based Frank-Wolfe with Nesterov acceleration (fast initial convergence)
2. FFT-based spectral gradient with mirror descent (many fast iterations)
3. Progressive length scaling (1600 -> 2500 -> 3500)
4. Multi-start with diverse initializations (sin^2, cos^2, modulated)
5. Coordinate polish with adaptive mass transfer
6. Adaptive time allocation based on progress
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None
_time_budget = 1710.0
_t0_global = 0.0


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  Solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def compute_grad(a, ac, top_k=25, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        if s > e:
            continue
        ii = np.arange(s, e + 1)
        jj = ki - ii
        grad[ii] += 2.0 * wi * a[jj]
    return grad


def spectral_opt(a0, tlim, top_k=25, beta=4.0, gamma=0.7, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = a.copy()
    n_avg = 1
    t_iter = 0
    while time.monotonic() - t0 < tlim:
        t_iter += 1
        ac = fast_autocorr(a)
        g = compute_grad(a, ac, top_k=top_k, beta=beta)
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        step = gamma / (np.sqrt(t_iter + 1) * gn)
        la = np.log(a + 1e-30)
        la_n = la - step * g
        ma = la_n.max()
        a_n = np.exp(la_n - ma)
        a_n = np.maximum(a_n, 0.0)
        sn = a_n.sum()
        if sn < 1e-15:
            a_n = a.copy()
        else:
            a_n = a_n / sn
        a = a_n
        Mn = fast_autocorr(a).max()
        if Mn < best_m:
            best_m = Mn
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            ac2 = avg_a / n_avg
            ac2 = np.maximum(ac2, 0.0)
            s2 = ac2.sum()
            if s2 > 1e-15:
                ac2 = ac2 / s2
                M2 = fast_autocorr(ac2).max()
                if M2 < best_m:
                    best_m = M2
                    best_a = ac2.copy()
        if verbose and t_iter % 2000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"     Spec iter={t_iter} ({el:.1f}s) J={J:.6f}")
    return best_a, best_m

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    t = 0
    while time.monotonic() - t0 < tlim:
        t += 1
        ac = fast_autocorr(y)
        Mc = ac.max()
        if Mc < 1e-14:
            break
        bv = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        con = [T @ bv <= Mc]
        prob = cp.Problem(cp.Maximize(cp.sum(bv)), con)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        bval = np.maximum(np.asarray(bv.value, dtype=np.float64), 0.0)
        bs = bval.sum()
        if bs < 1e-15:
            continue
        g = bval / bs
        alpha = 2.0 / (t + 2.0)
        yn = (1.0 - alpha) * y + alpha * g
        yn = np.maximum(yn, 0.0)
        sy = yn.sum()
        if sy < 1e-15:
            yn = y.copy()
        else:
            yn = yn / sy
        if t == 1:
            xn = g.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            xn = yn + bt * (yn - y)
            xn = np.maximum(xn, 0.0)
            sx = xn.sum()
            if sx < 1e-15:
                xn = yn.copy()
            else:
                xn = xn / sx
        y = yn
        x = xn
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        if verbose and t % 5 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_m

        print(f"\n=== P6: Coord n={n_cur} ({t6:.0f}s) ===")
        a_out, _ = coord_polish(norm_sum1(best_seq), t6, verbose=True)
        try_update(a_out, f"coord_{n_cur}")

    # Phase 7: Final spectral polish
    rem = _time_budget - (time.monotonic() - _t0_global)
    n_cur = len(best_seq)
    if rem > 30:
        t7 = min(rem * 0.30, 200.0)
        print(f"\n=== P7: Final Spec n={n_cur} ({t7:.0f}s) ===")
        a_out, _ = spectral_opt(norm_sum1(best_seq), t7, top_k=30, beta=5.0, gamma=0.5, verbose=True)
        try_update(a_out, f"spec_final_{n_cur}")

    # Phase 8: Final coord
    rem = _time_budget - (time.monotonic() - _t0_global)
    n_cur = len(best_seq)
    if rem > 20:
        t8 = min(rem * 0.4, 100.0)
        print(f"\n=== P8: Final Coord ({t8:.0f}s) ===")
        a_out, _ = coord_polish(norm_sum1(best_seq), t8, verbose=True)
        try_update(a_out, f"coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        el = time.monotonic() - _t0_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={el:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
