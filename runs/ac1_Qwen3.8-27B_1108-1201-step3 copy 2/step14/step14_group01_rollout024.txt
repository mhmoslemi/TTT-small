def fast_conv_self(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    S = a.sum()
    if S < 1e-15:
        return float('inf')

    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

        sc = evaluate_sequence(a_list)

    s = a.sum()
    if s < 1e-15:
        return a
    return a * target_sum / s

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

    n = len(y)
    size = 2 * n - 1

    conv_y = fast_conv_self(y)

        idx = np.argsort(conv_y)[-n_constraints:]
        T = build_T(y)
        T_screened = T[idx, :]
        M_screened = np.full(len(idx), M)
    else:
        T = build_T(y)
        T_screened = T
        M_screened = np.full(size, M)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_screened @ b_var <= M_screened]
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)
