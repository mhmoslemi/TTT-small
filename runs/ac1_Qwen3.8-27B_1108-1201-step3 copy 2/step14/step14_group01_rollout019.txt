"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive constraint screening (HiGHS) and Nesterov acceleration
2. Gradient-guided pair coordinate descent with vectorized delta search
3. Polyak averaging for stable convergence
4. Multi-length exploration with perturbation restarts
Key improvements: HiGHS-based LP (faster than cvxpy), dual-driven constraint selection,
longer LP phase, better CD j-selection using convolution peak structure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# ===================== FFT Convolution Helpers =====================

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

# ===================== Constraint Matrix =====================

def build_T_rows(a, rows):
    """Build specific rows of constraint matrix T where T[k,j] = a[k-j] if 0<=k-j<=n-1."""
    n = len(a)
    T = np.zeros((len(rows), n))
    for idx in range(len(rows)):
        k = rows[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[idx, lo:hi] = a[k - hi + 1: k - lo + 1][::-1]
    return T

# ===================== LP Solver =====================

def lp_solve_hg(a, M, active_indices):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b>=0. Returns b or None."""
    n = len(a)
    c = -np.ones(n)
    T_sub = build_T_rows(a, active_indices)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T_sub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            return np.maximum(res.x, 0.0)
        # Try without presolve if it failed
        res = linprog(c, A_ub=T_sub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': False, 'time_limit': 30.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

# ===================== Core LP + Nesterov Optimization =====================

def lp_nesterov_optimize(a0, time_limit, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive constraint screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s

    best_a = a.copy()
    best_maxb = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()  # Nesterov's y
    avg_b = np.zeros(n)
    n_avg = 0

    # Adaptive constraint set: start broad, tighten over time
    current_threshold = 0.90

    for t in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint selection
        if t < 50:
            # Initially use many constraints
            threshold_frac = 0.80
        elif t < 200:
            threshold_frac = 0.85 + 0.05 * min(1.0, (t - 50) / 150.0)
        else:
            # Gradually tighten
            threshold_frac = 0.90 + 0.08 * min(1.0, (t - 200) / 1000.0)

        active = np.where(conv_y >= M * threshold_frac)[0]
        # Always include the peak
        peak_idx = np.argmax(conv_y)
        if peak_idx not in active:
            active = np.concatenate([active, [peak_idx]])

        # Also include a few random ones for diversity
        if len(active) < n:
            n_extra = min(20, size - len(active))
            if n_extra > 0:
                remaining = np.setdiff1d(np.arange(size), active)
                extra = np.random.choice(remaining, size=n_extra, replace=False)
                active = np.concatenate([active, extra])

        b = lp_solve_hg(y, M, active)
        if b is None:
            # Fall back to full constraints if screened fails
            b = lp_solve_hg(y, M, np.arange(size))
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify no dropped constraints are violated
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 1e-3))[0]
            if len(violated) > 0:
                # Re-solve with violated constraints added
                new_active = np.unique(np.concatenate([active, violated]))
                if len(new_active) < size:
                    b2 = lp_solve_hg(y, M, new_active)
                    if b2 is not None and b2.sum() > 1e-15:
                        b = b2
                        b_sum = b.sum()
                        g = b * norm / b_sum
                # If full solve needed, skip this iteration (too expensive)

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = y.sum()
        y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            # Clip momentum to prevent overshoot
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_maxb:
            best_maxb = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_maxb:
            best_maxb = sc_a
            best_a = a.copy()

        # Polyak averaging
        avg_b += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_b / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_maxb:
                    best_maxb = sc_avg
                    best_a = avg_check.copy()
                # Periodically reset average to focus on recent iterates
                if n_avg % 300 == 0:
                    avg_b = avg_b * 0.5
                    n_avg = n_avg // 2

        if verbose and t % 200 == 0:
            score = 2.0 * n * best_maxb / (norm * norm)
            print(f"    LP t={t} ({elapsed:.0f}s): best_score={score:.8f}")

    return best_a, best_maxb

def lse_gradient_descent(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_maxb = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(40.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT cross-correlation
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)

            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.02 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_maxb:
                best_maxb = sc
                best_a = a.copy()

        if verbose:
            score = 2.0 * n * best_maxb / (norm * norm)
            print(f"    LSE stage {si} T={T_val:.1f}: best_score={score:.8f}")

    return best_a, best_maxb

# ===================== Pair Coordinate Descent =====================

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Gradient-guided pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_maxb = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Find peak lags for j selection
            peak_threshold = cur_max * 0.92
            peak_lags = np.where(conv >= peak_threshold)[0]

            # Build candidate j set
            j_candidates = []
            # Random candidates
            for _ in range(6):
                j_candidates.append(int(rng.integers(n)))
            # Structured: for each peak lag, find j that reduces it
            for kl in peak_lags[:8]:
                # Transferring mass from i to j changes conv[kl] by:
                # delta * (a[kl-j] + a[j-kl] - a[kl-i] - a[i-kl])
                # We want to maximize the reduction, so pick j where a[kl-j]+a[j-kl] is large
                # but a[kl-i]+a[i-kl] is fixed for given i, kl
                # So we want j that maximizes a[kl-j] + a[j-kl]
                # Try j near the "other side" of the peak
                j1 = (kl - i + n) % n
                j2 = (kl + i) % n
                j_candidates.append(j1)
                j_candidates.append(j2)

            # Deduplicate and remove i
            j_candidates = list(set([j for j in j_candidates if j != i]))

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates[:12]:  # Limit candidates for speed
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector g_k = a[k-j]+a[j-k]-a[k-i]-a[i-k]
                g = np.zeros(size)
                # a[k-j] for k in [j, j+n-1]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                # a[j-k] for k in [max(0,j-n+1), j]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a[k-i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                # -a[i-k]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta: minimize max(conv + delta*g + quadratic corrections)
                n_samples = 20
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    # Quadratic corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at i+j
                    ki = i + j
                    if ki < size:
                        # The linear approximation double-counts a[i]*a[j] and a[j]*a[i]
                        # Actually the cross term is: -a[i]*a[j]*delta (from removing a[i])
                        # + a[j]*a[i]*delta (from adding to a[j]) = 0 net for the cross
                        # But the self terms are handled above
                        pass
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_maxb - 1e-12:
                        best_maxb = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3 or no_improve == 5:
                # Perturb to escape local optimum
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_maxb / (norm * norm)
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): score={score:.8f}")

    return best_a, best_maxb

# ===================== Utility Functions =====================

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_score_normalized(a):
    """Compute the actual evaluation score given a normalized (sum=1) sequence."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

# ===================== Main Entry Point =====================

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP best-response with Nesterov acceleration as the
    primary optimizer, augmented with coordinate descent and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization: n={n_base}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # Phase 1: Quick LSE to polish (60s)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 60.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse_n{len(a_out)}")

    # Phase 2: Main LP Nesterov (long phase ~600s)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.55, 600.0)
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, verbose=True)
        try_update(a_out, f"lp1_n{len(a_out)}")

    # Phase 3: Pair CD polish (200s)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 3: Pair CD (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{len(a_out)}")

    # Phase 4: LP round 2 (200s)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 200.0)
    print(f"\n=== Phase 4: LP Round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, verbose=True)
        try_update(a_out, f"lp2_n{len(a_out)}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 100:
        # Try shorter
        n_short = max(1000, n_base - 200)
        t5a = min(remaining * 0.25, 80.0)
        if t5a > 15:
            init_short = interpolate_seq(best_seq, n_short)
            a_out, _ = lp_nesterov_optimize(init_short, t5a, verbose=False)
            try_update(a_out, f"lp_short_{n_short}")

        # Try longer
        remaining = time_budget - (time.monotonic() - t_global)
        n_long = n_base + 200
        t5b = min(remaining * 0.30, 100.0)
        if t5b > 15 and remaining > 60:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = lp_nesterov_optimize(init_long, t5b, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")

    # Phase 6: Perturbation + re-optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Perturbation restart (budget={remaining:.0f}s) ===")
    if remaining > 60:
        # Perturb current best
        a_perturb = best_seq.copy()
        n_pert = max(20, len(a_perturb) // 20)
        idx_pert = rng.choice(len(a_perturb), size=n_pert, replace=False)
        a_perturb[idx_pert] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
        a_perturb = np.maximum(a_perturb, 0.0)
        a_perturb = a_perturb / a_perturb.sum()

        t6 = min(remaining * 0.5, 120.0)
        if t6 > 15:
            a_out, _ = lp_nesterov_optimize(a_perturb, t6, verbose=False)
            try_update(a_out, "perturb_lp")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.6, 60.0), rng, verbose=False)
            try_update(a_out, "perturb_cd")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, min(remaining * 0.6, 80.0), verbose=False)
        try_update(a_out, "final_lp")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 40.0), rng, verbose=False)
            try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
