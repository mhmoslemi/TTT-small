def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))

def lp_best_response(a, M, solver):

    n = len(a)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) * c / n
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.maximum(v - lam, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * c / s
    else:
        result = np.ones(n) * c / n
    return result

def smoothed_grad_optimize(a0, time_limit, verbose=True):
    """
    Smoothed gradient descent with Nesterov acceleration and adaptive temperature.
    Uses FFT-based gradient computation for speed.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # FFT setup
    fft_size = 1 << (3 * n - 3).bit_length()

    # Nesterov state
    y = a.copy()
    y_prev = a.copy()

    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    adam_step = 0

    # Temperature schedule
    conv_init = fast_autocorr(a)
    T0 = max(conv_init.max() * 0.5, norm_val * 0.1)
    T_min = 0.05 * conv_init.max() / n

    t0 = time.monotonic()
    n_iter = 0
    patience = 0
    best_patience_score = best_score

    for step in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: geometric decay with floor
        decay = 0.9995
        T = max(T_min, T0 * (decay ** step))

        # Compute convolution at y
        conv_y = fast_autocorr(y)
        max_conv = conv_y.max()
        if max_conv < 1e-12:
            break

        # Softmax weights
        scaled = (conv_y - max_conv) / T
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w = w / w_sum

        # Compute gradient via cross-correlation
        w_padded = np.zeros(fft_size)
        w_padded[:len(w)] = w
        y_padded = np.zeros(fft_size)
        y_padded[:n] = y

        W_fft = rfft(w_padded, fft_size)
        Y_fft = rfft(y_padded, fft_size)

        # cross-correlation: sum_k w[k] * y[k-i]
        cross = irfft(W_fft * np.conj(Y_fft), fft_size)
        grad = 2.0 * cross[:n]

        # Adam update
        adam_step += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad * grad
        m_hat = m_adam / (1 - beta1 ** adam_step)
        v_hat = v_adam / (1 - beta2 ** adam_step)

        # Learning rate schedule
        lr = 0.05 * norm_val / (1.0 + 0.005 * adam_step) * (T / T0 + 0.3)

        if s_c < 1e-15:
            y_candidate = y.copy()
        else:
            y_candidate = y_candidate * norm_val / s_c

        # Nesterov update
        alpha = 2.0 / (step + 2.0)
        y_new = (1.0 - alpha) * y + alpha * y_candidate
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if step == 1:
            a = y_candidate.copy()
        else:
            momentum = (step - 1.0) / (step + 2.0)
            a = y_new + momentum * (y_new - y_prev)
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = y_new.copy()
            else:
                a = a * norm_val / s_a

        y_prev = y.copy()
        y = y_new
        n_iter += 1

        # Track best (use exact max, not smoothed)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # Restart
        if patience > 500:
            y = best_a.copy()
            y_prev = y.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            adam_step = 0
            patience = 0

                        delta_vec[lo1:hi1] -= a[np.arange(hi1 - lo1)]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 - i + n)
                    if hi2 > lo2:
                        src = np.arange(n - 1 - i, n - 1 - i + (hi2 - lo2))
                        valid = (src >= 0) & (src < n)
                        vals = np.zeros(hi2 - lo2)
                        vals[valid] = a[src[valid]]
                        delta_vec[lo2:hi2] -= vals

                    # Add mass to j
                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] += a[np.arange(hi3 - lo3)]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 - j + n)
                    if hi4 > lo4:
                        src4 = np.arange(n - 1 - j, n - 1 - j + (hi4 - lo4))
                        valid4 = (src4 >= 0) & (src4 < n)
                        vals4 = np.zeros(hi4 - lo4)
                        vals4[valid4] = a[src4[valid4]]
                        delta_vec[lo4:hi4] += vals4

                    delta_vec *= 1.0  # we're moving unit mass (the delta_d below scales it)

                    # Search for best d
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 12
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        # Fix diagonal terms
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov (35% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n1}")

    # === Phase 2: Smoothed Gradient (25% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Smoothed Gradient n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        a_out, _ = smoothed_grad_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sg_n{n2}")

    # === Phase 3: Pair CD (20% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t3, rng, window=30, verbose=True)
        do_try_update(a_out, f"pcd_n{n3}")

    # === Phase 4: LP Nesterov round 2 (15% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n4}")

    # === Phase 5: Smoothed Gradient round 2 (10% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Smoothed Gradient round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = smoothed_grad_optimize(a_norm, t5, verbose=False)
        do_try_update(a_out, f"sg2_n{n5}")

    # === Phase 6: Final Pair CD (remainder) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = remaining * 0.6
    print(f"\n=== Phase 6: Final Pair CD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n6) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t6, rng, window=50, verbose=True)
        do_try_update(a_out, f"pcd2_n{n6}")

    # === Phase 7: Try alternate length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base + 128
        t7 = remaining * 0.7
        print(f"\n=== Phase 7: Alternate length n={n_alt} (budget={t7:.0f}s) ===")
        if t7 > 30:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            t7a = t7 * 0.5
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t7a, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")

            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 20:
                n_sg = n_alt if best_seq is not None and len(best_seq) == n_alt else n6
                best_for_sg = best_seq if best_seq is not None and len(best_seq) == n_alt else best_seq
                a_norm_sg = best_for_sg * np.sqrt(2.0 * len(best_for_sg)) / best_for_sg.sum()
                a_out_sg, _ = smoothed_grad_optimize(a_norm_sg, min(remaining2 * 0.4, 40.0), verbose=False)
                do_try_update(a_out_sg, f"sg_n{len(best_for_sg)}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
