   The Lagrangian is: L(b, y, z) = 1^T b + y^T (M - T b) - z^T b
   where y >= 0 (for T b <= M) and z >= 0 (for b >= 0).

   From z^T b = 0: for each i, either b[i] = 0 or z[i] = 0 (i.e., (T^T y)[i] = 1).

def lp_best_response_fast(a, M):
    """Solve max sum(b) s.t. T@b <= M, b >= 0 using scipy linprog HiGHS."""
    n = len(a)
    size = 2 * n - 1
    # Build T matrix: T[k, i] = a[k-i] for valid k-i
    T = np.zeros((size, n))
    for i in range(n):
        end = min(n, size - i)
        T[i:i+end, i] = a[:end]
    # min -sum(b) s.t. T@b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=T, b_ub=np.full(size, M), bounds=(0, None),
                  method='highs', options={'presolve': True})
    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0.0)
        if b.sum() > 1e-15:
            return b
    return None

def nesterov_lp(a0, time_limit, verbose=True):
    """LP-based best response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    y = a.copy()
    a_momentum = a.copy()
    last_improve = t0
    stall_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_autocorr(y)
        M_curr = c_y.max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Line search with multiple candidates
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_obj = fast_autocorr(np.maximum((1 - alpha_base) * y + alpha_base * g, 0)).max()
        for trial in [alpha_base * 0.3, alpha_base * 0.5, alpha_base * 1.0,
                      alpha_base * 1.5, alpha_base * 2.0, alpha_base * 3.0]:

            cand = np.maximum((1 - trial) * y + trial * g, 0)

            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        alpha = best_alpha

        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov momentum
        momentum = (t - 1.0) / (t + 2.0)
        if t == 1:
            a_new = g.copy()
        else:
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

            else:
                a_new = a_new * norm / s_a

        a_momentum = a_new

        if score_y < best_score:

            last_improve = time.monotonic()

def subgradient_nesterov(a0, time_limit, n_peaks=5, verbose=True):
    """Multi-peak subgradient with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    y = a.copy()
    a_mom = a.copy()
    last_improve = t0
    stall = 0
    gamma = 1.0  # learning rate (will adapt)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_autocorr(y)
        max_c = c_y.max()
        if max_c < 1e-12:
            break

        # Find top peaks
        peak_thresh = max_c * 0.95
        peak_indices = np.where(c_y >= peak_thresh)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c_y)])
        peak_indices = peak_indices[:n_peaks * 3]

        # Compute weighted subgradient
        # For peak k, subgradient component i is y[k-i] (for valid k-i)
        # Weight by how close c[k] is to max
        weights = np.exp(10.0 * (c_y[peak_indices] - max_c) / max_c)
        weights = weights / weights.sum()

        # Compute subgradient via FFT
        # sg[i] = sum_k w_k * y[k-i] for valid k-i
        # This is a cross-correlation
        fft_size = 1 << (3 * n - 2).bit_length()
        y_pad = np.zeros(fft_size)
        y_pad[:n] = y
        w_pad = np.zeros(fft_size)
        w_pad[peak_indices] = weights
        Y_fft = rfft(y_pad, fft_size)
        W_fft = rfft(w_pad, fft_size)
        sg = irfft(Y_fft * np.conj(W_fft), fft_size)[:n]
        sg = 2.0 * sg  # factor of 2 from the quadratic

        # Apply Nesterov
        beta = (t - 1.0) / (t + 2.0)
        if t == 1:
            A = y.copy()
        else:
            A = y + beta * (y - a_mom)
            A = np.maximum(A, 0.0)
            s_A = A.sum()
            if s_A < 1e-15:
                A = y.copy()
            else:
                A = A * norm / s_A

        # Gradient step
        # Adaptive step size
        gamma_t = norm / (np.linalg.norm(sg) * np.sqrt(t + 1.0) + 1e-10)
        a_new = A - gamma_t * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = A.copy()
        else:
            a_new = a_new * norm / s_new

        # Nesterov combination
        theta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = (1.0 - 1.0/theta) * y + (1.0/theta) * a_new
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        y = y_new
        a_mom = a_new

        # Check best
        sc = fast_autocorr(y).max()
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
            last_improve = time.monotonic()
            stall = 0

        sc2 = fast_autocorr(a_mom).max()
        if sc2 < best_score:
            best_score = sc2
            best_a = a_mom.copy()
            last_improve = time.monotonic()
            stall = 0

        if t % 500 == 0:
            n_iters += 1
            if time.monotonic() - last_improve > 60:
                stall += 1
                if stall > 10:
                    break
                y = best_a * norm / best_a.sum()
                a_mom = y.copy()
            if verbose and n_iters % 20 == 0:
                print(f"    SG iter={n_iters * 500} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def freq_domain_refine(a0, time_limit, verbose=True):
    """Gerchberg-Saxton style frequency domain refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute spectrum
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A_fft = rfft(a_pad, fft_size)

        # Target: constant magnitude
        mag = np.abs(A_fft)
        target_mag = np.mean(mag)  # or could use median

        # Update phase, keep target magnitude
        phase = np.angle(A_fft)
        A_new = target_mag * np.exp(1j * phase)
        A_new[0] = np.abs(A_new[0])  # keep DC positive (real)
        A_new[-1] = np.abs(A_new[-1]) if fft_size > 1 else A_new[-1]

        # Inverse FFT
        a_time = irfft(A_new, fft_size)[:n]

        # Enforce non-negativity
        a_time = np.maximum(a_time, 0.0)

        # Renormalize
        s = a_time.sum()
        if s < 1e-15:
            break
        a = a_time * norm / s

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, seed=42, verbose=True):
    """Peak-aware coordinate descent with simultaneous peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    n_top_peaks = 8
    n_top_contrib = 4
    n_candidates = 5

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_top_peaks]
        max_b = b.max()

        improved = False

        # For each top peak, find top contributors and shift mass away
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:n_top_contrib]]

            # Find positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < n_candidates:
                candidates = np.where(np.arange(n) != top_idx[0])[0]
            # Sort by current value (prefer moving to larger positions to maintain structure)
            candidates = candidates[np.argsort(-a[candidates])][:n_candidates]

            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                for dst in candidates:
                    for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        stall = 0
                        break

            if stall % 10 == 0 and stall > 10:
                # Larger perturbation
                n_pert = max(5, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for ix in idx_pert:
                    a_trial[ix] *= rng.uniform(0.3, 2.0)
                if try_update(a_trial):
                    pass

            if stall > 500:
                break

    return best_a, best_score
