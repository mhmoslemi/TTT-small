def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluator."""
    try:
        a_list = [float(x) for x in a]

def normalize_to_sum(a, target_sum):
    """Normalize a to have the given target sum, maintaining non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def lse_gradient_step(a, T, norm_val):
    """
    One LSE gradient step. Returns (new_a, gradient, conv, w, score).
    a is normalized to sum = norm_val = sqrt(2n).
    """
    n = len(a)
    size = 2 * n - 1
    
    conv = fast_conv(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a.copy(), np.zeros(n), conv, np.ones(size)/size, max_conv
    
    shifted = (conv - max_conv) / T
    exp_conv = np.exp(shifted)
    w = exp_conv / exp_conv.sum()
    
    # Compute gradient: grad[j] = 2 * sum_i w[j+i] * a[i]
    # Using convolution: g = convolve(a, w)[n-1 : 2n-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = rfft(a, L)
    W_fft = rfft(w, L)
    conv_aw = irfft(A_fft * W_fft, L)[:n + size - 1]
    g = conv_aw[n - 1:2 * n - 1]
    grad = 2.0 * g
    
    return a, grad, conv, w, max_conv


def lse_optimize(a0, time_limit, verbose=True):
    """
    LSE smoothed gradient descent with Nesterov momentum.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm_val)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    n_stages = 25
    temps = np.geomspace(100.0, 0.02, n_stages)
    
    # Nesterov state
    y = a.copy()
    v = np.zeros(n)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Number of iterations at this temperature
        # More iterations at lower T (exploitation)
        frac = si / n_stages
        n_iters = int(max(20, min(300, remaining * (1.0 + 2.0 * frac))))
        
        # Learning rate: larger at high T
        base_eta = 1.0 / (T * n * 0.5)
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at current y (Nesterov lookahead point)
            _, grad, conv, w, max_conv = lse_gradient_step(y, T, norm_val)
            
            # Nesterov momentum
            momentum = 0.9  # could adapt
            v = momentum * v + (1.0 - momentum) * grad
            # Capped momentum for stability
            v_norm = np.linalg.norm(v)
            if v_norm > 10.0 * norm_val:
                v = v * 10.0 * norm_val / v_norm
            
            # Step
            eta = base_eta / (1.0 + 0.1 * it)  # slight decay within stage
            y_new = y - eta * v
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y = a.copy()  # reset
            else:
                y_new = y_new * norm_val / s_y
                y = y_new
            
            # Nesterov extrapolation for the next iterate
            mu = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / (2.0 * it + 2.0)
            a_new = y + mu * (y - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a = y.copy()
            else:
                a = a_new * norm_val / s_a
            
            # Track best
            score_y = max_conv  # approximate (at y, not after update)
            # Actually compute at y:
            score_y = fast_conv(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            score_a = fast_conv(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE stage {si:2d} T={T:7.2f} ({elapsed:.0f}s): best={best_score:.8f}")

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T where T[k] @ b = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. a*b[k] <= M for all k, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, norm_val)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    stuck = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck += 1
            if stuck > 30:
                break
            continue
        stuck = 0
        
        g = b * norm_val / b.sum()
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-NS t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

    if window is None:
        window = n  # full window

    best_score = conv.max()

    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            
            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i or a[i] < norm_val * 1e-12:
                        continue
                    
                    # Compute delta convolution for moving mass from i to j
                    # (a*a) change = 2*(a_new * a_old difference)
                    # After: a[i]-=d, a[j]+=d
                    # Delta in (a*a)[k]: 
                    #   -2*d*(a[k-i] + a[k-j]) + d^2*(delta_{k,2i} - 2*delta_{k,i+j} + delta_{k,2j})
                    # approximately (for small d): -2*d*(a[k-i] + a[k-j])
                    
                    # For the search, use a simpler approach:
                    # Try a few values of d and check the max
                    
                    best_d = 0.0
                    best_val = conv.max()
                    
                    # Sample d values
                    n_samples = 10
                    max_d = a[i] * 0.95
                    if max_d < 1e-15:
                        break
                    deltas = np.linspace(0.0, max_d, n_samples)
                    
                    for d in deltas:
                        if d <= 0 or d > a[i]:
                            continue
                        
                        # Update convolution
                        # (a_new * a_new)[k] = (a*a)[k] - 2d*(a[k-i]+a[k-j]) + d^2*...
                        # For simplicity, just compute the change at the max positions
                        # Actually, let's do the full update for accuracy
                        new_a = a.copy()
                        new_a[i] -= d
                        new_a[j] += d
                        s = new_a.sum()
                        new_a = new_a * norm_val / s
                        
                        if d > 0:
                            new_conv = fast_conv(new_a)
                            if new_conv.max() < best_val - 1e-14:
                                best_val = new_conv.max()
                                best_d = d
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        a = a * norm_val / s

                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm_val / a.sum()
                            conv = fast_conv(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

The key insight for efficient CD: when we change `a[i] -= d` and `a[j] += d`, the change in the autocorrelation is:

`delta_conv[k] = (a_new * a_new)[k] - (a * a)[k]`
`= ((a[i]-d)*a[k-i] + (a[j]+d)*a[k-j] + (cross terms with other elements))`
`= -d*a[k-i] - d*a[k-j] + d*(-a[k-i] - a[k-j]) + d^2*(...delta terms...)`

Wait, let me be more careful. The convolution `(a*a)[k] = sum_m a[m]*a[k-m]`.

After the change: `a'[m] = a[m] - d*delta_{m,i} + d*delta_{m,j}`.

`(a'*a')[k] = (a + d*(e_j - e_i)) * (a + d*(e_j - e_i))[k]`
`= (a*a)[k] + d*(a*e_j - a*e_i)[k] + d*(e_j*a - e_i*a)[k] + d^2*(e_j-e_i)*(e_j-e_i)[k]`
`= (a*a)[k] + d*(a[k-j] - a[k-i]) + d*(a[k-j] - a[k-i]) + d^2*(delta_{k,j} - delta_{k,i} - delta_{k,j+i} + delta_{k,i+j})`

Wait, `(a * e_j)[k] = sum_m a[m] * e_j[k-m] = a[k-j]` (if `0 <= k-j < n`).
Similarly `(a * e_i)[k] = a[k-i]`.

So `(a * (d*(e_j - e_i)))[k] = d*(a[k-j] - a[k-i])`.

And `((e_j - e_i) * (e_j - e_i))[k] = delta_{k,2j} - delta_{k,i+j} - delta_{k,i+j} + delta_{k,2i} = delta_{k,2j} - 2*delta_{k,i+j} + delta_{k,2i}`.

So:
`delta_conv[k] = 2d*(a[k-j] - a[k-i]) + d^2*(delta_{k,2j} - 2*delta_{k,i+j} + delta_{k,2i})`

This is O(1) per k (just looking up a[k-i] and a[k-j]), and the delta terms are point updates.

So the full delta_conv can be computed in O(n) time (iterating over k from 0 to 2n-2 and computing `2d*(a[k-j] - a[k-i])` for each k, plus the delta terms).

Then `new_max = max(conv + delta_conv)` can be computed in O(n) as well.

This is much faster than doing a full FFT for each candidate!
