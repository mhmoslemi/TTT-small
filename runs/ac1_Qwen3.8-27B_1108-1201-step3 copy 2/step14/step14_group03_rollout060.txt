def coordinate_descent(a, time_limit, rng, verbose=True):
    n = len(a)
    size = 2*n - 1
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    
    # Compute initial convolution
    c = fast_autocorr(a)
    best_score = c.max() / s**2
    best_a = a.copy()
    
    t0 = time.monotonic()
    n_passes = 0
    
    # Grid of relative changes
    t_grid = np.array([-0.9, -0.7, -0.5, -0.3, -0.1, 0.0, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0])
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-15:
                # Maybe set to small positive
                continue
            
            # v[k] = a[k-i] for valid k
            v = np.zeros(size)
            # k-i is in [0, n-1] when k is in [i, i+n-1]
            k_lo = i
            k_hi = min(size, i + n)
            v[k_lo:k_hi] = a[:k_hi-k_lo]
            
            # e_{2i}
            e_2i = np.zeros(size)
            if 2*i < size:
                e_2i[2*i] = 1.0
            
            # Candidate d values
            d_candidates = ai * t_grid
            d_candidates = d_candidates[d_candidates > -ai + 1e-15]  # Keep a[i] >= 0
            if len(d_candidates) == 0:
                continue
            
            # Evaluate objective for each candidate
            # c_new_j = c + 2*d_j*v + d_j^2 * e_2i
            # s_new_j = s + d_j
            # score_j = max(c_new_j) / s_new_j^2
            
            # Vectorized: d_candidates has shape (J,)
            # c + 2*d_j*v has shape (J, size)
            c_new = c[None, :] + 2.0 * d_candidates[:, None] * v[None, :] + (d_candidates**2)[:, None] * e_2i[None, :]
            s_new = s + d_candidates
            s_new = np.maximum(s_new, 1e-15)
            scores = c_new.max(axis=1) / (s_new**2)
            
            best_j = np.argmin(scores)
            best_d = d_candidates[best_j]
            best_score_new = scores[best_j]
            
            current_score = c.max() / s**2
            if best_score_new < current_score - 1e-15:
                # Apply update
                a[i] += best_d
                c = c_new[best_j]
                s = s + best_d

            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def get_solver():
    global _GLOBAL_SOLVER
    if _GLOBAL_SOLVER is not None:
        return _GLOBAL_SOLVER
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, warm_start=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0
    last_b = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, warm_start=last_b)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        last_b = b.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=True, fine=False):
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')

    c = fast_autocorr(a)
    best_score = c.max() / s**2
    best_a = a.copy()

    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    if fine:
        t_grid = np.array([-0.95, -0.6, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.6, 1.5])
    else:
        t_grid = np.array([-0.95, -0.5, -0.2, -0.05, 0.05, 0.2, 0.5, 1.0, 3.0])

    J = len(t_grid)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-18:
                continue

            v = np.zeros(size)
            k_lo = i
            k_hi = min(size, i + n)
            v[k_lo:k_hi] = a[:k_hi - k_lo]

            e_2i_val = 1.0 if 2 * i < size else 0.0

            d_candidates = ai * t_grid
            valid = d_candidates > -ai + 1e-18
            d_candidates = d_candidates[valid]
            if len(d_candidates) == 0:
                continue

            Jc = len(d_candidates)
            c_new = c[None, :] + 2.0 * d_candidates[:, None] * v[None, :]
            if e_2i_val > 0:
                c_new[:, 2 * i] += d_candidates**2

            s_new = s + d_candidates
            s_new = np.maximum(s_new, 1e-15)
            scores = c_new.max(axis=1) / (s_new**2)

            best_j = np.argmin(scores)
            best_d = d_candidates[best_j]
            best_score_new = scores[best_j]

            current_score = c.max() / s**2
            if best_score_new < current_score - 1e-16:
                a[i] += best_d
                if a[i] < 0:
                    a[i] = 0
                    best_d = -ai
                    a[i] = 0
                    c = fast_autocorr(a)
                    s = a.sum()
                else:
                    c = c_new[best_j]
                    s = s_new

                if best_score_new < best_score - 1e-16:
                    best_score = best_score_new
                    best_a = a.copy()
                    pass_improved = True

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

    return best_a, best_score


def lse_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(50.0, 0.5, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_conv = conv.max()
            shifted = (conv - max_conv) / T
            shifted = np.clip(shifted, -500, 500)
            exp_conv = np.exp(shifted)
            w = exp_conv / exp_conv.sum()

            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)
            cross1 = irfft(W_fft * np.conj(A_fft), fft_size)
            g1 = np.zeros(n)
            for i in range(n):
                k_lo = max(0, i)
                k_hi = min(n, i + n)
                if k_hi > k_lo:
                    g1[i] = np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])

            g2 = np.zeros(n)
            for i in range(n):
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(n, i + 1)
                if k_hi2 > k_lo2:
                    g2[i] = np.dot(w[k_lo2:k_hi2], a[i - np.arange(k_lo2, k_hi2)])

            grad = 2.0 * (g1 + g2)

            step = norm / (T * max(1, it)) / (1.0 + 0.05 * it)
            step = min(step, norm * 0.1 / n)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_perturb_and_cd(a0, time_limit, rng, n_perturbs=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    per_perturb_time = time_limit / (n_perturbs + 1)

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit - 5:
            break
        noise_scale = 0.05 / np.sqrt(n)
        a_perturbed = a + rng.standard_normal(n) * noise_scale * norm
        a_perturbed = np.maximum(a_perturbed, 0.0)
        sp = a_perturbed.sum()
        if sp < 1e-15:
            continue
        a_perturbed = a_perturbed * norm / sp

        remaining = min(per_perturb_time, time_limit - (time.monotonic() - t0))
        a_out, sc = coordinate_descent(a_perturbed, remaining, rng, verbose=False, fine=True)
        sc_norm = fast_autocorr(a_out).max()
        if sc_norm < best_score:
            best_score = sc_norm
            best_a = a_out.copy()
            a = a_out.copy()
            elapsed = time.monotonic() - t0
            print(f"    Perturb {p}: improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick coordinate descent to clean up parent (80s)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Quick CD n={n_base} (budget={remaining * 0.05:.0f}s) ===")
    t1 = remaining * 0.05
    if t1 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t1, rng, verbose=True, fine=False)
        do_try_update(a_out, "cd1")

    # Phase 2: LP Nesterov - main workhorse (45% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={remaining * 0.45:.0f}s) ===")
    t2 = remaining * 0.45
    if t2 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # Phase 3: CD after LP (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: CD after LP n={n_best} (budget={remaining * 0.15:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t3, rng, verbose=True, fine=True)
        do_try_update(a_out, f"cd2_n{n_best}")

    # Phase 4: LP Nesterov round 2 (20% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_best} (budget={remaining * 0.20:.0f}s) ===")
    t4 = remaining * 0.20
    if t4 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Random perturb + CD (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Perturb+CD n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_perturb_and_cd(a_norm, t5, rng, n_perturbs=4)
        do_try_update(a_out, f"perturb_n{n_best}")

    # Phase 6: LSE polish (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t6 = remaining * 0.10
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_polish(a_norm, t6, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 7: Final fine CD (remainder)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final fine CD n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, remaining, rng, verbose=True, fine=True)
        do_try_update(a_out, f"cd_final_n{n_best}")

    # Final: try alternate length if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n_base + 100
        print(f"\n=== Alt length n={n_alt} (budget={remaining * 0.4:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        t_alt = min(remaining * 0.4, 120.0)
        if t_alt > 20:
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t_alt, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20 and best_seq is not None and len(best_seq) == n_alt:
            a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
            a_out_alt2, _ = coordinate_descent(a_norm_alt2, min(remaining2 * 0.5, 40.0), rng, verbose=False, fine=True)
            do_try_update(a_out_alt2, f"cd_n{n_alt}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
