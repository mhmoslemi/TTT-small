    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Use a tight target
        eps = min(0.05, 0.005 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        
        # Get best response

            b = lp_best_response(y, M_curr, n)
            if b is None or b.sum() < 1e-15:
                break

        g = b * norm_val / b.sum()
        n_iters += 1

        alpha = 2.0 / (t + 2.0)

        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        a_new = y_new.copy()
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        a = a_new

        if score_y < best_score:

            best_a = y.copy()
            last_improve = time.monotonic()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

**Alternative approach: Optimal control / ODE viewpoint**

Think of `a` as a function on [0,1] (discretized). The convolution `a*a` at the "center" (lag n-1) is like the integral of `a(x)*a(1-x)` dx. We want this to be small while keeping `integral a = sqrt(2n)`.

**Alternative: Fourier domain approach**

In the Fourier domain, convolution becomes multiplication. If `A(f) = FFT(a)`, then `FFT(a*a) = A(f)^2` (with appropriate scaling). The max of `a*a` in the time domain is related to the L1 norm of `A(f)^2` in the frequency domain (by the convolution theorem in reverse).

So if we can make `|A(f)|` small for all f (except maybe f=0), we can bound the max.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_constraint_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    if T is None:
        T = build_constraint_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def multi_peak_subgradient_optimize(a0, time_limit, n_peaks=5, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 1.0
    patience = 0
    last_improve = time.monotonic()
    run_a = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ab = fast_autocorr(run_a)
        top_idx = np.argsort(ab)[::-1][:n_peaks]
        top_vals = ab[top_idx]
        total_top = top_vals.sum()
        if total_top < 1e-15:
            break
        weights = top_vals / total_top

        subgrad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    subgrad[j] += 2.0 * w * run_a[idx]

        g_mean = subgrad.mean()
        g_proj = subgrad - g_mean

        step = step_size / np.sqrt(t + 1)
        a_new = run_a - step * g_proj
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = run_a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            last_improve = time.monotonic()
            patience = 0
        else:
            patience += 1

        run_a = a_new

        if patience > 800:
            patience = 0
            step_size *= 0.7
            if step_size < 0.01:
                break

        if time.monotonic() - last_improve > 50.0:
            break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a * norm_val, best_score

def targeted_coord_descent(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ab = fast_autocorr(a)
        k_peak = int(np.argmax(ab))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) > 0:
            contrib = a[idx] * a[k_peak - idx]
            sorted_idx = idx[np.argsort(contrib)[::-1]]
            top_contrib = sorted_idx[:20]
        else:
            top_contrib = np.array([k_peak % n])

        top_peaks = np.argsort(ab)[::-1][:12]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets[:10]:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a * norm_val, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - x) ** 0.6
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-0.5 * ((x - 0.35) / 0.25) ** 2) + 0.4 * np.exp(-0.5 * ((x - 0.65) / 0.2) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP-Nesterov main (40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_curr} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.38
    if solver is not None and t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, eps_base=0.02, eps_max=0.05)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp1_n{n_curr}", t_global)

    # Phase 2: Multi-peak subgradient (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: Multi-peak subgrad n={n_curr} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = multi_peak_subgradient_optimize(a_norm, t2, n_peaks=5, verbose=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"subgrad_n{n_curr}", t_global)

    # Phase 3: LP-Nesterov round 2 with different params (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: LP-Nest round2 n={n_curr} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.15
    if solver is not None and t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, eps_base=0.03, eps_max=0.08)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp2_n{n_curr}", t_global)

    # Phase 4: Coordinate descent polish (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_curr} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t4, seed=12345)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"coord1_n{n_curr}", t_global)

    # Phase 5: Try longer sequence (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_curr + 300
        print(f"\n=== Phase 5: LP-Nest longer n={n_long} (budget={remaining:.0f}s) ===")
        t5 = remaining * 0.55
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        if solver is not None:
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, eps_base=0.02, eps_max=0.05)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Try shorter sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_short = max(800, n_curr - 400)
        print(f"\n=== Phase 6: LP-Nest shorter n={n_short} (budget={remaining:.0f}s) ===")
        t6 = remaining * 0.5
        init_short = interpolate_sequence(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        if solver is not None:
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, eps_base=0.02, eps_max=0.05)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_short_n{n_short}", t_global)

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_curr = len(best_seq)
        print(f"\n=== Phase 7: Final polish n={n_curr} (budget={remaining:.0f}s) ===")
        third = remaining / 3.0
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, third, seed=99999)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final1", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20 and solver is not None:
            n_curr2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_curr2) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm2, remaining2, solver, verbose=False, eps_base=0.015, eps_max=0.04)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lp_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
