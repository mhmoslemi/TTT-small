def coordinate_descent(a0, time_limit, rng=None, verbose=True):
    """Single-coordinate descent with exact convex line search.
    For coordinate i, the new autocorrelation is conv + d*A + d^2*B where
    A[k]=2*a[k-i] for valid k, B[2i]=1. The max over k is convex in d,
    so ternary search finds the optimum.
    """
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    if rng is None:
        rng = np.random.default_rng(42)

    conv = fast_autocorr(a).copy()
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        pass_start_max = conv.max()

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]

            ai = a[i]
            if ai < norm * 1e-12:
                continue

            k_lo = i
            k_hi = min(size, i + n)
            length = k_hi - k_lo

            a_slice = a[k_lo - i: k_hi - i]
            affected = conv[k_lo: k_hi]

            has_quad = (2 * i >= k_lo) and (2 * i < k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            # Precompute unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = conv[:k_lo].max()
            if k_hi < size:
                unaff_max = max(unaff_max, conv[k_hi:].max())

            # Determine search bounds
            d_lo = -ai
            # Upper bound: ensure quad term doesn't blow up
            if has_quad:
                current_max = max(affected.max(), unaff_max)
                disc = ai * ai - (conv[2 * i] - current_max)
                if disc > 0:
                    d_hi = -ai + np.sqrt(disc)
                else:
                    d_hi = ai * 0.5
            else:
                d_hi = ai

            if d_hi <= d_lo + 1e-14:
                d_hi = d_lo + ai * 0.1 if ai > 0 else 0.0

            # Ternary search (25 iterations gives ~1e-5 relative precision)
            tl = d_lo
            th = d_hi
            for _ in range(25):
                tm1 = tl + (th - tl) / 3.0
                tm2 = th - (th - tl) / 3.0

                aff1 = affected + 2.0 * a_slice * tm1
                if has_quad:
                    aff1[quad_idx] += tm1 * tm1
                m1 = max(aff1.max(), unaff_max)

                aff2 = affected + 2.0 * a_slice * tm2
                if has_quad:
                    aff2[quad_idx] += tm2 * tm2
                m2 = max(aff2.max(), unaff_max)

                if m1 < m2:
                    th = tm2
                else:
                    tl = tm1

            d_opt = (tl + th) / 2.0

            # Also check d=0 and d=-ai as candidates
            candidates = [d_opt, 0.0, d_lo]
            best_d = 0.0
            best_val = max(affected.max(), unaff_max)
            for d_c in candidates:
                if d_c < d_lo - 1e-14 or d_c > d_hi + 1e-14:
                    continue
                aff_c = affected + 2.0 * a_slice * d_c
                if has_quad:
                    aff_c[quad_idx] += d_c * d_c
                m_c = max(aff_c.max(), unaff_max)
                if m_c < best_val - 1e-13:
                    best_val = m_c
                    best_d = d_c

            if best_d != 0.0:
                a[i] += best_d
                new_aff = affected + 2.0 * a_slice * best_d
                if has_quad:
                    new_aff[quad_idx] += best_d * best_d
                conv[k_lo: k_hi] = new_aff
                pass_improved = True

        # Normalize after pass
        s = a.sum()
        if s < 1e-15:
            break
        scale = norm / s
        a *= scale
        conv *= scale * scale

        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_adam(a0, time_limit, verbose=True):
    """LSE smooth approximation minimized with Adam optimizer and temperature annealing."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute indices for gradient
    i_idx = np.arange(n)[:, None]
    l_idx = np.arange(n)[None, :]
    indices = i_idx + l_idx
    valid_mask = indices < size

    # Adam state
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    # Multi-stage annealing
    stages = [
        (3.0, 0.5, 2000),
        (0.5, 0.15, 2000),
        (0.15, 0.04, 3000),
        (0.04, 0.01, 3000),
    ]

    base_lr = 0.05 * norm / n

    for stage_idx, (T_start, T_end, n_steps) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break

        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Cosine schedule within stage
            progress = step / max(n_steps - 1, 1)
            T = T_start + (T_end - T_start) * (1 - np.cos(np.pi * progress)) / 2.0

            conv = fast_autocorr(a)
            max_conv = conv.max()

            # Compute weights (stable)
            shifted = (conv - max_conv) / T
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()

            # Gradient: grad[i] = 2 * sum_l w[i+l] * a[l]
            w_mat = np.where(valid_mask, w[indices], 0.0)
            grad = 2.0 * (w_mat @ a)

            # Adam update
            t_adam += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            lr = base_lr / (1.0 + 0.01 * t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            # Check improvement periodically
            if step % 50 == 0 or step == n_steps - 1:
                score = fast_autocorr(a).max()
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            score_now = fast_autocorr(a).max()
            if score_now < best_score:
                best_score = score_now
                best_a = a.copy()
            print(f"    LSE-Adam stage={stage_idx} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                continue

            # Pick j within window
            offset = rng.integers(1, window + 1)
            direction = rng.integers(0, 2)
            if direction == 0:
                j = (i + offset) % n
            else:
                j = (i - offset) % n
            if j == i:
                j = (i + 1) % n
            if a[j] >= norm * 0.9:
                continue

            # Compute delta_vec for transferring mass from i to j
            # delta = 2*(contribution_of_j - contribution_of_i) in the autocorrelation
            delta_vec = np.zeros(size)

            # Contribution of a[i] to autocorrelation (to be subtracted)
            lo1 = i
            hi1 = min(size, i + n)
            delta_vec[lo1:hi1] -= 2.0 * a[i + np.arange(hi1 - lo1) - i]
            # Wait, this isn't right. Let me think again.
            # If we decrease a[i] by d and increase a[j] by d:
            # Change in conv[k] = d * (2*a[j]_contribution[k] - 2*a[i]_contribution[k]) + d^2 * (B_j[k] - B_i[k])
            # where a[i]_contribution[k] = a[k-i] if valid (the coefficient of a[i] in conv[k])
            # Actually: conv[k] = sum_j a[j]*a[k-j]
            # d(conv[k])/da[i] = 2*a[k-i] (if valid) [as computed before]
            # So change = d * (2*a[k-j] - 2*a[k-i]) + d^2 * (B_j[k] - B_i[k])
            # where B_i[k] = 1 if k=2i, B_j[k] = 1 if k=2j

            # For simplicity, use the A vectors
            A_i = np.zeros(size)
            A_j = np.zeros(size)
            B_i = np.zeros(size)
            B_j = np.zeros(size)

            k_lo_i = max(0, i)
            k_hi_i = min(size, i + n)
            A_i[k_lo_i:k_hi_i] = 2.0 * a[k_lo_i - i: k_hi_i - i]
            if 2 * i < size:
                B_i[2 * i] = 1.0

            k_lo_j = max(0, j)
            k_hi_j = min(size, j + n)
            A_j[k_lo_j:k_hi_j] = 2.0 * a[k_lo_j - j: k_hi_j - j]
            if 2 * j < size:
                B_j[2 * j] = 1.0

            dA = A_j - A_i
            dB = B_j - B_i

            # Minimize max(conv + d*dA + d^2*dB) over d in [-a[i], a[i]*0.9]
            d_lo = -a[i]
            d_hi = a[i] * 0.9

            # Unaffected max (where dA=0 and dB=0, the value doesn't change)
            # Actually, dA is non-zero in [k_lo_i, k_hi_i) ∪ [k_lo_j, k_hi_j)
            # For simplicity, just use full vector
            tl, th = d_lo, d_hi
            for _ in range(15):
                tm1 = tl + (th - tl) / 3.0
                tm2 = th - (th - tl) / 3.0

                new_c1 = conv + tm1 * dA + tm1 * tm1 * dB
                m1 = new_c1.max()

                new_c2 = conv + tm2 * dA + tm2 * tm2 * dB
                m2 = new_c2.max()

                if m1 < m2:
                    th = tm2
                else:
                    tl = tm1

            d_opt = (tl + th) / 2.0

            # Check d_opt vs d=0
            new_c0 = conv  # d=0
            best_val = new_c0.max()
            best_d = 0.0

            new_co = conv + d_opt * dA + d_opt * d_opt * dB
            val_o = new_co.max()
            if val_o < best_val - 1e-12:
                best_val = val_o
                best_d = d_opt

            # Also check d = -a[i] (zero out coordinate i)
            new_cz = conv + d_lo * dA + d_lo * d_lo * dB
            val_z = new_cz.max()
            if val_z < best_val - 1e-12:
                best_val = val_z
                best_d = d_lo

            if best_d != 0.0:
                a[i] -= best_d
                a[j] += best_d
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[i] += best_d
                    a[j] -= best_d

        # Normalize
        s = a.sum()
        if s < 1e-15:
            break
        a *= norm / s
        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = evaluate_safe(p)
            print(f"Parent: n={len(p)}, score={p_score:.6f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.6f}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov (35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Coordinate Descent with exact line search (35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.35
    print(f"\n=== Phase 2: CD exact line search n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t2, rng=rng, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # Phase 3: LSE-Adam (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LSE-Adam n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_adam(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: CD again after LSE (12% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.12
    print(f"\n=== Phase 4: CD round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, rng=rng, verbose=False)
        do_try_update(a_out, f"cd2_n{n_best}")

    # Phase 5: Pair CD refinement (8% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.08
    print(f"\n=== Phase 5: Pair CD n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_refine(a_norm, t5, rng, window=30, verbose=False)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 6: Perturbation + CD restart (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Perturbation + CD (budget={remaining:.0f}s) ===")
        n_restarts = 3
        time_per_restart = remaining / n_restarts

        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 10:
                break
            # Perturb
            a_pert = best_seq.copy()
            noise_scale = 0.02 * a_pert.sum() / n_best
            noise = rng.normal(0, noise_scale, size=n_best)
            a_pert = np.maximum(a_pert + noise, 0.0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert = a_pert / s

            # CD on perturbed
            t_r = min(time_per_restart, time_budget - (time.monotonic() - t_global))
            if t_r > 10:
                a_norm_r = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
                a_out_r, _ = coordinate_descent(a_norm_r, t_r * 0.6, rng=rng, verbose=False)
                do_try_update(a_out_r, f"pert_cd_{r}_n{n_best}")

                remaining_r = time_budget - (time.monotonic() - t_global)
                if remaining_r > 10 and best_seq is not None and len(best_seq) == n_best:
                    a_norm_r2 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
                    a_out_r2, _ = coordinate_descent(a_norm_r2, min(remaining_r * 0.4, t_r * 0.4), rng=rng, verbose=False)
                    do_try_update(a_out_r2, f"pert_cd2_{r}_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
