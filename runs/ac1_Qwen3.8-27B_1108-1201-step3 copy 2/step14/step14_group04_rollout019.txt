        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            a_new = b.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa
        
        y = y_new
        a = a_new
        
        # Check scores
        sc_y = fast_conv(y).max()
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv(a).max()
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()
        
        # Ergodic average
        avg += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                sc_avg = fast_conv(avg_check).max()
                if sc_avg < best_sc:
                    best_sc = sc_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): lp={t_lp:.8f} best={best_sc:.8f}")
    
    return best_a, best_sc

def local_peak_shave(a0, time_limit, seed=42):
    """Local refinement: reduce autocorrelation peaks via coordinate moves."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv(a).max()
    a = a / s
    
    best_a = a.copy()
    best_sc = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        # Find top peaks
        top_idx = np.argsort(b)[::-1][:5]
        improved = False
        
        for k in top_idx:
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Try moving mass from src to a low position
            # Avoid positions that contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_idx:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < n // 4:
                candidates = np.where(np.arange(n) != src)[0]
            # Pick from lowest values
            candidates = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 10)]
            
            for _ in range(3):
                dst = candidates[rng.integers(len(candidates))]
                for frac in [0.001, 0.005, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] - amt < 0:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = a_t / a_t.sum()
                    sc = fast_conv(a_t).max()
                    if sc < best_sc - 1e-12:
                        a = a_t
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                if a_t[i] >= 0:
                    a_t = a_t / a_t.sum()
                    sc = fast_conv(a_t).max()
                    if sc < best_sc:
                        a = a_t
                        best_sc = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break
    
    return best_a, best_sc

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

def lse_fista(a0, time_limit, seed=0):
    """LSE-FISTA warmup for fast initial improvement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_sc = fast_conv(a).max()
    y = a.copy()
    staging = a.copy()
    t_fista = 1
    t0 = time.monotonic()
    
    for T in [10.0, 3.0, 1.0, 0.3, 0.1]:
        L = n / (T * 2.0)
        while time.monotonic() - t0 < time_limit:
            ac = fast_conv(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                break
            shifted = (ac - ac_max) / T
            shifted = np.clip(shifted, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            
            # Gradient of LSE w.r.t. y
            N = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N)
            Y_rev = rfft(y[::-1], N)
            cross = irfft(W * Y_rev, N)
            grad = 2.0 * cross[n - 1: n - 1 + n]
            
            step = 1.0 / L
            z = y - step * grad
            a_new = np.maximum(z, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / sn
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                sy = y_new.sum()
                if sy < 1e-15:
                    y_new = a_new.copy()
                else:
                    y_new = y_new / sy
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if t_fista % 200 == 0:
                sc = fast_conv(a).max()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
    
    sc = fast_conv(a).max()
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc
