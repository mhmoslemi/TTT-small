def build_T(a, n):
    """Build the convolution constraint matrix for given a."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lse_smoothed_obj_grad(x, n, T_temp):
    """LSE-smoothed objective and gradient for the normalized problem."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE approximation of max
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    lse_val = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    
    # Gradient via FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    # Objective: 2n * lse_val / s^2
    obj_val = 2.0 * n * lse_val / (s * s)
    
    # Gradient: need to be careful
    # d/dx [2n * lse / s^2] = 2n * (cross * s^2 - lse * 2s * a) / s^4
    # = 2n * (cross - 2 * lse * a / s) / s^2
    # Hmm, let me re-derive.
    # F = 2n * L / S^2 where L = lse(b), S = sum(a)
    # dL/da_i = sum_k w_k * db[k]/da_i = cross[i] (the cross-correlation)
    # dS/da_i = 1
    # dF/da_i = 2n * (cross[i] * S^2 - L * 2S) / S^4
    # = 2n * (cross[i] * S - 2L) / S^3
    
    grad = 2.0 * n * (cross * s - 2.0 * lse_val) / (s * s * s)
    return obj_val, grad

    n = len(a0)
    norm_val = np.sqrt(2.0 * n)

        a = a * norm_val / s0

        def obj_func(x, T_val=T):
            val, _ = lse_smoothed_obj_grad(x, n, T_val)
            return val

            _, g = lse_smoothed_obj_grad(x, n, T_val)
            return g

                                  jac=grad_func,

def simulated_annealing_polish(a0, time_limit, seed=42):
    """SA polish with peak-targeted moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T_start = 0.05
    T_end = 0.001
    total_iters = 5000
    stall = 0
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Compute current peak
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Find top contributors to peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_k = 10
        top_contrib = idx[np.argsort(contrib)[::-1]][:top_k]
        
        # Find low-value positions
        low_val = np.argsort(a)[:10]
        
        # Choose a move: transfer from high-contrib to low-value
        if rng.random() < 0.6:
            src = top_contrib[rng.integers(len(top_contrib))]
            dst = low_val[rng.integers(len(low_val))]
        else:
            src = rng.integers(n)
            dst = rng.integers(n)
        
        if src == dst or a[src] < 1e-10:
            continue
        
        frac = rng.uniform(0.01, 0.15)
        amt = a[src] * frac
        
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial * norm_val / s_t
        
        score = fast_autocorr(a_trial).max()
        
        # Accept or reject (SA criterion)
        delta = score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            a = a_trial.copy()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 200:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP best-response with adaptive tight M and Nesterov acceleration (primary)
2. LSE (log-sum-exp) smoothed gradient with L-BFGS-B for continuous refinement
3. Simulated annealing with peak-targeted perturbations for local polish
4. Multi-length exploration via interpolation
Key improvements: adaptive M aggressiveness, class-based callbacks (no closures),
better time allocation, and SA escape from local optima.
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_obj_grad(x, n, T_temp):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    lse_val = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    obj_val = 2.0 * n * lse_val / (s * s)
    grad = 2.0 * n * (cross * s - 2.0 * lse_val) / (s * s * s)
    return obj_val, grad


class LSECallback:
    def __init__(self, n, T_temp):
        self.n = n
        self.T_temp = T_temp

    def __call__(self, x):
        return self.obj(x)[0]

    def obj(self, x):
        return lse_obj_grad(x, self.n, self.T_temp)

    def grad(self, x):
        return lse_obj_grad(x, self.n, self.T_temp)[1]


def lse_optimize(a0, time_limit, T_vals):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for T in T_vals:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        cb = LSECallback(n, T)

        def obj_func(x):
            return cb.obj(x)[0]

        def grad_func(x):
            return cb.grad(x)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-8, 'maxls': 40})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                sc = a_c.sum()
                if sc > 1e-10:
                    a_c = a_c * norm_val / sc
                    score_c = fast_autocorr(a_c).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_c.copy()
                        a = a_c.copy()
            print(f"    LSE T={T:.1f}: best={best_score:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")
    return best_a, best_score


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    eps_base = 0.025

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break
        time_since_improve = time.monotonic() - last_improve
        if time_since_improve > 30:
            eps_base = min(0.06, eps_base * 1.25)
        else:
            eps_base = max(0.015, eps_base * 0.98)
        eps = min(eps_base, 0.004 * math.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        T = build_T(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score - 1e-10:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
        if time.monotonic() - last_improve > 50.0:
            stall_count += 1
            if stall_count > 4:
                break
        else:
            stall_count = 0
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score
                        :.8f} eps={eps:.4f}")
    return best_a, best_score


def sa_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    cur_score = best_score
    t0 = time.monotonic()
    T_curr = 0.03
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 5000.0)
        T_curr = 0.03 * (0.003 / 0.03) ** progress
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_k = 12
        top_contrib = list(idx[np.argsort(contrib)[::-1]][:top_k])
        low_val = list(np.argsort(a)[:12])

        if rng.random() < 0.55:
            src = top_contrib[rng.integers(len(top_contrib))]
            dst = low_val[rng.integers(len(low_val))]
        else:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))

        if src == dst or a[src] < 1e-10:
            continue
        frac = rng.uniform(0.01, 0.12)
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial * norm_val / s_t
        score = fast_autocorr(a_trial).max()
        delta = score - cur_score
        if delta < 0 or (T_curr > 1e-8 and rng.random() < math.exp(-delta / T_curr)):
            a = a_trial.copy()
            cur_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall = 1
        else:
            stall += 1
        if stall > 300:
            break
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + 40.0 * x * (1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP best-response with Nesterov as primary, LSE gradient as secondary,
    SA and coordinate descent for polish, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Phase 1: LP-Nest primary (40% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    if a_out is not None and a_out.sum() > 0.01:
        sc = evaluate_safe(a_out / a_out.sum())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out / a_out.sum()
            print(f"  *** NEW BEST: {sc:.6f} (n={n_best})")

    # Phase 2: LSE gradient refinement (15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 180.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LSE n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t2, [60.0, 30.0, 15.0, 8.0, 4.0])
        if a_out is not None and a_out.sum() > 0.01:
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out / a_out.sum()
                print(f"  *** NEW BEST: {sc:.6f} (n={n_best})")

    # Phase 3: LP-Nest round 2 (12%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 180.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nest r2 n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        if a_out is not None and a_out.sum() > 0.01:
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out / a_out.sum()
                print(f"  *** NEW BEST: {sc:.6f} (n={n_best})")

    # Phase 4: Coordinate descent (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CoordDesc n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t4, seed=777)
        if a_out is not None and a_out.sum() > 0.01:
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out / a_out.sum()
                print(f"  *** NEW BEST: {sc:.6f} (n={n_best})")

    # Phase 5: Try different length (12%)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_long = n_best + 400 if n_best < 2500 else n_best - 200
        n_long = max(800, min(n_long, 3000))
        t5 = min(remaining * 0.55, 250.0)
        if t5 > 60:
            print(f"\n=== Phase 5: LP-Nest n={n_long} (budget={t5:.0f}s) ===")
            init_long = interpolate_seq(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
            if a_out is not None and a_out.sum() > 0.01:
                sc = evaluate_safe(a_out / a_out.sum())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out / a_out.sum()
                    print(f"  *** NEW BEST: {sc:.6f} (n={n_long})")

    # Phase 6: SA polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        n_best = len(best_seq)
        t6 = remaining - 5
        print(f"\n=== Phase 6: SA n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = sa_polish(a_norm, t6, seed=54321)
        if a_out is not None and a_out.sum() > 0.01:
            sc = evaluate_safe(a_out / a_out.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out / a_out.sum()
                print(f"  *** NEW BEST: {sc:.6f} (n={n_best})")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
