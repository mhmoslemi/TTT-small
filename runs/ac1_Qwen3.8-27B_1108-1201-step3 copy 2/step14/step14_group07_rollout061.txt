            obj, grad = lse_objective_and_grad(a, n, T)
            if not np.isfinite(obj):
                break
            grad_proj = grad - np.mean(grad)
            if np.max(np.abs(grad_proj)) < 1e-12:
                break
            a_trial, alpha = backtracking_line_search(a, grad_proj, n, T)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
            a = a_trial
            iters_done += 1
            if iters_done > 5000:
                break
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE PG stage {si+1}/{n_stages} (T={T:.2f}): best={best_score:.8f}, iters={iters_done} ({elapsed:.0f}s)")
    return best_a

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T_mat[idx, i] = a[idx - i]
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated Frank-Wolfe with LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
        if t > 100 and (t % 500 == 0):
            if fast_autocorr(y).max() > best_score * 1.01:
                break
    return best_a

def coordinate_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Intensive gradient-guided coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        k_star = np.argmax(c)
        # Compute gradient at this point (subgradient using k_star)
        h = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                h[j] = 2.0 * a[m]
        S = a.sum()
        grad = 2.0 * n * (h / (S * S) - 2.0 * c[k_star] * a / (S * S * S))
        # Normalize gradient for ranking
        grad_rank = grad - np.mean(grad)
        # Find best (src, dst) pairs: decrease high-gradient, increase low-gradient
        src_candidates = np.argsort(grad_rank)[::-1][:30]  # high gradient (decrease these)
        dst_candidates = np.argsort(grad_rank)[:30]  # low gradient (increase these)
        improved = False
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-14:
                        continue
                    if try_update(a_trial):
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try peak-targeted moves
            top_peaks = np.argsort(c)[::-1][:8]
            for k_peak in top_peaks:
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[k_peak - idx]
                top_idx = idx[np.argsort(contrib)[::-1][:5]]
                for src in top_idx:
                    if a[src] < 1e-12:
                        continue
                    # Move to a position that doesn't contribute to this peak
                    peak_mask = np.zeros(n, dtype=bool)
                    for k2 in top_peaks[:5]:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        peak_mask[js2:je2 + 1] = True
                    candidates = np.where(~peak_mask)[0]
                    if len(candidates) == 0:
                        candidates = np.where(np.arange(n) != src)[0]
                    # Prefer candidates with low gradient
                    cand_sorted = candidates[np.argsort(grad_rank[candidates])][:10]
                    for dst in cand_sorted:
                        for frac in [0.005, 0.01, 0.03, 0.05, 0.1]:
                            amt = a[src] * frac
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            if a_trial[src] < -1e-14:
                                continue
                            if try_update(a_trial):
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random moves
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break
            if stall % 50 == 0:
                # Perturbation
                n_perturb = max(3, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.4, 1.6)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        best_a = a_trial.copy()
                        best_score = score
                        a = a_trial
            if stall > 500:
                break
    return best_a

def simulated_annealing(a0, time_limit, seed=42, verbose=False):
    """SA with coordinate moves and periodic re-optimization."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T_start = best_score * 0.01
    T_end = best_score * 0.0001
    n_total_steps = 50000
    step_num = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        T = T_start * (T_end / T_start) ** (step_num / max(1, n_total_steps))
        # Propose a move: change 1-3 random elements
        n_changes = rng.integers(1, 4)
        idx_changes = rng.choice(n, size=n_changes, replace=False)
        a_trial = a.copy()
        for idx in idx_changes:
            a_trial[idx] *= rng.uniform(0.5, 1.5)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            step_num += 1
            continue
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            step_num += 1
            continue
        a_trial = a_trial / s_t
        score = fast_autocorr(a_trial).max()
        delta = score - best_score
        if delta < 0:
            a = a_trial
            best_a = a_trial.copy()
            best_score = score
            no_improve = 0
        else:
            if T > 1e-15 and np.random.random() < np.exp(-delta / (T + 1e-15)):
                a = a_trial
            no_improve += 1
        step_num += 1
        # Periodic re-optimization
        if no_improve > 2000 and no_improve % 2000 == 0:
            a = best_a.copy()
            no_improve = 0
            # Quick LSE polish
            for T_lse in [5.0, 2.0, 1.0]:
                for _ in range(50):
                    if time.monotonic() - t0 > time_limit:
                        break
                    obj, grad = lse_objective_and_grad(a, n, T_lse)
                    if not np.isfinite(obj):
                        break
                    grad_proj = grad - np.mean(grad)
                    # Simple step
                    alpha = 0.1 / (n * 0.1 + 1)
                    a_new = a - alpha * grad_proj
                    a_new = np.maximum(a_new, 0.0)
                    s = a_new.sum()
                    if s > 1e-10:
                        a_new = a_new / s
                        sc = fast_autocorr(a_new).max()
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_new.copy()
                            no_improve = 0
                        a = a_new
    return best_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(12.0 * np.pi * x) - 0.05 * np.cos(24.0 * np.pi * x)
    elif choice == 8:
        # Random smooth
        a = np.ones(n)
        for _ in range(20):
            center = rng.integers(0, n)
            width = rng.integers(10, n // 4)
            for i in range(n):
                d = abs(i - center)
                if d < width:
                    a[i] += rng.uniform(-0.5, 0.5) * np.exp(-0.5 * (d / (width / 3.0)) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def propose_candidate(budget_s=1800.0):

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, eval_score={p_score:.6f}")

            best_eval_score = p_score

    if best_seq is None:

        best_eval_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, eval_score={best_eval_score:.6f}")

    n_base = len(best_seq)

    def do_try_update(a, label):

            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_eval_score:
            best_eval_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global

    # Phase 2: LP Frank-Wolfe with Nesterov

    if remaining > 60:
        n_cur = len(best_seq)

        for delta in [-100, -50, 50, 100, 200]:

            if 800 <= nl <= 2500 and nl not in lengths_to_try:

            do_try_update(a_out, f"len{nl}")

        do_try_update(a_out, "final_cd")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)

        print(f"\nFINAL: eval_score={best_eval_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Clamp to valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
