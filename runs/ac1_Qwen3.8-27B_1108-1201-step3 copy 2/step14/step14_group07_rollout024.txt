def build_sparse_T(a, n):
    """Build sparse constraint matrix T where T[k,j] = a[k-j] if valid."""
    size = 2 * n - 1
    rows = []
    cols = []

        rows.extend(idx.tolist())

    return csr_matrix((vals, (rows, cols)), shape=(size, n))

    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm_val / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Focus on top peaks
        peak_indices = np.argsort(b)[::-1][:8]
        improved = False

        for k_peak in peak_indices:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:4]]

            # Find safe destinations (away from all top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_indices[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            safe_dst = np.where(~peak_mask)[0]
            if len(safe_dst) < 10:
                safe_dst = np.where(np.arange(n) != peak_indices[0])[0]

            for src in top_src:
                if a[src] < 1e-12:
                    continue
                # Move mass from src to safe destinations
                candidates = safe_dst[np.argsort(-a[safe_dst])[:min(10, len(safe_dst))]]
                for dst in candidates:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        if try_update(trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random moves to escape
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                if try_update(trial):
                    break
            if stall > 200:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick scan of nearby lengths
    print("\n=== Phase 0: Multi-length scan ===")
    n_parent = len(best_seq) if best_seq is not None else 1600
    scan_lengths = sorted(set([n_parent, max(800, n_parent - 300), n_parent + 300, n_parent + 600]))
    scan_lengths = [n for n in scan_lengths if 600 <= n <= 3500]
    scan_results = {}

    for i_scan, n_scan in enumerate(scan_lengths):
        if time.monotonic() - t_global > time_budget * 0.10:
            break
        remaining_scan = time_budget * 0.10 - (time.monotonic() - t_global)
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - i_scan), 80.0)
        if scan_time < 10:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = normalize_a(init_seq, n_scan)
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: local_score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")
        do_update(a_out, f"scan_n{n_scan}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP at best length (main workhorse - 40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining*0.40:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = normalize_a(best_seq, n_best)
    a_out, _ = nesterov_lp_optimize(a_norm, t1, verbose=True)
    do_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: LSE optimization (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE n={n_best} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_norm = normalize_a(best_seq, n_best)
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=40.0, T_end=2.0, n_stages=5)
        do_update(a_out, f"lse_n{n_best}")

    # Phase 3: Try longer sequence (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 500
    print(f"\n=== Phase 3: LP n={n_long} (budget={min(remaining*0.15, 250):.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = normalize_a(init_long, n_long)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True)
        do_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Noisy restart LP (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP noisy n={n_best} (budget={min(remaining*0.10, 150):.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_noisy = best_seq + rng.normal(0, 0.03 * best_seq.mean(), n_best)
        a_noisy = normalize_a(a_noisy, n_best)
        a_out, _ = nesterov_lp_optimize(a_noisy, t4, verbose=False)
        do_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 5: Coordinate descent polish (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={min(remaining*0.10, 150):.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_norm = normalize_a(best_seq, n_best)
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        do_update(a_out, "coord_polish")

    # Phase 6: LSE low-temp polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE low-temp n={n_best} ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_norm = normalize_a(best_seq, n_best)
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=15.0, T_end=1.0, n_stages=4)
        do_update(a_out, "lse_lowtemp")

    # Phase 7: Final coordinate descent with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 10:
        a_norm = normalize_a(best_seq, n_best)
        a_out, _ = coordinate_descent_polish(a_norm, remaining, seed=77777)
        do_update(a_out, "coord_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
