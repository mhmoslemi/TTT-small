def build_toeplitz_matrix(a):
    """Build the convolution matrix T where (T @ b)[k] = conv(a,b)[k]"""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):
            if k < size:
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

def lp_best_response(a, M, T_matrix, solver):
    """Solve max sum(b) s.t. T*b <= M, b >= 0"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_matrix @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    grad = 2.0 * n * grad_conv / (s * s) - 4.0 * n * lse * a / (s ** 3)

def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = remaining / max(1, n_stages - si)

        x0 = a.copy()
        
        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val
        
        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score

def peak_direction_cd(a0, time_limit, seed=42):
    """Coordinate descent using analytical peak-reduction directions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    iterations = 0
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        b = fast_autocorr(a)
        
        # Find the top peaks
        top_peaks = np.argsort(b)[::-1][:8]
        max_peak = b[top_peaks[0]]
        
        improved = False
        
        # For each top peak, find the best mass transfer
        for k_peak in top_peaks[:4]:
            # Find which pairs (i, k_peak-i) contribute most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            
            if contrib.sum() < 1e-20:
                continue
                
            # Top contributing positions
            top_src = idx[np.argsort(contrib)[::-1][:8]]
            
            # Build peak mask (positions that affect any of the top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Try moving mass from top contributors to non-peak positions
            for src in top_src:
                if a[src] < 1e-14:
                    continue
                
                # Best destinations: minimize the "partner" value at the peak
                # Moving from src to dst changes peak by: -a[src]*a[k_peak-src] + a[k_peak-dst]*(change)
                # We want dst where a[k_peak-dst] is small (or dst is outside the peak range)
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(peak_mask)[0]
                
                # Score candidates by how much they'd reduce the peak
                # For dst, the change in peak k_peak is approximately:
                # delta = -a[src]*a[k_peak-src]/a[src] + a[k_peak-dst]*amt/a[src]
                # We want to minimize this, so prefer dst where a[k_peak-dst] is small
                scores = np.full(n, np.inf)
                valid_dst = np.zeros(n, dtype=bool)
                for dst in candidates:
                    change = 0
                    # Effect on peak k_peak
                    if 0 <= k_peak - src < n and k_peak - src != src:
                        pass  # src is already accounted for
                    if 0 <= k_peak - dst < n:
                        change += a[k_peak - dst]
                    else:
                        change += 0  # no effect on this peak
                    # Effect on other peaks
                    for k2 in top_peaks[1:4]:
                        if 0 <= k2 - dst < n:
                            change += 0.3 * a[k2 - dst]
                    scores[dst] = change
                    valid_dst[dst] = True
                
                best_dsts = candidates[np.argsort(scores[candidates])[:4]]
                
                for dst in best_dsts:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 3 == 0:
                n_pert = max(2, n // 200)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    a_trial[idx] = max(0, a_trial[idx] * rng.uniform(0.7, 1.3))
                if try_update(a_trial):
                    pass
            
            if stall % 10 == 0:
                # More aggressive perturbation
                n_pert = max(5, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    a_trial[idx] = max(0, a_trial[idx] * rng.uniform(0.5, 1.5))
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm / s_t
                    if try_update(a_trial):
                        pass
            
            if stall > 200:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        # Two-peak structure
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2) + 0.1
    elif choice == 8:
        # Cosine window with modulation
        x = np.linspace(0.0, 1.0, n)
        a = (1 + np.cos(np.pi * x)) * (0.8 + 0.2 * np.sin(6 * np.pi * x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

- To reduce it, we need to "spread out" the mass so that the products `a[i]*a[k-i]` are smaller

One key insight: **the peak is often at lag = n-1 (the center)**, where `conv(a,a)[n-1] = sum_i a[i]*a[n-1-i]`. This is the "autocorrelation at zero" if we think of the sequence as a signal.

So the real constraint is balancing the center peak with the off-center peaks. This is a multi-peak problem.

The entropy-regularized problem:

The solution is `b[i] = exp(epsilon * (1 - sum_j T[j,i]*lambda[j]))` where lambda are the dual variables.

def build_toeplitz_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        for j in range(n):
            k = i + j
            if k < size:
                rows[pos] = k
                cols[pos] = i
                vals[pos] = a[j]
                pos += 1
    rows = rows[:pos]
    cols = cols[:pos]
    vals = vals[:pos]
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def lp_best_response(a, M, T_matrix, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_matrix @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_obj_grad(x, n, T_temp):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    scaled = T_temp * b
    sm = scaled.max()
    exp_vals = np.exp(scaled - sm)
    exp_sum = exp_vals.sum()
    lse_val = (sm + np.log(exp_sum)) / T_temp
    obj = 2.0 * n * lse_val / (s * s)
    w = exp_vals / exp_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    A = rfft(a_pad, fft_size)
    W = rfft(w_pad, fft_size)
    cross = irfft(A * np.conj(W), fft_size)[:n]
    conv_aw = irfft(A * W, fft_size)[:n]
    grad_conv = cross + conv_aw
    grad = 2.0 * n * grad_conv / (s * s) - 4.0 * n * lse_val * a / (s ** 3)
    return obj, grad.ravel()

                for di in best_order:
                    dst = candidates[di]
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        at = np.maximum(a_trial, 0.0)
                        st = at.sum()
                        if st < 1e-10:
                            continue
                        at = at * norm / st
                        sc = fast_autocorr(at).max()
                        if sc < best_score - 1e-14:
                            a = at
                            best_a = at.copy()
                            best_score = sc
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                n_pert = max(3, n // 100)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for ix in idx_p:
                    a_trial[ix] = max(0.0, a_trial[ix] * rng.uniform(0.6, 1.4))
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial * norm / st
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = sc
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 12))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-15 * (x - 0.3) ** 2) + np.exp(-15 * (x - 0.7) ** 2) + 0.05
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 + np.cos(np.pi * x)) * (0.85 + 0.15 * np.sin(6.0 * np.pi * x))
    elif choice == 9:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.5 * np.sin(np.pi * x) + 0.2 * np.sin(2.0 * np.pi * x)
    elif choice == 10:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2) * (1.0 + 0.4 * np.cos(8.0 * np.pi * x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_global(a, state):
    """Update global best if a is better. state is a dict with best_seq, best_score, t_global."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        elapsed = time.monotonic() - state['t_global']
        label = state.get('label', 'update')
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-start exploration -> deep LP refinement -> peak CD -> LSE polish.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    state = {'best_seq': None, 'best_score': float('inf'), 't_global': t_global, 'label': 'init'}

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state['best_seq'] = parent.copy()
            state['best_score'] = p_score
    except Exception:
        parent = None

    n_base = len(state['best_seq']) if state['best_seq'] is not None else 1600

    # Phase 1: Multi-start exploration (20% budget)
    print(f"\n=== Phase 1: Multi-start exploration (base_n={n_base}) ===")
    p1_budget = min(time_budget * 0.22, 350.0)
    n_starts = 8
    other_ns = [n_base - 100, n_base + 100, n_base + 200]

    inits = []
    if state['best_seq'] is not None:
        # Start with parent at base length
        inits.append(('parent', n_base, state['best_seq']))
        # Interpolated to other lengths
        for nn in other_ns:
            ip = interpolate_sequence(state['best_seq'], nn)
            inits.append((f'interp_{nn}', nn, ip))
    # Add structured inits
    for i in range(4):
        nn = n_base if i % 2 == 0 else other_ns[i % len(other_ns)]
        si = structured_init(nn, rng)
        inits.append((f'struct_{i}_{nn}', nn, si))
    # Add a noisy version of parent
    if state['best_seq'] is not None:
        noisy = state['best_seq'] + rng.normal(0, 0.03 * state['best_seq'].mean(), n_base)
        noisy = np.maximum(noisy, 0.0)
        noisy = noisy / noisy.sum()
        inits.append(('noisy_parent', n_base, noisy))

    # Run short LP on each init
    time_per_start = p1_budget / len(inits)
    time_per_start = max(time_per_start, 10.0)
    time_per_start = min(time_per_start, 60.0)

    best_start_a = None
    best_start_score = float('inf')

    for i_init, (label, nn, init) in enumerate(inits):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.25:
            break
        remaining_p1 = p1_budget - (time.monotonic() - t_global)
        if remaining_p1 < 8:
            break
        t_alloc = min(time_per_start, remaining_p1, 50.0)
        if t_alloc < 8:
            break

        a_norm = init * np.sqrt(2.0 * nn) / init.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t_alloc, solver, verbose=False)
        print(f"    [{label}] n={nn}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

        if sc_out < best_start_score:
            best_start_score = sc_out
            best_start_a = a_out.copy()

        state['label'] = label
        try_update_global(a_out, state)

    # Phase 2: Deep LP refinement on best (38% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(state['best_seq'])
    print(f"\n=== Phase 2: Deep LP n={n_current} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.38, 600.0)
    if t2 > 20:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_current) / state['best_seq'].sum()
        state['label'] = 'lp_deep'
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        try_update_global(a_out, state)

    # Phase 3: Peak-direction CD (18% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(state['best_seq'])
    print(f"\n=== Phase 3: Peak CD n={n_current} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 15:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_current) / state['best_seq'].sum()
        state['label'] = 'peak_cd'
        a_out, _ = peak_cd(a_norm, t3, seed=12345)
        try_update_global(a_out, state)

    # Phase 4: LSE polish (15% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(state['best_seq'])
    print(f"\n=== Phase 4: LSE polish n={n_current} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_current) / state['best_seq'].sum()
        state['label'] = 'lse_polish'
        a_out, _ = lse_optimize(a_norm, t4, T_start=50.0, T_end=3.0, n_stages=5)
        try_update_global(a_out, state)

    # Phase 5: Second LP with different noise (12% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(state['best_seq'])
    print(f"\n=== Phase 5: LP noisy2 n={n_current} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        noise_scale = 0.015 * state['best_seq'].mean()
        a_noisy = state['best_seq'] + rng.normal(0, noise_scale, n_current)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_current) / a_noisy.sum()
        state['label'] = 'lp_noisy2'
        a_out, _ = nesterov_lp_optimize(a_noisy, t5, solver, verbose=False)
        try_update_global(a_out, state)

    # Phase 6: Final peak CD on best (rest of time minus safety margin)
    remaining = time_budget - (time.monotonic() - t_global) - 30.0
    n_current = len(state['best_seq'])
    print(f"\n=== Phase 6: Final peak CD n={n_current} (budget={max(0,remaining):.0f}s) ===")
    if remaining > 10:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_current) / state['best_seq'].sum()
        state['label'] = 'peak_cd_final'
        a_out, _ = peak_cd(a_norm, remaining, seed=99999)
        try_update_global(a_out, state)

    # Final
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs) and fs < state['best_score']:
            state['best_score'] = fs
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in state['best_seq']]
