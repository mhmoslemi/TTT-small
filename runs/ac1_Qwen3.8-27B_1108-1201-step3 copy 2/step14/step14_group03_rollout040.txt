def lp_best_response(a, M, solver):

def peak_targeted_cd(a0, time_limit, rng, verbose=True):

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]

        contributing = []
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributing.append((i, j, a[i] * a[j]))
        contributing.sort(key=lambda x: -x[2])

        top_pairs = contributing[:min(20, len(contributing))]

        for (i, j, prod) in top_pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-8:
                continue

            src_idx = i if a[i] >= a[j] else j
            dst_candidates = []
            for d in range(1, min(n // 4 + 1, 50)):
                for direction in [1, -1]:
                    cand = (src_idx + direction * d) % n
                    if cand != src_idx and a[cand] < norm * 0.3 / n:

            rng.shuffle(dst_candidates)
            tried = 0
            for dst in dst_candidates[:15]:
                tried += 1
                if tried > 15:
                    break
                if dst == src_idx:
                    continue

                eps = a[src_idx] * 0.3
                a_try = a.copy()
                a_try[src_idx] -= eps
                a_try[dst] += eps
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                c_try = fast_autocorr(a_try)
                if c_try.max() < best_score - 1e-10:
                    a = a_try
                    conv = c_try

                    pass_improved = True

                    break

        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0

            elapsed = time.monotonic() - t0
            print(f"    PCDW pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Momentum LSE optimization (fast, 8% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.08
    print(f"\n=== Phase 1: Momentum LSE n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = momentum_lse_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"mgd_n{n_base}")

    # Phase 2: LP Nesterov with adaptive M (main, 48% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.48
    print(f"\n=== Phase 2: LP Nesterov adaptive n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize_v2(a_norm, t2, solver, verbose=True, m_scale=0.95)
        do_try_update(a_out, f"lpv2_n{n_best}")

    # Phase 3: Peak-targeted CD (22% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.22
    print(f"\n=== Phase 3: Peak-targeted CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeted_cd(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"ptcd_n{n_best}")

    # Phase 4: Pair CD window (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Pair CD window n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t4, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcdw_n{n_best}")

    # Phase 5: Second LP Nesterov round (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize_v2(a_norm, t5, solver, verbose=True, m_scale=0.97)
        do_try_update(a_out, f"lpv2b_n{n_best}")

    # Phase 6: Try alternate length (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        n_alt = n_base + 200
        t6 = remaining * 0.5
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
        if t6 > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out_alt, _ = nesterov_lp_optimize_v2(a_norm_alt, min(t6, remaining * 0.4), solver, verbose=False, m_scale=0.95)
            do_try_update(a_out_alt, f"lpv2_n{n_alt}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 15 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = pair_cd_window(a_norm_alt2, min(remaining2 * 0.5, 20.0), rng, window=20, verbose=False)
                do_try_update(a_out_alt2, f"pcdw_n{n_alt}")

    # Final safety: ensure we return within time
    elapsed_total = time.monotonic() - t_global
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed_total:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
