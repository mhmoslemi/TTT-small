def build_conv_matrix(a):

    T = np.zeros((size, n))

def mirror_descent_lse(a0, time_limit, T_start=15.0, T_end=0.5, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    N = 1 << (3 * n - 2).bit_length()
    n_iters = max(5000, int(time_limit * 50))
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(it / n_iters, 1.0)
        T = T_start * (T_end / T_start) ** progress
        A_fft = rfft(a, N)
        conv = irfft(A_fft * A_fft, N)[:size]
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        exp_conv = np.exp((conv - max_conv) / T)
        w = exp_conv / exp_conv.sum()
        W_fft = rfft(w, N)
        X = irfft(np.conj(A_fft) * W_fft, N)
        grad = X[:n].copy()
        grad *= 2.0 / T
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-15:
            break
        eta = 1.0 / (grad_max * (1.0 + 0.5 * it))
        eta = min(eta, 2.0)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def lse_lsqb_objective(x, n, T):
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    conv = irfft(A * A, N)[:size]
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0
    shifted = (conv - max_c) / T
    m = shifted.max()
    lse = m + np.log(np.sum(np.exp(shifted - m)))
    return lse + max_c / T


def lse_lsqb_grad(x, n, T):
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    size = 2 * n - 1
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    conv = irfft(A * A, N)[:size]
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    exp_conv = np.exp((conv - max_c) / T)
    w = exp_conv / exp_conv.sum()
    W = rfft(w, N)
    X = irfft(np.conj(A) * W, N)
    raw_grad = X[:n].copy()
    raw_grad *= 2.0 / T
    dot = np.sum(raw_grad * a)
    dx = a * (raw_grad - dot)
    return dx

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 12

                        if 2 * i < size:

                        ss = a.sum()
                        if ss < 1e-15:

                            continue
                        a = a * norm / ss
                        conv = fast_autocorr(a)

                        if cur_max < best_score - 1e-13:

        if not pass_improved:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Mirror descent (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.15
    print(f"\n=== Phase 1: Mirror Descent n={n1} ({t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = mirror_descent_lse(best_seq, t1, T_start=10.0, T_end=0.5, verbose=True)
        do_try_update(a_out, f"md_n{n1}")

    # Phase 2: L-BFGS-B with LSE (30% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = remaining * 0.30
    print(f"\n=== Phase 2: LSE-LBFGS n={n2} ({t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lbfgsb_lse_optimize(best_seq, t2, T_stages=[4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25], verbose=True)
        do_try_update(a_out, f"lbfgs_n{n2}")

    # Phase 3: LP Nesterov (30% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = remaining * 0.30
    print(f"\n=== Phase 3: LP Nesterov n={n3} ({t3:.0f}s) ===")
    if t3 > 20:
        norm3 = np.sqrt(2.0 * n3)
        a_norm3 = best_seq * norm3 / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm3, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 4: Pair CD (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Pair CD n={n4} ({t4:.0f}s) ===")
    if t4 > 10:
        norm4 = np.sqrt(2.0 * n4)
        a_norm4 = best_seq * norm4 / best_seq.sum()
        a_out, _ = pair_cd(a_norm4, t4, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd_n{n4}")

    # Phase 5: Second LP Nesterov round (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n5} ({t5:.0f}s) ===")
    if t5 > 15:
        norm5 = np.sqrt(2.0 * n5)
        a_norm5 = best_seq * norm5 / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm5, t5, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n5}")

    # Phase 6: Final pair CD + alternate length (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n6 = len(best_seq)
        t6a = remaining * 0.5
        t6b = remaining * 0.5
        print(f"\n=== Phase 6a: Final Pair CD n={n6} ({t6a:.0f}s) ===")
        if t6a > 10:
            norm6 = np.sqrt(2.0 * n6)
            a_norm6 = best_seq * norm6 / best_seq.sum()
            a_out, _ = pair_cd(a_norm6, t6a, rng, window=40, verbose=True)
            do_try_update(a_out, f"pcd2_n{n6}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            n_alt = n_base + 150
            print(f"\n=== Phase 6b: Alternate length n={n_alt} ({remaining * 0.7:.0f}s) ===")
            t_alt = remaining * 0.7
            init_alt = interpolate_sequence(best_seq, n_alt)
            # Quick LP optimize at new length
            norm_alt = np.sqrt(2.0 * n_alt)
            a_alt = init_alt * norm_alt / init_alt.sum()
            t_alt_lp = min(t_alt * 0.6, 120.0)
            if t_alt_lp > 15:
                a_out_alt, _ = nesterov_lp_optimize(a_alt, t_alt_lp, solver, verbose=False)
                do_try_update(a_out_alt, f"lp_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and best_seq is not None and len(best_seq) == n_alt:
                norm_alt2 = np.sqrt(2.0 * n_alt)
                a_alt2 = best_seq * norm_alt2 / best_seq.sum()
                a_out_alt2, _ = pair_cd(a_alt2, min(remaining * 0.7, 60.0), rng, window=20, verbose=False)
                do_try_update(a_out_alt2, f"pcd_n{n_alt}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
