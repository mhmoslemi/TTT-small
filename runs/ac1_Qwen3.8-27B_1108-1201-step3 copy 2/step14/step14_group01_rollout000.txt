        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

                b2 = lp_solve(y, M, solver, active_new)

                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        y_new = np.maximum(y_new, 0.0)

        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        sc_y = compute_max_conv(y_new)
        # Nesterov restart: if objective increased significantly, reset momentum
        if t > 10 and sc_y > M * 1.05:

        else:
            momentum = (t - 1.0) / (t + 2.0)

        if t == 1:

        if verbose and t % 50 == 0:

def lse_gd(a0, time_limit, verbose=True):

    s = a.sum()

    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(30.0, 0.2, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(150, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                continue
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))

def coord_descent(a0, time_limit, rng, verbose=True):

            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:

            # The gradient w.r.t. a[i] points to where reducing a[i] helps most
            # Find j that maximizes the "transfer benefit"
            # Benefit of transferring delta from i to j:
            # Reduces conv by delta * (contribution of a[i] at peak positions)
            # Increases conv by delta * (contribution of a[j] at peak positions)

            # Simple heuristic: pick j that is far from the peak positions
            peak_idx = np.argsort(conv)[-5:]  # top 5 peak positions
            best_j = -1
            best_metric = float('inf')

                if j == i:
                    continue

                metric = 0.0

                    # Contribution of a[j] to conv[p]: a[max(0,min(n-1,p-j))] + a[max(0,min(n-1,j-p))]
                    k1 = p - j
                    k2 = j - p
                    c1 = a[k1] if 0 <= k1 < n else 0
                    c2 = a[k2] if 0 <= k2 < n else 0
                    metric += c1 + c2
                if metric < best_metric:
                    best_metric = metric
                    best_j = j

            if best_j < 0:
                continue
            j = best_j
            aj = a[j]

            # Binary search for optimal delta
            best_d = 0.0
            best_v = cur_max
            lo, hi = 0.0, ai * 0.8
            for _ in range(8):

                a[i] = ai - mid

                v = compute_max_conv(a)
                a[i] = ai
                a[j] = aj
                a = a * norm / a.sum()
                if v < best_v:

                    hi = mid

        if verbose and (idx + 1) % 200 == 0:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
