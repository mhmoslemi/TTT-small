def fast_autocorr(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    """Build the (2n-1) x n convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    a = np.asarray(a)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""

    try:

        return None

        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:

    return b_val

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final: also check the average of all iterates
    if n_avg > 0:
        avg_final = avg_a / n_avg
        s_f = avg_final.sum()
        if s_f > 1e-15:
            avg_final = avg_final * norm / s_f
            score_f = fast_autocorr(avg_final).max()
            if score_f < best_score:
                best_score = score_f
                best_a = avg_final.copy()

    return best_a, best_score


def soft_max_gradient_descent(a0, time_limit, verbose=True):
    """Polish using soft-max approximation of max with gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Temperature schedule
    n_stages = 8
    temps = np.geomspace(5.0, 0.1, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(50, int(remaining * 3))

        # FFT setup for gradient computation
        a_rev = a[::-1].copy()
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            max_conv = conv.max()
            
            # Compute softmax weights
            log_w = (conv - max_conv) / T
            log_w -= log_w.max()  # numerical stability
            w = np.exp(log_w)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

            # Compute gradient: grad[j] = 2 * sum_i a[i] * w[j+i]
            # Using FFT: cross-correlation of a and w
            a_rev_pad = np.zeros(fft_size)
            a_rev_pad[:n] = a_rev
            w_pad = np.zeros(fft_size)
            w_pad[:size] = w
            
            A_rev_fft = rfft(a_rev_pad, fft_size)
            W_fft = rfft(w_pad, fft_size)
            cross = irfft(A_rev_fft * W_fft, fft_size)
            grad = 2.0 * cross[n - 1 : 2 * n - 1]

            # Step size: proportional to T (Lipschitz constant ~ 1/T)
            step = T * 0.5 / (n + 1)
            
            # Gradient step
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s

            # Check improvement
            score_new = fast_autocorr(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            
            # Always accept (for the smooth approximation)
            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SoftMax T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent: transfer mass between pairs of positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Try transferring to nearby positions
            for offset in range(1, min(window, n) + 1):
                j = (i + offset) % n
                if j == i:
                    continue
                if a[i] < norm * 1e-10:
                    break

                # Compute delta in convolution if we move mass d from i to j
                delta_vec = np.zeros(size)
                
                # Effect of reducing a[i] by d: conv change is -d * (a shifted by i) in both directions
                # conv(a,a)[k] includes terms a[i]*a[k-i] and a[k-i]*a[i]
                # Reducing a[i] by d changes conv by -d*(a[k-i] + a[i-k]) for valid indices
                # i.e., -d * (a shifted to position i) - d * (a reversed shifted)
                
                # For the full convolution, reducing a[i] by d:
                # delta[k] -= d * a[k-i] (for 0<=k-i<n)
                # delta[k] -= d * a[i-k] (for 0<=i-k<n, i.e., k in [i-n+1, i])
                
                # Adding d to a[j]:
                # delta[k] += d * a[k-j] (for 0<=k-j<n)
                # delta[k] += d * a[j-k] (for 0<=j-k<n, i.e., k in [j-n+1, j])
                
                # Also: the a[i]*a[i] term changes: -2d*a[i] + d^2 (at k=2i... wait)
                # Actually, the cross terms a[i]*a[j] and a[j]*a[i] also change.
                # Let me just compute the full delta numerically for correctness.
                
                # Save original
                orig_i = a[i]
                orig_j = a[j]
                
                # Apply change
                a[i] = orig_i - d  # we'll try values of d
                # Actually, let me just compute the delta vector analytically
                
                # The change in conv when a[i] -> a[i]-d and a[j] -> a[j]+d:
                # New conv[k] - old conv[k] = 
                #   -d*a[k-i] - d*a[i-k] + d*a[k-j] + d*a[j-k]  (cross terms with other elements)
                #   + d^2 (at k = i+j, from the new a[j]*a[i] cross... wait this is getting complicated)
                
                # Simpler: just compute new conv directly for a few candidate d values
                break  # Skip to simpler approach below

            # Simpler approach: try a few d values
            for offset in range(1, min(window, n) + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Try moving mass from i to j
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.9, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        a[i] -= d
                        a[j] += d
                        s = a.sum()
                        a = a * norm / s  # renormalize
                        new_conv = fast_autocorr(a)
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d
                        a[j] -= d
                        a[i] += d
                        a = a * norm / a.sum()
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-14:
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            # Revert
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
