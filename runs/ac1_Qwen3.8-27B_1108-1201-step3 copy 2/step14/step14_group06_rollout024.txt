    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a, n):
    """Build the (2n-1) x n matrix T where T @ b = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    
    # Try to get dual variables
    dual = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value)
        if dual_val is not None:
            dual = np.maximum(np.asarray(dual_val, dtype=np.float64), 0.0)
    except:
        pass
    
    s = b_val.sum()
    if s < 1e-15:
        return None, dual
    return b_val, dual


def lp_best_response_adaptive(a, M_base, solver, max_trials=5):
    """Find the tightest M for which LP gives a meaningful solution."""
    n = len(a)
    T = build_conv_matrix(a, n)
    
    for trial in range(max_trials):
        frac = 1.0 - 0.01 * (trial + 1)
        M = M_base * frac
        if M < 1e-15:
            break
        b, dual = lp_best_response(a, M, solver, T)
        if b is not None:
            # Check if solution is "interesting" (different from a)
            a_norm = a / a.sum()
            b_norm = b / b.sum()
            correlation = np.dot(a_norm, b_norm) / (np.linalg.norm(a_norm) * np.linalg.norm(b_norm))
            if correlation < 0.999:
                return b, M, dual
            # Too similar, try tighter M
    # Fallback: use M_base
    b, dual = lp_best_response(a, M_base, solver, T)
    return b, M_base, dual


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with adaptive M."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    patience = 90.0  # seconds without improvement before stalling

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Build constraint matrix (expensive, but needed for LP)
        T = build_conv_matrix(y, n)
        
        # Adaptive M: try to tighten
        b, M_used, dual = lp_best_response_adaptive(y, M_curr, solver, max_trials=4)
        
        if b is None:
            # Fallback: use current M
            b, dual = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break
            M_used = M_curr

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov step size with adaptation
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Averaging trick
        if n_iters % 30 == 0:
            # Compute running average of recent a's
            pass  # Skip for simplicity

        # Stalling detection
        if time.monotonic() - last_improve > patience:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    Stall at iter={n_iters}, best={best_score:.10f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): M_curr={M_curr:.6f}, M_used={M_used:.6f}, best={best_score:.10f}")

    return best_a, best_score


def spectral_peak_shaving(a, time_limit, seed=42):
    """Use spectral analysis to identify and reduce peak-contributing components."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        M = b[k_peak]

        # Find which elements contribute most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        
        # Get top contributors
        top_k = min(20, len(idx))
        top_idx = idx[np.argsort(contrib)[-top_k:]]
        
        # Find elements not in top peaks (safe to increase)
        top_peaks = np.argsort(b)[::-1][:8]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        safe_targets = non_peak[np.argsort(a[non_peak])[:min(20, len(non_peak))]]

        improved = False
        for src in top_idx[:10]:
            if a[src] < 1e-10:
                continue
            for dst in safe_targets[:10]:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=4):
    """Optimize using log-sum-exp smoothed objective with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute FFT size
    fft_size = 1 << (2 * n - 1).bit_length()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(n_stages - si, 1), 120.0)

        # Define objective and gradient
        def obj_and_grad(x, T_v=T_val):
            a_v = np.maximum(x, 0.0)
            s = a_v.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            b = fast_autocorr(a_v)
            max_b = b.max()
            if max_b < 1e-15:
                return 1e10, np.zeros_like(x)
            shifted = (b - max_b) / T_v
            exp_vals = np.exp(shifted)
            lse = max_b + np.log(exp_vals.sum()) / T_v
            w = exp_vals / exp_vals.sum()
            
            # Gradient via cross-correlation
            w_full = np.zeros(fft_size)
            w_full[:2*n-1] = w
            a_full = np.zeros(fft_size)
            a_full[:n] = a_v
            W = rfft(w_full, fft_size)
            A = rfft(a_full, fft_size)
            cross = irfft(W * np.conj(A), fft_size)[:n]
            
            obj_val = 2.0 * n * lse / (s * s)
            grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a_v / s
            return obj_val, grad

        def obj_func(x):
            val, _ = obj_and_grad(x)
            return val

        def grad_func(x):
            _, g = obj_and_grad(x)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 800,
                                           'gtol': 1e-8, 'ftol': 1e-10,
                                           'maxls': 40})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                s_c = a_c.sum()
                if s_c > 1e-10:
                    a_c = a_c * norm_val / s_c
                    score_c = fast_autocorr(a_c).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_c.copy()
                        a = a_c.copy()
                        print(f"    LSE T={T_val:.1f}: score={score_c:.10f}")
        except Exception as e:
            print(f"    LSE T={T_val:.1f} error: {e}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma))  # Non-symmetric!
    elif choice == 6:
        # Asymmetric: ramp then flat
        x = np.linspace(0.0, 1.0, n)
        a = np.where(x < 0.3, x / 0.3, 1.0 - 0.3 * (x - 0.3))
    else:
        # Non-symmetric exponential
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.0 * x) + 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_solve(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Accelerated LP best-response with adaptive M tightening."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = float(b_y.max())
        if M_curr < 1e-12:
            break

        T = build_conv_matrix(y, n)

        # Adaptive M: try progressively tighter
        b = None
        for frac in [0.005, 0.01, 0.015, 0.025, 0.04, 0.06]:
            M_try = M_curr * (1.0 - frac)
            if M_try < 1e-15:
                break
            b_try = lp_solve(y, M_try, solver, T)
            if b_try is not None:
                # Check if solution is meaningfully different
                a_norm = y / y.sum()
                b_norm = b_try / b_try.sum()
                corr = float(np.dot(a_norm, b_norm) / (np.linalg.norm(a_norm) * np.linalg.norm(b_norm) + 1e-30))
                if corr < 0.9995:
                    b = b_try
                    break
                elif frac == 0.06:
                    b = b_try
        if b is None:
            b = lp_solve(y, M_curr, solver, T)
            if b is None:
                break

        g = b * norm_val / b.sum()
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        M_g = fast_autocorr(g).max()
        if M_curr > 0 and M_g > 0:
            ratio = min(float(M_g) / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            last_improve = time.monotonic()

        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 80.0:
            stall += 1
            if stall > 10:
                if verbose:
                    print(f"    Stall iter={n_iters}, best={best_score:.10f}")
                break
        else:
            stall = 0

        if verbose and n_iters % 50 == 0:
            print(f"    LP it={n_iters} ({time.monotonic()-t0:.0f}s): M={M_curr:.6f} best={best_score:.10f}")

    return best_a, best_score


def peak_coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak contributors."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Peak contributors
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_idx = idx[np.argsort(contrib)[::-1][:20]]

        # Safe targets (not in top peaks)
        top_peaks = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        safe_targets = non_peak[np.argsort(a[non_peak])[:min(20, len(non_peak))]]

        improved = False
        for src in top_idx[:12]:
            if a[src] < 1e-10:
                continue
            for dst in safe_targets[:12]:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.08, 0.12, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-10:
                        continue
                    a_t = a_t * norm_val / s_t
                    sc = float(fast_autocorr(a_t).max())
                    if sc < best_score - 1e-12:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(60):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-10:
                    continue
                a_t = a_t * norm_val / s_t
                sc = float(fast_autocorr(a_t).max())
                if sc < best_score - 1e-12:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 250:
                break

    return best_a, best_score


def lse_objective(x):
    n = _lse_n
    T_val = _lse_T
    fft_size = _lse_fft_size
    a_v = np.maximum(x, 0.0)
    s = a_v.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a_v)
    max_b = float(b.max())
    if max_b < 1e-15:
        return 1e10
    shifted = (b - max_b) / T_val
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T_val
    return 2.0 * n * lse / (s * s)


def lse_gradient(x):
    n = _lse_n
    T_val = _lse_T
    fft_size = _lse_fft_size
    a_v = np.maximum(x, 0.0)
    s = a_v.sum()
    if s < 1e-10:
        return np.zeros_like(x)
    b = fast_autocorr(a_v)
    max_b = float(b.max())
    if max_b < 1e-15:
        return np.zeros_like(x)
    shifted = (b - max_b) / T_val
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T_val
    w = exp_vals / exp_vals.sum()

    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a_v
    W = rfft(w_full, fft_size)
    A = rfft(a_full, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]

    obj_val = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a_v / s
    return grad


def lse_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=4):

    global _lse_n, _lse_T, _lse_fft_size

    best_a = a.copy()

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

        if time.monotonic() - t0 > time_limit:

        _lse_T = float(T_val)
        try:

                                 jac=lse_gradient,
                                 options={'maxiter': 200, 'maxfun': 800,
                                          'gtol': 1e-8, 'ftol': 1e-10, 'maxls': 40})

def structured_init(n, rng):

    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)

    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)

    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)

    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * np.exp(-1.5 * x) + 0.1
    a = np.maximum(a, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

    print(f"\n=== P2: CoordDesc n={nc} ({t2:.0f}s) ===")
    if t2 > 15:
        a_n = best_seq * np.sqrt(2.0 * nc) / best_seq.sum()
        a_out, _ = peak_coord_descent(a_n, t2, seed=seed)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"cd1_n{nc}", t0g)

    # Phase 3: LSE gradient (10%)
    rem = time_budget - (time.monotonic() - t0g)
    nc = len(best_seq)
    t3 = min(rem * 0.10, 140.0)
    print(f"\n=== P3: LSE n={nc} ({t3:.0f}s) ===")
    if t3 > 15:
        a_n = best_seq * np.sqrt(2.0 * nc) / best_seq.sum()
        a_out, _ = lse_optimize(a_n, t3, T_start=50.0, T_end=2.0, n_stages=4)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse1_n{nc}", t0g)

    # Phase 4: LP round 2 (12%)
    rem = time_budget - (time.monotonic() - t0g)
    nc = len(best_seq)
    t4 = min(rem * 0.12, 180.0)
    print(f"\n=== P4: LP-Nest r2 n={nc} ({t4:.0f}s) ===")
    if t4 > 25:
        a_n = best_seq * np.sqrt(2.0 * nc) / best_seq.sum()
        a_out, _ = nesterov_lp(a_n, t4, solver, verbose=False)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp2_n{nc}", t0g)

    # Phase 5: Longer sequence (16%)
    rem = time_budget - (time.monotonic() - t0g)
    n_long = nc + 400
    t5 = min(rem * 0.16, 280.0)
    print(f"\n=== P5: LP-Nest long n={n_long} ({t5:.0f}s) ===")
    if t5 > 50:
        init_l = interpolate_seq(best_seq, n_long)
        a_n = init_l * np.sqrt(2.0 * n_long) / init_l.sum()
        a_out, _ = nesterov_lp(a_n, t5, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lpL_n{n_long}", t0g)

    # Phase 6: Peak shaving final (8%)
    rem = time_budget - (time.monotonic() - t0g)
    nc = len(best_seq)
    t6 = min(rem * 0.10, 140.0)
    print(f"\n=== P6: CoordDesc final n={nc} ({t6:.0f}s) ===")
    if t6 > 15:
        a_n = best_seq * np.sqrt(2.0 * nc) / best_seq.sum()
        a_out, _ = peak_coord_descent(a_n, t6, seed=seed + 9999)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "cd_final", t0g)

    # Phase 7: Final LSE polish (remaining)
    rem = time_budget - (time.monotonic() - t0g)
    if rem > 15:
        nc = len(best_seq)
        t7 = min(rem * 0.9, 150.0)
        print(f"\n=== P7: LSE final n={nc} ({t7:.0f}s) ===")
        a_n = best_seq * np.sqrt(2.0 * nc) / best_seq.sum()
        a_out, _ = lse_optimize(a_n, t7, T_start=30.0, T_end=1.0, n_stages=3)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lse_final", t0g)

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t0g:.0f}s")
    else:
        print("ERROR: no valid sequence!")
        return [1.0] * 100

    return [float(max(0.0, x)) for x in best_seq]
