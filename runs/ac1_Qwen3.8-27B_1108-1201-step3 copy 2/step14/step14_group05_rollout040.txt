    T_red = np.zeros((n, n_half))

    # First term: a[k-h] where k >= h
    idx1 = ks - hs  # (n, n_half), values 0 to n-1 where valid
    mask1 = (ks >= hs) & (idx1 < n)

    idx2 = ks - n + 1 + hs  # (n, n_half)
    mask2 = (hs >= n - 1 - ks) & (idx2 >= 0) & (idx2 < n)

    T_red += np.where(mask2, a[idx2_safe], 0.0)
    
    return T_red
