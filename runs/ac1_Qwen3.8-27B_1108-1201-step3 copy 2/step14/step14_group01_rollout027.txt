def fast_conv_self(a):

def lp_solve_fast(a, M, solver, active):

    val = np.asarray(b_var.value, dtype=np.float64)

def project_simplex_neg(v, s_target):
    """Project v onto {x >= 0, sum(x) = s_target}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), s_target / len(v))
    return v * s_target / s

            break

            break

                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Adaptive step with backtracking
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))
            for trial in range(15):
                a_new = a - step * grad
                a_new = project_simplex_neg(a_new, norm)
                sc = fast_conv_self(a_new).max()
                if sc <= conv.max() + 0.05 * T_val:

                step *= 0.5

            a_new = a - step * grad
            a = project_simplex_neg(a_new, norm)

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

def mirror_descent_steps(a0, time_limit, verbose=True):
    """Mirror descent (entropy-regularized) on the simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Work in log-space for mirror descent
    # a_i = norm * p_i where p is a probability vector
    # MD step: p_i <- p_i * exp(-eta * g_i) / Z
    eta0 = 1.0 / np.sqrt(n)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Subgradient: use empirical distribution over top lags
        top_k = max(1, min(20, int(0.1 * size)))
        top_idx = np.argsort(conv)[-top_k:]
        # Weight by how close to max
        weights = np.exp((conv[top_idx] - max_c) / max(0.1, max_c * 0.1))
        weights = weights / weights.sum()

        # Gradient: g = sum_k w_k * T^T e_k
        # T^T e_k is a shifted version of a
        g = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            # (T^T e_k)[i] = a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            g[lo:hi] += wi * a[k - np.arange(lo, hi)]

        # Mirror descent step
        eta = eta0 / np.sqrt(t)
        p = a / norm  # probability vector
        log_p = np.log(np.maximum(p, 1e-30))
        log_p_new = log_p - eta * g
        log_p_new -= log_p_new.max()  # numerical stability
        p_new = np.exp(log_p_new)
        p_new = p_new / p_new.sum()
        a_new = p_new * norm

        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                hi = min(size, j + n)
                if hi > lo:
                    g_vec[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g_vec[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g_vec[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g_vec[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # 1D minimization with golden section + refinement
                best_d = 0.0
                best_v = cur_max
                n_samples = 25
                deltas = np.linspace(0.01 * ai, 0.99 * ai, n_samples)
                for d in deltas:
                    new_conv = conv + d * g_vec
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                # Refine around best
                if best_d > 0:
                    lo_d = max(0.0, best_d * 0.7)
                    hi_d = min(ai, best_d * 1.3)
                    for d in np.linspace(lo_d, hi_d, 15):
                        new_conv = conv + d * g_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                        v = new_conv.max()
                        if v < best_v:
                            best_v = v
                            best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve == 3:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = 1.0 - 0.5 * np.cos(2 * np.pi * 3 * x) * np.cos(np.pi * x)
    else:
        a = 0.4 + 0.4 * np.sin(np.pi * x) ** 2 + 0.2 * np.cos(2 * np.pi * 7 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 1: Proximal gradient with LSE smoothing (fast initial progress)
    n1 = len(best_seq)
    t1 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 1: PG-LSE n={n1} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = proximal_gradient_lse(best_seq, t1, verbose=True)
        try_update(a_out, f"pglse1_n{n1}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 2: LP Nesterov (main workhorse, first round)
    n2 = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n2}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 3: Mirror descent (different direction)
    n3 = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: Mirror Descent n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = mirror_descent_steps(best_seq, t3, verbose=True)
        try_update(a_out, f"md_n{n3}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 4: Pair CD polish
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd1_n{n4}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 5: LP Nesterov second round (from improved point)
    n5 = len(best_seq)
    t5 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n5}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: Try longer sequences
    n_long_candidates = [n_base + 300, n_base + 600]
    for n_long in n_long_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t6 = min(remaining * 0.20, 100.0)
        print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
        if t6 > 20:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = lp_nesterov_optimize(init_long, min(t6, 80.0), solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = pair_cd_optimize(init_long, min(remaining * 0.4, 40.0), rng, verbose=False)
                try_update(a_out2, f"pcd_long_{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Final alternating polish
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Quick PG step
        t7a = min(remaining * 0.25, 60.0)
        a_out, _ = proximal_gradient_lse(best_seq, t7a, verbose=False)
        try_update(a_out, "pg_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            # Quick CD
            t7b = min(remaining * 0.4, 50.0)
            a_out, _ = pair_cd_optimize(best_seq, t7b, rng, verbose=False)
            try_update(a_out, "pcd_final")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                # One more LP round
                t7c = min(remaining * 0.5, 40.0)
                a_out, _ = lp_nesterov_optimize(best_seq, t7c, solver, verbose=False)
                try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
