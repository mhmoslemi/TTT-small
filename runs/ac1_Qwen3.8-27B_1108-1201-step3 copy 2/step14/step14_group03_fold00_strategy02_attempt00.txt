    if rng is None:
        rng = np.random.RandomState(42)
    n = len(sequence)
    # Identify boundary points (local extremes)

    boundaries = [i for i in range(seq_size) if 
                  (sequence[i] > 0.99*max(sequence)) or 
                  (sequence[i] < 0.015*min(sequence))]
    
    # Apply specific geometric transformations
    mutations = {}

        current_val = sequence[i]
