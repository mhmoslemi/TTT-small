    """Minimize obj on [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    for _ in range(iterations):

            c = d
            fc = fd

            best_a[:] = x

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.8f}")

    return best_a, fast_autocorr(best_a), 1.0, 2.0 * n * float(fast_autocorr(best_a).max())


def try_coord_descent_short(a0, time_limit, seed, verbose=False):
    """Run coordinate descent with early stopping, return best."""
    n = len(a0)
    iters = max(5, int(time_limit / 0.05))
    a, b, s, J = coord_descent_optimize(a0, time_limit, verbose=verbose, seed=seed, max_passes=iters)
    return a, J


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: coordinate descent with multi-restart, mirror descent,
    and length scaling. Focuses on exact line search for precision.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_a = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            best_a = parent.copy()
            best_score = evaluate_safe(best_a)
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_a is None:
        n = 1600
        x = np.linspace(0, 1, n)
        best_a = np.sin(np.pi * x) ** 2
        best_a = best_a / best_a.sum()
        best_score = evaluate_safe(best_a)
        print(f"Default init: n={len(best_a)}, score={best_score:.6f}")

    n_best = len(best_a)

    def try_update(a, label):
        nonlocal best_a, best_score, n_best
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            n_best = len(a)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Mirror descent for global structure (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD n={n_best} ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 20:
        a_out, _, _, _ = mirror_descent_optimize(best_a, t1, top_k=25, beta=4.0, verbose=True, seed=42)
        try_update(a_out, f"md_n{n_best}")

    # === Phase 2: Intensive coordinate descent (45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: CD intensive n={len(best_a)} ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30:
        a_out, b_out, s_out, J_out = coord_descent_optimize(best_a, t2, verbose=True, seed=42)
        try_update(a_out, f"cd_n{len(best_a)}")

    # === Phase 3: Scale to longer length + CD (20% time) ===
    n_long = min(len(best_a) + 800, 3200)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Scale to n={n_long} + CD ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_long = interpolate_sequence(best_a, n_long)
        a_out, _, _, _ = coord_descent_optimize(a_long, t3 * 0.6, verbose=True, seed=123)
        try_update(a_out, f"cd_long_n{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _, _, _ = coord_descent_optimize(a_out, remaining * 0.8, verbose=False, seed=456)
            try_update(a_out2, f"cd_long2_n{n_long}")

    # === Phase 4: CD on best length (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 4: CD polish n={n_cur} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _, _, _ = coord_descent_optimize(best_a, t4, verbose=True, seed=789)
        try_update(a_out, f"cd_polish_n{n_cur}")

    # === Phase 5: Random restarts (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 5: Restarts n={n_cur} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 30:
        n_restarts = max(2, int(t5 / 30))
        time_per_restart = t5 / n_restarts
        rng = np.random.default_rng(2024)
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 30:
                break
            # Create perturbed version
            pert = best_a.copy()
            idx = rng.choice(n_cur, size=n_cur // 5, replace=False)
            pert[idx] *= rng.uniform(0.3, 3.0, size=len(idx))
            pert = np.maximum(pert, 0.0)
            ps = pert.sum()
            if ps < 1e-12:
                continue
            pert = pert / ps
            a_out, _, _, _ = coord_descent_optimize(pert, time_per_restart * 0.8, verbose=False, seed=1000 + r)
            try_update(a_out, f"restart{r}")

    # === Phase 6: Final CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 6: Final CD n={n_cur} ===")
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        a_out, _, _, _ = coord_descent_optimize(best_a, t6, verbose=True, seed=9999)
        try_update(a_out, f"cd_final_n{n_cur}")

    # Final validation
    if best_a is not None:
        fs = evaluate_safe(best_a)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_a]
