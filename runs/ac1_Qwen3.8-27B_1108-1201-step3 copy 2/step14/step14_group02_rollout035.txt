def compute_soft_subgrad(a, ac, top_k=50, beta=2.0):
    """Compute soft subgradient using top-K lags with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    # Get top K lags by value
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    
    # Exponential weighting based on how close to max
    max_val = top_vals[0]
    if max_val < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        # Weight: exp(-beta * (max - val) / max)
        w = np.exp(-beta * (max_val - top_vals) / max_val)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # subgradient of conv(a,a)[k] w.r.t. a[j] is 2*a[k-j] if 0<=k-j<n

            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

    s = a.sum()

def compute_soft_subgrad(a, ac, top_k=50, beta=2.0):
    """Soft subgradient using top-K lags with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)
    
    max_val = top_vals[0]
    if max_val < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = np.exp(-beta * (max_val - top_vals) / max_val)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def mirror_descent_optimize(a0, time_limit, top_k=50, beta=2.0, lr_scale=5.0, verbose=False):
    """Mirror descent with Nesterov acceleration and adaptive learning rate."""
    n = len(a0)

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_avg = a.copy()
    n_avg = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        g = compute_soft_subgrad(y, ac_y, top_k=top_k, beta=beta)

        step = lr_scale / (t + 50.0)
        
        # Mirror step
        log_y = np.log(y + 1e-30)
        log_new = log_y - step * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov
        if t == 1:
            y_new = a_new.copy()
        else:
            beta_n = (t - 1.0) / (t + 2.0)
            y_new = a_new + beta_n * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / sy
        y = y_new
        
        x_avg += y
        n_avg += 1
        
        M_new = fast_autocorr(y).max()
        if M_new < best_m:
            best_m = M_new
            best_a = y.copy()
        
        if n_avg % 200 == 0:
            avg_check = x_avg / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    avg_final = x_avg / n_avg
    avg_final = np.maximum(avg_final, 0.0)

        if M_avg < best_m:

            best_a = avg_final.copy()
    
    return best_a, best_m

def pair_exchange_polish(a0, time_limit, n_trials=50, verbose=False):
    """Local search using pair exchange moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

        # Find top lags
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]

            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            
            if src_val < 1e-15:
                continue

            # Target should be one that minimizes increase in peak lags
            # Score: for each candidate dst, compute how much it adds to top lags
            n_candidates = min(n, 100)
            candidates = rng.integers(0, n, size=n_candidates)
            
            best_score = float('inf')
            best_dst = -1
            
            # Precompute: for each top lag, the "sensitivity" of adding to position p
            # Adding epsilon at position p increases conv(k2) by 2*epsilon*a[|k2-p|] (approx)
            # We want to minimize the maximum increase across top lags
            
            # Vectorized score computation
            # For candidate dst, the increase at lag k2 is ~ 2*a[dst] contribution shift
            # Actually: moving mass from src to dst changes conv(k) by:
            # -2*a[k-src]*delta (removal) + 2*a[k-dst]*delta (addition) (approximately, for small delta)
            # Net change at lag k: 2*delta*(a[k-dst] - a[k-src])
            # We want this to be negative for all top k, or at least minimize the max positive
            
            peak_lags = top_k[:10]
            
            for dst in candidates:
                if dst == src:
                    continue
                # Compute max increase over peak lags
                max_increase = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    # Change at lag k2 from moving delta from src to dst
                    # Removal: -2*delta*a[k2-src] if valid
                    # Addition: +2*delta*a[k2-dst] if valid
                    delta_add = 0.0
                    delta_rem = 0.0
                    d_add = k2i - dst
                    d_rem = k2i - src
                    if 0 <= d_add < n:
                        delta_add = a[d_add]
                    if 0 <= d_rem < n:
                        delta_rem = a[d_rem]
                    net = delta_add - delta_rem
                    if net > max_increase:
                        max_increase = net
                
                if max_increase < best_score:
                    best_score = max_increase
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try several step sizes
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation

            j = rng.integers(n)

                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Pair exchange ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_init(n, init_type='sine', rng=None):

        a = np.ones(n)
    elif init_type == 'random':

        a = (np.cos(np.pi * (x - 0.5)) ** 2 + 0.1)

        a = np.ones(n)
    
    return norm_sum1(a)

def norm_sum1(a):

    s = a.sum()

def compute_softmax_grad_fft(a, ac, tau=50.0):
    """
    Compute gradient of soft-max of autocorrelation values using FFT.
    grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid indices)
    This equals 2 * cross_corr(a, weights)[j] = 2 * conv(flip(a), weights)[n-1+j]
    """
    n = len(a)
    max_ac = ac.max()
    exp_vals = np.exp(tau * (ac - max_ac))
    weights = exp_vals / exp_vals.sum()

    flip_a = a[::-1]
    conv_len = n + (2 * n - 2) - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    A = rfft(flip_a, fft_size)
    W = rfft(weights.astype(np.float64), fft_size)
    conv_result = irfft(A * W, fft_size)

    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def compute_topk_subgrad(a, ac, top_k=50, beta=2.0):
    """Top-K subgradient with exponential weighting (fallback for small n)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)

    max_val = top_vals[0]
    if max_val < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = np.exp(-beta * (max_val - top_vals) / max_val)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def pair_exchange_polish(a0, time_limit, verbose=False):

    n = len(a0)

            k_int = int(k)
            i_start = max(0, k_int - n + 1)

            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-25:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find best destination using sensitivity scoring
            n_candidates = min(n, 150)
            # Bias candidates away from peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            if len(candidates) < n_candidates // 2:
                candidates = np.arange(n)
            if len(candidates) > n_candidates:

                a_trial = np.maximum(a_trial, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        best_seq = generate_init(n_init, 'sine', rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: FFT mirror descent at parent length with multiple taus ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD-FFT n={n0} ===")
    t1_total = min(remaining * 0.25, 400.0)
    if t1_total > 30:
        tau_values = [30.0, 50.0, 80.0]
        t_per = t1_total / len(tau_values)
        for tau_val in tau_values:
            if time.monotonic() - t_global > time_budget:
                break
            if t_per < 15:
                break
            a_out, _ = mirror_descent_optimize(norm_sum1(best_seq), t_per,
                                               tau=tau_val, lr_scale=5.0,
                                               use_fft=True, verbose=True)
            try_update(a_out, f"md_tau{tau_val:.0f}_n{n0}")
            t_per = t1_total / len(tau_values)  # reset

    # === Phase 2: Pair exchange at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Pair exchange n={n0} ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        a_out, _ = pair_exchange_polish(norm_sum1(best_seq), t2, verbose=True)
        try_update(a_out, f"pair_n{n0}")

    # === Phase 3: Alternative initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Alt inits n={n0} ===")
    t3_total = min(remaining * 0.12, 150.0)
    if t3_total > 30:
        inits = ['triangular', 'bump', 'parabolic', 'cosine']
        t_per = t3_total / len(inits)
        for init_type in inits:
            if time.monotonic() - t_global > time_budget:
                break
            if t_per < 10:
                break
            a_init = generate_init(n0, init_type, rng)
            blend = 0.4 * norm_sum1(best_seq) + 0.6 * a_init
            blend = norm_sum1(blend)
            a_out, _ = mirror_descent_optimize(blend, t_per, tau=50.0,
                                               lr_scale=5.0, use_fft=True, verbose=False)
            try_update(a_out, f"md_{init_type}")

    # === Phase 4: Scale to 1.5x length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = min(int(n0 * 1.5), 4000)
    print(f"\n=== Phase 4: MD-FFT n={n1} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        a_out, _ = mirror_descent_optimize(norm_sum1(a_long), t4,
                                           tau=50.0, lr_scale=6.0,
                                           use_fft=True, verbose=True)
        try_update(a_out, f"md_n{n1}")

    # === Phase 5: Scale to 2x length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = min(int(n0 * 2.0), 6000)
    print(f"\n=== Phase 5: MD-FFT n={n2} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        a_long2 = interpolate_sequence(best_seq, n2)
        a_out, _ = mirror_descent_optimize(norm_sum1(a_long2), t5,
                                           tau=40.0, lr_scale=6.0,
                                           use_fft=True, verbose=True)
        try_update(a_out, f"md_n{n2}")

    # === Phase 6: Final refinement at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final refine n={n_best} ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 20:
        configs = [(40.0, 6.0), (60.0, 5.0), (20.0, 4.0)]
        t_per = t6 / len(configs)
        for tau_val, lr_val in configs:
            if time.monotonic() - t_global > time_budget:
                break
            if t_per < 10:
                break
            a_out, _ = mirror_descent_optimize(norm_sum1(best_seq), t_per,
                                               tau=tau_val, lr_scale=lr_val,
                                               use_fft=True, verbose=False)
            try_update(a_out, f"final_tau{tau_val:.0f}")

    # === Phase 7: Final pair exchange ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final pair n={n_best} ===")
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 15:
        a_out, _ = pair_exchange_polish(norm_sum1(best_seq), t7, verbose=True)
        try_update(a_out, f"final_pair_n{n_best}")

    # === Phase 8: One more MD round with best found so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Boost MD n={n_best} ===")
        t8 = min(remaining * 0.4, 120.0)
        if t8 > 20:
            a_out, _ = mirror_descent_optimize(norm_sum1(best_seq), t8,
                                               tau=50.0, lr_scale=5.0,
                                               use_fft=True, verbose=True)
            try_update(a_out, f"boost_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
