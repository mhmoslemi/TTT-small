**Mirror Descent with Negative Entropy:**

The Bregman divergence with negative entropy potential φ(x) = sum x_i log(x_i) is:
D_φ(x||y) = sum x_i log(x_i/y_i) - sum x_i + sum y_i

The mirror descent update for minimizing f subject to x in simplex:
x_{t+1} = argmin_{x in simplex} { η * <g_t, x> + D_φ(x || x_t) }

The solution is: x_{t+1}[i] = x_t[i] * exp(-η * g_t[i]) / sum_j x_t[j] * exp(-η * g_t[j])

This is the multiplicative weights update!

def fast_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_stable(x, tau=1.0):
    """Compute softmax(x/tau) stably."""
    x = np.asarray(x, dtype=np.float64) / tau
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def soft_max_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv[k])) w.r.t. a.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    size = 2 * n - 1
    conv = fast_conv(a)

    x = tau * conv
    x = x - x.max()
    e = np.exp(x)

    # = 2 * ifft(conj(FFT(p)) * FFT(a))[:n]

    a_padded[:n] = a
    P = rfft(p_padded)
    A = rfft(a_padded)
    grad_full = irfft(np.conj(P) * A, fft_size)
    return 2.0 * grad_full[:n]

        sg[lo:] = 2.0 * a[n-1:lo-1:-1]

    # Adaptive parameters
    tau = 5.0 * n  # Start with smooth gradient
    tau_max = 200.0 * n
    eta = 0.1  # Learning rate
    
    stagnation = 0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute soft-max gradient
        conv = fast_conv(a)
        current_max = conv.max()
        
        # Adaptive tau
        if tau < tau_max:
            tau *= 1.005
        
        # Compute gradient
        x = tau * conv
        x = x - x.max()
        e = np.exp(x)
        p = e / e.sum()
        
        # FFT-based gradient
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        p_padded = np.zeros(fft_size)
        p_padded[:size] = p
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        P = rfft(p_padded)
        A = rfft(a_padded)
        grad_full = irfft(np.conj(P) * A, fft_size)
        grad = 2.0 * grad_full[:n]
        
        # Adaptive step size
        grad_scale = grad.max()
        if grad_scale < 1e-15:
            break
        eta_eff = eta / grad_scale
        
        # Multiplicative update
        a_new = a * np.exp(-eta_eff * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        n_iters += 1
        
        # Check improvement
        score = current_max  # Use previous conv (slightly stale but cheap)
        # Actually let's compute fresh for the best tracking
        if n_iters % 10 == 0 or score < best_score:
            conv_check = fast_conv(a)
            score_check = conv_check.max()
            if score_check < best_score:
                best_score = score_check
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 10
        
        # Reduce eta if stagnating
        if stagnation > 500:
            eta *= 0.5
            stagnation = 0
            if eta < 0.005:
                break
        
        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            conv_now = fast_conv(a).max()
            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, curr={conv_now:.10f}, tau={tau/n:.1f}n")
    
    return best_a, best_score


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) > 0 and len(active_set) < size:
        # Build constraint matrix for active set only
        rows = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx2 = k - j
                if 0 <= idx2 < n:
                    rows[idx, j] = a[idx2]
        constraints = [rows @ b_var <= M]
    else:
        # Full constraint matrix
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j+n, j] = a
        active_set = np.arange(size)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set, None
    
    return b_val, active_set, prob
