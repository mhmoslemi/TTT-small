def lse_value_grad(a, T):
    """
    Compute LSE objective and exact gradient.
    Objective: F(a) = 2*n*lse(b) / (sum(a))^2 where b = conv(a,a)
    """
    n = len(a)
    b = fast_autocorr(a)
    b_max = b.max()
    exp_vals = np.exp(T * (b - b_max))
    exp_sum = exp_vals.sum()
    lse = b_max + np.log(exp_sum) / T
    w = exp_vals / exp_sum  # This is the "soft argmax" weights
    
    s = a.sum()
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient of lse wrt a:
    # dlse/da_i = sum_k w_k * 2 * a_{k-i} (for valid indices)
    # = 2 * (w_star_a)_i where (w_star_a)_i = sum_k w_k * a_{k-i}
    
    # Compute cross-correlation (w_star_a)[i] = sum_{j=0}^{n-1} w_{i+j} * a_j
    # Using FFT: this is conv(a, w_padded_rev)[offset]
    
    w_ext = np.zeros(3 * n - 2)
    w_ext[:2 * n - 1] = w
    w_rev = w_ext[::-1]
    
    fft_size = 1 << (4 * n - 3).bit_length()
    A_fft = rfft(a, fft_size)
    W_fft = rfft(w_rev, fft_size)
    full_conv = irfft(A_fft * W_fft, fft_size)[:4 * n - 3]

    cross = full_conv[indices]
    
    dlse_da = 2.0 * cross  # dlse/da_i = 2 * cross_i
    
    # Full objective gradient
    grad = (2.0 * n / (s * s)) * dlse_da - (4.0 * n * lse / (s ** 3)) * a
    
    return obj, grad

def project_simplex(v):

def normalize(a, target_sum=None):

    return a * target_sum / s

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    # Build the constraint matrix T where (T @ b)[k] = conv(a, b)[k]
    # T is a (2n-1) x n Toeplitz-like matrix
    # T[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    # This is the same as a Toeplitz matrix with first column = a (padded)
    
    # For efficiency, build as sparse
    from scipy.sparse import csr_matrix
    
    # Row k, column i: T[k,i] = a[k-i] if 0 <= k-i <= n-1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # Column i has non-zeros at rows i, i+1, ..., i+n-1 (up to 2n-2)
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False):
    """
    Nesterov-accelerated LP best-response iteration.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    t = 0
    stall = 0
    best_score_track = best_score
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Line search
        base_alpha = 2.0 / (t + 2.0)
        best_alpha = base_alpha
        best_obj = fast_autocorr(np.maximum((1 - base_alpha) * y + base_alpha * g, 0)).max()
        
        # Try a few more step sizes
        for trial in [base_alpha * 0.5, base_alpha * 0.75, base_alpha * 1.25, base_alpha * 1.5, base_alpha * 2.0]:
            trial = min(trial, 1.0)
            if trial <= 0:
                continue
            cand = np.maximum((1 - trial) * y + trial * g, 0.0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj - 1e-14:
                best_obj = sc
                best_alpha = trial
        
        alpha = best_alpha
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        y_new = normalize(y_new, norm)
        
        # Nesterov extrapolation
        if t == 1:
            a_new = y_new.copy()
        else:
            mu = (t - 1.0) / (t + 2.0)
            a_new = y_new + mu * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = candidate.copy()
                stall = 0
            else:
                stall += 1
        
        # Averaging
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_candidate = normalize(avg_a / n_avg, norm)
            sc = fast_autocorr(avg_candidate).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = avg_candidate.copy()
                stall = 0
        
        # Restart if stalling
        if stall > 100:
            y = normalize(best_a, norm)
            a = y.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0
            if verbose:
                print(f"    LP restart at t={t}, best={best_score:.8f}")
        
        if stall > 500:
            break
        
        if verbose and t % 200 == 0:
            print(f"    LP iter={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def adam_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8, verbose=False):
    """
    Adam optimizer on LSE surrogate with temperature annealing.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / max(1, n_stages - si)
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.1
        step = 0
        stall = 0
        best_stage_score = best_score
        best_stage_a = a.copy()
        
        while time.monotonic() - t0 < time_limit - 2:
            step += 1
            
            # Decay learning rate
            lr = 0.1 / (1 + 0.01 * step)
            
            obj, grad = lse_value_grad(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
            
            # Evaluate
            score = fast_autocorr(a_new).max()
            
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                stall = 0
            elif score < best_stage_score - 1e-12:
                best_stage_score = score
                best_stage_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            
            a = a_new
            
            if stall > 200:
                a = best_stage_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.05
                stall = 0
            
            if stall > 500:
                break
            
            if verbose and step % 500 == 0:
                print(f"    Adam T={T:.1f} step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
        
        # End of stage: use best for next stage
        a = normalize(best_stage_a, norm)
    
    return best_a, best_score

def multiplicative_updates(a0, time_limit, T=10.0, verbose=False):
    """
    Multiplicative weight updates (mirror descent with entropy).
    a_i <- a_i * exp(-eta * grad_i)
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.1
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        obj, grad = lse_value_grad(a, T)
        
        # Multiplicative update
        # Scale gradient relative to a to avoid overflow
        grad_normalized = grad / (a + 1e-10)
        grad_clipped = np.clip(grad_normalized, -10, 10)
        
        a_new = a * np.exp(-eta * grad_clipped)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, norm)
        
        score = fast_autocorr(a_new).max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Decay eta
        if step % 50 == 0:
            eta *= 0.95
        
        if stall > 300:
            if eta > 0.001:
                eta *= 0.5
                a = best_a.copy()
                stall = 0
            else:
                break
        
        if verbose and step % 200 == 0:
            print(f"    MU step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def gradient_coord_descent(a0, time_limit, seed=42, verbose=False):
    """
    Gradient-guided coordinate descent: move mass from high-gradient
    coordinates to low-gradient coordinates.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        s = a.sum()
        
        # Find the peak lags
        top_peaks = np.argsort(b)[::-1][:8]
        
        # Compute a "responsibility" score for each coordinate
        # based on how much it contributes to the top peaks
        responsibility = np.zeros(n)
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            responsibility[idx] += contrib / (b[k_peak] + 1e-15)
        
        # Coordinates with high responsibility are "bad" (contribute to peaks)
        # Coordinates with low responsibility are "good" (don't contribute)
        
        # Move mass from high-responsibility to low-responsibility coordinates
        n_moves = min(n // 4, 500)
        improved = False
        
        high_res = np.argsort(responsibility)[::-1][:n_moves]
        low_res = np.argsort(responsibility)[:n_moves]
        
        for _ in range(min(100, n_moves)):
            # Pick a random source (high responsibility) and destination (low responsibility)
            src = rng.choice(high_res)
            dst = rng.choice(low_res)
            
            if a[src] < 1e-12 or src == dst:
                continue
            
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize(a_trial, norm)
                
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_a = a_trial.copy()
                    best_score = score
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            # Try gradient-based moves
            obj, grad = lse_value_grad(a, 5.0)
            
            # Move mass from high-grad to low-grad coordinates
            high_grad = np.argsort(grad)[::-1][:n // 5]
            low_grad = np.argsort(grad)[:n // 5]
            
            for _ in range(50):
                src = rng.choice(high_grad)
                dst = rng.choice(low_grad)
                if a[src] < 1e-12 or src == dst:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = normalize(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
        
        if verbose and stall % 20 == 0:
            print(f"    GCD stall={stall} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
