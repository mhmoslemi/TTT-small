    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    return np.maximum(v - theta, 0.0)

                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            # Adapt momentum
            momentum = 0.85 + 0.1 * (si / n_stages)

        if verbose:
            print(f"    LSE-M stage {si} T={T_val:.1f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def build_T_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve_fast(a, M, solver, active_idx):
    """Solve LP with given active constraints."""
    n = len(a)
    size = 2 * n - 1
    T = build_T_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_idx is not None and 0 < len(active_idx) < size:
        constraints = [T[active_idx, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive screening."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    last_full_check = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive screening: use tighter threshold early, looser later
        progress = min(1.0, t / 200.0)
        if t % 50 == 0 or t < 50:
            # Periodic full check
            threshold = M * (0.7 + 0.15 * progress)
        else:
            threshold = M * (0.75 + 0.15 * progress)
        active = np.where(conv_y > threshold)[0]
        if len(active) < 10:
            active = np.where(conv_y > M * 0.5)[0]

        # Solve LP with screened constraints
        b = lp_solve_fast(y, M, solver, active)
        if b is None or b.sum() < 1e-15:
            b = lp_solve_fast(y, M, solver, None)
        if b is None or b.sum() < 1e-15:
            continue

        # Check if dropped constraints are violated
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, b)
            violated = np.where(check_conv > M * (1 + 1e-3))[0]
            if len(violated) > 0:
                b = lp_solve_fast(y, M, solver, None)
                if b is None or b.sum() < 1e-15:
                    continue

        b_sum = b.sum()
        g = b * norm_val / b_sum

        # Nesterov update with adaptive alpha
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum_coeff = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum_coeff * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgrad_cd_optimize(a0, time_limit, rng, verbose=True):
    """Subgradient-guided coordinate descent with exact 1D optimization."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Identify peak positions for subgradient
        peak_thresh = cur_max * 0.98
        peak_lags = np.where(conv >= peak_thresh)[0]

        # Compute subgradient (average over peak positions)
        subgrad = np.zeros(n)
        for kl in peak_lags:
            for i in range(n):
                val = 0.0
                ki = kl - i
                if 0 <= ki < n:
                    val += a[ki]
                ik = i - kl
                if 0 <= ik < n:
                    val += a[ik]
                subgrad[i] += val
        subgrad /= len(peak_lags)

        # Sort by subgradient: high subgrad coordinates should decrease
        # Low subgrad coordinates should increase
        order = np.argsort(subgrad)
        low_coords = order[:max(1, n // 5)]
        high_coords = order[-max(1, n // 5):]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-10:
                continue

            # Candidate j's: low subgradient + random
            j_candidates = []
            for j in low_coords[:5]:
                if j != i:
                    j_candidates.append(j)
            for _ in range(4):
                j = int(rng.integers(n))
                if j != i:
                    j_candidates.append(j)
            # Add structured candidates based on peak lags
            for kl in peak_lags[:3]:
                j1 = (kl - i) % n
                j2 = (kl + i) % n
                if j1 != i:
                    j_candidates.append(j1)
                if j2 != i:
                    j_candidates.append(j2)

            # Deduplicate
            seen = set()
            unique_js = []
            for j in j_candidates:
                if j not in seen and j != i:
                    seen.add(j)
                    unique_js.append(j)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in unique_js[:8]:
                aj = a[j]

                # Compute transfer vector
                # delta_conv[k] = d * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
                # plus quadratic corrections at k=2i (negative), k=2j (positive), k=i+j
                dv = np.zeros(size)
                # a[k-j] valid when k-j in [0,n-1] -> k in [j, j+n-1]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    dv[lo:hi] += a[0:hi - lo]
                # a[j-k] valid when j-k in [0,n-1] -> k in [j-n+1, j]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    dv[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                # -a[k-i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    dv[lo3:hi3] -= a[0:hi3 - lo3]
                # -a[i-k]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    dv[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal d in [0, ai]
                # The max of conv + d*dv (ignoring quadratic) is piecewise linear in d
                # Use ternary search on the smoothed version (works well in practice)
                # Or: evaluate at several candidate points
                n_samples = 20
                d_vals = np.linspace(0.01, ai * 0.99, n_samples)

                # Add the kink points (where two lines cross)
                # Approximate: check where dv[k]*d = max_conv - conv[k] for top k's
                top_k = np.argsort(conv)[-10:]
                for k in top_k:
                    if dv[k] < -1e-15:
                        d_kink = (cur_max - conv[k]) / (-dv[k])
                        if 0 < d_kink < ai:
                            d_vals = np.append(d_vals, d_kink)

                d_vals = np.clip(d_vals, 0.0, ai)
                d_vals = np.unique(d_vals)

                for d in d_vals:
                    if d < 1e-12:
                        continue
                    new_conv = conv + d * dv
                    # Quadratic corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    if i + j < size:
                        new_conv[i + j] = conv[i + j] + d * dv[i + j] - 2 * d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm_val / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm_val / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 4 or no_improve == 6:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_walk_polish(a0, time_limit, rng, verbose=True):
    """Random perturb + re-optimize to escape local optima."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Random Gaussian perturbation
        n_perturb = max(10, n // 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        scale = 0.1 + 0.05 * rng.random()
        a[idx] *= (1.0 + scale * rng.standard_normal(n_perturb))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a * norm_val / s

        # Quick LSE GD to recover
        T_val = 5.0
        for it in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            fft_size = 1 << (2 * n - 2).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            aw = irfft(A * W, fft_size)
            grad = 2.0 * (ca[:n] + aw[:n])

            step = 0.5 / np.sqrt(n)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm_val / s_new
            sc_new = fast_conv_self(a_new).max()
            if sc_new < max_c:
                a = a_new
            T_val *= 0.95

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            print(f"    RW t={time.monotonic()-t0:.0f}s: best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x * (1 - x) * 4)
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
    elif choice == 8:
        # Symmetric with dip in middle
        a = np.exp(-4.0 * (x - 0.2) ** 2) + np.exp(-4.0 * (x - 0.8) ** 2) + 0.3 * np.exp(-8.0 * (x - 0.5) ** 2)
    else:
        # Random smooth
        a = np.exp(-2.0 * (x - rng.random()) ** 2) + 0.5 * np.exp(-2.0 * (x - rng.random()) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE momentum GD (fast initial progress) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE-Momentum GD n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_momentum_gd(best_seq, t1, verbose=True)
        try_update(a_out, "lse_mom")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_nest")

    # === Phase 3: Subgradient CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: SubGrad CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = subgrad_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, "subgrad_cd")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp_nest2")

    # === Phase 5: Random walk + polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Random walk polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = random_walk_polish(best_seq, t5, rng, verbose=True)
        try_update(a_out, "rw_polish")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length exploration ===")
    if remaining > 100:
        for n_try in [len(best_seq) - 200, len(best_seq) + 200, len(best_seq) + 400]:
            if n_try < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.4, 120.0)
            if t_alloc < 30:
                break
            print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
            init_try = interpolate_seq(best_seq, n_try)
            # Quick LSE
            a_txt, _ = lse_momentum_gd(init_try, min(t_alloc * 0.4, 30.0), verbose=False)
            try_update(a_txt, f"lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            # LP
            a_out, _ = lp_nesterov_optimize(best_seq, min(remaining * 0.4, 60.0), solver, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # === Phase 7: Final polish cycle ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.3, 80.0)
        a_out, _ = lse_momentum_gd(best_seq, t7a, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.4, 60.0)
            a_out, _ = lp_nesterov_optimize(best_seq, t7b, solver, verbose=False)
            try_update(a_out, "lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7c = min(remaining * 0.5, 40.0)
            a_out, _ = subgrad_cd_optimize(best_seq, t7c, rng, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
