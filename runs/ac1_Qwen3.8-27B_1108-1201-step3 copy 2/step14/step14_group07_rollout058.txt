"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based LSE projected gradient
descent with adaptive temperature, targeted peak reduction, and multi-scale search.
Key innovations: O(n log n) gradient computation via cross-correlation theorem,
momentum-based projected gradient with backtracking, and iterative peak
elimination for final polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective_value(b, T):
    """Compute LSE value of b at temperature T."""
    b_max = b.max()
    if b_max < 1e-30:
        return 0.0
    shifted = T * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.exp(shifted).sum())
    return b_max + log_sum / T


def lse_weights(b, T):
    """Compute softmax weights for LSE gradients."""
    b_max = b.max()
    if b_max < 1e-30:
        return np.ones(len(b)) / len(b)
    shifted = T * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()


def lse_gradient_fft(a, w, n):
    """
    Compute gradient of LSE w.r.t. a using cross-correlation.
    grad[j] = sum_l w[j+l] * a[l] for l = 0..n-1 (with w out of range = 0)
    Uses FFT cross-correlation theorem.
    """
    # We need corr[w, a][j] = sum_m w[m+j] * a[m] for j=0..n-1
    # This requires w to be defined up to index (n-1)+(n-1) = 2n-2, which it is (length 2n-1)
    # Total needed length for correlation: len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    w_padded = np.zeros(fft_size)
    w_padded[:2*n-1] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = rfft(w_padded, fft_size)
    A = rfft(a_padded, fft_size)
    
    # Cross-correlation: IFFT(W * conj(A))
    # corr[j] = sum_m w[m+j] * a[m]
    corr = irfft(W * np.conj(A), fft_size)
    
    return corr[:n]


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    
    # Sort in decreasing order
    sorted_v = np.sort(v)[::-1]
    
    # Find rho
    css = np.cumsum(sorted_v)
    cond = sorted_v + (css - 1.0) / np.arange(1, n + 1) > 0
    rho = np.sum(cond)
    
    if rho == 0:
        return np.ones(n) / n
    
    theta = (css[int(rho) - 1] - 1.0) / rho
    result = np.maximum(v - theta, 0.0)
    # Normalize (should already sum to 1, but fix numerical errors)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def lse_pgdescent(a0, time_budget, T_start=50.0, T_end=1.0, 
                  momentum_sched=None, verbose=False):
    """
    LSE projected gradient descent with momentum and adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam-like state
    m_state = np.zeros(n)  # First moment
    v_state = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    total_iters = 0
    last_improve = t0
    stall_count = 0
    
    # Temperature schedule: geometric decay
    # We'll do multiple "stages" with different temperatures
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        stage_budget = (time_budget - (time.monotonic() - t0)) / (n_stages - stage)
        stage_t = time.monotonic()
        
        # Learning rate for this stage
        eta = 1.0 / (T * fast_autocorr(a).max() + 1e-10)
        eta = min(eta, 0.5)
        
        while time.monotonic() - stage_t < stage_budget:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Compute gradient
            b = fast_autocorr(a)
            w = lse_weights(b, T)
            grad = lse_gradient_fft(a, w, n)
            
            total_iters += 1
            
            # Adam update
            m_state = beta1 * m_state + (1 - beta1) * grad
            v_state = beta2 * v_state + (1 - beta2) * grad ** 2
            m_hat = m_state / (1 - beta1 ** total_iters)
            v_hat = v_state / (1 - beta2 ** total_iters)
            
            a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
            a_new = simplex_project(a_new)
            
            # Evaluate
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                last_improve = time.monotonic()
            
            a = a_new
            
            if total_iters % 200 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE-PGD: iter={total_iters}, T={T:.1f}, "
                      f"best_max_b={best_score:.8f}, elapsed={elapsed:.0f}s")
        
        # Update learning rate for next stage
        if verbose:
            print(f"    LSE stage {stage+1}/{n_stages} done: "
                  f"T={T:.1f}, best_max_b={best_score:.8f}")
    
    return best_a, best_score


def peak_reduction(a0, time_budget, top_k=5, n_trials=200, verbose=False):
    """
    Targeted peak reduction: iteratively reduce the top peaks in the autocorrelation.
    Uses mass transfer from high-contributing to low-contributing positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        improved = False
        
        for k_peak in top_peaks:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Find contributing pairs
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Find the top contributors
            top_contrib = idx[np.argsort(contrib)[::-1][:10]]
            
            # Find low-contribution positions (outside peak region)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(peak_mask)[0]
            
            # Try transferring from top contributors to candidates
            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                
                # Sort candidates by how "safe" they are (low contribution to peaks)
                # A position dst is "safe" if it doesn't contribute much to any top peak
                dst_contributions = np.zeros(len(candidates))
                for pos, dst in enumerate(candidates):
                    for k2 in top_peaks:
                        if 0 <= k2 - dst < n:
                            dst_contributions[pos] += a[k2 - dst]
                
                # Pick low-contribution destinations
                safe_dsts = candidates[np.argsort(dst_contributions)[:min(20, len(candidates))]]
                
                for dst in safe_dsts[:10]:
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_a = a_trial.copy()
                            best_score = score
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        break
                if stall == 0:
                    continue
            
            if stall > 500:
                break
    
    return best_a, best_score


def structured_init(n, seed=42):
    """Generate various structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        # Sine power
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        # Multi-frequency
        x = np.linspace(0, 2*np.pi, n)
        a = 0.5 + 0.3*np.cos(2*x) + 0.15*np.cos(5*x) + 0.05*np.cos(9*x)
    elif choice == 3:
        # Double Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-4*(x-0.25)**2) + 0.6*np.exp(-4*(x-0.75)**2)
    elif choice == 4:
        # Single Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-8*(x-0.5)**2)
    elif choice == 5:
        # Gaussian centered
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5*(idx/sigma)**2)
    elif choice == 6:
        # Sinusoidal modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3*np.sin(4*np.pi*x) + 0.1*np.sin(8*np.pi*x)
    elif choice == 7:
        # Near-constant with small perturbation
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    elif choice == 8:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5 + 0.5*(1-x) ** 0.5
    else:
        # Random
        a = rng.standard_normal(n) ** 2
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def fourier_init(n, rng):
    """
    Initialize using Fourier domain: create a sequence whose Fourier transform
    has controlled magnitude to produce a flat autocorrelation.
    """
    # We want a*a to be flat. In Fourier domain, FFT(a*a) ≈ |FFT(a)|^2 (for circular).
    # For linear convolution with zero-padding, it's similar.
    # Strategy: choose |A(f)| to be approximately constant, then pick random phases.
    
    fft_size = 1 << (2*n-2).bit_length()
    
    # Target: |A(f)|^2 should be approximately constant
    # |A(f)| = constant for all f
    # => a = IFFT(constant magnitude * random phases)
    
    freqs = np.fft.rfftfreq(fft_size)
    
    # Use a bandpass to concentrate energy
    band = np.abs(freqs) > 1.0/fft_size  # Remove DC
    # Or use a sigmoid to create a smooth transition
    f_norm = np.abs(freqs) * fft_size / 2
    band_weight = 0.3 + 0.7 * (1 - np.exp(-f_norm))  # Smooth lowpass removal
    
    magnitude = band_weight * np.sqrt(n / fft_size)  # Normalize
    phases = rng.uniform(0, 2*np.pi, size=len(magnitude))
    # Make DC component real and positive
    phases[0] = 0
    
    A = magnitude * np.exp(1j * phases)
    a_full = np.fft.irfft(A, fft_size)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lse_pgdescent_v2(a0, time_budget, verbose=False):
    """
    Improved LSE projected gradient descent with:
    - Backtracking line search
    - Better momentum (heavy ball)
    - Multiple restarts on stall
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Heavy ball momentum
    v_state = np.zeros(n)
    momentum = 0.9
    
    # Temperature stages
    n_stages = 10
    T_start = 100.0
    T_end = 0.5
    temps = np.geomspace(T_start, T_end, n_stages)
    
    total_iters = 0
    stall_count = 0
    last_improve = t0
    
    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        stage_budget = remaining / (n_stages - stage)
        stage_t = time.monotonic()
        
        # Initial learning rate for this stage
        b_init = fast_autocorr(a)
        eta = 0.5 / (T * b_init.max() + 1e-6)
        eta = min(eta, 0.1)
        
        v_state = np.zeros(n)  # Reset momentum each stage
        
        while time.monotonic() - stage_t < stage_budget:
            if time.monotonic() - t0 > time_budget:
                break
            
            b = fast_autocorr(a)
            w = lse_weights(b, T)
            grad = lse_gradient_fft(a, w, n)
            
            total_iters += 1
            
            # Heavy ball update
            v_state = momentum * v_state - eta * grad
            a_new = a + v_state
            a_new = simplex_project(a_new)
            
            # Check improvement
            score_new = 2.0 * n * fast_autocorr(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                last_improve = time.monotonic()
                stall_count = 0
            else:
                stall_count += 1
            
            a = a_new
            
            # Adaptive learning rate
            if total_iters % 50 == 0:
                if stall_count > 100:
                    eta *= 0.8
                    if eta < 1e-8:
                        break
                elif stall_count < 50:
                    eta *= 1.05
            
            if verbose and total_iters % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSEv2: iter={total_iters}, T={T:.1f}, eta={eta:.2e}, "
                      f"obj={best_score:.8f}, stall={stall_count}, t={elapsed:.0f}s")
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSEv2 stage {stage+1}/{n_stages}: T={T:.1f}, "
                  f"obj={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_weights(b, T):
    """Compute softmax weights for LSE: w[k] = exp(T*b[k]) / sum(exp(T*b))."""
    b_max = b.max()
    if b_max < 1e-30:
        return np.ones(len(b)) / len(b)
    shifted = T * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_vals / total


def lse_gradient_fft(a, w, n):
    """
    Compute dLSE/da[j] = sum_l w[j+l]*a[l] via FFT cross-correlation.
    Uses: corr = IFFT(FFT(w) * conj(FFT(a)))
    """
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = rfft(w_padded, fft_size)
    A = rfft(a_padded, fft_size)

    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    cond = sorted_v + (css - 1.0) / np.arange(1, n + 1) > 0
    rho = int(np.sum(cond))
    if rho == 0:
        return np.ones(n) / n
    theta = (css[rho - 1] - 1.0) / rho
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s
