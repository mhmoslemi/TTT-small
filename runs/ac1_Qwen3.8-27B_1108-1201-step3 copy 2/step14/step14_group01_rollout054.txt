        # Actually, for the gradient of max(conv(a,a)) w.r.t. a[i]:
        # d/d a[i] conv(a,a)[k] = a[k-i] + a[i-k] (when valid)
        # So grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k])
        
        # sum_k w[k] * a[k-i]: this is the cross-correlation of w and a at lag i
        # = (w cross a)[i] = sum_k w[k] * a[k+i] (circular) or sum_k w[k] * a[k-i]
        
        # Using FFT: cross_corr[i] = ifft(conj(FFT(w)) * FFT(a))[i]
        # This gives sum_k w[k] * a[k+i] (circular, with appropriate windowing)
        
        # For linear convolution with zero-padding (fft_size >= 2n-1):
        # ifft(FFT(w) * FFT(a)) = conv(w, a) (linear)
        # conv(w,a)[k] = sum_j w[j] * a[k-j] for 0 <= k-j < n
        # = sum_j w[j] * a[k-j]
        
        # We want s1[i] = sum_k w[k] * a[k-i]
        # Let j = k-i: s1[i] = sum_j w[j+i] * a[j]
        # = sum_j a[j] * w[j+i]
        # = cross_corr_a_w[i] where cross_corr_a_w[i] = sum_j a[j] * w[j+i]
        
        # cross_corr_a_w = ifft(conj(FFT(a)) * FFT(w)) (circular)
        # With zero-padding, this equals the linear cross-correlation for |i| < n.
        
        ca = irfft(np.conj(A) * W, fft_size)
        # ca[i] = sum_j a[j] * w[j+i] (circular, but linear for i in [-n+1, n-1] with zero-pad)
        s1 = ca[:n]  # s1[i] = sum_j a[j] * w[j+i] for i = 0, ..., n-1
        
        # sum_k w[k] * a[i-k]: let j = i-k, = sum_j w[i-j] * a[j]
        # = sum_j a[j] * w[i-j]
        # = conv(a, w)[i] (linear convolution at position i)
        g2 = wa[:n]  # wa[k] = sum_j a[j] * w[k-j], so wa[i] = sum_j a[j] * w[i-j]
        # Wait, wa = conv(w, a) or conv(a, w)?
        # ifft(FFT(w) * FFT(a)) = conv(w, a) [w convolved with a]
        # conv(w,a)[k] = sum_j w[j] * a[k-j]
        # We want sum_j a[j] * w[i-j] = conv(a, w)[i] = conv(w, a)[i] (convolution is commutative)
        # So g2[i] = wa[i] = sum_j w[j] * a[i-j] = sum_j a[j] * w[i-j] ✓
        
        grad = 2.0 * (s1 + g2)
        
        # Gradient step with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = norm / (T_val * np.sqrt(n)) * 0.5
        a_new = a.copy()
        for _ in range(15):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            new_max = fast_conv_self(a_new).max()
            if new_max <= max_c + 0.01:
                break
            step *= 0.5
        
        a = a_new
        s_new = a.sum()
        if s_new < 1e-15:
            a = np.ones(n) / n * norm
        a_max = fast_conv_self(a).max()
        if a_max < best_max:
            best_max = a_max
            best_a = a.copy()
    
    return best_a, best_max

def nesterov_lp_optimize(a0, time_limit, solver, rng=None, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * norm
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()  # Nesterov "momentum" point
    
    iteration = 0
    stagnant = 0
    prev_max = best_max
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        # Compute M from current y
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Solve LP best-response
        b = lp_best_response(y, M, solver, max_iter=8, max_active=150)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (iteration + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if iteration == 1:
            x_new = g.copy()
        else:
            momentum = (iteration - 1.0) / (iteration + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x
        
        y = y_new
        x = x_new
        
        # Evaluate
        max_y = fast_conv_self(y).max()
        if max_y < best_max:
            best_max = max_y
            best_a = y.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        max_x = fast_conv_self(x).max()
        if max_x < best_max:
            best_max = max_x
            best_a = x.copy()
            stagnant = 0
        
        # Perturb if stagnant
        if stagnant > 50 and rng is not None:
            n_perturb = max(10, n // 20)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_pert = best_a.copy()
            a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert * norm / s_p
                max_p = fast_conv_self(a_pert).max()
                if max_p < best_max * 1.01:  # accept if not too bad
                    y = a_pert.copy()
                    stagnant = 0
        
        if verbose and iteration % 50 == 0:
            print(f"    LP Nesterov it={iteration} ({elapsed:.0f}s): best={best_max:.8f}")
    
    return best_a, best_max

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * norm
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue
            
            # Select j candidates
            j_candidates = set()
            for _ in range(10):
                j_candidates.add(int(rng.integers(n)))
            # Structured candidates from peak lags
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            for kl in peak_lags[:3]:
                j_struct = (kl - i + n) % n
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct)
                j_candidates.add(j_struct2)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in list(j_candidates)[:8]:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector
                g = conv.copy()
                # Subtract contribution of a[i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3-lo3] * ai / ai  # a[k-i]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                # Add contribution of a[j]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi-lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # Find best delta
                best_d = 0.0
                best_v = cur_max
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8]:
                    d = ai * frac
                    new_conv = conv + d * g
                    if 2*i < size:
                        new_conv[2*i] -= 2*ai*d - d*d
                    if 2*j < size:
                        new_conv[2*j] += 2*aj*d + d*d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                
                if best_d > 0 and best_v < best_val - 1e-10:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] = ai - best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max - 1e-12:
                        best_max = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] = ai
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve >= 4:
                n_pert = max(5, n // 50)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_imp % 200 == 0 and total_imp > 0:
            print(f"    PCD imp={total_imp} ({time.monotonic()-t0:.0f}s): best={best_max:.8f}")
    
    return best_a, best_max
