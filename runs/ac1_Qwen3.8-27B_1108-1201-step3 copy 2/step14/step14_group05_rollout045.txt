            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_autocorr_gradient(a, weights):
    """
    Compute gradient of sum(weights[i] * b[i]) where b = convolve(a, a).
    Uses FFT for O(n log n) computation.
    weights: array of length 2n-1
    Returns: gradient array of length n
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(weights, fft_size)
    AR = rfft(a[::-1], fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def power_mean_gradient_descent(a0, time_limit, p_start=4.0, p_end=80.0, verbose=True):
    """
    Nesterov-accelerated gradient descent on the power-mean smoothed autocorrelation.
    f_p(a) = (mean(b^p))^(1/p) approximates max(b) for large p.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    v = np.zeros(n)  # Momentum
    beta = 0.9  # Momentum coefficient

    t0 = time.monotonic()
    iters = 0
    lr = 1.0 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        p = p_start * (p_end / p_start) ** elapsed_frac

        b = fast_autocorr(y)
        b_pos = np.maximum(b, 1e-30)

        # Power-mean weights: w_i = b_i^(p-1) / (sum(b^(p-1)))
        w = b_pos ** (p - 1)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient
        grad = compute_autocorr_gradient(a, w)

        # Project onto tangent space (sum constraint)
        grad -= np.mean(grad)

        # Backtracking line search
        f_curr = (np.mean(b_pos ** p)) ** (1.0 / p)
        best_lr = lr * 0.1
        for _ in range(8):
            a_trial = a - best_lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                best_lr *= 0.5
                continue
            a_trial = a_trial * norm / s_t
            b_trial = fast_autocorr(a_trial)
            f_trial = (np.mean(np.maximum(b_trial, 1e-30) ** p)) ** (1.0 / p)
            if f_trial < f_curr:
                break
            best_lr *= 0.5

        # Momentum update
        v = beta * v - best_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        iters += 1

        # Nesterov lookahead
        mu = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a + (beta / mu) * (a - a + v / (best_lr + 1e-30))  # simplified
        # Actually let's use standard Nesterov: y = a + momentum*(a - prev_a)
        # Simplified: just use y = a for the gradient evaluation
        y = a.copy()

        # Track best
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adaptive learning rate
        if iters % 200 == 0:
            # Reduce momentum slightly over time
            beta = max(0.5, 0.9 - 0.1 * elapsed_frac)

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PMGD iter={iters} ({elapsed:.0f}s): p={p:.1f}, best={best_score:.8f}, lr={best_lr:.2e}")

    return best_a, best_score


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, max_active=40):
    """
    Fessler-Parks accelerated LP Nesterov with ultra-aggressive constraint reduction.
    Only keeps top max_active active constraints for speed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.9

    # Pre-build the convolution matrix (sparse structure)
    # T[i, j] = a[j] if 0 <= i-j < n else 0
    # We'll compute T @ b on the fly using FFT

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Find active constraints (top max_active)
        b_y = fast_autocorr(y)
        tol = 1e-6 * M_curr
        active = b_y >= M_curr - tol
        n_active = int(active.sum())

        if n_active > max_active:
            active_idx = np.where(active)[0]
            # Keep the most binding ones
            vals = b_y[active_idx]
            top_k = max_active
            idx_order = np.argsort(vals)[::-1][:top_k]
            active_top = active_idx[idx_order]
        else:
            active_top = np.where(active)[0]

        # Build reduced constraint matrix and solve LP
        # T[b_idx, :] @ b <= M[b_idx]
        # T[i, j] = y[j] if 0 <= i-j < n
        # So T[i,:] = [y[i-(n-1)], ..., y[i]] shifted
        # For row i: columns j where 0 <= i-j < n, i.e., j in [max(0, i-n+1), min(n-1, i)]
        # T[i, j] = y[j] for j in that range

        b_var = cp.Variable(n, nonneg=True)
        constraints = []

        for i in active_top:
            js = max(0, i - n + 1)
            je = min(n - 1, i)
            row = np.zeros(n)
            for j in range(js, je + 1):
                row[j] = y[j]
            constraints.append(row @ b_var <= b_y[i])

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            n_iters += 1
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            n_iters += 1
            continue
        if b_var.value is None:
            n_iters += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            n_iters += 1
            continue

        g = b_val * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum step
        if t <= 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, active={len(active_top)}")

        # Adaptive restart
        if stall > 200:
            rng = np.random.default_rng(n_iters * 17 + 3)
            perturb_scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = max(1.0, gamma - 0.1)
            if verbose:
                print(f"    [LP Restart at iter {n_iters}]")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Iterative projection: alternate between flat-power-spectrum in freq domain
    and non-negativity in time domain. Similar to Gerchberg-Saxton.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0

    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        # Frequency domain: flatten the power spectrum
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat magnitude (proportional to 1/sqrt(n) for unit sum)
        target_mag = np.ones_like(mag) * (norm / np.sqrt(fft_size))
        # Blend between current and target
        blend = 0.3 + 0.4 * (iters % 100) / 100.0
        new_mag = blend * target_mag + (1 - blend) * mag
        A_new = new_mag * np.exp(1j * phase)

        # Back to time domain
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        a = a_new
        iters += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """
    Fast coordinate descent: try moving mass from peak-contributing positions
    to non-contributing positions with line search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing positions
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            n_stall += 1
            continue
        contrib = a[idx] * a[p_star - idx]
        max_contrib_idx = idx[np.argmax(contrib)]

        # Try several destination positions
        # Non-contributing: j < js or j > je
        n_nc_start = js
        n_nc_end = je + 1
        n_dest = min(8, n)
        best_trial = None
        best_trial_score = max_b

        for _ in range(n_dest):
            dst = rng.integers(n)
            if dst == max_contrib_idx:
                continue
            src_val = a[max_contrib_idx]
            if src_val < 1e-15:
                break

            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[max_contrib_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_trial.copy()

        if best_trial is not None and best_trial_score < best_score:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        else:
            # Try random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.05, 0.3)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                    else:
                        n_stall += 1
                else:
                    n_stall += 1
            else:
                n_stall += 1

        # Periodic large perturbation
        if n_stall > 100:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_nesterov(a0, time_limit, verbose=True):
    """
    LSE (log-sum-exp) smoothed gradient descent with Nesterov acceleration.
    Very fast per iteration due to FFT-based gradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.85

    t0 = time.monotonic()
    iters = 0
    lr = 5.0 / n

    T_init = best_score * 0.1
    T_final = best_score * 0.005

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress

        b = fast_autocorr(y)
        b_max = b.max()
        if b_max < 1e-15:
            break

        # LSE weights
        w = np.exp((b - b_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient via FFT
        grad = compute_autocorr_gradient(a, w)
        # Scale gradient: the LSE gradient is T * w, but we use w directly
        # df_LSE/da = T * sum_i (w_i/T) * db_i/da = sum_i w_i * db_i/da
        # So grad is already correct (it's the gradient of T*log(sum(exp(b/T))))
        # Actually: d(T*log(sum(exp(b_i/T))))/da_j = sum_i exp(b_i/T)/sum(exp(b_i/T)) * db_i/da_j
        # = sum_i w_i * db_i/da_j
        # Which is exactly what compute_autocorr_gradient gives us with weights=w.

        # Project onto tangent space
        grad -= np.mean(grad)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        step = lr / (1.0 + iters * 0.001)

        # Backtracking
        f_curr = T * np.log(np.exp((b - b_max) / T).sum()) + b_max
        trial_lr = step
        for _ in range(5):
            a_trial = a - trial_lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                trial_lr *= 0.5
                continue
            a_trial = a_trial * norm / s_t
            b_trial = fast_autocorr(a_trial)
            f_trial = T * np.log(np.exp((b_trial - b_trial.max()) / T).sum()) + b_trial.max()
            if f_trial < f_curr:
                break
            trial_lr *= 0.5

        # Nesterov update
        v = beta * v - trial_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        y = a.copy()  # Simplified: no separate Nesterov lookahead
        iters += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adapt beta
        beta = max(0.5, 0.85 - 0.2 * progress)

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Neu iter={iters} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_initializations(n, n_starts=5):
    """Generate diverse initial sequences."""
    inits = []

    # Sinusoidal
    x = np.linspace(0.0, 1.0, n)
    s1 = np.sin(np.pi * x) ** 1.5
    inits.append(s1 / s1.sum())

    # Gaussian
    s2 = np.exp(-((x - 0.5) ** 2) / (0.05))
    inits.append(s2 / s2.sum())

    # Raised cosine
    s3 = 0.5 * (1 + np.cos(2 * np.pi * x))
    s3 = np.maximum(s3, 1e-10)
    inits.append(s3 / s3.sum())

    # Bimodal
    s4 = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    inits.append(s4 / s4.sum())

    # Random smooth
    rng = np.random.default_rng(123)
    raw = np.abs(rng.normal(0, 1, n))
    # Smooth it
    kernel = np.ones(10) / 10
    s5 = np.convolve(raw, kernel, mode='same')
    s5 = np.maximum(s5, 1e-10)
    inits.append(s5 / s5.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    - LSE Nesterov (fast gradient-based)
    - Power-mean gradient descent
    - LP Nesterov with ultra-aggressive constraint reduction
    - Spectral shaping
    - Coordinate descent polish
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        inits = generate_initializations(n_base)
        best_init = inits[0]
        best_init_score = float('inf')
        for init in inits:
            sc = evaluate_safe(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE Nesterov (fast, 25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_nesterov(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse1_n{n_cur}")

    # === Phase 2: Power-mean gradient descent (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: PowerMean GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = power_mean_gradient_descent(a_norm, t2, p_start=4.0, p_end=60.0, verbose=True)
        do_try_update(a_out, f"pmgd_n{n_cur}")

    # === Phase 3: LP Nesterov with aggressive constraint reduction (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t3, solver, verbose=True, max_active=30)
        do_try_update(a_out, f"lp_n{n_cur}")

    # === Phase 4: Coordinate descent polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CoordDesc n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t4, seed=777, verbose=True)
        do_try_update(a_out, f"cd_n{n_cur}")

    # === Phase 5: LP Nesterov round 2 with fewer constraints (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t5, solver, verbose=True, max_active=20)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_try_list = [n_base - 100, n_base + 100, n_base + 200, n_base - 50]
    n_try_list = [x for x in n_try_list if 500 <= x <= 3000]

    for n_try in n_try_list:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n=== Phase 6: Try n={n_try} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.4, 150.0)
        if t6 < 20:
            continue
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        # Quick LSE optimization
        a_out, _ = lse_nesterov(a_norm, min(t6 * 0.5, 60.0), verbose=False)
        do_try_update(a_out, f"lse_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            continue
        # Quick LP
        n_cur_try = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur_try) / best_seq.sum()
        t6b = min(remaining * 0.3, 60.0)
        if t6b > 15:
            a_out, _ = lp_nesterov_optimize(a_norm, t6b, solver, verbose=False, max_active=25)
            do_try_update(a_out, f"lp_n{n_try}")

    # === Phase 7: Final LSE polish (remaining time * 0.3) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.3, 150.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_nesterov(a_norm, t7, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_cur}")

    # === Phase 8: Final LP (remaining time * 0.3) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.3, 120.0)
    if t8 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t8, solver, verbose=True, max_active=20)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final CD n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.3, 80.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t9, seed=31415, verbose=True)
        do_try_update(a_out, f"cd_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
