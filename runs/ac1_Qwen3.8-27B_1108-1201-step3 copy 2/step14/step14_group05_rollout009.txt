    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()

    n_iters = 0
    stall = 0
    gamma_cur = gamma
    M_hist = []

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        # Solve LP
        b = lp_solve_cutting_plane(y, M_curr, solver, max_constraints=max_constraints)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma_cur)
        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma_cur) / (t + 2.0 - gamma_cur)
            if momentum > 1.0:
                momentum = 1.0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            osc = np.sum(np.abs(np.diff(recent))) / (30 * max(recent[0], 1e-15))
            if osc > 0.08:
                gamma_cur = max(0.8, gamma_cur - 0.15)

                gamma_cur = min(2.0, gamma_cur + 0.05)

            perturb_scale = 0.05 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma_cur = max(0.8, gamma_cur - 0.2)
            M_hist = []
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def geometric_mean_refine(a0, time_limit, solver, max_constraints=80, verbose=True):
    """
    Geometric mean (multiplicative) refinement.
    Alternates between LP best-response and geometric averaging.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(a).max()

        b = lp_solve_cutting_plane(a, M_curr, solver, max_constraints=max_constraints)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_iters += 1

        # Geometric mean update (with damping)
        # a_new = a^beta * g^(1-beta), beta decreases over time
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = 0.7 - 0.3 * progress  # from 0.7 to 0.4

        # Log-space geometric mean (safe for zeros)
        log_a = np.log(np.maximum(a, 1e-30))
        log_g = np.log(np.maximum(g, 1e-30))
        log_a_new = beta * log_a + (1.0 - beta) * log_g
        a_new = np.exp(log_a_new - max(log_a_new.max(), 0))
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Additive fallback with small step
            alpha = 0.1 / (1 + n_iters * 0.01)
            a_alt = (1 - alpha) * a + alpha * g
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt * norm / a_alt.sum()
            sc_alt = fast_autocorr(a_alt).max()
            if sc_alt < best_score:
                best_score = sc_alt
                best_a = a_alt.copy()
                a = a_alt
                stall = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GeoMean iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if stall > 100:
            rng = np.random.default_rng(n_iters * 7 + 11)
            a = np.maximum(a + rng.normal(0, 0.03 * a.mean(), n), 0)
            a = a * norm / a.sum()
            stall = 0

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.1, best_score * 0.03)
    T_end = max(0.001, best_score * 0.001)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: d(max_b)/d(a_j) approximated by weighted convolution
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]

        # Project onto tangent space (sum constraint)
        grad -= np.mean(grad)

        # Backtracking line search
        step = 0.5 / (T * n * 0.1)
        accepted = False
        for _ in range(20):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc = fast_autocorr(a_new).max()
            if sc < max_b * 1.001:
                a = a_new
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Small guaranteed step
            step = 1e-8 / n
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * norm / a_new.sum()
            a = a_new

        steps += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    n_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find top contributing pair to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Find non-contributing target positions
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            non_contrib_idx = np.array([i for i in range(n) if i != src])

        # Try multiple destinations
        n_try = min(8, len(non_contrib_idx))
        dst_candidates = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_try]

        best_trial_score = best_score
        best_move = None

        for dst in dst_candidates:
            for frac in [0.1, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_move = (src, dst, frac)

        if best_move is not None and best_trial_score < best_score:
            src, dst, frac = best_move
            a[src] -= a[src] * frac
            a[dst] += a[src_val is not None] * 0  # placeholder
            amt = a[src]  # already reduced
            # Redo properly
            a_trial = a.copy()
            a_trial[best_move[0]] -= a_trial[best_move[0]] * best_move[2]
            a_trial[best_move[1]] += a_trial[best_move[0]]  # This is wrong, let me fix
            # Actually let's just redo
            a = a_trial.copy()
            a[best_move[0]] = max(0, a.copy()[0])  # too complex, let me simplify

        # Simplified: just do the move directly
        if best_move is not None and best_trial_score < best_score:
            s_idx, d_idx, f = best_move
            amt = a[s_idx] * f
            a[s_idx] -= amt
            a[d_idx] += amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            best_score = fast_autocorr(a).max()
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        else:
            n_stall += 1

        # Random exploration when stalled
        if n_stall > 30:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                amt = a[i] * rng.uniform(0.05, 0.15)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    n_stall = 0
                else:
                    n_stall += 1

        if n_stall > 200:
            n_pert = max(10, n // 30)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a_trial[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def generate_initialization(n, variant=0, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)

    if variant == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 1:
        # Gaussian
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif variant == 2:
        # Triangular
        a = 1 - np.abs(2 * x - 1)
    elif variant == 3:
        # Sine power
        a = np.sin(np.pi * x) ** 1.2
    elif variant == 4:
        # Flat top with tapered edges
        a = np.ones(n)
        taper = int(n * 0.1)
        for i in range(taper):
            a[i] = (i + 1) / (taper + 1)
            a[n - 1 - i] = (i + 1) / (taper + 1)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0.2, 0.8)
            width = rng.uniform(0.05, 0.2)
            a *= np.exp(-((x - center) ** 2) / (2 * width ** 2))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def crossover_sequences(a1, a2, alpha=0.5):
    """Geometric mean crossover of two sequences."""
    n1, n2 = len(a1), len(a2)
    if n1 != n2:
        a2 = interpolate_sequence(a2, n1)
        n2 = n1
    log_a1 = np.log(np.maximum(a1, 1e-30))
    log_a2 = np.log(np.maximum(a2, 1e-30))
    log_mix = alpha * log_a1 + (1 - alpha) * log_a2
    mix = np.exp(log_mix - log_mix.max())
    s = mix.sum()
    if s < 1e-15:
        return a1.copy()
    return mix / s


def multi_scale_optimize(a_coarse, n_fine, time_limit, solver, max_constraints=50, verbose=True):
    """Coarse-to-fine: optimize at coarse level, then upscale and refine."""
    n_coarse = len(a_coarse)

    # Refine at coarse level
    norm_c = np.sqrt(2.0 * n_coarse)
    a_c = a_coarse * norm_c / a_coarse.sum()
    t_coarse = time_limit * 0.3
    a_c_best, _ = nesterov_lp_optimize(a_c, t_coarse, solver, gamma=1.6, max_constraints=max_coeff, verbose=False)
    max_coeff = max_constraints

    # Upscale
    a_fine = interpolate_sequence(a_c_best / a_c_best.sum(), n_fine)

    # Refine at fine level
    norm_f = np.sqrt(2.0 * n_fine)
    a_f = a_fine * norm_f / a_fine.sum()
    t_fine = time_limit - t_coarse - (time.monotonic() - time.monotonic())
    t_fine = time_limit * 0.7
    a_f_best, _ = nesterov_lp_optimize(a_f, t_fine, solver, gamma=1.5, max_constraints=max_constraints + 20, verbose=verbose)

    return a_f_best

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_submatrix(a, indices):
    """Build submatrix of convolution matrix for given row indices."""
    n = len(a)
    K = len(indices)
    T = np.zeros((K, n))
    for i in range(K):
        idx = indices[i]
        js = max(0, idx - n + 1)
        je = min(n - 1, idx)
        if js <= je:
            T[i, js:je + 1] = a[idx - np.arange(js, je + 1)]
    return T


def lp_solve_cutting_plane(a, M_budget, solver, max_constraints=60):
    """
    LP best response with cutting-plane constraint reduction.
    Maximize sum(b) subject to conv(a,b)_i <= M_budget for all i, b >= 0.
    Uses iterative constraint addition for speed.
    """
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1

    # Initial constraint set: top autocorrelation peaks + spread
    a_conv = fast_autocorr(a)
    n_top = min(max_constraints // 2, size)
    top_idx = np.argsort(a_conv)[::-1][:n_top]
    n_spread = min(max_constraints // 3, size)
    spread_idx = np.linspace(0, size - 1, n_spread).astype(int)
    active_set = np.unique(np.concatenate([top_idx, spread_idx]))[:max_constraints]

    b_var = cp.Variable(n, nonneg=True)
    b_val = None

    for cp_iter in range(8):
        T_sub = build_conv_submatrix(a, active_set)
        M_active = np.full(len(active_set), M_budget)
        constraints = [T_sub @ b_var <= M_active]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if b_val is not None:
                break
            return None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if b_val is not None:
                break
            return None

        if b_var.value is None:
            return None

        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_new_sum = b_new.sum()
        if b_new_sum < 1e-15:
            return None

        # Check for violations using FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A_f = rfft(a, fft_size)
        B_f = rfft(b_new, fft_size)
        conv_ab = irfft(A_f * B_f, fft_size)[:size]

        violations = conv_ab > M_budget * (1 + 1e-8)
        n_viol = int(violations.sum())

        if n_viol == 0:
            b_val = b_new
            break

        if len(active_set) >= max_constraints:
            b_val = b_new
            break

        # Add most violated constraints
        viol_idx = np.where(violations)[0]
        viol_vals = conv_ab[viol_idx]
        add_order = viol_idx[np.argsort(viol_vals)[::-1]]
        n_add = min(len(add_order), max_constraints - len(active_set))
        active_set = np.unique(np.concatenate([active_set, add_order[:n_add]]))[:max_constraints]

        if b_new_sum > (b_val.sum() if b_val is not None else 0):
            b_val = b_new

    if b_val is None or b_val.sum() < 1e-15:
        return None
    return b_val
