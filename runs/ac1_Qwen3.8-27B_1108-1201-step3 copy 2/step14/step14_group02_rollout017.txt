def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]

    except Exception:

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def frank_wolfe_lp(a0, time_limit, solver, verbose=True, max_iters=None):
    """
    Nesterov-accelerated Frank-Wolfe with LP best-response.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_nest = a.copy()
    iter_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if max_iters is not None and iter_count >= max_iters:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break

        # Solve inner LP: max sum(b) s.t. (y*b)_k <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraint matrix T: T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        # T @ b gives the convolution y * b
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_nest = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x_nest).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nest.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def multiplicative_weights(a0, time_limit, verbose=True, top_k=10, beta=3.0):
    """
    Multiplicative weights / mirror descent on the simplex.
    Uses the subgradient from the top-k active constraints.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Polyak step size
    gamma = 1.0  # will be tuned
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top-k active constraints
        top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
        top_vals = ac[top_idx]
        
        # Soft subgradient
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(w)) / len(w)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]

        # Normalized gradient (for mirror descent)
        step = gamma / (np.sqrt(t) + 1.0)
        
        # Multiplicative update: a_j <- a_j * exp(-step * grad_j)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MW t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=False):

    Targeted peak shaving: iteratively identify the worst peak and

    """

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:

        M_curr = ac.max()
        k_star = int(np.argmax(ac))

            break

        src_val = a[src]

            stall += 1
            continue

        # Score each candidate destination by how much it increases the max autocorr
        # Use a fast approximation: compute the "sensitivity" of each position
        
        # For efficiency, sample candidates

def gradient_descent_refine(a0, time_limit, verbose=False):
    """
    Projected gradient descent with careful step size selection.
    Uses the exact subgradient of the max function.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    # Backtracking line search projected gradient
    step = 0.01
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Exact subgradient: gradient at the active constraint(s)
        # Find all lags within 10% of max
        threshold = M_curr * 0.9
        active = np.where(ac >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        
        # Average subgradient over active constraints
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * a[j]
        grad /= len(active)
        
        # Projected gradient step on simplex
        # a_new = proj_simplex(a - step * grad)
        a_trial = a - step * grad
        
        # Project onto simplex (a >= 0, sum(a) = 1)
        # Simple projection: clip to 0, renormalize
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s
        
        M_trial = fast_autocorr(a_trial).max()
        
        if M_trial < best_m:
            best_m = M_trial
            best_a = a_trial.copy()
            a = a_trial
            # Increase step slightly
            step = min(step * 1.05, 0.1)
        else:
            # Decrease step
            step = max(step * 0.7, 1e-6)
            # Occasionally try a random perturbation
            if it % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GD it={it} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
