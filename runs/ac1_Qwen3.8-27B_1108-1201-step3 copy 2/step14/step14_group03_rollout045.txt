def build_conv_matrix(a, n_constraints=None):
    """Build the convolution matrix T where (T@b)_k gives the k-th element of a*b."""
    n = len(a)
    size = 2 * n - 1
    if n_constraints is not None and n_constraints < size:
        # Only include a subset of constraints (those with highest a*a values)
        aa = fast_autocorr(a)
        idx = np.argsort(aa)[::-1][:n_constraints]
        idx = np.sort(idx)
    else:
        idx = np.arange(size)
    T = np.zeros((len(idx), n))
    for k in idx:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            T[list(idx).index(k), j] = a[j]
    return T, idx


def build_conv_matrix_fast(a):
    """Build convolution matrix efficiently using structured layout."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = min(size, j + n)
        T[lo:hi, j] = a[j]
        lo2 = max(0, n - 1 - j)
        hi2 = min(size, n - 1 + j + 1)
        # The reflected part: a_j contributes to position k = n-1-(n-1-j) ... 
        # Actually for linear convolution: a_j * b_i contributes to position j+i
        # So a_j contributes to positions j, j+1, ..., j+n-1 (from b_0 to b_{n-1})
        # That's it - each a_j only appears in one contiguous block!

    return T

    T = build_conv_matrix_fast(a)

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()

    stuck_count = 0

    last_improve_iter = 0

        if M_curr < 1e-12:

                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:

            a_new = g.copy()
        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        a = a_new

        # Track best

def lse_momentum_optimize(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent with momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Multi-stage temperature annealing
    n_stages = 8
    temps = np.geomspace(20.0, 0.5, n_stages)
    momentum_buf = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(500, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            # Smoothed objective: (1/T) * log(sum exp(T*(conv - max_conv)/conv_sum))
            scaled = T * (conv - max_conv) / (conv.sum() + 1e-30)
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()

            # Gradient: d/d(a_i) [sum_k w_k * conv_k] 
            # = 2 * sum_k w_k * a_{k-i} (for valid k-i in [0,n-1])
            # Compute via FFT (cross-correlation of w and a)
            fft_size = 1 << (3 * n - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            cross = irfft(W * np.conj(A), fft_size)
            grad_direct = 2.0 * cross[:n]

            # Also account for the normalization (sum a = norm)
            # Using Lagrangian: subtract mean component
            grad = grad_direct - np.mean(grad_direct)

            # Adam-style update
            momentum_buf = beta1 * momentum_buf + (1 - beta1) * grad
            step_size = norm / (T * np.sqrt(n)) * 0.5
            a_new = a - step_size * momentum_buf / (np.sqrt(np.abs(momentum_buf).mean() + 1e-10))
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent: move mass from position i to position j."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue

                    # Compute change in convolution if we move delta from i to j
                    # New conv = old conv + delta * (delta_vec)
                    # where delta_vec[k] = 2*(a_{k-j} - a_{k-i}) for valid indices
                    delta_vec = np.zeros(size)
                    # Contribution of a_j (increased by delta):
                    # conv gets +delta * a_m at position j+m for m in [0,n-1]
                    # and +delta * a_m at position (n-1-m)+j ... 
                    # Actually: (a+delta*e_j - delta*e_i) * (a+delta*e_j - delta*e_i)
                    # ≈ a*a + delta*(e_j*a + a*e_j - e_i*a - a*e_i) to first order
                    # The change at position k is: delta*(a_{k-j} + a_{k-i} - a_{k-i} - a_{k-j})
                    # Wait, that's zero to first order in the cross terms... 
                    # Let me be more careful.
                    # (a') = a + delta*(e_j - e_i)
                    # (a'*a')_k = (a + delta*(e_j - e_i))^T conv ... 
                    # = a*a_k + delta*(a*e_j)_k + delta*(e_j*a)_k - delta*(a*e_i)_k - delta*(e_i*a)_k
                    #   + delta^2*(e_j*a - e_i*a - e_j*e_i + e_i*e_j)_k
                    # = conv_k + delta*(a_{k-j} + a_{j+k-n+1}... ) 
                    # This is getting complicated. Let me just compute it directly.
                    
                    # Simpler: just compute the delta vector directly
                    # For position k in [0, 2n-2]:
                    # (a'*a')_k - (a*a)_k where a' = a + delta*(e_j - e_i)
                    # = delta * (a_{k-j} + a_{k-i-reflected} - a_{k-i} - a_{k-j-reflected})
                    # + delta^2 * (terms involving delta)
                    
                    # Let's use the formula: d(conv_k)/da_i = 2*a_{k-i} if 0<=k-i<n
                    # So the change is: delta * 2 * (a_{k-j} - a_{k-i}) for valid k
                    # (ignoring the delta^2 terms for the search)
                    
                    lo_j = max(0, j)
                    hi_j = min(size, j + n)
                    if hi_j > lo_j:
                        delta_vec[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
                    lo_i = max(0, i)
                    hi_i = min(size, i + n)
                    if hi_i > lo_i:
                        delta_vec[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]
                    
                    # Binary search for best delta in [0, a[i]]
                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas[1:]:  # skip d=0
                        new_conv_max = (conv + d * delta_vec).max()
                        if new_conv_max < best_val - 1e-12:
                            best_val = new_conv_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                        else:
                            # Revert
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)
                        break  # break offset loop after a move

        if not pass_improved:
            break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov at parent length (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Try longer sequence (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400  # Try a moderately longer sequence
    t2 = remaining * 0.35
    print(f"\n=== Phase 2: Longer sequence n={n_long} (budget={t2:.0f}s) ===")
    if t2 > 60 and best_seq is not None:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm_long = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        # Split time between LP Nesterov and LSE
        t2a = t2 * 0.6
        t2b = t2 * 0.4
        if t2a > 30:
            a_out_long, _ = nesterov_lp_optimize(a_norm_long, t2a, solver, verbose=True)
            do_try_update(a_out_long, f"lp_long_n{n_long}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20 and best_seq is not None and len(best_seq) == n_long:
            a_norm_long2 = best_seq * np.sqrt(2.0 * n_long) / best_seq.sum()
            a_out_long2, _ = lse_momentum_optimize(a_norm_long2, min(t2b, remaining2 * 0.8), verbose=True)
            do_try_update(a_out_long2, f"lse_long_n{n_long}")

    # === Phase 3: LSE gradient at best length (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LSE momentum n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm3 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out3, _ = lse_momentum_optimize(a_norm3, t3, verbose=True)
        do_try_update(a_out3, f"lse_n{n_best}")

    # === Phase 4: LP Nesterov round 2 at best length (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm4 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out4, _ = nesterov_lp_optimize(a_norm4, t4, solver, verbose=True)
        do_try_update(a_out4, f"lp2_n{n_best}")

    # === Phase 5: Pair CD polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 10:
        a_norm5 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out5, _ = pair_cd_polish(a_norm5, remaining, rng, window=25, verbose=True)
        do_try_update(a_out5, f"pcd_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
