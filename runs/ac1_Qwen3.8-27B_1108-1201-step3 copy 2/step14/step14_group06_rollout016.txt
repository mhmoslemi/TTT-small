    n = len(a)
    S = a.sum()
    ffsize = 1 << (2*n-1).bit_length()
    a = np.maximum(a.copy(), 0.0)
    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, ffsize)
        P = np.abs(A)**2
        
        # Flatten the power spectrum
        P_flat = np.maximum(P, 1e-20)
        P_mod = P_flat ** (1.0/alpha)
        
        # Reconstruct with modified magnitude
        A_mod = np.sqrt(P_mod) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_mod)[:n])

        peak = fast_autocorr(a_new).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_peak

def build_constraint_matrix(a, n):

        start = i
        end = min(i + n, size)

    return T

def spectral_equalize(a0, time_limit, alpha=0.85, n_inner=50):
    """Spectral equalization: flatten power spectrum while preserving phase."""
    n = len(a0)
    S = a0.sum()

        P_mod = np.maximum(P_mod, 1e-30)

            peak_a = fast_autocorr(a).max()
            if peak_a < best_peak:
                best_peak = peak_a
                best_a = a.copy()
            break

    return best_a, best_peak

def subgradient_optimize(a0, time_limit, solver=None):
    """Subgradient method with backtracking line search on max convolution."""
    n = len(a0)
    S = a0.sum()
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(a0.copy(), 0.0)
    a = a * norm_val / max(a.sum(), 1e-15)

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()

    step_base = 0.1
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Subgradient: 2 * a[k_star - i] for valid i
        # Equivalent to: grad[m] = 2*a[k_star - m] if 0 <= k_star-m < n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_star - m]

        # Also check if multiple peaks are active
        threshold = peak_val * 0.999
        active_peaks = np.where(c >= threshold)[0]
        if len(active_peaks) > 1:
            for k2 in active_peaks:
                lo2 = max(0, k2 - n + 1)
                hi2 = min(n - 1, k2)
                for m in range(lo2, hi2 + 1):
                    grad[m] += 2.0 * a[k2 - m]
            grad /= len(active_peaks)

        # Adaptive step size
        step = step_base / np.sqrt(t + 1.0)

        # Try the step with backtracking
        for tries in range(10):
            a_try = a - step * grad
            # Project onto {a >= 0, sum(a) = norm_val}
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try * norm_val / s_try
            peak_try = fast_autocorr(a_try).max()

            if peak_try < peak_val:
                a = a_try
                if peak_try < best_peak:
                    best_peak = peak_try
                    best_a = a.copy()
                    no_improve = 0
                break
            else:
                step *= 0.5

        if no_improve > 200:
            # Perturb to escape local minim
            rng_local = np.random.default_rng(t)
            pert = rng_local.normal(0, 0.01, n)
            a = a + pert
            a = np.maximum(a, 0.0)
            a = a * norm_val / max(a.sum(), 1e-15)
            no_improve = 0
            if fast_autocorr(a).max() < best_peak:
                best_peak = fast_autocorr(a).max()
                best_a = a.copy()
        else:
            # Track no-improve counter loosely
            if peak_val >= best_peak:
                no_improve += 1

    return best_a, best_peak

            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_smoothed_optimize(a0, time_limit):
    """Log-sum-exp smoothed optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for T_val in [60.0, 20.0, 5.0, 1.0]:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def obj_func(x, T_use=T_val):
            a_use = np.maximum(x, 0.0)
            s_use = a_use.sum()
            if s_use < 1e-10:
                return 1e10
            c = fast_autocorr(a_use)
            max_c = c.max()
            if max_c < 1e-15:
                return 1e10
            shifted = (c - max_c) / T_use
            log_sum = max_c + np.log(np.exp(shifted).sum()) * T_use / len(c)
            return 2.0 * n * log_sum / (s_use * s_use)

        def grad_func(x, T_use=T_val):
            a_use = np.maximum(x, 0.0)
            s_use = a_use.sum()
            if s_use < 1e-10:
                return np.zeros(n)
            c = fast_autocorr(a_use)
            max_c = c.max()
            if max_c < 1e-15:
                return np.zeros(n)
            shifted = (c - max_c) / T_use
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()

            # Gradient of log-sum-exp smoothing times the chain rule
            # d/dx of (2n/(s^2)) * LSE(c(x))
            # = (2n/(s^2)) * sum_k w_k * dc[k]/dx - (4n*LSE)/(s^3) * x
            # dc[k]/dx[m] = 2*a[k-m] (if valid) -> use cross-correlation
            A_use = rfft(a_use, fft_size)
            W_use = rfft(np.zeros(fft_size), fft_size)  # placeholder

            # Compute gradient via: for each m, grad[m] = 2*sum_k w[k]*a[k-m]
            # This is 2 * (w * a) circular cross-correlation, taken first n
            w_padded = np.zeros(fft_size)
            w_padded[:2*n-1] = w
            a_padded = np.zeros(fft_size)
            a_padded[:n] = a_use

            W_fft = rfft(w_padded, fft_size)
            A_fft = rfft(a_padded, fft_size)
            cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

            lse_val = max_c + np.log(exp_vals.sum()) * T_use / len(c)
            obj_val = 2.0 * n * lse_val / (s_use * s_use)
            grad = 2.0 * n * 2.0 * cross / (s_use * s_use) - 2.0 * obj_val * a_use / s_use
            return grad

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-7, 'maxls': 30})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    peak_c = fast_autocorr(a_cand).max()
                    if peak_c < best_score:
                        best_score = peak_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        print(f"    LSE T={T_val:.1f}: peak={peak_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T_val:.1f} error: {e}")

    return best_a, best_score


def peak_targeted_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        k_peak = int(np.argmax(c))

        # Identify peak-contributing elements
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_contrib = np.argsort(contrib)[::-1]
        top_src = idx[sorted_contrib[:12]]

        # Identify target elements (low contribution to any peak)
        peak_mask = np.zeros(n, dtype=bool)
        top_peaks = np.argsort(c)[::-1][:12]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        targets = non_peak[np.argsort(a[non_peak])[:12]]

        improved = False
        for src in top_src:
            if a[src] < 1e-10:
                continue
            for dst in targets:
                if dst == src:
                    continue
                for frac in [0.03, 0.07, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbations
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Try a larger random perturbation
                pert = rng.normal(0, 0.05, n)
                a = a + pert
                a = np.maximum(a, 0.0)
                a = a * norm_val / max(a.sum(), 1e-15)
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                stall = 0
            elif stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Asymmetric: high on left, low on right (exploits anti-correlation)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.4 * np.exp(-6.0 * (x - 0.8) ** 2)
    else:
        # Multi-peak
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8*(x-0.15)**2) + 0.7*np.exp(-8*(x-0.5)**2) + 0.4*np.exp(-8*(x-0.85)**2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: spectral equalization + LP Nesterov + subgradient + coord descent."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2000
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral equalization (NEW method for diversification)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 1: Spectral equalization n={n_curr} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = spectral_equalize_adaptive(a_norm, t1, n_alphas=4)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"spectral_n{n_curr}", t_global)

    # Phase 2: LP Nesterov with aggressive M (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: LP-Nest aggressive n={n_curr} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, aggressive=True)
        best_seq, best_score = attempt_update((best_seq, best_score), f"lp_agg_n{n_curr}", t_global)

    # Phase 3: Subgradient method
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: Subgradient n={n_curr} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t3)
        best_seq, best_score = attempt_update((best_seq, best_score), f"subgrad_n{n_curr}", t_global)

    # Phase 4: LSE smoothed optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: LSE smoothed n={n_curr} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = lse_smoothed_optimize(a_norm, t4)
        best_seq, best_score = attempt_update((best_seq, best_score), f"lse_n{n_curr}", t_global)

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_curr} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = peak_targeted_coord_descent(a_norm, t5, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), f"cd1_n{n_curr}", t_global)

    # Phase 6: LP round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 6: LP-Nest round 2 n={n_curr} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False, aggressive=True)
        best_seq, best_score = attempt_update((best_seq, best_score), f"lp_r2_n{n_curr}", t_global)

    # Phase 7: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 7: Longer sequence n={n_long} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, aggressive=True)
        best_seq, best_score = attempt_update((best_seq, best_score), f"lp_long_n{n_long}", t_global)

    # Phase 8: Final polish - alternate methods
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_curr} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        half = remaining / 2.0

        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = peak_targeted_coord_descent(a_norm, half, seed=99999)
        best_seq, best_score = attempt_update((best_seq, best_score), "cd_final1", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            n_curr2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_curr2) / best_seq.sum()
            a_out, _ = subgradient_optimize(a_norm2, remaining2)
            best_seq, best_score = attempt_update((best_seq, best_score), "subgrad_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
