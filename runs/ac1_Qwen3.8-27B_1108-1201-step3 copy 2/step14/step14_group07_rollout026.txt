`2 * shift(a, k*) - mu/a = nu * 1`

1. Start with current `a`

5. Update `a[i] = max(0, mu/(g[i]-nu))`
6. Repeat

        it += 1
        mu = max(0.001, 0.5 * (0.5) ** (it / 200.0))
        
        b = fast_autocorr(a)
        k_star = int(b.argmax())

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
        
        # Water-filling: find nu such that sum(max(0, mu/(g-nu))) = norm
        g_max = g.max()
        if g_max < 1e-15:
            break
        
        # Bisection on nu
        lo_nu = 0.0
        hi_nu = g_max
        for _ in range(40):
            mid = (lo_nu + hi_nu) / 2.0
            val = np.maximum(0.0, mu / np.maximum(g - mid, 1e-15))
            val = np.where(g > mid, mu / (g - mid), 0.0)
            if val.sum() > norm:
                lo_nu = mid
            else:
                hi_nu = mid
        
        nu = (lo_nu + hi_nu) / 2.0
        a_new = np.where(g > nu, mu / (g - nu), 0.0)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Damped update
        alpha = 0.3
        a_trial = np.maximum((1 - alpha) * a + alpha * a_new, 0.0)
