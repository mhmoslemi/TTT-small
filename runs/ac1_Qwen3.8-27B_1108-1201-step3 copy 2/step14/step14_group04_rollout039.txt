"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-FISTA warm-up (fast initial convergence)
2. Accelerated greedy best response (fast iterations, no LP needed)
3. LP-Nesterov (exact best response for polishing)
4. Smooth LSE gradient descent (refinement)
5. Targeted single-peak shaving (final polish)

Key innovation: Greedy best response approximation that avoids LP solves,
allowing 10-100x more iterations in the same time budget.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s

def greedy_best_response(a, M, K=5):
    """
    Greedy approximation of LP best response.
    For each of the top K peaks, compute a feasible direction,
    then take element-wise minimum.
    """
    n = len(a)
    r = fast_autocorr(a)
    
    # Find top K peak locations
    top_k = np.argsort(r)[::-1][:K]
    
    # Initialize b as very large (will be reduced by constraints)
    b = np.full(n, M * n)  # large upper bound
    
    for k in top_k:
        # Constraint: sum_j a[j]*b[k-j] <= M
        # Valid j: 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        
        if j_min > j_max:
            continue
        
        # The b-indices involved: b[k-j] for j in [j_min, j_max]
        # = b[k-j_min], b[k-j_min-1], ..., b[k-j_max]
        # = b[k-j_min : k-j_max-1 : -1]... hmm, let me think differently
        
        # b_index = k - j, so as j goes from j_min to j_max,
        # b_index goes from k-j_min down to k-j_max
        b_indices = k - np.arange(j_min, j_max + 1)
        a_coeffs = a[j_min:j_max + 1]
        
        # Greedy: set b[b_index] = M / (num_indices * a[j])
        # This makes the constraint tight: sum a[j]*b[k-j] = sum M/num = M
        count = len(b_indices)
        for idx in range(count):
            bi = b_indices[idx]
            aj = a_coeffs[idx]
            if aj > 1e-15:
                candidate = M / (count * aj)
            else:
                candidate = M * 10  # large if a[j] is tiny
            b[bi] = min(b[bi], candidate)
    
    # Ensure non-negative
    b = np.maximum(b, 0.0)
    
    return b

def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = n / max(T * 2.0, 1.0)
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            r = fast_autocorr(y)
            r_max = r.max()
            if r_max < 1e-15:
                break
            shifted = (r - r_max) / T
            shifted = np.clip(shifted, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            
            # Gradient: grad[m] = sum_k w[k]*a[k-m] + a[m]
            N_fft = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N_fft)
            Y = rfft(y, N_fft)
            cross = irfft(W * np.conj(Y), N_fft)[:n]
            grad = cross + y
            
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(np.maximum(y_new, 0.0), 1.0)
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"  LSE-FISTA stage={stage_i} T={T:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def accelerated_greedy(a0, time_limit, verbose=True, K=5):
    """
    Main optimization: accelerated greedy best response.
    Uses Nesterov acceleration with greedy (non-LP) best response.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    ema = a.copy()
    ema_beta = 0.01
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = greedy_best_response(y, M_curr, K=K)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # EMA
        ema = (1 - ema_beta) * ema + ema_beta * a
        if n_iters % 50 == 0:
            ema_norm = ema * norm / ema.sum()
            score_ema = fast_autocorr(ema_norm).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_norm.copy()
        
        # Regular averaging
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = avg_check * norm / avg_check.sum()
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Accel-Greedy iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def smooth_peak_descent(a0, time_limit, verbose=True, seed=42):
    """
    Smooth peak reduction using LSE gradient with decreasing temperature.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T = 0.5  # initial temperature
    T_decay = 0.999
    eta = 0.01  # initial step size
    momentum = 0.9
    v = np.zeros(n)  # momentum buffer
    stall = 0
    
    N_fft = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        r = fast_autocorr(a)
        r_max = r.max()
        
        if r_max < best_score * 1.0001:
            stall += 1
        
        # LSE weights
        shifted = (r - r_max) / T
        shifted = np.clip(shifted, -50.0, 50.0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Gradient
        W = rfft(w, N_fft)
        A = rfft(a, N_fft)
        cross = irfft(W * np.conj(A), N_fft)[:n]
        grad = cross + a
        
        # SGD with momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.05)  # slightly increase step
        else:
            a = a_new  # still accept to explore
            eta *= 0.95  # reduce step
            stall += 1
        
        T = max(T * T_decay, 0.01)
        
        if stall > 200:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 10
            if stall > 500:
                break
        
        if verbose and int(time.monotonic() - t0) %
