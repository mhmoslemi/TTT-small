import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                print(f"  Using solver: {s}")
                return s
        except Exception:

    _solver = cp.ECOS

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_fast(a, M, solver, active_set):
    """Solve the LP: max sum(b) s.t. T(a)[:, active_set]^T @ b <= M, b >= 0"""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(30.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(300, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= conv.max() + 0.01 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

    n = len(a0)

def gradient_polish(a0, time_limit, rng, verbose=True):
    """Direct gradient-based polish with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Use decreasing temperature for the softmax surrogate
    n_stages = 8
    temps = np.geomspace(5.0, 0.05, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(500, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            # Softmax weights on the convolution
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            # Project gradient to maintain sum constraint
            grad = grad - np.mean(grad)

            # Backtracking line search
            step = 1.0 / (T_val * np.sqrt(n))
            a_new = a.copy()
            for trial in range(15):
                a_candidate = a_new - step * grad
                a_candidate = np.maximum(a_candidate, 0.0)
                s_c = a_candidate.sum()
                if s_c < 1e-15:
                    step *= 0.5
                    continue
                a_candidate = a_candidate * norm / s_c
                sc = fast_conv_self(a_candidate).max()
                if sc <= max_c + 1e-8:
                    break
                step *= 0.5

            a_new = a_new - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    Grad polish stage {si} T={T_val:.3f}: best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = 0.4 + 0.3 * np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x) + 0.1*np.cos(6*np.pi*x)
    else:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(3*np.pi*x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    # Multiple averaging windows
    windows = [10, 30, 60, 120, 250]
    buffers = {w: [] for w in windows}

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening - start sparse, grow
        n_active = min(size, 30 + t * 3)
        if n_active >= size:
            active = np.arange(size)
        else:
            threshold_idx = np.argpartition(conv_y, -n_active)[-n_active:]
            active = threshold_idx

        b = lp_solve_active(y, M, solver, active)
        if b is None:
            b = lp_solve_active(y, M, solver, np.arange(size))
            if b is None:
                continue

        # Check constraint violations and re-solve if needed
        if len(active) < size:
            conv_ab = fast_conv_cross(y, b)
            violations = np.where(conv_ab > M * (1 + 5e-4))[0]
            if len(violations) > 0:
                new_active = np.union1d(active, violations)
                b2 = lp_solve_active(y, M, solver, new_active)
                if b2 is not None:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update with varying alpha
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best from y and a
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Update averaging buffers
        for w in windows:
            buffers[w].append(a.copy())
            if len(buffers[w]) > w:
                buffers[w].pop(0)

        # Check averaged iterates every 5 steps
        if t % 5 == 0:
            for w in windows:
                if len(buffers[w]) == w:
                    avg = np.mean(buffers[w], axis=0)
                    s_avg = avg.sum()
                    if s_avg > 1e-15:
                        avg = avg * norm / s_avg
                        sc_avg = fast_conv_self(avg).max()
                        if sc_avg < best_score:
                            best_score = sc_avg
                            best_a = avg.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def spectral_flatten(a0, time_limit, verbose=True):
    """Iteratively flatten the power spectrum of the sequence."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    # FFT size for the autocorrelation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multiple stages with decreasing strength
    n_stages = 10
    strengths = np.linspace(0.3, 0.05, n_stages)

    total_steps = 5000
    steps_per_stage = max(10, total_steps // n_stages)

    for si, strength in enumerate(strengths):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = min(steps_per_stage, int(remaining * 5))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            A = rfft(a, fft_size)
            mag = np.abs(A)
            mag[mag < 1e-15] = 1e-15

            # Target: reduce the dynamic range of the magnitude
            # Use a soft flattening: A_new = A * (target/mag)^strength
            target = np.median(mag[1:]) if len(mag) > 1 else mag[0]

            ratio = target / mag
            ratio = np.clip(ratio, 0.1, 10.0)
            A_new = A * ratio ** strength
            # Preserve DC
            A_new[0] = A[0]

            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new

            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose:
            print(f"    Spectral stage {si} strength={strength:.3f}: best={best_score:.8f}")

    return best_a, best_score

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(600, int(remaining * 12)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            # Project to maintain sum
            grad = grad - np.mean(grad)

            step = 2.0 / (T_val * np.sqrt(n) * np.linalg.norm(grad) + 1e-10)
            a_new = a.copy()
            for trial in range(15):
                a_candidate = a_new - step * grad
                a_candidate = np.maximum(a_candidate, 0.0)
                s_c = a_candidate.sum()
                if s_c < 1e-15:
                    step *= 0.5
                    continue
                a_candidate = a_candidate * norm / s_c
                sc = fast_conv_self(a_candidate).max()
                if sc <= max_c + 1e-7:
                    break
                step *= 0.5

            a_new = a_new - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    Grad polish stage {si} T={T_val:.4f}: best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = 0.4 + 0.3 * np.cos(2*np.pi*x) + 0.2*np.cos(4*np.pi*x) + 0.1*np.cos(6*np.pi*x)
    elif choice == 9:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(3*np.pi*x))
    elif choice == 10:
        a = 1.0 - 0.3 * np.cos(2*np.pi*x) + 0.1*np.cos(4*np.pi*x)
    else:
        a = np.exp(-2.0*(x-0.5)**2) * (1 + 0.3*np.cos(4*np.pi*x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE for initial improvement
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # Phase 2: Main LP Nesterov (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 2: LP Nesterov main n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_main_n{len(best_seq)}")

    # Phase 3: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Spectral flatten n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = spectral_flatten(best_seq, t3, verbose=True)
        try_update(a_out, f"spectral_n{len(best_seq)}")

    # Phase 4: LP round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Gradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Gradient polish (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = gradient_polish(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"grad_n{len(best_seq)}")

    # Phase 6: Try different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 300
    if remaining > 120:
        t6 = min(remaining * 0.40, 150.0)
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
        if t6 > 30:
            init_alt = interpolate_seq(best_seq, n_alt)
            a_out, _ = nesterov_lp_optimize(init_alt, min(t6, 100.0), solver, verbose=False)
            try_update(a_out, f"lp_alt_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                a_out2, _ = gradient_polish(best_seq, min(remaining * 0.4, 50.0), rng, verbose=False)
                try_update(a_out2, f"grad_alt")

    # Phase 7: Final micro-iterations
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final micro-iterations (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 50.0), verbose=False)
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, min(remaining * 0.7, 50.0), solver, verbose=False)
            try_update(a_out, f"lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = gradient_polish(best_seq, min(remaining * 0.8, 30.0), rng, verbose=False)
            try_update(a_out, f"grad_final")

    # Final evaluation and return
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
