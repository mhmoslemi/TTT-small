import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(z):
    """Project z onto {a >= 0, sum(a) = 1}."""
    n = len(z)
    # Binary search for tau
    z_sorted = np.sort(z)[::-1]
    cssv = np.cumsum(z_sorted)
    rho = np.nonzero(z_sorted * np.arange(1, n+1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0)

def fw_step(a, k_star):
    """Find the FW vertex: argmin_j a_{|k*-j|}."""
    n = len(a)
    k = int(k_star)
    dists = np.abs(k - np.arange(n))
    # g_j = 2*a_{|k*-j|} if |k*-j| < n else 0
    valid = dists < n
    vals = np.where(valid, a[dists[np.minimum(dists, n-1)]], np.inf)
    # Find argmin
    return np.argmin(vals)

def nesterov_fw(a0, time_limit, verbose=True):
    n = len(a0)
    a = project_simplex(a0.astype(float))
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        ac_y = fast_autocorr(y)
        k_star = np.argmax(ac_y)
        
        # FW vertex
        j_star = fw_step(y, k_star)
        s = np.zeros(n)
        s[j_star] = 1.0
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s
        y_new = project_simplex(y_new)
        
        if t == 1:
            x_new = s.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = project_simplex(x_new)
        
        y = y_new
        x = x_new
        
        # Check improvement
        ac_x = fast_autocorr(x)
        mx = ac_x.max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        ac_y2 = fast_autocorr(y)
        my = ac_y2.max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  FW t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def soft_subgradient(a, ac, top_k=5, temperature=0.1):
    """Compute a soft subgradient using top-k lag values."""
    n = len(a)
    # Get top k lag values
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    
    # Softmax weights
    weights = np.exp((top_vals - top_vals[0]) / temperature)
    weights /= weights.sum()
    
    # Subgradient is weighted average over top k
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        dists = np.abs(k_int - np.arange(n))
        valid = dists < n

        ac_a = fast_autocorr(a)
        g = compute_soft_subgradient(a, ac_a, top_k=top_k, power=power)
        a_new = project_simplex(a - step * g)
        ac_new = fast_autocorr(a_new)
        m_new = ac_new.max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            step = min(step * 1.05, 1.0)
        else:
            stall += 1
            step *= 0.7

        a = a_new

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PG t={t} ({elapsed:.1f}s): J={J:.6f}, step={step:.2e}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PG done: {t-1} iters, {elapsed:.1f}s, J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j = np.arange(n)
            dists = np.abs(k_int - j)
            valid = dists < n
            cost = np.where(valid, a[dists], 1e10)

            src_candidates = np.where(a > 1e-8)[0]
            if len(src_candidates) == 0:
                break
            src_scored = cost[src_candidates] * a[src_candidates]
            src = src_candidates[np.argmax(src_scored)]

            dst_candidates = np.where(cost < cost[src] * 0.8)[0]
            if len(dst_candidates) == 0:
                dst_candidates = np.where(cost == cost.min())[0]
            dst = dst_candidates[np.argmin(cost[dst_candidates])]

            if src == dst:
                continue

            for frac in [0.05, 0.02, 0.01, 0.005, 0.002]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j2 = rng.integers(n)
            if i != j2 and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j2] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def safe_evaluate(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = safe_evaluate(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent_norm = p / p.sum()
            best_seq = parent_norm.copy()
            best_score = safe_evaluate(parent_norm)
            print(f"Parent: n={len(parent_norm)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = safe_evaluate(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Nesterov FW at parent length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov FW n={n0} ===")
    t1 = min(remaining * 0.40, 700.0)
    if t1 > 30:
        a_in = project_simplex(best_seq)
        a_out, _ = nesterov_fw(a_in, t1, top_k=5, power=2.0, verbose=True)
        try_update(a_out, f"fw_n{n0}")

    # === Phase 2: Mirror descent at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = project_simplex(best_seq)
        a_out, _ = mirror_descent(a_in, t2, top_k=5, power=2.0, gamma=0.3, verbose=True)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 3: Projected gradient at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Projected Gradient n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 180.0)
    if t3 > 15:
        a_in = project_simplex(best_seq)
        a_out, _ = projected_gradient(a_in, t3, top_k=5, power=2.0, verbose=True)
        try_update(a_out, f"pg_n{len(best_seq)}")

    # === Phase 4: Try larger length (interpolate + optimize) ===
    n1 = min(n0 + 800, 3200)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Scale to n={n1} ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30:
        a_long = interpolate_to_length(best_seq, n1)
        a_in = project_simplex(a_long)
        a_out, _ = nesterov_fw(a_in, t4 * 0.6, top_k=5, power=2.0, verbose=True)
        try_update(a_out, f"fw_n{n1}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_in = project_simplex(best_seq)
            a_out, _ = mirror_descent(a_in, t4 * 0.4, top_k=5, power=2.0, gamma=0.3, verbose=False)
            try_update(a_out, f"md_n{len(best_seq)}_p4")

    # === Phase 5: Coordinate polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate Polish n={n_best} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_in = project_simplex(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Second round FW with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: FW round 2 (top_k=8, power=1.5) n={n_best} ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 30:
        a_in = project_simplex(best_seq)
        a_out, _ = nesterov_fw(a_in, t6, top_k=8, power=1.5, verbose=True)
        try_update(a_out, f"fw2_n{n_best}")

    # === Phase 7: Try even larger length ===
    n2 = min(n_best + 1000, 4000)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Scale to n={n2} ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30 and n2 > n_best:
        a_long2 = interpolate_to_length(best_seq, n2)
        a_in = project_simplex(a_long2)
        a_out, _ = nesterov_fw(a_in, t7 * 0.7, top_k=5, power=2.0, verbose=True)
        try_update(a_out, f"fw_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = project_simplex(best_seq)
            a_out, _ = coordinate_polish(a_in, remaining * 0.8, verbose=False)
            try_update(a_out, f"coord_n{len(best_seq)}_p7")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord n={n_best} ===")
    t8 = min(remaining * 0.3, 100.0)
    if t8 > 10:
        a_in = project_simplex(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Alternative initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Alt init n={n_best} ===")
        x = np.linspace(0, 1, n_best)
        for alpha_val in [0.5, 1.0, 2.0, 3.0]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            alt = (1.0 - (2.0 * x - 1.0) ** 2) ** alpha_val
            alt = np.maximum(alt, 0.0)
            s = alt.sum()
            if s < 1e-15:
                continue
            alt = alt / s
            t9 = min(remaining * 0.3, 60.0)
            a_out, _ = nesterov_fw(alt, t9, top_k=5, power=2.0, verbose=False)
            try_update(a_out, f"alt_{alpha_val:.1f}")

    # === Phase 10: Final FW with best params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 10: Final FW n={n_best} ===")
        t10 = min(remaining * 0.7, 300.0)
        a_in = project_simplex(best_seq)
        a_out, _ = nesterov_fw(a_in, t10, top_k=5, power=2.0, verbose=True)
        try_update(a_out, f"fw_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n = 1600
        x = np.linspace(0, 1, n)
        fallback = np.sin(np.pi * x)
        fallback = np.maximum(fallback, 0.0)
        fallback = fallback / fallback.sum()
        return [float(v) for v in fallback]

    return [float(v) for v in best_seq]
