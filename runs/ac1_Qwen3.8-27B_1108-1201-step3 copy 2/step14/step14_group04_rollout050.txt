import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (default): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def build_conv_matrix(a):
    """Build T matrix where (T @ b) = conv(a, b) for b of same length as a."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        T[k, lo:hi + 1] = a[k - lo:k - hi + 1][::-1]
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = build_conv_matrix(a)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP-Nesterov with tail averaging and restart logic."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    prev_y = a.copy()
    avg_a = np.zeros(n)
    tail_sum = np.zeros(n)
    tail_count = 0
    n_tail_window = 50
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    stall_threshold = 200
    best_stall_score = best_score

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        t = n_iters
        alpha = 2.0 / (t + 2.0)
        prev_y = y.copy()
        y = (1.0 - alpha) * y + alpha * g
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = prev_y.copy()
        else:
            y = y * norm / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y + momentum * (y - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s_a

        # Track scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_stall_score = best_score
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a_new).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            best_stall_score = best_score
            stall_count = 0

        # Running average
        avg_a += a_new
        avg_n = n_iters
        if avg_n % 10 == 0:
            avg_check = avg_a / avg_n
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_autocorr(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    best_stall_score = best_score
                    stall_count = 0

        # Tail average
        tail_sum += a_new
        tail_count += 1
        if tail_count > n_tail_window:
            tail_sum -= prev_y  # approximate removal of oldest (not perfect but helps)
        if tail_count % 30 == 0 and tail_count >= n_tail_window:
            tail_avg = tail_sum / tail_count
            s_t = tail_avg.sum()
            if s_t > 1e-15:
                tail_avg = tail_avg * norm / s_t
                sc_t = fast_autocorr(tail_avg).max()
                if sc_t < best_score:
                    best_score = sc_t
                    best_a = tail_avg.copy()
                    best_stall_score = best_score
                    stall_count = 0

        # Restart if stalled for too long
        if stall_count > stall_threshold:
            # Perturb and restart
            rng = np.random.default_rng(n_iters)
            perturbation = rng.random(n) * 0.02
            y_new = y * (1.0 + (perturbation - perturbation.mean()))
            y_new = np.maximum(y_new, 0.0)
            s_yn = y_new.sum()
            if s_yn > 1e-15:
                y = y_new * norm / s_yn
            prev_y = y.copy()
            avg_a = np.zeros(n)
            tail_sum = np.zeros(n)
            tail_count = 0
            n_iters = 1
            stall_count = 0
            stall_threshold = max(100, stall_threshold * 0.7)

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def soft_min_polish(a0, time_limit, beta_start=50.0, beta_end=500.0, verbose=True):
    """Polish using gradient descent on softmax approximation of max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_total = max(1, int(time_limit * 30))  # estimate iterations
    lr_base = 0.5
    stall = 0

    fft_size = 1 << (4 * n - 2).bit_length()

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        frac = it / max(1, n_total)
        beta = beta_start + (beta_end - beta_start) * frac

        # Compute autocorrelation
        ac = fast_autocorr(a)

        # Softmax weights
        ac_shifted = (ac - ac.max()) / beta
        ac_shifted = np.clip(ac_shifted, -50.0, 50.0)
        exp_ac = np.exp(ac_shifted)
        w = exp_ac / exp_ac.sum()

        # Compute gradient via FFT
        # grad[j] = 2 * sum_k w[k] * a[k-j]
        # Use: flip w, convolve with a, extract appropriate slice
        fw = w[::-1]
        # conv(fw, a) has length (2n-1) + n - 1 = 3n-2
        # grad[j] = 2 * conv(fw, a)[2n-2-j]
        A_fft = rfft(a, fft_size)
        F_fft = rfft(fw, fft_size)
        conv_full = irfft(A_fft * F_fft, fft_size)
        # conv_full[m] = sum_k fw[k] * a[m-k] for valid indices
        # We need conv_full[2n-2-j] for j=0..n-1
        grad = 2.0 * conv_full[2 * n - 2: 2 * n - 2 + n][::-1]

        # Projected gradient step
        lr = lr_base / (1.0 + 0.01 * it)
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try * norm / s_try

        sc_try = fast_autocorr(a_try).max()
        if sc_try < best_score - 1e-12:
            a = a_try
            best_score = sc_try
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller step
            lr2 = lr * 0.3
            a_try2 = a - lr2 * grad
            a_try2 = np.maximum(a_try2, 0.0)
            s_try2 = a_try2.sum()
            if s_try2 > 1e-15:
                a_try2 = a_try2 * norm / s_try2
                sc_try2 = fast_autocorr(a_try2).max()
                if sc_try2 < best_score - 1e-12:
                    a = a_try2
                    best_score = sc_try2
                    best_a = a.copy()
                    stall = 0

        if stall > 100:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMin iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_pair_swap(a0, time_limit, seed=42, max_rounds=5000):
    """Local optimization via pair swaps and small perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    # Precompute autocorrelation
    ac = fast_autocorr(a)
    peak_k = np.argmax(ac)
    peak_val = ac[peak_k]

    while time.monotonic() - t0 < time_limit and stall < 200:
        improved = False
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:4]

        for k in top_k:
            # For peak at lag k, contribution is sum_i a[i]*a[k-i]
            # To reduce it, move mass from high-contribution pairs to low-contribution positions
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find a good destination: positions with low autocorrelation contribution
            # Use the gradient: positions where reducing a[j] helps most
            # Approximate: positions far from the peak-contributing indices
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                lo2 = max(0, k2 - n + 1)
                hi2 = min(n - 1, k2)
                peak_mask[lo2:hi2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            # Pick destination from low-value positions among candidates
            dst_pool = candidates[np.argsort(a[candidates])][:max(5, len(candidates) // 10)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    ac = fast_autocorr(a)
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            ac = fast_autocorr(a)
                            stall = 0
            stall += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization combining LP, soft-min, and local methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a / s
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_start = len(best_seq)

    # === Phase 1: LP-Nesterov at starting length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP-Nesterov n={n_start} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_start) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_start}")

    # === Phase 2: Soft-min polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Soft-min polish n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = soft_min_polish(a_norm, t2, beta_start=30.0, beta_end=300.0, verbose=True)
        try_update(a_out, f"softmin_n{n_best}")

    # === Phase 3: Local pair swap ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Local search n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_pair_swap(a_norm, t3, seed=12345)
        try_update(a_out, f"local_n{n_best}")

    # === Phase 4: LP polish after local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_best + 300
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 60:
        print(f"\n=== Phase 5: LP at n={n2} (budget={t5:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp3_n{n2}")

    # === Phase 6: Soft-min on longer if it helped ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Soft-min n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = soft_min_polish(a_norm, t6, beta_start=50.0, beta_end=500.0, verbose=True)
        try_update(a_out, f"softmin2_n{n_best}")

    # === Phase 7: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.08, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final local search n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_pair_swap(a_norm, t7, seed=99999)
        try_update(a_out, f"local2_n{n_best}")

    # === Phase 8: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t8 = min(remaining * 0.6, 150.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t8, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
