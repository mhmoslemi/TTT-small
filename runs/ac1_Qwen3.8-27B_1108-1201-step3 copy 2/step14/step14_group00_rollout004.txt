def normalize(a, target_sum):

    s = a.sum()

        return np.ones_like(a) * target_sum / len(a)

def get_active_rows(a, active_set):

def lp_best_response_lazy(a, M, solver, active_set, max_lazy_rounds=5):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    for round_idx in range(max_lazy_rounds):
        T = get_active_rows(a, active_set)
        constraints = [T @ b_var <= M, cp.sum(b_var) <= 20.0 * np.sqrt(2.0 * n)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None, active_set

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, active_set

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            return None, active_set

        # Check non-active constraints for violations
        if len(active_set) < size:
            non_active = np.setdiff1d(np.arange(size), active_set)
            T_non = get_active_rows(a, non_active)
            vals = T_non @ b_val
            violated = np.where(vals > M * (1 + 1e-5))[0]
            if len(violated) > 0:
                # Add violated constraints to active set
                new_active = np.union1d(active_set, non_active[violated])
                active_set = new_active
                continue
            else:
                break
        else:
            break

    # Update active set: keep constraints that are near-tight
    T = get_active_rows(a, active_set)
    margin = M * 0.02
    tight_mask = (T @ b_val) > (M - margin)
    new_active = active_set[tight_mask]
    if len(new_active) < 5:
        # Keep at least the top 10 tightest
        vals = T @ b_val
        top_idx = np.argsort(vals)[::-1][:min(10, len(active_set))]
        new_active = active_set[top_idx]
    b_val, new_active

    return b_val, new_active


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0, max_iters=100000):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Nesterov state
    x = a.copy()  # current iterate
    x_prev = a.copy()  # previous iterate
    y = a.copy()  # extrapolated point

    t0 = time.monotonic()
    active_set = np.arange(2 * n - 1)  # Start with all constraints
    stall = 0
    n_solves = 0

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t == 1:
            y = x.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.99)
            y = x + momentum * (x - x_prev)
            y = normalize(y, norm)

        M = fast_autocorr(y).max()
        if M < 1e-12:
            break

        # Solve LP at y
        b, active_set = lp_best_response_lazy(y, M, solver, active_set)
        if b is None:
            # Try with full active set
            active_set = np.arange(2 * n - 1)
            b, active_set = lp_best_response_lazy(y, M, solver, active_set)
            if b is None:
                continue

        n_solves += 1
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Update
        x_prev = x.copy()
        x = g

        score = fast_autocorr(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Ergodic average check
        if n_solves % 20 == 0:
            # We could track average, but for simplicity just check x
            pass

        # Restart if stalled
        if stall > 80:
            x = best_a.copy()
            x_prev = best_a.copy()
            y = best_a.copy()
            active_set = np.arange(2 * n - 1)
            stall = 0
            t = 1

        if verbose and n_solves % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP solve={n_solves} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def smooth_grad_projected(a0, time_limit, verbose=True, seed=0):
    """Smoothed projected gradient using log-sum-exp approximation of max."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # FFT setup for gradient computation
    fft_size = 1 << (3 * n - 2).bit_length()
    a_flip = a[::-1]

    t0 = time.monotonic()
    p = 5.0  # Smoothing parameter (temperature inverse)
    lr = 0.1  # Initial learning rate
    n_iters = 0
    no_improve = 0

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Increase p over time (sharper approximation of max)
        p = 5.0 + 2.0 * np.sqrt(it)
        p = min(p, 500.0)

        conv = fast_autocorr(a)
        c_max = conv.max()

        # Softmax weights
        exp_p = np.exp(p * (conv - c_max))
        w = exp_p / exp_p.sum()

        # Compute gradient via FFT
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        AF = rfft(a_flip, fft_size)

        wa = irfft(W * A, fft_size)
        wa_flip = irfft(W * AF, fft_size)

        # grad_i = sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
        # = (w * a_flip)_{n-1+i} + (w * a)_i
        u = wa_flip[n - 1:n - 1 + n]
        v = wa[:n]
        grad = u + v

        # Backtracking line search
        accept = False
        for ls in range(10):
            step = lr / (2.0 ** ls)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / sn

            conv_new = fast_autocorr(a_new)
            score_new = conv_new.max()

            if score_new < c_max - 1e-12 * c_max:
                a = a_new
                accept = True
                break
            elif ls == 0:
                pass  # Try smaller step

        if not accept:
            lr *= 0.5
            if lr < 1e-10:
                lr = 0.1
                p = max(5.0, p * 0.5)

        n_iters += 1
        score = conv_new.max() if accept else c_max
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            break

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGP it={n_iters} p={p:.1f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, seed=0):
    """Single-coordinate descent with exact line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            if a[i] < 1e-10:
                continue

            # Compute gradient direction for coordinate i
            # If we change a[i] by d (and adjust sum by spreading to others),
            # the change in conv is: d * (g_i) + d^2 * (indicator at 2i)
            # Simple approach: try moving mass from i to a random j
            j = int(rng.integers(n))
            if j == i:
                continue
            if a[j] + a[i] < 1e-15:
                continue

            # Direction: move delta from i to j
            # conv_new[k] = conv[k] + delta*(g[k]) + delta^2*h[k]
            # where g[k] = a[k-j] - a[k-i] + a[j-k] - a[i-k]
            # and h[k] = [k==2j] - [k==2i]

            # Compute g vector efficiently
            # g[k] = a[k-j] - a[k-i] + a[j-k] - a[i-k]
            # = (a shifted to j) - (a shifted to i) + (a shifted to j, reversed) - (a shifted to i, reversed)

            # For efficiency, just evaluate at strategic points
            # Find the peak and try to reduce it
            k_star = int(np.argmax(conv))

            # Compute g at k_star
            g_kstar = 0.0
            ki = k_star - i
            if 0 <= ki < n:
                g_kstar -= a[ki]
            ik = i - k_star
            if 0 <= ik < n:
                g_kstar -= a[ik]
            kj = k_star - j
            if 0 <= kj < n:
                g_kstar += a[kj]
            ij = j - k_star
            if 0 <= ij < n:
                g_kstar += a[ij]

            if g_kstar >= 0:
                # Moving mass from i to j would increase the peak; skip
                continue

            # Estimate optimal delta (first order)
            # Want conv[k_star] + delta * g_kstar to be minimal
            # But delta <= a[i]
            delta_max = a[i]
            # First-order: delta_opt = -conv[k_star] / g_kstar (but capped)
            # Better: try to bring conv[k_star] down to the median level
            target = np.median(conv)
            delta_est = (conv[k_star] - target) / (-g_kstar) if -g_kstar > 1e-15 else 0
            delta_est = min(delta_est, delta_max * 0.5)
            delta_est = max(delta_est, delta_max * 0.01)

            # Try a few candidate deltas
            best_d = 0.0
            best_val = conv.max()
            candidates = [delta_est * 0.25, delta_est * 0.5, delta_est * 0.75, delta_est, delta_max * 0.1]
            for d in candidates:
                if d <= 0 or d > a[i]:
                    continue
                # Evaluate new max (approximate: only check near k_star)
                # Full check is expensive, so check a window around k_star
                new_conv_approx = conv.copy()
                # Update all positions affected by changing a[i] and a[j]
                # This is O(n) per candidate
                for k in range(size):
                    delta_g = 0.0
                    ki2 = k - i
                    if 0 <= ki2 < n:
                        delta_g -= a[ki2]
                    ik2 = i - k
                    if 0 <= ik2 < n:
                        delta_g -= a[ik2]
                    kj2 = k - j
                    if 0 <= kj2 < n:
                        delta_g += a[kj2]
                    ij2 = j - k
                    if 0 <= ij2 < n:
                        delta_g += a[ij2]
                    h = 0.0
                    if k == 2 * j:
                        h += 1.0
                    if k == 2 * i:
                        h -= 1.0
                    new_conv_approx[k] = conv[k] + d * delta_g + d * d * h

                new_max = new_conv_approx.max()
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_d = d

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                sn = a.sum()
                if sn < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a = a * norm / sn
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
