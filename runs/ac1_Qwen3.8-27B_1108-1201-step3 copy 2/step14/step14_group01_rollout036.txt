            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a, target_sum=None):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    if target_sum is not None:
        return a * target_sum / s
    return a / s


def build_T_masked(a, active_indices):
    """Build constraint matrix for active indices only."""
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if hi > lo:
            T[row, lo:hi] = a[hi - 1 - np.arange(lo, hi)]
            # T[row, i] = a[k - i] for valid i
            for i in range(lo, hi):
                T[row, i] = a[k - i]
    return T


def lp_solve(a, M, solver, active=None, tol=1e-7):
    """Solve the LP: max sum(b) s.t. conv(a,b)[k] <= M for active k, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active is not None and 0 < len(active) < size:
        T = build_T_masked(a, active)
        constraints = [T @ b_var <= M]
    else:
        T = np.zeros((size, n))
        for m in range(n):
            T[m:m + n, m] = a
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lp_solve_full(a, M, solver):
    """Solve full LP without screening."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M, b_var >= 0])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(80.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, int(remaining * 4))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            aw = irfft(A * W, fft_size)
            grad = 2.0 * (ca[:n] + aw[:n])

            s_cur = a.sum()
            lse_val = max_c + T_val * np.log(exp_c.sum())
            obj_grad = 2.0 * n * (grad * s_cur - 2.0 * lse_val / s_cur) / (s_cur * s_cur)

            gn = np.linalg.norm(obj_grad)
            if gn < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))

            for trial in range(8):
                a_new = a - step * obj_grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.02 * T_val:
                    break
                step *= 0.5

            a_new = a - step * obj_grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, adaptive=True):
    """Improved LP-Nesterov with adaptive momentum and periodic refresh."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    a_prev = a.copy()
    momentum = 0.0
    base_momentum = 0.9
    no_improve = 0
    full_lp_interval = max(20, n // 50)
    screen_threshold = 0.80

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t % full_lp_interval == 0:
            active = None
        else:
            threshold = M * screen_threshold
            active = np.where(conv_y > threshold)[0]
            if len(active) < max(20, n // 10):
                threshold = M * 0.5
                active = np.where(conv_y > threshold)[0]

        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1 + 1e-3))[0]
            if len(violated) > 0:
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Adaptive Nesterov update
        if adaptive:
            alpha = 2.0 / (t + 2.0)
            # Adaptive momentum: increase when improving, decrease when stalling
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            momentum = (t - 1.0) / (t + 2.0)
            # Cap momentum to prevent overshooting
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        else:
            a_new = g.copy()

        # Periodic "snap" to LP solution
        if t % (full_lp_interval * 5) == 0:
            a_new = g.copy()
            y = g.copy()

        y = y_new
        a = a_new

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart mechanism
        if no_improve > 200:
            # Perturb and restart
            noise = 0.05 * norm * np.random.default_rng(t).standard_normal(n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert * norm / s_p
                sc_p = fast_conv_self(a_pert).max()
                if sc_p < best_score:
                    a = a_pert
                    y = a_pert.copy()
                    no_improve = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def cd_polish(a0, time_limit, rng, verbose=True):
    """Coordinate descent polish: transfer mass between pairs to reduce peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score_local(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_threshold = cur_max * 0.9
        peak_lags = np.where(conv >= peak_threshold)[0]

        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Find best j to transfer to
            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            # Candidate j's: random + structured
            j_candidates = list(rng.integers(n, size=6))
            for kl in peak_lags[:3]:
                j_struct = (kl - i) % n
                if j_struct != i:
                    j_candidates.append(j_struct)
            j_candidates = list(set(j_candidates))

            for j in j_candidates:
                if j == i or a[j] >= ai * 0.5:
                    continue
                aj = a[j]

                # Gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Optimize delta
                n_samples = 20
                deltas = np.linspace(0.01 * ai, 0.99 * ai, n_samples)
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val - 1e-12:
                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 8:
                break
            if no_improve_passes >= 3:
                n_pert = max(5, n // 80)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_p = a.sum()
                if s_p > 1e-15:
                    a = a * norm / s_p
        else:
            no_improve_passes = 0

        if verbose and idx % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_score_local(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)


def structured_init(n, rng):
    """Try various structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = (1.0 - (2.0 * x - 1.0) ** 2) ** 1.5
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = 0.7 + 0.3 * np.cos(2 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.3
    elif choice == 4:
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
    else:
        a = 0.5 + 0.4 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses multi-phase optimization:
    1. LSE-GD for fast initial improvement
    2. LP-Nesterov (main workhorse) with adaptive acceleration
    3. Multi-length exploration (extend sequence)
    4. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LSE GD (fast warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 2: LP-Nesterov main (workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 50:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 3: CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: CD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = cd_polish(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"cd_n{len(best_seq)}")

    # Phase 4: LP-Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP-Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 5: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t5 = min(remaining * 0.30, 250.0)
    print(f"\n=== Phase 5: Longer n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 60 and remaining > 150:
        init_long = interpolate_seq(best_seq, n_long)
        t5a = min(t5 * 0.6, 150.0)
        a_out, _ = nesterov_lp_optimize(init_long, t5a, solver, verbose=True)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 80.0)
            a_out, _ = cd_polish(best_seq, t5b, rng, verbose=False)
            try_update(a_out, f"cd_long_{n_long}")

    # Phase 6: Try even longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = n_base + 800
    t6 = min(remaining * 0.40, 200.0)
    print(f"\n=== Phase 6: Longer2 n={n_long2} (budget={t6:.0f}s) ===")
    if t6 > 60 and remaining > 120:
        init_long2 = interpolate_seq(best_seq, n_long2)
        a_out, _ = nesterov_lp_optimize(init_long2, min(t6, 150.0), solver, verbose=True)
        try_update(a_out, f"lp_long2_{n_long2}")

    # Phase 7: Final polish on the best found
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.4, 80.0)
        a_out, _ = lse_gradient_descent(best_seq, t7a, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 100.0)
            a_out, _ = cd_polish(best_seq, t7b, rng, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
