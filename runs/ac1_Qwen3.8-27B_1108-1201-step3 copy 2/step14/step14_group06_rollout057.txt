def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fsz = 1 << (size - 1).bit_length()
    A = rfft(a, fsz)
    return irfft(A * A, fsz)[:size]

def build_T(y, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        s = i
        e = min(i + n, size)
        T[s:e, i] = y[e - i - 1 : s - i - 1 : -1] if e > s else np.array([])
    return T


def build_T_correct(y, n):
    """Build T such that (T @ b)[k] = sum_i y[k-i]*b[i] = (y*b)_k."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        # b[i] contributes to positions k where 0 <= k-i < n, i.e., i <= k < i+n
        # (y*b)_k = sum_i y[k-i] * b[i], valid when 0 <= k-i < n
        # For fixed i: k ranges from i to i+n-1
        # y[k-i] = y[0]..y[n-1] as k goes i..i+n-1
        start = i
        end = min(i + n, size)
        if end > start:
            T[start:end, i] = y[:end - start]
    return T


def lp_best_response(y, M, solver, T=None):
    n = len(y)
    if T is None:
        T = build_T_correct(y, n)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.asarray(b_var.value, dtype=np.float64)
    if bv is None:
        return None
    bv = np.maximum(bv, 0.0)
    if bv.sum() < 1e-15:
        return None
    return bv


def lse_obj_grad(x, n, T_val):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shift = (b - max_b) / T_val
    exps = np.exp(shift)
    lse = max_b + np.log(exps.sum()) / T_val
    w = exps / exps.sum()
    fsz = 1 << (2 * n - 1).bit_length()
    wf = np.zeros(fsz)
    wf[:2 * n - 1] = w
    af = np.zeros(fsz)
    af[:n] = a
    cross = irfft(rfft(wf, fsz) * np.conj(rfft(af, fsz)), fsz)[:n]
    obj = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj * a / s
    return obj, grad


def lse_refine(a0, t_limit, T_lo=40.0, T_hi=4.0, stages=4):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    best_a = a.copy()
    best_sc = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_lo, T_hi, stages)
    for T in temps:
        if time.monotonic() - t0 > t_limit:
            break
        rem = t_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        sf, gf = lse_obj_grad(a.copy(), n, T)
        try:
            res = sp_minimize(sf, a.copy(), jac=gf, method='L-BFGS-B',
                              options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-7})
            if res.x is not None:
                cand = np.maximum(res.x, 0.0)
                cs = cand.sum()
                if cs > 1e-10:
                    cand = cand * norm / cs
                    sc = fast_autocorr(cand).max()
                    if sc < best_sc:
                        best_sc = sc
                        best_a = cand.copy()
                        a = cand.copy()
                        print(f"    LSE T={T:.1f}: {sc:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} err: {e}")
    return best_a

def coord_descent(a0, t_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    best_a = a.copy()
    best_sc = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < t_limit:
        b = fast_autocorr(a)
        # Find top peaks and their contributing indices
        top_peaks = np.argsort(b)[::-1][:20]
        peak_set = set()
        for k in top_peaks[:10]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                if a[i] > best_sc * 1e-6 / max(a[k - i], 1e-12):
                    peak_set.add(i)
        # Elements NOT in peak set and with small values
        non_peak = [i for i in range(n) if i not in peak_set]
        non_peak.sort(key=lambda i: a[i])
        low_targets = non_peak[:20]
        # Elements in peak set with large values (sources)
        peak_list = sorted(peak_set, key=lambda i: -a[i])
        sources = peak_list[:20]
        improved = False
        for src in sources:
            if improved:
                break
            if a[src] < 1e-10:
                continue
            for dst in low_targets:
                if dst == src or improved:
                    continue
                # Try local quadrature for optimal fraction
                best_frac = 0.1
                best_local_sc = fast_autocorr(a).max()
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
                    amt = a[src] * frac
                    if amt > a[src]:
                        amt = a[src] * 0.99
                    tr = a.copy()
                    tr[src] -= amt
                    tr[dst] += amt
                    tr = np.maximum(tr, 0.0)
                    st = tr.sum()
                    if st < 1e-10:
                        continue
                    tr = tr * norm / st
                    sc = fast_autocorr(tr).max()
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_frac = frac
                if best_local_sc < best_sc - 1e-13:
                    amt = a[src] * best_frac
                    a = a.copy()
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    a = a * norm / st
                    best_sc = fast_autocorr(a).max()
                    best_a = a.copy()
                    improved = True
        if not improved:
            # Random perturbations
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.15)
                amt = a[i] * frac
                tr = a.copy()
                tr[i] -= amt
                tr[j] += amt
                tr = np.maximum(tr, 0.0)
                st = tr.sum()
                if st < 1e-10:
                    continue
                tr = tr * norm / st
                sc = fast_autocorr(tr).max()
                if sc < best_sc - 1e-13:
                    a = tr.copy()
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    return best_a


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def attempt_update(bs, cand, label, tg):
    """Update best state. bs = (seq, score)."""
    seq, sc = bs
    if cand is None:
        return seq, sc
    a = np.maximum(np.asarray(cand, np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return seq, sc
    a = a / s
    ev = evaluate_safe(a)
    if np.isfinite(ev) and ev < sc:
        sc = ev
        seq = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-tg:.0f}s)")
    return seq, sc

            best_seq = pn.copy()
            best_score = ps
    except Exception:
        pass
    # Arithmetic progression init (the key improvement)
    for n_try in [1600, 2000]:
        a_ap = np.arange(1, n_try + 1, dtype=np.float64)
        # Scale to avoid clipping at 1000
        if a_ap.max() > 1000.0:
            a_ap = a_ap * (1000.0 / a_ap.max())
        a_ap = a_ap / a_ap.sum()
        sc = evaluate_safe(a_ap)
        print(f"AP init n={n_try}: score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_ap.copy()
            print(f"  -> Using AP as best start: {sc:.6f}")
    if best_seq is None:
        n = 1600
        x = np.linspace(0, 1, n)
        best_seq = np.sin(np.pi * x) ** 0.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default sin init: score={best_score:.6f}")
    n_best = len(best_seq)
    # Phase 1: LP Nesterov (main)
    rem = tb - (time.monotonic() - tg)
    print(f"\n=== P1: LP-Nest n={n_best} ({rem*0.45:.0f}s) ===")
    t1 = rem * 0.45
    an = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    out = nesterov_lp(an, t1, solver, verbose=True)
    best_seq, best_score = attempt_update((best_seq, best_score), out, f"lp1_n{n_best}", tg)
    # Phase 2: LSE
    rem = tb - (time.monotonic() - tg)
    n_best = len(best_seq)
    print(f"\n=== P2: LSE n={n_best} ({min(rem*0.12,150):.0f}s) ===")
    t2 = min(rem * 0.12, 150.0)
    if t2 > 30:
        an = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        out = lse_refine(an, t2)
        best_seq, best_score = attempt_update((best_seq, best_score), out, f"lse_n{n_best}", tg)
    # Phase 3: Coordinate descent
    rem = tb - (time.monotonic() - tg)
    n_best = len(best_seq)
    print(f"\n=== P3: CD n={n_best} ({min(rem*0.12,180):.0f}s) ===")
    t3 = min(rem * 0.12, 180.0)
    if t3 > 20:
        an = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        out = coord_descent(an, t3, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), out, f"cd1_n{n_best}", tg)
    # Phase 4: LP Nesterov round 2
    rem = tb - (time.monotonic() - tg)
    n_best = len(best_seq)
    print(f"\n=== P4: LP-Nest2 n={n_best} ({min(rem*0.12,150):.0f}s) ===")
    t4 = min(rem * 0.12, 150.0)
    if t4 > 30:
        an = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        out = nesterov_lp(an, t4, solver, verbose=False)
        best_seq, best_score = attempt_update((best_seq, best_score), out, f"lp2_n{n_best}", tg)
    # Phase 5: Try longer sequence
    rem = tb - (time.monotonic() - tg)
    n_long = len(best_seq) + 400
    print(f"\n=== P5: LP-Nest long n={n_long} ({min(rem*0.15,200):.0f}s) ===")
    t5 = min(rem * 0.15, 200.0)
    if t5 > 60:
        il = interp_seq(best_seq, n_long)
        an = il * np.sqrt(2.0 * n_long) / il.sum()
        out = nesterov_lp(an, t5, solver, verbose=True)
        best_seq, best_score = attempt_update((best_seq, best_score), out, f"lp_long_n{n_long}", tg)
    # Phase 6: Final polish
    rem = tb - (time.monotonic() - tg)
    n_best = len(best_seq)
    print(f"\n=== P6: Polish n={n_best} ({rem:.0f}s) ===")
    if rem > 20:
        third = rem / 3.0
        an = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        out = coord_descent(an, third, seed=99999)
        best_seq, best_score = attempt_update((best_seq, best_score), out, "cd_fin1", tg)
        rem2 = tb - (time.monotonic() - tg)
        if rem2 > 15:
            n_best2 = len(best_seq)
            an2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            out = lse_refine(an2, min(rem2, 100.0), T_lo=30.0, T_hi=2.0, stages=3)
            best_seq, best_score = attempt_update((best_seq, best_score), out, "lse_fin", tg)
        rem3 = tb - (time.monotonic() - tg)
        if rem3 > 15:
            n_best3 = len(best_seq)
            an3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            out = coord_descent(an3, rem3, seed=777)
            best_seq, best_score = attempt_update((best_seq, best_score), out, "cd_fin2", tg)
    # Final eval
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.6f}, n={len(best_seq)}, t={time.monotonic()-tg:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
