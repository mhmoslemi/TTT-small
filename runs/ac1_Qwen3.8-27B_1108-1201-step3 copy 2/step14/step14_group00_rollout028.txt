def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""

    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_rows(a, k_indices):
    """Build rows of convolution matrix T where (T@b)_k = (a*b)_k for given k."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((0, n))


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: maximize sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is None:
        active_set = np.arange(size)

    if len(active_set) == 0:
        return None, active_set

    T = build_conv_rows(a, active_set)
    if len(T) == 0:
        return None, active_set

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set

    return b_val, active_set

    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L = 1 << (2 * n - 1).bit_length()

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT and power spectrum
        A = np.fft.fft(a, L)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag * mag

        # Compute autocorrelation from power spectrum
        r = np.fft.ifft(power).real[:2*n-1]
        r = np.maximum(r, 0)

        # Identify peaks
        r_max = r.max()
        r_mean = r.sum() / len(r)
        # Target: flatten towards mean (or slightly above)
        target = r_mean * 1.05

        # Compute "excess" - how much each lag exceeds target
        excess = np.maximum(r - target, 0)

        if excess.max() < 1e-10 * r_max:
            # Well flattened
            break

        # Compute gradient of sum(excess^2) w.r.t. power spectrum
        # r_k = sum_m power[m] * exp(2*pi*i*m*k/L) (circular)
        # For linear lags: r_k for k=0..n-1 and r_{L-k} for k=n..L-1
        # Gradient w.r.t. power[m] = 2 * sum_k excess[k] * cos(2*pi*m*k/L) (for the relevant lags)

        # Use a subset of lags for efficiency
        # Focus on lags where excess is significant
        significant = np.where(excess > 0.01 * r_max)[0]
        if len(significant) == 0:
            break
        if len(significant) > 500:
            # Take top 500
            top_idx = np.argsort(excess)[-500:]
            significant = top_idx

        # Compute spectral correction
        # For each significant lag k, the contribution to power[m] is proportional to
        # excess[k] * cos(2*pi*m*k/L)
        # We want to REDUCE power at frequencies that constructively interfere at peak lags
        # and INCREASE at frequencies that destructively interfere

        m_vals = np.arange(len(significant))
        k_vals = significant
        # Build the correction matrix (sparse approximation)
        # correction[m] = sum_k excess[k] * cos(2*pi*m*k/L)
        # Use FFT-based approach: correction = IFFT(excess_padded)
        excess_padded = np.zeros(L)
        excess_padded[:len(significant)] = excess[significant]
        # Also mirror for the linear lag structure
        for k in significant:
            if k > 0 and L - k < L:
                excess_padded[L - k] = excess[excess[significant] >= 0][-1] if False else excess_padded[L-k]

        # Simpler: use the full excess vector (padded to L)
        excess_full = np.zeros(L)
        for k in range(min(len(excess), n)):
            excess_full[k] = excess[k]
            if k > 0:
                excess_full[L - k] = excess[k]

        # The spectral correction direction
        correction = np.fft.ifft(excess_full).real * L
        # Normalize correction
        corr_max = np.abs(correction).max()
        if corr_max < 1e-15:
            break
        correction = correction / corr_max

        # Adaptive step size
        eta = 0.1 * (1.0 + 0.5 * np.exp(-it / 100.0))

        # Update power spectrum (multiplicative, projected to positive)
        power_new = power * np.exp(-eta * correction)
        power_new = np.maximum(power_new, 1e-30)

        # Reconstruct A
        mag_new = np.sqrt(power_new)
        A_new = mag_new * np.exp(1j * phase)

        # Convert to time domain
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        # Symmetry blend
        a_new = 0.9 * a_new + 0.1 * a_new[::-1]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Reject and reduce step
            eta *= 0.5
            a = a  # Keep old

        if verbose and it % 200 == 0:
            print(f"    Spectral iter={it} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with exact line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Try several random partners
            for trial in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector g where new_conv = conv + delta*g (first order)
                g = np.zeros(size)
                # Remove mass from i
                for k in range(n):
                    # a[k] * a[i] contributes to conv[k+i] (if k+i < size)
                    if k + i < size:
                        g[k + i] -= a[k]
                    # a[i] * a[k] contributes to conv[k+i]
                    # (same as above for symmetric conv)
                    if i + k < size:
                        g[i + k] -= a[k]
                # Wait, this double counts. Let me redo.
                # conv[m] = sum_t a[t]*a[m-t]
                # If we change a[i] -> a[i] - d, the change in conv[m] is:
                #   -d * a[m-i] - d * a[i-m] (for valid indices)
                # If we change a[j] -> a[j] + d, the change in conv[m] is:
                #   +d * a[m-j] + d * a[j-m] (for valid indices)

                # Redo g properly
                g = np.zeros(size)
                for m in range(size):
                    # Contribution from a[i]:
                    # a[i] appears as a[t] when t=i: pair with a[m-i]
                    # a[i] appears as a[m-t] when m-t=i, i.e., t=m-i: pair with a[m-i]
                    contrib_i = 0.0
                    if 0 <= m - i < n:
                        contrib_i += a[m - i]
                    if 0 <= m - i < n:
                        contrib_i += a[m - i]  # symmetric
                    # Actually: conv[m] = sum_t a[t]*a[m-t]
                    # Derivative w.r.t. a[i]: sum over t where t=i or m-t=i
                    # t=i: a[i]*a[m-i], derivative w.r.t. a[i] is a[m-i]
                    # m-t=i => t=m-i: a[m-i]*a[i], derivative w.r.t. a[i] is a[m-i]
                    # So total: 2*a[m-i] if 0<=m-i<n, else a[i] if m-i==i (i.e., m=2i)
                    if 0 <= m - i < n:
                        if m - i == i:
                            g[m] += -a[i]  # self-pair, derivative of a[i]^2 is 2*a[i], but we move d so it's 2*(-d)*a[i]... 
                            # Actually: if we set a[i] -> a[i]-d, the term a[i]*a[i] becomes (a[i]-d)^2, change = -2*a[i]*d + d^2
                            # First order: -2*a[i]
                            g[m] = -2 * a[i] if m == 2 * i else g[m]
                        else:
                            g[m] += -2 * a[m - i]
                    # Hmm this is getting complicated. Let me just use finite differences for small n
                    # or a simpler approach.

                # Simpler approach: just try a few delta values
                aj = a[j]
                best_d = 0.0
                best_val = conv.max()
                candidates = [ai * f for f in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]]
                for d in candidates:
                    if d >= ai:
                        continue
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += d
                        a[j] -= d
                        continue
                    a_temp = a * norm / s
                    new_conv = fast_autocorr(a_temp)
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d
                        a = a_temp.copy()
                        conv = new_conv
                    else:
                        s_old = a.sum()
                        a = (a * norm / s_old).copy() if s_old > 1e-15 else a
                        # Restore
                        a[i] += d
                        a[j] -= d

                if best_d > 0:
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-12:
                        best_score = cur
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Undo
                        a[i] += best_d
                        a[j] -= best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                break  # Only try first successful partner per i

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and (it := 0) == 0:
            pass

    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, verbose=True):
    """Faster pair CD using analytical first-order approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0

    def compute_grad(a_curr, i):
        """Compute d(conv)/d(a_i) as a vector of length size."""
        g = np.zeros(size)
        for m in range(size):
            # conv[m] = sum_t a[t]*a[m-t]
            # d(conv[m])/d(a[i]) = a[m-i] (if 0<=m-i<n) + a[m-i] (if 0<=m-i<n, symmetric)
            # But we need to be careful: a[t]*a[m-t], derivative w.r.t. a[i]:
            # - when t=i: a[m-i] (if 0 <= m-i < n)
            # - when m-t=i, i.e., t=m-i: a[t] = a[m-i] (if 0 <= m-i < n)
            # If i == m-i (i.e., m=2i), these are the same term, so only count once with factor 2
            val = 0.0
            mi = m - i
            if 0 <= mi < n:
                val += a[mi]
            if 0 <= mi < n and mi != i:
                val += a[mi]
            # Wait, the convolution is conv[m] = sum_{t: 0<=t<n, 0<=m-t<n} a[t] * a[m-t]
            # d/d(a[i]) of this = sum over terms where a[i] appears
            # a[i] appears as a[t] when t=i: gives a[m-i] (if 0<=m-i<n)
            # a[i] appears as a[m-t] when m-t=i, t=m-i: gives a[m-i] (if 0<=m-i<n)
            # If t=i AND t=m-i (i.e., m=2i), it's the same term a[i]*a[i], derivative is 2*a[i]
            if 0 <= mi < n:
                if mi == i:
                    val = 2.0 * a[i]
                else:
                    val = 2.0 * a[mi]
            g[m] = val
        return g

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            for trial in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # First-order: new_conv[m] ≈ conv[m] + d * (grad_j[m] - grad_i[m])
                # where grad_i = d(conv)/d(a_i), grad_j = d(conv)/d(a_j)
                # We move d from i to j: a[i]-=d, a[j]+=d
                # Change in conv[m] ≈ d * (d(conv[m])/d(a[j]) - d(conv[m])/d(a[i]))
                # = d * (2*a[m-j] - 2*a[m-i]) (approximately, ignoring boundary effects)

                # For efficiency, compute the direction vector
                # dir[m] = 2*a[m-j] (if valid) - 2*a[m-i] (if valid)
                dir = np.zeros(size)
                for m in range(size):
                    mj = m - j
                    mi = m - i
                    if 0 <= mj < n and mj != j:
                        dir[m] += 2.0 * a[mj]
                    elif 0 <= mj < n and mj == j:
                        dir[m] += 2.0 * a[j]
                    if 0 <= mi < n and mi != i:
                        dir[m] -= 2.0 * a[mi]
                    elif 0 <= mi < n and mi == i:
                        dir[m] -= 2.0 * a[i]

                # Binary search for optimal d in [0, ai]
                # Minimize max(m) (conv[m] + d * dir[m])
                lo, hi = 0.0, ai
                best_d = 0.0
                best_val = conv.max()

                # Try a few strategic points first
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    d = frac * ai
                    new_max = (conv + d * dir).max()
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    # Refine with binary search around best_d
                    d_lo = max(0, best_d * 0.8)
                    d_hi = min(ai, best_d * 1.2)
                    for _ in range(10):
                        d_mid = (d_lo + d_hi) / 2
                        val_mid = (conv + d_mid * dir).max()
                        if val_mid < (conv + d_lo * dir).max():
                            d_lo = d_mid
                        else:
                            d_hi = d_mid
                    best_d = d_lo

                    if best_d > 1e-15:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s > 1e-15:
                            a = a * norm / s
                        conv = fast_autocorr(a)
                        cur = conv.max()
                        if cur < best_score - 1e-12:
                            best_score = cur
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            # Undo
                            a[i] += best_d
                            a[j] -= best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)
                        break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            print(f"    PCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, seed_type=0):
    """Generate diverse structured initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    choice = seed_type % 10

    if choice == 0:
        a = np.sin(np.pi * x) ** 0.7
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 2:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 0.5 + 0.2 * np.cos(6 * np.pi * x)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.4 + 0.6 * np.sin(np.pi * x)
    elif choice == 7:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) ** 0.3
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 0.7 * x) + 0.2 * np.cos(2 * np.pi * 1.3 * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
