def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_T_sparse(y):
    """Build sparse T matrix for conv(y, b) constraints.
    T[k, j] = y[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(y)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(y[k - j])
    T = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T.tocsr()

def fw_lp_step(y, M, solver, T=None):
    """One Frank-Wolfe LP step: maximize sum(b) s.t. conv(y,b) <= M, b >= 0.
    Returns normalized b (sum=1) or None on failure.
    """
    n = len(y)
    if T is None:
        T = build_T_sparse(y)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    # Actually, if sum(b) = 1, then maximizing sum(b) is trivial.
    # We want to maximize sum(b) s.t. conv(y,b) <= M, b >= 0.
    # The optimal sum(b) tells us how much we can scale up.
    # Let's reformulate: find b >= 0, conv(y,b) <= M, maximize sum(b).
    # Then normalize: g = b / sum(b).
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def nesterov_fw(a0, time_limit, solver, verbose=True, use_line_search=False):
    """Frank-Wolfe with Nesterov acceleration.
    
    Minimizes f(a) = max_k conv(a,a)[k] over the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov point
    x = a.copy()  # primal point
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build T matrix for current y
        T = build_T_sparse(y)
        
        # LP step
        b_val = fw_lp_step(y, M_curr, solver, T)
        if b_val is None:
            # Fallback: use uniform direction
            g = np.ones(n) / n
        else:
            g = b_val
        
        # Standard FW step
        alpha = 2.0 / (t + 2.0)
        
        if use_line_search:
            # Line search: try several step sizes
            best_t = alpha
            best_f = float('inf')
            for cand_t in [0.1, 0.2, 0.3, 0.5, 0.7, alpha, 1.0]:
                cand_t = min(cand_t, 1.0)
                a_try = (1 - cand_t) * y + cand_t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                f_try = fast_autocorr(a_try).max()
                if f_try < best_f:
                    best_f = f_try
                    best_t = cand_t
            alpha = best_t
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def spectral_smooth(a, time_limit, verbose=True, alpha=0.1, n_iter=None):
    """Spectral smoothing: flatten the frequency magnitude spectrum.
    
    Works by:
    1. Compute FFT of padded sequence
    2. In frequency domain, push |A|^2 towards its mean (for non-DC)
    3. Transform back, project to non-negative simplex
    """
    n = len(a)
    a_curr = norm_sum1(a)
    best_a = a_curr.copy()
    best_m = fast_autocorr(a_curr).max()
    t0 = time.monotonic()
    
    N = 1 << (2 * n - 1).bit_length()  # FFT size
    iter_count = 0
    
    # Compute initial spectrum
    a_pad = np.zeros(N)
    a_pad[:n] = a_curr
    A = np.fft.fft(a_pad)
    
    # Target: uniform |A|^2 for k >= 1
    # |A[0]| = sum(a) = 1 (normalized)
    
    for it in range(n_iter or 100000):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count += it
        
        # Compute non-DC average
        non_dc = np.abs(A[1:])**2
        mean_power = non_dc.mean() if len(non_dc) > 0 else 0
        
        if mean_power < 1e-30:
            break
        
        # Smooth: A_new[k] = A[k] * (mean_power / |A[k]|^2)^alpha for k >= 1
        # This pushes magnitudes towards sqrt(mean_power)
        A_new = A.copy()
        magnitudes = np.abs(A[1:])
        # Avoid division by zero
        safe_mag = np.maximum(magnitudes, 1e-10)
        scale = (np.sqrt(mean_power) / safe_mag) ** alpha
        A_new[1:] = A[1:] * scale
        
        # Maintain symmetry for real signal
        # For real signal: A[N-k] = conj(A[k])
        # We need to ensure this after modification
        # Actually, since we're modifying both A[k] and A[N-k] independently,
        # we should average them to maintain reality.
        for k in range(1, N // 2 + 1):
            idx_conj = N - k if N - k < N else 0
            if idx_conj != k:
                avg = (A_new[k] + np.conj(A_new[idx_conj])) / 2
                A_new[k] = avg
                A_new[idx_conj] = np.conj(avg)
        
        # Transform back
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Failed, break
            break
        a_new = a_new / s_new
        
        # Update spectrum
        a_pad[:n] = a_new
        a_pad[n:] = 0
        A = np.fft.fft(a_pad)
        
        # Update best
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, alpha={alpha}")
    
    return best_a, best_m

def spectral_smooth_v2(a, time_limit, verbose=True, alpha=0.05):
    """Spectral smoothing v2: use soft clipping instead of power scaling."""
    n = len(a)
    a_curr = norm_sum1(a)
    best_a = a_curr.copy()
    best_m = fast_autocorr(a_curr).max()
    t0 = time.monotonic()
    
    N = 1 << (2 * n - 1).bit_length()
    
    a_pad = np.zeros(N)
    a_pad[:n] = a_curr
    A = np.fft.fft(a_pad)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute non-DC power
        non_dc_power = np.abs(A[1:])**2
        mean_p = non_dc_power.mean()
        max_p = non_dc_power.max()
        
        if max_p < 1e-30:
            break
        
        # Soft clipping: reduce components above threshold
        threshold = max_p * (1 - alpha)  # slightly below max
        # Or: use median as threshold
        # median_p = np.median(non_dc_power)
        # threshold = median_p * 2
        
        magnitudes = np.abs(A[1:])
        # For components above threshold, reduce them
        # A_new[k] = A[k] * (threshold / |A[k]|^2)^beta
        beta = 0.3
        safe_mag = np.maximum(magnitudes, 1e-10)
        ratio = np.where(safe_mag > np.sqrt(threshold), 
                        (np.sqrt(threshold) / safe_mag) ** beta, 
                        1.0)
        
        A_new = A.copy()
        A_new[1:] = A[1:] * ratio
        
        # Enforce Hermitian symmetry
        for k in range(1, N // 2 + 1):
            j = N - k
            if j < N and j != k:
                avg = (A_new[k] + np.conj(A_new[j])) / 2
                A_new[k] = avg
                A_new[j] = np.conj(avg)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a_pad[:n] = a_new
        a_pad[n:] = 0
        A = np.fft.fft(a_pad)
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral2 it={it+1} ({elapsed:.1f}s): J={J:.6f}, max_p/mean={max_p/mean_p:.3f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, max_stall=200):
    """Coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        
        # Find the peak lag
        peak_lag = np.argmax(ac)
        peak_val = ac[peak_lag]
        
        # Find the pair (i, j) with i+j = peak_lag that maximizes a[i]*a[j]
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n - 1, peak_lag)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_lag - idx]
        src_i = idx[np.argmax(contrib)]
        src_j = peak_lag - src_i
        
        if contrib.max() < 1e-20:
            stall += 1
            continue
        
        # Try moving mass from the larger of (src_i, src_j) to a "quiet" position
        if a[src_i] >= a[src_j]:
            src = src_i
        else:
            src = src_j
        
        # Find quiet positions (low autocorrelation contribution)
        # Score each position by how much it would increase the peak
        # Quick heuristic: positions far from the peak lag
        # Or: positions where a is currently small
        
        # Sample candidate destinations
        n_cand = min(n, 500)
        candidates = rng.integers(0, n, size=n_cand)
        
        best_dst = -1
        best_score = float('inf')
        
        for dst in candidates:
            if dst == src:
                continue
            # Estimate the change in peak autocorrelation
            # Moving epsilon from src to dst changes:
            # conv[lag] by -epsilon * (a[lag-src] + a[src-lag]) + epsilon * (a[lag-dst] + a[dst-lag])
            delta = 0.0
            for lag in [peak_lag]:
                d1 = lag - dst
                d2 = dst - lag
                add = 0.0
                if 0 <= d1 < n:
                    add += a[d1]
                if 0 <= d2 < n:
                    add += a[d2]
                d1 = lag - src
                d2 = src - lag
                rem = 0.0
                if 0 <= d1 < n:
                    rem += a[d1]
                if 0 <= d2 < n:
                    rem += a[d2]
                delta = add - rem
            
            if delta < best_score:
                best_score = delta
                best_dst = dst
        
        if best_dst < 0:
            stall += 1
            continue
        
        # Try several step sizes
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        
        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord stall={stall} ({elapsed:.1f}s): J={J:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)  # Leave buffer
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # === Phase 1: LP Frank-Wolfe at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    print(f"\n=== Phase 1: FW n={n_current} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_fw(a_in, t1, solver, verbose=True)
        try_update(a_out, f"fw_n{n_current}")
    
    # === Phase 2: Spectral smoothing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    print(f"\n=== Phase 2: Spectral n={n_current} ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_smooth_v2(a_in, t2, verbose=True, alpha=0.05)
        try_update(a_out, f"spec_n{n_current}")
    
    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    print(f"\n=== Phase 3: Coord n={n_current} ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True)
        try_update(a_out, f"coord_n{n_current}")
    
    # === Phase 4: FW + Spectral at n=2400 ===
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: FW+Spec n={n1} ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        
        # FW first
        t4a = t4 * 0.6
        a_out, _ = nesterov_fw(a_in, t4a, solver, verbose=True)
        try_update(a_out, f"fw_n{n1}")
        
        # Then spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            t4b = min(remaining * 0.1, 80.0)
            a_out, _ = spectral_smooth_v2(a_in, t4b, verbose=True, alpha=0.03)
            try_update(a_out, f"spec_n{n1}")
    
    # === Phase 5: Coord at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    print(f"\n=== Phase 5: Coord n={n_current} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True)
        try_update(a_out, f"coord2_n{n_current}")
    
    # === Phase 6: Scale to n=3200 ===
    n2 = 3200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Scale n={n2} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_long2 = interpolate_sequence(best_seq, n2)
        a_in = norm_sum1(a_long2)
        
        # Spectral at larger length
        t6a = t6 * 0.5
        a_out, _ = spectral_smooth_v2(a_in, t6a, verbose=True, alpha=0.03)
        try_update(a_out, f"spec_n{n2}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, remaining * 0.5, verbose=True)
            try_update(a_out, f"coord_n{n2}")
    
    # === Phase 7: Final FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    print(f"\n=== Phase 7: Final FW n={n_current} ===")
    if n_current <= 2500 and remaining > 60:
        t7 = min(remaining * 0.3, 180.0)
        if t7 > 3
