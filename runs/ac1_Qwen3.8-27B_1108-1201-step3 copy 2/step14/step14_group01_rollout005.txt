    for si, p_val in enumerate(p_values):

            if w_sum < 1e-30:

            W = rfft(w, fft_size)

            aw = irfft(A * W, fft_size)

            step = base_lr / max(1.0, grad_norm / norm)

            s_new = a_new.sum()
            if s_new < 1e-15:

            a_new = a_new * norm / s_new
            sc_new = fast_conv_self(a_new).max()

        # Adaptive constraint screening
        threshold_frac = 0.80 + 0.10 * min(1.0, t / 500.0)
        active = np.where(conv_y >= M * threshold_frac)[0]
        if len(active) < 50:
            active = np.where(conv_y >= M * 0.60)[0]
        if len(active) < 20:
            active = None

        b, dual = lp_solve(y, M, solver, active, warm_start=prev_b)
        if b is None:
            b, dual = lp_solve(y, M, solver, None)
        if b is None:
            continue
        prev_b = b.copy()

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Verify dropped constraints
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = check_conv[~np.isin(np.arange(size), active)]
            if len(violated) > 0 and violated.max() > M * 1.01:
                b2, _ = lp_solve(y, M, solver, None, warm_start=b)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    best_j_cache = {}

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Find peak lags for structured candidate selection
            peak_threshold = cur_max * 0.97
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_struct = int((kl - i) % n)
                j_candidates.add(j_struct)
                j_struct2 = int((kl + i) % n)
                j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.01, ai * 0.99, n_samples)
                best_d = 0.0
                best_vj = cur_max

                # Vectorized: compute max of conv + d*g for all d at once
                # new_conv[k] = conv[k] + d * g[k] (linear in d)
                # Plus second-order at 2i and 2j
                # max over k of (conv[k] + d*g[k]) is piecewise linear
                # For each k: value = conv[k] + d*g[k], increasing in d if g[k]>0
                # The max is determined by the k with largest g[k] among those near the peak

                # Quick approach: evaluate at a few key points
                gs = g[np.where(np.abs(g) > 1e-10)[0]]
                if len(gs) == 0:
                    continue
                # The critical d is where the two "racing" lines cross
                # Approximate: try d values where g changes sign relative to conv
                for d in deltas:
                    new_max_linear = (conv + d * g).max()
                    # Second order correction
                    if 2 * i < size:
                        val_2i = conv[2*i] - 2*ai*d + d*d
                        if val_2i > new_max_linear:
                            new_max_linear = val_2i
                    if 2 * j < size:
                        val_2j = conv[2*j] + 2*aj*d + d*d
                        if val_2j > new_max_linear:
                            new_max_linear = val_2j
                    if new_max_linear < best_vj:
                        best_vj = new_max_linear
                        best_d = d

                if best_d > 0 and best_vj < best_v - 1e-10:
                    best_v = best_vj
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(3, n // 80)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD impr={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def perturb_block(a, rng, strength=0.3):
    """Zero out a random block and redistribute mass."""
    n = len(a)
    block_size = max(3, n // 30)
    start = int(rng.integers(0, n - block_size))
    a_pert = a.copy().astype(np.float64)
    mass = a_pert[start:start + block_size].sum()
    a_pert[start:start + block_size] = 0.0
    # Redistribute to random positions
    targets = rng.choice(n, size=block_size, replace=True)
    a_pert[targets] += mass / block_size
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-15:
        a_pert = a_pert * np.sqrt(2.0 * n) / s
    return a_pert

def perturb_gaussian(a, rng, strength=0.05):
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_pert = a * (1.0 + noise)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-15:
        a_pert = a_pert * np.sqrt(2.0 * n) / s
    return a_pert

def perturb_mirror(a, rng, strength=0.2):
    """Mirror a random subsequence."""
    n = len(a)
    length = max(5, n // 10)
    start = int(rng.integers(0, n - length))
    a_pert = a.copy().astype(np.float64)
    segment = a_pert[start:start + length].copy()
    a_pert[start:start + length] = (1 - strength) * segment + strength * segment[::-1]
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-15:
        a_pert = a_pert * np.sqrt(2.0 * n) / s
    return a_pert

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1680.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            norm = np.sqrt(2.0 * len(best_seq))
            best_seq = best_seq * norm / best_seq.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: p-norm gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: p-norm GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = pnorm_gradient_descent(best_seq, t1, p_start=4.0, p_end=150.0, n_stages=6, verbose=True)
        try_update(a_out, f"pnorm1_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{len(best_seq)}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd1_n{len(best_seq)}")

    # Phase 4: Multi-restart with diverse perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 4
    t_per_restart = min(remaining / (n_restarts + 1), 150.0)
    print(f"\n=== Phase 4: Multi-restart ({n_restarts} restarts, {t_per_restart:.0f}s each) ===")
    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        if ri == 0:
            pert = perturb_block(best_seq, rng, strength=0.25)
            label = "block"
        elif ri == 1:
            pert = perturb_gaussian(best_seq, rng, strength=0.03)
            label = "gauss"
        elif ri == 2:
            pert = perturb_mirror(best_seq, rng, strength=0.15)
            label = "mirror"
        else:
            # Combine: small block + small gaussian
            pert = perturb_block(best_seq, rng, strength=0.15)
            pert = perturb_gaussian(pert, rng, strength=0.02)
            label = "block+gauss"

        t_r = min(t_per_restart, remaining - 10.0)
        if t_r <= 0:
            break

        # Run LP Nesterov on perturbed point
        a_out, _ = nesterov_lp_optimize(pert, t_r * 0.7, solver, verbose=False)
        try_update(a_out, f"restart_{label}_{ri}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and a_out is not None:
            t_cd = min(remaining * 0.3, 40.0)
            a_out2, _ = pair_cd_optimize(best_seq, t_cd, rng, verbose=False)
            try_update(a_out2, f"cd_restart_{ri}")

    # Phase 5: LP Nesterov round 2 on best
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # Phase 6: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 200
    t6 = min(remaining * 0.25, 150.0)
    print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 60:
        init_alt = interpolate_seq(best_seq, n_alt)
        a_out, _ = nesterov_lp_optimize(init_alt, min(t6, 100.0), solver, verbose=False)
        # Normalize for evaluation
        if a_out is not None:
            a_eval = a_out / a_out.sum()
            try_update(a_eval, f"alt_n{n_alt}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t_final = min(remaining * 0.6, 80.0)
        a_out, _ = pair_cd_optimize(best_seq, t_final, rng, verbose=False)
        try_update(a_out, "pcd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_lse = min(remaining * 0.5, 40.0)
            a_out, _ = pnorm_gradient_descent(best_seq, t_lse, p_start=20.0, p_end=100.0, n_stages=3, verbose=False)
            try_update(a_out, "pnorm_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
