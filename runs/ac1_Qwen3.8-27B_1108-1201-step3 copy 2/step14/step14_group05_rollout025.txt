def lse_gradient_fft(a, T_temp):
    """Compute gradient of LSE-smoothed max of conv(a,a) using FFT."""
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    max_c = c.max()
    p = np.exp(T_temp * (c - max_c))
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(2*n-1)
        p[np.argmax(c)] = 1.0
    else:
        p = p / p_sum
    
    # grad_j = sum_m a_m * p_{j+m} + sum_m a_m * p_{j-m}
    # First term: cross-correlation of a with p at lag j
    # Second term: convolution of a with p at position j (for j in [0, n-1])
    
    fft_size = 1 << (3*n - 2).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    P = rfft(p.astype(np.float64), fft_size)
    
    # Cross-correlation: IFFT(A * conj(P))
    # For real signals: IFFT(A * P) gives convolution, IFFT(A * conj(P)) gives cross-corr
    # But rfft doesn't directly give conj. Let me use the full FFT.

    # Simpler: just use the full complex FFT
    A_c = np.fft.fft(a.astype(np.float64), fft_size)
