def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_conv_matrix_full(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_rows(a, k_indices):
    n = len(a)
    k_indices = np.asarray(k_indices, dtype=int)
    rows = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            for j in range(lo, hi):
                rows[idx, j] = a[k - j]
    return rows


def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None

    duals = None
    try:
        if len(constraints) > 0:
            dual_val = constraints[0].dual_value
            if dual_val is not None:
                duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass

    return b_val, active_set, duals

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    t0 = time.monotonic()

    active_set = None
    stall_count = 0
    stall_threshold = 80
    M_multiplier = 1.05  # Start with slightly relaxed M

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max() * M_multiplier
        if M_curr < 1e-12:
            break

        # Adaptive active set: use dual-based or threshold-based
        if active_set is None or t % 50 == 0:
            if t < 200:
                tol = 0.05
            else:
                tol = 0.02 + 0.01 * np.exp(-(t - 200) / 1000.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y > conv_y.max() * 0.7)[0]

        b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        # Check for violations and expand active set if needed
        if b is not None:
            full_conv = build_conv_matrix_full(y) @ b
            if np.any(full_conv > M_curr * (1 + 1e-6)):
                viol_idx = np.where(full_conv > M_curr * (1 + 1e-6))[0]
                if active_set is not None:
                    active_set = np.union1d(active_set, viol_idx)
                else:
                    active_set = viol_idx
                if len(active_set) > 0.9 * (2 * n - 1):
                    active_set = None
                b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        # Fall back to full LP periodically
        if b is None:
            b, active_set_new, duals = lp_best_response(y, M_curr, solver, None)
            active_set = None

        if b is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue

        active_set = active_set_new if active_set_new is not None else active_set
        # Update active set from duals
        if duals is not None and duals.size > 0:
            mask = duals > 1e-8
            if mask.sum() > 0 and active_set_new is not None:
                active_set = active_set_new[mask]

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.92)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Subgradient kick every 300 iterations
        if t % 300 == 0:
            conv_a = fast_autocorr(a)
            k_star = int(np.argmax(conv_a))
            grad = np.zeros(n)
            for i in range(n):

                    grad[i] = 2.0 * a[j]
            # Normalize gradient

            if g_norm > 1e-15:
                grad = grad / g_norm
            # Perturb: move away from the peak-contributing elements
            kick_scale = 0.05
            a_kicked = a - kick_scale * grad
            a_kicked = np.maximum(a_kicked, 0.0)
            s_k = a_kicked.sum()
            if s_k > 1e-15:
                a_kicked = a_kicked * norm / s_k

                    best_a = a_kicked.copy()
                    a = a_kicked
                    y = a_kicked
                # Even if not better, keep the direction for future
                # Blend slightly
                a = 0.95 * a + 0.05 * a_kicked
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                y = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Restart if stalled

            a = best_a.copy()
            active_set = None
            stall_count = 0

                elapsed = time.monotonic() - t0
                print(f"    Restart at iter {t} ({elapsed:.0f}s), M_mult={M_multiplier:.3f}")

        # Gradually tighten M multiplier
        M_multiplier = max(1.0, M_multiplier - 0.00005)

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_fw(a0, time_limit, verbose=False):
    """Frank-Wolfe with subgradient from the peak - fast and simple."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0
    stall_limit = 500

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient of c_{k*} w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += y[j]
        grad *= 2.0

        # FW step: vertex v minimizes grad @ v
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall = 0

        if stall > stall_limit:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0

        if verbose and t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector g where new_conv = conv + d*g (first order)
                g = np.zeros(size)
                # Remove from i
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    src1 = np.arange(lo1, hi1)
                    g[lo1:hi1] -= a[np.minimum(np.maximum(src1 - i, 0), n - 1)]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    src2 = n - 1 - np.arange(lo2, hi2)
                    valid = (src2 >= 0) & (src2 < n)
                    if any(valid):
                        g[lo2:hi2][valid] -= a[src2[valid]]

                # Add to j
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    src3 = np.arange(lo3, hi3)
                    g[lo3:hi3] += a[np.minimum(np.maximum(src3 - j, 0), n - 1)]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    if any(valid4):
                        g[lo4:hi4][valid4] += a[src4[valid4]]

                # Try several delta values
                candidates = [ai * 0.1, ai * 0.3, ai * 0.5, ai * 0.7, ai * 0.9, ai]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0:
                        continue
                    new_max = (conv + d * g).max()
                    # Second-order correction
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        pert = perturb_sequence(best_seq, rng, scale=0.02 + 0.02 * r)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False, seed=100 + r)
        do_try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 5: Try longer length n+400 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    print(f"\n=== Phase 5: LP Nesterov n={n2} ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 40:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, seed=7)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 6: Pair CD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pair CD n={n_best} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t6, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # === Phase 7: LP Nesterov final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP Nesterov final n={n_best} ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, seed=42)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Try even longer if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq) + 800
    if remaining > 120:
        print(f"\n=== Phase 8: LP Nesterov n={n3} ===")
        t8 = min(remaining * 0.30, 200.0)
        if t8 > 40:
            init_long2 = interpolate_sequence(best_seq, n3)
            a_norm = init_long2 * np.sqrt(2.0 * n3) / init_long2.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, seed=17)
            do_try_update(a_out, f"lp_n{n3}")

    # === Phase 9: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final Pair CD n={n_best} ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_coordinate_descent(a_norm, t9, rng, verbose=True)
            do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
