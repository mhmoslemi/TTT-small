def lp_best_response_fast(y, M, solver, max_iter=50):
    """Solve LP best response using constraint generation."""
    n = len(y)
    size = 2*n - 1
    
    # Start with just non-negativity and a few key constraints
    # (the ones most likely to be active)
    active = np.zeros(size, dtype=bool)
    # Always include the peak constraint
    active[np.argmax(y * y)] = True  # rough estimate
    # Add a few random ones
    for _ in range(20):
        active[np.random.randint(size)] = True
    
    b = None
    for it in range(max_iter):
        T = build_conv_matrix(y)
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(
            cp.Maximize(cp.sum(b_var)),
            [T[active, :] @ b_var <= M[active]]
        )
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return b
        
        if b_var.value is None:
            return b
        b = np.maximum(np.asarray(b_var.value), 0)
        
        # Check for violated constraints
        circ = T @ b
        violated = circ > M * (1 + 1e-6)
        
        if not violated.any():
            break
        
        # Add violated constraints (up to a limit)
        n_violated = violated.sum()
        if n_violated > 20:
            # Keep the most violated
            viol_idx = np.where(violated)[0]
            viol_amount = circ[viol_idx] - M
            top = viol_idx[np.argsort(viol_amount)[::-1][:20]]
            for i in top:
                active[i] = True
        else:
            active[violated] = True
    
    return b

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, indices=None):
    """Build convolution matrix rows for given indices only."""
    n = len(a)
    size = 2 * n - 1
    if indices is None:
        indices = np.arange(size)
    T = np.zeros((len(indices), n))
    for i, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[i, js:je+1] = a[k - np.arange(js, je+1)]
    return T


def cross_correlation(w, a):
    """Compute grad_j = sum_k w[k] * a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    size = len(w)
    fft_size = 1 << (n + size - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv = irfft(W * A_rev, fft_size)
    return conv[n-1:n-1+n]

def nesterov_lp(a0, time_limit, solver, verbose=True, use_cg=True, cg_iter=30, max_active_init=30):
    """Fessler-Parks accelerated LP with constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.9
    M_hist = []
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        if use_cg:
            b = lp_best_response_cg(y, M_curr, solver, max_cg_iter=cg_iter, init_active=max_active_init)
        else:
            b = lp_best_response_full(y, M_curr, solver)
        
        if b is None:
            n_iters += 1
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            if momentum > 1.0:
                momentum = 1.0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            if len(recent) > 2:
                diffs = np.abs(np.diff(recent))
                osc = diffs.mean() / max(recent[0], 1e-15)
                if osc > 0.03:
                    gamma = max(1.2, gamma - 0.1)
                elif osc < 0.005:
                    gamma = min(2.0, gamma + 0.05)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}, stall={stall}")

        # Restart
        if stall > 200:
            perturb_scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.7
            M_hist = []

    return best_a, best_score


def mwa_optimize(a0, time_limit, verbose=True, T_weight=None):
    """Multiplicative Weights Update with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 0.5 / n
    
    # Nesterov momentum
    y = a.copy()
    prev_y = a.copy()
    T_w = T_weight if T_weight is not None else max(0.01, best_score * 0.05)
    stall = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        b = fast_autocorr(y)
        max_b = b.max()
        
        if max_b < 1e-15:
            break
        
        # Compute softmax weights
        w = np.exp((b - max_b) / T_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient via cross-correlation
        grad = cross_correlation(w, y)
        
        # MWA update: a_j *= exp(-eta * grad_j)
        # Clip to avoid numerical issues
        log_update = -eta * np.maximum(grad, 0)
        log_update = np.clip(log_update, -50, 0)
        multiplicative_factor = np.exp(log_update)
        
        y_new = y * multiplicative_factor
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov step
        t = steps + 1
        momentum = (t + 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - prev_y)
        a_new = np.maximum(a_new, 0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = a_new
        
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
            stall = 0
            # Slightly increase eta on improvement
            eta = min(eta * 1.02, 5.0 / n)
        else:
            stall += 1
            # Decrease eta
            eta = max(eta * 0.95, 0.01 / n)
        
        steps += 1
        
        # Animate T_w: decrease over time
        progress = (time.monotonic() - t0) / time_limit
        T_w = max(0.001, best_score * 0.05 * (1 - progress))
        
        # Restart if stalled
        if stall > 100:
            y = np.maximum(y + rng.normal(0, 0.01 * y.mean(), n), 0)
            y = y * norm / y.sum()
            prev_y = y.copy()
            eta = 0.5 / n
            stall = 0
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWA step={steps} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.2e}, T_w={T_w:.4f}")

    return best_a, best_score


def peak_shave_multi(a0, time_limit, seed=42, verbose=True):
    """Multi-peak targeted shaving with bisection line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top 3 peaks
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:3]
        
        # Focus on the worst peak
        p_star = int(top_peaks[0])

        # Find the most contributing pair for this peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        if js > je:
            n_stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        if contrib.max() < 1e-18:
            n_stall += 1
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Find best destination: positions that don't contribute to top peaks
        # A position j contributes to peak p if js <= j <= je for that peak
        harmful = np.zeros(n, dtype=bool)
        for p in top_peaks:
            hs = max(0, p - n + 1)
            he = min(n - 1, p)
            harmful[hs:he+1] = True
        harmful[src] = True

        safe_pos = np.where(~harmful)[0]
        if len(safe_pos) == 0:
            safe_pos = np.array([i for i in range(n) if i != src])

        # Pick destination from safe positions with lowest value
        n_select = min(20, len(safe_pos))
        dst_candidates = safe_pos[np.argsort(a[safe_pos])[:n_select]]
        dst = dst_candidates[rng.integers(len(dst_candidates))]

        # Bisection line search
        lo, hi = 0.0, src_val
        best_frac = 0.0
        best_trial_score = max_b
        for _ in range(20):
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[src] = src_val - mid
            a_trial[dst] += mid
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = mid
            if sc <= max_b:
                lo = mid
            else:
                hi = mid

        if best_frac > 1e-10:
            a_trial = a.copy()
            a_trial[src] = src_val - best_frac
            a_trial[dst] += best_frac
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()

            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            n_stall += 1

        # Periodic random exploration
        if n_stall > 80:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in idx_pert:
                j = rng.integers(n)
                amt = a_trial[i] * rng.uniform(0.05, 0.3)
                a_trial[i] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.005:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
            n_stall = 0

        if n_stall > 400:
            break

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def quick_lp_iterations(a, solver, n_iters=30):
    """Run a few LP iterations quickly to assess a starting point."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for it in range(n_iters):
        M_curr = fast_autocorr(a).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response_cg(a, M_curr, solver, max_cg_iter=15, init_active=20)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        t = it + 1
        alpha = min(2.0 / (t + 1), 0.5)
        a_new = (1 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        a = a_new
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining LP Nesterov, MWA, peak shaving, and multi-length exploration.
    Strategy: Quick length sweep -> Main LP optimization -> MWA -> Peak shave -> Final polish.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = (1.0 + np.cos(np.pi * x)) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Quick length sweep (60s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_sweep = min(remaining * 0.05, 60.0)
    if t_sweep > 10:
        print(f"\n=== Phase 0: Length sweep (budget={t_sweep:.0f}s) ===")
        n_candidates = list(range(max(1200, n_base - 200), n_base + 201, 50))
        if n_base not in n_candidates:
            n_candidates.append(n_base)
        n_candidates = sorted(set(n_candidates))
        
        t_per_n = t_sweep / len(n_candidates)
        for n_try in n_candidates:
            if time.monotonic() - t_global > time_budget - 60:
                break
            init_try = interpolate_sequence(best_seq, n_try)
            a_norm = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            a_out, sc_out = quick_lp_iterations(a_norm, solver, n_iters=20)
            do_try_update(a_out, f"sweep_n{n_try}")
            print(f"    n={n_try}: score={sc_out:.6f}")

    # === Phase 1: Main LP optimization (40% of remaining) ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n_cur} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, use_cg=True, cg_iter=25, max_active_init=25)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: MWA optimization (15% of remaining) ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.12
    print(f"\n=== Phase 2: MWA n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mwa_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"mwa_n{n_cur}")

    # === Phase 3: LP round 2 (20% of remaining) ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, use_cg=True, cg_iter=20, max_active_init=20)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Peak shaving (15% of remaining) ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Peak Shave n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_multi(a_norm, t4, seed=777, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 5: Try a different length ===
    n_alt = n_cur + 50
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.12
    print(f"\n=== Phase 5: LP n={n_alt} (budget={t5:.0f}s) ===")
    if t5 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True, use_cg=True, cg_iter=20, max_active_init=20)
        do_try_update(a_out, f"lp_n{n_alt}")

    # === Phase 6: MWA on best ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: MWA n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mwa_optimize(a_norm, t6, verbose=True)
        do_try_update(a_out, f"mwa2_n{n_cur}")

    # === Phase 7: Final LP polish (with full constraints) ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = remaining * 0.10
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True, use_cg=False)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shave ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = max(remaining * 0.15, 20.0)
    print(f"\n=== Phase 8: Final Peak Shave n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_multi(a_norm, t8, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # === Phase 9: Very short LP burst ===
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = max(remaining * 0.5, 10.0)
    print(f"\n=== Phase 9: Final LP burst n={n_cur} (budget={t9:.0f}s) ===")
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t9, solver, verbose=True, use_cg=True, cg_iter=15, max_active_init=15)
        do_try_update(a_out, f"lp_burst_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
