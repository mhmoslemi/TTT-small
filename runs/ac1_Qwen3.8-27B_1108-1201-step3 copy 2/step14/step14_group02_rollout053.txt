"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with exact subgradient structure and adaptive temperature (continuation)
2. LP-based Frank-Wolfe for medium sizes
3. Coordinate descent polishing
4. Frequency-domain spectrum flattening
5. Multi-length strategy with smart interpolation

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a).flatten()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_soft_subgrad(a, ac, top_k=30, beta=6.0):
    """
    Compute soft subgradient of max_k ac[k].
    Uses softmax weighting over top-k lags, with temperature controlled by beta.
    grad[m] = 2 * sum_k w_k * a[k-m] * I(k-n+1 <= m <= k)
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weights with temperature
    beta_eff = beta
    w = np.exp(beta_eff * (top_vals - top_vals.max()))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # Compute gradient: grad[m] = 2 * sum_k w_k * a[k-m] (for valid k-m)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)  # exclusive
        if m_start >= m_end:
            continue
        m_range = np.arange(m_start, m_end)
        a_vals = a[k_int - m_range]
        grad[m_range] += 2.0 * wi * a_vals
    
    return grad

def mirror_descent_continuation(a0, time_limit, verbose=True):
    """
    Mirror descent with adaptive temperature (continuation method).
    Start with smooth approximation (small beta), gradually increase.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Continuation schedule: beta goes from 2 to 20 over the run
    # top_k goes from 10 to 50
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = 2.0 + 18.0 * progress ** 0.5
        top_k = int(10 + 40 * progress)
        
        # Step size: use 1/sqrt(t) schedule
        step = 1.0 / (np.sqrt(t) * best_m * n + 1e-10)
        
        ac = fast_autocorr(a)
        g = compute_soft_subgrad(a, ac, top_k=top_k, beta=beta)
        
        # Mirror descent step (exponential family / multiplicative weights)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        # Project: normalize to sum 1
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Track best
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        # Averaging
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = norm_sum1(avg_a)
            M_avg = fast_autocorr(avg_check).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, top_k={top_k}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Solve LP: max sum(b) s.t. T@b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = y
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = norm_sum1(y_new)
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = norm_sum1(x_new)
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lag = int(np.argmax(ac))
        
        improved = False
        
        # Get subgradient at the peak lag
        # grad[m] = 2*a[top_lag - m] for m in [top_lag-n+1, top_lag]
        m_start = max(0, top_lag - n + 1)
        m_end = min(n, top_lag + 1)
        grad = np.zeros(n)
        m_range = np.arange(m_start, m_end)
        grad[m_range] = 2.0 * a[top_lag - m_range]
        
        # Find the coordinate with largest gradient (most "expensive")
        # and move mass from there to a coordinate with smallest gradient
        if grad.max() > 1e-20:
            src = int(np.argmax(grad))
            
            # Find best destination: minimize the increase in max autocorrelation
            # The new gradient at dst would be 2*a[top_lag - dst] if we add mass there
            # We want dst where a[top_lag - dst] is small (if 0 <= top_lag-dst < n)
            
            # Score each candidate by how much it would increase the peak
            if 0 <= top_lag < n:
                # Adding to dst increases (conv)[top_lag] by 2*a[top_lag-dst]*delta
                # We want to minimize this
                if dst_val := (top_lag - np.arange(n)):
                    pass
            
            # Simpler: score by a[abs(top_lag - dst)] (approximate connectivity)
            dists = np.abs(top_lag - np.arange(n))
            # Prefer positions far from the peak's "support"
            scores = a[np.clip(dists, 0, n-1)] + 1e-10
            # Also consider not being in the support of the gradient
            in_support = np.zeros(n, dtype=bool)
            in_support[m_start:m_end] = True
            scores[in_support] += a[in_support]  # extra penalty for being in support
            
            candidates = np.argsort(scores)[:50]  # top 50 best candidates
            rng.shuffle(candidates)
            
            for dst in candidates:
                if dst == src:
                    continue
                
                # Try moving some mass
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
            
            if improved:
                stall = 0
                continue
        
        # Also try random moves
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-12:
            frac = float(rng.uniform(0.01, 0.05))
            amt = a[i] * frac
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                a[i] += amt
                a[j] -= amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        
        if not improved:
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def frequency_flatten(a0, time_limit, verbose=True):
    """
    Frequency-domain spectrum flattening (Gerchberg-Saxton style).
    Try to make |F[k]| more uniform to reduce peak autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2*n-1).bit_length()
    
    # Target: flatten the power spectrum
    # |F[k]|^2 should ideally be constant
    # We use a "compression" approach: replace |F[k]| with |F[k]|^alpha for alpha < 1
    
    alpha = 0.5  # compression factor
    alpha_min = 0.1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # FFT
        A = rfft(a, fft_size)
        
        # Compress magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        mag_compressed = mag ** alpha
        A_new = mag_compressed * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = irfft(A_new, fft_size)
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        
        # Gradually reduce alpha (sharpen)
        alpha = max(alpha_min, alpha * 0.999)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Freq t={t} ({elapsed:.1f}s): J={J:.6f}, alpha={alpha:.3f}")
    
    return best_a, best_m

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
