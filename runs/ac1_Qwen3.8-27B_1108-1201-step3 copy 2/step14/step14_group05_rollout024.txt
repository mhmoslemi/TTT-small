def lse_descent(a0, time_limit, verbose=True):

        grad -= np.mean(grad)

def interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return result / ss

        # Adaptive gamma
        if len(M_history) >= 50:
            recent = M_history[-40:]
            oscillation = np.sum(np.abs(np.diff(recent))) / (40 * max(recent[0], 1e-15))
            if oscillation > 0.03:
                gamma = max(1.0, gamma - 0.05)
            elif oscillation < 0.005:
                gamma = min(1.95, gamma + 0.02)

        # Restart if stuck
        if stall > 250:
            rng = np.random.default_rng(t * 7 + 3)
            scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, scale * 0.5, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.85
            M_history = []
            if verbose:
                print(f"    [Restart at iter {t}]")

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NP iter={t} ({elapsed:.0f}s): best_M={best_M:.8f}, M_y={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")

    return best_a, best_M


def lse_descent(a0, time_limit, verbose=True):
    """LogSumExp-smoothed gradient descent with annealing temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t_start = time.monotonic()
    steps = 0

    T_start = max(0.05, best_M * 0.08)
    T_end = max(0.0005, best_M * 0.001)

    while time.monotonic() - t_start < time_limit:
        progress = min(1.0, (time.monotonic() - t_start) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient via FFT: grad_j = sum_k w_k * dc_k/da_j
        # dc_k/da_j = 2*a_{k-j} (approx, using circular conv trick)
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_result = irfft(W * AR, fft_size)
        grad = 2.0 * conv_result[n - 1:2 * n - 1]

        # Project to maintain sum
        grad = grad - np.mean(grad)

        step = 1.0 / (T * n * 0.4)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        M_test = fast_autocorr(a).max()
        if M_test < best_M:
            best_M = M_test
            best_a = a.copy()

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.5f}, best_M={best_M:.8f}")

    return best_a, best_M


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving: move mass from peak contributors to low areas."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t_start = time.monotonic()
    n_stall = 0
    n_iter = 0

    while time.monotonic() - t_start < time_limit:
        n_iter += 1
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Contributors to peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 100:
                break
            continue

        # Try top contributors
        top_k = min(8, len(idx))
        src_candidates = idx[np.argsort(contrib)[::-1][:top_k]]

        # Non-contributing positions
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True

        found = False
        for src in src_candidates:
            if found:
                break
            nc_mask = non_contrib.copy()
            nc_mask[src] = True
            nc_idx = np.where(nc_mask)[0]
            if len(nc_idx) < 3:
                nc_idx = np.array([i for i in range(n) if i != src])

            # Pick a few low-value destinations
            n_sel = min(6, len(nc_idx))
            low_vals = np.argsort(a[nc_idx])[:n_sel * 2]
            dst_pool = nc_idx[low_vals]

            for dst in dst_pool[rng.choice(len(dst_pool), size=min(4, len(dst_pool)), replace=False)]:
                for frac in [0.08, 0.15, 0.25, 0.4]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    M_test = fast_autocorr(a_trial).max()
                    if M_test < best_M:
                        a = a_trial
                        best_M = M_test
                        best_a = a.copy()
                        found = True
                        n_stall = 0
                        break
                if found:
                    break

        if not found:
            n_stall += 1

        # Random perturbation when stuck
        if n_stall > 40:
            n_pert = max(5, n // 40)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                if a_trial[i] > 1e-15:
                    amt = a_trial[i] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                M_test = fast_autocorr(a_trial).max()
                if M_test < best_M * 1.015:
                    a = a_trial
                    if M_test < best_M:
                        best_M = M_test
                        best_a = a.copy()
                        found = True
            n_stall = 0

        if n_stall > 300:
            break

        if verbose and n_iter % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    PS iter={n_iter} ({elapsed:.0f}s): best_M={best_M:.8f}, stall={n_stall}")

    return best_a, best_M


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return result / ss


def try_update_state(best_seq, best_score, a, label, t_global):
    """Try to update best_seq/best_score with candidate a. Returns (new_seq, new_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a.copy(), sc
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: Fessler-Parks LP Nesterov with constraint generation,
    multi-length exploration, LSE smoothing, and peak shaving polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1730.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Main LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.32, 520.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({t1:.0f}s) ===")
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    # === Phase 2: Try longer length (n+400) ===
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} ({t2:.0f}s) ===")
    if t2 > 30:
        init_long = interpolate_seq(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp_n{n2}", t_global)

    # === Phase 3: LP round 2 at current best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.18, 260.0)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} ({t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Try even longer (n+800) if time permits ===
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 4: LP Nesterov n={n3} ({t4:.0f}s) ===")
    if t4 > 30:
        init_long3 = interpolate_seq(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp_n{n3}", t_global)

    # === Phase 5: LSE gradient descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 5: LSE descent n={n_cur} ({t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_descent(a_norm, t5, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lse_n{n_cur}", t_global)

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 6: PeakShave n={n_cur} ({t6:.0f}s) ===")
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t6, seed=777, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"peak_n{n_cur}", t_global)

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final LP n={n_cur} ({t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp_fin_n{n_cur}", t_global)

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.10, 80.0)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} ({t8:.0f}s) ===")
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t8, seed=31415, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"peak_fin_n{n_cur}", t_global)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
