def fast_autocorr(a):
    """Compute full linear autocorrelation of a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix T where T@b = conv(a,b)[:2n-1]."""
    n = len(a)
    size = 2 * n - 1
    # T[k,j] = a[k-j] if 0 <= k-j < n
    # Column j has nonzeros at rows j, j+1, ..., min(j+n-1, size-1)
    indptr = np.zeros(n + 1, dtype=np.int64)
    j = 0
    for j in range(n):
        length = min(n, size - j)
        indptr[j + 1] = indptr[j] + length
    
    nnz = indptr[-1]
    data = np.empty(nnz)
    # For column j, the values are a[0], a[1], ..., a[length-1]
    pos = 0
    for j in range(n):
        length = min(n, size - j)
        data[pos:pos+length] = a[:length]
        pos += length
    
    # For CSR, we need rows and columns
    # Actually let's use CSC (compressed sparse column) which is more natural here
    from scipy.sparse import csc_matrix
    # Build row indices
    rows = np.empty(nnz, dtype=np.int64)
    pos = 0
    for j in range(n):
        length = min(n, size - j)
        rows[pos:pos+length] = np.arange(j, j + length)
        pos += length
    
    T = csc_matrix((data, rows, indptr), shape=(n, size))
    # Actually, I want T to be size x n, not n x size
    # Let me just build it as dense for simplicity since n<=2000
    return None  # Use dense instead


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M (element-wise), b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build the constraint matrix (dense, but efficient for n <= 2000)
    T = np.zeros((size, n))
    for j in range(n):
        end = min(n, size - j)
        T[j:j+end, j] = a[:end]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()  # Nesterov "lookahead" point
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue
        stall = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step with line search
        alpha_base = 2.0 / (t + 2.0)
        # Try a small set of step sizes
        candidates = [alpha_base, min(1.0, alpha_base * 2), min(1.0, alpha_base * 0.5)]
        best_alpha = alpha_base
        best_obj = fast_autocorr(np.maximum((1 - alpha_base) * y + alpha_base * g, 0)).max()
        for da in candidates[1:]:
            cand = np.maximum((1 - da) * y + da * g, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand * norm / sc
            obj = fast_autocorr(cand).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = da

        alpha = best_alpha
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if t % 200 == 0:
            if time.monotonic() - last_improve > 30:
                if verbose:
                    print(f"    NLP: stall at iter={n_iters}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
                # Restart from best
                y = best_a.copy()
                a = best_a.copy()
                stall = 0
            elif verbose and t % 500 == 0:
                print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

        if stall > 100:
            break

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient: d/da[i] of lse = (1/T) * sum_k w_k * d/da[i] b_k
    # b_k = sum_j a[j] * a[k-j]
    # d/da[i] b_k = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n) [but i-k = -(k-i), so for i=k, it's a[0])
    # Actually: d/da[i] (a*a)_k = a[k-i] + a[i-k] when both indices are valid
    # But (a*a)_k = sum_j a[j]*a[k-j], so d/da[i] = a[k-i] (from j=i term) + a[i-k] (from k-j=i, j=k-i term)
    # Wait: d/da[i] sum_j a[j]*a[k-j]
    # = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
    # = a[k-i] + a[k-i] ... no
    # = a[k-i] (when j=i) + (when k-j=i, i.e., j=k-i: a[k-i])
    # So if k-i is in [0,n-1] AND i is in [0,n-1] (which it always is since i is our index):
    # d/da[i] b_k = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n) = 2*a[k-i] (if 0<=k-i<n)
    # Wait no. Let me be more careful.
    # b_k = sum_{j: valid} a[j] * a[k-j] where valid means 0<=j<n and 0<=k-j<n
    # d/da[i] b_k = sum_{j: valid} [delta_{ji} * a[k-j] + a[j] * delta_{k-j,i}]
    # = a[k-i] (if i valid and k-i valid) + a[k-i] (if k-i valid and i valid)
    # Both conditions are the same: 0<=i<n (always true) and 0<=k-i<n
    # So d/da[i] b_k = 2*a[k-i] if 0 <= k-i < n, else 0
    # BUT WAIT: when i = k-i, i.e., 2i = k, we get a[i] * a[i] and the derivative is 2*a[i]
    # which is consistent with the formula (2*a[k-i] = 2*a[i])
    # So the formula d/da[i] b_k = 2*a[k-i] (if 0<=k-i<n) is correct.
    
    # Gradient of lse w.r.t. a[i] = (1/T) * sum_k w_k * 2*a[k-i] (if 0<=k-i<n)
    # = (2/T) * sum_k w_k * a[k-i] (for valid k-i)
    # = (2/T) * (w * a_reversed)[i] ... let me think
    # sum_k w_k * a[k-i] where 0<=k-i<n means k is in [i, i+n-1]
    # = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is a cross-correlation of w and a
    
    # To compute this efficiently:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the "l" correlation of a and w
    
    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    # cross-correlation: sum_j a[j] * w[i+j]
    # This is like a full cross-correlation
    # In FFT terms: if we compute ifft(conj(A) * W), we get the cross-correlation
    # Actually: (a ★ w)[k] = sum_j a[j] * conj(w[k-j]) = ifft(conj(A) * W)
    # We want: sum_j a[j] * w[i+j] = sum_j a[j] * w[(i+j)] 
    # Let's think of it as: for fixed i, sum_j a[j] * w[i+j]
    # Let m = i+j, then j = m-i, sum_{m=i}^{i+n-1} a[m-i] * w[m]
    # = (a_reversed * w)[something] ... this is getting complicated
    # Let me just use the direct convolution formula
    # sum_j a[j] * w[i+j] = (a * w_reversed)[n-1-i] ... no
    # Actually, let's just compute it via FFT:
    # Define c[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i = 0, ..., n-1
    # This is the "cross-correlation" of a (reversed) and w, evaluated at specific points
    # c[i] = sum_j a[j] * w[i+j] = sum_j a_reversed[n-1-j] * w[i+j]
    # Let k = n-1-j, j = n-1-k:
    # c[i] = sum_k a_reversed[k] * w[i+n-1-k]
    # = (a_reversed * w)[i+n-1]  (full convolution)
    
    a_rev = a[::-1]
    c = irfft(conj(rfft(a_rev, fft_size)) * rfft(w_pad, fft_size), fft_size)
    # c[m] = sum_k a_rev[k] * w[m-k] for valid indices
    # We want c[i+n-1] for i = 0, ..., n-1
    cross = c[n-1:n-1+n]
    
    grad_conv = (2.0 / T_temp) * cross
    grad = grad_conv / (s * s) - 2.0 * obj * a / s
    return obj, grad


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=0.5, n_stages=5):
    """Optimize using LSE-smoothed objective with scipy L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        def obj_func(x):
            a_clip = np.maximum(x, 0.0)
            s = a_clip.sum()
            if s < 1e-10:
                return 1e10
            b = fast_autocorr(a_clip)
            max_b = b.max()
            if max_b < 1e-15:
                return 1e10
            exp_vals = np.exp(T * (b - max_b))
            lse = max_b + np.log(exp_vals.sum()) / T
            return 2.0 * n * lse / (s * s)

        def grad_func(x):
            a_clip = np.maximum(x, 0.0)
            s = a_clip.sum()
            if s < 1e-10:
                return np.zeros(n)
            b = fast_autocorr(a_clip)
            max_b = b.max()
            if max_b < 1e-15:
                return np.zeros(n)
            exp_vals = np.exp(T * (b - max_b))
            w = exp_vals / exp_vals.sum()
            lse = max_b + np.log(exp_vals.sum()) / T
            obj_val = 2.0 * n * lse / (s * s)
            
            # Cross-correlation
            fft_size = 1 << (3 * n - 2).bit_length()
            a_pad = np.zeros(fft_size)
            a_pad[:n] = a_clip
            w_pad = np.zeros(fft_size)
            w_pad[:2*n-1] = w
            a_rev = a_clip[::-1]
            c = irfft(conj(rfft(a_rev, fft_size)) * rfft(w_pad, fft_size), fft_size)
            cross = c[n-1:n-1+n]
            
            grad_conv = (2.0 / T) * cross
            grad = grad_conv / (s * s) - 2.0 * obj_val * a_clip / s
            return grad

        x0 = a.copy()
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'time_limit': stage_time,
                                           'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score - 1e-14:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception as e:
            print(f"    LSE stage {si} failed: {e}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Fast coordinate descent focusing on active set (top peaks)."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    n_cd_iters = 0
    while time.monotonic() - t0 < time_limit:
        if n_cd_iters % 200 == 0:
            b = fast_autocorr(a)
            top_peaks = np.argsort(b)[::-1][:10]
            # Build active set: indices that contribute to top peaks
            active = set()
            for k in top_peaks:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    active.add(j)
            active = list(active)
        
        n_cd_iters += 1
        improved = False
        
        # Try moving mass from high-contribution indices to low-peak indices
        for _ in range(50):
            # Pick a random pair with some structure
            if len(active) > 0:
                src = active[rng.integers(len(active))]
            else:
                src = rng.integers(n)
            if a[src] < 1e-12:
                continue
            # Pick destination: prefer indices NOT in active set
            not_active = np.array([i for i in range(n) if i not in active][:n//2]) if len(active) < n else np.array([])
            if len(not_active) > 0:
                dst = not_active[rng.integers(len(not_active))]
            else:
                dst = rng.integers(n)
                if dst == src:
                    continue
            
            frac = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2])
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if try_update(a_trial):
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-10.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = 1.0 + 0.2 * np.sin(6 * np.pi * x) + 0.1 * np.sin(12 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def mirror_descent_step(a, b, M, eta):
    """Mirror descent step: a_new[i] = a[i] * exp(-eta * subgrad[i]) / Z"""
    n = len(a)
    # Subgradient of max conv at a
    # Find the peak
    k_star = np.argmax(b)
    subgrad = 2.0 * np.zeros(n)
    for i in range(n):
        ki = k_star - i
        if 0 <= ki < n:
            subgrad[i] = 2.0 * a[ki]
        # Also check the other direction (for non-symmetric)
        ki2 = i - k_star
        if 0 <= ki2 < n:
            subgrad[i] += 2.0 * a[ki2]
    
    # Mirror descent with exponential weights
    a_new = a * np.exp(-eta * subgrad)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new
