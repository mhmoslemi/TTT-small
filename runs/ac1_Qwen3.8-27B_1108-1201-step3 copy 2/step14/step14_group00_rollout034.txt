def fast_conv_square(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def eval_direct(a):
    """Compute the objective directly without calling evaluate_sequence."""

    a = np.minimum(a, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_conv_square(a)
    return float(2.0 * n * conv.max() / (s * s))

def build_active_constraints(a, active_indices):
    """Build constraint matrix rows for specific lag indices.
    Row k represents: sum_j a[k-j] * b_j <= M for valid j.
    Returns (m x n) matrix.
    """
    n = len(a)
    m = len(active_indices)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            coeff = a[k - j]
            if coeff > 0:
                T[idx, j] = coeff
    return T


def lp_best_response(a, M, solver, active_set=None, max_iters=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0.
    Returns (b, active_set_used) or (None, active_set).
    """
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_active_constraints(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        active_set = np.arange(size)
        T = build_active_constraints(a, active_set)
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set
    return b_val, active_set


def check_violations(a, b, M, active_set, size):
    """Check which non-active constraints are violated by b."""
    n = len(a)
    all_idx = np.arange(size)
    non_active = np.setdiff1d(all_idx, active_set)
    if len(non_active) == 0:
        return np.array([], dtype=int)
    # Compute (a*b)_k for non-active k using FFT
    conv_ab = fast_conv_square_check(a, b)
    violations = np.where(conv_ab[non_active] > M * (1 + 1e-7))[0]
    return non_active[violations]


def fast_conv_square_check(a, b):
    """Compute convolution of a and b (not necessarily same)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    B = rfft(np.asarray(b, dtype=np.float64), N)
    return irfft(A * B, N)[:size]

    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv_square(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    active_set = None
    stall_count = 0
    stall_threshold = 80
    n_lp_solves = 0
    n_wf_steps = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_square(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break

        k_peak = int(np.argmax(conv_y))

        if hi - lo >= 2:

            mean_v = vals.mean()

        n_wf_steps += 1

def coordinate_descent(a0, time_limit, rng, verbose=True, max_passes=200):
    """Pair coordinate descent with line search."""
    n = len(a0)
    size = 2 * n - 1
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_conv_square(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    no_improve_passes = 0

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            # Try a few random partners
            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute delta*conv effect
                # Moving delta from i to j changes conv by:
                # -delta * (a shifted by i) on both sides
                # +delta * (a shifted by j) on both sides
                # For the max, we track the change at each lag k:
                # delta_conv[k] = -delta*(a[max(0,k-i):...]) ... simplified
                # Use linear approximation for candidate deltas

                best_d = 0.0
                best_val = conv.max()

                # Try geometric deltas
                for frac in [0.1, 0.25, 0.4, 0.6, 0.8, 0.95, 1.0]:
                    d = ai * frac
                    if d <= 0:
                        continue
                    # Estimate new conv using rank-2 update
                    # conv_new[k] = conv[k] - d*(a[k-i] + a[k-i-n+...]) + d*(a[k-j] + ...)
                    # Simplified: use FFT for accuracy
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial * norm / st
                    conv_trial = fast_conv_square(a_trial)
                    new_max = conv_trial.max()
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_conv_square(a)
                    if conv.max() < best_score - 1e-14:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_conv_square(a)

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break
        else:
            no_improve_passes = 0

        if verbose and p % 50 == 0:
            print(f"    CD pass={p} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.04):

    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:

    return a_pert / s

            try_update(a_out2, f"cd_pert{r}_n{n_cur2}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 4: Longer length n={n_long} ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 50:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=7)
        try_update(a_out, f"lp_n{n_long}")
        # CD on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and best_seq is not None:
            n_best = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out2, _ = coordinate_descent(a_norm2, min(60, remaining - 30), rng, verbose=False)
            try_update(a_out2, f"cd_n{n_best}")

    # === Phase 5: Try shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, len(best_seq) - 300)
    if remaining > 100:
        print(f"\n=== Phase 5: Shorter length n={n_short} ===")
        t5 = min(remaining * 0.20, 180.0)
        if t5 > 40:
            init_short = interpolate_sequence(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, seed=99)
            try_update(a_out, f"lp_n{n_short}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} ===")
    t6 = min(remaining * 0.25, 250.0)
    if t6 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=42)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} ===")
    t7 = min(remaining * 0.40, 300.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, rng, verbose=True)
        try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 8: One more LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Extra LP n={n_best} ===")
        t8 = min(remaining * 0.6, 150.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, seed=777)
        try_update(a_out, f"lp_extra_n{n_best}")
        # Final CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_norm2 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out2, _ = coordinate_descent(a_norm2, remaining - 10, rng, verbose=False)
            try_update(a_out2, f"cd_veryfinal_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0, 1.0, 1.0]

    return [float(x) for x in best_seq]
