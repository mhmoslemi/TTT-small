def gradient_descent_polish(a0, time_limit, verbose=True):

            cross = irfft(W * A, fft_size)[:n]  # This gives (w*a)_j for the "forward" part

                grad = grad / gnorm

            # Adaptive learning rate
            lr = lr_base / (1.0 + 0.05 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE beta={beta:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimized(a, time_limit, rng, window=25, verbose=True):
    """
    Optimized pair coordinate descent with vectorized delta computation.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            max_offset = min(window, n - 1)
            for offset in range(1, max_offset + 1):
                j = (i + offset) % n
                if j == i:
                    continue
                if a[i] < norm * 1e-10:
                    break

                # Compute autocorrelation change when moving d from i to j
                # delta[k] = (new_conv[k] - old_conv[k]) / d (for linear part)
                # = a[k-j] + a[j-k] - a[k-i] - a[i-k] (with bounds)
                # Self terms: at k=2i: -2a[i] + d, at k=2j: +2a[j] + d

                # Vectorized computation of delta
                delta = np.zeros(size)
                # a[k-j] for valid k: k-j in [0,n-1] => k in [j, j+n-1]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    delta[lo:hi] += a[lo - j:hi - j]
                lo = max(0, j - n + 1)
                hi = min(size, j + 1)
                if hi > lo:
                    delta[lo:hi] += a[j - (lo:hi)]

                lo = max(0, i)
                hi = min(size, i + n)
                if hi > lo:
                    delta[lo:hi] -= a[lo - i:hi - i]
                lo = max(0, i - n + 1)
                hi = min(size, i + 1)
                if hi > lo:
                    delta[lo:hi] -= a[i - (lo:hi)]

                # Find best d
                best_d = 0.0
                best_val = conv.max()
                n_samples = 10
                for d in np.linspace(0.05 * a[i], 0.95 * a[i], n_samples):
                    new_conv = conv + d * delta
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def make_diverse_init(n, rng, parent=None):
    """Generate a diverse random initialization."""
    init_type = rng.integers(0, 6)
    if init_type == 0 or (parent is None and init_type == 0):
        # Uniform
        a = np.ones(n) / n
    elif init_type == 1:
        # Random
        a = rng.random(n)
    elif init_type == 2:
        # Triangular
        x = np.arange(n) / n
        a = 1.0 - np.abs(2.0 * x - 1.0) + 0.1 * rng.random(n)
    elif init_type == 3:
        # Cosine
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) + 0.05 * rng.random(n)
    elif init_type == 4:
        # Sparse random
        a = rng.random(n)
        mask = rng.random(n) < 0.3
        a[mask] = 0
    elif init_type == 5:
        # Bipartite: high in first half, low in second
        a = np.ones(n)
        a[n // 2:] = 0.3
        a += 0.05 * rng.random(n)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = n
    return a * np.sqrt(2.0 * n) / (s / np.sqrt(2.0 * n)) * (s / np.sqrt(2.0 * n))
    # Simplify: just normalize
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * np.sqrt(2.0 * n) / n
    else:
        a = a * np.sqrt(2.0 * n) / s
    return a


def perturb_structured(a, pert_type, rng):
    """Apply a structured perturbation to a solution."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = a.copy()

    if pert_type == 0:
        # Random subset zeroing
        n_zero = n // 4
        zero_idx = rng.choice(n, n_zero, replace=False)
        a[zero_idx] = 0
    elif pert_type == 1:
        # Multiplicative noise
        noise = 0.7 + 0.6 * rng.random(n)
        a = a * noise
    elif pert_type == 2:
        # Additive noise
        a = a + 0.1 * rng.standard_normal(n) * np.mean(a)
    elif pert_type == 3:
        # Smooth then add anti-smooth (creates oscillation)
        from scipy.ndimage import uniform_filter1d
        smooth = uniform_filter1d(a, size=max(3, n // 50), mode='wrap')
        a = a + 0.3 * (a - smooth)
    elif pert_type == 4:
        # Frequency-domain perturbation
        fft_size = 1 << (2 * n - 2).bit_length()
        A = rfft(a, fft_size)
        # Reduce some high-frequency components
        mask = rng.random(fft_size // 2 + 1) < 0.2
        A[mask] *= 0.5
        a = np.real(irfft(A, fft_size)[:n])
    elif pert_type == 5:
        # Block swap
        block_size = max(1, n // 20)
        b1 = rng.integers(0, n - 2 * block_size)
        b2 = rng.integers(0, n - 2 * block_size)
        if b2 < b1:
            b1, b2 = b2, b1
        if b2 - b1 >= block_size:
            a[b1:b1 + block_size], a[b2:b2 + block_size] = a[b2:b2 + block_size].copy(), a[b1:b1 + block_size].copy()
    else:
        # Inversion (complement)
        a = a.max() - a + 1e-6

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"Starting n={n_base}, score={best_score:.6f}")

    # Phase 1: Main LP Nesterov from parent (35% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov from parent (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, "lp_parent")

    # Phase 2: Diverse initializations (25% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = remaining * 0.25
    print(f"\n=== Phase 2: Diverse inits (total budget={t2_total:.0f}s) ===")
    n_diverse = 4
    t2_each = t2_total / n_diverse
    for di in range(n_diverse):
        if time.monotonic() - t_global > time_budget - 60:
            break
        if t2_each < 15:
            break
        print(f"  Diverse init {di + 1}/{n_diverse}...")
        a_init = make_diverse_init(n_base, rng, parent=best_seq)
        a_out, _ = nesterov_lp_optimize(a_init, t2_each, solver, verbose=False)
        do_try_update(a_out, f"diverse_{di}")

    # Phase 3: Structured perturbations + re-optimize (25% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3_total = remaining * 0.25
    print(f"\n=== Phase 3: Perturb + re-optimize (total budget={t3_total:.0f}s) ===")
    if best_seq is not None and t3_total > 30:
        n_perturb = 5
        t3_each = t3_total / n_perturb
        n_best = len(best_seq)
        for pi in range(n_perturb):
            if time.monotonic() - t_global > time_budget - 40:
                break
            if t3_each < 10:
                break
            a_pert = perturb_structured(best_seq, pi, rng)
            a_norm_pert = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
            a_out, _ = nesterov_lp_optimize(a_norm_pert, t3_each, solver, verbose=False)
            do_try_update(a_out, f"perturb_{pi}")

    # Phase 4: Gradient polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: LSE gradient polish (budget={t4:.0f}s) ===")
    if t4 > 10 and best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_polish(a_norm, t4, verbose=True)
        do_try_update(a_out, "lse_polish")

    # Phase 5: Final LP Nesterov refinement (4% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.04
    print(f"\n=== Phase 5: Final LP Nesterov (budget={t5:.0f}s) ===")
    if t5 > 10 and best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, "lp_final")

    # Phase 6: Pair CD polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.6
    print(f"\n=== Phase 6: Pair CD polish (budget={t6:.0f}s) ===")
    if t6 > 10 and best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_optimized(a_norm, t6, rng, window=30, verbose=True)
        do_try_update(a_out, "pcd_final")

    # Phase 7: One more LP round if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, remaining * 0.7, solver, verbose=False)
        do_try_update(a_out, "lp_last")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
