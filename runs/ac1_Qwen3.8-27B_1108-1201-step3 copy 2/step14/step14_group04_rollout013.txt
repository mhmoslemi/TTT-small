import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    from scipy.fft import rfft, irfft
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_grad(y, T):
    from scipy.fft import rfft, irfft
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lse_fista(a0, T_schedule, time_limit, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L_init = n / 30.0
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build T matrix implicitly via constraints
    # T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # For efficiency, build as a sparse matrix or use the structure
    # Actually for n=1600, building the (2n-1) x n matrix is 3199 x 1600 = ~5M entries
    # Most are zero (each row has at most n non-zeros, but only the diagonal band)
    # Let's use cvxpy's ability to handle this

    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        if M_target < current_peak * 0.98:

        t_nest = 1
        y_restart = y.copy()  # For restart
        prev_score = fast_autocorr(y).max()
        stall_count = 0
        
        while time.monotonic() - t0 < time_limit:
            M_curr = fast_autocorr(y).max()
            # Use the tighter of M_target and current peak * 1.05
            M_eff = min(M_target, M_curr * 1.05)
            
            b = lp_best_response(y, M_eff, solver)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            g = b * norm / b_sum
            n_iters += 1
            
            alpha = 2.0 / (t_nest + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            if t_nest == 1:
                a_new = g.copy()
            else:
                momentum = (t_nest - 1.0) / (t_nest + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a

            a = a_new

                best_a = y.copy()

                best_a = a.copy()

                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

            if score_y > prev_score * 1.001:
                stall_count += 1
                if stall_count > 10:
                    # Restart
                    y = y_restart.copy()
                    t_nest = 1
                    stall_count = 0
                    break
            else:
                stall_count = max(0, stall_count - 1)
                if score_y < prev_score * 0.999:
                    y_restart = y.copy()
            prev_score = score_y
            
            if verbose and n_iters % 30 == 0:
                elapsed = time.monotonic() - t0

            # but with the current y as starting point
            if time.monotonic() - t0 > time_limit:
                break
        
        # Periodically check time and update current_peak
        current_peak = best_score

The idea of M-continuation is:
- Start with a loose constraint: T@b <= M_loose (where M_loose > current_peak)
- This gives a "wider" feasible set, so the LP optimum is more spread out
- Gradually tighten M toward the current peak

The benefit is that starting with a loose M might lead to a better exploration of the space before converging.

**Key insight**: The score is 2n * max_conv / S^2. At the optimum, by the minimax principle, all the "active" conv values should be equal to the max. The LP best response g maximizes sum(g) subject to T_f @ g <= M. At the fixed point, f = g and T_f @ f <= M with some constraints active.

The quality of the solution is determined by how many constraints are active and how balanced they are. If only one constraint is active (the peak), the solution might not be optimal. We want many constraints to be active at the same level.

**New idea: "Peak spreading" via dual**

The dual of the LP (max sum(b) s.t. T@b <= M, b >= 0) is:
min M * sum(y) s.t. T^T @ y >= 1, y >= 0

The optimal dual variables y* tell us the "weights" on each lag. If y* is concentrated on one lag, the solution is "peak-dominated." If y* is spread out, the solution is more balanced.

We can use this insight: after solving the LP, examine the dual variables. If they're too concentrated, we can add a regularization term to encourage spreading.

But modifying the LP to include dual regularization is complex. Let me think of a simpler approach.

Instead of solving the LP once with M = current_peak, solve it with multiple M values (e.g., M, 1.1*M, 1.2*M) and combine the solutions. The idea is that different M values emphasize different constraints, and combining them gives a more balanced solution.

2. LP Nesterov with adaptive restarts and M-continuation (main workhorse)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:

    current_peak_init = fast_autocorr(y).max()
    continuation_factor = 1.3
    continuation_decay = 0.97
    min_continuation = 1.0

    y_restarted = y.copy()
    best_y_at_restart = best_score
    prev_scores = []

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        M_eff = min(M_curr * continuation_factor, M_curr + 1e-10)

        b = lp_best_response(y, M_eff, solver)
        if b is None:
            # Try with slightly larger M
            b = lp_best_response(y, M_curr * 1.5, solver)
            if b is None:
                break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum
        n_iters += 1

        t = n_iters
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if n_iters == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Adaptive restart logic
        prev_scores.append(score_y)
        if len(prev_scores) > 20:
            prev_scores.pop(0)

        # Restart if no improvement in last 15 iterations
        if len(prev_scores) >= 15:
            if min(prev_scores[5:]) >= max(prev_scores[:5]) * 1.0005:
                y = best_a.copy()
                n_iters = 0
                avg_a = np.zeros(n)
                n_avg = 0
                prev_scores = []
                y_restarted = y.copy()
                continue

        # Decay continuation factor
        continuation_factor = max(continuation_factor * continuation_decay, min_continuation)

        if verbose and n_iters % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Adv iter={n_iters} cont={continuation_factor:.3f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, seed=42, verbose=True):
    """Reduce peaks of autocorrelation by targeted mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        # Find top peaks
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            # Find contributing pairs for this lag
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue

            # Source: highest contributing index
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: positions NOT contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]

            # Sort by current value (prefer low-value destinations)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            # Try a few destinations
            n_try = min(len(candidates_sorted), 5)
            dst_idx_list = candidates_sorted[rng.integers(0, len(candidates_sorted), size=n_try)]

            for dst in dst_idx_list:
                if dst == src:
                    continue
                for frac in [0.0003, 0.001, 0.003, 0.01]:
                    amt = src_val * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 1000:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            if int((time.monotonic() - t0) * 2) % 2 == 0 and stall % 100 == 0:
                print(f"    Peak Eq ({elapsed:.0f}s): best={best_score:.10f}, stall={stall}")

    return best_a, best_score


def stochastic_local_search(a0, time_limit, seed=999, verbose=True):
    """Simulated annealing-style local search on the peak."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    current_score = best_score
    t0 = time.monotonic()
    T_start = 1e-6
    T_end = 1e-10
    total_iters = 5000
    n_accepted = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.001, 0.02)
        amt = a[i] * frac
        if amt < 1e-18:
            continue

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * norm / s_t
        score = fast_autocorr(a_trial).max()

        delta = score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            current_score = score
            n_accepted += 1
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SLS ({elapsed:.0f}s): new best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def mixed_lp_subgradient(a0, time_limit, solver, verbose=True):
    """LP Nesterov mixed with direct subgradient steps on the max function."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    # Every 3rd iteration, take a subgradient step instead of LP
    use_sg_every = 3
    sg_step_size = 0.02

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        M_curr = fast_autocorr(y).max()

        if n_iters % use_sg_every == 0:
            # Subgradient step
            ac = fast_autocorr(y)
            max_val = ac.max()
            tol = 0.01 * max_val
            active = np.where(ac >= max_val - tol)[0]
            if len(active) > 10:
                active = active[np.argsort(ac[active])[::-1][:10]]

            sg = np.zeros(n)
            for k in active:
                for i in range(n):
                    j = k - i
                    if 0 <= j <= n - 1:
                        sg[i] += 2.0 * y[j]
            sg /= len(active)

            y_new = y - sg_step_size * sg
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_new
            y = y_new
        else:
            # LP step
            b = lp_best_response(y, M_curr, solver)
            if b is None:
                continue
            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g = b * norm / b_sum

            t = n_iters
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_new
            y = y_new

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        avg_a += y
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MixedLP-SG iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with adaptive LP, peak equalization, and stochastic refinement."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up (short) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP Nesterov Advanced (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.38, 600.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Advanced n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_advanced(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Peak Equalization (novel) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Peak Equalization n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_equalization(a_norm, t3, seed=42, verbose=True)
        try_update(a_out, f"peak_eq_n{n_best}")

    # === Phase 4: LP Nesterov again (post peak-eq) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP Polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_advanced(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: LSE-FISTA fine-tuning (very low T) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        print(f"\n=== Phase 5: LSE-FISTA fine n={n_best} (budget={t5:.0f}s) ===")
        T_sched = [0.1, 0.05, 0.02, 0.01]
        a_out, _ = lse_fista(best_seq, T_sched, t5, verbose=True)
        try_update(a_out, f"lse_fine_n{n_best}")

    # === Phase 6: Peak Equalization round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.10, 130.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Peak Eq round2 n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_equalization(a_norm, t6, seed=7777, verbose=True)
        try_update(a_out, f"peak_eq2_n{n_best}")

    # === Phase 7: Stochastic local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Stochastic LS n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = stochastic_local_search(a_norm, t7, seed=12345, verbose=True)
        try_update(a_out, f"sls_n{n_best}")

    # === Phase 8: Mixed LP + Subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.10, 120.0)
    if t8 > 20:
        print(f"\n=== Phase 8: Mixed LP+SG n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mixed_lp_subgradient(a_norm, t8, solver, verbose=True)
        try_update(a_out, f"mixed_n{n_best}")

    # === Phase 9: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n2 = len(best_seq) + 200
        t9 = min(remaining * 0.4, 150.0)
        if t9 > 30:
            print(f"\n=== Phase 9: LP at n={n2} (budget={t9:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = lp_nesterov_advanced(a_norm, t9, solver, verbose=True)
            try_update(a_out, f"lp_long_n{n2}")

    # === Phase 10: Final peak equalization + SLS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.6, 80.0)
        if t10 > 10:
            print(f"\n=== Phase 10: Final PE n={n_best} (budget={t10:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_equalization(a_norm, t10, seed=54321, verbose=True)
            try_update(a_out, "final_pe")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
