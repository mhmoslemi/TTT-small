def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_active_set(a, M, solver, max_active=150, refine_iters=8):
    """LP with dual-guided active set refinement."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Initial: all constraints
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    # Refine active set using dual-guided selection
    for it in range(refine_iters):
        circ = T @ b_val
        tol = 1e-8 * M + 1e-12
        active = circ >= M - tol
        n_active = int(active.sum())
        
        if n_active < 5:
            break
        
        # Select top max_active most violated
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True
        
        # Solve reduced LP
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        max_viol = circ_new.max() - M
        
        if max_viol <= 1e-6 * M:
            b_val = b_new
            break
        else:
            # Add most violated constraint
            worst = int(np.argmax(circ_new))
            active[worst] = True
            b_val = b_new
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def spectral_refine(a0, time_limit, verbose=True):
    """
    Frequency-domain refinement: minimize max autocorrelation by flattening the spectrum.
    
    Key insight: max(conv(a,a)) is related to the peak of the inverse FFT of |A|^2.
    We use gradient descent on a smoothed version of the max, computed via FFT.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Annealing schedule for temperature
    T_start = best_score * 0.05
    T_end = max(0.001, best_score * 0.002)
    
    # Momentum for gradient descent
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Softmax weights for the "max" approximation
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient computation via FFT
        # d/d a_j [sum_k w_k * conv(a,a)_k] 
        # = sum_k w_k * 2 * a_{k-j} (for valid indices)
        # This is 2 * conv(w, a)[n-1 : 2n-1] (cross-correlation)
        a_rev = a[::-1]
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1].copy()
        
        # Project gradient onto tangent space (maintain sum constraint)
        grad = grad - np.mean(grad)
        
        # Adam optimizer
        steps += 1
        lr = 0.01 / n
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad * grad
        m_hat = m_t / (1 - beta1 ** steps)
        v_hat = v_t / (1 - beta2 ** steps)
        update = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}")
        
        # Periodic restart if stuck
        if steps % 2000 == 0:
            recent_best = best_score
            if abs(max_b - best_score) < 1e-8:
                rng = np.random.default_rng(steps)
                a = np.maximum(a + rng.normal(0, 0.005 * a.mean(), n), 0)
                a = a * norm / a.sum()
                m_t = np.zeros(n)
                v_t = np.zeros(n)
        
        # Early stop if converged
        if max_b - best_score < 1e-10:
            break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def alternating_projection(a0, time_limit, solver, verbose=True):
    """
    Alternating projection:
    1. Project onto {sum(a) = norm, a >= 0}
    2. Project onto {max(conv(a,a)) <= M} via LP
    3. Decrease M gradually
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    M = best_score * 1.001  # Start slightly above current
    M_target = best_score * 0.99  # Target: slightly below current
    M_step = best_score * 0.001
    iters = 0
    
    while time.monotonic() - t0 < time_limit and M > M_target:
        # Solve LP: find b that minimizes max(conv(y,b)) ... 
        # Actually, the AP approach: given M, find the closest point to a in {conv(b,b)_k <= M for all k, sum(b)=norm, b>=0}
        # This is a QP, not an LP. Let's use LP as a relaxation.
        
        # Instead: solve max sum(b) s.t. T@b <= M, b >= 0, sum(b) = norm
        # If feasible, the solution gives a direction. If not, increase M.
        
        T = build_conv_matrix(a)  # Note: this uses current a for the matrix
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M, cp.sum(b_var) == norm]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(b_var - a)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            M += M_step
            iters += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            M += M_step
            iters += 1
            continue
        
        if b_var.value is None:
            M += M_step
            iters += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_val = b_val * norm / b_val.sum()
        
        score = fast_autocorr(b_val).max()
        if score < best_score:
            best_score = score
            best_a = b_val.copy()
            a = b_val
            M -= M_step * 0.5  # Try to tighten
        else:
            # Blend
            blend = 0.5
            a_new = (1 - blend) * a + blend * b_val
            a_new = np.maximum(a_new, 0)
            a_new = a_new * norm / a_new.sum()
            sc = fast_autocorr(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        
        iters += 1
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AP iter={iters} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_refined(a, M, solver, max_active=180, refine_iters=10):
    """LP with iterative active set refinement using violation-based selection."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    # Start with all constraints
    constraints = [T @ b_var <= M]

    try:
        prob.solve(solver=solver, verbose=False)

        return None

        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    # Iteratively refine active set
    for it in range(refine_iters):
        circ = T @ b_val
        tol = max(1e-9 * M, 1e-12)
        active = circ >= M - tol
        n_active = int(active.sum())

        if n_active < 3:
            break

        # Select top max_active most violated/tight constraints
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True
        elif n_active < max_active // 2 and it > 0:
            # Too few active: add some near-active
            near = circ >= M * 0.999
            if int(near.sum()) > n_active and int(near.sum()) <= max_active:
                active = near

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

            break
        else:

            active[worst] = True
            b_val = b_new

                a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.3f}, stall={stall}")

        if stall > 250:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = gamma_init
            M_hist = []
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def spectral_refine(a0, time_limit, verbose=True):
    """
    Frequency-domain Adam gradient descent on LogSumExp-smoothed max autocorrelation.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adam hyperparameters
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    base_lr = 0.05 / n

    # Temperature schedule
    T_start = max(best_score * 0.03, 0.1)
    T_end = max(0.01, best_score * 0.003)

    # Precompute FFT size
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Softmax weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: d/d a_j [sum_k w_k * conv(a,a)_k] = 2 * cross_corr(w, a)[j]
        a_rev = a[::-1]
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1].copy()

        # Project onto sum constraint
        grad = grad - np.mean(grad)

        # Gradient for non-negativity: zero gradient where a is near zero
        mask = a > 1e-10
        grad[~mask] = np.maximum(grad[~mask], 0)

        # Adam step with adaptive learning rate
        steps += 1
        lr = base_lr * (1.0 + 0.5 * (1.0 - progress))
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad * grad
        m_hat = m_t / (1 - beta1 ** steps)
        v_hat = v_t / (1 - beta2 ** steps)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}, max_b={max_b:.6f}")

        # Periodic perturbation if stuck
        if steps % 5000 == 0 and abs(max_b - best_score) < 1e-7:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.003 * a.mean(), n), 0)
            a = a * norm / a.sum()
            m_t = np.zeros(n)
            v_t = np.zeros(n)

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Multi-destination peak shaving with line search and structured perturbation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find all top peaks
        sorted_peaks = np.argsort(b)[::-1][:5]

        # For the main peak, find contributing pairs
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 150:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Non-contributing positions for this peak
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            non_contrib_idx = np.array([i for i in range(n) if i != src])

        # Also consider non-contributing for secondary peaks
        all_non_contrib = non_contrib.copy()
        for p2 in sorted_peaks[1:]:
            js2 = max(0, p2 - n + 1)
            je2 = min(n - 1, p2)
            all_non_contrib[js2:je2 + 1] = False
        all_nc_idx = np.where(all_non_contrib)[0]
        if len(all_nc_idx) > 0:
            non_contrib_idx = all_nc_idx

        # Try multiple (src, dst, frac) combinations
        best_trial = max_b
        best_move = None
        n_select = min(20, len(non_contrib_idx))
        sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]

        for dst in sorted_dst:
            for frac in [0.15, 0.25, 0.4, 0.6]:
                amt = src_val * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial:
                    best_trial = sc
                    best_move = (src, dst, frac)

        if best_move is not None and best_trial < max_b:
            src_i, dst_i, frac_i = best_move
            a_trial = a.copy()
            a_trial[src_i] -= a[src_i] * frac_i
            a_trial[dst_i] += a[src_i] * frac_i
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                else:
                    n_stall += 1
        else:
            n_stall += 1
            # Random exploration when stuck
            if n_stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.05, 0.3)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                        else:
                            n_stall += 1

        if n_stall > 250:
            # Large structured perturbation
            n_pert = max(10, n // 30)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                if a[idx] > 1e-10:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = int(rng.integers(n))
                    a_trial[idx] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.005:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def q_projection_refine(a0, time_limit, solver, verbose=True, max_constraints=200):
    """
    QP-based projection: minimize ||b - a||^2 subject to conv constraints.
    More direct than LP for finding the best update direction.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0

    # Start M slightly above current max, decrease over time
    M = best_score * 1.005
    M_min = best_score * 0.99
    T = build_conv_matrix(a)

    while time.monotonic() - t0 < time_limit and M > M_min:
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M, cp.sum(b_var) == norm]

        prob = cp.Problem(cp.Minimize(cp.sum_squares(b_var - a)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            M += best_score * 0.001
            iters += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            M += best_score * 0.001
            iters += 1
            continue

        if b_var.value is None:
            M += best_score * 0.001
            iters += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            M += best_score * 0.001
            iters += 1
            continue
        b_val = b_val * norm / s_b

        score = fast_autocorr(b_val).max()

        if score <= M * 1.001:
            # Feasible: try to tighten M
            if score < best_score:
                best_score = score
                best_a = b_val.copy()
            a = b_val
            M = max(M_min, score * 0.999)
        else:
            # Infeasible: relax
            M = max(M, score * 1.001)
            # Blend as a fallback
            blend = 0.3
            a_new = (1 - blend) * a + blend * b_val
            a_new = np.maximum(a_new, 0)
            a_new = a_new * norm / a_new.sum()
            sc = fast_autocorr(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        iters += 1
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP iter={iters} ({elapsed:.0f}s): M={M:.6f}, best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining LP Nesterov, spectral refinement, QP projection,
    peak shaving, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_active=150, gamma_init=1.9)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_refine(a_norm, t2, verbose=True)
        do_try_update(a_out, f"spectral_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (from improved starting point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, max_active=120, gamma_init=1.85)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: QP Projection ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: QP Projection n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = q_projection_refine(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"qp_n{n_cur}")

    # === Phase 5: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_candidates = [n_base + 50, n_base + 100, n_base + 200]
    for n_try in n_candidates:
        if remaining < 80:
            break
        t5 = min(remaining * 0.08, 80.0)
        if t5 < 20:
            break
        print(f"\n=== Phase 5: LP Nesterov n={n_try} (budget={t5:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, max_active=100, gamma_init=1.9)
        do_try_update(a_out, f"lp_n{n_try}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Spectral on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Spectral n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_refine(a_norm, t6, verbose=True)
        do_try_update(a_out, f"spectral2_n{n_cur}")

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t7, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, max_active=100, gamma_init=1.8)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.10, 80.0)
    if t9 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t9, seed=271828, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
