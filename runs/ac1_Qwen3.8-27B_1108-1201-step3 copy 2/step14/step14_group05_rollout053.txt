def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:
        a_list = [float(max(0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:

        sc = evaluate_sequence(a_list)

def build_conv_rows(a, row_indices):
    """Build rows of the convolution matrix T for specific row indices.
    T[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    rows = len(row_indices)
    T = np.zeros((rows, n))
    for i, k in enumerate(row_indices):
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        if js_start <= js_end:
            js = np.arange(js_start, js_end + 1)
            T[i, js] = a[k - js]
    return T


def solve_lp(a, M, n_active_max=150):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0 using active set LP.
    Returns optimal b or None on failure."""
    n = len(a)
    size = 2 * n - 1

    # Find active constraints: start with top n_active_max
    ab = fast_autocorr(a)
    M_val = M if M is not None else ab.max()
    top_idx = np.argsort(ab)[::-1][:n_active_max]

    # Solve with these constraints
    T = build_conv_rows(a, top_idx)
    M_active = np.ones(len(top_idx)) * M_val

    # LP: max 1'b s.t. T b <= M_active, b >= 0
    # In linprog form: min -1'b s.t. -T b >= -M_active, b >= 0
    c_obj = -np.ones(n)
    A_ub = -T
    b_ub = -M_active
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None

    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None

    # Check if solution satisfies all constraints
    circ = fast_autocorr(a) * 0 + (T @ b_val)  # only check active
    # Actually check full convolution
    full_circ = np.convolve(a, b_val)
    if full_circ.max() > M_val * (1 + 1e-6):
        # Solution violates non-active constraints, add them
        violated = np.where(full_circ > M_val * (1 + 1e-9))[0]
        if len(violated) > 0:
            all_idx = np.unique(np.concatenate([top_idx, violated[:n_active_max - len(top_idx)]]))
            all_idx = all_idx[:n_active_max]
            T2 = build_conv_rows(a, all_idx)
            M2 = np.ones(len(all_idx)) * M_val
            res2 = linprog(c_obj, A_ub=-T2, b_ub=-M2, bounds=bounds, method='highs')
            if res2.status == 0:
                b_val = np.maximum(res2.x, 0.0)
                if b_val.sum() < 1e-15:
                    return None

    return b_val


def nesterov_lp_optimize(a0, time_limit, verbose=True, n_active=150, gamma_init=1.9):
    """Accelerated LP Nesterov with parabolic extrapolation and warm starts."""
    n = len(a0)

    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    prev_prev_y = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = gamma_init

    # History for parabolic extrapolation
    y_hist = [a.copy()]

    while time.monotonic() - t_start < time_limit:
        n_iters += 1
        t = n_iters

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break

        # Solve LP
        b = solve_lp(y, M_curr, n_active_max=n_active)
        if b is None:
            if n_iters % 100 == 0 and verbose:
                print(f"    LP failed at iter {n_iters}")
            # Perturb and retry
            rng = np.random.default_rng(n_iters)
            y = np.maximum(y + rng.normal(0, 0.01 * y.mean(), n), 0)
            y = y * norm / y.sum()
            prev_y = y.copy()
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Parabolic interpolation for momentum
        if len(y_hist) >= 3:
            y1, y2, y3 = y_hist[-3], y_hist[-2], y_hist[-1]
            # Simple parabolic: y_new = y3 + (y3 - y2) + 0.1*(y2 - y1) (extrapolation)
            momentum_vec = (y3 - y2) + 0.1 * (y2 - y1)
        elif len(y_hist) >= 2:
            momentum_vec = y_hist[-1] - y_hist[-2]
        else:
            momentum_vec = np.zeros(n)

        # Nesterov-type step
        alpha = 2.0 / (t + 2.0 - gamma)
        if alpha > 0.4:
            alpha = 0.4
        if alpha < 0.01:
            alpha = 0.01

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Apply momentum (damped)
        damp = min(0.3, 0.1 * alpha * 10)
        a_new = y_new + damp * momentum_vec
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        prev_prev_y = prev_y.copy()
        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track history (limit length)
        y_hist.append(y.copy())
        if len(y_hist) > 5:
            y_hist.pop(0)

        # Update best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma

        # Actually, the cleanest way:
        # grad[m] = 2 * (w cross-corr a)[m] where cross-corr is defined as:
        # (w cross-corr a)[m] = sum_j w[j+m] * a[j] (sum over valid j)
        # This equals sum_j w[j+m] * a[j] = sum_k w[k] * a[k-m] (with k=j+m)

        # So I need: for m=0,...,n-1: sum_k w[k] * a[k-m]
        # = (the cross-correlation of w with a, at lag m)
        # = ifft(FFT(w) * conj(FFT(a))) evaluated appropriately

        # Standard: corr(w,a)[m] = sum_j w[j+m]*a[j] = ifft(conj(FFT(w)) * FFT(a))[m]
        # Wait: ifft(conj(W)*A)[m] = sum_k conj(W[k])*A[k]*exp(2pi i km/N)
        # This gives the circular cross-correlation.

        # For linear cross-correlation, I need to be careful about padding.
        # w has length 2n-1, a has length n.
        # I want corr[m] = sum_j w[j+m]*a[j] for m = 0,...,n-1
        # This is a "truncated" cross-correlation.

        # Simplest: use np.correlate or just compute via FFT with proper padding.

        # Let me use: pad both to length >= 2n-1+n-1 = 3n-2 for linear conv.
        fft_size2 = 1 << (3 * n).bit_length()
        W2 = rfft(w, fft_size2)
        A2 = rfft(a, fft_size2)
        # Linear cross-correlation: ifft(conj(W2) * A2)
        # This gives: sum_j w[j]*a[j+m] for m = 0, 1, ... (the "positive lags")
        # Actually ifft(conj(W)*A) gives the cross-correlation sum_j w[j+m]*a[j]
        # Let me verify: conv(w, a)[k] = sum_j w[j]*a[k-j]
        # corr = ifft(conj(W)*A)[m] = sum_j w[j+m]*a[j] (circular, with zero padding = linear)

        corr = irfft(np.conj(W2) * A2, fft_size2)
        # corr[m] = sum_j w[j+m]*a[j] for j=0,...,n-1 (and w[j+m] = 0 if j+m >= 2n-1)
        # This is what we want for the first term!
        # grad[m] = 2 * corr[m] (approximately, for the first term)
        # But we also need the second term: sum_j w[j]*a[2m-j]
        # = sum_j w[j]*a[j'] where j' = 2m-j, so j = 2m-j'
        # = sum_j' w[2m-j']*a[j'] = corr2[2m] where corr2[d] = sum_j w[d-j]*a[j]
        # = conv(w, a)[2m]... hmm this is getting circular.

        # Actually by the symmetry of the problem, if w is "nearly symmetric" (which it will be
        # when the max is spread out), then the two terms are approximately equal.
        # For now, let's just use grad = 2 * corr[:n] as an approximation.
        # This is exact when w is symmetric, which is a good approximation.

        grad = 2.0 * corr[:n]

        # Project gradient onto tangent space
        grad -= np.mean(grad)

        # Polyak step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        polyak_step = (max_b - M_lower) / grad_norm_sq
        step = min(eta, max(polyak_step, eta_min))

        # Take step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()

        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            eta = min(eta * 1.02, 5.0 / n)
        else:
            # Annealing accept
            T_ann = best_score * 0.002 * (1.0 - progress)
            if np.random.random() < np.exp(-(score_new - best_score) / max(T_ann, 1e-12)):
                a = a_new
            eta *= step_decay

        steps += 1
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.2e}")

        # Escape mechanism
        if steps % 500 == 0 and eta < 1e-6 / n:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.005 * a.mean(), n), 0)
            a = a * norm / a.sum()
            eta = 0.5 / n

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving: move mass from peak-contributing to non-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t_start = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find top contributing pair to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            n_stall += 1
            continue
        contrib = a[idx] * a[p_star - idx]
        src = idx[np.argmax(contrib)]

        # Find non-contributing positions (positions that don't affect the peak)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True  # include src as potential dst (unlikely but safe)

        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

        if len(nc_idx) == 0:
            break

        # Try several destination positions and fractions
        best_trial = best_score
        best_move = None

        n_try = min(20, len(nc_idx))
        # Prefer low-value non-contributing positions
        sorted_nc = nc_idx[np.argsort(a[nc_idx])[:n_try]]

        for dst in sorted_nc[:10]:
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial:
                    best_trial = sc
                    best_move = (src, dst, frac)

        if best_move is not None and best_trial < best_score:
            src, dst, frac = best_move
            a[src] -= a[src] * frac
            a[dst] += a_old_src * frac if 'a_old_src' in dir() else a[src] + a[src]*frac/(1-frac)
            # Simpler: just redo
            a_trial = a.copy()
            amt = a_trial[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * norm / a_trial.sum()
            a = a_trial
            best_score = best_trial
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        else:
            n_stall += 1
            # Random exploration
            if n_stall > 30:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.2)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0

        if n_stall > 150:
            # Larger perturbation
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                if a_trial[idx] > 0:
                    amt = a_trial[idx] * rng.uniform(0.1, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def generate_heuristic(n, style='ramp'):
    """Generate a heuristic initial sequence."""
    if style == 'ramp':
        a = np.arange(n, 0, -1, dtype=np.float64)
    elif style == 'exp':
        a = np.exp(-np.linspace(0, 3, n))
    elif style == 'bump':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / (2 * 0.1 ** 2))
    elif style == 'tri':
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - np.abs(2 * x - 1) * 2)
    elif style == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining LP Nesterov, subgradient, peak shaving,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.exp(-((x - 0.25) ** 2) / (2 * 0.15 ** 2))
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Heuristic init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting best: {best_score:.6f}, n={n_base}")

    # === Phase 1: Multi-length quick survey (90s) ===
    print(f"\n=== Phase 1: Multi-length survey ===")
    t1_total = min(90.0, time_budget * 0.05)
    n_candidates = [max(1000, n_base - 100), n_base, n_base + 100, n_base + 200, n_base + 400]
    n_candidates = sorted(set(n_candidates))
    t1_each = t1_total / len(n_candidates)

    survey_results = {}
    for n_try in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_alloc = min(t1_each, remaining - 10)
        if t_alloc < 5:
            break
        print(f"  Surveying n={n_try} (budget={t_alloc:.0f}s)")
        init = interpolate_sequence(best_seq, n_try)
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t_alloc, verbose=False, n_active=100, gamma_init=1.8)
        do_try_update(a_out, f"survey_n{n_try}")
        survey_results[n_try] = best_score

    # === Phase 2: Intensive LP at best length (50% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = remaining * 0.45
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t2, verbose=True, n_active=200, gamma_init=1.9)
    do_try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 3: Subgradient (15% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Subgradient n={n_cur} (budget={t3:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = subgradient_optimize(a_norm, t3, verbose=True)
    do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 4: LP round 2 (20% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur} (budget={t4:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t4, verbose=True, n_active=250, gamma_init=1.7)
    do_try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: Peak shaving (10% of remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Peak Shave n={n_cur} (budget={t5:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = peak_shave(a_norm, t5, seed=12345, verbose=True)
    do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_cur + 200
    t6 = remaining * 0.15
    if t6 > 30:
        print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6 * 0.6, verbose=True, n_active=150, gamma_init=1.9)
        do_try_update(a_out, f"lp_long_n{n_long}")

        # Peak shave on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = peak_shave(a_norm, min(remaining * 0.4, 60), seed=42, verbose=False)
            do_try_update(a_out, f"peak_long_n{n_cur}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t7 = remaining * 0.5
        print(f"\n=== Phase 7: Final LP polish n={n_cur} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, verbose=True, n_active=300, gamma_init=1.5)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t8 = remaining
        print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=98765, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
