"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent (multiplicative weights) with log-sum-exp gradient and acceleration
2. Pair coordinate descent with bisection line search
3. Multi-length exploration with interpolation
4. Adaptive beta schedule and perturbation-based restarts
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_smooth_grad(a, beta):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    c = fast_autocorr(a)
    size = len(c)
    c_shifted = c - c.max()
    log_w = beta * c_shifted
    max_lw = log_w.max()
    exp_lw = np.exp(log_w - max_lw)
    Z = exp_lw.sum()
    if Z < 1e-300:
        p = np.zeros(size)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_lw / Z
    p_rev = p[::-1]
    fft_size = 1 << (size + n - 2).bit_length()
    P = np.fft.rfft(p_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_pa = np.fft.irfft(P * A, fft_size)
    grad = 2.0 * conv_pa[size - 1 - np.arange(n)]
    return grad, c, p


def mirror_descent_optimize(a0, time_limit, seed=0, verbose=True):
    """Accelerated smoothed mirror descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta = 20.0 + 200.0 * (1.0 - np.exp(-t / 300.0))
        grad, c, p = compute_smooth_grad(y, beta)

        eta = 1.0 / (1.0 + 0.05 * t)
        log_upd = -eta * grad
        log_upd = np.clip(log_upd, -60, 60)
        w = a * np.exp(log_upd)
        w = np.maximum(w, 0.0)
        sw = w.sum()
        if sw < 1e-15:
            w = a.copy()
        else:
            w = w / sw

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * w
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            a = w.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a = y_new + momentum * (y_new - y)
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa < 1e-15:
                a = y_new.copy()
            else:
                a = a / sa

        y = y_new

        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        sc_y = 2.0 * n * fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_c = avg_a / n_avg
            s_avg = avg_c.sum()
            if s_avg > 1e-15:
                avg_c = avg_c / s_avg
                sc_avg = 2.0 * n * fast_autocorr(avg_c).max()

                    best_a = avg_c.copy()
                    stall = 0

            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0
            if verbose:
                print(f"    MD restart at t={t} ({time.monotonic()-t0:.0f}s)")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.10f} beta={beta:.1f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, seed=0, verbose=True):
    """Pair coordinate descent with bisection line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                gain = np.zeros(2 * n - 1)
                lo1 = max(0, i)
                hi1 = min(2 * n - 1, i + n)
                if hi1 > lo1:
                    src1 = np.arange(lo1, hi1) - i
                    gain[lo1:hi1] -= a[src1]
                lo2 = max(0, n - 1 - i)
                hi2 = min(2 * n - 1, n + i)
                if hi2 > lo2:
                    src2 = n - 1 - np.arange(lo2, hi2)
                    valid = (src2 >= 0) & (src2 < n)
                    if any(valid):
                        vals = np.zeros(hi2 - lo2)
                        vals[valid] = a[src2[valid]]
                        gain[lo2:hi2] -= vals

                lo3 = max(0, j)
                hi3 = min(2 * n - 1, j + n)
                if hi3 > lo3:
                    src3 = np.arange(lo3, hi3) - j
                    gain[lo3:hi3] += a[src3]
                lo4 = max(0, n - 1 - j)
                hi4 = min(2 * n - 1, n + j)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    if any(valid4):
                        vals4 = np.zeros(hi4 - lo4)
                        vals4[valid4] = a[src4[valid4]]
                        gain[lo4:hi4] += vals4

                # Bisection for optimal t in [0, ai]
                lo_t = 0.0
                hi_t = ai
                cur_max = conv.max()

                # Check if moving mass helps at all
                new_max_mid = (conv + ai * 0.5 * gain).max()
                di = 2 * i
                if di < 2 * n - 1:
                    new_max_mid = max(new_max_mid, conv[di] - 2 * a[i] * ai * 0.5 + (ai * 0.5) ** 2)
                dj = 2 * j
                if dj < 2 * n - 1:
                    new_max_mid = max(new_max_mid, conv[dj] + 2 * a[j] * ai * 0.5 + (ai * 0.5) ** 2)

                if new_max_mid >= cur_max - 1e-14:
                    continue

                # Bisection
                for _ in range(20):
                    mid_t = (lo_t + hi_t) / 2.0
                    new_max = (conv + mid_t * gain).max()
                    di = 2 * i
                    if di < 2 * n - 1:
                        new_max = max(new_max, conv[di] - 2 * a[i] * mid_t + mid_t * mid_t)
                    dj = 2 * j
                    if dj < 2 * n - 1:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * mid_t + mid_t * mid_t)
                    if new_max < cur_max - 1e-14:
                        lo_t = mid_t
                        cur_max = new_max
                    else:
                        hi_t = mid_t

                best_d = lo_t
                if best_d > 1e-14:
                    a[i] -= best_d
                    a[j] += best_d
                    conv = fast_autocorr(a)
                    new_score = 2.0 * n * conv.max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_seq(a, rng, scale=0.05):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def fw_step(a, c, n):
    """Single Frank-Wolfe step: find argmin of subgradient, interpolate."""
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        v = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            v += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            v += a[j2]
        grad[i] = 2.0 * v
    i_min = int(np.argmin(grad))
    return i_min
