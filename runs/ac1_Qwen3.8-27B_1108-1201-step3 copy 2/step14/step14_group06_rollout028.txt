def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

_global_solver = None


def get_solver():

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_T(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def normalize(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * (target_sum / max(len(a), 1))
    return a * target_sum / s


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    if T is None:
        T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):

    y = a.copy()

    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()

def mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_stages = 6
    temps = np.geomspace(60.0, 2.0, n_stages)

    for si, T_sm in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        n_iters = 1500
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr(a)
            shifted = T_sm * (b - b.max())
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()

            fft_size = 1 << (2 * n - 1).bit_length()
            w_full = np.zeros(fft_size)
            w_full[:2 * n - 1] = w
            a_full = np.zeros(fft_size)
            a_full[:n] = a
            W_fft = rfft(w_full, fft_size)
            A_fft = rfft(a_full, fft_size)
            grad = irfft(W_fft * A_fft, fft_size)[:n]

            eta = 0.4 / (it + 20.0)
            log_update = -eta * grad
            log_update -= log_update.max()
            update = np.exp(log_update)
            a_new = a * update
            s = a_new.sum()
            if s < 1e-15:
                a_new = np.ones(n) * norm_val / n
            else:
                a_new = a_new * norm_val / s

            a = a_new
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            print(f"    Mirror T={T_sm:.1f}: best={best_score:.8f}")

    return best_a, best_score


def coord_descent(a0, time_limit, seed=42, verbose=True):

    norm_val = np.sqrt(2.0 * n)

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_src = sorted_idx[:12]

        top_peaks = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        low_targets = non_peak[np.argsort(a[non_peak])[:12]]

        improved = False
        for src in top_src:
            if a[src] < 1e-10:
                continue
            for dst in low_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35]:

                    a_trial[dst] += amt
                    a_trial = normalize(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(60):
                i = int(rng.integers(n))
                j = int(rng.integers(n))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:

    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:

        return float(sc)
    except Exception:
        return float('inf')


def try_update_state(best_seq, best_score, candidate, label, t_global):
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on main length (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_cur} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.42
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    # Phase 2: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 220.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t2, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"mirror_n{n_cur}", t_global)

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.13, 160.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t3, seed=42, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"coord1_n{n_cur}", t_global)

    # Phase 4: LP Nesterov round 2 (refined starting point)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # Phase 5: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_cur + 400
    print(f"\n=== Phase 5: LP-Nest longer n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.22, 300.0)
    if t5 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Final polish - alternate coord descent and mirror
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_cur} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        third = remaining / 3.0

        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, third, seed=99999, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "coord_final1", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            n_cur2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
            a_out, _ = mirror_descent(a_norm2, min(remaining2, 100.0), verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "mirror_final", t_global)

        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_cur3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_cur3) / best_seq.sum()
            a_out, _ = coord_descent(a_norm3, remaining3, seed=777, verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, "coord_final2", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
