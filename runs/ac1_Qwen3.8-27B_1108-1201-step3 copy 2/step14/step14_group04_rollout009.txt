For convex problems, the LP-based von Neumann minimax is essentially a subgradient method where the subgradient is computed exactly via the LP. The advantage of the LP is that it gives the "exact" subgradient (in the sense of the supporting hyperplane), which leads to better convergence.

The analytical subgradient (using only the active peaks) is also exact at the optimum (by KKT conditions, the subgradient should be a weighted average of the active peak gradients, and the weights are the dual variables of the LP).

So both approaches should converge to the same point. The LP gives better per-iteration progress, while the analytical subgradient is faster per iteration.

My plan: alternate between the two. Use subgradient FISTA for quick progress, then LP for precise progress.

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def compute_subgradient(y, ac=None):
    """Compute subgradient of max_k conv(y,y)_k at point y."""
    n = len(y)
    if ac is None:
        ac = fast_autocorr(y)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n)
    # Find peaks within tolerance
    tol = max(1e-8 * M, 1e-10)
    peak_mask = ac >= M - tol
    peak_indices = np.where(peak_mask)[0]
    
    grad = np.zeros(n)
    for k in peak_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * y[k - idx]
    return grad, M


def subgrad_fista(a0, S, time_limit, verbose=False):

    if sc < best_score:

    if verbose:

    return best_a, best_score


def lp_best_response(a, M, solver, active_frac=1.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Option to only include active constraints for speed."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix
    # T[k,j] = a[k-j] if 0 <= k-j < n
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    
    if active_frac < 1.0:
        # Only include constraints that are "active" (close to M)
        ac = T_mat @ a  # = conv(a,a)
        threshold = M * active_frac
        active = ac >= threshold
        if active.sum() < 1:
            active = np.ones(len(active), dtype=bool)
        T_mat = T_mat[active]
    
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov minimax iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()

                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:

def alternating_refine(a0, time_limit, solver, verbose=True):
    """Alternate between subgradient FISTA and LP Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Alternate: 60s subgrad, 120s LP, 60s subgrad, 120s LP, ...
    block_sizes = [60, 120, 60, 120, 60, 120]
    block_idx = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        block_time = min(block_sizes[block_idx % len(block_sizes)], remaining)
        block_idx += 1
        
        if block_idx % 2 == 1:
            # Subgradient FISTA
            a_out, sc = subgrad_fista(a, norm, block_time, verbose=False)
        else:
            # LP Nesterov
            a_out, sc = lp_nesterov(a, block_time, solver, verbose=False)
        
        a = np.maximum(a_out, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * norm / s
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        elapsed = time.monotonic() - t0
        if verbose:
            print(f"    Alt block {block_idx} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall_count}")
        
        if stall_count > 8:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
