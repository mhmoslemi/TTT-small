def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def ramp_init(n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True,
                         m_reduction_start=0.05, m_reduction_end=0.12):
    """LP best-response with Nesterov acceleration and adaptive M reduction."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = ramp_init(n) * norm_val
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    best_M_reduction = m_reduction_start

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive M reduction: start gentle, get more aggressive over time
        progress = min(1.0, t / 200.0)
        m_red = m_reduction_start + (m_reduction_end - m_reduction_start) * progress
        # Also increase with iterations to force more improvement
        m_red = min(m_red + 0.001 * np.sqrt(t), 0.20)

        if use_tight_M:
            M_used = M_curr * (1.0 - m_red)
        else:
            M_used = M_curr

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            # Fallback: try with current M
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                # Try slightly looser
                b = lp_best_response(y, M_curr * 1.01, solver, T)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size based on improvement ratio
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.5)
            alpha = min(alpha / max(ratio, 0.03), 1.0)

        # Nesterov update
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Averaged iterate (helps with convex convergence)
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Stall detection

            stall_count += 1

def subgradient_refine(a0, time_limit, n_active=10):
    """Refine using subgradient of the max autocorrelation value."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = ramp_init(n) * norm_val
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    step_size = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find active set (positions near the max)
        threshold = max_b * 0.999
        active_k = np.where(b >= threshold)[0]
        if len(active_k) > n_active:
            active_k = active_k[np.argsort(b[active_k])[::-1]][:n_active]

        # Compute subgradient: for a_k in active set, grad_i = sum_{k} a_{k-i} (for valid indices)
        grad = np.zeros(n)
        for k in active_k:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                grad[j] += a[k - j]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Projected subgradient step
        a_trial = a - step_size * grad * max_b
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            step_size *= 0.5
            continue
        a_trial = a_trial * norm_val / s_t

        score = fast_autocorr(a_trial).max()
        if score < best_score - 1e-12:
            a = a_trial.copy()
            best_score = score
            best_a = a.copy()
            stall = 0
            step_size = min(step_size * 1.1, 0.05)
        else:
            stall += 1
            step_size *= 0.7
            if stall > 100:
                break

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = ramp_init(n) * norm_val
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Elements contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find low-value elements to receive mass
        low_val = np.argsort(a)[:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining LP, subgradient, and coordinate descent with ramp init."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = ramp_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Ramp init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov on parent/best length (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining * 0.38:.0f}s) ===")
    t1 = remaining * 0.38
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                    use_tight_M=True, m_reduction_start=0.06,
                                    m_reduction_end=0.15)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out,
                                          f"lp_p1_n{n_best}", t_global)

    # === Phase 2: LP Nesterov on ramp init at same length (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_ramp = len(best_seq)
    print(f"\n=== Phase 2: LP-Nest ramp init n={n_ramp} (budget={remaining * 0.20:.0f}s) ===")
    t2 = remaining * 0.20
    ramp_a = ramp_init(n_ramp) * np.sqrt(2.0 * n_ramp)
    a_out2, _ = nesterov_lp_optimize(ramp_a, t2, solver, verbose=True,
                                     use_tight_M=True, m_reduction_start=0.04,
                                     m_reduction_end=0.10)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out2,
                                          f"lp_ramp_n{n_ramp}", t_global)

    # === Phase 3: Try longer sequence (n+400) (18% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 3: LP-Nest longer n={n_long} (budget={remaining * 0.18:.0f}s) ===")
    t3 = remaining * 0.18
    if t3 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm_long = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out3, _ = nesterov_lp_optimize(a_norm_long, t3, solver, verbose=True,
                                         use_tight_M=True, m_reduction_start=0.05,
                                         m_reduction_end=0.12)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out3,
                                              f"lp_long_n{n_long}", t_global)

    # === Phase 4: LSE gradient refinement (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE gradient n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out4, _ = lse_scipy_optimize(a_norm, t4, T_start=60.0, T_end=2.0, n_stages=5)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out4,
                                              f"lse_n{n_best}", t_global)

    # === Phase 5: Subgradient refinement (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Subgradient n={n_best} (budget={remaining * 0.05:.0f}s) ===")
    t5 = min(remaining * 0.05, 80.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out5, _ = subgradient_refine(a_norm, t5, n_active=8)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out5,
                                              f"subgrad_n{n_best}", t_global)

    # === Phase 6: Targeted coordinate descent (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent n={n_best} (budget={remaining * 0.07:.0f}s) ===")
    t6 = min(remaining * 0.07, 100.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out6, _ = targeted_coord_descent(a_norm, t6, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out6,
                                              f"coord1_n{n_best}", t_global)

    # === Phase 7: Final LP round on best (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP polish n={n_best} (budget={remaining * 0.50:.0f}s) ===")
    t7 = remaining * 0.50
    if t7 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out7, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True,
                                         use_tight_M=True, m_reduction_start=0.10,
                                         m_reduction_end=0.18)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out7,
                                              f"lp_final_n{n_best}", t_global)

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final coord descent (budget={remaining * 0.6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out8, _ = targeted_coord_descent(a_norm, remaining * 0.6, seed=99999)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out8, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
