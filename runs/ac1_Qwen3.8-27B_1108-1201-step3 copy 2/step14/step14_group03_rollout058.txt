import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Compute linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]

def objective_value(a):
    """Compute the objective without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def build_toeplitz(a):
    """Build the Toeplitz matrix T such that (T @ b) = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def fw_line_search(a, b, n_evals=30):
    """
    Find the best step size tau in [0, 1] for the Frank-Wolfe step.
    a_new = (1-tau)*a + tau*b
    Minimize the objective.
    """
    best_tau = 0.0
    best_score = objective_value(a)
    
    # Binary search + golden section hybrid
    lo, hi = 0.0, 1.0
    # First, check if tau=1 is better
    score_1 = objective_value(b)
    if score_1 < best_score:
        best_tau = 1.0
        best_score = score_1
        hi = 1.0
    else:
        # The minimum is in (0, 1)
        # Use golden section
        gr = (np.sqrt(5) - 1) / 2
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        for _ in range(n_evals):
            fc = objective_value((1-c)*a + c*b)
            fd = objective_value((1-d)*a + d*b)
            if fc < fd:
                hi = d
                d = c
                c = hi - gr * (hi - lo)
            else:
                lo = c
                c = d
                d = lo + gr * (hi - lo)
            mid = (lo + hi) / 2
            fm = objective_value((1-mid)*a + mid*b)
            if fm < best_score:
                best_score = fm
                best_tau = mid
    
    return best_tau, best_score

def multiplicative_weight_optimize(a0, time_limit, verbose=True):

    n = len(a0)

    n_iters = 0

        c = fast_convolve(a, a)
        max_c = c.max()
        # Subgradient: 2 * (row of conv matrix at the argmax position)
        k_star = c.argmax()
        # gradient[i] = 2 * a[k_star - i] (for valid i)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad / max_c)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = objective_value(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        n_iters += 1
        
        # Decay eta
        eta *= 0.999
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")
    
    return best_a, best_score

def _fast_convolve(a, b):
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _objective(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    # Affected range: k in [i, i+n-1]
    c_aff = c[i:i + n] + 2.0 * d * a
    max_aff = c_aff.max()
    # Unaffected range
    if i > 0:
        max_unaff_left = c[:i].max()
    else:
        max_unaff_left = -np.inf
    if i + n < size:
        max_unaff_right = c[i + n:].max()
    else:
        max_unaff_right = -np.inf
    max_unaff = max(max_unaff_left, max_unaff_right) if np.isfinite(max_unaff_left) or np.isfinite(max_unaff_right) else -np.inf
    # Correction at k=2i
    max_corr = -np.inf
    if 2 * i < size:
        max_corr = c[2 * i] + 2.0 * d * a[i] + d * d
    max_total = max(max_aff, max_unaff, max_corr)
    s_new = 1.0 + d
    if s_new < 1e-10:
        return float('inf')
    return 2.0 * n * max_total / (s_new * s_new)


def _coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    c = _fast_autocorr(a)
    size = 2 * n - 1
    best_a = a.copy()
    best_score = 2.0 * n * c.max()
    t0 = time.monotonic()

    n_passes = 0
    no_improve = 0
    n_updates = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        n_updates_this_pass = 0

        # Precompute prefix and suffix max of c
        prefix_max = np.maximum.accumulate(c)
        suffix_max = np.maximum.accumulate(c[::-1])[::-1]

        perm = rng.permutation(n)
        n_recent_updates = 0

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            v = a[i]

            # Determine search range for d (change to a[i])
            if v < 1e-14:
                lo_d = 0.0
                hi_d = 0.01
            else:
                lo_d = -0.95 * v
                hi_d = 0.95 * v

            # Evaluate at several points
            n_evals = 15
            d_candidates = np.linspace(lo_d, hi_d, n_evals)
            scores = np.array([_cd_score_delta(c, a, n, size, i, d) for d in d_candidates])

            # Find best candidate
            min_idx = np.argmin(scores)
            best_d = d_candidates[min_idx]
            best_s = scores[min_idx]

            if best_s >= best_score * (1.0 - 1e-10):
                continue

            # Refine with bisection around best candidate
            if min_idx > 0 and min_idx < n_evals - 1:
                search_lo = d_candidates[min(min_idx, min_idx + 1 if scores[min_idx] <= scores[min_idx + 1] else min_idx - 1)]
                search_hi = d_candidates[max(min_idx, min_idx - 1 if scores[min_idx] <= scores[min_idx - 1] else min_idx + 1)]
                if search_lo > search_hi:
                    search_lo, search_hi = search_hi, search_lo
            elif min_idx == 0:
                search_lo = d_candidates[0]
                search_hi = d_candidates[1]
            else:
                search_lo = d_candidates[-2]
                search_hi = d_candidates[-1]

            for _ in range(10):
                mid = (search_lo + search_hi) / 2.0
                s_mid = _cd_score_delta(c, a, n, size, i, mid)
                s_lo = _cd_score_delta(c, a, n, size, i, search_lo)
                if s_mid < s_lo:
                    search_lo = mid
                else:
                    search_hi = mid

            best_d = (search_lo + search_hi) / 2.0
            best_s = _cd_score_delta(c, a, n, size, i, best_d)

            if best_s < best_score - 1e-12 and best_d != 0.0:
                a[i] += best_d
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] -= best_d
                    continue
                a = a / s_new
                c = _fast_autocorr(a)
                cur_score = 2.0 * n * c.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    pass_improved = True
                    n_updates += 1
                    n_updates_this_pass += 1
                    # Update prefix/suffix max periodically
                    if n_recent_updates >= 32:
                        prefix_max = np.maximum.accumulate(c)
                        suffix_max = np.maximum.accumulate(c[::-1])[::-1]
                        n_recent_updates = 0
                else:
                    # Revert
                    a[i] -= best_d / (1.0 + best_d)  # approximate revert
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    c = _fast_autocorr(a)

        no_improve += 0 if pass_improved else 1
        if no_improve >= 5:
            break

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, updates={n_updates}")

    return best_a, best_score


def _multiplicative_weight(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _objective(a)
    t0 = time.monotonic()
    eta = 1.0
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        c = _fast_autocorr(a)
        max_c = c.max()
        if max_c < 1e-15:
            break
        k_star = int(c.argmax())

        # Subgradient: 2 * (a shifted)
        grad = np.zeros(n)
        # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        grad[lo:hi] = 2.0 * a[k_star - (np.arange(lo, hi))]

        # Multiplicative update with softmax-like decay
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad / max_c
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = _objective(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        n_iters += 1

        # Decay eta slowly
        eta *= 0.998
        if eta < 0.01:
            eta = 0.01
