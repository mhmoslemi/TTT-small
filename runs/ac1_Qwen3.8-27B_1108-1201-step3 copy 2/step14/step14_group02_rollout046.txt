def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_translation_matrix_row(y, k):
    """Compute the row of T corresponding to lag k."""
    n = len(y)
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return row
    idx = np.arange(js, je + 1)
    row[idx] = y[k - idx]
    return row

    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    iter_count = 0
    
    for t in range(1, max_iters or 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select top-K constraints
        top_idx = np.argsort(ac)[::-1][:top_K]

        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in top_idx:
            k_int = int(k)
            row = compute_translation_matrix_row(y, k_int)
            if row.sum() > 0:
                constraints.append(row @ b_var <= M_curr)

            break
            
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)

            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        sv = v.sum()

        g = v / sv

        gap = M_curr - (ac @ g)  # This isn't quite right

def coordinate_descent(a0, time_limit, verbose=True):
    """Gradient-guided coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find the most contributing pair for the peak
        peak_k = int(np.argmax(ac))
        
        # Compute gradient-like information
        # For moving mass from i to j, the change in conv[k] is:
        # delta = a[i]*(a[k-j]-a[k-i]) + a[j]*(a[k-i]-a[k-j]) ... (approximate)
        # More precisely, moving eps from i to j:
        # conv[k] changes by: eps*(a[k-j] - a[k-i]) + eps*(a[j-k] - a[i-k]) (for the cross terms)
        # And the self terms change: -eps^2*(a[i-k]+a[j-k]+...) (higher order)
        
        # Find the peak contribution
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        src_candidates = np.arange(js, je + 1)
        contrib = a[src_candidates] * a[peak_k - src_candidates]
        
        if contrib.sum() < 1e-20:
            stall += 1
            continue
        
        src = src_candidates[np.argmax(contrib)]
        
        # Find best destination
        # Try moving to positions that reduce the peak
        n_try = min(n, 100)
        if n_try < n:
            # Prefer positions far from peak contributions
            peak_regions = np.zeros(n)
            for k2 in np.argsort(ac)[::-1][:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                peak_regions[s2:e2+1] += ac[k2i]
            candidates = np.argsort(peak_regions)[:n_try]
        else:
            candidates = np.arange(n)
        
        improved = False
        best_trial_m = best_m
        
        for dst in candidates:
            if dst == src:
                continue
            
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
        else:
            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolated_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    # === Phase 1: FW with line search at parent length ===
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FW n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = fw_with_linesearch(best_seq, t1, solver, top_K=80, verbose=True)
        try_update(a_out, f"fw_n{n0}")
    
    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: CD n={len(best_seq)} ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 20:
        a_out, _ = coordinate_descent(best_seq, t2, verbose=True)
        try_update(a_out, f"cd_n{len(best_seq)}")
    
    # === Phase 3: Perturb and re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Perturb+FW n={n_cur} ===")
    n_perturb = 5
    t3_per = min(remaining / (n_perturb * 2 + 1), 40.0)
    for i in range(n_perturb):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        # Perturb
        perturbed = best_seq.copy()
        idx = rng.choice(n_cur, size=n_cur//4, replace=False)
        noise = rng.uniform(-0.05, 0.05, size=len(idx))
        perturbed[idx] += noise * best_seq[idx].mean()
        perturbed = norm_sum1(perturbed)
        
        # Short FW
        a_out, _ = fw_with_linesearch(perturbed, t3_per, solver, top_K=60, verbose=False)
        try_update(a_out, f"perturb_{i}_n{n_cur}")
    
    # === Phase 4: Scale to 2400 ===
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: FW n={n1} ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_long = interpolated_sequence(best_seq, n1)
        a_out, _ = fw_with_linesearch(a_long, t4, solver, top_K=80, verbose=True)
        try_update(a_out, f"fw_n{n1}")
    
    # === Phase 5: CD at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: CD n={n_cur} ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 20:
        a_out, _ = coordinate_descent(best_seq, t5, verbose=True)
        try_update(a_out, f"cd2_n{n_cur}")
    
    # === Phase 6: Scale to 3200 ===
    n2 = 3200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: FW n={n2} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_long2 = interpolated_sequence(best_seq, n2)
        a_out, _ = fw_with_linesearch(a_long2, t6, solver, top_K=60, verbose=True)
        try_update(a_out, f"fw_n{n2}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_cur} ===")
    if n_cur <= 2500 and remaining > 60:
        t7 = min(remaining * 0.3, 180.0)
        a_out, _ = fw_with_linesearch(best_seq, t7, solver, top_K=100
