    if ac_max < 1e-15:

    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad(y, T):
    """Gradient of the log-sum-exp smoothed max of autocorrelation."""
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def project_simplex(v, target_sum=1.0):
    """Project onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    """LSE-FISTA optimization with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 30.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 8.0), n / (T * 1.5))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def enhanced_lp_nesterov(a0, time_limit, solver, restart_interval=400, verbose=True):
    """
    Enhanced LP-Nesterov with:
    - Backtracking line search for step size
    - Geometric ergodic averaging
    - Multi-M exploration (occasionally try tighter M)
    - Periodic restarts with small perturbations
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    rng = np.random.default_rng(12345)

    # Geometric averaging
    avg_a = np.zeros(n)
    avg_weight_total = 0.0
    avg_weight_current = 1.0
    gamma = 0.9995  # geometric decay factor

    t0 = time.monotonic()
    n_iters = 0
    t = 1

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Solve LP with current M
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            t += 1
            continue

        # Multi-M exploration: every 50 iters, also try tighter M
        if t % 50 == 0:
            b_tight = lp_best_response(y, M_curr * 0.97, solver)
            if b_tight is not None and b_tight.sum() > b.sum() * 0.5:
                # Blend the two responses
                b = 0.5 * b + 0.5 * b_tight

        b_sum = b.sum()
        if b_sum < 1e-15:
            t += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Backtracking line search
        y_score = fast_autocorr(y).max()
        best_alpha = 1.0 / (t + 1.0)
        alpha_candidates = [1.0 / (t + 1.0), 2.0 / (t + 2.0), 0.5, 0.25, 0.1, 0.05]
        # Sort by preference (smaller steps first for stability)
        alpha_candidates = sorted(set(alpha_candidates))

        chosen_alpha = alpha_candidates[0]
        for alpha in alpha_candidates:
            y_candidate = (1.0 - alpha) * y + alpha * g
            y_candidate = np.maximum(y_candidate, 0.0)
            s_y = y_candidate.sum()
            if s_y < 1e-15:
                continue
            y_candidate = y_candidate * norm / s_y
            cand_score = fast_autocorr(y_candidate).max()
            if cand_score <= y_score + 0.01 * y_score:
                chosen_alpha = alpha
                break
            chosen_alpha = alpha

        alpha = chosen_alpha
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov momentum for x
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Geometric ergodic averaging
        avg_a = avg_weight_current * a + (1.0 - avg_weight_current) * avg_a
        avg_weight_total += avg_weight_current
        avg_weight_current = 1.0 / (1.0 + gamma * average_weight_total_inverse())

        # Actually, let's use a simpler geometric average
        # Reset: maintain a running geometric average
        # avg_a = sum_t (gamma^(T-t) * a_t) / sum_t (gamma^(T-t))
        # Simplified: avg_a = (1-gamma)*a + gamma*avg_a
        avg_a = (1.0 - gamma) * a + gamma * avg_a
        if n_iters <= 1:
            avg_a = a.copy()

        # Check geometric average score periodically
        if n_iters % 15 == 0:
            avg_check = avg_a.copy()
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Periodic restart: add small perturbation
        if t % restart_interval == 0:
            perturbation_scale = norm * 0.01 / np.sqrt(t)
            noise = rng.standard_normal(n) * perturbation_scale
            y_perturbed = y + noise
            y_perturbed = np.maximum(y_perturbed, 0.0)
            s_p = y_perturbed.sum()
            if s_p > 1e-15:
                y_perturbed = y_perturbed * norm / s_p
                score_p = fast_autocorr(y_perturbed).max()
                if score_p < best_score * 1.01:
                    y = y_perturbed
                # Reset Nesterov counter by keeping t but noting the jump

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        t += 1

    return best_a, best_score


def average_weight_total_inverse():
    return 1.0


def coordinate_descent_polish(a0, time_limit, seed=777):
    """
    Coordinate descent polish: targeted peak reduction via mass redistribution.
    Uses systematic exploration of peak-contributing pairs.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    no_improve_counter = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks (excluding lag 0 and 2n-2 which are always small)
        peaks = np.argsort(b)[::-1]
        # Skip the energy peak at n-1 if it's the only one, focus on off-diagonal
        top_peaks = []
        for k in peaks[:20]:
            if k != n - 1 or len(top_peaks) > 3:
                top_peaks.append(k)
            if len(top_peaks) >= 6:
                break

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Try multiple source positions (top contributors)
            src_order = idx[np.argsort(contrib)[::-1]][:5]

            for src in src_order:
                # Find good destinations: positions that don't contribute to top peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]

                # Sort by current value (prefer moving to low-value positions)
                candidates_sorted = candidates[np.argsort(a[candidates])]
                n_try = max(3, len(candidates_sorted) // 8)
                try_dsts = candidates_sorted[:n_try]

                for dst in try_dsts:
                    if dst == src:
                        continue
                    for frac in [0.0003, 0.001, 0.003, 0.008, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-18:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            continue
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            no_improve_counter = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve_counter += 1
            # Random perturbation to escape local optima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            no_improve_counter = 0
            stall += 1
            if stall > 1500 or no_improve_counter > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length while preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: CD Polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t3, seed=42)
        try_update(a_out, f"cd1_n{n_best}")

    # === Phase 4: Second LP pass (refined starting point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP Pass 2 n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_lp_nesterov(a_norm, t4, solver,
                                         restart_interval=300, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 5: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 300
    t5 = min(remaining * 0.14, 180.0)
    if t5 > 40:
        print(f"\n=== Phase 5: LP at n={n2} (budget={t5:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = enhanced_lp_nesterov(a_norm, t5, solver,
                                         restart_interval=250, verbose=False)
        try_update(a_out, f"lp3_n{n2}")

    # === Phase 6: CD polish on longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 15:
        print(f"\n=== Phase 6: CD Polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t6, seed=9999)
        try_update(a_out, f"cd2_n{n_best}")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.6, 120.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = enhanced_lp_nesterov(a_norm, t7, solver,
                                             restart_interval=500, verbose=False)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
