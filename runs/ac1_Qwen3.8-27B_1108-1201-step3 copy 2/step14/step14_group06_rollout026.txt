    n = len(v)

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    LP-Nest stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def projected_gradient_descent(a0, time_limit, T_start=300.0, T_end=0.3, n_stages=10):
    """Projected LSE gradient descent with BB step size."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = time_limit / n_stages
        stage_end = t0 + (si + 1) * stage_time

        if si == 0:
            lr = 0.5
        else:
            lr = 0.3

        a_prev = a.copy()
        grad_prev = None
        stall = 0
        n_iter = 0

        while time.monotonic() < stage_end:
            lse_val, grad, c_max = lse_grad_autocorr(a, T)

            # BB step size
            if grad_prev is not None:
                s_vec = a - a_prev
                y_vec = grad - grad_prev
                sy = np.dot(s_vec, y_vec)
                yy = np.dot(y_vec, y_vec)
                if yy > 1e-20:
                    bb_lr = max(sy, 1e-10) / yy
                    lr = np.clip(bb_lr, 1e-6, 2.0)
            a_prev = a.copy()
            grad_prev = grad.copy()
            n_iter += 1

            # Gradient step
            a_new = a - lr * grad
            # Project
            a_new = project_simplex(a_new, S)

            score = fast_autocorr(a_new).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                lr *= 0.6
                stall += 1
                if stall > 80:
                    break

        if n_iter > 0 and (si + 1) % 3 == 0:
            print(f"    PGD T={T:.1f} stage {si+1}/{n_stages}: best={best_score:.8f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find low-value positions away from peak
        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False
        # Try targeted moves
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.005, 0.1))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Try swap moves
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2 * (0.7 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.4 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP-Nesterov + Projected Gradient + Coordinate Descent + Multi-length."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (main workhorse) - 45% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                     use_tight_M=True, eps_base=0.02, eps_growth=0.003)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp1_n{n_best}", t_global)

    # Phase 2: Projected Gradient Descent - 20% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: PGD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = projected_gradient_descent(a_norm, t2, T_start=300.0, T_end=0.3, n_stages=10)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"pgd_n{n_best}", t_global)

    # Phase 3: Coordinate Descent - 10% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"coord1_n{n_best}", t_global)

    # Phase 4: LP Nesterov round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest r2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False,
                                         use_tight_M=True, eps_base=0.015, eps_growth=0.002)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp2_n{n_best}", t_global)

    # Phase 5: PGD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: PGD r2 n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = projected_gradient_descent(a_norm, t5, T_start=100.0, T_end=0.1, n_stages=6)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"pgd2_n{n_best}", t_global)

    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    print(f"\n=== Phase 6: LP-Nest longer n={n_long} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 300.0)
    if t6 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True,
                                         use_tight_M=True, eps_base=0.02, eps_growth=0.003)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # Phase 7: Final PGD + Coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        half = remaining / 2.0
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = projected_gradient_descent(a_norm, half, T_start=50.0, T_end=0.1, n_stages=5)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "pgd_final", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_norm2, remaining2, seed=12345)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
