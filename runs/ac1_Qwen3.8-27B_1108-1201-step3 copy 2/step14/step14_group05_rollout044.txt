"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov with constraint generation (separation oracle)
2. LogSumExp-smoothed gradient descent with adaptive temperature
3. Direct peak shaving with multi-source/multi-destination moves

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")

def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve(a, b):
    """Compute full linear convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence score."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_convolution(a, b):
    """Compute T_a @ b where T_a is the convolution matrix built from a.
    This equals convolve(a, b) (full linear convolution)."""
    return fast_convolve(a, b)


def get_active_set(a, M, tol_frac=0.01):
    """Find constraints where (a*b)_k is close to M (for b=a, i.e., the autocorrelation).
    Returns indices where autocorr >= M * (1 - tol_frac)."""
    autocorr = fast_autocorr(a)
    threshold = M * (1.0 - tol_frac)
    active = np.where(autocorr >= threshold)[0]
    return active


def build_sub_conv_matrix(a, active_indices):
    """Build the submatrix T[active_indices, :] of the convolution matrix."""
    n = len(a)
    T_sub = np.zeros((len(active_indices), n))
    for row, i in enumerate(active_indices):
        # T[i, j] = a[i-j] if 0 <= i-j < n, else 0
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        if j_start <= j_end:
            T_sub[row, j_start:j_end+1] = a[i - np.arange(j_start, j_end + 1)]
    return T_sub


def lp_solve(a, M, solver, active_indices=None, max_active=300):
    """Solve the best response LP with constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    
    if active_indices is None:
        # Find active set
        autocorr = fast_autocorr(a)
        max_val = autocorr.max()
        threshold = max_val * 0.95
        active_indices = np.where(autocorr >= threshold)[0]
    
    # Limit to max_active constraints (keep the most binding)
    if len(active_indices) > max_active:
        autocorr = fast_autocorr(a)
        vals = autocorr[active_indices]
        top = np.argsort(vals)[::-1][:max_active]
        active_indices = active_indices[top]
    
    if len(active_indices) == 0:
        active_indices = np.array([np.argmax(fast_autocorr(a))])
    
    # Build submatrix
    T_sub = build_sub_conv_matrix(a, active_indices)
    M_sub = np.full(len(active_indices), M)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M_sub]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, max_active=200, max_active_round2=100):
    """Fessler-Parks accelerated LP Nesterov with constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8
    M_history = []
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        
        # Compute current M for y
        M_curr = fast_autocorr(y).max()
        M_history.append(M_curr)
        
        if M_curr < 1e-12:
            break
        
        # Solve LP for best response
        b = lp_solve(y, M_curr, solver, max_active=max_active)
        if b is None:
            # Fallback: use subgradient step
            autocorr = fast_autocorr(y)
            p = np.argmax(autocorr)
            # Subgradient: 2 * y[p - j] for valid j
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            g = np.zeros(n)
            g[js:je+1] = 2.0 * y[p - np.arange(js, je + 1)]
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g * norm / gs
            else:
                g = y.copy()
        else:
            b_sum = b.sum()
            g = b * norm / b_sum
        
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(max(momentum, 0.0), 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0
        
        # Adaptive gamma
        if len(M_history) > 100:
            recent = M_history[-50:]
            diffs = np.diff(recent)
            osc = np.sum(np.abs(diffs)) / (50 * max(recent[0], 1e-15))
            if osc > 0.05:
                gamma = max(1.0, gamma - 0.1)
            elif osc < 0.01:
                gamma = min(2.0, gamma + 0.05)
        
        # Restart if stalled
        if stall > 200:
            rng = np.random.default_rng(n_iters * 31 + 7)
            perturb = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.5
            M_history = []
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPNest iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")
    
    return best_a, best_M


def lse_gradient_descent(a0, time_limit, verbose=True, T_start=None, T_end=None):
    """LogSumExp-smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    if T_start is None:
        T_start = best_M * 0.1
    if T_end is None:
        T_end = best_M * 0.005
    
    t0 = time.monotonic()
    steps = 0
    a_rev = a[::-1]
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute autocorrelation
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Compute weights
        shifted = b - max_b  # for numerical stability
        w = np.exp(shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
        # = 2 * (w * a_rev)[n-1+m]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient onto tangent space of {sum = norm}
        grad -= np.mean(grad)
        
        # Adaptive step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        step = 1.0 / (n * 0.1)  # base step
        # Polyak-type adjustment
        objective = T * np.log(w_sum) + max_b  # f_T(a)
        progress_obj = objective - best_M  # upper bound on f_T - f_T*
        if progress_obj > 0:
            polyak_step = progress_obj / grad_norm_sq
            step = min(step, polyak_step)
        step = max(step, 1e-8 / n)
        
        # Take step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        a_rev = a[::-1]
        steps += 1
        
        # Check improvement
        score = fast_autocorr(a).max()
        if score < best_M:
            best_M = score
            best_a = a.copy()
