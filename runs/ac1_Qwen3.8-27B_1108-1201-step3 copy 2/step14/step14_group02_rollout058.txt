    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def soft_subgradient(a, top_k=30, beta=3.0):
    """Compute soft subgradient using top-k peaks with exponential weighting."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Exponential weighting based on peak height
    w = top_vals ** beta

    if w_sum < 1e-30:

        w = w / w_sum

            a_idx = k_int - j_idx
            g[j_idx] += 2.0 * wi * a[a_idx]

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_conv(a, b=None):
    """Compute conv(a, b) or conv(a, a) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    """Build the convolution matrix T where T@b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a, M, solver='highs'):
    """Solve max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_T(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = T
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method=solver, bounds=[(0, None)]*n)
    if res.success:
        return res.x
    return None

"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

_global_best_score = float('inf')


def fast_conv(a, b=None):
    """Compute full convolution of a and b (or a*a) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute full autocorrelation (convolution) of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize array to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix T where T @ b = conv(a, b). T has shape (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


def compute_soft_grad(a, ac, top_k=40, beta=3.0):
    """Compute soft subgradient using weighted combination of top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)

    # Weight by peak height with exponential sharpening
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            j_idx = np.arange(j_start, j_end)
            a_idx = k_int - j_idx
            g[j_idx] += 2.0 * wi * a[a_idx]
    return g


def mirror_descent(a0, time_limit, top_k=40, beta=3.0, eta0=0.5, verbose=True):
    """Mirror descent with soft subgradient, adaptive step, and Polyak averaging."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    theta = np.log(a + 1e-30)
    avg_a = a.copy()
    n_avg = 1
    iter_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        g = compute_soft_grad(a, ac, top_k=top_k, beta=beta)

        # Adaptive step size: decrease slowly
        eta = eta0 / (1.0 + 0.005 * t)

        # Mirror descent update in log-space
        theta = theta - eta * g
        theta_shifted = theta - theta.max()
        a_new = np.exp(np.clip(theta_shifted, -50, 50))
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        iter_count += 1

        # Track best
        M_new = fast_autocorr(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # Polyak averaging
        avg_a = avg_a + (a - avg_a) / n_avg
        n_avg += 1

        # Periodically check averaged solution
        if n_avg % 50 == 0:
            a_avg = norm_sum1(avg_a)
            M_avg = fast_autocorr(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    MD iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_M


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    n = len(a)
    T = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=T, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None


def nesterov_fw(a0, time_limit, verbose=True, max_iters=50000):
    """Nesterov-accelerated Frank-Wolfe with LP best-response."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0
    no_improve = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        g = solve_lp_direction(y, M_curr)
        if g is None:
            # Fallback: use gradient-based direction
            grad = compute_soft_grad(y, ac, top_k=20, beta=2.0)
            j_min = np.argmin(grad)
            g = np.zeros(n)
            g[j_min] = 1.0

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Check improvements
        prev_best = best_M
        my = fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_M:
            best_M = mx
            best_a = x.copy()

        if best_M < prev_best:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FW iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")

        # Early stop if no improvement for a long time
        if no_improve > 200 and iter_count > 50:
            break

    return best_a, best_M


def away_step_fw(a0, time_limit, verbose=True, max_iters=30000):
    """Away-step Frank-Wolfe with LP best-response."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    v_prev = None  # Previous vertex for away-step
    iter_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(x)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        g = solve_lp_direction(x, M_curr)
        if g is None:
            grad = compute_soft_grad(x, ac, top_k=20, beta=2.0)
            j_min = np.argmin(grad)
            g = np.zeros(n)
            g[j_min] = 1.0

        gamma = 2.0 / (t + 2.0)

        # Standard FW step
        z = (1.0 - gamma) * x + gamma * g

        # Away-step: minimize f((1-λ)*z + λ*(x - v_prev)) over λ in [0,1]
        if v_prev is not None:
            best_obj = fast_autocorr(z).max()
            best_lambda = 1.0
            # Try a few values of lambda (away-step parameter)
            for lam in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
                trial = (1.0 - lam) * z + lam * (x - v_prev)
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st < 1e-15:
                    continue
                trial = trial / st
                obj = fast_autocorr(trial).max()
                if obj < best_obj:
                    best_obj = obj
                    best_lambda = lam
            z = (1.0 - best_lambda) * z + best_lambda * (x - v_prev)
            z = np.maximum(z, 0.0)
            sz = z.sum()
            if sz < 1e-15:
                z = x.copy()
            else:
                z = z / sz

        x = z
        v_prev = g
        iter_count += 1

        M_new = fast_autocorr(x).max()
        if M_new < best_M:
            best_M = M_new
            best_a = x.copy()

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    ASF iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_M


def coordinate_search(a0, time_limit, rng_seed=42, verbose=True):
    """Stochastic coordinate search with simulated annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    temperature = 0.01
    cool_rate = 0.9995
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peaks
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        # Choose source: coordinate that contributes to peaks
        weights = np.zeros(n)
        for k in top_k[:10]:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            j_idx = np.arange(j_start, j_end)
            a_idx = k_int - j_idx
            weights[j_idx] += a[a_idx] * ac[k_int]

        total_w = weights.sum()
        if total_w < 1e-30:
            src = rng.integers(n)
        else:
            probs = weights / total_w
            src = rng.choice(n, p=probs)

        if a[src] < 1e-15:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue

        # Choose destination: low-contribution coordinate
        dst_weights = np.zeros(n)
        for k in top_k[:10]:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            j_idx = np.arange(j_start, j_end)
            a_idx = k_int - j_idx
            dst_weights[j_idx] += a[a_idx]

        # Inverse: prefer low contribution
        dst_probs = 1.0 / (dst_weights + 0.1)
        dst_probs /= dst_probs.sum()
        dst = rng.choice(n, p=dst_probs)

        if dst == src:
            continue

        # Try a random transfer fraction
        frac = rng.uniform(0.01, 0.15)
        amt = a[src] * frac

        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st

        M_trial = fast_autocorr(a_trial).max()

        # Acceptance criterion
        if M_trial < best_M:
            a = a_trial
            best_M = M_trial
            best_a = a.copy()
            temperature *= 0.99  # Cool faster on improvement
        elif M_trial < M_curr or rng.random() < np.exp(-(M_trial - M_curr) / temperature):
            a = a_trial

        temperature *= cool_rate
        iter_count += 1

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CS iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, T={temperature:.4f}")

    return best_a, best_M

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Stochastic CS n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_search(a_in, t5, rng_seed=777, verbose=True)
        try_update(a_out, f"cs_n{len(best_seq)}")

    # === Phase 6: Second round of MD + FW at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: MD+FW round 2 n={n_best} ===")
    if n_best <= 2000:
        t6a = min(remaining * 0.12, 150.0)
        if t6a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, t6a, top_k=50, beta=4.0, eta0=0.3, verbose=False)
            try_update(a_out, f"md2_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.10, 150.0)
        if t6b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_fw(a_in, t6b, verbose=True, max_iters=10000)
            try_update(a_out, f"nfw2_n{n_best}")

    # === Phase 7: Try larger length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_try = min(int(n0 * 1.5), 4000)
    print(f"\n=== Phase 7: Larger length n={n_try} ===")
    if remaining > 120 and n_try > n0:
        a_long = interpolate_sequence(best_seq, n_try)
        a_in = norm_sum1(a_long)

        t7 = min(remaining * 0.20, 250.0)
        t7_md = t7 * 0.5
        t7_fw = t7 * 0.5

        if t7_md > 20:
            a_out, _ = mirror_descent(a_in, t7_md, top_k=40, beta=3.0, eta0=0.4, verbose=True)
            try_update(a_out, f"md_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_try <= 2500:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_fw(a_in, min(remaining * 0.5, 200.0), verbose=True, max_iters=10000)
            try_update(a_out, f"nfw_n{n_try}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish n={n_best} ===")
    if remaining > 30:
        t8a = min(remaining * 0.4, 200.0)
        if t8a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = targeted_coordinate_polish(a_in, t8a, verbose=False)
            try_update(a_out, f"final_tcp_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and n_best <= 2500:
            t8b = min(remaining * 0.5, 150.0)
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_fw(a_in, t8b, verbose=False, max_iters=5000)
            try_update(a_out, f"final_nfw_n{n_best}")

    # === Phase 9: Final MD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final MD n={n_best} ===")
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, t9, top_k=60, beta=5.0, eta0=0.2, verbose=False)
            try_update(a_out, f"final_md_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = targeted_coordinate_polish(a_in, remaining * 0.7, verbose=False)
            try_update(a_out, f"final_tcp2_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
