```python
def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:

        return np.ones(n) * target_sum / n
    return target_sum * v / s
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP minimax with adaptive active set (top-K constraints only)
2. Nesterov acceleration for O(1/t) convergence
3. Periodic full-convolution checks to maintain active set accuracy
4. Coordinate descent polish
5. Multi-length exploration (n=1550 to 1700)
Key innovation: reduced LP (K constraints instead of 2n-1) enables 10-50x more iterations
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# Global solver reference
_lp_solver = 'HiGHS'

def fast_autocorr(a):
    """Fast autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_rows(a, indices):
    """Build rows of convolution matrix: row i is the coefficient vector for lag indices[i]."""
    n = len(a)
    K = len(indices)
    A_mat = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        if js_start <= js_end:
            # A_mat[i, j] = a[k-j] for j in [js_start, js_end]
            a_idx = np.arange(k - js_end, k - js_start + 1)
            A_mat[i, js_start:js_end+1] = a[a_idx][::-1]
    return A_mat

def solve_reduced_lp(a, M, active_indices):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for k in active_indices, b >= 0
    Returns b* or None."""
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n
    
    A_mat = build_conv_rows(a, active_indices)
    
    # Primal: max 1'b s.t. A_mat @ b <= M*ones(K), b >= 0
    # Use scipy linprog: min -1'b s.t. A_mat @ b <= M, -b <= 0
    c = -np.ones(n)
    A_ub = A_mat  # A_mat @ b <= M
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_minimax_active(a0, time_limit, K_init=20, K_max=300, check_interval=30, verbose=True):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    b_full = fast_autocorr(a)
    best_peak = b_full.max()
    y = a.copy()
    
    t0 = time.monotonic()
    K = K_init
    n_iters = 0
    active_set = np.argsort(b_full)[::-1][:K]
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Periodically update active set with full convolution
        if t % check_interval == 1:
            b_full = fast_autocorr(y)
            M = b_full.max()
            # Get top K indices
            active_set = np.argsort(b_full)[::-1][:K]
            # Gradually increase K
            K = min(K + 2, K_max)
        else:
            M = fast_autocorr(y).max()
        
        if M < 1e-14:
            break
        
        # Solve reduced LP
        b_star = solve_reduced_lp(y, M, active_set)
        if b_star is None:
            # Try with smaller active set
            if K > 5:
                active_set_small = active_set[:max(5, K//2)]
                b_star = solve_reduced_lp(y, M, active_set_small)
            if b_star is None:
                continue
        
        b_sum = b_star.sum()
        g = b_star / b_sum  # normalize to sum=1
        
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Track best (check both y and a)
        peak_y = fast_autocorr(y).max()
        if peak_y < best_peak:
            best_peak = peak_y
            best_a = y.copy()

    if peak_a < best_peak:

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent: transfer mass between pairs to reduce peak."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:4]
        improved = False
        
        # Try to reduce each top peak
        for k_idx in top_k:
            k = k_idx
            js_start = max(0, k - n + 1)
            js_end = min(n - 1, k)
            idx = np.arange(js_start, js_end + 1)
            if len(idx) == 0:
                continue
            
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destination (low values, not contributing to peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                peak_mask[s2:e2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick a few random candidates and the lowest-valued ones
            n_try = min(5, len(candidates))
            random_cands = rng.choice(candidates, size=min(n_try, len(candidates)), replace=False)
            low_cands = candidates[np.argsort(a[candidates])[:n_try]]
            dst_candidates = np.unique(np.concatenate([random_cands, low_cands]))
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-20 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    peak = fast_autocorr(a_trial).max()
                    if peak < best_peak - 1e-15:
                        a = a_trial
                        best_peak = peak
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP minimax with adaptive active set (top-K constraints only) - 10-50x faster than full LP
2. Nesterov acceleration for O(1/t) convergence
3. Periodic full-convolution checks to maintain active set accuracy
4. Coordinate descent polish for final improvement
5. Multi-length exploration
Key innovation: reduced LP (K constraints instead of 2n-1) enables far more iterations in the same time.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Fast autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_rows(a, indices):
    """Build rows of convolution matrix for given lag indices."""
    n = len(a)
    K = len(indices)
    A_mat = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        if js_start <= js_end:
            a_idx = np.arange(k - js_end, k - js_start + 1)[::-1]
            A_mat[i, js_start:js_end + 1] = a[a_idx]
    return A_mat


def solve_reduced_lp(a, M, active_indices):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for k in active_indices, b >= 0."""
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n
    A_mat = build_conv_rows(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.status != 0 or res.x is None:
        return None
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        a_out, _ = lp_minimax_active(best_seq, t2, K_init=60, K_max=500, check_interval=10, verbose=True)
        best_seq, best_score = try_update_sequence(a_out, f"lp2_n{n_best}", best_seq, best_score, t_global)

    # === Phase 3: Coordinate descent polish (12%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: CD n={n_best} (budget={t3:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t3, seed=12345, verbose=True)
        best_seq, best_score = try_update_sequence(a_out, f"cd_n{n_best}", best_seq, best_score, t_global)

    # === Phase 4: Try different lengths (12%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_current = len(best_seq)
    length_options = [n_current - 150, n_current + 150, n_current + 300]
    for n_try in length_options:
        if n_try <= 100 or n_try > 3000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t4 = min(remaining * 0.4, 250.0)
        if t4 < 30:
            break
        print(f"\n=== Phase 4: LP n={n_try} (budget={t4:.0f}s) ===")
        init_n = interpolate_sequence(best_seq, n_try)
        a_out, _ = lp_minimax_active(init_n, t4, K_init=30, K_max=200, check_interval=20, verbose=False)
        best_seq, best_score = try_update_sequence(a_out, f"lp_n{n_try}", best_seq, best_score, t_global)
        # Quick CD on the new length result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t4b = min(remaining * 0.3, 60.0)
            if a_out is not None:
                a_out2, _ = coordinate_descent(a_out, t4b, seed=777, verbose=False)
                best_seq, best_score = try_update_sequence(a_out2, f"cd_n{n_try}", best_seq, best_score, t_global)

    # === Phase 5: Final LP pass on best (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t5 = min(remaining * 0.7, 200.0)
        print(f"\n=== Phase 5: Final LP n={n_best} (budget={t5:.0f}s) ===")
        a_out, _ = lp_minimax_active(best_seq, t5, K_init=50, K_max=400, check_interval=15, verbose=False)
        best_seq, best_score = try_update_sequence(a_out, f"lp_final_n{n_best}", best_seq, best_score, t_global)
        # Quick CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = coordinate_descent(best_seq, min(remaining * 0.8, 60.0), seed=55555, verbose=False)
            best_seq, best_score = try_update_sequence(a_out2, "cd_final", best_seq, best_score, t_global)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
