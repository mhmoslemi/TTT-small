    n = len(a)
    if T is None:
        T = build_T(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_grad(a, n, temp):
    """Compute LSE-smoothed max of conv(a,a) and gradient."""
    c = fast_conv(a)
    c_max = c.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n)
    shifted = temp * (c - c_max)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum = c_max + np.log(exp_vals.sum()) / temp
    w = exp_vals / exp_vals.sum()

    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    # = 2 * sum_i w[j+i] * a[i]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * corr[:n]
    return log_sum, grad


def lse_optimize(a0, time_limit, t_start, t_end, n_stages, n):
    """Multi-stage LSE optimization."""
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0) * norm_val
    best_a = a.copy()
    best_score = fast_conv(best_a).max()
    t0 = time.monotonic()

    temps = np.geomspace(t_start, t_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        stage_time = min(remaining / max(n_stages - si, 1), 120.0)
        if stage_time < 5:
            break

        def obj_func(x, Tv=T):
            a_n = np.maximum(x, 0.0)
            s = a_n.sum()
            if s < 1e-10:
                return 1e10
            a_norm = a_n / s
            val, _ = lse_grad(a_norm, n, Tv)
            return val

        def grad_func(x, Tv=T):
            a_n = np.maximum(x, 0.0)
            s = a_n.sum()
            if s < 1e-10:
                return np.zeros(n)
            a_norm = a_n / s
            _, g = lse_grad(a_norm, n, Tv)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 500, 'maxfun': 2000,
                                          'gtol': 1e-10, 'ftol': 1e-12,
                                          'maxls': 50})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                s_c = a_c.sum()
                if s_c > 1e-10:
                    a_c = a_c * norm_val / s_c
                    sc = fast_conv(a_c).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_c.copy()
                        a = a_c.copy()
        except Exception:
            pass

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and perturbation restarts."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0) * norm_val
    best_a = a.copy()
    best_score = fast_conv(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(12345)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_conv(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive M schedule: aggressive early, conservative late
        frac = min(t / 5000.0, 1.0)
        if frac < 0.3:
            eps = 0.04
        elif frac < 0.7:
            eps = 0.015
        else:
            eps = 0.005
        M_used = M_curr * (1.0 - eps)

        T_mat = build_T(y, n)
        b = lp_solve(y, M_used, solver, T_mat)
        if b is None:
            b = lp_solve(y, M_curr, solver, T_mat)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        M_new = fast_conv(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)
            alpha = max(alpha, 0.01)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_conv(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Perturbation restart when stalled
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 3:
                # Perturb and restart
                noise = rng.normal(0, 0.05 * best_score / n, size=n)
                y_pert = y + noise
                y_pert = np.maximum(y_pert, 0.0)
                s_p = y_pert.sum()
                if s_p > 1e-10:
                    y = y_pert * norm_val / s_p
                    a = y.copy()
                    avg_a = np.zeros(n)
                    n_avg = 0
                stall_count = 0
                last_improve = time.monotonic()
                if verbose:
                    print(f"    [perturb restart] iter={n_iters}")
            elif verbose and stall_count == 3:
                print(f"    [stall detected] iter={n_iters}, best={best_score:.8f}")
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def grad_guided_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent guided by LSE gradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0) * norm_val
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    temp = 20.0

    while time.monotonic() - t0 < time_limit:
        # Compute gradient to guide moves
        _, grad = lse_grad(a / norm_val, n, temp)

        # Find top contributing elements to peaks
        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_src = sorted_idx[:12]

        # Find low-gradient targets (safe to increase)
        sorted_targets = np.argsort(grad)[:12]

        improved = False
        for src in top_src:
            if a[src] < 1e-12:
                continue
            for dst in sorted_targets:
                if dst == src:
                    continue
                for frac in [0.03, 0.07, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] = a[src] - amt
                    a_trial[dst] = a_trial[dst] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves as fallback
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_conv(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Anneal temperature and try perturbation
            temp = max(1.0, temp * 0.9)
            stall += 1
            if stall % 10 == 0:
                noise = rng.normal(0, 0.02, size=n)
                a_pert = a + noise * a.std()
                a_pert = np.maximum(a_pert, 0.0)
                s_p = a_pert.sum()
                if s_p > 1e-10:
                    a_pert = a_pert * norm_val / s_p
                    sc = fast_conv(a_pert).max()
                    if sc < best_score:
                        a = a_pert
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 100:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 6.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Two-lobe structure
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2)
    else:
        # Flat with modulation
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update(best_seq, best_score, candidate, label, t_global):
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = evaluate_sequence([float(x) for x in a_norm])
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    # Initialize from parent or default
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_sequence([float(x) for x in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP-Nesterov (primary workhorse) - 35% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_cur} (budget={remaining * 0.35:.0f}s) ===")
    t1 = remaining * 0.35
    a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    # Phase 2: LSE gradient optimization - 15%
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE gradient n={n_cur} (budget={remaining * 0.15:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t2, 100.0, 2.0, 6, n_cur)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lse_n{n_cur}", t_global)

    # Phase 3: Gradient-guided coordinate descent - 10%
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_cur} (budget={remaining * 0.10:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = grad_guided_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"coord1_n{n_cur}", t_global)

    # Phase 4: LP-Nesterov round 2 - 15%
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest round 2 n={n_cur} (budget={remaining * 0.15:.0f}s) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # Phase 5: Try longer sequence - 15%
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 5: LP-Nest longer n={n_long} (budget={min(remaining * 0.15, 250.0):.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 60:
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Final polish with alternating methods
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_cur} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Coord descent
        third = remaining * 0.3
        if third > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = grad_guided_coord_descent(a_norm, third, seed=99999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_f1", t_global)

        # LSE with lower temps
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            n_cur2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
            a_out, _ = lse_optimize(a_norm2, min(remaining2 * 0.3, 80.0), 30.0, 0.5, 4, n_cur2)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lse_f", t_global)

        # Final LP
        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 20:
            n_cur3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_cur3) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm3, remaining3 * 0.4, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_f", t_global)

        # Final coord
        remaining4 = time_budget - (time.monotonic() - t_global)
        if remaining4 > 10:
            n_cur4 = len(best_seq)
            a_norm4 = best_seq * np.sqrt(2.0 * n_cur4) / best_seq.sum()
            a_out, _ = grad_guided_coord_descent(a_norm4, remaining4, seed=777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_f2", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
