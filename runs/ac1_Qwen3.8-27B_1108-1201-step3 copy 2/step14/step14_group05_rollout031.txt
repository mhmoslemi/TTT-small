dg/da[j] = 2*n * sum_k softmax(T*b)[k] * 2*a[k-j] (for valid k-j)
= 4*n * sum_k w[k] * a[k-j]

where w[k] = exp(T*b[k]) / sum_l exp(T*b[l])

This is a weighted average of shifted/reversed copies of a!

And the mirror descent update is:
a_new[j] = a[j] * exp(-eta * gradient[j]) / normalizer

As T -> infinity, this converges to the subgradient method using only the argmax.
As T -> 0, it uses all peaks equally.

This is a smooth approximation that should converge nicely.

Now, the key question is: what step size eta to use?

For mirror descent on a convex problem with step size eta, the convergence rate is O(1/eta) + O(eta * L) where L is the smoothness constant. The optimal eta balances these.

For our problem, the "smoothness" depends on the scale of the convolution. Let's use:
eta = 1 / (max_contribution) where max_contribution = max_j |gradient[j]|

Or use a line search / backtracking rule.

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    stall = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fft_size_g = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature: starts moderate, ends very sharp
        T = T_start * (T_end / T_start) ** (progress ** 0.5)
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Softmax weights (LSE gradient direction)
        b_clip = np.clip(T * (b - max_b), -500, 500)
        w = np.exp(b_clip)
        w_sum = w.sum()
        if w_sum < 1e-30:
            T = T / 10  # Reduce T if underflow
            continue
        w = w / w_sum
        
        # Gradient via FFT: grad[j] = 2 * sum_k w[k] * a[k-j]
        w_rev = w[::-1]
        A_g = rfft(a, fft_size_g)
        W_g = rfft(w_rev, fft_size_g)
        conv_g = irfft(A_g * W_g, fft_size_g)
        grad = 2.0 * conv_g[2*n-2 : 2*n-2+n].copy()
        
        # Tangent space projection
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Adaptive step size
        grad_norm = np.sqrt(np.dot(grad, grad))
        if grad_norm < 1e-15:
            break
        
        # Step size: decrease with progress, scale with T
        eta = 1.0 / (T * grad_norm * 0.5 + 1e-10)
        eta = min(eta, 2.0 / np.sqrt(n))
        eta = max(eta, 1e-6)
        
        # Multiplicative update
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -30, 30)
        a_new = a * np.exp(exp_arg)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score_new = fast_autocorr(a_new).max()
        
        # Accept/reject with annealing
        if score_new <= best_score:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Annealing acceptance
            T_anneal = best_score * 0.005 * (1.0 - progress)
            if T_anneal > 1e-10 and rng.random() < np.exp(-(score_new - best_score) / T_anneal):
                a = a_new
                stall += 1
            else:
                stall += 1
                eta *= 0.8
        
        # Restart if stuck too long
        if stall > 200:
            perturb = rng.normal(0, 0.05 * a.mean(), n)
            a = np.maximum(a + perturb, 1e-10)
            a = a / a.sum()
            stall = 0
            
        steps += 1
        
        if verbose and steps % 500 == 0:
            print(f"    LSD step={steps} ({elapsed:.0f}s): T={T:.0f}, eta={eta:.2e}, best={best_score:.6f}, curr={score_new:.6f}, stall={stall}")
    
    return best_a, best_score


def topk_subgradient_nesterov(a0, time_limit, K=5, verbose=True, seed=42):
    """Top-K subgradient with Nesterov acceleration."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Nesterov state
    y = a.copy()
    y_prev = a.copy()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Step size
    step_size = 1.0 / n
    best_lower = best_score * 0.99  # Optimistic lower bound
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        b = fast_autocorr(y)
        max_b = b.max()
        
        # Find top K peaks
        K_eff = min(K, size)
        top_idx = np.argsort(b)[::-1][:K_eff]
        top_vals = b[top_idx]
        
        # Softmax weighting over top K
        T_w = max(0.1, (max_b - top_vals[-1] + 1e-10))
        w_k = np.exp((top_vals - max_b) / T_w)
        w_k = w_k / w_k.sum()
        
        # Compute combined subgradient
        # For each peak p, gradient contribution at j is: 2*a[p-j] for valid j
        # Total gradient: sum_k w_k * 2 * a[p_k - j]
        # = 2 * sum_k w_k * shifted_reversed_a[j]
        grad = np.zeros(n)
        for k in range(K_eff):
            p = top_idx[k]
            wk = w_k[k]
            if wk < 1e-8:
                continue
            # grad[j] += 2 * wk * a[p-j] for 0 <= p-j < n
            j_start = max(0, p - n + 1)
            j_end = min(n - 1, p)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                grad[js] += 2.0 * wk * a[p - js]
        
        # Project to tangent space
        grad = grad - grad.mean()
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        
        # Polyak step size
        gap = max_b - best_lower
        if gap > 0:
            polyak_step = gap / (grad_norm_sq + 1e-30)
            step = min(step_size, polyak_step)
        else:
            step = step_size
        step = max(step, 1e-8)
        
        # Take step
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov update for y
        t = steps + 1
        if t == 1:
            y_new = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + np.sqrt(1 + 8.0 * t * t) / 2.0 - 1.0)
            # Standard Nesterov: momentum = (t-1) / (1 + sqrt(1+2t))... let me use simpler
            momentum = (np.sqrt(t) - 1.0) / (np.sqrt(t - 1) + 1.0) if t > 1 else 0.0
            # Actually, let's use the Fessler-Parks style:
            momentum = min(1.0, (t + 1.0) / (t + 2.0))
            y_new = a_new + momentum * (a_new - y_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y
        
        y_prev = y.copy()
        y = y_new
        steps += 1
        
        # Track best
        score_y = fast_autocorr(y).max()
        score_a = fast_autocorr(a_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_lower = min(best_lower, score_y * 0.999)
            step_size = min(step_size * 1.1, 5.0 / n)
        elif score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            best_lower = min(best_lower, score_a * 0.999)
        
        # Decay step size
        step_size *= 0.999
        
        if verbose and steps % 500 == 0:
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.6f}, step={step:.2e}")
    
    return best_a, best_score


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, max_constraints=500):
    """LP-based Fessler-Parks Nesterov with constraint reduction."""
    if not HAS_CP or solver is None:
        return a0, fast_autocorr(a0).max()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    y = a.copy()
    y_prev = a.copy()
    
    size = 2 * n - 1
    # Build Toeplitz matrix once
    T_mat = np.zeros((size, n))
    for j in range(n):
        T_mat[j:j+n, j] = a  # This is wrong, need to rebuild at each step
    
    # Actually, T depends on a, so we need to rebuild it each iteration.
    # For efficiency, build it on the fly.
    
    gamma = 1.8
    
    while time.monotonic() - t0 < time_limit:
        t_iter = n_iters + 1
        elapsed = time.monotonic() - t0
        
        # Compute current peak
        b_y = fast_autocorr(y)
        M = b_y.max()
        if M < 1e-12:
            break
        
        # Build convolution matrix for current y
        # T[k, j] = coefficient of (y*b)[k] with respect to b[j]
        # (y*b)[k] = sum_j y[k-j] * b[j] for valid k-j
        # So T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        T_mat = np.zeros((size, n))
        for j in range(n):
            # T[k, j] = y[k-j] for k = j, j+1, ..., j+n-1
            T_mat[j:j+n, j] = y
        
        # Select constraints: top max_constraints by (T @ b_opt prev) or just all
        # For simplicity, use all constraints
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M]
        obj = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            n_iters += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            n_iters += 1
            continue
        if b_var.value is None:
            n_iters += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            n_iters += 1
            continue
        
        # Normalize b to be a probability vector
        g = b_val / b_sum
        n_iters += 1
        
        # Fessler-Parks update
        alpha = 2.0 / (t_iter + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 1e-30)
        s_y = y_new.sum()
        y_new = y_new / s_y
        
        # Nesterov momentum
        if t_iter == 1:
            a_new = g.copy()
        else:
            momentum = (t_iter + 1.0 - gamma) / (t_iter + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 1e-30)
            s_a = a_new.sum()
            a_new = a_new / s_a
        
        y_prev = y.copy()
        y = y_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a_new).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        
        if verbose and n_iters % 50 == 0:
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.6f}, M={M:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving via targeted mass transfer."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing pairs for peak p_star
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 50:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Try multiple destinations
        best_trial = None
        best_trial_score = max_b
        
        # Random destinations
        for _ in range(5):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                sc = fast_autocorr(a_t).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_t.copy()
        
        # Also try "spread" move: reduce src, add to multiple low positions
        low_positions = np.argsort(a)[:10]
        low_positions = [p for p in low_positions if p != src]
        if low_positions:
            dst = low_positions[int(rng.integers(len(low_positions)))]
            for frac in [0.1, 0.2, 0.3]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                sc = fast_autocorr(a_t).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_t.copy()
        
        if best_trial is not None and best_trial_score < best_score:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            n_stall = 0
        else:
            n_stall += 1
            # Annealing
            if best_trial is not None:
                T_ann = best_score * 0.002 * (1.0 - progress)
                if T_ann > 1e-10:
                    delta = best_trial_score - best_score
                    if rng.random() < np.exp(-delta / max(T_ann, 1e-12)):
                        a = best_trial
        
        if n_stall > 100:
            # Larger perturbation
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    amt = a[i] * rng.uniform(0.1, 0.4)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            n_stall = 0
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
