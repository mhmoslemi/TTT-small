            best_score = score_y
            best_a = y.copy()

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov it={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, window=30, verbose=True):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm_target)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm_target * 1e-10:
                continue

            for offset in range(1, min(window + 1, n)):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm_target * 1e-10:
                        break

                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[:hi1 - lo1]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        src = n - 1 - np.arange(lo2, hi2)
                        valid = (src >= 0) & (src < n)
                        if valid.any():
                            delta_vec[lo2:hi2][valid] += a[src[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] -= a[:hi3 - lo3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        src4 = n - 1 - np.arange(lo4, hi4)
                        valid4 = (src4 >= 0) & (src4 < n)
                        if valid4.any():
                            delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        nv = new_conv.max()
                        if nv < best_val - 1e-13:
                            best_val = nv
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = normalize_to(a, norm_target)
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            improved = True
                            no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Adam softmax (50% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    print(f"\n=== Phase 1: Adam softmax n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc = adam_softmax_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"adam_n{n_base}")

    # Phase 2: Peeling (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Peeling n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = peeling_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"peel_n{n_best}")

    # Phase 3: LP Nesterov (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LP Nesterov n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = lp_nesterov_phase(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # Phase 4: Adam again with lower temperatures (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Adam round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = adam_softmax_optimize(a_norm, t4, verbose=False)
        do_try_update(a_out, f"adam2_n{n_best}")

    # Phase 5: CD polish (5% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.05
    print(f"\n=== Phase 5: CD polish n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = pair_cd_polish(a_norm, t5, rng, window=25, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
