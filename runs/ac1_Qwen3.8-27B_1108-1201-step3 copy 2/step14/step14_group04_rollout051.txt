        n_tail += 1
        if n_tail > tail_window:
            tail_a -= a  # This is wrong, need a circular buffer
            # Actually let's just use full average and a separate recent window

        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        # Periodic micro-restart with LSE-FISTA
        if micro_restart_every > 0 and n_iters % micro_restart_every == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > micro_restart_time + 1.0:
                a_norm = best_a / best_a.sum()
                a_restart, _ = lse_fista(a_norm, T=0.1, time_limit=micro_restart_time, verbose=False, max_iters=200)
                a_restart = a_restart * norm / a_restart.sum()
                sc_r = fast_autocorr(a_restart).max()
                if sc_r < best_score:
                    best_score = sc_r
                    best_a = a_restart.copy()
                    # Restart LP from improved point
                    y = best_a.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest it={n_iters} ({elapsed:.0f}s): best={best_score:.8f} M={M_curr:.6f}")

    return best_a, best_score


def lse_fista_anneal(a0, T_schedule, time_limit, verbose=False):
    """LSE-FISTA with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        stage_time = min(remaining * 0.4, 60.0)
        a_out, sc_out = lse_fista(a, T, stage_time, verbose=False, max_iters=30000)
        a_out = a_out * norm / a_out.sum()
        sc = fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a = a_out
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA anneal stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_peak_shave(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0003, 0.0007, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with adaptive LP, LSE polishing, and length exploration."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov enhanced (main workhorse, ~50% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.50, 800.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP-Nesterov enhanced n={n_best} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t1, solver, verbose=True,
                                         micro_restart_every=100, micro_restart_time=0.04)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 2: LSE-FISTA annealing polish (~15% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: LSE-FISTA anneal n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq / best_seq.sum()
        T_sched = [2.0, 0.5, 0.15, 0.05, 0.02, 0.01]
        a_out, _ = lse_fista_anneal(a_norm, T_sched, t2, verbose=True)
        try_update(a_out, f"lse1_n{n_best}")

    # === Phase 3: LP again after LSE polish (~15% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t3, solver, verbose=True,
                                         micro_restart_every=60, micro_restart_time=0.03)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Very low-T LSE-FISTA for final precision (~10% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Low-T LSE n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq / best_seq.sum()

        try_update(a_out, f"lse_lowT_n{n_best}")

                t_r = min(remaining3 * 0.25, 40.0)
                noise = rng.normal(0, 0.001, n_best)
                a_pert = best_seq + noise
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                a_out, _ = lse_fista(a_pert, T=0.005, time_limit=t_r, verbose=False, max_iters=50000)
                try_update(a_out, f"lse_restart{restart_seed}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Targeted peak shaving on best (~5% budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Targeted shave n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t6, seed=55555)

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 15:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_enhanced(a_norm, t7, solver, verbose=False,

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t8 = min(remaining * 0.3, 50.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final shave n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = targeted_peak_shave(a_norm, t8, seed=99999)
            try_update(a_out, "shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
