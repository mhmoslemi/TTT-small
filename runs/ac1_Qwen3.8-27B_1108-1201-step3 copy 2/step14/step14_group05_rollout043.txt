    n = len(a)
    w_rev = w[::-1]

    """Compute LSE weights: w[k] = exp(autocorr[k]/T) / sum(exp(autocorr[l]/T))"""
    shifted = (autocorr - autocorr.max()) / T

def mirror_descent_step(a, grad, eta):
    """Entropy mirror descent step: a * exp(-eta*grad), renormalized"""
    # Project gradient onto sum=0 hyperplane for scale invariance

    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-15:

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _compute_grad(a, w):
    """g[j] = 2 * sum_k w[k] * a[k-j] for valid k-j in [0,n-1]."""
    n = len(a)
    w_rev = w[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_rev, fft_size)
    conv = irfft(A * W, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _lse_weights(autoc, T):

def _mirror_step(a, grad, eta):

    return new_a * (a.sum() / s)

        a_new = _mirror_step(a, grad, eta)

            best_a = a.copy()

def _peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving: move mass from peak contributors to non-contributors."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

        autoc = _fast_autocorr(a)

        K = min(3, 2 * n - 1)

            je = min(n - 1, p)

            if len(idx) == 0:
                continue
            contrib = a[idx] * a[p - idx]
            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:

            non_contrib[src] = True
            nc_idx = np.where(non_contrib)[0]
            if len(nc_idx) == 0:
                nc_idx = np.array([i for i in range(n) if i != src])

            # Try moving to lowest-value non-contributing positions
            n_try = min(5, len(nc_idx))
            sorted_dst = nc_idx[np.argsort(a[nc_idx])][:n_try]

            best_score = max_M
            best_move = None
            for dst in sorted_dst:
                for frac in [0.1, 0.2, 0.3]:
                    amt = src_val * frac
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st < 1e-15:
                        continue
                    trial = trial * norm / st
                    sc = _fast_autocorr(trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_move = (src, dst, frac)

            if best_move is not None:
                src_m, dst_m, frac_m = best_move
                a_tmp = a.copy()
                a_tmp[src_m] -= a[src_m] * frac_m
                a_tmp[dst_m] += a[src_m] * frac_m
                a_tmp = np.maximum(a_tmp, 0)
                st = a_tmp.sum()
                a_tmp = a_tmp * norm / st
                sc = _fast_autocorr(a_tmp).max()
                if sc < best_M:
                    a = a_tmp
                    best_M = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    break
                else:
                    # Annealing acceptance
                    T = best_M * 0.002 * (1.0 - elapsed_frac)
                    if rng.random() < np.exp(-(sc - best_M) / max(T, 1e-10)):
                        a = a_tmp
                        n_stall += 1
                    else:
                        n_stall += 1
            else:
                n_stall += 1

        # Random exploration when stuck
        if n_stall > 30:
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.15)
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial = trial * norm / st
                        sc = _fast_autocorr(trial).max()
                        if sc < best_M:
                            a = trial
                            best_M = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                            break
                        elif sc < best_M * 1.01:
                            T = best_M * 0.003
                            if rng.random() < np.exp(-(sc - best_M) / max(T, 1e-10)):
                                a = trial
            n_stall = 0

        if n_stall > 200:
            # Large perturbation
            idx_pert = rng.choice(n, size=max(10, n // 50), replace=False)
            trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                trial[idx] -= amt
                trial[j] += amt
            trial = np.maximum(trial, 0)
            st = trial.sum()
            if st > 1e-15:
                trial = trial * norm / st
                sc = _fast_autocorr(trial).max()
                if sc < best_M * 1.05:
                    a = trial
                    if sc < best_M:
                        best_M = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            print(f"    PeakShave improve={n_improve} ({time.monotonic()-t0:.0f}s): best_M={best_M:.7f}")

    return best_a, best_M


def _interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def _multiscale_init(n_target, time_budget):
    """Multi-scale entropy mirror descent initialization."""
    scales = [64, 128, 256, 512, 1024]
    scales = [s for s in scales if s < n_target if s >= 32]
    if n_target not in scales and len(scales) < 6:
        scales.append(n_target)
    elif n_target < scales[-1]:
        scales = [s for s in scales if s <= n_target]
        scales.append(n_target)

    t0 = time.monotonic()
    a = None

    for n in scales:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        # Time per scale: proportional to n
        alloc = min(remaining * 0.4, 0.05 * n ** 0.7)
        alloc = max(alloc, 2.0)

        if a is None:
            # Initialize with a smooth bump
            x = np.linspace(0, 1, n)
            a = np.sin(np.pi * x) ** 1.5
            a = np.maximum(a, 0.01)
            a = a / a.sum()
        else:
            a = _interpolate_seq(a, n)

        # Optimize at this scale using mirror descent
        norm = np.sqrt(2.0 * n)
        a = a * norm / a.sum()
        a, _ = _mirror_descent_optimize(a, alloc, verbose=False)
        print(f"    Multi-scale: n={n}, M={_fast_autocorr(a).max():.7f}")

    # Interpolate to target
    if a is not None:
        a = _interpolate_seq(a, n_target)
    else:
        a = np.ones(n_target) / n_target

    return a


def _hybrid_optimize(a0, time_limit, solver, verbose=True):
    """
    Hybrid optimizer: alternates between mirror descent (fast) and LP Nesterov (accurate).
    Uses adaptive switching based on convergence rate.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive parameters
    eta = 1.0 / n  # mirror descent step size
    lp_every = 15  # do LP every 15 mirror steps
    it = 0
    last_improve = 0
    prev_best = best_M

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)
        it += 1

        if it - last_improve > 2000:
            print(f"    Hybrid: no improvement in 2000 iters, adjusting...")
            # Increase exploration
            eta *= 2.0
            eta = min(eta, 5.0 / n)
            lp_every = max(5, lp_every // 2)
            last_improve = it

        # Mirror descent steps
        steps_this_round = lp_every
        for _ in range(steps_this_round):
            if time.monotonic() - t0 >= time_limit:
                break

            autoc = _fast_autocorr(a)
            M_curr = autoc.max()

            # Adaptive temperature
            T = M_curr * max(0.003, 0.03 * (1.0 - elapsed_frac) ** 1.5)
            w = _lse_weights(autoc, T)
            grad = _compute_grad(a, w)

            # Mirror step with backtracking
            a_new = _mirror_step(a, grad, eta)
            M_new = _fast_autocorr(a_new).max()

            if M_new <= M_curr * 1.001:
                a = a_new
                eta = min(eta * 1.05, 5.0 / n)
            else:
                # Try smaller step
                a_new2 = _mirror_step(a, grad, eta * 0.3)
                M_new2 = _fast_autocorr(a_new2).max()
                if M_new2 < M_curr:
                    a = a_new2
                eta *= 0.7

            M_check = _fast_autocorr(a).max()
            if M_check < best_M:
                best_M = M_check
                best_a = a.copy()
                last_improve = it

        # Periodic LP correction
        if it % lp_every == 0 or it == 1:
            if time.monotonic() - t0 >= time_limit:
                break
            autoc = _fast_autocorr(a)
            # Use more active constraints for LP
            max_active = min(50, 2 * n - 1)
            active_idx = np.argsort(autoc)[::-1][:max_active]
            b = _lp_best_response(a, autoc, active_idx, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    g = b * norm / b_sum
                    # Blend with current
                    t_blend = 0.3
                    a_blend = (1 - t_blend) * a + t_blend * g
                    a_blend = np.maximum(a_blend, 0)
                    sb = a_blend.sum()
                    if sb > 1e-15:
                        a_blend = a_blend * norm / sb
                        M_blend = _fast_autocorr(a_blend).max()
                        if M_blend < best_M:
                            a = a_blend
                            best_M = M_blend
                            best_a = a.copy()
                            last_improve = it

        if verbose and it % 100 == 0:
            print(f"    Hybrid it={it} ({elapsed:.0f}s): best_M={best_M:.7f}, eta={eta:.2e}, lp_every={lp_every}")

    return best_a, best_M


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-scale init + hybrid optimization + peak shaving."""
    global height_sequence_1  # noqa: F821 (will be provided externally)

    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    n_base = 1600

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create initial
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Multi-scale initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-scale init (budget={remaining:.0f}s) ===")
    ms_budget = min(remaining * 0.08, 80.0)
    if ms_budget > 10:
        ms_a = _multiscale_init(n_base, ms_budget)
        update_best(ms_a, "multiscale")

    # === Phase 2: Hybrid optimization (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Hybrid opt n={n_base} (budget={remaining:.0f}s) ===")
    h1_budget = min(remaining * 0.35, 500.0)
    if h1_budget > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _hybrid_optimize(a_norm, h1_budget, solver, verbose=True)
        update_best(a_out, f"hybrid1_n{n_cur}")

    # === Phase 3: LP Nesterov (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    lp1_budget = min(remaining * 0.20, 280.0)
    if lp1_budget > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, lp1_budget, solver, verbose=True,
                                         max_active=30, restart_every=200)
        update_best(a_out, f"lp1_n{n_cur}")

    # === Phase 4: Mirror descent (complementary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Mirror descent n={n_cur} (budget={remaining:.0f}s) ===")
    md_budget = min(remaining * 0.10, 120.0)
    if md_budget > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _mirror_descent_optimize(a_norm, md_budget, verbose=True)
        update_best(a_out, f"mirror_n{n_cur}")

    # === Phase 5: Hybrid round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Hybrid r2 n={n_cur} (budget={remaining:.0f}s) ===")
    h2_budget = min(remaining * 0.15, 180.0)
    if h2_budget > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _hybrid_optimize(a_norm, h2_budget, solver, verbose=True)
        update_best(a_out, f"hybrid2_n{n_cur}")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    for n_try in [n_base - 50, n_base + 50]:
        if remaining < 60:
            break
        if n_try < 100:
            continue
        print(f"\n  Trying n={n_try}...")
        a_interp = _interpolate_seq(best_seq, n_try)
        a_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()
        t_alloc = min(remaining * 0.08, 80.0)
        a_out, _ = _hybrid_optimize(a_norm, t_alloc, solver, verbose=False)
        update_best(a_out, f"hybrid_n{n_try}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Peak shaving n={n_cur} (budget={remaining:.0f}s) ===")
    ps_budget = min(remaining * 0.20, 150.0)
    if ps_budget > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, ps_budget, seed=777, verbose=True)
        update_best(a_out, f"peak1_n{n_cur}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    lp2_budget = min(remaining * 0.30, 200.0)
    if lp2_budget > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, lp2_budget, solver, verbose=True,
                                         max_active=25, restart_every=150)
        update_best(a_out, f"lp2_n{n_cur}")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final peak shave n={n_cur} (budget={remaining:.0f}s) ===")
    ps2_budget = min(remaining * 0.40, 150.0)
    if ps2_budget > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, ps2_budget, seed=31415, verbose=True)
        update_best(a_out, f"peak2_n{n_cur}")

    # === Phase 10: Last chance hybrid ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 10: Last hybrid n={n_cur} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _hybrid_optimize(a_norm, remaining * 0.8, solver, verbose=True)
        update_best(a_out, f"hybrid_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
