def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_grad(a, T):
    """
    Compute gradient of T * logsumexp(conv(a,a) / T) w.r.t. a.
    This is a smooth approximation of max_k conv_k.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    
    # Softmax weights (numerically stable)
    conv_shift = conv - conv.max()
    weights = np.exp(conv_shift / T)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(size)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    
    # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) for valid indices
    # term2[i] = (weights * a)_i  [linear convolution, take first n elements]
    # term1[i] = (flip(weights) * a)_{2n-2-i}
    
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    
    conv_wa = irfft(W * A, fft_size)  # length 3n-2
    term2 = conv_wa[:n]
    
    w_flip = weights[::-1]
    Wf = rfft(w_flip, fft_size)
    conv_wf_a = irfft(Wf * A, fft_size)
    term1 = conv_wf_a[2 * n - 2 : 2 * n - 2 + n : -1]  # indices 2n-2, 2n-3, ..., n-1
    # Wait, that's wrong. I need conv_wf_a[2n-2-i] for i=0,...,n-1
    # So indices: 2n-2, 2n-3, ..., 2n-2-(n-1) = n-1
    term1 = conv_wf_a[n-1 : 2 * n - 1][::-1]
    
    # Wait: conv_wf_a[2n-2-i] for i=0 gives conv_wf_a[2n-2]
    # for i=n-1 gives conv_wf_a[2n-2-(n-1)] = conv_wf_a[n-1]
    # So it's conv_wf_a[n-1 : 2n-1][::-1]
    # Which is conv_wf_a[2n-2], conv_wf_a[2n-3], ..., conv_wf_a[n-1]
    
    grad = term1 + term2
    return grad


def smooth_obj(a, T):
    """Compute T * logsumexp(conv(a,a) / T)."""
    n = len(a)
    conv = fast_autocorr(a)
    conv_shift = conv - conv.max()
    return T * (np.log(np.sum(np.exp(conv_shift / T))) + conv.max() / T)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - target_sum) / rho_hat
    return np.maximum(v - theta, 0.0)


def fista_smooth(a0, T_start, T_end, n_iters, proj_sum=1.0, verbose=False, t0=None, time_limit=None):
    """
    FISTA on the smooth approximation T*logsumexp(conv/T) with simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * proj_sum / n
    else:
        a = a * proj_sum / s
    
    y = a.copy()
    z = a.copy()
    t_k = 1.0
    best_a = a.copy()
    
    # Compute initial Lipschitz constant estimate
    # The Hessian of T*logsumexp(conv/T) is bounded by... let's use conservative estimate
    # Actually, let's use backtracking line search instead
    
    if t0 is None:
        t0 = time.monotonic()
    
    for k in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        frac = k / max(n_iters - 1, 1)
        T_curr = T_start * (T_end / T_start) ** frac
        
        # Gradient at y
        grad = smooth_grad(y, T_curr)
        
        # FISTA step with backtracking
        step = 0.01
        z_old = z.copy()
        for _ in range(50):
            z_new = y - step * grad
            z_new = project_simplex(z_new, proj_sum)
            # Check descent (Wolfe-ish condition)
            f_new = smooth_obj(z_new, T_curr)
            f_y = smooth_obj(y, T_curr)
            if f_new <= f_y - 0.5 * step * np.dot(grad, z_new - y):
                break
            step *= 0.5
        
        # Update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = z_new + ((t_k - 1.0) / t_new) * (z_new - z_old)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = z_new.copy()
        else:
            y = y * proj_sum / s_y
        t_k = t_new
        z = z_new
        
        # Track best
        f_z = smooth_obj(z, T_curr)
        if f_z < smooth_obj(best_a, T_curr):
            best_a = z.copy()
        
        if verbose and k % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * f_z
            print(f"    FISTA k={k} T={T_curr:.4f} ({elapsed:.0f}s): obj={score:.8f}")
    
    return best_a
