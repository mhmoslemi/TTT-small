def build_sparse_T(a, n):
    """Build sparse convolution constraint matrix T where (T@b)_k = (a*b)_k."""
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        rows.extend(idx)
        cols.extend([i] * len(idx))
        vals.extend(a[idx - i])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response_sparse(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_sparse_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    b_clipped = np.maximum(b, eps)
    b_p = b_clipped ** p
    sum_bp = b_p.sum()
    if sum_bp < 1e-30:
        return 1e10, np.zeros(n)
    # Objective: (sum b^p)^(1/p) / s^2
    obj_val = sum_bp ** (1.0 / p) / (s * s)
    # d(obj)/da_i = (1/p) * sum_bp^(1/p - 1) * p * b^(p-1) * (db/da_i) / s^2
    #              - 2 * (sum_bp)^(1/p) * a_i / s^3
    # where db_k/da_i = a_{k-i} + a_{k+n-i} (for the full convolution)
    # But since we only have the first n values of b contributing to gradient:
    # db/da_i = sum_k b_k^(p-1) * p * (T[k,i])
    # Use FFT-based cross-correlation
    p_weights = b_clipped ** (p - 1)
    # db_k / da_i for k in 0..2n-2: contribution is a_{k-i} if valid
    # We need: grad_i = sum_k p * b_k^(p-1) * T[k,i]
    # T[k,i] = a[k-i] if 0<=k-i<n
    # So grad_i = p * sum_k b_k^(p-1) * a[k-i] (valid range)
    # This is a cross-correlation between p_weights (on 2n-1 grid) and a (on n grid)
    fft_size = 1 << (2 * n - 1).bit_length()
    pw_full = np.zeros(fft_size)
    pw_full[:2 * n - 1] = p_weights
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    PW = rfft(pw_full, fft_size)
    AF = rfft(a_full, fft_size)
    # Cross-correlation: sum_k pw_full[k] * a_full[k-i]
    # = ifft(conj(AF) * PW) ... let me think more carefully
    # We want c_i = sum_k pw_full[k] * a_full[k-i]
    # This is the cross-correlation of pw_full with a_full
    # c = ifft(conj(A_full) * P_full) gives c_i = sum_k a_full[k] * conj(pw_full[k-i])
    # Actually for real signals: corr[i] = sum_k a[k] * pw[k+i] = ifft(conj(AF) * PW)[i]
    # We want: sum_k pw[k] * a[k-i] = sum_k pw[k+i] * a[i] ... hmm
    # Let me use: c_i = sum_k pw_full[k] * a_full[k-i]
    # Define r = flip(a_full), then c_i = sum_k pw_full[k] * r[n-1-(k-i)]
    # This is getting complicated. Let me just use the direct approach with FFT:
    # c_i = sum_k pw[k] * a[k-i] = (pw * a_flipped)[i + n - 1] where * is convolution
    # Actually: sum_k pw[k] * a[k-i] = (pw convolved with flip(a))[i]
    a_flipped = a_full[::-1]
    A_flipped_fft = rfft(a_flipped, fft_size)
    cross = irfft(PW * A_flipped_fft, fft_size)
    # cross[i] = sum_k pw[k] * a_flipped[k-i] = sum_k pw[k] * a_full[n-1-(k-i)]
    # = sum_k pw[k] * a_full[n-1-k+i]
    # Let j = k-i: = sum_j pw[j+i] * a_full[n-1-j]
    # Hmm, let me just do it differently.
    # We want g_i = sum_{k=0}^{2n-2} pw[k] * T[k,i] where T[k,i] = a[k-i] if 0<=k-i<n
    # = sum_{k} pw[k] * a[k-i] (with a indexed 0..n-1)
    # = (pw * a)[i] where * is cross-correlation (not convolution)
    # Cross-correlation: (pw ⋆ a)[i] = sum_k pw[k] * a[k-i]
    # In FFT: ifft(FFT(pw) * conj(FFT(a))) ... no
    # (f ⋆ g)[n] = sum_k f[k] g[k-n]
    # FFT(f ⋆ g) = conj(FFT(f)) * FFT(g) ... no, that's for convolution
    # Actually: FFT(f ⋆ g) = conj(FFT(f)) * FFT(g) is wrong
    # Let me use: (f ⋆ g)[n] = sum_k f[k+n] g[k] = (flip(f) * g)[n] where * is conv
    # So: fft(f ⋆ g) = fft(flip(f)) * fft(g)
    # flip(f) = f[::-1], fft(flip(f)) = conj(fft(f)) (for real f, with appropriate length)
    # Actually for real f: fft(flip(f)) = conj(fft(f)) only if we include the [0] term properly
    # Let me just use the direct formula: cross_corr = irfft(conj(PW) * AF, fft_size)
    # This gives: c[i] = sum_k pw[k] * a[(k-i) mod fft_size]
    # For our purposes (with zero padding), this should work for i in 0..n-1
    cross = irfft(np.conj(PW) * AF, fft_size)[:n]
    # cross[i] = sum_k pw[k] * a[(k-i) mod fft_size]
    # For 0 <= i < n and k in 0..2n-2: if k-i is in 0..n-1, a[k-i] is correct
    # If k-i < 0 or k-i >= n, a[k-i] should be 0 (zero padding handles this)
    
    sum_bp_pow = sum_bp ** (1.0 / p - 1.0)
    grad = p * sum_bp_pow * cross / (s * s) - 2.0 * sum_bp ** (1.0 / p) * a / (s ** 3)
    return obj_val, grad


def pnorm_scipy_optimize(a0, time_limit, p_values=(20, 50, 100)):
    """Optimize using p-norm smoothed objective with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for p in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def obj_func(x):
            val, _ = pnorm_objective_and_grad(x, n, p)
            return val

        def grad_func(x):
            _, g = pnorm_objective_and_grad(x, n, p)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-8, 'ftol': 1e-12,
                                           'maxls': 50})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        print(f"    p-norm p={p}: score={score_c:.8f}")
        except Exception as e:
            print(f"    p-norm p={p} error: {e}")

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    """Log-sum-exp smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    log_sum = max_b + np.log(exp_sum) / T_temp
    w = exp_vals / exp_sum

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(np.conj(W_fft) * A_fft, fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_scipy_optimize(a0, time_limit, T_start=60.0, T_end=2.0, n_stages=4):
    """LSE smoothed optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-8, 'ftol': 1e-12,
                                           'maxls': 50})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_val / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def continuation_lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP-Nesterov with continuation: start loose, tighten M aggressively."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    last_improve = t0
    stall_count = 0
    n_iters = 0

    y = a.copy()
    ema_a = a.copy()
    ema_beta = 0.95

    # Continuation: start with larger eps (looser), decrease over time
    eps_start = 0.08
    eps_end = 0.01
    n_cont = 30  # number of continuation steps

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Continuation: interpolate eps from start to end
        cont_frac = min(1.0, t / float(n_cont * 50))
        eps = eps_start + (eps_end - eps_start) * cont_frac
        # Add some oscillation to help escape
        eps_eff = eps * (1.0 + 0.3 * np.sin(t * 0.1))
        eps_eff = max(0.001, min(0.15, eps_eff))
        M_used = M_curr * (1.0 - eps_eff)

        T = build_sparse_T(y, n)
        b = lp_best_response_sparse(y, M_used, solver, T)
        if b is None:
            b = lp_best_response_sparse(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov step size
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)
        alpha = min(alpha, 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # EMA tracking
        ema_a = ema_beta * ema_a + (1.0 - ema_beta) * a
        s_ema = ema_a.sum()
        if s_ema > 1e-15:
            ema_a = ema_a * norm_val / s_ema

        # Check all candidates
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        score_ema = fast_autocorr(ema_a).max()
        if score_ema < best_score:
            best_score = score_ema
            best_a = ema_a.copy()
            last_improve = time.monotonic()

        # Restart from best if stalled
        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 3:
                # Restart from best_a
                y = best_a.copy()
                a = best_a.copy()
                ema_a = best_a.copy()
                stall_count = 0
                last_improve = time.monotonic()
                if verbose:
                    print(f"    Restart at iter={n_iters}, best={best_score:.8f}")
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Cont iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eps={eps_eff:.4f}")

    return best_a, best_score


def gradient_guided_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent guided by p-norm gradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_peak = int(np.argmax(b))

        # Compute p-norm gradient for guidance (p=20)
        _, grad = pnorm_objective_and_grad(a, n, 20)

        # Find top contributing pairs at peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find elements with most negative gradient (should increase)
        neg_grad_idx = np.argsort(grad)[:20]
        pos_grad_idx = np.argsort(grad)[::-1][:20]

        # Strategy: move mass from high-gradient (should decrease) to low-gradient (should increase)
        improved = False

        # Try moving from peak contributors to negative-gradient positions
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in neg_grad_idx:
                if dst == src:
                    continue
                best_move_score = best_score
                best_move = None
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_move_score:
                        best_move_score = score
                        best_move = a_trial.copy()
                if best_move is not None and best_move_score < best_score - 1e-12:
                    a = best_move
                    best_score = best_move_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random moves guided by gradient
            for _ in range(120):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                # Bias towards moving from high-grad to low-grad
                if grad[i] < grad[j]:
                    i, j = j, i
                if a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = continuation_lp_nesterov(a_norm, t1, solver, verbose=True)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_cont_n{n_best}", t_global)

    # Phase 2: p-norm gradient optimization - 15% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: p-norm gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pnorm_scipy_optimize(a_norm, t2, p_values=(20, 50, 100))
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"pnorm_n{n_best}", t_global)

    # Phase 3: LSE gradient - 10% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=50.0, T_end=2.0, n_stages=4)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lse_n{n_best}", t_global)

    # Phase 4: Gradient-guided coordinate descent - 10% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Grad coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_guided_coord_descent(a_norm, t4, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"coord1_n{n_best}", t_global)

    # Phase 5: LP round 2 from improved point - 15% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = continuation_lp_nesterov(a_norm, t5, solver, verbose=False)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 6: Multi-length exploration - 10% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 60:
        # Try longer
        n_long = n_best + 300
        t6a = min(remaining * 0.5, 150.0)
        if t6a > 40:
            print(f"  Trying n={n_long}...")
            init_long = interpolate_sequence(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = continuation_lp_nesterov(a_norm, t6a, solver, verbose=False)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_long}", t_global)

        # Try shorter (sometimes shorter is better due to less boundary effect)
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 40:
            n_short = max(800, n_best - 200)
            t6b = min(remaining2 * 0.5, 100.0)
            if t6b > 30:
                print(f"  Trying n={n_short}...")
                init_short = interpolate_sequence(best_seq, n_short)
                a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
                a_out, _ = continuation_lp_nesterov(a_norm, t6b, solver, verbose=False)
                best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_short_n{n_short}", t_global)

    # Phase 7: Final polish - remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        # Short coord descent
        t7a = min(remaining * 0.4, 60.0)
        if t7a > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = gradient_guided_coord_descent(a_norm, t7a, seed=99999)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final1", t_global)

        # Short LSE
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = lse_scipy_optimize(a_norm2, min(remaining2 * 0.5, 80.0),
                                           T_start=30.0, T_end=1.0, n_stages=3)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lse_final", t_global)

        # Final coord descent
        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_best3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            a_out, _ = gradient_guided_coord_descent(a_norm3, remaining3, seed=777)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final2", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
