    return 2.0 * p * r[:n]

def softmax_normalize(v):
    """Exponentiate and normalize (with overflow protection)."""
    v = v - v.max()
    v = np.exp(v)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def md_lp_optimize(a0, time_limit, p_start=4.0, p_end=200.0, anneal_iters=3000,
                   eta0=0.5, verbose=True, warmup=100):
    """Mirror descent with Lp-norm objective and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    y = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # p schedule: start at p_start, increase to p_end over anneal_iters
        if t <= anneal_iters:
            frac = (t - 1) / max(anneal_iters - 1, 1)
            p = p_start * (p_end / p_start) ** frac
        else:
            p = p_end
        
        # Compute gradient at y (Nesterov lookahead)
        c_y = fast_conv(y)
        c_y_pos = np.maximum(c_y, 0.0)
        if p > 1:
            d = c_y_pos ** (p - 1)
        else:
            d = np.ones_like(c_y)
        
        N = 1 << (len(c_y) + n - 1).bit_length()
        D = rfft(d, N)
        A_f = rfft(y, N)
        r = irfft(D * np.conjugate(A_f), N)
        grad = 2.0 * p * r[:n]
        
        # Step size with warmup
        if t <= warmup:
            eta = eta0 * t / warmup  # ramp up
        else:
            eta = eta0
        
        # Actually, let me use a decreasing step: eta = eta0 / (1 + 0.01*t)
        # Or adaptive: eta = eta0 / (t ** 0.5)
        eta = eta0 / (1.0 + 0.005 * t)
        
        # Mirror descent step from y
        log_step = np.log(y + 1e-30) - eta * grad
        a_new = softmax_normalize(log_step)
        
        # Nesterov momentum
        if t == 1:
            y_new = a_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y_new = a_new + beta * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        # Track best
        M_a = fast_conv(a).max()
        if M_a < best_m:
            best_m = M_a
            best_a = a.copy()
        M_y = fast_conv(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t}, p={p:.1f} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def subgradient_polish(a0, time_limit, top_k=10, verbose=True):
    """Direct subgradient method on the max with top-k smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    y = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute autocorrelation at y
        c_y = fast_conv(y)
        M = c_y.max()
        
        # Find top-k peaks
        k_max = min(top_k, len(c_y))
        top_idx = np.argsort(c_y)[::-1][:k_max]
        top_vals = c_y[top_idx]
        
        # Soft weights: exponential of (val / M)
        w = np.exp((top_vals - M) / (0.1 * M + 1e-30))
        w = w / w.sum()
        
        # Compute weighted subgradient
        # grad[i] = 2 * sum_k w_k * y[k-i]
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j = k_int - idx
            grad[idx] += 2.0 * wi * y[j]
        
        # Step size
        eta = 0.3 / (1.0 + 0.01 * t)
        
        # Mirror descent step
        log_step = np.log(y + 1e-30) - eta * grad
        a_new = softmax_normalize(log_step)
        
        # Nesterov
        if t == 1:
            y_new = a_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y_new = a_new + beta * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        M_a = fast_conv(a).max()
        if M_a < best_m:
            best_m = M_a
            best_a = a.copy()
        M_y = fast_conv(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        M = c.max()
        
        # Find highest peak
        k_peak = np.argmax(c)
        k_int = int(k_peak)
        
        # For this peak, find the pair (i, j=k-i) with max contribution
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_int - idx]
        
        if contrib.max() < 1e-30:
            stall += 1
            continue
        
        peak_i = idx[np.argmax(contrib)]
        
        # Try moving mass from peak_i to a random other position
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            dst = rng.integers(n)
            if dst == peak_i:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[peak_i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[peak_i] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    break
                else:
                    break  # try smaller frac? No, if this doesn't work, break
        
        if stall > 0:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD stall={stall} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving shape and sum=1."""
    n = len(a)
    if n == new_n:
        return a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a = a / s
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s_new
