Actually, one more important consideration: the LP `max 1^T b s.t. T@b ≤ M, b ≥ 0` has a special structure. The matrix T is the "autocorrelation matrix" of a, and the right-hand side is all M. This means we're looking for the "maximum throughput" signal that can pass through a "channel" with capacity M at each output.

An alternative to the full LP: use the **multiplicative weights update (MWU)**. The MWU for this problem is:
- Initialize b = 1/n
- For t = 1, 2, ...:
  - Compute violations: v_k = max(0, (T@b)_k - M)
  - Update: b_j ← b_j * (1 - η * sum_k v_k * T[k,j] / M) for each j
  - Normalize: b ← b / sum(b)

This is O(n^2) per iteration (to compute T@b and the update), but doesn't require an LP solver. With a good learning rate η, it converges to an approximate solution.

The MWU gives an (1-ε)-approximate solution after O(1/ε^2) iterations. For our purposes, even a rough approximation (ε = 0.1) might be enough as a direction for Frank-Wolfe.

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum-1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgrad(a, ac, top_k=10):
    """Compute subgradient of max(a*a) using top-k active constraints."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Use softmax weighting
    max_val = top_vals.max()
    w = np.exp(top_vals - max_val)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad[j] += 2 * wi * a[k-j] if 0 <= k-j < n
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            grad[js] += 2.0 * wi * a[k_int - js]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_k_lp=200):
    """
    Nesterov-accelerated Frank-Wolfe with LP best response.
    Uses reduced constraints (top-K) for speed.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov momentum point
    x = a.copy()  # Current iterate
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break
        
        # Build reduced T matrix (top-K constraints)
        K = min(top_k_lp, 2 * n - 1)
        top_constr = np.argsort(ac)[::-1][:K]
        
        T = np.zeros((K, n))
        for idx, k in enumerate(top_constr):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                T[idx, j_start:j_end] = a[k_int - np.arange(j_start, j_end)]
        
        # Solve LP: max 1^T b s.t. T@b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check improvement
        My = fast_autocorr(y).max()
        if My < best_M:
            best_M = My
            best_a = y.copy()
        
        Mx = fast_autocorr(x).max()
        if Mx < best_M:
            best_M = Mx
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def mirror_descent_optimize(a0, time_limit, verbose=True):
    """Mirror descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    n_avg = 1
    
    # Adaptive parameters
    beta = 2.0  # Temperature (start low = sharp, increase for exploration)
    gamma = 1.0  # Step size scale
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        # Adaptive top_k: start with fewer, increase over time
        top_k = min(5 + t // 100, 30)
        
        # Compute soft subgradient
        g = compute_subgrad(a, ac, top_k=top_k)
        
        # Mirror descent update (multiplicative)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 500 == 0:

            J = 2.0 * n * best_M
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M
