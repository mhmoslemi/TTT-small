def build_conv_matrix(a):

def lp_best_response(a, M, solver, max_active=None):
    """Solve the LP: max sum(b) s.t. T(a)b <= M, b >= 0"""
    import cvxpy as cp

    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if max_active is None:
        constraints = [T @ b_var <= M]
    else:

        if max_active < size:
            # Keep the most binding constraints
            idx = np.argsort(row_sums)[::-1][:max_active]
            constraints = [T[idx, :] @ b_var <= M[idx]]
        else:
            constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        return None

    if b_var.value is None:

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    return b_val

    t0 = time.monotonic()

        M_curr = fast_autocorr(y).max()

def mirror_descent(a0, time_limit, verbose=True):
    """Exponential (mirror) descent for non-negative optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 0.1  # step size
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Subgradient for the active constraint
        # db[p_star]/da[j] = 2*a[p_star-j] for valid j (when j != p_star-j)
        # Simplified: the gradient direction to reduce b[p_star]
        grad = np.zeros(n)
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[p_star - idx]
        
        # Also include secondary peaks
        sorted_idx = np.argsort(b)[::-1][:3]
        for p in sorted_idx:
            if p == p_star:
                continue
            weight = 0.3 * (b[p] / max_b)
            js2 = max(0, p - n + 1)
            je2 = min(n - 1, p)
            idx2 = np.arange(js2, je2 + 1)
            grad[idx2] += weight * 2.0 * a[p - idx2]
        
        # Project gradient to tangent space (sum constraint)
        grad -= np.mean(grad)
        
        # Mirror descent update (exp form)
        # a_new = a * exp(-eta * grad) / sum(a * exp(-eta * grad))
        exponent = -eta * grad
        # Clip for numerical stability
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a_new = np.maximum(a_new, 0.0)
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            eta = min(eta * 1.1, 1.0)
        else:
            eta *= 0.95
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent step={steps} ({elapsed:.0f}s): best={best_score:.8f}, max_b={max_b:.6f}, eta={eta:.4f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            cur_max_active = max(80, max_active)

        b = lp_best_response(y, M_curr, solver, max_active=cur_max_active)
        if b is None:
            n_iters += 1
            stall += 1
            if stall > 40:
                _structured_restart(y, a, prev_y, n, norm, rng)
                stall = 0
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            n_iters += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm / max(s_y, 1e-15)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm / max(s_a, 1e-15)

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, stall={stall}")

        if stall > 150:
            _structured_restart(y, a, prev_y, n, norm, rng)
            stall = 0

    return best_a, best_score

def _structured_restart(y, a, prev_y, n, norm, rng):
    """Structured restart using sinusoidal and frequency-aware perturbations."""
    strategy = rng.integers(3)
    if strategy == 0:
        # Low-frequency sinusoidal modulation
        freq = rng.uniform(1.0, 6.0)
        x = np.linspace(0, 2 * np.pi, n)
        phase = rng.uniform(0, 2 * np.pi)
        perturb = 0.04 * np.sin(freq * x + phase)
        y = np.maximum(y * (1.0 + perturb), 0)
        y = y * norm / y.sum()
    elif strategy == 1:
        # Frequency-domain perturbation
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(y, fft_size)
        # Add noise to high frequencies
        mask = np.zeros(len(A), dtype=bool)
        n_high = len(A) // 4
        mask[n_high:] = True
        A[mask] += rng.normal(0, 0.01 * np.max(np.abs(A)), np.sum(mask))
        y = irfft(A, fft_size)[:n]
        y = np.maximum(y, 0)
        y = y * norm / y.sum()
    else:
        # Random component swap
        n_swap = max(3, n // 30)
        idx = rng.choice(n, size=n_swap, replace=False)
        y_t = y.copy()
        for i in idx:
            j = rng.integers(n)
            if i != j:
                amt = y[i] * rng.uniform(0.1, 0.4)
                y_t[i] -= amt
                y_t[j] += amt
        y_t = np.maximum(y_t, 0)
        y = y_t * norm / max(y_t.sum(), 1e-15)

    a = np.maximum(a + 0.3 * (y - a), 0)
    a = a * norm / max(a.sum(), 1e-15)
    prev_y[:] = y

def mirror_descent(a0, time_limit, verbose=True):
    """Exponential (mirror) descent for non-negative optimization with multi-peak guidance."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 0.15

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Multi-peak subgradient with softmax weighting
        K = min(5, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:K]
        peak_vals = b[sorted_idx]
        T_param = max(0.05 * max_b, 1e-10) * (1.0 - 0.5 * progress)
        w = np.exp((peak_vals - max_b) / T_param)
        w = w / w.sum()

        grad = np.zeros(n)
        for m, p in enumerate(sorted_idx):
            if w[m] < 1e-6:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[m] * a[p - idx]

        # Project to tangent space
        grad -= np.mean(grad)

        # Mirror descent update
        exponent = -eta * grad
        exponent = np.clip(exponent, -40, 40)
        a_new = a * np.exp(exponent)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a_new = np.maximum(a_new, 0.0)

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            eta = min(eta * 1.05, 0.8)
        else:
            eta *= 0.97

        if verbose and steps % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror step={steps} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score

def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with multi-destination line search and annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    n_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing pair with max product
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 300:
                break
            continue

        src = idx[np.argmax(contrib)]

        # Find non-contributing positions
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True

        candidates = np.where(non_contrib)[0]
        if len(candidates) == 0:
            candidates = np.array([i for i in range(n) if i != src])

        # Try multiple destinations
        n_try = min(5, len(candidates))
        sorted_dst = candidates[np.argsort(a[candidates])][:n_try]

        best_trial = max_b
        best_move = None

        for dst in sorted_dst:
            for frac in [0.1, 0.2, 0.3, 0.4]:
                amt = a[src] * frac
                if amt < 1e-18:
                    break
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                a_t = a_t * norm / a_t.sum()
                sc = fast_autocorr(a_t).max()
                if sc < best_trial:
                    best_trial = sc
                    best_move = (src, dst, frac)

        if best_move is not None:
            src, dst, frac = best_move
            a_t = a.copy()
            a_t[src] -= a[src] * frac
            a_t[dst] += a[src] * frac
            a_t = np.maximum(a_t, 0)
            a_t = a_t * norm / a_t.sum()
            a = a_t
            best_score = best_trial
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        else:
            n_stall += 1

        # Annealing-based exploration
        if n_stall > 30:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                amt = a[i] * rng.uniform(0.05, 0.25)
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = np.maximum(a_t, 0)
                a_t = a_t * norm / a_t.sum()
                sc = fast_autocorr(a_t).max()
                T = best_score * 0.002 * (1.0 - elapsed_frac)
                if sc < best_score:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                elif T > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T):
                    a = a_t
                    n_stall = 0
                else:
                    n_stall += 1

        if n_stall > 250:
            # Larger structured perturbation
            n_pert = max(5, n // 30)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_t = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_t[idx] -= amt
                a_t[j] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t * norm / a_t.sum()
            sc = fast_autocorr(a_t).max()
            if sc < best_score * 1.02:
                a = a_t
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Hybrid optimization: LP-Nesterov + mirror descent + peak shaving
    with adaptive time allocation and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 450.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, max_active=200, gamma=1.8)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: MirrorDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # === Phase 3: LP Nesterov with different gamma ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov gamma=1.5 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 220.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, max_active=150, gamma=1.5)
        do_try_update(a_out, f"lp_g15_n{n_cur}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t4, seed=999, verbose=True)
        do_try_update(a_out, f"peak1_n{n_cur}")

    # === Phase 5: Try different length ===
    n2 = n_base + 80
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True, max_active=150, gamma=1.8)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 6: LP Nesterov with all constraints (final push) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP all-constraints n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, max_active=None, gamma=2.0)
        do_try_update(a_out, f"lp_all_n{n_cur}")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t7, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
