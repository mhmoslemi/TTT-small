    q_flip_idx = np.arange(size - 1, -1, -1)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(30, int(remaining * 4))

        # FISTA parameters
        t_fista = 1.0
        a_prev = a.copy()

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            # Extrapolation
            if it == 0:
                y = a.copy()
            else:
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
                momentum = (t_fista - 1.0) / t_new
                y = a + momentum * (a - a_prev)
                y = np.maximum(y, 0.0)
                sy = y.sum()
                if sy < 1e-15:
                    y = a.copy()
                else:
                    y = y * norm / sy
                t_fista = t_new

            # Compute autocorrelation at y
            conv_y = fast_autocorr(y)
            max_conv = conv_y.max()

            # Softmax weights
            p = np.exp((conv_y - max_conv) / T)
            p_sum = p.sum()
            if p_sum < 1e-30:
                break
            p /= p_sum

            # Compute gradient: h[i] = (2/T) * sum_m y[m] * p[i+m]
            # Using: h[i] = (2/T) * (y*q)[2n-2-i] where q = p[::-1]
            q = p[q_flip_idx]
            y_pad = np.zeros(fft_size)
            y_pad[:n] = y
            q_pad = np.zeros(fft_size)
            q_pad[:size] = q
            conv_yq = irfft(rfft(y_pad, fft_size) * rfft(q_pad, fft_size), fft_size)

            # h[i] = conv_yq[2n-2-i] for i=0..n-1
            h = (2.0 / T) * conv_yq[2 * n - 2 - np.arange(n)]

            # Step size: conservative based on gradient magnitude
            grad_norm = np.linalg.norm(h)
            if grad_norm < 1e-15:
                a_new = a.copy()
            else:
                # Use a step that's related to the problem scale
                step = (norm / (grad_norm * np.sqrt(n))) * 0.8
                # Cap step to avoid instability
                step = min(step, 0.5 / (max_conv / T + 1e-10))
                a_new = y - step * h

            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a_prev = a.copy()
            a = a_new

            # Track best
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Softmax-FISTA T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that targets the dominant autocorrelation lags."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Find dominant lags (top K)
        K = min(10, size)
        top_lags = np.argsort(conv)[-K:]
        top_lags_set = set(top_lags.tolist())

        # Compute "pressure" on each index from dominant lags
        pressure = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    pressure[i] += a[i] * a[j]

        # Sort indices by pressure (high pressure first - candidates to reduce)
        # Low pressure indices are candidates to increase
        order = np.argsort(pressure)
        high_press = order[n // 2:]
        low_press = order[:n // 2]

        # Try moving mass from high-pressure to low-pressure indices
        n_attempts = min(n, 200)
        tried = 0
        rng_idx = 0

        while tried < n_attempts and time.monotonic() - t0 < time_limit:
            i = high_press[rng_idx % len(high_press)]
            j = low_press[(rng_idx * 7) % len(low_press)]
            rng_idx += 1
            tried += 1

            if i == j or a[i] < norm * 1e-8:
                continue

            # Try moving small amount from i to j
            # Compute new max if we do this
            delta_max = a[i] * 0.3
            best_d = 0.0
            best_val = conv.max()
            n_samples = 8
            deltas = np.linspace(0.0, delta_max, n_samples)

            for d in deltas:
                if d <= 0:
                    continue
                # Update conv efficiently
                # conv changes: remove contribution from a[i], add to a[j]
                # New conv = conv - d*a shifted by i + d*a shifted by j (approx)
                # More precisely:
                # (a_new * a_new)[k] where a_new[i] = a[i]-d, a_new[j] = a[j]+d
                # = conv[k] - d*(a[k-i] + a[i-k]) - d*(a[k-j]... ) 
                # This is complex; let's just recompute for the candidate
                a_test = a.copy()
                a_test[i] -= d
                a_test[j] += d
                a_test = a_test * norm / a_test.sum()
                new_conv = fast_autocorr(a_test)
                new_max = new_conv.max()
                if new_max < best_val - 1e-12:
                    best_val = new_max
                    best_d = d

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PTC pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_window(a, time_limit, rng, window=25, verbose=True):
    """Standard pair coordinate descent with window."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, min(window, n) + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i or a[i] < norm * 1e-10:
                        continue

                    # Try moving mass from i to j
                    best_d = 0.0
                    best_val = conv.max()
                    max_move = a[i] * 0.5
                    n_samples = 10
                    deltas = np.linspace(0.0, max_move, n_samples)

                    for d in deltas:
                        if d <= 0:
                            continue
                        a[i] -= d
                        a[j] += d
                        s_tmp = a.sum()
                        if s_tmp < 1e-15:
                            a[i] += d
                            a[j] -= d
                            continue
                        a_scaled = a * norm / s_tmp
                        new_max = fast_autocorr(a_scaled).max()
                        a[i] += d
                        a[j] -= d
                        if new_max < best_val - 1e-12:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_restart_polish(a0, time_limit, rng, solver, verbose=True):
    """Try random perturbations and re-optimize locally."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_restarts = min(20, max(3, int(time_limit / 5)))
    per_restart_time = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        # Perturb
        eps = 0.05 * (1.0 + 0.5 * r / n_restarts)
        noise = rng.uniform(1.0 - eps, 1.0 + eps, size=n)
        a_pert = a * noise
        a_pert = np.maximum(a_pert, 0.0)
        sp = a_pert.sum()
        if sp < 1e-15:
            continue
        a_pert = a_pert * norm / sp

        # Quick softmax FISTA on perturbed
        a_out, sc = softmax_fista(a_pert, per_restart_time * 0.7, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    Restart {r}: improved to {best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov - main workhorse (40% of total time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Softmax FISTA (25% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Softmax FISTA n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = softmax_fista(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sf_n{n_best}")

    # Phase 3: Peak-targeted CD (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Peak Targeted CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeted_cd(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"ptc_n{n_best}")

    # Phase 4: Pair CD polish (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t4, rng, window=20, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 5: Random restart + local optimization (10% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Random Restart n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_restart_polish(a_norm, t5, rng, solver, verbose=True)
        do_try_update(a_out, f"rr_n{n_best}")

    # Phase 6: Second LP Nesterov round (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.15
    print(f"\n=== Phase 6: LP Nesterov round 2 n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 7: Final Softmax FISTA + CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t7a = remaining * 0.5
        t7b = remaining * 0.4
        print(f"\n=== Phase 7a: Final Softmax FISTA (budget={t7a:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = softmax_fista(a_norm, t7a, verbose=True)
        do_try_update(a_out, f"sf_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            n_best = len(best_seq)
            print(f"\n=== Phase 7b: Final Pair CD (budget={remaining:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_cd_window(a_norm, remaining * 0.8, rng, window=30, verbose=True)
            do_try_update(a_out, f"pcd_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
