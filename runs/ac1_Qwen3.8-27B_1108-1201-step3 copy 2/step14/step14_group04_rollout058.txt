def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lp_best_response(a, M, solver):

        prob.solve(solver=solver, verbose=False)

        return None

        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_nonneg_sum(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_fista(a0, time_limit, verbose=False):

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True, diversity_every=200):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()
    y = a.copy()

    # Sliding window for ergodic averaging
    win_size = 100
    window = np.zeros((win_size, n))
    window_idx = 0
    window_filled = 0

    t0 = time.monotonic()
    n_iters = 0
    rng = np.random.default_rng(12345)
    no_improve_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Update sliding window
        window[window_idx] = a
        window_idx = (window_idx + 1) % win_size
        window_filled = min(window_filled + 1, win_size)

        # Check scores
        prev_best = best_score
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Sliding window average check
        if n_iters % 8 == 0 and window_filled >= 20:
            if window_idx == 0:
                avg_a = window.mean(axis=0)
            else:
                avg_a = window[:window_idx].sum(axis=0) / (win_size - window_idx) + window[window_idx:].sum(axis=0) / (win_size - window_idx)
                # Actually simpler: just average all filled entries
                avg_a = window.mean(axis=0)
            s_avg = avg_a.sum()

                avg_a = avg_a * norm / s_avg
                score_avg = fast_autocorr(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()

        if n_iters % diversity_every == 0:
            # Small random perturbation
            pert = rng.standard_normal(n) * 0.01
            y_pert = y + pert
            y_pert = np.maximum(y_pert, 0.0)
            s_pert = y_pert.sum()
            if s_pert > 1e-15:
                y_pert = y_pert * norm / s_pert
                score_pert = fast_autocorr(y_pert).max()
                if score_pert < best_score:
                    best_score = score_pert
                    best_a = y_pert.copy()

        if best_score < prev_best:
            no_improve_count = 0
        else:
            no_improve_count += 1

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def polish_fft(a0, time_limit, verbose=True):
    """FFT-based projected subgradient polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    step_size = 0.3
    patience = 0
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        ac_max = ac.max()
        if ac_max < 1e-14:
            break

        # Soft active set weighting
        temp = max(ac_max * 0.003, 1e-10)
        weights = np.exp((ac - ac_max) / temp)
        weights = weights / weights.sum()

        # Compute subgradient via FFT cross-correlation
        W = rfft(weights, fft_size)
        A_rev = rfft(a[::-1], fft_size)
        cross = irfft(W * A_rev, fft_size)
        grad = 2.0 * cross[n - 1: n - 1 + n]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Try multiple step sizes
        improved = False
        for scale in [1.0, 0.5, 0.2, 0.1, 0.05]:
            effective_step = step_size * scale
            a_trial = a - effective_step * grad / grad_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                step_size = min(step_size * 1.05, 1.5)
                break

        n_iters += 1
        if not improved:
            step_size *= 0.8
            patience += 1
            if patience > 50:
                break
        else:
            patience = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        t6 = min(remaining * 0.5, 120.0)
        if t6 > 30:
            print(f"\n=== Phase 6: Final LP n={n_cur} (budget={t6:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t6, solver, verbose=False, diversity_every=500)
            try_update(a_out, "lp_final")

    # === Phase 7: Final polish with whatever time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 10:
        t7 = min(remaining * 0.4, 40.0)
        if t7 > 8:
            print(f"\n=== Phase 7: Final Polish n={n_cur} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = polish_fft(a_norm, t7, verbose=False)
            try_update(a_out, "polish_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: uniform sequence
        return [1.0 / 100] * 100

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
