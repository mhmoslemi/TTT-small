def lse_gradient(a, T):
    """Compute gradient of LSE(conv(a,a)/T) w.r.t. a.
    
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    # Shift to avoid overflow
    shifted = (conv - max_conv) / T
    w = np.exp(shifted)
    w = w / w.sum()
    # Cross-correlation
    grad = 2.0 / T * cross_corr_fft(a, w)
    return grad, w, conv


class AdamOptimizer:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, x, grad, bound_lo=0.0):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(x)
            self.v = np.zeros_like(x)
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad * grad
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        x_new = x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        x_new = np.maximum(x_new, bound_lo)
        return x_new

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b gives the full convolution of a and b."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: maximize sum(b) subject to T@a*b <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def single_element_cd(a, time_limit, rng, verbose=False):

def fast_autocorr(a):
    """Fast linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(a, b):
    """Cross-correlation: result[i] = sum_k b[k] * a[k-i] for valid k.
    a has length n, b has length 2n-1 (or padded).
    Returns vector of length n.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(B * np.conj(A), fft_size)[:n]

def adam_step(x, grad, state, lr, beta1, beta2, eps):
    """Single Adam step. State is a dict with m, v, t."""
    if 'm' not in state:
        state['m'] = np.zeros_like(x)
        state['v'] = np.zeros_like(x)
        state['t'] = 0
    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1.0 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1.0 - beta2) * grad * grad
    m_hat = state['m'] / (1.0 - beta1 ** t)
    v_hat = state['v'] / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return np.maximum(x_new, 0.0)

            if w_sum < 1e-30:
                w = np.zeros(2 * n - 1)
                w[n - 1] = 1.0
            else:
                w = w / w_sum

            grad = (2.0 / T) * cross_corr_fft(a, w)

            a_new = adam_step(a, grad, state, lr, 0.9, 0.999, 1e-8)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            if it % 40 == 0 or it == n_steps - 1:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                # Reset state occasionally to avoid stale momentum
                if it % 200 == 0 and it > 0:
                    state = {}

        if verbose and (si % 3 == 0 or si == len(temps) - 1):
            elapsed = time.monotonic() - t0
            score = fast_autocorr(a).max()
            print(f"    LSE-Adam T={T:.3f} ({elapsed:.0f}s): score={score:.8f}, best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T such that (T @ b)[k] = sum_j a[|k-j| related]... 
    Actually T @ b gives the full linear convolution of a and b."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov accelerated optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a, time_limit, rng, window=30, verbose=True):
    """Windowed pair coordinate descent."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            n_offsets = min(window, n // 3)
            if n_offsets < 1:
                continue
            n_try = min(4, n_offsets)
            offsets = rng.choice(n_offsets, size=n_try, replace=False) + 1

            for offset in offsets:
                j = (i + offset) % n
                if j == i:
                    continue
                if a[i] < norm * 1e-10:
                    break

                # Compute influence vectors
                inf_i = np.zeros(size)
                lo_i = i
                hi_i = min(size, i + n)
                inf_i[lo_i:hi_i] = 2.0 * a[:hi_i - lo_i]

                inf_j = np.zeros(size)
                lo_j = j
                hi_j = min(size, j + n)
                inf_j[lo_j:hi_j] = 2.0 * a[:hi_j - lo_j]

                delta_vec = inf_j - inf_i

                # Try a few values of d
                max_d = a[i] * 0.9
                if max_d < 1e-15:
                    continue
                best_d = 0.0
                best_max_val = conv.max()
                n_samples = 12
                for d in np.linspace(0.05 * max_d, max_d, n_samples):
                    new_conv = conv + d * delta_vec
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if i + j < size:
                        new_conv[i + j] -= 2.0 * d * d
                    new_max = new_conv.max()
                    if new_max < best_max_val - 1e-12:
                        best_max_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Hybrid optimizer:
    1. LSE Mirror Descent (fast, no LP) - 25% time
    2. LP Nesterov (exact best response) - 30% time
    3. LSE-Adam (smoothed gradient, adaptive) - 15% time
    4. Pair CD polish - 15% time
    5. LP Nesterov round 2 - 5% time
    6. Final Pair CD - 10% time
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE Mirror Descent (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.25
    n1 = n_base
    print(f"\n=== Phase 1: LSE Mirror Descent n={n1} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = lse_mirror_descent(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse_md_n{n1}")

    # === Phase 2: LP Nesterov (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.30
    n2 = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 25:
        a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_nest_n{n2}")

    # === Phase 3: LSE-Adam (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    n3 = len(best_seq)
    print(f"\n=== Phase 3: LSE-Adam n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = lse_adam_optimize(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_adam_n{n3}")

    # === Phase 4: Pair CD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    n4 = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n4} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t4, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd_n{n4}")

    # === Phase 5: LP Nesterov round 2 (5-10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.08
    n5 = len(best_seq)
    print(f"\n=== Phase 5: LP Nesterov r2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False)
        do_try_update(a_out, f"lp2_n{n5}")

    # === Phase 6: Final Pair CD with larger window ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.30
    n6 = len(best_seq)
    print(f"\n=== Phase 6: Final Pair CD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n6) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t6, rng, window=50, verbose=True)
        do_try_update(a_out, f"pcd2_n{n6}")

    # === Phase 7: Try alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for n_alt in [n_base + 150, n_base + 300, n_base - 100]:
            if n_alt < 200 or n_alt > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"\n=== Phase 7: Alt length n={n_alt} (remaining={remaining:.0f}s) ===")
            t7 = min(remaining * 0.35, 120.0)
            if t7 < 20:
                continue
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()

            # LSE mirror descent for quick optimization
            t7a = t7 * 0.6
            a_out_alt, _ = lse_mirror_descent(a_norm_alt, t7a, verbose=False)
            do_try_update(a_out_alt, f"lse_md_n{n_alt}")

            # Then LP if time remains
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 30 and best_seq is not None and len(best_seq) == n_alt:
                t7b = min(remaining2 * 0.4, t7 * 0.4)
                if t7b > 15:
                    a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                    a_out_alt2, _ = nesterov_lp_optimize(a_norm_alt2, t7b, solver, verbose=False)
                    do_try_update(a_out_alt2, f"lp_n{n_alt}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
