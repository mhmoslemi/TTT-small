        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        sc_a = _autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        
        # Restart if stalled
        if stall > stall_thresh:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0
        
        if verbose and t % 200 == 0:
            print(f"    NL t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def _subgrad_argmin(a, k_star):
    """Find j that minimizes the subgradient at peak k_star."""
    n = len(a)
    best_val = float('inf')
    best_j = 0
    for j in range(n):
        v = 0.0
        idx1 = k_star - j
        if 0 <= idx1 < n:
            v += a[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n and idx2 != idx1:
            v += a[idx2]
        if v < best_val:
            best_val = v
            best_j = j
    return best_j

def _subgrad_fw(a0, t_limit, verbose=False):
    """Simple subgradient Frank-Wolfe."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    y = _normalize(a0, S)
    a = y.copy()
    best_a = y.copy()
    best_score = _autocorr(y).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > t_limit or _time_left() < 30:
            break
        
        conv_y = _autocorr(y)
        k_star = int(np.argmax(conv_y))
        j_min = _subgrad_argmin(y, k_star)
        v = np.zeros(n)
        v[j_min] = S
        
        alpha = 2.0 / (t + 2.0)
        y_new = _normalize((1 - alpha) * y + alpha * v, S)
        
        if t == 1:
            a_new = v
        else:
            mu = min((t-1.0)/(t+2.0), 0.9)
            a_new = _normalize(y_new + mu*(y_new - y), S)
        
        y, a = y_new, a_new
        sc = _autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            print(f"    SF t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score
