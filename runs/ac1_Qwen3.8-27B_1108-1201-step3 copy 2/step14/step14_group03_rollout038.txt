    a = a * target_sum / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        top_k = max(5, size // 10)
        sort_idx = np.argsort(conv)[::-1]
        active_lags = set(sort_idx[:top_k].tolist())

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < target_sum * 5e-11:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < target_sum * 5e-11:
                        break

                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[0:hi1 - lo1]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        delta_vec[lo2:hi2] += a[n - 1 - (hi2 - 1):n - 1 - (lo2 - 1)][::-1]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] -= a[0:hi3 - lo3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        delta_vec[lo4:hi4] -= a[n - 1 - (hi4 - 1):n - 1 - (lo4 - 1)][::-1]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 12
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 1e-15:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * target_sum / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * target_sum / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def phase_restart_explore(a0, time_limit, solver, rng, verbose=True):
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s0

    best_a = a.copy()
    best_score = eval_internal(a)
    t0 = time.monotonic()

    n_restarts = 5
    per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        mix = 0.1 + 0.3 * r / n_restarts
        noise = rng.random(n) * mix
        a_restart = a + noise
        a_restart = np.maximum(a_restart, 0.0)
        s_r = a_restart.sum()
        if s_r < 1e-15:
            a_restart = a.copy()
        else:
            a_restart = a_restart * target_sum / s_r

        t_sub = min(per_restart, remaining * 0.8)
        if r % 2 == 0:
            a_out, sc = phase_smooth_gradient(a_restart, t_sub, verbose=False)
        else:
            a_out, _ = phase_lp_nesterov(a_restart, t_sub, solver, verbose=False)
            a_out = np.maximum(a_out, 0)
            s_o = a_out.sum()
            if s_o < 1e-15:
                a_out = a.copy()
            else:
                a_out = a_out * target_sum / s_o
            sc = eval_internal(a_out)

        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart {r+1} ({elapsed:.0f}s): new best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Smooth gradient descent (35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: Smooth Gradient n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm1 = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out1, _ = phase_smooth_gradient(a_norm1, t1, verbose=True)
        update_best(a_out1, "smooth_grad")

    # Phase 2: LP Nesterov (25% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out2, _ = phase_lp_nesterov(a_norm2, t2, solver, verbose=True)
        update_best(a_out2, "lp_nest")

    # Phase 3: Targeted CD (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Targeted CD n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm3 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out3, _ = phase_targeted_cd(a_norm3, t3, rng, window=40, verbose=True)
        update_best(a_out3, "cd")

    # Phase 4: LP Nesterov again (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm4 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out4, _ = phase_lp_nesterov(a_norm4, t4, solver, verbose=True)
        update_best(a_out4, "lp2")

    # Phase 5: Restart exploration (7% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = remaining * 0.07
    print(f"\n=== Phase 5: Restart Explore n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm5 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out5, _ = phase_restart_explore(a_norm5, t5, solver, rng, verbose=True)
        update_best(a_out5, "restart")

    # Phase 6: Final CD polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = max(0, remaining - 30)
    print(f"\n=== Phase 6: Final CD n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm6 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out6, _ = phase_targeted_cd(a_norm6, t6, rng, window=60, verbose=True)
        update_best(a_out6, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
