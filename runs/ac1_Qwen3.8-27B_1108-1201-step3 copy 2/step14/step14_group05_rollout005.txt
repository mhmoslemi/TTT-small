For a "perfect" sequence, all `conv(a,a)[k]` would be equal to some value `c`. Then `max_b = c` and we'd want to minimize `c`.

We have the constraint `sum_k conv(a,a)[k] = (sum a)^2` (by Parseval's theorem for the discrete case, or just by expanding the double sum).

1. **The LP solve is the bottleneck**. For n=1600, it's a 1600-variable LP with up to 3199 constraints. Modern solvers can handle this in ~10-50ms, but we need many iterations.

   - Direct optimization of the "flatness" using a softmax surrogate
   - Coordinate descent on the sequence
   - Spectral methods (design the DFT to be flat, which makes the autocorrelation flat)

Wait, the **spectral method** is interesting! By Welch's theorem / Fourier analysis:
`sum_k conv(a,a)[k]^2 = (1/N) sum |A(f)|^4` where A(f) is the DFT.

If we want `conv(a,a)[k]` to be as flat as possible (all equal to c), then we need the "circular autocorrelation" to be flat, which means the power spectrum `|A(f)|^2` should be constant. This is the "CAZAC" (Constant Amplitude Zero Auto-Correlation) or more precisely "low PAPR" sequence problem.

But we have a LINEAR (not circular) convolution, and the sequence must be non-negative. So the spectral approach is only an approximation.

2. Take the absolute value or rectify to make it non-negative
3. Use this as initialization for the LP-based optimization

Actually, for non-negative sequences, the DFT is the DTFT of a non-negative signal, which has properties related to positive-definite functions.

Let me think of another approach: **proximal gradient on the LSE surrogate**.

The function `f(a) = max_k conv(a,a)[k]` can be approximated by `g_T(a) = T * log(sum exp(conv(a,a)[k]/T))`. As T→0, this approaches the max. The gradient of `g_T` is:
`∇g_T(a) = sum_k w_k ∇conv(a,a)[k]` where `w_k = exp(conv(a,a)[k]/T) / sum_j exp(conv(a,a)[j]/T)`.

And `∇conv(a,a)[k]` with respect to `a[j]` is `2*a[k-j]` (for valid indices) - it's a shifted copy of `a`!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral initialization for diverse flat-autocorrelation starting points
2. Fast LP-based mirror descent with adaptive constraint management
3. LSE gradient descent with backtracking line search
4. Length exploration and interpolation
5. Peak shaving for final polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

_global_solver = None

        return _global_solver

        try:

            prob.solve(solver=s)

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _spectral_init(n, seed=0):
    """Generate a sequence with approximately flat autocorrelation using random phases."""
    rng = np.random.default_rng(seed)
    # Random Fourier phases -> flat power spectrum -> flat autocorrelation
    # Use only real frequencies for rfft
    nyq = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, nyq)
    # Construct spectrum with constant amplitude
    spectrum = np.exp(1j * phases)
    # Make it Hermitian symmetric for real output
    if n % 2 == 0:
        full_spec = np.zeros(n, dtype=complex)
        full_spec[:nyq] = spectrum
        full_spec[nyq:] = np.conj(spectrum[1:nyq][::-1])
    else:
        full_spec = np.zeros(n, dtype=complex)
        full_spec[:nyq] = spectrum
        full_spec[nyq:] = np.conj(spectrum[1:nyq][::-1])
    
    a = np.real(ifft(full_spec))
    a = np.abs(a)  # Make non-negative
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def _interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _lp_solve(a, M, solver, n_constraints=150):
    """LP best response with progressive constraint addition."""
    n = len(a)
    size = 2 * n - 1
    T = _build_conv_matrix(a)
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with top constraints
    circ = T @ a  # = conv(a, a)
    sorted_idx = np.argsort(circ)[::-1]
    n_init = min(n_constraints, size)
    active = sorted_idx[:n_init]
    
    constraints = [T[active] @ b_var <= M[active]]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    b_val = None
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    except Exception:
        return None
    
    if b_val is None or b_val.sum() < 1e-15:
        return None
    
    # Progressive constraint addition
    for _ in range(4):
        all_circ = T @ b_val
        violated = all_circ > M + 1e-6
        if not violated.any():
            break
        viol_idx = np.where(violated)[0]
        # Add worst violations
        if len(viol_idx) > n_constraints:
            vals = all_circ[viol_idx]
            keep = viol_idx[np.argsort(vals)[::-1][:n_constraints // 2]]
        else:
            keep = viol_idx
        active = np.unique(np.concatenate([active, keep]))
        active = active[:n_constraints * 2]
        
        rhs = np.maximum(M[active], all_circ[active] + 1e-8)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active] @ b_var <= rhs])
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                if b_new.sum() > b_val.sum() * 0.9:
                    b_val = b_new
        except Exception:
            break
    
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_nesterov(a0, time_limit, solver, verbose=True, n_constraints=150):
    """Fessler-Parks accelerated LP mirror descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.9
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = _fast_autocorr(y).max()
        
        b = _lp_solve(y, M_curr, solver, n_constraints=n_constraints)
        if b is None:
            stall += 1
            if stall > 20:
                # Restart
                rng = np.random.default_rng(n_iters * 31 + 7)
                y = np.maximum(y + rng.normal(0, 0.02 * y.mean(), n), 0)
                y = y * norm / y.sum()
                a = y.copy()
                prev_y = y.copy()
                stall = 0
            continue
        
        g = b * norm / b.sum()
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.4)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm / y_new.sum()
        
        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * norm / a_new.sum()
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        sc_y = _fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        sc_a = _fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            print(f"    LP iter={n_iters}: best={best_score:.8f}, M={M_curr:.6f}")
        
        # Restart if stalled
        if stall > 200:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.8
    
    return best_a, best_score


def _lse_gradient(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = best_score * 0.1
    T_end = max(best_score * 0.001, 1e-6)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = _fast_autocorr(a)
        max_b = b.max()
        
        # Softmax weights
        shifted = (b - max_b) / T
        # Numerical stability
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient via FFT: grad[j] = 2 * sum_k w[k] * a[k-j] (shifted)
        # This is 2 * conv(w, a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient (normalize a to fixed sum, so gradient should be on tangent)
        grad -= np.mean(grad)
        
        # Backtracking line search
        step = 1.0 / (T * n)
        score_curr = max_b
        for _ in range(20):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            sc_new = _fast_autocorr(a_new).max()
            if sc_new < score_curr * (1 - 1e-5):
                a = a_new
                break
            step *= 0.5
        
        steps += 1
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and steps % 200 == 0:
            print(f"    LSE step={steps}: T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score


def _peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing positions
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * 0.1
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = _fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                continue
        
        src = idx[np.argmax(contrib)]
        
        # Find good destinations (non-contributing to this peak)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = True
        
        if non_contrib.sum() == 0:
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[src] = False
        
        dst_idx = np.where(non_contrib)[0]
        dst = dst_idx[rng.integers(len(dst_idx))]
        
        # Line search
        best_trial = max_b
        best_frac = 0
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
            amt = a[src] * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * norm / a_trial.sum()
            sc = _fast_autocorr(a_trial).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac
        
        if best_frac > 0:
            a_trial = a.copy()
            a_trial[src] -= a[src] * best_frac
            a_trial[dst] += a[src] * best_frac
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * norm / a_trial.sum()
            sc = _fast_autocorr(a_trial).max()
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 100:
            # Multiple random moves
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.1, 0.4)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = _fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break
            stall = 0
        
        if verbose and stall % 2000 == 0 and stall > 0:
            print(f"    PeakShave stall={stall}: best={best_score:.8f}")
    
    return best_a, best_score
