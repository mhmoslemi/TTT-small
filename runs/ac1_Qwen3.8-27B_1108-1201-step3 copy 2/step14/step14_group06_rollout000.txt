def build_T(a, n):
    """Build the constraint matrix T where (T@b)[k] = sum_i b[i]*a[k-i] for valid (i,k-i)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        # b[i] contributes to positions k where k-i is in [0, n-1], i.e., k in [i, i+n-1]
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[start - i : end - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    
    # Get dual variables
    dual_y = None
    try:
        dual_y = np.asarray(constraints[0].dual_value, dtype=np.float64)
        if dual_y is not None:
            dual_y = np.maximum(dual_y, 0.0)
    except Exception:
        pass
    
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, dual_y

            else:

                non_peak_sorted = non_peak[np.argsort(a[non_peak])[:30]]
            
            # Try moving mass from top contributors to non-peak positions
            for src in top_idx[:10]:
                if a[src] < 1e-12:
                    continue
                for dst in non_peak_sorted[:10]:
                    if dst == src:
                        continue
                    for frac in [0.03, 0.07, 0.12, 0.2, 0.35]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation attempts
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    
    return best_a, best_score

def log_space_optimize(a0, time_limit):
    """Adam optimizer in log-space for the LSE-smoothed objective."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    
    # Work in log-space: z = log(a)
    z = np.log(a)
    
    # Adam optimizer
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for temp in [50.0, 20.0, 10.0, 5.0, 2.0, 1.0]:
        stage_limit = min((time_limit - (time.monotonic() - t0)) / 6.0, 120.0)
        stage_t0 = time.monotonic()
        
        while time.monotonic() - stage_t0 < stage_limit:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_curr = np.exp(z)
            s = a_curr.sum()
            if s < 1e-10:
                break
            a_normed = a_curr * norm_val / s
            
            b = fast_autocorr(a_normed)
            max_b = b.max()
            if max_b < 1e-15:
                break
            
            # LSE approximation of max
            shifted = (b - max_b) / temp
            exp_vals = np.exp(shifted)
            lse = max_b + temp * np.log(exp_vals.sum())
            w = exp_vals / exp_vals.sum()
            
            # Gradient of LSE w.r.t. a
            # d_lse/da[i] = sum_k w[k] * d(conv(a,a)[k])/da[i]
            # d(conv(a,a)[k])/da[i] = a[k-i] (if valid) + a[k-i] (the "other" side)
            # Actually for conv(a,a)[k] = sum_j a[j]*a[k-j], d/da[i] = a[k-i] + a[k-i]... no
            # conv(a,a)[k] = sum_j a[j] * a[k-j]
            # d/da[i] conv(a,a)[k] = a[k-i] (from the first factor when j=i) + a[k-i] (from the second factor when k-j=i, i.e., j=k-i)
            # Wait, that's only if i != k-i. If i == k-i (i.e., k=2i), then it's 2*a[i].
            # Actually: d/da[i] sum_j a[j]*a[k-j] = sum_j (delta(j,i)*a[k-j] + a[j]*delta(k-j,i))
            # = a[k-i] + a[k-i] = 2*a[k-i] (if 0<=k-i<n)
            # Wait no: delta(k-j, i) means k-j=i, so j=k-i.
            # So d/da[i] = a[k-i] + a[k-i] = 2*a[k-i]... 
            # Hmm, that's not right either. Let me be more careful.
            # conv(a,a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
            # d/da[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j,i)]
            # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, from j=k-i)
            # Wait, the second term: delta(k-j, i) is 1 when k-j=i, i.e., j=k-i.
            # So the second term is a[k-i] * delta(k-i is valid) ... 
            # Actually: sum_j a[j] * delta(k-j, i) = a[j] evaluated at j=k-i = a[k-i] (if valid)
            # And: sum_j delta(j,i) * a[k-j] = a[k-i] (if valid)
            # So total: 2 * a[k-i] (if 0<=k-i<n)
            # BUT this double-counts when i = k-i, i.e., k=2i. In that case it should be 2*a[i], not... 
            # Actually no, the formula is correct: d/da[i] of a[i]*a[i] is 2*a[i].
            
            # So grad_a[i] = sum_k w[k] * 2 * a[k-i] (for valid k-i)
            # This is 2 * (w * a)[i] in the "correlation" sense
            
            # Use FFT for efficiency
            fft_size = 1 << (2 * n - 1).bit_length()
            w_full = np.zeros(fft_size)
            w_full[:2*n-1] = w
            a_full = np.zeros(fft_size)
            a_full[:n] = a_normed
            
            W_fft = rfft(w_full, fft_size)
            A_fft = rfft(a_full, fft_size)
            # cross-correlation: sum_k w[k] * a[k-i]
            # This is like irfft(W_fft * conj(A_fft))
            cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
            # Actually let me verify: 
            # If c[i] = sum_k w[k] * a[k-i], this is the cross-correlation (w ⋆ a)[i]
            # In frequency domain: DFT(c) = conj(DFT(w)) * DFT(a) ... or DFT(w) * conj(DFT(a))?
            # The cross-correlation (w ⋆ a)[i] = sum_k w[k+i] * a[k]
            # Hmm, let me just do it directly for correctness (n is at most ~2000, so O(n^2) is fine)
            
            # Direct computation for correctness:
            grad_a = np.zeros(n)
            for i in range(n):
                for k in range(2*n-1):
                    j = k - i
                    if 0 <= j < n:
                        grad_a[i] += 2.0 * w[k] * a_normed[j]
            
            # Objective in log-space: f(z) = 2*n*lse(z) / (sum(exp(z)))^2
            # df/dz[i] = f/a[i] * a[i]/sum(a)^2 * ... 
            # Actually, let's just use the chain rule:
            # f = 2n * LSE / s^2 where s = sum(a)
            # df/dz[i] = (df/da[i]) * (da[i]/dz[i]) = (df/da[i]) * a[i]
            # df/da[i] = 2n * (1/s^2) * grad_a[i] - 2 * (2n*LSE/s^2) * (1/s)
            # = 2n/s^2 * grad_a[i] - 4n*LSE/s^3
            
            obj_val = 2.0 * n * lse / (s * s)
            grad_z = (2.0 * n / (s * s)) * grad_a - (2.0 * obj_val) * a_normed / s
            # Chain rule: df/dz[i] = df/da[i] * a[i]
            grad_z = grad_z * a_normed  # Wait, da[i]/dz[i] = a[i], and a[i] here is the current a
            # Actually a_normed = a * norm_val / s, and z is the log of the original a
            # Let me just use: da/dz = a (element-wise)
            grad_z = (2.0 * n / (s * s)) * grad_a * a_curr - (2.0 * obj_val) * a_normed * a_curr / s
            
            # Hmm this is getting complicated. Let me simplify by just optimizing in a-space with the constraint handled by projection.
            # Actually, let me just do it in a-space with L-BFGS-B on the LSE objective.
            break  # Skip the rest of this loop for now, use a different approach
        
        # Reinitialize Adam for new temperature
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
    
    # Fall back to L-BFGS-B in a-space
    def obj_and_grad(x):
        a_x = np.maximum(x, 1e-10)
        s = a_x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(n)
        a_n = a_x * norm_val / s
        b = fast_autocorr(a_n)
        max_b = b.max()
        if max_b < 1e-15:
            return 1e10, np.zeros(n)
        shifted = (b - max_b) / temp
        exp_vals = np.exp(shifted)
        lse = max_b + temp * np.log(exp_vals.sum())
        w = exp_vals / exp_vals.sum()
        
        # grad_a[i] = 2 * sum_k w[k] * a_n[k-i] (for valid k-i)
        grad_a = np.zeros(n)
        for i in range(n):
            for k in range(2*n-1):
                j = k - i
                if 0 <= j < n:
                    grad_a[i] += 2.0 * w[k] * a_n[j]
        
        obj_val = 2.0 * n * lse / (s * s)
        # d_obj/da[i] = 2n/s^2 * grad_a[i] - 2*obj_val/s * a_n[i]
        # But we're optimizing x (not normalized), and a_n = x * norm_val / s
        # This is getting complex. Let me just return obj_val and grad_a as an approximation.
        grad = (2.0 * n / (s * s)) * grad_a - (2.0 * obj_val) * a_n / s
        return obj_val, grad
    
    try:
        res = scipy_minimize(obj_and_grad, a.copy(), method='L-BFGS-B',
                             jac=True,
                             options={'maxiter': 200, 'maxfun': 500,
                                      'gtol': 1e-8, 'ftol': 1e-10})
        if res.x is not None:
            a_cand = np.maximum(res.x, 0.0)
            s_c = a_cand.sum()
            if s_c > 1e-10:
                a_cand = a_cand * norm_val / s_c
                sc = fast_autocorr(a_cand).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
    except Exception:
        pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Uniform with noise
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0)
    else:
        # Chebyshev-like
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
