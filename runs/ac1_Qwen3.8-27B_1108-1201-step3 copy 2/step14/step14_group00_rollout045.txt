def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return a_new / ss

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            el = time.monotonic() - t_global0
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, type_id=1)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    remaining = lambda: time_budget - (time.monotonic() - t_global0)

    # === Phase 1: LP Nesterov on parent length ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP+Nesterov n={n_cur} ===")
    t1 = min(remaining() * 0.30, 500.0)
    if t1 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0)
        try_update(a_out, f"NLP1_n{n_cur}")

    # === Phase 2: Pair CD polish ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Pair CD n={n_cur} ===")
    t2 = min(remaining() * 0.12, 180.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t2, seed=1)
        try_update(a_out, f"PCD1_n{n_cur}")

    # === Phase 3: Diversified restarts ===
    print(f"\n=== Phase 3: Diversified restarts ===")
    t3_total = min(remaining() * 0.22, 350.0)
    n_restarts = 3
    t3_each = t3_total / n_restarts
    for r in range(n_restarts):
        if remaining() < 80:
            break
        t3 = min(t3_each, remaining() - 60)
        if t3 < 30:
            break
        n_cur = len(best_seq)
        # Different perturbation scales
        sc = 0.03 + 0.03 * r
        pert = perturb(best_seq, rng, scale=sc)
        # Also try a fresh structured init at same length
        if r == 1:
            pert = structured_init(n_cur, rng, type_id=r)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=False, seed=100 + r)
        try_update(a_out, f"R{r}_n{n_cur}")

    # === Phase 4: Longer length ===
    if remaining() > 120:
        n2 = len(best_seq) + 400
        print(f"\n=== Phase 4: LP+Nesterov n={n2} ===")
        t4 = min(remaining() * 0.18, 280.0)
        if t4 > 30:
            init_long = interpolate_to_length(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True, seed=7)
            try_update(a_out, f"NLP_n{n2}")

    # === Phase 5: Pair CD on best ===
    n_cur = len(best_seq)
    if remaining() > 60:
        print(f"\n=== Phase 5: Pair CD n={n_cur} ===")
        t5 = min(remaining() * 0.12, 150.0)
        if t5 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = pair_cd(a_norm, t5, seed=2)
            try_update(a_out, f"PCD2_n{n_cur}")

    # === Phase 6: Final LP polish on best ===
    n_cur = len(best_seq)
    if remaining() > 50:
        print(f"\n=== Phase 6: Final NLP n={n_cur} ===")
        t6 = min(remaining() * 0.20, 200.0)
        if t6 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, seed=99)
            try_update(a_out, f"NLP_final_n{n_cur}")

    # === Phase 7: Shorter length exploration ===
    if remaining() > 60:
        n_short = max(800, len(best_seq) - 400)
        print(f"\n=== Phase 7: NLP n={n_short} ===")
