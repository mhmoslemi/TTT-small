import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v, s=1.0):
    """Project v onto {x : sum(x) = s, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with LogSumExp smoothing (fast, many iterations)
2. LP-based Fessler-Parks Nesterov (precise, fewer iterations)
3. Peak shaving for final polish
4. Multi-start with diverse initializations
Key innovation: FFT-based gradient computation enables 10x more iterations
than LP-based approach, with adaptive temperature annealing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v, s=1.0):
    """Project v onto {x : sum(x) = s, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, T):
    """Compute gradient of T*log(sum_k exp(c_k/T)) where c = conv(a,a)."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    scaled = (c - max_c) / T
    # Clip for numerical stability
    scaled = np.clip(scaled, -700, 700)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * (w cross_corr a)[m]
    # Using FFT: (w cross_corr a)[m] = ifft(fft(w) * conj(fft(a)))[m]
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    return 2.0 * corr


def lse_value(a, T):
    """Compute T*log(sum_k exp(c_k/T))."""
    c = fast_autocorr(a)
    max_c = c.max()
    scaled = (c - max_c) / T
    scaled = np.clip(scaled, -700, 700)
    return max_c + T * np.log(np.exp(scaled).sum())


def fista_lse(a0, time_limit, verbose=True):
    """FISTA with LSE smoothing and adaptive temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max() * 2 * n
    t0 = time.monotonic()

    # Initial LSE temperature
    c0 = fast_autocorr(a)
    T = max(c0.max() * 0.1, 1e-6 / n)
    T_min = 1e-4 / n

    y = a.copy()
    prev_y = a.copy()
    t_moment = 1.0
    L = 2.0 * n + 10.0 / T  # Initial Lipschitz estimate

    iters = 0
    stall_count = 0
    T_history = []

    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        g = lse_gradient(y, T)

        # Backtracking line search
        step = 1.0 / L
        for _ in range(20):
            z = y - step * g
            z_proj = simplex_project(z, 1.0)
            # Check descent condition (simplified: just check gradient direction)
            g_z = lse_gradient(z_proj, T)
            actual_decrease = np.dot(g, y - z_proj)
            predicted_decrease = 0.5 * step * np.dot(g * g)
            if actual_decrease >= 0.7 * predicted_decrease or step < 1e-10:
                break
            step *= 0.5
            L = max(L * 1.5, 1.0)

        y_new = simplex_project(y - step * g, 1.0)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_moment * t_moment))
        a_new = y_new + ((t_moment - 1.0) / t_new) * (y_new - prev_y)
        a_new = simplex_project(a_new, 1.0)

        prev_y = y
        y = y_new
        t_moment = t_new
        iters += 1

        # Track actual score
        score_y = fast_autocorr(y).max() * 2 * n
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Anneal temperature every 200 iterations
        if iters % 200 == 0:
            T *= 0.7
            T = max(T, T_min)
            L = 2.0 * n + 10.0 / T
            T_history.append(T)

            if verbose and iters % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FISTA iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.2e}, L={L:.0f}")

        # Restart if stalled
        if stall_count > 1000:
            rng = np.random.default_rng(iters * 7 + 3)
            pert = rng.normal(0, 0.02 / np.sqrt(n), n)
            y = simplex_project(y + pert, 1.0)
            prev_y = y.copy()
            t_moment = 1.0
            stall_count = 0
            T = max(T * 2, c0.max() * 0.05)
            L = 2.0 * n + 10.0 / T

        # If T is very small and we're converging, break
        if T < T_min * 1.1 and stall_count > 200:
            break

    return best_a, best_score


def multiplicative_weight(a0, time_limit, verbose=True):
    """
    Mirror descent with negative entropy (multiplicative weight updates).
    Uses subgradient of the actual max for the final phase.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max() * 2 * n
    t0 = time.monotonic()
    iters = 0

    # Adaptive step size
    lr = 0.01
    lr_min = 1e-6 / n

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()

        # Use LSE-weighted subgradient (smooth approximation to max gradient)
        T = max(max_c * 0.02, 1e-8 / n)
        scaled = (c - max_c) / T
        scaled = np.clip(scaled, -700, 700)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-300:
            w /= w_sum
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0

        # Gradient: 2 * cross_correlation(w, a)
        N = 1 << (3 * n - 2).bit_length()
        W = rfft(w, N)
        A = rfft(a, N)
        grad = 2.0 * irfft(W * np.conj(A), N)[:n]

        # Multiplicative update: a_new[j] = a[j] * exp(-lr * grad[j])
        scaled_grad = -lr * grad
        scaled_grad = np.clip(scaled_grad, -50, 50)
        a_new = a * np.exp(scaled_grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = fast_autocorr(a_new).max() * 2 * n

        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            lr = min(lr * 1.1, 0.1)
        elif score_new < best_score * 1.001:
            a = a_new
            lr *= 0.8
        else:
            lr = max(lr * 0.5, lr_min)

        iters += 1
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.2e}")

    return best_a, best_score


def lp_solve(a, M, solver=None):
    """Solve the best-response LP: max sum(b) s.t. conv(b,a) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1

    # Build convolution matrix T where (T@b)_k = conv(b,a)_k
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        T[k_start:k_end, j] = a[0:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def find_best_solver():
    """Find the best available LP solver."""
    if not HAS_CVXPY:

    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"    LP solver: {s}")
                return s
        except Exception:
            continue
    return cp.SCS


def lp_fessler_parks(a0, time_limit, solver, verbose=True, max_constraints=None):
    """Fessler-Parks accelerated LP Nesterov method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max() * 2 * n
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    t_moment = 1.0
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c_y = fast_autocorr(y)
        M = c_y.max()
        if M < 1e-15:
            break

        # Solve LP
        b = lp_solve(y, M, solver)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # Normalized best response
        iters += 1

        # Fessler-Parks step (Nesterov with gamma=2)
        t = iters
        alpha = 2.0 / (t + 1.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = t / (t + 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max() * 2 * n
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max() * 2 * n
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.2e}")

        # Restart if stalled

            rng = np.random.default_rng(iters * 13 + 7)

            y = simplex_project(y + pert, 1.0)
            a = simplex_project(a + pert, 1.0)
            prev_y = y.copy()
            t_moment = 1.0
            stall = 0

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Direct peak shaving: move mass from peak contributors to non-contributors."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max() * 2 * n
    t0 = time.monotonic()
    n_stall = 0
    n_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        p_star = int(np.argmax(c))

        # Find contributing indices for peak p_star
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            break

        src = idx[np.argmax(contrib)]

        # Find good destinations (low value, not contributing to peak)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = False
        nc_idx = np.where(non_contrib)[0]

        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

        # Pick destination from low-value non-contributing positions
        n_sel = min(5, len(nc_idx))
        sorted_dst = nc_idx[np.argsort(a[nc_idx])][:n_sel]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        # Try different transfer fractions
        src_val = a[src]
        best_trial = best_score
        best_frac = 0
        for frac in [0.03, 0.05, 0.08, 0.12, 0.2]:
            amt = src_val * frac
            if amt < 1e-15:
                break
            a_t = a.copy()
            a_t[src] = max(0, a_t[src] - amt)
            a_t[dst] += amt
            s_t = a_t.sum()
            a_t = a_t / s_t
            sc = fast_autocorr(a_t).max() * 2 * n
            if sc < best_trial:
                best_trial = sc
                best_frac = frac

        if best_frac > 0:
            a_t = a.copy()
            a_t[src] = max(0, a_t[src] - a_t[src] * best_frac)
            a_t[int(dst)] += a_t[src] * 0  # already applied via copy logic
            # Redo properly
            a_t = a.copy()
            amt = a_t[src] * best_frac
            a_t[src] -= amt
            a_t[int(dst)] += amt
            s_t = a_t.sum()
            a_t = a_t / s_t
            sc = fast_autocorr(a_t).max() * 2 * n
            if sc < best_score:
                a = a_t
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = np.maximum(a_t, 0)
                s_t = a_t.sum()
                if s_t > 0:
                    a_t = a_t / s_t
                    sc = fast_autocorr(a_t).max() * 2 * n
                    if sc < best_score:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                    else:
                        n_stall += 1
                else:
                    n_stall += 1
            else:
                n_stall += 1

        # Large perturbation if very stalled
        if n_stall > 100:
            n_pert = max(3, n // 100)
            perm = rng.permutation(n)
            for _ in range(n_pert):
                i, j = perm[0], perm[1]
                if a[i] > 0:
                    amt = a[i] * rng.uniform(0.1, 0.4)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 0:
                a = a / s_a
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_cosine_init(n, alpha=1.0):
    """Cosine-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** alpha
    s = a.sum()
    return a / s


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * sigma_frac ** 2))
    s = a.sum()
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        n_base = 1600

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # Find LP solver
    solver = find_best_solver() if HAS_CVXPY else None

    # === PHASE 1: FISTA with LSE smoothing (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: FISTA-LSE n={n_cur} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_init = best_seq.copy()
        a_out, _ = fista_lse(a_init, t1, verbose=True)
        try_update(a_out, f"fista_n{n_cur}")

    # === PHASE 2: LP Fessler-Parks (precise refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Fessler-Parks n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30 and solver is not None:
        a_out, _ = lp_fessler_parks(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # === PHASE 3: FISTA round 2 (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: FISTA-LSE r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_out, _ = fista_lse(best_seq, t3, verbose=True)
        try_update(a_out, f"fista2_n{n_cur}")

    # === PHASE 4: Multiplicative weight method ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Multiplicative Weight n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = multiplicative_weight(best_seq, t4, verbose=True)
        try_update(a_out, f"mw_n{n_cur}")

    # === PHASE 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Peak Shaving n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_out, _ = peak_shave(best_seq, t5, seed=999, verbose=True)
        try_update(a_out, f"peak_n{n_cur}")

    # === PHASE 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = len(best_seq) + 80
    print(f"\n=== Phase 6: FISTA n={n_alt} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 20:
        init_alt = interpolate_seq(best_seq, n_alt)
        a_out, _ = fista_lse(init_alt, t6, verbose=True)
        try_update(a_out, f"fista_n{n_alt}")

    # === PHASE 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20 and solver is not None:
        a_out, _ = lp_fessler_parks(best_seq, t7, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === PHASE 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 80.0)
    if t8 > 10:
        a_out, _ = peak_shave(best_seq, t8, seed=31415, verbose=True)
        try_update(a_out, f"peak_final_n{n_cur}")

    # === PHASE 9: Extra FISTA if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Extra FISTA n={n_cur} (budget={remaining:.0f}s) ===")
        a_out, _ = fista_lse(best_seq, remaining * 0.8, verbose=True)
        try_update(a_out, f"fista_extra_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
