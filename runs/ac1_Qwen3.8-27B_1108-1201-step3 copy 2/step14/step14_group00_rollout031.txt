Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with incremental autocorrelation updates
2. LP best-response with Nesterov acceleration
3. Structured initializations and multi-length exploration
4. Perturbation-based restarts

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a_init, float('inf')
    
    conv = fast_autocorr(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (S * S)
    best_a = a.copy()
    
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    
    # Scale factors to try
    scales = np.array([0.1, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0])
    
    while time.monotonic() - t0 < max_time:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > max_time:
                break
            
            ai = a[i]
            if ai < 1e-14:
                continue
            
            k_lo = max(0, i)
            k_hi = min(2 * n - 1, i + n)
            if k_hi <= k_lo:
                continue
            k_range = np.arange(k_lo, k_hi)
            j_range = k_range - i  # values of j = k - i, all in [0, n-1]
            
            # Save old values
            old_conv_segment = conv[k_lo:k_hi].copy()
            a_j_range = a[j_range].copy()
            
            current_obj = 2.0 * n * M / (S * S)
            best_local_obj = current_obj
            best_delta = 0.0
            
            for scale in scales:

                if abs(delta) < 1e-15:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                conv[k_lo:k_hi] = old_conv_segment + 2.0 * delta * a_j_range
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += delta * delta
                
                M_new = conv.max()
                obj_new = 2.0 * n * M_new / (S_new * S_new)
                
                if obj_new < best_local_obj - 1e-14:
                    best_local_obj = obj_new
                    best_delta = delta
            
            # Restore conv
            conv[k_lo:k_hi] = old_conv_segment

                S += best_delta
                conv[k_lo:k_hi] = old_conv_segment + 2.0 * best_delta * a_j_range
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += best_delta * best_delta
                M = conv.max()
                cur_obj = 2.0 * n * M / (S * S)
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def coord_descent_single(a_init, max_time, rng, verbose=True, print_interval=200):
    """Single-coordinate scaling coordinate descent with incremental conv updates."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a_init, float('inf')
    conv = fast_autocorr(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    scales = np.array([0.2, 0.4, 0.6, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.3, 1.5, 2.0])
    size = 2 * n - 1

    while time.monotonic() - t0 < max_time:
        n_passes += 1
        pass_best = best_obj
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > max_time:
                break
            ai = a[i]
            if ai < 1e-14:
                continue
            k_lo = max(0, i)
            k_hi = min(size, i + n)
            if k_hi <= k_lo:
                continue
            j_range = np.arange(k_lo, k_hi) - i
            old_seg = conv[k_lo:k_hi].copy()
            a_j = a[j_range].copy()
            cur_obj = 2.0 * n * M / (S * S)
            best_local = cur_obj
            best_d = 0.0

            for sc in scales:
                d = ai * (sc - 1.0)
                if abs(d) < 1e-15:
                    continue
                S_n = S + d
                if S_n < 0.01:
                    continue
                conv[k_lo:k_hi] = old_seg + 2.0 * d * a_j
                if 2 * i < size:
                    conv[2 * i] += d * d
                Mn = conv.max()
                obj = 2.0 * n * Mn / (S_n * S_n)
                if obj < best_local - 1e-14:
                    best_local = obj
                    best_d = d

            conv[k_lo:k_hi] = old_seg
            if best_d != 0.0:
                a[i] += best_d
                S += best_d
                conv[k_lo:k_hi] = old_seg + 2.0 * best_d * a_j
                if 2 * i < size:
                    conv[2 * i] += best_d * best_d
                M = conv.max()
                cur = 2.0 * n * M / (S * S)
                if cur < best_obj - 1e-14:
                    best_obj = cur
                    best_a = a.copy()

        if best_obj < pass_best - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % print_interval == 0:
            el = time.monotonic() - t0
            print(f"      CD1 pass={n_passes} ({el:.0f}s): obj={best_obj:.10f}")

    return best_a, best_obj


def coord_descent_pair(a_init, max_time, rng, verbose=True, print_interval=100):
    """Pair swap coordinate descent: move mass between two coordinates."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        return a_init, float('inf')
    conv = fast_autocorr(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < max_time:
        n_passes += 1
        pass_best = best_obj
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > max_time:
                break
            ai = a[i]
            if ai < 1e-14:
                continue
            # Gain for removing from i
            ki_lo = max(0, i)
            ki_hi = min(size, i + n)
            gi = np.zeros(size)
            ji = np.arange(ki_lo, ki_hi) - i
            gi[ki_lo:ki_hi] = a[ji]

            # Try a few random partners
            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]
                kj_lo = max(0, j)
                kj_hi = min(size, j + n)
                gj = np.zeros(size)
                jj = np.arange(kj_lo, kj_hi) - j
                gj[kj_lo:kj_hi] = a[jj]

                h = gj - gi  # net gain per unit delta moved from i to j

                # Try fractions of ai to move
                for frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
                    d = ai * frac
                    if d < 1e-14:
                        continue
                    # Conv change: d * (gj - gi) + quadratic terms
                    # Quadratic: -d^2 at k=2i (if valid), +d^2 at k=2j (if valid)
                    new_conv = conv + d * h
                    if 2 * i < size:
                        new_conv[2 * i] -= d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    Mn = new_conv.max()
                    obj = 2.0 * n * Mn / (S * S)
                    if obj < best_obj - 1e-13:
                        a[i] -= d
                        a[j] += d
                        conv[kj_lo:kj_hi] += d * a[jj].copy()
                        if 2 * j < size:
                            conv[2 * j] += d * d
                        conv[ki_lo:ki_hi] -= d * a[ji].copy()
                        if 2 * i < size:
                            conv[2 * i] -= d * d
                        M = conv.max()
                        best_obj = 2.0 * n * M / (S * S)
                        best_a = a.copy()
                        break

        if best_obj < pass_best - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                break

        if verbose and n_passes % print_interval == 0:
            el = time.monotonic() - t0
            print(f"      CDP pass={n_passes} ({el:.0f}s): obj={best_obj:.10f}")

    return best_a, best_obj


def build_conv_matrix(a, active_set=None):
    """Build convolution matrix rows for given active set indices."""
    n = len(a)
    size = 2 * n - 1
    if active_set is None:
        active_set = np.arange(size)
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            r = k - j
            if 0 <= r < n:
                T[idx, j] = a[r]
    return T


def lp_solve(a, M, solver, active_set=None, max_time_lp=5.0):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    if active_set is None:
        active_set = np.arange(size)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a, active_set)
    constraints = [T @ b_var <= M, cp.sum(b_var) >= 0.01]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-12:
        return None, active_set
    return b_val, active_set


def lp_nesterov(a_init, max_time, solver, verbose=True, seed=0, print_interval=200):
    """LP best-response with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > max_time:
            break
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive screening: keep constraints where conv is near max
        if active_set is None or t % 50 == 0:
            tol = 0.05 + 0.02 * np.exp(-t / 300.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y > M_curr * 0.7)[0]

        b, active_set = lp_solve(y, M_curr, solver, active_set)
        if b is None:
            if t % 10 == 0:

            if b is None:
                continue
            active_set = None

        b_sum = b.sum()
        if b_sum < 1e-12:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        sy = fast_autocorr(y).max()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        sa = fast_autocorr(a).max()
        if sa < best_score:
            best_score = sa
            best_a = a.copy()
            stall = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_c = avg_a / n_avg
            sc = fast_autocorr(avg_c).max()
            if sc < best_score:
                best_score = sc

                stall = 0

        if stall > 80:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall = 0

        if verbose and n_iters % print_interval == 0:
            el = time.monotonic() - t0

def structured_init(n, rng, method=0):
    """Generate structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    if method == 0:
        a = np.sin(np.pi * x) ** 0.7
    elif method == 1:
        a = 1.0 / (np.sqrt(np.maximum(x, 0.01) * np.maximum(1 - x, 0.01)) + 0.5)
    elif method == 2:
        a = np.cos(np.pi * x / 2.0) ** 2 + 0.3 * np.sin(np.pi * x)
    elif method == 3:
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.75) ** 2) + 0.2 * np.exp(-8.0 * (x - 0.5) ** 2)
    elif method == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif method == 5:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2 + 0.2 * np.sin(4 * np.pi * x) ** 2
    elif method == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) * np.exp(-3 * (x - 0.5) ** 2)
    else:
        a = np.ones(n) * (0.8 + 0.2 * rng.random(n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, scale=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_p = a * noise
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t1 = min(remaining * 0.20, 250.0)
    if t1 > 30:
        a_out, _ = coord_descent_single(best_seq, t1, rng, verbose=True)
        try_update(a_out, f"cd1_n{n_base}")

    # === Phase 2: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Pair CD n={n_cur} ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        a_out, _ = coord_descent_pair(best_seq, t2, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")

    # === Phase 3: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov n={n_cur} ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        norm_val = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm_val / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t3, solver, verbose=True, seed=0)
        try_update(a_out, f"lp_n{n_cur}")

    # === Phase 4: Coord descent after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CD polish n={n_cur} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = coord_descent_single(best_seq, t4, rng, verbose=True, print_interval=300)
        try_update(a_out, f"cd2_n{n_cur}")

    # === Phase 5: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Perturbation restarts ===")
    n_restarts = 3
    t5_each = min(remaining * 0.15 / n_restarts, 100.0)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(t5_each, remaining - 60)
        if t5 < 20:
            break
        n_cur = len(best_seq)
        pert = perturb_sequence(best_seq, rng, scale=0.04 + 0.03 * r)
        norm_val = np.sqrt(2.0 * n_cur)
        a_norm = pert * norm_val / pert.sum()
        a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=False, seed=200 + r)
        try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 6: Longer length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Longer length n={n_long} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        norm_val = np.sqrt(2.0 * n_long)
        a_norm = init_long * norm_val / init_long.sum()
        a_out, _ = lp_nesterov(a_norm, t6, solver, verbose=True, seed=7)
        try_update(a_out, f"lp_long_n{n_long}")
        # Quick CD polish on the longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = coord_descent_single(best_seq, min(remaining * 0.1, 60.0), rng, verbose=False)
            try_update(a_out2, f"cd_long_n{len(best_seq)}")

    # === Phase 7: Try different structured inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Structured inits ===")
    if remaining > 60:
        for meth in [1, 2, 4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_try = n_base
            a_try = structured_init(n_try, rng, method=meth)
            norm_val = np.sqrt(2.0 * n_try)
            a_norm = a_try * norm_val / a_try.sum()
            t_try = min(remaining * 0.15, 80.0)
            if t_try < 15:
                break
            a_out, _ = lp_nesterov(a_norm, t_try, solver, verbose=False, seed=500 + meth)
            try_update(a_out, f"init{meth}_n{n_try}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_cur} ===")
    if remaining > 20:
        t8a = min(remaining * 0.4, 120.0)
        if t8a > 15:
            a_out, _ = coord_descent_single(best_seq, t8a, rng, verbose=False)
            try_update(a_out, "final_cd")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.5, 100.0)
            if t8b > 15:
                norm_val = np.sqrt(2.0 * len(best_seq))
                a_norm = best_seq * norm_val / best_seq.sum()
                a_out, _ = lp_nesterov(a_norm, t8b, solver, verbose=False, seed=999)
                try_update(a_out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = compute_objective(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
