    # 2. conv_result = convolution of a_rev and w (linear convolution, length n + (2n-1) - 1 = 3n-2)
    # 3. cross = conv_result[n-1 : 2n-1]  (indices n-1 to 2n-2, which is n elements)
    
    # For FFT-based convolution:
    # fft_size >= 3n-1 (use next power of 2)
    # a_rev_padded = [a_rev, 0, 0, ...] (length fft_size)
    # w_padded = [w, 0, 0, ...] (length fft_size)
    # conv_result = irfft(rfft(a_rev_padded) * rfft(w_padded))[:3n-2]
    # cross = conv_result[n-1 : 2n-1]
    
    # Then grad_LSE_i = (2.0 / T) * cross[i]
    # full_grad_i = (2.0 * n / s**2) * grad_LSE_i - 4.0 * n * LSE * a[i] / s**3
    
    # This should be efficient!
    
    # Let me put it all together now.
    
    w = exp_vals / exp_vals.sum()
    
    # Compute cross-correlation: cross_i = sum_j a_j * w_{i+j}
    a_rev = a[::-1]
    fft_size2 = 1 << (3 * n - 1).bit_length()
    a_rev_padded = np.zeros(fft_size2)
    a_rev_padded[:n] = a_rev
    w_padded2 = np.zeros(fft_size2)
    w_padded2[:2*n-1] = w
    conv_result = irfft(rfft(a_rev_padded, fft_size2) * rfft(w_padded2, fft_size2))[:3*n-2]
    cross = conv_result[n-1 : 2*n-1]
    
    grad_LSE = (2.0 / T) * cross
    
    # Full objective gradient
    obj_val = 2.0 * n * lse / (s * s)
    grad = (2.0 * n / (s * s)) * grad_LSE - (4.0 * n * lse / (s ** 3)) * a
    
    return obj_val, grad.ravel()
