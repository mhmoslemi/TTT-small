def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

            # Try different transfer amounts
            best_d = 0.0
            best_new_max = conv.max()
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                d = a[i] * frac
                # Approximate new peak value
                new_peak = conv[k_peak] + d * (gj - gi)
                # Check if any other position becomes worse
                # (approximate: only check the new worst few positions)
                if new_peak < best_new_max - 1e-13:
                    best_new_max = new_peak
                    best_d = d

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a /= s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_a = a.copy()
                    improved_this_pass = True

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and (iteration + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD iter={iteration} ({elapsed:.0f}s): obj={best_obj:.8f}")

    return best_a, best_obj


def structured_init(n, rng, style=0):
    """Generate structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.sin(np.pi * x) ** 0.8
    elif style == 1:
        a = np.cos(np.pi * x * 0.5) ** 2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif style == 3:
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1 - np.abs(2 * x - 1)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.5
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize_simplex(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return normalize_simplex(a_interp)


def perturb_sequence(a, rng, scale=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    return normalize_simplex(a * noise)


def propose_candidate():
    """Main search function."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize_simplex(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, style=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Nesterov FW with weighted subgradients ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov FW Weighted, n={n_base} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = nesterov_fw_weighted(best_seq, t1, verbose=True, seed=0)
        try_update(a_out, "fw_w_nested")

    # === Phase 2: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov, n={n_cur} ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, verbose=True, seed=1)
        try_update(a_out, "lp1")

    # === Phase 3: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Pair CD, n={n_cur} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, _ = pair_cd(best_seq, t3, verbose=True, seed=2)
        try_update(a_out, "pcd1")

    # === Phase 4: FW on perturbed best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: FW on perturbed, n={n_cur} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        pert = perturb_sequence(best_seq, rng, scale=0.03)
        a_out, _ = nesterov_fw_weighted(pert, t4, verbose=False, seed=3)
        try_update(a_out, "fw_pert")

    # === Phase 5: LP on perturbed ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP on perturbed, n={n_cur} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        pert2 = perturb_sequence(best_seq, rng, scale=0.05)
        a_out, _ = nesterov_lp(pert2, t5, verbose=False, seed=4)
        try_update(a_out, "lp_pert")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_cur + 400
    print(f"\n=== Phase 6: LP on longer, n={n_long} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp(init_long, t6, verbose=True, seed=5)
        try_update(a_out, "lp_long")

    # === Phase 7: Final FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final FW, n={n_cur} ===")
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 20:
        a_out, _ = nesterov_fw_weighted(best_seq, t7, verbose=False, seed=6)
        try_update(a_out, "fw_final")

    # === Phase 8: Final Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PCD, n={n_cur} ===")
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 15:
        a_out, _ = pair_cd(best_seq, t8, verbose=True, seed=7)
        try_update(a_out, "pcd_final")

    # === Phase 9: Two-peak FW with different weights ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: FW variant, n={n_cur} ===")
        t9 = min(remaining * 0.30, 150.0)
        if t9 > 30:
            # Start from a different initialization style
            init_alt = structured_init(n_cur, rng, style=3)
            # Blend with best
            blend = 0.5 * best_seq + 0.5 * init_alt
            a_out, _ = nesterov_fw_weighted(blend, t9, verbose=False, seed=10)
            try_update(a_out, "fw_blend")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
