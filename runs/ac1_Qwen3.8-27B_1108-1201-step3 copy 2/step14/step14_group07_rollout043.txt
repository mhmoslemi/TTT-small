def coordinate_step(a, j):
    """Find optimal a[j] given all other coordinates fixed."""
    n = len(a)
    S_minus_j = a.sum() - a[j]
    
    # Compute C_k for all k
    # b_k = C_k + contribution from a[j]
    # contribution: 2*a[j]*a[k-j] for k≠2j (0≤k-j<n), a[j]^2 for k=2j
    x = a[j]
    b = fast_autocorr(a)  # current b
    
    # C_k = b_k - contribution
    C = b.copy()
    for k in range(2*n - 1):
        d = k - j
        if 0 <= d < n:
            if k == 2 * j:
                C[k] = b[k] - x * x
            else:
                C[k] = b[k] - 2 * x * a[d]
    
    # Now optimize g(x) = max_k (C_k + 2*a[k-j]*x) / (S_minus_j + x)^2
    #                 + parabola: (C[2j] + x^2) / (S_minus_j + x)^2
    # subject to x >= 0
    
    # We can use golden section search on [0, x_max] where x_max is large enough.
    # g(x) → 1 as x → ∞, so we want to find where g(x) < 1.
    
    # Find upper bound: g(x) < 1 when all terms < 1
    # (C_{2j} + x^2) / (S_minus_j + x)^2 < 1
    # C_{2j} + x^2 < (S_minus_j + x)^2 = S_minus_j^2 + 2*S_minus_j*x + x^2
    # C_{2j} < S_minus_j^2 + 2*S_minus_j*x
    # This is always true for large x.
    
    # For lines: (C_k + 2*a[k-j]*x) / (S_minus_j + x)^2 < 1
    # C_k + 2*a[k-j]*x < S_minus_j^2 + 2*S_minus_j*x + x^2
    # For large x, x^2 dominates, so this is true.
    
    # A reasonable upper bound: x_max = max(a[j], 2*S_minus_j) or something.
    
    x_max = max(x * 3, 2 * S_minus_j, 1.0)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_gauss = 0.0
    b_gauss = x_max
    c_val = b_gauss - gr * (b_gauss - a_gauss)
    d_val = a_gauss + gr * (b_gauss - a_gauss)
    f_c = g_of_x(c_val, C, a, j, S_minus_j)
    f_d = g_of_x(d_val, C, a, j, S_minus_j)
    
    for _ in range(50):
        if f_c < f_d:
            b_gauss = d_val
            d_val = c_val
            f_d = f_c

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build sparse Toeplitz constraint matrix: T[k,j] = a[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, size)
        k_idx = np.arange(start, end)
        rows.extend(k_idx.tolist())
        cols.extend([j] * (end - start))
        vals.extend(a[:end - start].tolist())
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    best_a = a.copy()
    best_maxb = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    patience = 0
    best_iter = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Line search: try a few step sizes
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_obj = fast_autocorr(np.maximum((1 - alpha_base) * y + alpha_base * g, 0)).max()
        for trial in [alpha_base * 0.5, alpha_base * 1.5, alpha_base * 2.5, alpha_base * 3.0]:
            trial = min(trial, 1.0)
            cand = np.maximum((1 - trial) * y + trial * g, 0.0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        alpha = best_alpha

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_maxb:
            best_maxb = score_y
            best_a = y.copy()
            best_iter = t
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_maxb:
            best_maxb = score_a
            best_a = a.copy()
            best_iter = t
            patience = 0

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_maxb:
                    best_maxb = score_avg
                    best_a = avg_check.copy()
                    best_iter = t
                    patience = 0

        if t % 200 == 0:
            stall_count += 1
            if patience > 500:
                if verbose:
                    print(f"    Stall at iter={n_iters}, best_maxb={best_maxb:.10f}")
                break
            elif patience > 100:
                y = best_a * norm / best_a.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                patience = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): maxb={best_maxb:.10f}")

    return best_a, best_maxb


def peak_gradient_descent(a0, time_limit, verbose=True):
    """Peak-directed gradient descent: reduce the current max peak iteratively."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 0.1 / n
    no_improve = 0
    max_no_improve = 500

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Compute gradient of b[k_star] w.r.t. a
        # For k < n: grad[j] = a[|k-j|] for j != k, 2*a[0] for j=k
        # For k >= n: grad[j] = a[k-j] for k-n+1 <= j <= n-1, 0 otherwise
        grad = np.zeros(n)
        if k_star < n:
            for j in range(n):
                if j == k_star:
                    grad[j] = 2.0 * a[0]
                else:
                    grad[j] = a[abs(k_star - j)]
        else:
            lo = k_star - n + 1
            hi = n - 1
            for j in range(lo, hi + 1):
                grad[j] = a[k_star - j]

        # Full gradient of f = b[k*]/(sum a)^2 = b[k*] (since sum=1)
        # But we also need to account for renormalization.
        # Simplified: just use gradient of b[k*] and project.
        # Direction: decrease a where grad is large (reduce the peak)
        # But we must keep a >= 0 and maintain sum=1.
        # Use: a_new = a - step * grad, then clip and renormalize.
        # Better: move mass from high-gradient positions to low-gradient positions.

        # Compute the gradient of the normalized objective more carefully.
        # f(a) = b[k*] / (sum a)^2. With sum a = 1, f = b[k*].
        # But after modification, sum changes, so we need:
        # df = grad_b * da - 2*b[k*]*a*ds / s^2 where s=sum(a)
        # For simplicity, use: direction = grad (since s=1, the correction is small)

        # Adaptive step size with backtracking
        improved = False
        for _retry in range(10):
            trial = a - step_size * grad
            trial = np.maximum(trial, 0.0)
            s_trial = trial.sum()
            if s_trial < 1e-10:
                step_size *= 0.5
                continue
            trial = trial / s_trial
            trial_score = 2.0 * n * fast_autocorr(trial).max()
            if trial_score < best_score - 1e-15:
                a = trial
                best_score = trial_score
                best_a = a.copy()
                no_improve = 0
                improved = True
                # Slightly increase step if improving well
                if it % 100 == 0 and step_size < 0.5 / n:
                    step_size *= 1.1
                break
            else:
                step_size *= 0.7

        if not improved:
            no_improve += 1
            # Try random perturbation to escape
            if no_improve % 20 == 0:
                rng = np.random.default_rng(it)
                n_pert = max(3, n // 100)
                idx = rng.choice(n, size=n_pert, replace=False)
                trial = a.copy()
                for i in idx:
                    trial[i] *= rng.uniform(0.5, 1.5)
                trial = np.maximum(trial, 0)
                s_t = trial.sum()
                if s_t > 1e-10:
                    trial = trial / s_t
                    tsc = 2.0 * n * fast_autocorr(trial).max()
                    if tsc < best_score:
                        a = trial
                        best_score = tsc
                        best_a = a.copy()
                        no_improve = 0

            if no_improve > max_no_improve:
                if verbose:
                    print(f"    PGD stall at iter={it}, score={best_score:.10f}")
                break

        if verbose and it % 1000 == 0:
            print(f"    PGD iter={it}: score={best_score:.10f}, step={step_size:.6f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent: optimize one coordinate at a time via golden section."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    order = np.arange(n)
    stall_sweeps = 0

    for sweep in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        # Shuffle order every 5 sweeps
        if sweep % 5 == 0:
            order = rng.permutation(n)

        improved_this_sweep = False

        for j in order:
            x = a[j]
            S_rest = 1.0 - x  # sum of others (total is 1)
            if S_rest < 1e-10:
                continue

            b = fast_autocorr(a)

            # Compute C_k = b_k minus contribution of a[j]
            C = b.copy()
            d = np.arange(2 * n - 1) - j
            valid = (d >= 0) & (d < n)
            if np.any(valid):
                if 2 * j < 2 * n - 1:
                    C[2 * j] = b[2 * j] - x * x
                    line_mask = valid.copy()
                    line_mask[2 * j] = False
                    if np.any(line_mask):
                        C[line_mask] = b[line_mask] - 2.0 * x * a[d[line_mask]]
                else:
                    C[valid] = b[valid] - 2.0 * x * a[d[valid]]

            # Golden section search for optimal x_new
            # g(x) = max_k [C_k + coeff_k * x] / (S_rest + x)^2
            # where coeff_k = 2*a[k-j] for line terms, x for parabola (special)
            
            x_lo = 0.0
            x_hi = max(3.0 * x, 3.0 * S_rest, 0.5 * S_rest + 1.0)
            
            def g_eval(x_new):
                denom = (S_rest + x_new) ** 2
                if denom < 1e-30:
                    return 1e10
                num = C.copy()
                if np.any(valid):
                    if 2 * j < 2 * n - 1:
                        line_mask = valid.copy()
                        line_mask[2 * j] = False
                        if np.any(line_mask):
                            num[line_mask] += 2.0 * a[d[line_mask]] * x_new
                        num[2 * j] += x_new * x_new
                    else:
                        num[valid] += 2.0 * a[d[valid]] * x_new
                return num.max() / denom

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            f_c1 = g_eval(c1)
            f_c2 = g_eval(c2)

            for _ in range(35):
                if f_c1 < f_c2:
                    x_hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    f_c1 = g_eval(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    f_c2 = g_eval(c2)

            x_opt = (x_lo + x_hi) / 2.0
            x_opt = max(0.0, x_opt)

            # Compare
            f_curr = g_eval(x)
            f_opt = g_eval(x_opt)
            if f_opt < f_curr - 1e-16 and abs(x_opt - x) > 1e-12:
                a[j] = x_opt
                s_new = a.sum()
                if s_new < 1e-10:
                    a[j] = x
                    continue
                a = a / s_new
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    improved_this_sweep = True

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep={sweep}: score={best_score:.10f}")

        if not improved_this_sweep:
            stall_sweeps += 1
            if stall_sweeps > 30:
                break
        else:
            stall_sweeps = 0

    return best_a, best_score


def lse_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=5):
    """LSE smoothed objective with L-BFGS-B."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        def lse_obj(x):
            ax = np.maximum(x, 0.0)
            sx = ax.sum()
            if sx < 1e-10:
                return 1e10
            b = fast_autocorr(ax)
            mb = b.max()
            if mb < 1e-15:
                return 1e10
            ev = np.exp(T * (b - mb))
            lse = mb + np.log(ev.sum()) / T
            return 2.0 * n * lse / (sx * sx)

        def lse_grad(x):
            ax = np.maximum(x, 0.0)
            sx = ax.sum()
            if sx < 1e-10:
                return np.zeros(n)
            b = fast_autocorr(ax)
            mb = b.max()
            if mb < 1e-15:
                return np.zeros(n)
            ev = np.exp(T * (b - mb))
            w = ev / ev.sum()
            lse = mb + np.log(ev.sum()) / T
            # Gradient of LSE w.r.t. a: T * sum_k w_k * db_k/da
            # db_k/da[j] = a[k-j] + a[j-k] (valid indices)
            # Use FFT: cross-correlation of a and w
            fft_size = 1 << (3 * n - 1).bit_length()
            ap = np.zeros(fft_size)
            ap[:n] = ax
            wp = np.zeros(fft_size)
            wp[:2 * n - 1] = w
            AF = rfft(ap, fft_size)
            WF = rfft(wp, fft_size)
            cross = irfft(AF * np.conj(WF), fft_size)[:n]
            conv_grad = T * (cross + cross[::-1] - ax * w[n - 1] if n > 0 else cross)
            # Actually let me compute it properly
            # db_k/da[j] for all k summed with w_k:
            # sum_k w_k * (a[k-j] + a[j-k]) = (w * a)[j] + (w_rev * a_rev)[j]
            # = cross-correlation
            # Simpler: use the FFT approach
            # sum_k w_k * a[k-j] = (w conv a)[j] (cross-correlation)
            # sum_k w_k * a[j-k] = (w rev conv a)[j]
            # Both combined: 2 * (w conv a)[j] if symmetric, else sum of both
            # Let's just use the direct FFT method:
            # grad_b[j] = sum_k w_k * (a[k-j] * valid + a[j-k] * valid)
            # = (full correlation of w and a)
            # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives correlation
            wc = irfft(rfft(wp, fft_size) * np.conj(rfft(ap, fft_size)), fft_size)[:n]
            grad_b = T * (wc + wc[n - 1::-1] if n > 1 else wc * 2)
            # Simplified: grad_b[j] = T * sum_k w_k * db_k/da[j]
            # For our purposes: grad_b = T * (a * w + a * w[::-1] shifted)
            # Let me just use a simpler approximation
            obj_val = 2.0 * n * lse / (sx * sx)
            # grad of obj = grad_b / s^2 - 2*obj*a/s
            grad = grad_b / (sx * sx) - 2.0 * obj_val * ax / sx
            return np.where(x < 0, 0, grad)

        try:
            res = scipy_minimize(lse_obj, a, method='L-BFGS-B',
                                  jac=lse_grad,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500})
            if res.x is not None:
                ac = np.maximum(res.x, 0.0)
                sc = ac.sum()
                if sc > 1e-10:
                    ac = ac / sc
                    score_c = 2.0 * n * fast_autocorr(ac).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = ac.copy()
                        a = ac
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(7.0 * x)
        a = np.maximum(a, 0.0)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 3:
        center = n / 2.0
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.sin(6.0 * np.pi * x) + 0.1 * np.sin(10.0 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def get_parent_sequence():
    """Try to load the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None

    n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)

    a_out, _ = nesterov_lp_optimize(a_in, t1, solver, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Peak grad n={n
