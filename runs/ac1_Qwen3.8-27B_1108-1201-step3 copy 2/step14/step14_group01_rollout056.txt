            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_mc:.8f} M={M:.6f}")

    return best_a, best_mc

def multiplicative_update(a0, time_limit, verbose=True):
    """Multiplicative (exponentiated gradient) update for maintaining non-negativity."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_mc = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Softmax weights with decreasing temperature
        T = max(0.5, 5.0 / (1.0 + it * 0.05))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -80.0, 80.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()

        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        term1 = irfft(W * np.conj(A), fft_size)[:n]
        term2 = irfft(A * W, fft_size)[:n]
        grad = 2.0 * (term1 + term2)

        # Multiplicative update: a_i <- a_i * exp(-step * grad_i)
        step = 1.0 / (np.linalg.norm(grad) + 1e-10) * 0.5
        log_update = np.exp(-step * grad)
        a_new = a * log_update
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new * norm / s_new

        mc = compute_max_conv(a_new)
        if mc < best_mc:
            best_mc = mc
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 50 == 0:
            print(f"    MultUpdate it={it}: mc={mc:.8f} best={best_mc:.8f}")

    return best_a, best_mc

def peak_flattening_cd(a0, time_limit, rng, verbose=True):
    """Targeted coordinate descent: reduce the highest peaks in self-convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_mc = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imps = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()

        # Get top peaks
        peak_count = min(10, size)
        peak_indices = np.argsort(conv)[::-1][:peak_count]

        pass_improved = False

        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            # conv[pk] = sum_j a[j] * a[pk - j] (valid j)
            # Find the pair (j, pk-j) that contributes most
            j_min = max(0, pk - n + 1)
            j_max = min(pk, n - 1)
            if j_max < j_min:
                continue

            contributions = a[j_min:j_max + 1] * a[pk - np.arange(j_min, j_max + 1)]
            if len(contributions) == 0:
                continue

            max_ci = np.argmax(contributions)
            j1 = j_min + max_ci
            j2 = pk - j1

            # Transfer mass from the larger of a[j1], a[j2] to a random position
            if a[j1] >= a[j2]:
                src = j1
            else:
                src = j2

            if a[src] < 1e-15:
                continue

            # Try several transfer amounts and destinations
            best_delta = 0.0
            best_new_max = cur_max
            best_dst = -1

            n_dst = 6
            dsts = rng.integers(0, n, size=n_dst)

            for dst in dsts:
                if dst == src:
                    continue
                # Try transferring fraction of a[src]
                for frac in [0.3, 0.5, 0.7, 0.9]:
                    d = a[src] * frac
                    a[src] -= d
                    a[dst] += d
                    a = a * norm / a.sum()
                    new_max = compute_max_conv(a)
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = d
                        best_dst = dst
                    a[dst] -= d
                    a[src] += d
                    a = a * norm / a.sum()

            if best_dst >= 0 and best_delta > 0:
                a[src] -= best_delta
                a[best_dst] += best_delta
                a = a * norm / a.sum()
                new_mc = compute_max_conv(a)
                if new_mc < best_mc - 1e-12:
                    best_mc = new_mc
                    best_a = a.copy()
                    pass_improved = True
                    total_imps += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve >= 4:
                # Perturb
                n_pert = max(5, n // 100)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
        else:
            no_improve = 0

        if verbose and total_imps % 50 == 0 and total_imps > 0:
            elapsed = time.monotonic() - t0
            print(f"    PFC improvements={total_imps} ({elapsed:.0f}s): best={best_mc:.8f}")

    return best_a, best_mc

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-4.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(3 * np.pi * x) * np.cos(np.pi * x)
    elif choice == 8:
        a = np.exp(-6.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-15.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-15.0 * (x - 0.8) ** 2)
    else:
        a = np.ones(n) * (0.5 + 0.3 * np.cos(2 * np.pi * x))
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE FISTA (fast initial progress)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: LSE FISTA n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_fista(best_seq, t1, verbose=True)
        try_update(a_out, "fista")

    # Phase 2: LP Nesterov (main workhorse) - longer allocation
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Peak-flattening CD
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Peak CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_flattening_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "peakcd")

    # Phase 4: Second LP round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=False)
        try_update(a_out, "lp2")

    # Phase 5: Multiplicative update polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: Multiplicative Update n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = multiplicative_update(best_seq, t5, verbose=False)
        try_update(a_out, "mult")

    # Phase 6: Try different length (slightly longer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 200
    t6 = min(remaining * 0.30, 150.0)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t6, 80.0), solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")

    # Phase 7: Try shorter length too
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, len(best_seq) - 200)
    t7 = min(remaining * 0.20, 100.0)
    print(f"\n=== Phase 7: Shorter length n={n_short} (budget={t7:.0f}s) ===")
    if t7 > 20 and remaining > 60:
        init_short = interpolate_seq(best_seq, n_short)
        a_out, _ = nesterov_lp_optimize(init_short, min(t7, 60.0), solver, verbose=False)
        try_update(a_out, f"lp_short_{n_short}")

    # Phase 8: Final peak CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.5, 120.0)
    print(f"\n=== Phase 8: Final Peak CD n={len(best_seq)} (budget={t8:.0f}s) ===")
    if t8 > 15:
        a_out, _ = peak_flattening_cd(best_seq, t8, rng, verbose=False)
        try_update(a_out, "peakcd_final")

    # Phase 9: Very final quick LP
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.8, 80.0)
    print(f"\n=== Phase 9: Final LP n={len(best_seq)} (budget={t9:.0f}s) ===")
    if t9 > 15:
        a_out, _ = nesterov_lp_optimize(best_seq, t9, solver, verbose=False)
        try_update(a_out, "lp_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
