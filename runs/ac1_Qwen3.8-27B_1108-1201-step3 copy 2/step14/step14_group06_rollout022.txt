    S = np.sqrt(2*n)
    
    # Compute contributions to the peak
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    contrib = a[idx] * a[k_peak - idx]
    total_peak = contrib.sum()
    
    # Find the best (src, dst) pair to reduce the peak
    best_improvement = 0
    best_move = None
    
    # Source: positions with high contribution to the peak
    top_contrib = idx[np.argsort(contrib)[::-1]][:20]
    
    # Destination: positions that don't contribute to the peak
    # (i.e., positions where a[j] * a[k_peak - j] = 0 or small)
    peak_mask = np.zeros(n, dtype=bool)
    peak_mask[js:je+1] = True
    non_peak = np.where(~peak_mask)[0]
    
    # Also consider positions that contribute to other peaks minimally
    # For simplicity, use positions with small values
    non_peak_sorted = non_peak[np.argsort(a[non_peak])][:20]
    
    for src in top_contrib:
        for dst in non_peak_sorted:
            if dst == src:
                continue
            # Try moving a small amount from src to dst
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * S / a_trial.sum()
                
                b_trial = fast_autocorr(a_trial)
                score = b_trial.max()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    
    # Gradient via FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)
        
        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)

        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g
        
        try:

                        best_a = a_candidate.copy()

        except Exception:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if s < 1e-15:

def oscillatory_init(n, rng):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE (log-sum-exp) smoothed gradient with L-BFGS-B for continuous refinement
2. Peak-targeted surgery with momentum: identifies top contributing pairs across multiple peaks
   and performs optimal mass transfers that simultaneously reduce several peaks
3. Multi-start with diverse oscillatory/structured initializations to explore different basins
4. Adaptive multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and its gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_optimize(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=7):
    """Multi-stage LSE optimization with L-BFGS-B and decreasing temperature."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        T_val = float(T)

        def obj_func(x, TV=T_val):
            val, _ = lse_objective_and_grad(x, n, TV)
            return val

        def grad_func(x, TV=T_val):
            _, g = lse_objective_and_grad(x, n, TV)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 300, 'maxfun': 800,
                                          'gtol': 1e-10, 'ftol': 1e-12,
                                          'maxls': 50})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
        except Exception:
            pass
    return best_a, best_score


def multi_peak_surgery(a0, time_limit, seed=42, n_top_peaks=8, n_candidates=12):
    """
    Multi-peak targeted surgery: identifies top contributing pairs across
    several peaks simultaneously and finds mass transfers that reduce the
    global maximum. Uses pressure-based scoring to guide source/destination
    selection, with adaptive fractional steps.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    fracs_small = [0.02, 0.04, 0.07, 0.1, 0.15]
    fracs_med = [0.05, 0.1, 0.15, 0.25, 0.4]

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break
        iter_count += 1
        b = fast_autocorr(a)
        current_max = b.max()
        top_peaks = np.argsort(b)[::-1][:n_top_peaks]
        # Build pressure vector: for each position i, how much does it contribute
        # to the top peaks (weighted by peak height)
        pressure = np.zeros(n)
        for k_peak in top_peaks:
            pk_val = b[k_peak]
            if pk_val < current_max * 0.7:
                break
            weight = pk_val
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            pressure[idx] += weight * a[k_peak - idx]
        # Source candidates: high a[i] * pressure[i] (reducing these helps most)
        vp = a * pressure
        top_reduce = np.argsort(vp)[::-1]
        top_reduce = top_reduce[a[top_reduce] > 1e-8][:n_candidates]
        # Destination candidates: low pressure (adding here hurts least)
        low_pressure = np.argsort(pressure)[:n_candidates]
        improved = False
        # Try systematic transfers
        for src in top_reduce:
            if a[src] < 1e-8:
                continue
            for dst in low_pressure:
                if dst == src:
                    continue
                for frac in fracs_small:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try larger fractions
            for src in top_reduce[:6]:
                if a[src] < 1e-8:
                    continue
                for dst in low_pressure[:6]:
                    if dst == src:
                        continue
                    for frac in fracs_med:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            # Random kick to escape
            for _ in range(40):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                frac = float(rng.uniform(0.01, 0.06))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                # Occasionally do a larger random perturbation
                if stall % 50 == 0:
                    for _ in range(10):
                        i = int(rng.integers(n))
                        j = int(rng.integers(n))
                        if i == j or a[i] < 1e-8:
                            continue
                        frac = float(rng.uniform(0.1, 0.3))
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            break
        else:
            stall = 0
        if iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MPS iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
    return best_a, best_score


def oscillatory_init(n, rng):
    """Generate oscillatory initial sequence to flatten convolution."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    n_freqs = int(rng.integers(4, 9))
    for _ in range(n_freqs):
        f = float(rng.uniform(1.0, n / 2.0))
        amp = float(rng.uniform(0.05, 0.3))
        phase = float(rng.uniform(0, 2 * np.pi))
        a += amp * np.cos(2.0 * np.pi * f * x / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        idx = np.arange(n) - n // 2
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) * 7.0 / n) + 0.1 * np.cos(2.0 * np.pi * np.arange(n) * 13.0 / n)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-50.0 * (x - 0.2) ** 2) + np.exp(-50.0 * (x - 0.8) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state with candidate."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining:
    1. LSE gradient optimization (smooth continuous refinement)
    2. Multi-peak targeted surgery (fast, focused local search)
    3. Multi-start with diverse initializations (explore different basins)
    4. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')
    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    n_best = len(best_seq)
    # Phase 1: LSE from parent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.25
    print(f"\n=== Phase 1: LSE n={n_best} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = lse_optimize(a_norm, t1, T_start=100.0, T_end=0.3, n_stages=7)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_n{n_best}", t_global)
    # Phase 2: Multi-peak surgery (main workhorse - fast iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.35
    print(f"\n=== Phase 2: MultiPeakSurg n={n_best} (budget={t2:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = multi_peak_surgery(a_norm, t2, seed=42, n_top_peaks=8, n_candidates=12)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"mps_n{n_best}", t_global)
    # Phase 3: LSE round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.10
    print(f"\n=== Phase 3: LSE r2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t3, T_start=30.0, T_end=0.1, n_stages=4)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse2_n{n_best}", t_global)
    # Phase 4: Multi-start with oscillatory inits
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: Multi-start oscillatory (budget={t4:.0f}s) ===")
    if t4 > 30:
        n_starts = min(5, int(t4 / 30))
        per_start = t4 / max(n_starts, 1)
        for si in range(n_starts):
            remaining_ms = time_budget - (time.monotonic() - t_global)
            if remaining_ms < 20:
                break
            nt = min(per_start, remaining_ms - 10)
            if nt < 15:
                break
            n_try = n_best
            init = oscillatory_init(n_try, rng)
            a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
            # Quick LSE then surgery
            t_lse = nt * 0.5
            t_surg = nt * 0.5
            a_lse, _ = lse_optimize(a_norm, t_lse, T_start=50.0, T_end=2.0, n_stages=3)
            a_surg, _ = multi_peak_surgery(a_lse, t_surg, seed=1000 + si, n_top_peaks=6, n_candidates=8)
            best_seq, best_score = attempt_update(best_seq, best_score, a_surg, f"osc_s{si}_n{n_try}", t_global)
    # Phase 5: Final surgery polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        third = remaining / 3.0
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_surgery(a_norm, third, seed=99999, n_top_peaks=6, n_candidates=10)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "mps_final1", t_global)
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out2, _ = multi_peak_surgery(a_norm2, remaining2, seed=55555, n_top_peaks=5, n_candidates=8)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out2, "mps_final2", t_global)
    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    return [float(x) for x in best_seq]
