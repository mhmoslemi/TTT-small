        # Periodic gradient refinement every 20 LP iterations
        if n_iters % 20 == 0 and (time.monotonic() - t0) < time_limit * 0.8:
            gd_a, gd_score = gradient_phase(y, 3.0, n_steps=30, T_init=best_score * 0.3)
            if gd_score < best_score:
                best_score = gd_score
                best_a = gd_a.copy()
                y = gd_a * norm / gd_a.sum()
                a = y.copy()
        
        # Perturbation restart if stuck
        if n_iters % 50 == 0 and best_score > last_best - 1e-10:
            rng_pert = np.random.default_rng(n_iters)
            if rng_pert.random() < 0.3:
                eps = best_score * 0.01
                y_pert = y + rng_pert.normal(0, eps / n, n)
                y_pert = np.maximum(y_pert, 0)
                s_p = y_pert.sum()
                if s_p > 1e-15:
                    y_pert = y_pert * norm / s_p
                    sc_p = fast_autocorr(y_pert).max()
                    if sc_p < best_score:
                        y = y_pert
                        a = y_pert.copy()
        
        last_best = best_score
        
        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_polish(a0, time_limit, rng, window=25, verbose=True):
    """Pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-12:
                continue
            
            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i or a[j] < norm * 1e-14:
                        continue
                    if a[i] < norm * 1e-12:
                        break
                    
                    # Try moving small amounts from i to j
                    # Use a few trial values
                    best_d = 0.0
                    best_max = conv.max()
                    n_trials = 8
                    for trial in range(1, n_trials + 1):
                        d = ai * trial / (n_trials + 1) * 0.8
                        if d <= 0 or d >= a[i]:
                            continue
                        # Approximate new conv
                        # delta_conv[k] = -d*(a[k-i+n-1] + a[i+k-n+1]) + d*(a[k-j+n-1] + a[j+k-n+1])
                        # (ignoring d^2 terms for approximation)
                        new_conv = conv.copy()
                        for k in range(size):
                            idx1 = k - i + n - 1
                            idx2 = i + k - n + 1
                            idx3 = k - j + n - 1
                            idx4 = j + k - n + 1
                            delta = 0.0
                            if 0 <= idx1 < n:
                                delta -= a[idx1]
                            if 0 <= idx2 < n:
                                delta -= a[idx2]
                            if 0 <= idx3 < n:
                                delta += a[idx3]
                            if 0 <= idx4 < n:
                                delta += a[idx4]
                            new_conv[k] -= d * delta * 2
                        
                        new_max = new_conv.max()
                        if new_max < best_max - 1e-14:
                            best_max = new_max
                            best_d = d
                    
                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur = conv.max()
                        if cur < best_score - 1e-14:
                            best_score = cur
                            best_a = a.copy()
                            pass_improved = True
        
        if not pass_improve:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

Wait, the pair_cd_polish is O(n * window * size) per pass which for n=1600, window=25, size=3199 is about 1600*25*3199 ≈ 128M operations per pass. That's too slow.

Let me optimize the pair CD. The key insight is that when we move mass d from position i to position j, the change in the autocorrelation is:

`delta_c[k] = -d * (a[k-i+n-1] + a[i+k-n+1] - d*[k==2(i-n+1)]...) + d * (a[k-j+n-1] + a[j+k-n+1] - d*[k==2(j-n+1)]...)`

Ignoring the d^2 terms (which are small for small d):
`delta_c[k] ≈ d * (a[k-j+n-1] + a[j+k-n+1] - a[k-i+n-1] - a[i+k-n+1])`

This is a fixed vector (independent of d), so we can compute it once and then check `max(conv + d * delta_c)` for various d. The max of `conv + d * delta_c` is a piecewise linear function of d, and we can find the best d by checking where the maximum changes.

Actually, for the pair CD, let me use a simpler but faster approach:
- Precompute the "signature" of each position (how the autocorrelation changes when mass is added/removed there)
- For a pair (i,j), the combined signature is sig_j - sig_i
- Find the best d by checking max(conv + d * (sig_j - sig_i))

The signature of position i: `sig_i[k] = a[k-i+n-1] + a[i+k-n+1]` (with bounds checking)

This is O(n * size) to precompute all signatures, which is O(n^2 * 2) = O(2n^2) ≈ 5M for n=1600. That's fine.

Then for each pair, we just do a dot-product-like check which is O(size) = O(n). With n * window pairs, that's O(n^2 * window) per pass ≈ 1600 * 25 * 1600 = 64M. Still a bit much but feasible if we vectorize.

Actually, let me just use a simpler approach: for each position i, compute the "gradient" of the max autocorrelation with respect to a[i]. Then do coordinate descent: for each i, find the best value of a[i] given all other values fixed.

For a fixed i, if we change a[i] to a[i] + d, the autocorrelation changes by:
`c_new[k] = c_old[k] + d * (a[k-i+n-1] + a[i+k-n+1]) + d^2 * [k == 2i-n+1]` (approximately)

Actually the d^2 term only appears at k = 2i - n + 1 (if valid), where c[k] gets an additional d^2.

So the new max is `max(max(c_old + d * sig_i), c_old[2i-n+1] + 2*d*a[i] + d^2)` (if 2i-n+1 is valid).

We can find the optimal d by binary search or by checking a few candidate values.

This single-coordinate approach is O(n * size) per sweep = O(n^2 * 2) ≈ 5M per sweep. With many sweeps in 1800s, this is very feasible.

But single-coordinate descent might be slow to converge. Pair CD is better because it can "see" the interaction between two positions.

Let me use a hybrid: 
- Precompute signatures
- For each pair (i, j) with j close to i (within window), compute the combined signature
- Find the best d by checking max(conv + d * combined_sig) at a few values of d
- This is O(size) per pair, and with n * window pairs, total is O(n * window * size) per pass

For n=1600, window=25, size=3199: 1600 * 25 * 3199 ≈ 128M per pass. With numpy vectorization, this is maybe 0.1-0.5 seconds per pass. So in 300 seconds, we can do 600-3000 passes. That should be enough for polish.

Actually wait, I can vectorize this much better. For a fixed i, the signatures for j = i+1, i+2, ..., i+window can be computed at once using the precomputed signature array.
