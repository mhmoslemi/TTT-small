def fast_conv_self(a):

def lse_gd(a0, time_limit, verbose=True):
    """LSE smooth gradient descent"""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = np.geomspace(30.0, 0.2, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, int(remaining * 3))

            if time.monotonic() - t0 > time_limit:

            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient computation
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]

            # Line search

            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
            
            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new

def lp_solve(a, M, solver, active):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix (sparse for efficiency)
    # conv(a,b)[k] = sum_i a[i] * b[k-i] (where valid)
    # For each k, the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i] <= M
    # Let j = k-i: sum_{j} a[k-j] * b[j] <= M, where k-j in [0,n-1] => j in [k-n+1, k]
    
    # For efficiency, build the matrix explicitly for active constraints only
    b_var = cp.Variable(n, nonneg=True)
    
    # Build T matrix for active constraints
    # T[k, j] = a[k-j] if k-j in [0, n-1], else 0
    n_active = len(active)
    T = np.zeros((n_active, n))
    for idx, k in enumerate(active):
        # T[k,j] = a[k-j] for j where 0 <= k-j <= n-1
        # i.e., j in [k-(n-1), k] intersect [0, n-1]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            T[idx, lo:hi+1] = a[hi:lo-1:-1]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                b = y - (norm / (T_val * np.sqrt(n))) * grad / gn
                b = np.maximum(b, 0.0)
            else:
                b = y.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = y.copy()
            b_sum = y.sum()
        g = b * norm / b_sum
        
        # Check dropped constraints
        if len(active_set) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 1e-3):
                # Add violated constraints
                violated = np.where(check > M * (1 + 1e-3))[0]
                active_set = np.unique(np.concatenate([active_set, violated]))
                b = lp_solve(y, M, solver, active_set)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): y={sc_y:.8f}, a={sc_a:.8f}, best={best_score:.8f}, |A|={len(active_set)}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):

        pass_improved = False

            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Find candidate j's
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            # Gradient-informed: j that reduces the max convolution the most
            peak_lags = np.where(conv >= cur_max * 0.9)[0][:5]
            for kl in peak_lags:
                j_struct = (kl - i) % n
                j_struct2 = (kl + i) % n
                j_candidates.add(int(j_struct))
                j_candidates.add(int(j_struct2))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                for d in deltas:
                    if d < 1e-10:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j

                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                new_max = compute_max_conv(a)
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()

                a = np.maximum(a, 0.0)

        else:
            no_improve = 0

            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

def structured_init(n, rng):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE smooth gradient descent for fast initial progress

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_max_conv(a):
    return float(fast_conv_self(a).max())


def eval_fn(a):
    a_list = [float(max(0.0, min(1000.0, x))) for x in a]

                    continue
                a_new = a_new * norm / s_new
                sc = compute_max_conv(a_new)
                if sc <= max_c + 0.005 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def build_T_rows(a, active):
    n = len(a)
    size = 2 * n - 1
    n_active = len(active)
    T = np.zeros((n_active, n))
    for idx, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            T[idx, lo:hi + 1] = a[hi:lo - 1:-1]
    return T


def lp_solve(a, M, solver, active):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_rows(a, active)
    if T.shape[0] == 0:
        return a / a.sum()
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)


def lse_fallback(y, norm, T_val=1.0):
    n = len(y)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(y)
    max_c = conv.max()
    shifted = (conv - max_c) / T_val
    shifted = np.maximum(shifted, -80.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    W = rfft(w, fft_size)
    A = rfft(y, fft_size)
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    g2 = irfft(A * W, fft_size)[:n]
    grad = 2.0 * (g1 + g2)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return y.copy()
    step = norm / (T_val * np.sqrt(n))
    b = y - step * grad
    b = np.maximum(b, 0.0)
    bs = b.sum()
    if bs < 1e-15:
        return y.copy()
    return b * norm / bs


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    active_set = np.where(fast_conv_self(y) > fast_conv_self(y).max() * 0.7)[0]
    no_improve = 0
    prev_best = best_score

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint selection
        threshold = M * 0.8
        candidate_active = np.where(conv_y > threshold)[0]
        active_set = np.unique(np.concatenate([active_set, candidate_active]))
        if len(active_set) > 400:
            active_set = candidate_active if len(candidate_active) >= 20 else active_set[:400]
        if len(active_set) < 10:
            active_set = np.where(conv_y > M * 0.5)[0]
        if len(active_set) == 0:
            active_set = np.arange(size)

        b = lp_solve(y, M, solver, active_set)
        if b is None:
            b = lp_solve(y, M, solver, np.arange(size))
        if b is None:
            b = lse_fallback(y, norm)

        b_sum = b.sum()
        if b_sum < 1e-15:
            b = y.copy()
            b_sum = y.sum()
        g = b * norm / b_sum

        # Check dropped constraints
        if len(active_set) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * (1 + 5e-4):
                violated = np.where(check > M * (1 + 5e-4))[0]
                active_set = np.unique(np.concatenate([active_set, violated]))
                if len(active_set) > 500:
                    active_set = np.where(fast_conv_self(y) > M * 0.6)[0]
                b2 = lp_solve(y, M, solver, active_set)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Convergence check
        if best_score < prev_best - 1e-8:
            no_improve = 0
            prev_best = best_score
        else:
            no_improve += 1
            if no_improve == 200:
                # Perturb to escape
                idx = np.random.default_rng(t).choice(n, size=max(5, n // 50), replace=False)
                y[idx] *= (1.0 + 0.1 * np.random.default_rng(t + 1).standard_normal(len(idx)))
                y = np.maximum(y, 0.0)
                y = y * norm / y.sum()
                active_set = np.where(fast_conv_self(y) > fast_conv_self(y).max() * 0.6)[0]
                no_improve = 0
                prev_best = best_score

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}, |A|={len(active_set)}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Candidate j selection
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            peak_lags = np.where(conv >= cur_max * 0.85)[0][:6]
            for kl in peak_lags:
                j_candidates.add(int((kl - i) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 25
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                for d in deltas:
                    if d < 1e-10:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = float(new_conv.max())
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = int(j)

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = compute_max_conv(a)
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 10:
                break
            if no_improve >= 5:
                n_perturb = max(3, n // 80)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
        else:
            no_improve = 0

        if verbose and total_imp % 200 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_imp} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Chebyshev-like
        a = np.cos(np.arccos(np.abs(2.0 * x - 1.0))) ** 2
    else:
        # Multi-peak Gaussian
        centers = [0.2, 0.5, 0.8]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-20.0 * (x - c) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        if not isinstance(a, np.ndarray):
            a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE GD (quick warmup)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gd(best_seq, t1, verbose=False)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = nesterov_lp(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: LP restart with perturbation
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: LP restart (budget={t3:.0f}s) ===")
    if t3 > 60:
        a_perturb = best_seq.copy()
        n_perturb = max(10, len(a_perturb) // 15)
        idx = rng.choice(len(a_perturb), size=n_perturb, replace=False)
        a_perturb[idx] *= (1.0 + 0.12 * rng.standard_normal(n_perturb))
        a_perturb = np.maximum(a_perturb, 0.0)
        a_perturb = a_perturb / a_perturb.sum()

        t3a = t3 * 0.6
        a_out, _ = nesterov_lp(a_perturb, t3a, solver, verbose=False)
        try_update(a_out, "lp_restart1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            a_perturb2 = best_seq.copy()
            a_perturb2 *= (1.0 + 0.06 * rng.standard_normal(len(a_perturb2)))
            a_perturb2 = np.maximum(a_perturb2, 0.0)
            a_perturb2 = a_perturb2 / a_perturb2.sum()
            t3b = min(remaining * 0.35, 80.0)
            a_out, _ = nesterov_lp(a_perturb2, t3b, solver, verbose=False)
            try_update(a_out, "lp_restart2")

    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 4: Pair CD (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = pair_cd(best_seq, t4, rng, verbose=True)
        try_update(a_out, "pcd")

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 120:
        for n_try in [n_base + 200, n_base + 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            init_long = interpolate_seq(best_seq, n_try)
            t5 = min(remaining * 0.25, 150.0)
            a_out, _ = nesterov_lp(init_long, t5, solver, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t6a = min(remaining * 0.45, 150.0)
        a_out, _ = nesterov_lp(best_seq, t6a, solver, verbose=False)
        try_update(a_out, "lp_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.7, 120.0)
            a_out, _ = pair_cd(best_seq, t6b, rng, verbose=False)
            try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
