def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgrad(a, active_indices):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    g[i] = 2 * sum_{k in active} a[k-i] * I(0 <= k-i < n)
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        j_lo = k - i_hi
        j_hi = k - i_lo
        g[i_lo:i_hi + 1] += 2.0 * a[j_lo:j_hi + 1][::-1]
    return g


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    """Compute objective 2*n*max_b/(sum_a)^2 locally without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        rows.extend(idx)
        cols.extend([i] * len(idx))
        vals.extend(a[idx - i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_subgrad_between=True):
    """LP best-response with Nesterov acceleration and subgradient steps between LP calls."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    lp_call_count = 0
    subgrad_step_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Every few iterations, do a subgradient step to make progress without LP
        if use_subgrad_between and t % 3 == 0:
            M_curr = fast_autocorr(y).max()
            if M_curr < 1e-12:
                break
            c = fast_autocorr(y)
            active = np.where(c >= M_curr * (1 - 1e-6))[0]
            g = compute_subgrad(y, active)
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                alpha = 2.0 / (t + 2.0)
                # Scale step to be relative to current scale
                step = alpha * y.sum() / (g_norm * np.sqrt(n))
                y_trial = y - step * g
                y_trial = np.maximum(y_trial, 0.0)
                s_yt = y_trial.sum()
                if s_yt > 1e-15:
                    y_trial = y_trial * norm / s_yt
                    sc = fast_autocorr(y_trial).max()
                    if sc < M_curr:
                        y = y_trial
                        if sc < best_score:
                            best_score = sc
                            best_a = y_trial.copy()
                            last_improve = time.monotonic()
                subgrad_step_count += 1
                continue

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        lp_call_count += 1

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Line search around Nesterov step
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_obj = fast_autocorr((1 - alpha_base) * y + alpha_base * g).max()
        for trial_factor in [0.5, 0.75, 1.25, 1.5, 2.0]:
            trial_alpha = min(alpha_base * trial_factor, 1.0)
            cand = (1 - trial_alpha) * y + trial_alpha * g
            cand = np.maximum(cand, 0.0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_alpha
        alpha = best_alpha

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:

            best_a = a.copy()

def coordinate_descent_analytic(a0, time_limit, seed=42):
    """Coordinate descent with analytical 2-point optimization.
    For each pair (src, dst), find the optimal fraction to transfer by
    minimizing the peak of the resulting autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:8]

        improved = False

        # Build peak influence mask
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks[:5]:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        # For each top peak, identify top contributors and try moving mass
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]

            low_contrib = np.where(~peak_mask)[0]
            if len(low_contrib) == 0:
                low_contrib = np.where(~peak_mask | (np.arange(n) != top_contrib[0]))[0]
            low_sorted = low_contrib[np.argsort(a[low_contrib])][:max(4, len(low_contrib) // 5)]

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                for dst in low_sorted[:5]:
                    # Try multiple fractions with analytical refinement
                    for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random perturbations
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            # Larger perturbation occasionally
            if stall % 50 == 0:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                try_update(a_trial)

            if stall > 500:
                break

    return best_a, best_score


def subgrad_fista(a0, time_limit, verbose=True):
    """FISTA-style accelerated subgradient method for the max-autocorrelation problem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_prev = a.copy()
    k = 1
    step_base = 1.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        c_y = fast_autocorr(y)
        M = c_y.max()
        if M < 1e-12:
            break

        active = np.where(c_y >= M * (1 - 1e-5))[0]
        g = compute_subgrad(y, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        alpha = step_base / np.sqrt(k)
        for _bt in range(10):
            x_new = y - alpha * g
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                alpha *= 0.5
                continue
            x_new = x_new * norm / s_x
            sc = fast_autocorr(x_new).max()
            if sc < M + alpha * g_norm * 2.0:
                break
            alpha *= 0.5

        # FISTA update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        y_new = x_new + ((k - 1.0) / t_k) * (x_new - x_prev)
        y_new = np.maximum(y_new, 0.0)
        s_yn = y_new.sum()
        if s_yn > 1e-15:
            y_new = y_new * norm / s_yn

        x_prev = x_new
        y = y_new
        k += 1

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        score_x = fast_autocorr(x_new).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()

        if verbose and k % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA k={k} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def spectral_perturb(a, rng, amount=0.1):
    """Perturb in frequency domain to change shape while preserving non-negativity."""
    n = len(a)
    a_s = a / a.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a_s, fft_size)
    # Random phase perturbation
    m = len(A)
    for i in range(1, m):
        A[i] *= (1.0 + amount * rng.standard_normal())
    a_new = irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lse_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    """LSE-smoothed L-BFGS-B optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining, 180.0)

        x0 = a.copy()

        def obj_func(x):
            aa = np.maximum(x.reshape(n), 0.0)
            s = aa.sum()
            if s < 1e-10:
                return 1e10
            b = fast_autocorr(aa)
            max_b = b.max()
            if max_b < 1e-15:
                return 1e10
            exp_vals = np.exp(T * (b - max_b))
            lse = max_b + np.log(exp_vals.sum()) / T
            return 2.0 * n * lse / (s * s)

        def grad_func(x):
            aa = np.maximum(x.reshape(n), 0.0)
            s = aa.sum()
            if s < 1e-10:
                return np.zeros(n)
            b = fast_autocorr(aa)
            max_b = b.max()
            if max_b < 1e-15:
                return np.zeros(n)
            exp_vals = np.exp(T * (b - max_b))
            w = exp_vals / exp_vals.sum()
            lse = max_b + np.log(exp_vals.sum()) / T

            fft_size = 1 << (3 * n - 2).bit_length()
            a_padded = np.zeros(fft_size)
            a_padded[:n] = aa
            w_padded = np.zeros(fft_size)
            w_padded[:2 * n - 1] = w
            A_fft = rfft(a_padded, fft_size)
            W_fft = rfft(w_padded, fft_size)
            cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

            conv_grad = 4.0 * cross
            obj_val = 2.0 * n * lse / (s * s)
            grad = conv_grad / (s * s) - 2.0 * obj_val * aa / s
            return grad.ravel()

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP-Nesterov at base length (main workhorse, ~40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_subgrad_between=True)
    do_try_update(a_out, f"lp_nesterov_n{n_base}")

    # Phase 2: Subgradient FISTA on best ~15% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FISTA subgrad n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_fista(a_norm, t2, verbose=True)
        do_try_update(a_out, f"fista_n{n_best}")

    # Phase 3: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-length exploration (budget={remaining:.0f}s) ===")
    lengths_to_try = []
    for dn in [-200, -100, 100, 200, 400]:
        nl = n_best + dn
        if 800 <= nl <= 3000 and nl != n_best:
            lengths_to_try.append(nl)
    lengths_to_try = sorted(set(lengths_to_try))

    for i_len, nl in enumerate(lengths_to_try):
        if time.monotonic() - t_global > time_budget * 0.85:
            break
        remaining_len = time_budget * 0.85 - (time.monotonic() - t_global)
        if remaining_len < 30:
            break
        t_len = min(remaining_len / max(1, len(lengths_to_try) - i_len), 180.0)
        if t_len < 15:
            break

        init = interpolate_sequence(best_seq, nl)
        a_n = init * np.sqrt(2.0 * nl) / init.sum()
        a_out, _ = nesterov_lp_optimize(a_n, t_len, solver, verbose=False, use_subgrad_between=True)
        do_try_update(a_out, f"lp_n{nl}")

    # Phase 4: LSE optimization on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t4, T_start=40.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 5: Spectral perturbation + re-optimize
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral perturb + reopt n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 60:
        t5 = min(remaining * 0.12, 150.0)
        if t5 > 20:
            for attempt in range(3):
                if time.monotonic() - t_global > time_budget * 0.92:
                    break
                pert = spectral_perturb(best_seq, rng, amount=0.05 * (attempt + 1))
                a_np = pert * np.sqrt(2.0 * n_best) / pert.sum()
                t_attempt = t5 / (3 - attempt)
                a_out, _ = nesterov_lp_optimize(a_np, t_attempt, solver, verbose=False, use_subgrad_between=True)
                do_try_update(a_out, f"spectral{attempt}_n{n_best}")

    # Phase 6: Coordinate descent polish (important for final improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_analytic(a_norm, t6, seed=12345)
        do_try_update(a_out, "coord_polish_1")

    # Phase 7: Another LP pass from polished result
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP from polished n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False, use_subgrad_between=True)
        do_try_update(a_out, f"lp_polished_n{n_best}")

    # Phase 8: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.5
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_analytic(a_norm, t8, seed=99999)
        do_try_update(a_out, "coord_final")

    # Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.4, 100.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lse_optimize(a_norm, t9, T_start=10.0, T_end=0.5, n_stages=3)
            do_try_update(a_out, "lse_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
