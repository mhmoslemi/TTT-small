def fast_conv(a, b=None):
    """Compute full convolution of a with b (or itself if b is None)."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = n + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def lp_nesterov_improved(a0, time_limit, solver, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_nonneg(a0, norm)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    y = a.copy()
    
    # EMA parameters
    ema_alpha = 0.05
    ema_avg = a.copy()
    
    # Full ergodic average
    erg_avg = a.copy()
    n_erg = 0
    
    t0 = time.monotonic()
    n_iters = 0
    m_inflation = 1.001  # Slightly inflate M
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_conv(y).max()
        if M_curr < 1e-14:
            break
        
        # Adaptive M inflation: larger early, smaller later
        frac = (time.monotonic() - t0) / time_limit
        m_inf = 1.005 * (1 - frac) + 1.0 + 0.001 * frac
        
        b = lp_best_response(y, M_curr * m_inf, solver)
        if b is None:
            # Try with exact M
            b = lp_best_response(y, M_curr, solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        t = n_iters
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new
        
        # Update averages
        erg_avg += y
        n_erg += 1
        erg_avg_norm = erg_avg * norm / max(erg_avg.sum(), 1e-30)
        
        ema_avg = (1 - ema_alpha) * ema_avg + ema_alpha * y
        ema_avg = np.maximum(ema_avg, 0.0)
        ema_avg = ema_avg * norm / max(ema_avg.sum(), 1e-30)
        
        # Check scores
        for candidate, label in [(y, 'y'), (erg_avg_norm, 'erg'), (ema_avg, 'ema')]:
            sc = fast_conv(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        # Periodic perturbation
        if n_iters % 200 == 0 and n_iters > 100:
            rng = np.random.default_rng(n_iters)
            pert = rng.normal(0, 0.01, n)
            y_pert = y + pert
            y_pert = np.maximum(y_pert, 0.0)
            y_pert = y_pert * norm / max(y_pert.sum(), 1e-30)

            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        # Find peak lags
        peak_lags = np.argsort(conv_val)[::-1][:8]
        improved = False

                continue
            # Find contributing pairs

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:

            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            
            # Source: position with highest contribution to this peak
            src = idx[np.argmax(contrib)]
            
            # Find all peak-contributing positions

            if len(candidates) == 0:

                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = normalize_nonneg(a_trial, norm)
                    score = fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 2000:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_polish(a0, time_limit, seed=99):
    """Subgradient method for max of bilinear forms with averaging."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_nonneg(a0, norm)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg = a.copy()
    n_avg = 0
    step_size = norm / n
    
    while time.monotonic() - t0 < time_limit:
        conv_val = fast_conv(a)
        # Find the argmax lag
        k_star = np.argmax(conv_val)
        M_star = conv_val[k_star]
        
        # Subgradient of max_k conv_k at a: it's the gradient of conv_{k*} at a
        # grad_k(a)[j] = a[k-j] (for valid k-j)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] = a[k_j]
        
        # Also account for the normalization constraint
        # We're minimizing max_k conv_k(a) subject to sum(a) = norm, a >= 0
        # Use projected subgradient
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_nonneg(a_new, norm)
        
        # Adaptive step size: decrease over time
        t = time.monotonic() - t0
        step_size = norm / (n + t * n / time_limit)
        
        a = a_new
        avg += a
        n_avg += 1
        
        if n_avg % 10 == 0:
            avg_curr = avg * norm / max(avg.sum(), 1e-30)
            sc = fast_conv(avg_curr).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_curr.copy()
        
        sc_a = fast_conv(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
    
    return best_a, best_score
