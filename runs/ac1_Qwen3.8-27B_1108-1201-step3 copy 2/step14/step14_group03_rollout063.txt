```
def spectral_init(n, alpha=0.0):
    L=2*n-1; M=1 << (L-1).bit_length()
    cflat=np.zeros(M); cflat[:L]=1.0/L
    C=rfft(cflat,M)
    m=np.arange(M//2+1)
    lam=np.pi*m*(n-1)/M
    real_proj=np.real(C*np.exp(2j*lam))
    mag=np.abs(C)+alpha
    phase=-lam + (np.pi/2)*(real_proj<0)
    A=np.sqrt(mag)*np.exp(1j*phase)
    # Nyquist
    if M%2==0:
       A[-1]=np.sqrt(max(0.0, float(np.real(C[-1]))))
    a=np.fft.irfft(A, M).real[:n]
    a=np.maximum(a,0)
    s=a.sum()
    if s<1e-8:

       a=np.maximum(np.fft.irfft(np.sqrt(np.abs(C)),M).real[:n],0)
       s=a.sum()
    if s<1e-8: return np.ones(n)/n
    a = a/s
    return a
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The method is primarily a best-ray one-coordinate local search.  For a fixed
coordinate i, we minimize the exact objective over the ray obtained by changing
a[i] and renormalizing the rest of the vector.  This ray minimization is done
by a coarse candidate sweep plus a short ternary refinement.

Auxiliary phases:
- log-domain mirror descent on a smoothed log-sum-exp surrogate,
- restarts from smoothed/noisy/spectral variants,
- a modest length-extension trial,
- final fine coordinate polishing.

The routine is designed to return the best sequence found before the time
budget expires.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


_COARSE_RATIOS = np.array(
    [0.0, 0.10, 0.25, 0.50, 0.75, 0.90, 0.95, 0.98, 0.99,
     1.01, 1.02, 1.05, 1.10, 1.25, 1.50, 2.00, 3.00, 5.00,
     10.0, 20.0, 50.0],
    dtype=np.float64
)
_BASE_RATIOS = np.array(
    [0.0, 0.05, 0.25, 1.0, 5.0, 20.0, 100.0, 500.0, 2000.0],
    dtype=np.float64
)
_ZERO_RATIOS = np.array(
    [0.0, 0.01, 0.05, 0.25, 1.0, 5.0, 20.0, 100.0],
    dtype=np.float64
)


def normalize_nonneg(a, target_sum=1.0):
    """Clip to [0, inf) and renormalize to target_sum."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64) * target_sum
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) * (target_sum / arr.size)
    return arr * (target_sum / s)


def fast_full_conv(a):
    """Linear convolution a*a via zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def prefix_suffix_max(c):
    """Prefix maxima and suffix maxima of c."""
    pref = np.maximum.accumulate(c)
    suff = np.maximum.accumulate(c[::-1])[::-1]
    return pref, suff


def generate_candidates(x0, base_rates):
    """Candidate unnormalized values x for coordinate update."""
    if x0 > 1e-14:
        x1 = np.array([x0], dtype=np.float64)
        X = np.concatenate((x0 * _COARSE_RATIOS, base_rates, x1))
    else:
        X = base_rates.copy()
    X = X[(X >= 0.0) & (X <= 1.0)]
    if X.size == 0:
        X = np.array([max(0.0, min(1.0, float(x0)))], dtype=np.float64)
    return X


def eval_coord_candidates(a, c, pref, suff, i, X):
    """
    Evaluate the normalized max-convolution for changing coordinate i to each
    value in X, assuming other coordinates are kept unchanged before
    renormalization.
    """
    n = a.size
    L = 2 * n - 1
    x0 = float(a[i])

    X = np.asarray(X, dtype=np.float64)
    dx = X - x0
    denom = 1.0 + dx
    valid = denom > 1e-12
    if not np.any(valid):
        return None
    if not np.all(valid):
        X = X[valid]
        dx = dx[valid]
        denom = denom[valid]

    left = pref[i - 1] if i > 0 else -np.inf
    right = suff[i + n] if i + n < L else -np.inf
    out = float(left if left > right else right)

    c_seg = c[i:i + n]
    l_seg = 2.0 * a
    l_seg = l_seg.copy()
    l_seg[i] = 0.0

    with np.errstate(over="ignore", divide="ignore", invalid="ignore"):
        vals = (c_seg[None, :] + dx[:, None] * l_seg[None, :]) / denom[:, None]

    vals[:, i] = -np.inf
    inside = np.max(vals, axis=1)

    dx2 = X * X - x0 * x0
    pval = (c[2 * i] + dx2) / denom
    total = np.maximum(inside, pval)

    if np.isfinite(out):
        total = np.maximum(total, out / denom)

    total[~np.isfinite(total)] = np.inf
    return total


def eval_coord_scalar(a, c, pref, suff, i, x):
    """Scalar version of coordinate-ray objective evaluation."""
    n = a.size
    L = 2 * n - 1
    x0 = float(a[i])
    x = float(x)

    denom = 1.0 + (x - x0)
    if denom <= 1e-12:
        return np.inf

    dx = x - x0
    dx2 = x * x - x0 * x0

    left = pref[i - 1] if i > 0 else -np.inf
    right = suff[i + n] if i + n < L else -np.inf
    out = float(left if left > right else right)

    # Max inside the affected segment, excluding the pure square lag 2*i.
    arr = c[i:i + n] + dx * (2.0 * a)
    arr[i] = -np.inf
    inside = float(arr.max())

    pval = (c[2 * i] + dx2) / denom
    total = inside if inside > pval else pval

    if np.isfinite(out):
        tv = out / denom
        if tv > total:
            total = tv

    if not np.isfinite(total):
        return np.inf
    return float(total)


def refine_coord_best(a, c, pref, suff, i, best_x, base):
    """Short ternary refinement around the best coordinate-ray candidate."""
    best_val = eval_coord_scalar(a, c, pref, suff, i, best_x)
    x0 = float(a[i])

    if best_x > 1e-12:
        low = max(0.0, best_x - max(2.0 * base, 0.25 * best_x))
        high = min(1.0, best_x + max(2.0 * base, 0.50 * best_x))
    else:
        low = 0.0
        high = min(1.0, max(20.0 * base, 0.05, 2.0 * x0))

    if not (high > low):
        return best_val, best_x

    for _ in range(18):
        m1 = low + (high - low) / 3.0
        m2 = high - (high - low) / 3.0
        f1 = eval_coord_scalar(a, c, pref, suff, i, m1)
        f2 = eval_coord_scalar(a, c, pref, suff, i, m2)
        if f1 < f2:
            high = m2
        else:
            low = m1
        if (high - low) < base * 1e-7:
            break

    xm = 0.5 * (low + high)
    fm = eval_coord_scalar(a, c, pref, suff, i, xm)
    if fm < best_val:
        return fm, xm
    return best_val, best_x


def apply_coord_update(a, c, i, x):
    """
    Apply the coordinate-ray update:
      - set unnormalized coordinate i to x,
      - keep other coordinates unchanged,
      - renormalize the whole sequence to sum 1,
      - update c = a*a exactly (up to floating point).
    """
    n = a.size
    L = 2 * n - 1
    x0 = float(a[i])
    x = float(x)

    denom = 1.0 + (x - x0)
    if denom <= 1e-12:
        return

    inv = 1.0 / denom
    inv_scale = inv * inv
    dx = x - x0
    dx2 = x * x - x0 * x0

    if i > 0:
        c[:i] *= inv_scale

    seg = c[i:i + n]
    new_seg = seg.copy()
    new_seg += dx * (2.0 * a)
    new_seg[i] += dx2
    c[i:i + n] = new_seg * inv_scale

    if i + n < L:
        c[i + n:] *= inv_scale

    a[i] = x
    a *= inv

    ss = float(a.sum())
    if ss > 1e-15 and abs(ss - 1.0) > 1e-12:
        invss = 1.0 / ss
        a *= invss
        c *= invss * invss


def coordinate_descent(a_input, c_input, time_limit, rng, verbose=True,
                       label="", improve_tol=1e-12):
    """
    Best-ray one-coordinate descent for the normalized objective
    max_k (a*a)_k.  Returns (a, c, score), where score = 2*n*max(c).
    """
    a = normalize_nonneg(a_input, target_sum=1.0)
    n = a.size
    if n == 0:
        a = np.array([1.0], dtype=np.float64)
        n = 1

    if c_input is None:
        c = fast_full_conv(a)
    else:
        c = np.asarray(c_input, dtype=np.float64).ravel()
        if c.size != 2 * n - 1 or not np.all(np.isfinite(c)):
            c = fast_full_conv(a)

    L = 2 * n - 1
    if c.sum() < 1e-6:
        c = fast_full_conv(a)

    if not np.all(np.isfinite(c)):
        c = fast_full_conv(a)

    pref, suff = prefix_suffix_max(c)
    cur_max = float(c.max())
    base = 1.0 / float(n)
    base_rates = base * _BASE_RATIOS
    base_rates = base_rates[(base_rates >= 0.0) & (base_rates <= 1.0)]

    best_a = a.copy()
    best_c = c.copy()
    best_max = cur_max

    t0 = time.monotonic()
    pass_num = 0
    no_improve_passes = 0
    total_improvements = 0

    while (time.monotonic() - t0) < time_limit:
        pass_num += 1
        pass_improvements = 0
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            x0 = float(a[i])
            X = generate_candidates(x0, base_rates)
            totals = eval_coord_candidates(a, c, pref, suff, i, X)
            if totals is None:
                continue

            idx = int(np.argmin(totals))
            val = float(totals[idx])

            if val < cur_max - improve_tol:
                xbest = float(X[idx])
                val, xbest = refine_coord_best(a, c, pref, suff, i, xbest, base)

                if val < cur_max - 0.5 * improve_tol:
                    apply_coord_update(a, c, i, xbest)
                    pref, suff = prefix_suffix_max(c)
                    cur_max = float(pref[-1])
                    pass_improvements += 1
                    total_improvements += 1

                    if cur_max < best_max - improve_tol:
                        best_max = cur_max
                        best_a = a.copy()
                        best_c = c.copy()
                        if verbose:
                            print(
                                f"  [{label}] coord improve: "
                                f"score={2.0 * n * best_max:.10f} "
                                f"n={n} imp={total_improvements}"
                            )

        if pass_improvements == 0:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

        if pass_num % 10 == 0:
            a = normalize_nonneg(a, target_sum=1.0)
            c = fast_full_conv(a)
            pref, suff = prefix_suffix_max(c)
            cur_max = float(c.max())
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                best_c = c.copy()
                if verbose:
                    print(
                        f"  [{label}] coord pass {pass_num}: "
                        f"score={2.0 * n * best_max:.10f}"
                    )

    # Final sanity recompute for the best stored pair.
    best_a = normalize_nonneg(best_a, target_sum=1.0)
    best_c = fast_full_conv(best_a)
    best_max = float(best_c.max())
    return best_a, best_c, 2.0 * n * best_max


def softmax_stable(c, T):
    """Stable softmax over convolution coefficients."""
    c = np.asarray(c, dtype=np.float64)
    T = float(T)
    if T <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w

    q = (c - float(c.max())) / T
    q = q - float(q.max())
    w = np.exp(q)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / s


def hankel_product(a, w):
    """
    Compute g_i = sum_t a_t w_{i+t} for i=0..n-1, where w has length 2n-1.
    This is the unweighted gradient factor for log-sum-exp of a*a.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    conv_len = 3 * n - 2
    size = 1 << (conv_len - 1).bit_length()
    ar = a[::-1]
    full = irfft(rfft(ar, size) * rfft(w, size), size)
    return full[n - 1:n - 1 + n]


def log_descent(a_input, time_limit, rng, verbose=True, label=""):
    """
    Log-domain mirror descent on smoothed log-sumexp surrogates.
    Returns (a, c, score).
    """
    a = normalize_nonneg(a_input, target_sum=1.0)
    n = a.size
    if n == 0:
        a = np.array([1.0], dtype=np.float64)
        n = 1

    c = fast_full_conv(a)
    cur_max = float(c.max())

    best_a = a.copy()
    best_c = c.copy()
    best_max = cur_max

    t0 = time.monotonic()
    if best_max <= 1e-15:
        return best_a, best_c, 2.0 * n * best_max

    rel = np.array([0.40, 0.20, 0.10, 0.05, 0.02, 0.01], dtype=np.float64)
    temps = best_max * rel
    temps = np.maximum(temps, 1e-10)

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break

        eta = 0.5
        stagnation = 0

        while time.monotonic() - t0 < time_limit:
            c = fast_full_conv(a)
            max_c = float(c.max())

            if max_c < best_max - 1e-12:
                best_max = max_c
                best_a = a.copy()
                best_c = c.copy()
                if verbose:
                    print(f"  [{label}] log current best: score={2.0 * n * best_max:.10f}")

            w = softmax_stable(c, T)
            hprod = hankel_product(a, w)
            g = 2.0 * hprod
            mean_g = float(np.dot(a, g))
            h = a * (g - mean_g) / T
            h = np.clip(h, -30.0, 30.0)

            cand = a * np.exp(-eta * h)
            cand = normalize_nonneg(cand, target_sum=1.0)
            cc = fast_full_conv(cand)
            new_max = float(cc.max())

            if new_max < max_c - 1e-14:
                a = cand
                c = cc
                max_c = new_max
                stagnation = 0
                if eta < 2.0:
                    eta *= 1.05
                if max_c < best_max - 1e-12:
                    best_max = max_c
                    best_a = a.copy()
                    best_c = c.copy()
                    if verbose:
                        print(f"  [{label}] log improve: score={2.0 * n * best_max:.10f} T={T:.3e}")
            else:
                eta *= 0.5
                stagnation += 1
                if stagnation > 40 or eta < 1e-8:
                    break

    best_a = normalize_nonneg(best_a, target_sum=1.0)
    best_c = fast_full_conv(best_a)
    best_max = float(best_c.max())
    return best_a, best_c, 2.0 * n * best_max


def spectral_smooth(a, sigma):
    """Circular Gaussian smoothing in the DFT domain."""
    a = normalize_nonneg(a, target_sum=1.0)
    n = a.size
    if n < 3 or sigma <= 0:
        return a
    f = np.fft.rfftfreq(n) * n
    gain = np.exp(-0.5 * (f / float(sigma)) ** 2)
    b = np.fft.irfft(np.fft.rfft(a) * gain, n).real
    return normalize_nonneg(b, target_sum=1.0)


def noisy_variant(a, sigma, rng):
    """Multiplicative lognormal perturbation."""
    a = normalize_nonneg(a, target_sum=1.0)
    n = a.size
    mult = np.exp(rng.normal(0.0, float(sigma), n))
    b = a * mult
    return normalize_nonneg(b, target_sum=1.0)


def interpolate_sequence(a, new_n):
    """Linear interpolation/resampling to a new length."""
    a = normalize_nonneg(a, target_sum=1.0)
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    return normalize_nonneg(b, target_sum=1.0)


def spectral_init(n, alpha=0.0):
    """
    Spectral initialization: take a flat impulse response target, square-root
    its DFT with a half-linear phase, inverse transform, then take the
    non-negative part and normalize.
    """
    if n <= 1:
        return np.array([1.0], dtype=np.float64)

    L = 2 * n - 1
    M = 1 << (L - 1).bit_length()

    c_flat = np.zeros(M, dtype=np.float64)
    c_flat[:L] = 1.0 / float(L)
    C = rfft(c_flat, M)

    m = np.arange(M // 2 + 1, dtype=np.float64)
    lam = np.pi * m * (n - 1) / float(M)

    # C should be real_times e^{-i 2 lam}; use its sign to choose half-phase.
    real_proj = np.real(C * np.exp(2j * lam))
    mag = np.abs(C) + float(alpha)
    phase = -lam + (np.pi / 2.0) * (real_proj < 0.0)

    A = np.sqrt(np.maximum(mag, 0.0)) * np.exp(1j * phase)

    # Nyquist frequency must be real for a real inverse transform.
    if M % 2 == 0:
        nyq_real = float(np.real(C[-1]))
        A[-1] = np.sqrt(max(0.0, nyq_real)) + 0j

    a = np.fft.irfft(A, M).real[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())

    if s < 1e-10:
        # Fallback: positive part of magnitude-only square root.
        A2 = np.sqrt(np.abs(C))
        A2[-1] = float(np.sqrt(max(0.0, float(np.real(C[-1])))))
        a = np.fft.irfft(A2, M).real[:n]
        a = np.maximum(a, 0.0)
        s = float(a.sum())

    if s < 1e-10:
        return np.ones(n, dtype=np.float64) / float(n)

    return a / s


def blend_sequences(a, b, alpha):
    """Convex blend of two nonnegative sequences of possibly different length."""
    a = normalize_nonneg(a, target_sum=1.0)
    b = normalize_nonneg(b, target_sum=1.0)
    if a.size != b.size:
        b = interpolate_sequence(b, a.size)
    c = (1.0 - float(alpha)) * a + float(alpha) * b
    return normalize_nonneg(c, target_sum=1.0)


def evaluate_safe(a):
    """Call the external evaluator safely."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        vals = [float(x) for x in arr]
        if len(vals) == 0:
            return np.inf
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search driver.  Must return a Python list of non-negative floats.
    """
    rng = np.random.default_rng(int(seed))
    t_start = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 60.0)

    best_a = None
    best_c = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Load parent construction if available.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-12:
                parent = parent / float(parent.sum())
                parent_c = fast_full_conv(parent)
                parent_score = 2.0 * parent.size * float(parent_c.max())
                best_a = parent
                best_c = parent_c
                best_score = parent_score
                print(f"Loaded parent: n={parent.size}, score={best_score:.10f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback initialization.
    # ------------------------------------------------------------------
    if best_a is None:
        n0 = 1600
        sp = spectral_init(n0, alpha=0.1)
        init = blend_sequences(sp, np.ones(n0, dtype=np.float64) / float(n0), 0.25)
        best_a = init
        best_c = fast_full_conv(best_a)
        best_score = 2.0 * best_a.size * float(best_c.max())
        print(f"Default init: n={best_a.size}, score={best_score:.10f}")

    def consider(seq, c, score, label):
        # This is an inner helper for bookkeeping only; it does not use
        # closures except for explicit passed values.  To obey the no-closure
        # rule strictly, avoid using it; inline updates below instead.
        pass

    # ------------------------------------------------------------------
    # Phase 1: coordinate descent from parent.
    # ------------------------------------------------------------------
    remaining = time_budget - (time.monotonic() - t_start)
    lim = min(remaining * 0.35, 900.0)
    if lim > 30.0:
        a, c, s = coordinate_descent(
            best_a, best_c, lim, rng,
            verbose=True, label="coord1", improve_tol=1e-12
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a
            best_c = c
            print(f"Phase 1 best: score={best_score:.10f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Phase 2: log-domain mirror descent from current best.
    # ------------------------------------------------------------------
    remaining = time_budget - (time.monotonic() - t_start)
    lim = min(remaining * 0.15, 300.0)
    if lim > 30.0:
        a, c, s = log_descent(best_a, lim, rng, verbose=True, label="log1")
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a
            best_c = c
            print(f"Phase 2 best: score={best_score:.10f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Phase 3: coordinate descent again after smoothing.
    # ------------------------------------------------------------------
    remaining = time_budget - (time.monotonic() - t_start)
    lim = min(remaining * 0.30, 800.0)
    if lim > 30.0:
        a, c, s = coordinate_descent(
            best_a, best_c, lim, rng,
            verbose=True, label="coord2", improve_tol=1e-12
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a
            best_c = c
            print(f"Phase 3 best: score={best_score:.10f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Phase 4: restarts from variants and modest length extension.
    # ------------------------------------------------------------------
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 120.0:
        nbest = best_a.size
        variants = []

        v1 = spectral_smooth(best_a, 2.0)
        variants.append((v1, "smooth2"))

        v2 = spectral_smooth(best_a, 8.0)
        variants.append((v2, "smooth8"))

        v3 = noisy_variant(best_a, 0.10, rng)
        variants.append((v3, "noise10"))

        sp = spectral_init(nbest, alpha=0.25)
        v4 = blend_sequences(best_a, sp, 0.35)
        variants.append((v4, "spectral_blend"))

        if nbest <= 1800:
            n2 = int(min(2400, nbest + max(200, nbest // 8)))
            v5 = interpolate_sequence(best_a, n2)
            variants.append((v5, f"len{n2}"))

        variants = variants[:5]

        if variants:
            reserve = 60.0
            avail = time_budget - (time.monotonic() - t_start)
            t_var = max(30.0, (avail - reserve) / max(1, len(variants)))
            t_var = min(t_var, 280.0)

            for vseq, vname in variants:
                elapsed = time.monotonic() - t_start
                rem = time_budget - elapsed
                if rem < reserve + 30.0:
                    break

                lim = min(t_var, rem - reserve - 30.0)
                if lim < 20.0:
                    break

                vseq = normalize_nonneg(vseq, target_sum=1.0)
                vc = fast_full_conv(vseq)
                vscore = 2.0 * vseq.size * float(vc.max())

                if not np.isfinite(vscore):
                    continue

                # If the variant is clearly worse, spend a little time on a
                # smoothed log-descent first to pull it into a reasonable basin.
                if vseq.size == best_a.size and vscore > best_score * 1.25:
                    mid = min(lim * 0.35, 120.0)
                    if mid > 20.0:
                        vseq, vc, vscore = log_descent(
                            vseq, mid, rng,
                            verbose=False, label=vname + "_log"
                        )
                        if np.isfinite(vscore) and vscore < best_score:
                            best_score = vscore
                            best_a = vseq
                            best_c = vc
                            print(f"Variant [{vname}] log improved: score={best_score:.10f}")

                elapsed = time.monotonic() - t_start
                rem = time_budget - elapsed
                lim2 = max(20.0, rem - reserve - 20.0)

                a, c, s = coordinate_descent(
                    vseq, vc, lim2, rng,
                    verbose=False, label=vname, improve_tol=1e-12
                )
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_a = a
                    best_c = c
                    print(f"Variant [{vname}] improved: score={best_score:.10f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Phase 5: final fine coordinate polishing.
    # ------------------------------------------------------------------
    remaining = time_budget - (time.monotonic() - t_start)
    reserve = 45.0
    if remaining > reserve + 30.0:
        lim = remaining - reserve
        a, c, s = coordinate_descent(
            best_a, best_c, lim, rng,
            verbose=True, label="final", improve_tol=1e-13
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a
            best_c = c
            print(f"Final polish best: score={best_score:.10f}, n={best_a.size}")

    # ------------------------------------------------------------------
    # Final safety and external evaluation.
    # ------------------------------------------------------------------
    if best_a is None:
        return [1.0]

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    best_a = normalize_nonneg(best_a, target_sum=1.0)

    ext_score = evaluate_safe(best_a)
    if np.isfinite(ext_score) and ext_score < best_score:
        best_score = ext_score

    print(f"FINAL returned score estimate={best_score:.10f}, n={len(best_a)}")
    return [float(x) for x in best_a]
```
