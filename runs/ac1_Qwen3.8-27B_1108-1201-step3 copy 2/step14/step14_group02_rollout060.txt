    where b = conv(a, a).

    y = a.copy()

    fft_size = 1 << (3 * n - 1 + 1).bit_length()

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(a, ac, tau, fft_size):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a using FFT."""
    n = len(a)
    ac_shifted = ac - ac.max()
    p = np.exp(tau * ac_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(np.conj(A) * P, fft_size)
    return corr[:n]

def lp_frank_wolfe_step(a, M, solver, n_lp_vars=None):
    """Single LP Frank-Wolfe step: find best response direction."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build T matrix: (T @ b)[k] = sum_j a[j] * b[k-j]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    constraints = [T @ b_var <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate-wise polish: try moving mass from high to low contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 500
    top_k_count = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find peak lags
        peak_lags = np.argsort(ac)[::-1][:top_k_count]
        improved = False

        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination using peak sensitivity
            best_dst = -1
            best_score = float('inf')

            # Sample candidate destinations
            n_cand = min(n, 300)
            candidates = rng.integers(0, n, size=n_cand)

            for dst in candidates:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    d = k2i - dst
                    if 0 <= d < n:
                        sens += a[d]
                    d2 = dst - k2i
                    if 0 <= d2 < n:
                        sens += a[d2]

                rem_sens = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    d = k2i - src
                    if 0 <= d < n:
                        rem_sens += a[d]
                    d2 = src - k2i
                    if 0 <= d2 < n:
                        rem_sens += a[d2]

                net = sens - rem_sens
                if net < best_score:
                    best_score = net
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()

    if verbose:
        J = 2.0 * n * best_M
        print(f"    Coord polish ({time.monotonic()-t0:.1f}s): J={J:.7f}")

    return best_a, best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.7f}")

    n0 = len(best_seq)

    # === Phase 1: FISTA smooth gradient at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA smooth grad n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_optimize_v2(a_in, t1, tau_init=2.0, tau_end=80.0, verbose=True)
        try_update(a_out, f"fista_n{n0}")

    # === Phase 2: Heavy-ball smooth gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Heavy-ball NGD n={n0} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_optimize(a_in, t2, tau_init=3.0, tau_max=100.0, verbose=True)
        try_update(a_out, f"ngd_n{n0}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Frank-Wolfe n={n0} ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # === Phase 4: Coordinate polish at current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord polish n={n_cur} ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Scale to longer sequence ===
    n1 = max(n0 + 400, 2000)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: FISTA n={n1} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        a_out, _ = nesterov_smooth_optimize_v2(a_in, t5, tau_init=2.0, tau_end=60.0, verbose=True)
        try_update(a_out, f"fista_n{n1}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, min(remaining * 0.3, 60.0), verbose=False)
            try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Even longer ===
    n2 = max(n1 + 400, 2800)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: FISTA n={n2} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 30:
        a_long2 = interpolate_sequence(best_seq, n2)
        a_in = norm_sum1(a_long2)
        a_out, _ = nesterov_smooth_optimize_v2(a_in, t6, tau_init=2.0, tau_end=50.0, verbose=True)
        try_update(a_out, f"fista_n{n2}")

    # === Phase 7: Final FISTA polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final FISTA n={n_best} ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_optimize_v2(a_in, t7, tau_init=5.0, tau_end=150.0, verbose=True)
        try_update(a_out, f"fista_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final FISTA with very high tau ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final FISTA high-tau n={n_best} ===")
        t9 = min(remaining * 0.3, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_smooth_optimize_v2(a_in, t9, tau_init=20.0, tau_end=300.0, verbose=True)
            try_update(a_out, f"fista_hightau_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
