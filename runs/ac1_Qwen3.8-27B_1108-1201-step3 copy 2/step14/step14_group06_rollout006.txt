def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a, n):
    """Build the cross-correlation constraint matrix T where (T @ b)[k] = sum_i a[k-i]*b[i]."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    # Try to get dual variables
    dual_vals = None
    try:
        if prob.constraints[0].dual_value is not None:
            dual_vals = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_vals


def lse_obj_grad(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_optimize(a0, time_limit, temps=(100.0, 30.0, 10.0, 3.0, 1.0)):
    """Multi-temperature LSE optimization with L-BFGS-B."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def obj_func(x):
            val, _ = lse_obj_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_obj_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 200, 'maxfun': 800,
                                          'gtol': 1e-9, 'ftol': 1e-12, 'maxls': 40})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                sc = a_cand.sum()
                if sc > 1e-10:
                    a_cand = a_cand / sc
                    score_c = 2.0 * n * fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        print(f"    LSE T={T:.1f}: score={best_score:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_nest = a.copy()
    n_iters = 0
    last_improve = t0
    stall_rounds = 0
    rng = np.random.default_rng(12345)

    for t in range(1, n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current max autocorrelation
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-15:
            break

        # Cosine M schedule: start with small reduction, increase over time
        progress = min(1.0, t / (0.7 * n_max_iter))
        eps = 0.02 + 0.12 * (1.0 - np.cos(np.pi * progress)) / 2.0
        M_used = M_curr * (1.0 - eps)

        # Build constraint matrix and solve LP
        T = build_T(y, n)
        b, dual = lp_best_response(y, M_used, solver, T)
        if b is None:
            # Retry with looser M
            M_used2 = M_curr * (1.0 - 0.01)
            b, _ = lp_best_response(y, M_used2, solver, T)
            if b is None:
                b = lp_best_response(y, M_curr, solver, T)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b / b_sum
        n_iters += 1

        # Nesterov step size
        alpha = 2.0 / (t + 2.0)

        # Adapt alpha based on improvement
        M_new = fast_autocorr(g).max()
        if M_curr > 1e-15 and M_new > 1e-15:
            ratio = M_new / M_curr
            if ratio < 0.9:
                alpha = min(alpha * 1.5, 1.0)
            elif ratio > 1.1:
                alpha = alpha * 0.5

        # Nesterov update
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a_nest = a_new

        # Periodic diversification
        if t % 200 == 0 and t > 500:
            noise_scale = 0.005 * np.mean(y)
            y_pert = y + rng.normal(0, noise_scale, n)
            y_pert = np.maximum(y_pert, 0.0)
            sp = y_pert.sum()
            if sp > 1e-10:
                y_pert = y_pert / sp
                score_pert = 2.0 * n * fast_autocorr(y_pert).max()
                if score_pert < 2.0 * n * M_curr:
                    y = y_pert

        # Check improvement
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_rounds = 0
        else:
            stall_rounds += 1

        score_a = 2.0 * n * fast_autocorr(a_nest).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a_nest.copy()
            last_improve = time.monotonic()
            stall_rounds = 0

        if time.monotonic() - last_improve > 90.0:
            if stall_rounds > 200:
                break

        if n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def group_coord_descent(a0, time_limit, seed=42):
    """
    Group coordinate descent: identify peak-contributing pairs and move mass
    from high-product pairs to low-product pairs.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Find contributing pairs for the peak lag
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        # Also look at top few peaks
        top_peaks = np.argsort(b)[::-1][:8]

        improved = False
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            contrib2 = a[idx2] * a[k2 - idx2]

            # Sort by contribution (high to low)
            sorted_idx = idx2[np.argsort(contrib2)[::-1]]
            high_contrib = sorted_idx[:10]
            low_contrib = sorted_idx[-10:]

            for src in high_contrib:
                if a[src] < 1e-8:
                    continue
                for dst in low_contrib:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-10:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random pair moves
            for _ in range(200):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.005, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Monotonically increasing (exploits asymmetry)
        x = np.linspace(0.0, 1.0, n)
        a = x ** 2
    else:
        # Bimodal with gap
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.2) ** 2) + np.exp(-20.0 * (x - 0.8) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fine_tune(a0, time_limit, seed=0):
    """
    Very fine-grained optimization: try tiny perturbations of individual
    coordinates and small groups.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Focus on coordinates that are "active" (non-negligible)
    active = np.where(a > 1e-6)[0]
    if len(active) == 0:
        active = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        peak_idx = set(np.arange(js, je + 1).tolist())

        # Try small multiplicative perturbations of peak-contributing coords
        improved = False
        candidates = list(peak_idx) if len(peak_idx) < 200 else list(rng.choice(list(peak_idx), 100, replace=False))
        rng.shuffle(candidates)

        for i in candidates[:50]:
            if i >= n or a[i] < 1e-8:
                continue
            for mult in [0.97, 0.985, 1.015, 1.03]:
                a_trial = a.copy()
                a_trial[i] *= mult
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try swap-based moves: redistribute between two coords
            for _ in range(100):
                i = rng.choice(active[:min(len(active), n)])
                j = rng.integers(n)
                if i == j:
                    continue
                delta = rng.choice([-1, 1]) * rng.uniform(0.001, 0.01) * a.max()
                a_trial = a.copy()
                a_trial[i] += delta
                a_trial[j] -= delta
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break  # Local minimum reached at this granularity

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update(candidate_a, label):
        nonlocal best_seq, best_score
        if candidate_a is None:
            return
        a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Main LP-Nesterov (50% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.48
    a_norm = best_seq.copy()  # already normalized to sum=1
    a_out, _ = nesterov_lp(a_norm, t1, solver, n_max_iter=50000)
    try_update(a_out, f"lp1_n{n_best}")

    # Phase 2: LSE optimization (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = lse_optimize(best_seq.copy(), t2, temps=(100.0, 30.0, 10.0, 3.0, 1.0))
        try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Group coordinate descent (12% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Group Coord Descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = group_coord_descent(best_seq.copy(), t3, seed=42)
        try_update(a_out, f"coord_n{n_best}")
        # Second pass with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t3b = min(remaining * 0.08, 100.0)
            if t3b > 10:
                a_out, _ = group_coord_descent(best_seq.copy(), t3b, seed=777)
                try_update(a_out, f"coord2_n{n_best}")

    # Phase 4: LP-Nesterov round 2 (12% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 30:
        a_out, _ = nesterov_lp(best_seq.copy(), t4, solver, n_max_iter=30000)
        try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Multi-length exploration (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    lengths_to_try = [n_best + 400, n_best + 800]
    for extra_len in lengths_to_try:
        n_try = n_best + extra_len
        if n_try > 4000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(remaining * 0.35, 180.0)
        print(f"  Trying n={n_try} (budget={t5:.0f}s)")
        init_long = interpolate_seq(best_seq, n_try)
        a_out, _ = nesterov_lp(init_long, t5, solver, n_max_iter=30000)
        try_update(a_out, f"lp_long_n{n_try}")
        if a_out is not None:
            # Quick LSE on the longer sequence
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5b = min(remaining * 0.2, 60.0)
                a_out2, _ = lse_optimize(a_out, t5b, temps=(50.0, 10.0, 3.0))
                try_update(a_out2, f"lse_long_n{n_try}")

    # Phase 6: Fine-tune (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Fine-tune n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 15:
        t6 = min(remaining * 0.4, 120.0)
        a_out, _ = fine_tune(best_seq.copy(), t6, seed=0)
        try_update(a_out, "fintune1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.3, 100.0)
            a_out, _ = fine_tune(best_seq.copy(), t6b, seed=999)
            try_update(a_out, "fintune2")

        # Final LSE with very low temperature
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.5, 80.0)
            a_out, _ = lse_optimize(best_seq.copy(), t6c, temps=(5.0, 2.0, 1.0, 0.5))
            try_update(a_out, "lse_final")

        # Final coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = group_coord_descent(best_seq.copy(), remaining, seed=31415)
            try_update(a_out, "coord_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
