def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T, n=None):
    if n is None:
        n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    Y_rev = np.fft.rfft(y[::-1], N)
    cross = np.fft.irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build the convolution constraint matrix efficiently
    # conv(a, b)[k] <= M for k = 0, ..., 2n-2
    # conv(a,b)[k] = sum_i a[i] * b[k-i] (valid indices)
    # For each k, the constraint is: sum_{i in valid} a[i] * b[k-i] <= M
    # This is linear in b: for each lag k, coefficient of b[j] is a[k-j] if valid
    
    # Use sparse matrix for efficiency
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    T_mat = cp.nominal_sparse(rows, cols, vals, (2 * n - 1, n))
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # For each lag k, conv(a,b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j]*b[j]
    # We build this as a series of row operations
    # Row k of the matrix has entries: at column j, the value is a[k-j] if 0<=k-j<n
    
    # More efficient: use the fact that each row is a (shifted) copy of a
    # Row k: columns j from max(0,k-n+1) to min(n-1,k), values a[k-j]
    
    # Build as sparse
    n_rows = 2 * n - 1
    nnz = sum(min(n-1, k) - max(0, k-n+1) + 1 for k in range(n_rows))
    
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    
    T_mat = cp.cvxpy.constraints.parameters.Parameter((n_rows, n), value=np.zeros((1,1)))
    # Actually, let's just use the sparse matrix directly
    import scipy.sparse as sp
    T_sparse = sp.coo_matrix((vals, (rows, cols)), shape=(n_rows, n)).tocsr()
    
    T_cvx = cp.Constant(T_sparse)
    constraints = [T_cvx @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_adaptive(a0, time_limit, solver, verbose=True):
    """LP best-response with adaptive Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    best_lp_score = best_score
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size: use backtracking to ensure descent
        alpha = 2.0 / (t + 2.0)
        
        # Try multiple step sizes
        best_trial = None
        best_trial_score = fast_autocorr(y).max()
        
        for trial_alpha in [alpha, alpha * 0.5, alpha * 1.5, alpha * 2.0]:
            y_trial = (1.0 - trial_alpha) * y + trial_alpha * g
            y_trial = np.maximum(y_trial, 0.0)
            s_y = y_trial.sum()
            if s_y < 1e-15:
                continue
            y_trial = y_trial * norm / s_y
            sc = fast_autocorr(y_trial).max()
            if sc < best_trial_score - 1e-12:
                best_trial_score = sc
                best_trial = y_trial
        
        if best_trial is not None:
            y_new = best_trial
        else:
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall = 0

        # Sliding window average
        if n_avg > 200 and n_avg % 100 == 0:
            # Use last 200 iterates (approximate via current average minus old)
            # Simplified: just check the current a
            pass

        if verbose and n_iters % 80 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f} stall={stall}")

        if stall > 5000:
            break

    return best_a, best_score


def subgradient_polyak(a0, time_limit, verbose=True):
    """Subgradient method with Polyak step size for the max-autocorrelation problem."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    # Polyak requires an upper bound; use current score as proxy
    J_star_est = best_score * 0.99  # optimistic lower bound on improvement
    
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Find all active peaks (within 1% of max)
        threshold = ac_max * 0.99
        active_peaks = np.where(ac >= threshold)[0]
        
        # Compute subgradient: average of gradients at active peaks
        # grad of ac[p] w.r.t. a[i] = a[p-i] (for valid i)
        grad = np.zeros(n)
        for p in active_peaks:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            for j in range(js, je + 1):
                i = p - j
                grad[j] += a[i]
        grad /= len(active_peaks)
        
        # Polyak step size
        J_current = ac_max
        gap = J_current - J_star_est
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        step = gap / grad_norm_sq
        step = min(step, 0.1)  # cap step size
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still move, just don't update best
            stall += 1
        
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_autocorr(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f} stall={stall}")
        
        if stall > 3000:
            # Reduce aggressiveness
            J_star_est = best_score * 0.999
            stall = 0
        
        if stall > 10000:
            break
    
    return best_a, best_score


def peak_targeted_local_search(a0, time_limit, seed=42, verbose=True):
    """Local search that targets the top peaks of the autocorrelation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    # Pre-compute which indices contribute to which peaks
    # For a peak at position p, a[j] contributes with weight a[p-j]
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Get top peaks
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argsort(ac)[::-1][:top_k]
        
        improved = False
        
        for p in peak_indices:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            
            # Contribution of each a[j] to this peak: a[j] * a[p-j]
            contrib = a[idx] * a[p - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Source: index with highest contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Destination: find indices that contribute least to ANY top peak
            # Build a "damage" matrix: for each position j, how much does it contribute to top peaks?
            damage = np.zeros(n)
            for p2 in peak_indices:
                js2 = max(0, p2 - n + 1)
                je2 = min(n - 1, p2)
                for j in range(js2, je2 + 1):
                    i2 = p2 - j
                    damage[j] += a[j] * a[i2]
            
            # Find low-damage positions (excluding src)
            damage[src] = np.inf
            # Also prefer positions with low current value (less disruption)
            sorted_dst = np.argsort(damage + 0.01 * a)
            n_candidates = max(3, n // 10)
            candidates = sorted_dst[:n_candidates]
            
            dst = candidates[rng.integers(len(candidates))]
            
            # Try different fractions
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape local minimum
            stall += 1
            n_moves = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_moves):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > norm * 1e-8:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 2000:
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Local search done ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def multi_peak_lp_reduce(a0, time_limit, solver, verbose=True):
    """Instead of single-M LP, solve with multiple M values and average."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        # Solve LP with M = M_curr
        b1 = lp_best_response(y, M_curr, solver)
        if b1 is None:
            continue
        
        # Also solve with slightly tighter M to get a different direction
        M_tight = M_curr * 0.95
        b2 = lp_best_response(y, M_tight, solver)
        
        b_sum1 = b1.sum()
        if b_sum1 < 1e-15:
            continue
        
        if b2 is not None and b2.sum() > 1e-15:
            # Weighted combination
            w1 = b_sum1 / (b_sum1 + b2.sum())
            w2 = 1.0 - w1
            g = (w1 * b1 + w2 * b2) * norm / (w1 * b_sum1 + w2 * b2.sum())
        else:
            g = b1 * norm / b_sum1
        
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        score_y = fast_autocorr(y_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
        
        y = y_new
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeakLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def subgradient_with_averaging(a0, time_limit, verbose=True):
    """Improved subgradient with Nesterov averaging and adaptive step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = float(2.0 * n * best_score_ if False else fast_autocorr(a).max())
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Step size schedule: c / sqrt(t)
    c = 0.5
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        ac_max = ac.max()
        
        # Active set: peaks within 2% of max
        threshold = ac_max * 0.98
        active = np.where(ac >= threshold)[0]
        
        # Subgradient: average gradient at active peaks
        grad = np.zeros(n)
        for p in active:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            for j in range(js, je + 1):
                i = p - j
                grad[j] += a[i]
        grad /= max(1, len(active))
        
        # Step size
        step = c / np.sqrt(t)
        step = min(step, 0.05)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        avg_a += a
        n_avg += 1
        
        if n_avg % 50 == 0:
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if n_avg % 500 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    sc_avg = fast_autocorr(avg_check).max()
                    if sc_avg < best_score:
                        best_score = sc_avg
                        best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradAvg t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining multiple optimization strategies:
    1. LSE-FISTA warm-up
    2. LP best-response with adaptive step (primary method)
    3. Subgradient with averaging
    4. Peak-targeted local search
    5. Multi-length exploration
    6. Final polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    parent = None
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Use a smooth initial profile
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # === Phase 1: LSE-FISTA warm-up (fast convergence) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [15.0, 4.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")
    
    # === Phase 2: LP Nesterov (main workhorse, longest phase) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.42, 650.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov-Adaptive n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_adaptive(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")
    
    # === Phase 3: Subgradient with averaging (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        print(f"\n=== Phase 3: SubGrad-Avg n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_with_averaging(a_norm, t3, verbose=True)
        try_update(a_out, f"subgrad_n{n_best}")
    
    # === Phase 4: Peak-targeted local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Local Search n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeted_local_search(a_norm, t4, seed=42, verbose=True)
        try_update(a_out, f"local_n{n_best}")
    
    # === Phase 5: LP again (post local search) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_adaptive(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")
    
    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 200
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 25 and remaining > 80:
        print(f"\n=== Phase 6: LP at n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_adaptive(a_norm, t6, solver, verbose=True)
        try_update(a_out, f"lp_long_n{n2}")
    
    # === Phase 7: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.08, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final local search n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeted_local_search(a_norm, t7, seed=77777, verbose=True)
        try_update(a_out, "final_local")
    
    # === Phase 8: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t8 = min(remaining * 0.4, 120.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_adaptive(a_norm, t8, solver, verbose=False)
            try_update(a_out, "lp_final")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
