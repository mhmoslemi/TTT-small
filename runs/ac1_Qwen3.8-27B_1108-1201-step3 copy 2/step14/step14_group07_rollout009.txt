            if s_c < 1e-15:
                continue
            cand = cand * norm / s_c
            val = fast_autocorr(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = trial_a
        
        y_new = np.maximum((1-best_alpha)*y + best_alpha*g, 0)
        y_new = normalize_seq(y_new, norm)
        
        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            mom = (t-1.0) / (t+2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = normalize_seq(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_M:
                best_M = sc
                best_a = candidate.copy()
                last_improve = time.monotonic()
        
        # Averaging (Polyak)
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_cand = normalize_seq(avg_a / n_avg, norm)
            sc_avg = fast_autocorr(avg_cand).max()
            if sc_avg < best_M:
                best_M = sc_avg
                best_a = avg_cand.copy()
                last_improve = time.monotonic()
        
        # Stall detection and restart
        if t % 500 == 0 and time.monotonic() - last_improve > 60:
            if verbose:
                print(f"    NLP stall at iter={n_iters}, M={best_M:.10f}")
            # Restart from best
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            last_improve = time.monotonic()
            if time.monotonic() - t0 > time_limit * 0.8:
                break
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): M={best_M:.10f}")
    
    return best_a, best_M

def lse_obj_and_grad(x, n, T):
    """Log-sum-exp surrogate objective and gradient"""
    a = np.maximum(x.reshape(-1), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE smooth approximation of max
    scaled = T * (b - max_b)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    log_sum = max_b + np.log(np.exp(scaled).sum()) / T
    
    obj = 2.0 * n * log_sum / (s * s)
    
    # Gradient of LSE w.r.t. a
    # d(LSE)/da = (1/T) * sum_k exp(T*(b(k)-max_b)) * db(k)/da / sum_k exp(...)
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()  # weights for each lag
    
    # db(k)/da(j) = a(k-j) if 0 <= k-j < n (for the k where the contribution is non-zero)
    # Actually: b(k) = sum_i a[i]*a[k-i], db(k)/da(j) = a(k-j) + a(j-k) ... 
    # Wait, for convolution: b(k) = sum_i a[i]*a[k-i]
    # db(k)/da(j): a[j] appears when i=j (term a[j]*a[k-j]) and when k-i=j i.e. i=k-j (term a[k-j]*a[j])
    # So db(k)/da(j) = a[k-j] + a[k-j] = 2*a[k-j] if k-j is valid and j != k-j
    # If j = k-j (i.e., j = k/2), then db(k)/da(j) = a[j]
    
    # For the gradient: grad[j] = sum_k w[k] * db(k)/da(j)
    # = sum_k w[k] * 2*a[k-j] (approximately, ignoring the j=k/2 case)
    # This is essentially a weighted convolution of w and a
    
    # Compute: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * (w * a_reversed)(j) ... let me think more carefully
    
    # b(k) = sum_i a[i] a[k-i]
    # db(k)/da(j) = a[k-j] (from i=j term) + a[j-k] (from k-i=j term, i.e., a[k-j]*a[j] differentiated w.r.t. a[j])
    # Wait, a[k-i] when i varies: d/d(a[j]) of a[k-i] = delta(k-i, j) = 1 if i = k-j
    # So: db(k)/da(j) = a[k-j] (from the a[i] term when i=j) + a[k-j] (from the a[k-i] term when k-i=j)
    # = 2*a[k-j] if 0 <= k-j < n and j != k/2
    # = a[j] if j = k/2 (i.e., k = 2j)
    
    # For simplicity, use the approximation: grad ≈ 2 * conv(w, a_rev)
    # where a_rev[i] = a[n-1-i]
    
    # Actually, let me compute it directly:
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # Note: a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
    # And a[j-k] is valid when 0 <= j-k < n, i.e., j-n < k <= j
    
    # This is essentially: grad = conv(w, a) + conv(w, reverse(a)) ... 
    # Hmm, let me think differently.
    
    # b(k) = <a, shift_k(a)> where shift shifts a by k positions
    # db(k)/da = shift_k(a) + reverse_shift_k(a)
    
    # For the convolution: b(k) = sum_i a[i] a[k-i]
    # db(k)/da(j) = a[k-j] (if valid) + a[k-j] (if valid) = 2*a[k-j] (if valid)
    # But we need to be careful: if j = k-j, i.e., 2j = k, then it's just a[j] (counted once)
    
    # Approximation: grad[j] ≈ 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # = 2 * sum_{m: 0<=m<n} w[j+m] * a[m]
    # This is a cross-correlation of w and a
    
    # Let's compute using FFT:
    # cross[j] = sum_m w[j+m] * a[m] for valid m
    # This is the correlation of w and a at lag j
    
    # Using FFT: cross = irfft(conj(rfft(a)) * rfft(w)) ... 
    # Actually, cross[j] = sum_m a[m] * w[j+m]
    # Let's pad both to length >= 2n-1
    fft_size = 1 << (2*n - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    
    # cross[j] = sum_m a[m] w[j+m] for m=0..n-1, j=0..2n-2
    # This is (a * w_reversed) in a sense... 
    # Actually: (conv(a, w))[j] = sum_m a[m] w[j-m]
    # We want sum_m a[m] w[j+m] = (conv(a, w_rev))[j] where w_rev[m] = w[-m]... 
    
    # Let me just use: cross[j] = sum_m a[m] w[m+j]
    # = (a flipped, convolved with w)[j + n - 1] ... this is getting complicated
    
    # Simpler: use the direct formula with FFT
    # Define: g[j] = sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
    # This is the cross-correlation. Using FFT:
    # g = ifft(conj(FFT(a_padded)) * FFT(w_padded)) ... 
    
    # Actually the cleanest way:
    # g[j] = sum_m a[m] * w[j+m]
    # Let c = a flipped: c[m] = a[n-1-m]
    # Then g[j] = sum_m c[n-1-m] * w[j+m]
    # = (c * w)[n-1+j] (convolution)
    # So g = conv(c, w)[n-1 : n-1+n] = conv(c, w)[n-1 : 2n-1]
    
    c = a[::-1]
    conv_cw = fast_autocorr(np.concatenate([c, w]))  # This gives conv of c and w... 
    
    # Actually, let me just use np.convolve for this (n=1600 is fine)
    # g[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j = 0, ..., 2n-2
    # This is the cross-correlation
    # = conv(a_flipped, w)[-n+1 : n-1] ... 
    
    # You know what, let me just compute it with a simple approach:
    # grad[j] = sum_k w[k] * db(k)/da(j)
    # ≈ 2 * sum_k w[k] * a[k-j] (for the "cross terms")
    # 
    # The sum sum_k w[k] * a[k-j] is the convolution (w * a_flip)[j] where a_flip[i] = a[-i]
    # 
    # Let me define: h = [0, 0, ..., a[0], a[1], ..., a[n-1], 0, 0, ...] (padded)
    # And w is defined on [0, 2n-2]
    # 
    # (w * h)[j] = sum_k w[k] h[j-k] = sum_k w[k] a[j-k] (when j-k is in [0,n-1])
    # = sum_{m: 0<=j-m<n} w[j-m] a[m]
    # = sum_m a[m] w[j-m]
    # 
    # This is (conv(w, a))[j] in the standard sense!
    # 
    # But we want sum_m a[m] w[j+m] = sum_m a[m] w[2j-(j-m)] ... 
    # 
    # OK let me just do it the direct way for the gradient:
    
    # grad[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j] (approximation)
    # Let's compute this as a convolution
    # sum_{k=j}^{j+n-1} w[k] a[k-j] = sum_{m=0}^{n-1} w[j+m] a[m]
    # = (correlation of w and a at offset j)
    
    # Using FFT: correlation = ifft(conj(FFT(a)) * FFT(w))
    # But we need to be careful about the definition
    
    # Let me just use numpy for this since n is manageable
    # For n=1600, a direct O(n^2) computation would be 2.5M operations, which is fine
    
    # Actually, let's use the FFT approach properly:
    # We want: g[j] = sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * w[j+m]
    # This is the "upper triangle" of the outer product, summed along diagonals
    # It's the cross-correlation of a and w
    
    # Cross-correlation: (a ⋆ w)[j] = sum_m a[m] w[m+j]
    # Using FFT: ifft(conj(fft(a_padded)) .* fft(w_padded))
    
    a_fft = rfft(a, fft_size)
    w_fft = rfft(w, fft_size)
    cross = irfft(np.conj(a_fft) * w_fft, fft_size)[:n]
    
    # This gives cross[j] = sum_m a[m] w[m+j] (for the linear cross-correlation)
    # Wait, I need to verify: irfft(conj(fft(a)) * fft(w)) gives the circular cross-correlation
    # For linear: we need sufficient zero-padding (fft_size >= 2n-1 is enough? Let's use 2n)
    
    # Actually the formula is:
    # ifft(conj(FFT(a)) * FFT(w))[j] = sum_m a[m] w[m+j] (circular)
    # With sufficient padding, this equals the linear cross-correlation
    
    # So cross[j] ≈ sum_m a[m] w[m+j] for j = 0, ..., 2n-2 (with our padding)
    
    # Now grad[j] ≈ 2 * cross[j] for j = 0, ..., n-1
    # But we also need the "self-term" correction for even k=2j
    # For now, use the approximation
    
    grad_conv = 2.0 * cross[:n]
    
    # Full gradient of obj = 2n * LSE / s^2
    # d(obj)/da[j] = 2n * (dLSE/da[j] * s^2 - LSE * 2s * a[j]) / s^4
    # = 2n * (grad_conv[j] / s^2 - 2 * LSE * a[j] / s^3)
    # = 2 * obj * (grad_conv[j] / (2*LSE) - a[j]/s)  ... let me redo
    
    # obj = 2n * LSE / s^2
    # d(obj)/da[j] = 2n * (dLSE/da[j] * s^2 - LSE * 2s * a[j]) / s^4
    # = 2n * dLSE/da[j] / s^2 - 4n * LSE * a[j] / s^3
    # = (2n/s^2) * grad_conv[j] - (4n*LSE/s^3) * a[j]
    
    grad = (2.0 * n / (s * s)) * grad_conv - (4.0 * n * log_sum / (s ** 3)) * a
    
    return obj, grad.ravel()

def lse_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    """Optimize using LSE surrogate with L-BFGS-B"""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / max(1, n_stages - si)
        
        x0 = a.copy()
        
        def obj_func(x):
            return lse_obj_and_grad(x, n, T)[0]
        
        def grad_func(x):
            return lse_obj_and_grad(x, n, T)[1]
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, 100)] * n,
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'time_limit': stage_time,
                                           'gtol': 1e-8})
            if res.x is not None:
                a_cand = normalize_seq(res.x, norm)
                sc = fast_autocorr(a_cand).max()
                if sc < best_M:
                    best_M = sc
                    best_a = a_cand.copy()
                    a = a_cand
                else:
                    a = a_cand  # Still use as starting point for next stage
        except Exception as e:
            print(f"    LSE stage {si} error: {e}")
    
    return best_a, best_M

def gradient_descent_subgrad(a0, time_limit, verbose=False):
    """Subgradient method for max autocorrelation minimization"""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Weuse a "smoothed subgradient" approach
    # Instead of just the max, use a soft max (top-k average)
    k_top = max(5, n // 20)  # number of top lags to consider
    
    step_size = 0.1
    decay = 0.999
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        # Get the top-k lags
        top_idx = np.argsort(b)[::-1][:k_top]
        # Weight by how much above the median they are
        b_median = np.median(b)
        weights = np.maximum(b[top_idx] - b_median, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(k_top) / k_top
        else:
            weights = weights / weights.sum()
        
        # Compute subgradient: weighted average of gradients of top lags
        # grad_k[j] = 2 * a[k-j] (approximately)
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            # db(ki)/da[j] = a[ki-j] if 0 <= ki-j < n
            j_start = max(0, ki - (n - 1))
            j_end = min(n - 1, ki)
            indices = np.arange(j_start, j_end + 1)
            grad[indices] += wi * 2.0 * a[ki - indices]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Step in negative gradient direction
        a_new = a - step_size * grad / g_norm
        
        # Project: make non-negative and rescale
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, norm)
        
        # Check improvement
        sc = fast_autocorr(a_new).max()
        if sc < best_M:
            best_M = sc
            best_a = a_new.copy()
        a = a_new
        
        step_size *= decay
        
        if verbose and it % 5000 == 0:
            print(f"    Subgrad iter={it}, M={best_M:.10f}, step={step_size:.6f}")
    
    return best_a, best_M

def coordinate_descent(a0, time_limit, seed=42, verbose=False):
    """
    Gradient-guided coordinate descent with local search.
    Key improvements:
    - Identify peak lags and their main contributors
    - Try both "reduce peak" and "increase valley" moves
    - Use local 1D optimization for each coordinate
    - Periodic random perturbations
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    def try_update(trial):
        nonlocal a, best_a, best_M, no_improve
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            return False
        trial = trial * norm / s
        sc = fast_autocorr(trial).max()
        if sc < best_M - 1e-15:
            a = trial
            best_a = trial.copy()
            best_M = sc
            no_improve = 0
            return True
        no_improve += 1
        return False
    
    moves = 0
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        b = fast_autocorr(a)
        
        # Find peak lags
        n_peaks = max(8, n // 50)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        # Strategy 1: Reduce mass at positions that contribute to peaks
        for k_peak in top_peaks:
            # Find positions that contribute to this peak
            js = max(0, k_peak - (n - 1))
            je = min(n - 1, k_peak)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            
            # Top contributors
            top_contrib = idx[np.argsort(contrib)[::-1][:min(10, len(idx))]]
            
            for src in top_contrib:
                if a[src] < 1e-10:
                    continue
                
                # Find destination: position that doesn't contribute much to any peak
                # Compute "peak exposure" for each position
                peak_exposure = np.zeros(n)
                for k2 in top_peaks[:max(4, n_peaks//2)]:
                    js2 = max(0, k2 - (n-1))
                    je2 = min(n-1, k2)
                    for i in range(js2, je2+1):
                        peak_exposure[i] += a[i] * a[k2-i]
                
                # Sort by peak exposure (ascending) - we want to move to low-exposure positions
                candidates = np.argsort(peak_exposure)[:n//4]
                
                # Try moving to the best candidates
                for dst in candidates[:8]:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        if try_update(trial):
                            improved = True
                            moves += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                continue
        
        # Strategy 2: Local 1D optimization
        if not improved:
            # Pick a position and try to optimize its value
            # Focus on positions with high peak exposure
            b = fast_autocorr(a)
            top_peaks2 = np.argsort(b)[::-1][:5]
            
            # Get the top contributing position
            for k_peak in top_peaks2:
                js = max(0, k_peak - (n-1))
                je = min(n-1, k_peak)
                if js > je:
                    continue
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[k_peak - idx]
                best_pos = idx[np.argmax(contrib)]
                
                # Try different values for a[best_pos]
                # The constraint is that the sum stays the same
                # So if we increase a[best_pos] by d, we decrease some other by d
                cur_val = a[best_pos]
                if cur_val < 1e-10:
                    continue
                
                # Try reducing a[best_pos] and distributing to random positions
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amt = cur_val * frac
                    trial = a.copy()
                    trial[best_pos] -= amt
                    # Distribute to random positions (not contributing to this peak)
                    other_peaks_mask = np.zeros(n, dtype=bool)
                    for k2 in top_peaks2:
                        js2 = max(0, k2 - (n-1))
                        je2 = min(n-1, k2)
                        other_peaks_mask[js2:je2+1] = True
                    
                    safe_positions = np.where(~other_peaks_mask)[0]
                    if len(safe_positions) == 0:
                        safe_positions = np.where(np.arange(n) != best_pos)[0]
                    
                    # Distribute proportionally to current values
                    target = safe_positions[:20]
                    target_vals = a[target].copy()
                    if target_vals.sum() < 1e-10:
                        target_vals = np.ones(len(target)) * 0.01
                    distribution = target_vals / target_vals.sum() * amt
                    for ti, tv in zip(target, distribution):
                        trial[ti] += tv
                    
                    if try_update(trial):
                        improved = True
                        moves += 1
                        break
                if improved:
                    break
        
        # Strategy 3: Random perturbation
        if not improved:
            n_perturb = max(2, n // 100)
            indices = rng.choice(n, size=n_perturb, replace=False)
            trial = a.copy()
            for idx in indices:
                trial[idx] *= rng.uniform(0.5, 1.5)
            trial = np.maximum(trial, 0)
            if try_update(trial):
                improved = True
                moves += 1
        
        # Strategy 4: Block move - shift a block of mass
        if not improved and moves % 50 == 0:
            block_size = max(3, n // 50)
            start = rng.integers(0, n - block_size)
            end = start + block_size
            block = a[start:end].copy()
            
            # Try moving this block to a random location
            new_start = rng.integers(0, n - block_size)
            trial = a.copy()
            trial[start:end] = 0
            # Handle overlap
            if new_start < end and start < new_start + block_size:
                # Overlap: just add
                trial[new_start:new_start+block_size] += block
            else:
                trial[new_start:new_start+block_size] += block
            # Redistribute to keep sum
            if try_update(trial):
                improved = True
                moves += 1
        
        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
    
    if verbose:
        print(f"    CD: {moves} improvements, M={best_M:.10f}")
    
    return best_a, best_M

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    f_padded = np.concatenate([f_norm, [f_norm[-1]]])  # Pad for interpolation
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x[:-1] if len(old_x) == old_n + 1 else old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3*np.cos(2*np.pi*2*x) + 0.15*np.cos(2*np.pi*5*x) + 0.05*np.cos(2*np.pi*9*x)
    elif choice == 3:
        a = np.exp(-4*(x-0.25)**2) + 0.6*np.exp(-4*(x-0.75)**2)
    elif choice == 4:
        a = np.exp(-8*(x-0.5)**2)
    elif choice == 5:
        idx = np.arange(n) - n//2
        a = np.exp(-0.5*(idx/(n/5.0))**2)
    elif choice == 6:
        a = 1.0 + 0.3*np.sin(4*np.pi*x) + 0.1*np.sin(8*np.pi*x)
    elif choice == 7:
        a = 1.0 + 0.5*np.cos(2*np.pi*x)
    elif choice == 8:
        a = np.maximum(0, np.sin(3*np.pi*x))
    elif choice == 9:
        a = np.exp(-2*(x-0.3)**2) + np.exp(-2*(x-0.7)**2) + 0.3
    elif choice == 10:
        # Triangular
        a = 1.0 - np.abs(2*x - 1)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multiplicative_update(a0, time_limit, verbose=False):
    """
    Multiplicative (mirror descent) update for minimizing max autocorrelation.
    Uses the "natural gradient" direction based on the autocorrelation peaks.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.5  # Step size
    n_iters = 0
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        M_curr = b.max()
        
        # Compute "pressure" at each position: how much it contributes to peaks
        # pressure[i] = sum_{k: b[k] > threshold} db[k]/da[i]
        threshold = 0.8 * M_curr
        active_lags = np.where(b > threshold)[0]
        
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(b)])
        
        pressure = np.zeros(n)
        for k in active_lags:
            weight = (b[k] - threshold) / (M_curr - threshold + 1e-15)
            js = max(0, k - (n-1))
            je = min(n-1, k)
            idx = np.arange(js, je+1)
            pressure[idx] += weight * a[k - idx]
        
        # Multiplicative update: a[i] *= (1 - eta * pressure[i]) / (1 - eta * mean_pressure)
        mean_p = pressure.mean()
        ratio = (1.0 - eta * pressure) / (1.0 - eta * mean_p + 1e-15)
        ratio = np.clip(ratio, 0.1, 10.0)  # Prevent extreme updates
        
        a_new = a * ratio
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, norm)
        
        sc = fast_autocorr(a_new).max()
        if sc < best_M:
            best_M = sc
            best_a = a_new.copy()
        
        a = a_new
        n_iters += 1
        
        # Decay step size
        eta *= 0.9995
        
        if verbose and n_iters % 5000 == 0:
            print(f"    MU iter={n_iters}, M={best_M:.10f}, eta={eta:.6f}")
        
        if eta < 0.001:
            break
    
    return best_a, best_M

    Main optimization function. Uses a combination of:
    1. LP-based Nesterov (primary)

    4. Coordinate descent (local polish)
    5. Multi-length exploration

    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver_cached()

            parent = p / p.sum()
            p_score = evaluate_safe(parent)

            best_seq = a.copy()
