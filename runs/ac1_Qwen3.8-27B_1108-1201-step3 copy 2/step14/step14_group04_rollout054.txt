def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: for each lag k, the constraint is sum_j a[j] * b[k-j] <= M
    # This is a Toeplitz-like matrix
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def spectral_flatten(a0, time_limit, seed=42):
    """
    Frequency-domain spectral flattening:
    Alternate between flattening |FFT(a)|^2 and projecting to a >= 0.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        # Compute FFT
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: make power as flat as possible (excluding DC)
        # Use iterative reweighted approach
        # Soft-flat: weight by 1/sqrt(power + eps) to pull extremes toward mean
        eps = 1e-10
        non_dc = np.arange(1, len(power))
        p_non_dc = power[non_dc]
        target = np.median(p_non_dc)
        
        # Compute flatness metric
        flatness = (p_non_dc.max() / max(p_non_dc.min(), eps))
        
        # Adjust phases/magnitudes to flatten
        # Stride: scale each frequency bin's magnitude toward target
        scale = np.ones(len(power))
        for idx in non_dc:
            if power[idx] > target:
                # Reduce
                scale[idx] = target / (power[idx] + eps) ** 0.3
            elif power[idx] < target * 0.5:
                # Slightly increase (less aggressive)
                scale[idx] = (target / (power[idx] + eps)) ** 0.1
        
        A_scaled = A * scale
        a_new = irfft(A_scaled, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        if sc < best_score - 1e-12:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

                break
    
    return best_a, best_score


def targeted_redistribution(a0, time_limit, seed=12345):

    Targeted peak reduction: identify autocorrelation peaks and redistribute mass
    from high-product pairs to low-impact positions.
    """

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks (excluding lag 0 which is always the largest)
        sorted_lags = np.argsort(b)[::-1]
        top_lags = sorted_lags[:8]
        
        improved = False
        for k in top_lags:
            if k < 0 or k > 2 * n - 2:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Find the pair with maximum contribution
            src_j = idx[np.argmax(contrib)]
            src_k = k - src_j
            
            # Find positions to move mass to (positions that don't contribute to top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_lags:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True
            
            # Candidates: positions not in peak mask, with low a values
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(a < np.median(a))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Sort by a value (move to positions with less mass)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            
            for frac in [0.0003, 0.001, 0.003, 0.01, 0.02, 0.05]:
                amt = a[src_j] * frac
                if amt < 1e-15:
                    continue
                dst = candidates_sorted[rng.integers(0, min(len(candidates_sorted), 20))]
                
                a_trial = a.copy()
                a_trial[src_j] -= amt
                a_trial[dst] += amt
                if a_trial[src_j] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if a.max() > norm * 1e-8:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt

                        if s_t > 1e-15:

                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(cp.square(x))), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_nonneg_sum(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_proximal_gradient(a0, time_limit, n_stages=4, verbose=True):
    """
    Proximal gradient on the log-sum-exp of autocorrelation.
    Uses FISTA acceleration with temperature annealing.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = project_nonneg_sum(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = [10.0, 3.0, 1.0, 0.4, 0.15]

        if time.monotonic() - t0 > time_limit:
            break
        
        n_size = len(a)

        L = n_size / (3.0 * T)
        
        y = a.copy()
        t_fista = 1

        max_stage_time = time_limit * 0.25 if n_stages > 1 else time_limit

            # Compute gradient of LSE
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:

            W = rfft(w, N_fft)

import numpy as np

_global_solver = None

def get_lp_solver():

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_nn(v, target_sum):
    """Project to non-negative with given sum."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s


def lp_bres(a, M, solver):
    """LP best response: max sum(b) s.t. conv(a,b)<=M, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_sw(a0, time_limit, solver, verbose=True):

        M = M_base * M_factor

        else:

            a_new = g.copy()
        else:

            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new

        window[window_idx, :] = a
        window_idx = (window_idx + 1) % window_size
        window_filled = min(window_filled + 1, window_size)

        full_avg += a
        full_count += 1

        score_a = fast_autocorr(a).max()

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_stall = 0
            prev_best = best_score
        
        # Check sliding window average (every 20 iterations once window is full)
        if window_filled >= window_size and n_iters % 20 == 0:
            sw_avg = window.mean(axis=0)
            s_sw = sw_avg.sum()
            if s_sw > 1e-15:

                    best_stall = 0

def gradient_peak_shave(a0, time_limit, seed=42):
    """
    Gradient-guided peak shaving:
    1. Compute autocorrelation gradient for each peak
    2. Move mass in the direction that most reduces the peak
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = project_nn(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:5]
        improved = False
        
        for k in top_k:
            # Compute gradient of ac[k] with respect to a
            # grad[m] = a[k-m] + a[m] (for valid m)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            
            # Find the position with max contribution
            src = idx[np.argmax(contrib)]
            
            # Find destination: minimize impact on other peaks
            # Compute "sensitivity" of each position to the top peaks
            sensitivity = np.zeros(n)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    sensitivity[idx2] += a[k2 - idx2]
            
            # Choose destination with lowest sensitivity
            candidates = np.where(a > 1e-10)[0]
            if len(candidates) == 0:
                break
            low_sens = sorted(candidates, key=lambda m: sensitivity[m] + a[m] * 0.1)
            dst_candidates = low_sens[:max(5, n // 100)]
            
            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                dst = dst_candidates[rng.integers(len(dst_candidates))]
                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 400:
                break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
