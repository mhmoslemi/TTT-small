import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:

    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_toeplitz(a, n_constraints=None):
    """Build the Toeplitz matrix for the LP constraints.
    If n_constraints is given, only include the top n_constraints rows."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1
    
    if n_constraints is not None and n_constraints < size:
        # Select top n_constraints lags
        top_idx = np.argsort(b)[::-1][:n_constraints]
    else:
        top_idx = np.arange(size)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in top_idx:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    row_map = {k: i for i, k in enumerate(top_idx)}
    rows = [row_map[r] for r in rows]
    
    T = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    return T, top_idx, b[top_idx]


def lp_best_response_sparse(a, M, solver, n_constraints=None):
    """Solve the LP best-response problem with sparse matrix."""
    n = len(a)
    b = fast_autocorr(a)
    
    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(300, n))
    
    # Select constraints: top n_constraints by value, plus always include the max
    size = 2 * n - 1
    if n_constraints < size:
        top_idx = np.argsort(b)[::-1][:n_constraints]
    else:
        top_idx = np.arange(size)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    
    T = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:

def nesterov_lp(a0, time_limit, solver, verbose=True, n_constraints=None, alpha_decay='standard'):
    """Nesterov-accelerated LP best-response with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    best_time = t0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        b = lp_best_response_sparse(y, M_curr, solver, n_constraints=n_constraints)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        
        g = b * norm / b_sum
        n_iters += 1
        stall = 0
        
        # Nesterov step with line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        y_trial = (1 - alpha) * y + alpha * g
        y_trial = np.maximum(y_trial, 0.0)
        s_y = y_trial.sum()
        if s_y > 1e-15:
            y_trial = y_trial * norm / s_y
        best_obj = fast_autocorr(y_trial).max()
        
        # Try a few other alphas
        for trial in [alpha * 0.5, alpha * 1.5, alpha * 2.0, alpha * 0.25]:
            if trial > 1.0:
                continue
            y_t = (1 - trial) * y + trial * g
            y_t = np.maximum(y_t, 0.0)
            s_t = y_t.sum()
            if s_t < 1e-15:
                continue
            y_t = y_t * norm / s_t
            sc = fast_autocorr(y_t).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        
        y_new = (1 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            best_time = time.monotonic()
        
        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            best_time = time.monotonic()
        
        # Restart if stalled
        if t % 500 == 0 and time.monotonic() - best_time > time_limit * 0.3:
            y = best_a.copy()
            a = y.copy()
            t = max(t // 2, 1)  # partially reset momentum
        
        if verbose and n_iters % 200 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def subgradient_polish(a0, time_limit, n_active=50, verbose=True):
    """Subgradient descent on max(conv(a,a)) with projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    stall = 0
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find active lags (within 5% of max, or top n_active)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > n_active:
            active = np.argsort(b)[::-1][:n_active]
        
        # Compute subgradient: average of gradients at active lags
        # g_j = (1/|active|) * sum_{k in active} 2*a_{k-j} (for valid k-j)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        g /= len(active)
        
        # Adaptive step size
        a_norm = np.linalg.norm(a)
        if a_norm < 1e-15:
            break
        eta = 1.0 / (np.sqrt(it) * a_norm)
        
        # Step
        a_trial = a - eta * g
        a_trial = np.maximum(a_trial, 0.0)
        
        # Project onto sum = norm
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_t
        
        sc = fast_autocorr(a_trial).max()
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            stall = 0
            a = a_trial
        else:
            stall += 1
            if stall > 200:
                # Reset step size and try again
                eta *= 0.5
                stall = 0
        
        if it % 5000 == 0 and verbose:
            print(f"    Subgrad iter={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_toeplitz_sparse(a, n_constraints=None):
    """Build sparse Toeplitz matrix and select which constraints to use."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    if n_constraints is not None and n_constraints < size:
        # Always include the argmax, plus top n_constraints-1 others
        top_idx = np.argsort(b)[::-1][:n_constraints]
    else:
        top_idx = np.arange(size)

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])

    T = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    return T, top_idx


def lp_best_response(a, M, solver, n_constraints=500):
    """Solve the LP: max sum(b) s.t. b>=0, (a*b)_k <= M for selected k."""
    n = len(a)
    T, top_idx = build_toeplitz_sparse(a, n_constraints)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, n_constraints=500):
    """Nesterov-accelerated LP best-response with adaptive line search and restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a_arr = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a_arr.sum()
    if s0 < 1e-15:
        a_arr = np.ones(n) * norm / n
    else:
        a_arr = a_arr * norm / s0

    best_a = a_arr.copy()
    best_score = fast_autocorr(a_arr).max()
    best_time = time.monotonic()

    y = a_arr.copy()
    t0 = time.monotonic()
    n_iters = 0
    t = 1

    while time.monotonic() - t0 < time_limit:
        if t > 500000:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break

        b = lp_best_response(y, M_curr, solver, n_constraints=n_constraints)
        if b is None:
            t += 1
            if t % 1000 == 0:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            t += 1
            continue

        g = b * norm / b_sum
        n_iters += 1

        # Line search around Nesterov step
        alpha_nesterov = 2.0 / (t + 2.0)
        best_alpha = alpha_nesterov
        best_obj = np.inf

        candidates = [alpha_nesterov, alpha_nesterov * 0.5, alpha_nesterov * 1.5,
                      alpha_nesterov * 2.0, alpha_nesterov * 0.25, 1.0 / (t + 2)]
        for alpha in candidates:
            if alpha <= 0 or alpha > 1:
                continue
            y_trial = (1 - alpha) * y + alpha * g
            y_trial = np.maximum(y_trial, 0.0)
            s_y = y_trial.sum()
            if s_y < 1e-15:
                continue
            y_trial = y_trial * norm / s_y
            sc = fast_autocorr(y_trial).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = alpha

        y_new = (1 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov momentum
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a_arr = a_new
        t += 1

        # Track best
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            best_time = time.monotonic()

        sc_a = fast_autocorr(a_arr).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_arr.copy()
            best_time = time.monotonic()

        # Periodic check for stalling
        if t % 300 == 0:
            if time.monotonic() - best_time > time_limit * 0.4:
                y = best_a.copy()
                a_arr = best_a.copy()
                t = max(t // 3, 1)

        if verbose and n_iters % 100 == 0:
            print(f"    NLP iter={n_iters} (t={t}, {time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_active=80, verbose=True):
    """
    Subgradient descent on max(conv(a,a)) with projection onto simplex.
    Uses the exact subgradient formula: at maximizing lag k, grad_j = 2*a[k-j].
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0
    base_eta = 1.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        # Select active lags
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        if len(active) > n_active:
            active = np.argsort(b)[::-1][:n_active]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient vectorized
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            js2 = int(js)
            je2 = int(je)
            g[js2:je2 + 1] += 2.0 * a[k - np.arange(js2, je2 + 1)]
        g /= len(active)

        # Adaptive step size with backtracking
        eta = base_eta / np.sqrt(it + 1)
        max_iter_backtrack = 10
        for _ in range(max_iter_backtrack):
            a_trial = a - eta * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                eta *= 0.5
                continue
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc <= max_b + eta * np.linalg.norm(g):
                break
            eta *= 0.5

        sc = fast_autocorr(a_trial).max()
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial

        it += 1

        if it % 2000 == 0 and verbose:
            print(f"    Subgrad iter={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

        if it > 200000:
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42, n_peaks=10, verbose=True):
    """
    Coordinate descent targeting the top peaks of the convolution.
    For each peak, try to reduce it by moving mass from contributing positions
    to non-contributing positions.
    """

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False

        for k_peak in top_peaks:
            if b[k_peak] < best_score * 0.99:
                continue

            # Find contributing indices
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]

            # Find safe destinations (indices that don't contribute to top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != k_peak)[0]
            # Prefer candidates with low values (to minimize new peaks)
            candidates_sorted = candidates[np.argsort(a[candidates])][:min(20, len(candidates))]

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                for dst in candidates_sorted[:8]:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random perturbation
                n_perturb = max(3, n // 100)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx_p in indices:
                    a_trial[idx_p] *= rng.uniform(0.5, 1.5)
                try_update(a_trial)

            # Try moving mass from high to low within the same "region"
            if stall % 20 == 0:
                src = int(np.argmax(a))
                dst_pool = np.argsort(a)[:n // 10]
                dst = int(dst_pool[rng.integers(len(dst_pool))])
                if src != dst and a[src] > 1e-10:
                    for frac in [0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            break

            if stall > 500:
                break

        if verbose and stall % 100 == 0:
            print(f"    CD stall={stall} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, remaining - 5, seed=99999, n_peaks=6, verbose=False)
        do_try_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in np.maximum(best_seq, 0.0)]
