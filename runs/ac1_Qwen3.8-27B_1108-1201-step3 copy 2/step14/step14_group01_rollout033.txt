"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_grad(a, beta):
    """Compute gradient of LSE-smoothed max of conv(a,a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv = fast_conv_self(a)
    max_c = conv.max()
    shifted = (conv - max_c) * beta
    exp_c = np.exp(np.maximum(shifted, -100.0))
    w = exp_c / exp_c.sum()
    # Gradient = 2 * cross_correlation(a, w)[i] for i in [0, n)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]
    return grad, conv

def adam_smoothed_optimize(a0, time_limit, verbose=True):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    
    # Temperature schedule
    n_stages = 8
    temps = np.geomspace(20.0, 2.0, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(50, min(500, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv = lse_grad(a, T_val)
            
            # Adam update
            m_hat = beta1 * m_hat + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
            m_hat_hat = m_hat / (1 - beta1 ** (it + 1))
            v_hat_hat = v_hat / (1 - beta2 ** (it + 1))
            
            a_new = a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + epsilon)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            # Check improvement
            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Adam stage {si} T={T_val:.1f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_solve(a, M, solver, active=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    n_iter = 0
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        # Start with tighter threshold, relax if LP fails
        threshold = M * 0.9
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 10:
            active = None  # Use full constraints
        
        b = lp_solve(y, M, solver, active)
        if b is None or not np.isfinite(b).all():
            b = lp_solve(y, M, solver, None)
        if b is None or not np.isfinite(b).all():
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify constraints (if screening was used)
        if active is not None:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.01:
                b = lp_solve(y, M, solver, None)
                if b is None:
                    continue
                b_sum = b.sum()
                if b_sum < 1e-15:
                    continue
                g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        n_iter += 1
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent on exact objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        perm = rng.permutation(n)
        pass_improved = False
        conv = fast_conv_self(a)
        cur_max = conv.max()
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find best j to transfer to
            # Try a few random j's + gradient-informed
            j_candidates = [int(rng.integers(n)) for _ in range(5)]
            # Add gradient-informed: where would reducing a[i] help most?
            # The contribution of a[i] to conv[k] is a[k-i] (for valid k)
            # So reducing a[i] helps most at k where a[k-i] is large
            peak_k = np.argmax(conv[np.maximum(0, i):np.minimum(size, i + n)])
            j_star = (i + int(peak_k) - i) % n  # This is just peak_k % n... 
            # Actually, the gain from transferring to j is related to where j is
            # relative to the peaks
            for pk in np.argsort(conv)[-3:]:
                j_c = (pk - i) % n
                if 0 <= j_c < n:
                    j_candidates.append(int(j_c))
            
            j_candidates = [j for j in j_candidates if j != i]
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in set(j_candidates):
                aj = a[j]
                # Try several transfer amounts
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    d = ai * frac
                    a[i] = ai - d
                    a[j] = aj + d
                    a_tmp = a * norm / a.sum()
                    v = fast_conv_self(a_tmp).max()
                    if v < best_val:
                        best_val = v
                        best_d = d
                        best_j = j
                    a[i] = ai
                    a[j] = aj
                    a = a * norm / a.sum()
            
            if best_j >= 0 and best_d > 0 and best_val < cur_max - 1e-12:
                a[i] = ai - best_d
                a[best_j] = a[best_j] + best_d
                a = a * norm / a.sum()
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    conv = a and fast_conv_self(a)
                    cur_max = conv.max()
                else:
                    a[best_j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_score
