def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_adam_optimize(a0, time_limit, verbose=True):
    """Adam-based LSE gradient descent with adaptive temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(300, int(remaining * 8)))
        lr = norm / (n ** 0.5) * 0.1
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            t_adam += 1
            
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            # LSE weights
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k])
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # sum_k w[k] * a[k+i] (circular cross-correlation)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            
            # sum_k w[k] * a[k-i] = conv(a, w)[i] (linear, since fft_size >= 2n-1)
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            
            grad = 2.0 * (g1 + g2)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE-Adam stage {si} T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def build_T_matrix(a):
    """Build the constraint matrix for LP: T[k, i] such that (T @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    size = 2 * n - 1
    # T[k, i] = a[k - i] if 0 <= k - i < n, else 0
    T = np.zeros((size, n))
    for m in range(n):
        # When b[m] contributes, it appears in conv at positions m, m+1, ..., m+n-1
        # with values a[0], a[1], ..., a[n-1]
        T[m:m + n, m] = a
    return T

def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Use constraint generation if too many constraints
    if active_set is not None and len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            # T[idx, i] = a[k - i] if 0 <= k - i < n
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    T[idx, i] = a[j]
        constraints = [T @ b_var <= M]
    else:
        T = build_T_matrix(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov accelerated LP best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Constraint generation state
    active_set = None
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Update active set: constraints where conv > 0.8 * M
        threshold = M * 0.8
        active_set = np.where(conv_y > threshold)[0]
        if len(active_set) < 10:
            active_set = np.where(conv_y > M * 0.5)[0]
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            # Try full constraints
            b = lp_best_response(y, M, solver, None)
        
        if b is None:
            b = y.copy()  # fallback: no improvement
        else:
            # Check if inactive constraints are violated
            if active_set is not None and len(active_set) < size:
                check = fast_conv_cross(y, b)
                violated = np.where(check > M * 1.001)[0]
                if len(violated) > 0:
                    # Add violated constraints and re-solve
                    active_set = np.union1d(active_set, violated)
                    b2 = lp_best_response(y, M, solver, active_set)
                    if b2 is not None:
                        b = b2
                    else:
                        b2 = lp_best_response(y, M, solver, None)
                        if b2 is not None:
                            b = b2
            
            b_sum = b.sum()
            if b_sum < 1e-15:
                b = y.copy()
            else:
                g = b * norm / b_sum
       
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    """Multiplicative (Vegas/Faddeev) updates for non-negative optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    eta = 0.05  # learning rate
    n_stages = 20
    etas = np.geomspace(eta, eta * 0.01, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(100, int(remaining * 3)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            # Compute the "pressure" on each coordinate
            # p[i] = sum_k max(0, conv[k] - M*0.95) * (a[k-i] + a[i-k]) / a[i]
            # Multiplicative update: a[i] *= exp(-eta * p[i])
            
            # Simpler: for each coordinate i, compute the weighted sum of violations
            W = rfft(np.maximum(conv - M * 0.95, 0), 1 << (2*n-2).bit_length())
            A = rfft(a, 1 << (2*n-2).bit_length())
            
            fft_size = 1 << (2 * n - 2).bit_length()
            W = rfft(np.maximum(conv - M * 0.95, 0), fft_size)
            A = rfft(a, fft_size)
            
            # pressure[i] = sum_k w[k] * (a[k-i] + a[i-k])
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            # Multiplicative update
            a = a * np.exp(-eta_s * pressure / M)
            a = np.maximum(a, 1e-10)  # keep positive
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    Vegas stage {si} eta={eta_s:.4f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with gradient-informed candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find best j to transfer to
            # Use gradient info: j that reduces the peak most
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            
            # Candidate j's
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j_candidates.add((kl - i + n) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain: how does conv change?
                # Transfer delta from i to j changes conv by:
                # delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k]) + second order
                g = np.zeros(size)
                
                # a[k-j] contribution
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                
                # a[j-k] contribution  
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # -a[k-i]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                
                # -a[i-k]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find best delta in [0, ai * 0.9]
                d_max = ai * 0.9
                n_samples = 20
                deltas = np.linspace(0, d_max, n_samples)
                
                for d in deltas[1:]:
                    new_conv = conv + d * g
                    # Second order corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                # Perturb

                idx_p = rng.choice(n, size=n_perturb, replace=False)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            
            grad = 2.0 * (g1 + g2)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (si % 4 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si} T={T_val:.3f}: best={best_score:.8f}")
    
    return best_a, best_score

def build_T_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and 0 < len(active_set) < size:
        T = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            row_start = max(0, k - n + 1)
            row_end = min(n, k + 1)
            for i in range(row_start, row_end):
                T[idx, i] = a[k - i]
        constraints = [T @ b_var <= M]
    else:
        T = build_T_matrix(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        threshold = M * 0.75
        active_set = np.where(conv_y > threshold)[0]
        if len(active_set) < 10:
            active_set = np.where(conv_y > M * 0.5)[0]
        if len(active_set) > 500:
            active_set = np.where(conv_y > M * 0.9)[0]
            if len(active_set) < 5:
                active_set = None
        
        b = lp_best_response(y, M, solver, active_set)
        
        if b is None:
            b = lp_best_response(y, M, solver, None)
        
        if b is None:
            continue
        
        if active_set is not None and len(active_set) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * 1.001)[0]
            if len(violated) > 0:
                active_set = np.union1d(active_set, violated)
                b2 = lp_best_response(y, M, solver, active_set)
                if b2 is not None:
                    b = b2
                else:
                    b2 = lp_best_response(y, M, solver, None)
                    if b2 is not None:
                        b = b2
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score

def vegas_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 25
    etas = np.geomspace(0.1, 0.001, n_stages)
    
    for si, eta_s in enumerate(etas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(80, int(remaining * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_conv_self(a)
            M = conv.max()
            if M < 1e-15:
                break
            
            w = np.maximum(conv - M * 0.9, 0.0)
            ww = w / (w.sum() + 1e-15)
            
            W = rfft(ww, fft_size)
            A = rfft(a, fft_size)
            
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            pressure = ca + aw
            
            a = a * np.exp(-eta_s * pressure)
            a = np.maximum(a, 1e-12)
            a = a * norm / a.sum()
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and si % 5 == 0:
            print(f"    Vegas stage {si} eta={eta_s:.5f}: best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            peak_lags = np.where(conv >= cur_max * 0.85)[0]
            
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:10]:
                j_candidates.add(int((kl - i + n) % n))
                j_candidates.add(int((kl + i) % n))
            j_candidates.discard(i)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                
                g = np.zeros(size)
                
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                d_max = ai * 0.8
                n_samples = 25
                deltas = np.linspace(0.01, d_max, n_samples)
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0 and best_v < best_score - 1e-10:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-10:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                n_perturb = max(5, n // 80)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                if no_improve >= 10:
                    break
        else:
            no_improve = 0
        
        if verbose and total_improvements % 500 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improv={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def sparsify(a, threshold_frac=0.001):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a_norm = a / s
    threshold = threshold_frac * np.max(a_norm)
    mask = a_norm > threshold
    if np.sum(mask) < n // 4:
        mask = a_norm > threshold * 0.1
    a_s = np.where(mask, a, 0.0)
    s_new = a_s.sum()
    if s_new < 1e-15:
        return a
    return a_s / s_new

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 20)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = np.exp(-30.0 * (x - c1) ** 2) + np.exp(-30.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-5.0 * x) + 0.4
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        n_peaks = rng.integers(3, 10)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            width = rng.uniform(0.02, 0.1)
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    elif choice == 9:
        a = np.sinc(x - 0.5) ** 2 + 0.1
    elif choice == 10:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.005 * np.abs(rng.standard_normal(n))
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def symmetrize(a):
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s * (a.sum() / s)
