def mirror_descent_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a * norm / s
    
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive temperature schedule
    n_total_steps = 5000
    T0 = 5.0
    T_end = 0.1
    
    for step in range(n_total_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = step / n_total_steps
        T = T0 * (T_end / T0) ** progress
        
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # Compute smoothed gradient
        exp_vals = np.exp((conv - max_conv) / T)
        w = exp_vals / exp_vals.sum()
        
        # Gradient via FFT cross-correlation
        A = rfft(a, fft_size)
        W = rfft(np.pad(w, (0, fft_size - size)), fft_size)
        # cross-correlation: we want sum_j w[j+i] * a[j]
        # This is the cross-correlation of w and a at lag i
        # Using FFT: corr = ifft(conj(fft(a)) * fft(w)) but need to be careful
        # Actually: corr[i] = sum_j a[j] * w[j+i] = ifft(conj(A) * W)[i] (circular)
        # But we need linear, so we pad sufficiently
        cross = irfft(np.conj(A) * W, fft_size)
        grad = 2.0 * cross[:n]
        
        # Multiplicative update
        eta = norm / (T * np.linalg.norm(grad)) if np.linalg.norm(grad) > 1e-10 else 1.0/n
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step={step} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

The key insight is that the problem is a **convex** optimization (minimize the maximum of linear functions, subject to a >= 0 and sum(a) = norm). Wait, is it convex?

max_k conv(a)[k] where conv(a)[k] = sum_j a[j]*a[k-j]. Each conv(a)[k] is a quadratic form, not linear. So the objective is not convex in general.

However, the LP best-response approach works because it's solving a related problem: given a fixed a, maximize sum(b) subject to conv(a,b) <= M. This is linear in b (since conv(a,b)[k] = sum_j a[j]*b[k-j] is linear in b). So it IS a linear program.

The overall Nesterov method is for a "zero-sum game" or a saddle point problem, not a convex optimization. The convergence is sublinear.

For the local optimization near the optimum, the problem is essentially: given that certain conv positions are at the maximum, how to adjust a to reduce them while not increasing others. This is a quadratic programming problem (minimize ||Delta||^2 subject to linear constraints on the change in conv).

**New idea: QP-based local optimization**

At each step:
1. Compute conv(a), identify the "hot" set H = {k : conv[k] >= max_conv - epsilon}
2. Solve: minimize ||Delta||_2^2
   subject to: for k in H, d(conv[k])/da * Delta <= 0 (don't increase hot positions)
                 for k not in H, d(conv[k])/da * Delta <= max_conv - conv[k] - epsilon (don't push others above max)
                 sum(Delta) = 0 (maintain normalization)
                 a + Delta >= 0

This is a QP with a special structure. The gradient d(conv[k])/da[i] = 2*a[k-i] (as computed earlier).

For the "hot" set, the constraint is: sum_i 2*a[k-i] * Delta[i] <= 0 for k in H.
For other positions: sum_i 2*a[k-i] * Delta[i] <= max_conv - conv[k] - epsilon.

This QP can be solved with cvxpy or a custom active-set method.

The advantage of this approach is that it directly reduces the maximum by solving a well-posed convex subproblem at each iteration.

    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # Identify hot set: positions within 1% of max
        threshold = max_conv * 0.995
        hot = np.where(conv >= threshold)[0]

        # For each k, the constraint is: 2 * sum_i a[k-i] * Delta[i] <= rhs[k]
        # where rhs[k] = 0 if k in hot, else max_conv - conv[k]
        # And sum(Delta) = 0

        # This is the k-th row of the convolution matrix (times 2)
        
        # For efficiency, only include constraints for positions near the hot set
        # and a few others
        # Actually, let's include all 2n-1 constraints (it's fine for n=1600)
        
        T_mat = build_conv_matrix(a)  # T[k, i] = a[k-i] if valid
        G = 2.0 * T_mat  # G[k, i] = 2 * a[k-i]
        
        rhs = np.zeros(size)
        for k in range(size):
            if k in hot:  # This is O(size) which is fine
                rhs[k] = 0.0  # Don't increase
            else:
                rhs[k] = max_conv - conv[k]  # Don't exceed current max
        
        # + epsilon margin
        # Actually, for hot positions, we want to DECREASE, so rhs should be negative
        # Let's use: G @ Delta <= rhs where rhs[k] = -0.01*max_conv for hot, 
        # and rhs[k] = max_conv - conv[k] - 0.001*max_conv for others
        
        margin = 0.01 * max_conv
        for k in hot:
            rhs[k] = -margin  # Must decrease by at least margin
        for k in range(size):
            if k not in hot:
                rhs[k] = max_conv - conv[k] - 0.001 * max_conv
        
        # QP: min ||Delta||^2 s.t. G @ Delta <= rhs, sum(Delta) = 0, a + Delta >= 0
        Delta = cp.Variable(n)
        objective = cp.Minimize(cp.sum_squares(Delta))
        constraints = [G @ Delta <= rhs, cp.sum(Delta) == 0, a + Delta >= 0]
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if n_iters > 0:
                break
            continue
        
        Delta_val = np.asarray(Delta.value, dtype=np.float64)
        a_new = a + Delta_val
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * norm / s
        
        new_max = fast_autocorr(a_new).max()
        if new_max < best_score:
            best_score = new_max
            best_a = a_new.copy()
        
        a = a_new

            margin *= 0.5
            if margin < 1e-6:

            continue
        
        if verbose and n_iters % 5 == 0:

            print(f"    QP iter={n_iters} ({elapsed:.0f}s): max={new_max:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

    t0 = time.monotonic()

    r0 = 0.1 * norm

        if remaining < 3:

        G = 2.0 * T_mat
        rhs = max_conv - conv  # >= 0

        objective = cp.Minimize(cp.sum_squares(Delta))
        constraints = [
            G @ Delta <= rhs,
            cp.sum(Delta) == 0,
            a + Delta >= 0,
            cp.norm(Delta, 2) <= r0
        ]
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)

            r0 *= 0.5
            if r0 < 1e-8 * norm:
                break
            continue

            if r0 < 1e-8 * norm:
                break
            continue

        Delta_norm = np.linalg.norm(Delta_val)

        found = False
        for alpha in [1.0, 0.5, 0.25, 0.1]:
