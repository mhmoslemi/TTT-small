def subgradient_autocorr(a, k):

    # d/d a[j] = a[k-j] (if 0 <= k-j < n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] = a[k - j]
    return grad

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    # Use sparse representation
    # For constraint k: sum_i a[i]*b[k-i] <= M
    # The coefficient of b[j] in constraint k is a[k-j]
    # So T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T[k, js:je+1] = a[k - np.arange(js, je+1)]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def project_onto_simplex(v, target_sum):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    rho_val = np.max(np.where(cond)[0]) + 1 if np.any(cond) else 0
    if rho_val == 0:
        return v / v.sum() * target_sum
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    projected = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    s = projected.sum()
    if s > 1e-15:
        projected = projected * target_sum / s
    else:
        projected = np.ones(n) * target_sum / n
    return projected

        je = min(n - 1, k)
        if js <= je:

    return log_sum, grad

def project_onto_simplex(v, target_sum):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) * target_sum / n
    rho_val = idx[-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    projected = np.maximum(v - theta, 0.0)
    s = projected.sum()
    if s > 1e-15:
        projected = projected * target_sum / s
    else:
        projected = np.ones(n) * target_sum / n
    return projected
