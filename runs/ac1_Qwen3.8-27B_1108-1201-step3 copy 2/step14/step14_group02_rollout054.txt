Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with LP oracle and golden-section line search
2. Subgradient mirror descent with Polyak step
3. Coordinate descent with optimal step selection
4. Fourier-parameterized initialization
5. Multi-length exploration (1600, 2000, 2400)
6. Aggressive final polishing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def obj_value(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (S * S)

def line_search(a, d, ac_a, ac_d, cross_ad, max_iter=50):
    """Golden section search for optimal alpha in [0,1]."""
    n = len(a)
    
    def f_of_alpha(alpha):
        # conv((1-a)*a + alpha*d, ...) at each lag
        # = (1-alpha)^2 * ac_a + 2*alpha*(1-alpha) * cross_ad + alpha^2 * ac_d
        val = (1-alpha)**2 * ac_a + 2*alpha*(1-alpha) * cross_ad + alpha**2 * ac_d
        return val.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f_of_alpha(c)
    fd = f_of_alpha(d_pt)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = f_of_alpha(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f_of_alpha(d_pt)
    
    alpha = (lo + hi) / 2
    return alpha

def compute_cross(a, b):
    """Compute cross-correlation: sum_i a[i]*b[k-i] for all k."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    # Cross-correlation: ifft(fft(a) * conj(fft(b)))... 
    # Actually for convolution: ifft(fft(a) * fft(b))
    # But we want sum_i a[i]*b[k-i] which is (a * b)[k] (convolution)
    return irfft(A * B, fft_size)[:size]

def fw_lp_optimize(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with LP oracle and golden-section line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # LP oracle: maximize sum(b) s.t. T@b <= M, b >= 0
        # T[k,i] = a[|k-i|] for valid indices
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraints efficiently
        # T[k,i] = a[k-i] if 0 <= k-i < n, else 0
        T = np.zeros((2*n-1, n))
        for j in range(n):
            T[j:j+n, j] = a
        
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: simple FW vertex
            k_star = int(np.argmax(ac))
            # subgradient direction: 2*a[k_star - j]
            sg = np.zeros(n)
            valid = np.arange(max(0, k_star-n+1), min(n, k_star+1))
            sg[valid] = 2 * a[k_star - valid]
            j_min = int(np.argmin(sg)) if sg.max() > 0 else 0
            g = np.zeros(n)
            g[j_min] = 1.0
        else:
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                # Fallback
                k_star = int(np.argmax(ac))
                sg = np.zeros(n)
                valid = np.arange(max(0, k_star-n+1), min(n, k_star+1))
                sg[valid] = 2 * a[k_star - valid]
                j_min = int(np.argmin(sg)) if sg.max() > 0 else 0
                g = np.zeros(n)
                g[j_min] = 1.0
            else:
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                b_sum = b_val.sum()
                if b_sum < 1e-15:
                    g = np.ones(n) / n
                else:
                    g = b_val / b_sum
        
        # Line search
        ac_a = fast_autocorr(a)
        ac_g = fast_autocorr(g)
        cross = compute_cross(a, g)
        
        alpha = line_search(a, g, ac_a, ac_g, cross)
        
        a_new = (1 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FW t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, alpha={alpha:.4f}")
        
        # Stopping criterion
        gap = ac_a.max() - (a @ (2*a - 2*g + g))  # approximate gap
        # Actually, for FW: gap = f(a) - min_{b in S} <grad, b>
        # = 2n*M - 2n*min_j v[j] where v = subgradient
        # Let's just check stagnation
        if t > 100 and abs(obj_value(a) - best_obj) < 1e-12:
            pass  # continue anyway
    
    return best_a, best_obj
