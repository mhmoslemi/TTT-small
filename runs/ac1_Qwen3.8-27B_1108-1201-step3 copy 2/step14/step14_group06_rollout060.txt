"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LSE (log-sum-exp) smoothed optimization with adaptive temperature annealing (45% time)
2. LP best-response with Nesterov acceleration and tight M targets (25% time)
3. Targeted coordinate descent with peak analysis (20% time)
4. Multi-length exploration and final polish (10% time)
Key improvements: FFT-based fast autocorrelation, efficient O(n^2) gradient via 
structured dot products, adaptive annealing schedule, and better peak targeting.
"""

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]

def lse_obj_and_grad(a, T):
    n = len(a)
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_autocorr(a)
    max_c = c.max()
    if max_c < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T * (c - max_c)
    max_shifted = shifted.max()
    exp_vals = np.exp(shifted - max_shifted)
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse_val = max_c + (max_shifted + np.log(exp_sum)) / T
    obj = 2.0 * n * lse_val / (sum_a * sum_a)
    G = np.zeros(n)
    for j in range(n):
        m1 = min(n, 2 * n - 1 - j)
        if m1 > 0:
            G[j] += np.dot(a[:m1], w[j:j + m1])
        if j >= 0:
            G[j] += np.dot(a[:j + 1], w[j::-1])
    grad = 2.0 * n * G / (sum_a * sum_a) - 2.0 * obj * a / sum_a
    return obj, grad


def lse_optimize(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=None):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    if n_stages is None:
        n_stages = max(4, int(np.log2(T_start / T_end)))
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        stage_budget = min(remaining / (n_stages - si), 200.0)

        def obj_wrap(x, T_val=T):
            a_pos = np.maximum(x, 0.0)
            s = a_pos.sum()
            if s < 1e-10:
                return 1e10
            a_norm = a_pos * norm_val / s
            val, _ = lse_obj_and_grad(a_norm, T_val)
            return val

        def grad_wrap(x, T_val=T):
            a_pos = np.maximum(x, 0.0)
            s = a_pos.sum()
            if s < 1e-10:
                return np.zeros(n)
            a_norm = a_pos * norm_val / s
            _, g = lse_obj_and_grad(a_norm, T_val)
            return g

        n_calls = 0
        while time.monotonic() - t0 < t0 + stage_budget:
            if time.monotonic() - t0 > time_limit:
                break
            try:
                res = scipy_minimize(obj_wrap, a.copy(), method='L-BFGS-B',
                                     jac=grad_wrap, options={'maxiter': 200,
                                                             'maxfun': 500,
                                                             'gtol': 1e-7,
                                                             'ftol': 1e-9,
                                                             'maxls': 30})
                n_calls += 1
                if res.x is not None:
                    a_cand = np.maximum(res.x, 0.0)
                    s_c = a_cand.sum()
                    if s_c > 1e-10:
                        a_cand = a_cand * norm_val / s_c
                        score = fast_autocorr(a_cand).max()
                        if score < best_score:
                            best_score = score
                            best_a = a_cand.copy()
                            print(f"      LSE T={T:.1f} call={n_calls}: score={score:.8f}")
                        a = a_cand.copy()
            except Exception:
                break
            if time.monotonic() - t0 > t0 + stage_budget:
                break
    return best_a, best_score


def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T_mat=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T_mat is None:
        T_mat = build_constraint_matrix(a, n)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * x) + 0.2 * np.cos(2.0 * np.pi * 7.0 * x)
    else:
        x = np.linspace(0.0, 1.0, n)
        k1, k2 = rng.integers(1, 10), rng.integers(1, 10)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k1 * x) + 0.2 * np.cos(2.0 * np.pi * k2 * x + 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, remaining * 0.6, seed=99999)
        best_seq, best_score = attempt_update(best_seq, best_score, "cd_final1", t_global)
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_norm2, remaining2, seed=77777)
            best_seq, best_score = attempt_update(best_seq, best_score, "cd_final2", t_global)
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    return [float(x) for x in best_seq]
