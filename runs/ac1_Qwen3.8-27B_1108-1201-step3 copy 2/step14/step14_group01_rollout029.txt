def lse_gd(a0, time_limit, verbose=False):
    """LSE smooth gradient descent for fast initial progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature annealing
    stages = [(50.0, 100), (20.0, 100), (7.0, 100), (3.0, 100), (1.0, 200), (0.3, 200)]

    for T_val, n_steps in stages:
        if time.monotonic() - t0 > time_limit:
            break
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient computation
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            aw = irfft(A * W, fft_size)
            grad = 2.0 * (ca[:n] + aw[:n])

            # Adaptive step with backtracking
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n))
            for _ in range(15):
                a_new = np.maximum(a - step * grad, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a = np.maximum(a - step * grad, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve_fast(a, M, solver, active_indices=None):
    """Solve the LP: max sum(b) s.t. b>=0, T@b <= M (using active constraints)."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_indices is not None and 0 < len(active_indices) < size:
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def nesterov_lp(a0, time_limit, solver, verbose=True):

    norm = np.sqrt(2.0 * n)

    t0 = time.monotonic()

    fft_size = 1 << (size - 1).bit_length()

    y = a.copy()  # Nesterov "lookahead" point
    prev_a = a.copy()

    # Adaptive parameters
    base_alpha = 2.0 / 5.0  # Start with moderate step
    alpha_decay = 0.9999
    min_alpha = 0.01

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current M at y
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Mix of very active and moderately active
        sort_idx = np.argsort(conv_y)[::-1]
        k_active = max(50, min(size, int(0.15 * size)))
        active = sort_idx[:k_active]
        # Also ensure we include the constraint at the max
        if sort_idx[0] not in active:
            active[0] = sort_idx[0]

        # Solve LP
        b = lp_solve_fast(y, M, solver, active)
        if b is None:

            b = lp_solve_fast(y, M, solver, active_small)
        if b is None:
            # Try full
            b = lp_solve_fast(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum  # Scaled best response

                    if b_sum > 1e-15:

        alpha_t = max(alpha_decay ** t, min_alpha) * base_alpha

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = (1.0 - alpha_t) * y + alpha_t * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

                a_new = a_new * norm / s_a

        prev_a = a
        a = a_new

        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()

        if t % 100 == 0:

def gradient_polish(a, norm, n_steps):

def spectral_smooth(a, n_iter=10, strength=0.5):

        # c[k] = IFFT(|A|^2)[k], we want to reduce max c[k]

        # The peak at lag p is caused by spectral power at frequencies where
        # the phase alignment creates constructive interference at that lag
        # Simplified: reduce power proportional to |A(f)|^2 * cos(2*pi*f*p/n)
        
        f = np.arange(fft_size // 2 + 1) / fft_size
        # Weight to reduce: high where cos(2*pi*f*p/n) > 0 and |A|^2 is large
        weight = np.ones(len(A))
        # Modulate to reduce the specific peak
        modulation = np.cos(2 * np.pi * f * fail_peak)
        weight = 1.0 - strength * np.maximum(modulation, 0) * (P / P.sum()) * n
        weight = np.maximum(weight, 0.1)
        
        A_new = A * np.sqrt(weight)
        a_new = np.real(irfft(A_new, fft_size)[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * norm / s

    return a

def targeted_cd(a0, time_limit, rng, verbose=True):
    """Targeted pair coordinate descent focusing on the worst peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_idx = np.argmax(conv)

        # Find the best transfer to reduce the peak
        # Try transferring from high-value positions to low-value positions
        # that would reduce the peak
        i_idx = int(rng.integers(n))
        ai = a[i_idx]
        if ai < norm * 1e-6:
            continue

        # Determine direction: from i to j or from j to i
        best_j = -1
        best_delta = 0.0
        best_val = cur_max

        # Try multiple j candidates
        candidates = set()
        for _ in range(12):
            candidates.add(int(rng.integers(n)))
        # Structured candidates based on peak
        for offset in range(-3, 4):
            j1 = (peak_idx - i_idx + offset) % n
            j2 = (peak_idx + i_idx + offset) % n
            candidates.add(j1)
            candidates.add(j2)
        candidates.discard(i_idx)

        for j in candidates:
            aj = a[j]
            
            # Compute gain: how does transferring delta from i to j affect the peak?
            # conv[k] changes by: delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
            # + second order: -delta^2 at k=2i, +delta^2 at k=2j
            # (approximately, for small delta)
            
            # Compute linear gain at the peak
            k = peak_idx
            g_peak = 0.0
            if 0 <= k - j < n:
                g_peak += a[k - j]
            if 0 <= j - k < n:
                g_peak += a[j - k]
            if 0 <= k - i < n:
                g_peak -= a[k - i]
            if 0 <= i - k < n:
                g_peak -= a[i - k]

            if g_peak > 0:
                # Transferring from i to j would increase the peak, skip
                continue
            
            # Find optimal delta
            # conv[k] + delta * g_peak (approximately linear for small delta)
            # Also need to check other peaks don't increase too much
            
            # Try a range of deltas
            max_delta = ai  # Can't transfer more than we have
            n_samples = 10
            for d in np.linspace(0.1 * max_delta, max_delta, n_samples):
                # Apply transfer
                a[i_idx] = ai - d
                a[j] = aj + d
                s_new = a.sum()
                if s_new > 1e-15:
                    a_temp = a * norm / s_new
                    v = fast_conv_self(a_temp).max()
                    if v < best_val:
                        best_val = v
                        best_delta = d
                        best_j = j
                a[i_idx] = ai
                a[j] = aj
                a = a * norm / a.sum()

        if best_j >= 0 and best_delta > 0:
            a[i_idx] = ai - best_delta
            a[best_j] = a[best_j] + best_delta
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            new_max = fast_conv_self(a).max()
            if new_max < best_max - 1e-10:
                best_max = new_max
                best_a = a.copy()
                no_improve = 0
            else:
                # Revert
                a[best_j] -= best_delta
                a[i_idx] += best_delta
                a = a * norm / a.sum()
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 20:
            # Perturb
            n_perturb = max(10, n // 30)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            no_improve = 0

        if verbose and (t - 200) % 500 == 0:
            pass

    return best_a, best_max

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    prev_a = a.copy()
    running_avg = a.copy()
    n_avg = 1
    base_momentum = 0.85

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        sort_idx = np.argsort(conv_y)[::-1]
        frac_active = max(0.08, 0.25 - 0.001 * t)
        k_active = max(40, min(size, int(frac_active * size)))
        active = sort_idx[:k_active]

        b = lp_solve(y, M, solver, active)
        if b is None:
            active2 = sort_idx[:max(20, k_active // 3)]
            b = lp_solve(y, M, solver, active2)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if len(active) < size:
            check = fast_conv_cross(y, g)
            if check.max() > M * 1.3:
                active3 = sort_idx[:min(size, max(100, int(0.4 * size)))]
                b3 = lp_solve(y, M, solver, active3)
                if b3 is not None:
                    b3_sum = b3.sum()
                    if b3_sum > 1e-15:
                        g = b3 * norm / b3_sum

        alpha_t = 2.0 / (t + 2.0)
        momentum = min(base_momentum + 0.0001 * t, 0.99)

        if t == 1:
            a_new = g.copy()
            y_new = g.copy()
        else:
            y_new = (1.0 - alpha_t) * y + alpha_t * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        prev_a = a
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()

        running_avg = (n_avg * running_avg + a) / (n_avg + 1)
        n_avg += 1

        if t % 50 == 0:
            avg_check = running_avg.copy()
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_max:
                    best_max = sc_avg
                    best_a = avg_check.copy()

        if t % 80 == 0:
            a = gradient_step(a, norm, T_val=2.0, n_steps=3)
            running_avg = (n_avg - 1) * running_avg / max(n_avg, 1) + a
            n_avg += 1
            sc_polish = fast_conv_self(a).max()
            if sc_polish < best_max:
                best_max = sc_polish
                best_a = a.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best_max={best_max:.8f}")

    return best_a, best_max

def spectral_refine(a0, n_iter, strength, norm_val):
    n = len(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a = a0.copy()

    for _ in range(n_iter):
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        c = irfft(A * A, fft_size)[:size]
        peak_val = c.max()
        if peak_val < 1e-15:
            break
        peak_idx = np.argmax(c)

        f = np.arange(len(A)) / fft_size
        modulation = np.cos(2.0 * np.pi * f * peak_idx)
        weight = 1.0 - strength * np.maximum(modulation, 0.0)
        weight = np.maximum(weight, 0.05)
        A_new = A * np.sqrt(weight)
        a_new = np.real(irfft(A_new, fft_size)[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * norm_val / s

    return a

def targeted_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    no_improve = 0
    total_imp = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        peak_idx = int(np.argmax(conv))

        i_idx = int(rng.integers(n))
        ai = a[i_idx]
        if ai < norm * 1e-8:
            no_improve += 1
            if no_improve > 30:
                n_perturb = max(5, n // 40)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a[idx] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                no_improve = 0
            continue

        candidates = set()
        for _ in range(10):
            candidates.add(int(rng.integers(n)))
        for off in range(-4, 5):
            candidates.add((peak_idx - i_idx + off) % n)
            candidates.add((peak_idx + i_idx + off) % n)
            candidates.add((off + i_idx) % n)
        candidates.discard(i_idx)

        best_j = -1
        best_delta = 0.0
        best_val = cur_max

        for j in candidates:
            if j == i_idx:
                continue
            aj = a[j]
            k = peak_idx
            g_peak = 0.0
            if 0 <= k - j < n:
                g_peak += a[k - j]
            if 0 <= j - k < n:
                g_peak += a[j - k]
            if 0 <= k - i_idx < n:
                g_peak -= a[k - i_idx]
            if 0 <= i_idx - k < n:
                g_peak -= a[i_idx - k]

            if g_peak > 0.0:
                continue

            for frac in [0.2, 0.5, 0.7, 0.9, 1.0]:
                d = ai * frac
                a[i_idx] = ai - d
                a[j] = aj + d
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i_idx] = ai
                    a[j] = aj
                    continue
                a_temp = a * norm / s_new
                v = fast_conv_self(a_temp).max()
                if v < best_val:
                    best_val = v
                    best_delta = d
                    best_j = j
                a[i_idx] = ai
                a[j] = aj
                a = a * (norm / a.sum()) if a.sum() > 1e-15 else a

        if best_j >= 0 and best_delta > 0:
            a[i_idx] = ai - best_delta
            a[best_j] = a[best_j] + best_delta
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            new_max = fast_conv_self(a).max()
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                no_improve = 0
                total_imp += 1
            else:
                a[best_j] -= best_delta
                a[i_idx] += best_delta
                a = a * norm / a.sum()
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break

        if verbose and total_imp % 200 == 0 and total_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD imp={total_imp} ({elapsed:.0f}s): best={best_max:.8f}")

    return best_a, best_max

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.35 * np.cos(2 * np.pi * 3 * x) + 0.12 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.2 + 0.8 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.3
    elif choice == 7:
        a = 0.4 + 0.4 * np.sin(3 * np.pi * x) * np.cos(np.pi * x)
        a = np.maximum(a, 0.0)
    elif choice == 8:
        a = np.ones(n)
        a[rng.choice(n, size=n // 4, replace=False)] *= 0.3
    else:
        a = 0.3 + 0.7 * rng.random(n)
        a = np.convolve(a, np.ones(5) / 5, mode='same')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Nesterov with adaptive
    acceleration, gradient polishing, spectral refinement, targeted CD,
    and multi-start/multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LSE GD (8%)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} ({t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gd(best_seq, t1, verbose=False)
        try_update(a_out, f"lse_{len(a_out)}")

    # Phase 2: LP-Nesterov main (40%)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_{len(a_out)}")

    # Phase 3: Spectral refinement + re-optimize (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: Spectral + re-opt n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 20:
        norm_val = np.sqrt(2.0 * len(best_seq))
        n_spec = max(3, int(t3 * 0.3))
        a_spec = spectral_refine(best_seq, n_spec, 0.25, norm_val)
        try_update(a_spec, "spec1")
        a_spec2 = spectral_refine(best_seq, n_spec, 0.5, norm_val)
        try_update(a_spec2, "spec2")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = nesterov_lp(best_seq, min(remaining * 0.7, 80.0), solver, verbose=False)
            try_update(a_out, f"lp_spec_{len(a_out)}")

    # Phase 4: LP-Nesterov round 2 (20%)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP-Nesterov r2 n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_{len(a_out)}")

    # Phase 5: Targeted CD (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Targeted CD n={len(best_seq)} ({t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = targeted_cd(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"cd_{len(a_out)}")

    # Phase 6: Multi-start with different init (8%)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-start ({remaining:.0f}s) ===")
    if remaining > 60:
        t_each = min(remaining * 0.25, 60.0)
        for ms in range(3):
            if time_budget - (time.monotonic() - t_global) < 30:
                break
            init_ms = structured_init(n_base, rng)
            a_init, _ = lse_gd(init_ms, min(t_each * 0.3, 20.0), verbose=False)
            a_out, _ = nesterov_lp(a_init, min(t_each * 0.7, 40.0), solver, verbose=False)
            try_update(a_out, f"ms{ms}_{len(a_out)}")

    # Phase 7: Longer length exploration (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 500
    t7 = min(remaining * 0.50, 150.0)
    print(f"\n=== Phase 7: Long n={n_long} ({t7:.0f}s) ===")
    if t7 > 40 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lse_gd(init_long, min(t7 * 0.2, 30.0), verbose=False)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out2, _ = nesterov_lp(a_out, min(remaining * 0.6, 100.0), solver, verbose=False)
            try_update(a_out2, f"long_{n_long}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out3, _ = targeted_cd(a_out2, min(remaining * 0.5, 30.0), rng, verbose=False)
                try_update(a_out3, f"cd_long_{n_long}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish ({remaining:.0f}s) ===")
    if remaining > 20:
        a_out, _ = lse_gd(best_seq, min(remaining * 0.4, 40.0), verbose=False)
        try_update(a_out, "final_lse")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = targeted_cd(best_seq, min(remaining * 0.6, 30.0), rng, verbose=False)
            try_update(a_out, "final_cd")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = nesterov_lp(best_seq, min(remaining * 0.5, 25.0), solver, verbose=False)
            try_update(a_out, "final_lp")

    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
