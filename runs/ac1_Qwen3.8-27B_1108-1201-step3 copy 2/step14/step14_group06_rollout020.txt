def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to_sqrt(a, n):
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * np.sqrt(2.0 * n) / s

def eg_optimize(a0, time_limit, eta0=5.0, decay=0.0001, verbose=True):
    """Exponentiated gradient optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = normalize_to_sqrt(a, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-12:
            break
        
        # Find top few peaks for smoother gradient
        top_k = min(10, len(c))
        top_indices = np.argsort(c)[::-1][:top_k]
        # Weight peaks by their value
        peak_weights = c[top_indices]
        peak_weights = peak_weights / peak_weights.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for idx, pk_idx in enumerate(top_indices):
            # g_j += peak_weights[idx] * a_{pk_idx - j}
            # a_{pk_idx - j} is valid when 0 <= pk_idx - j <= n-1
            j_start = max(0, pk_idx - n + 1)
            j_end = min(n - 1, pk_idx)
            if j_start <= j_end:
                vals = a[pk_idx - j_start:pk_idx - j_end + 1][::-1]
                g[j_start:j_end+1] += peak_weights[idx] * vals
        
        # Exponentiated gradient update
        eta = eta0 / (1.0 + t * decay)
        # Scale gradient to have reasonable magnitude
        g_norm = np.sqrt(np.sum(g**2))
        if g_norm > 1e-15:
            g_scaled = g / g_norm
            update = np.exp(-eta * g_scaled)
            a = a * update
            a = np.maximum(a, 1e-15)  # prevent underflow
            a = normalize_to_sqrt(a, n)
        
        n_iters += 1
        
        score = c_max  # already computed
        # Actually need to recompute since a changed
        # But for speed, check every 10 iterations
        if n_iters % 10 == 0 or n_iters < 50:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    # Final check
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def eg_optimize_v2(a0, time_limit, verbose=True):
    """Improved EG with adaptive step size and peak tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = normalize_to_sqrt(a, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Track peak history for smoothing
    peak_history = []
    learning_rate = 1.0 / max(a.max(), 0.1)
    
    n_iters = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Softmax over peaks for smooth gradient
        T_smooth = c_max * 0.1  # temperature
        shifted = (c - c_max) / T_smooth
        w = np.exp(shifted)
        w = w / w.sum()
        
        # Compute gradient using all lags weighted by softmax
        g = np.zeros(n)
        # For each lag k, g_j += w[k] * a[k-j] (for valid j)
        # This is equivalent to: g = conv(w_padded, a_reflected) restricted to [0, n-1]
        # More efficiently: g[j] = sum_k w[k] * a[k-j] for valid k-j
        # = sum_{i=0}^{n-1} a[i] * w[j+i] (with w[k] = 0 for k < 0 or k >= 2n-1)
        
        # Use FFT: g = IFFT(FFT(w) * conj(FFT(a_padded)))[:n]
        fft_size = 1 << (2 * n - 1).bit_length()
        w_full = np.zeros(fft_size)
        w_full[:2*n-1] = w
        a_full = np.zeros(fft_size)
        a_full[:n] = a
        
        W = rfft(w_full, fft_size)
        A = rfft(a_full, fft_size)
        # g[j] = sum_k w[k] * a[k-j] = (w * a_reflected)[j] where a_reflected[i] = a[n-1-i]... 
        # Actually, let me think more carefully.
        # g[j] = sum_k w[k] * a[k-j] for 0 <= k-j <= n-1
        # Let i = k-j, then k = i+j, so g[j] = sum_i a[i] * w[i+j]
        # This is the cross-correlation of a and w!
        # cross_corr[j] = sum_i a[i] * w[i+j]
        # In FFT: cross_corr = IFFT(conj(FFT(a)) * FFT(w)) (with appropriate padding)
        
        W = rfft(w_full, fft_size)
        A = rfft(a_full, fft_size)
        cross = irfft(np.conj(A) * W, fft_size)[:n]
        g = cross
        
        # Adaptive step size
        g_max = g.max()
        if g_max > 1e-15:
            eta = min(learning_rate, 2.0 / (g_max * n**0.5))
            update = np.exp(-eta * g / g_max)
            a = a * update
            a = np.maximum(a, 1e-30)
            a = normalize_to_sqrt(a, n)
        
        n_iters += 1
        
        if n_iters % 5 == 0 or n_iters < 20:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG2 iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a * norm_val / max(a.sum(), 1e-15)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        
        # Tight M
        eps = min(0.05, 0.005 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        
        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break
        
        g = b * norm_val / b.sum()
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm_val / max(y_new.sum(), 1e-15)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * norm_val / max(a_new.sum(), 1e-15)
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        if time.monotonic() - last_improve > 90.0:
            break
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coord_descent(a0, time_limit, seed=42, verbose=False):
    """Targeted coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a * norm_val / max(a.sum(), 1e-15)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        k_peak = int(np.argmax(c))
        
        # Find top contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]
        
        # Find low-value positions (targets for redistribution)
        low_positions = np.argsort(a)[:20]
        
        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_positions:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try random moves
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5)**2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3)**2) + 0.7 * np.exp(-5.0 * (x - 0.7)**2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma)**2)
    elif choice == 6:
        # Random with smoothing
        a = rng.random(n)
        # Smooth with a kernel
        kernel = np.exp(-0.5 * (np.arange(-5,6) / 3.0)**2)
        a = np.convolve(a, kernel, mode='same')
    else:
        x = np.linspace(0.0, 1.0, n)
        # W-shaped
        a = (1 - np.abs(2*x - 1)) ** 2 + 0.3 * np.sin(4*np.pi*x)**2
        a = np.maximum(a, 0.0)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
