3. FFT-accelerated pair coordinate descent for finishing touches
4. Multi-length exploration with interpolation-based warm starts
5. Multiple initialization strategies for robustness
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a, norm_val):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_val / len(a)
    return a * norm_val / s


def build_conv_matrix(a, active_indices):
    """Build the constraint matrix rows for active indices only."""
    n = len(a)
    n_active = len(active_indices)
    A_mat = np.zeros((n_active, n))
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            A_mat[idx, i_start:i_end + 1] = a[(i_end - i_start)::-1] if i_end > i_start else a[i_start:i_end + 1]
    return A_mat


def lp_solve_hg(a, M, active_indices):
    """Solve max sum(b) s.t. (a*b)[k] <= M for k in active_indices, b >= 0 using HiGHS."""
    n = len(a)
    A_mat = build_conv_matrix(a, active_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
        return None
    except Exception:
        return None


def nesterov_lp_optimize(a0, time_limit, verbose=True):
    """Main optimization: Nesterov-accelerated LP best-response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    a = a * norm / s

    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    size = 2 * n - 1
    full_indices = np.arange(size)
    iteration = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        # Use a threshold that adapts: start tight, loosen if LP doesn't improve
        threshold_ratio = 0.75 + 0.1 * (0.5 + 0.5 * np.sin(iteration * 0.05))
        active = np.where(conv_y >= M * threshold_ratio)[0]
        # Always include k=0 and a few neighbors
        for extra in [0, 1, size - 1, size - 2]:
            if extra not in active:
                active = np.append(active, extra)
        active = np.sort(active)

        if len(active) > size // 2:
            active = full_indices

        b = lp_solve_hg(y, M, active)
        if b is None:
            # Fallback: full constraints
            b = lp_solve_hg(y, M, full_indices)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Check for violations of dropped constraints
        if len(active) < size:
            check_conv = fast_conv_cross(y, g)
            violated = np.where(check_conv > M * (1.0 + 1e-5))[0]
            if len(violated) > 0:
                new_active = np.union1d(active, violated)
                b2 = lp_solve_hg(y, M, new_active)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (iteration + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if iteration == 1:
            a_new = g.copy()
        else:
            momentum = (iteration - 1.0) / (iteration + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()
            no_improve_count = 0

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={iteration} ({elapsed:.0f}s): best_max={best_max:.8f}")

        # Periodically try averaging
        if iteration % 100 == 0:
            avg = (y + a) / 2.0
            avg = normalize(avg, norm)
            sc_avg = fast_conv_self(avg).max()
            if sc_avg < best_max:
                best_max = sc_avg
                best_a = avg.copy()

    return best_a, best_max


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with gradient-informed search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imps = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            cur_max = conv.max()

            # Candidate j's: random + structured
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            # Structured: near peak lags
            peak_threshold = cur_max * 0.92
            peak_lags = np.where(conv >= peak_threshold)[0]
            for kl in peak_lags[:3]:
                j_s = (kl - i) % n
                j_candidates.add(j_s)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:10]:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Minimize max over delta in [0, ai]
                # Use golden-section-like approach with the key points
                n_samples = 20
                deltas = np.linspace(0.0, ai, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d == 0:
                        continue
                    new_conv = conv + 2.0 * d * g
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if (i + j) < size:
                        new_conv[i + j] -= 2.0 * d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 1e-15:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max - 1e-13:
                        best_max = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imps += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(3, n // 100)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
        else:
            no_improve = 0

        if verbose and total_imps % 200 == 0 and total_imps > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imps={total_imps} ({elapsed:.0f}s): best_max={best_max:.8f}")

    return best_a, best_max


def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.2 + 0.3
    elif style == 2:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.7) ** 2) + 0.3
    elif style == 3:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5 + 0.2
    elif style == 4:
        a = 0.4 + 0.4 * np.sin(3 * np.pi * x) ** 2 + 0.2 * np.cos(7 * np.pi * x)
    elif style == 5:
        a = np.cos(np.pi * x / 2.0) ** 2 * (1.0 + 0.3 * np.sin(4 * np.pi * x))
    elif style == 6:
        # Multi-peak
        peaks = rng.uniform(0.1, 0.9, size=4)
        a = np.ones(n) * 0.3
        for p in peaks:
            a += 0.7 * np.exp(-50.0 * (x - p) ** 2)
    else:
        a = np.ones(n) * 0.5 + 0.3 * rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, style=0)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP-Nesterov main optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_work = best_seq.copy()
        a_out, _ = nesterov_lp_optimize(a_work, t1, verbose=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Another LP-Nesterov round (different screening)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP-Nesterov round 2 n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_work = best_seq.copy()
        a_out, _ = nesterov_lp_optimize(a_work, t2, verbose=True)
        try_update(a_out, f"lp2_n{n2}")

    # Phase 3: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n3}")

    # Phase 4: Try a different length (longer)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_base + 200
        t4 = min(remaining * 0.30, 250.0)
        print(f"\n=== Phase 4: Longer length n={n_long} (budget={t4:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t4, 150.0), verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out2, _ = pair_cd_optimize(interp_a if 'interp_a' in dir() else best_seq, min(remaining * 0.5, 60.0), rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 5: Final LP round on best
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.40, 200.0)
    print(f"\n=== Phase 5: Final LP n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, verbose=False)
        try_update(a_out, f"lp_final_n{n5}")

    # Phase 6: Final pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = len(best_seq)
    t6 = min(remaining * 0.50, 150.0)
    print(f"\n=== Phase 6: Final PCD n={n6} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=False)
        try_update(a_out, f"pcd_final_n{n6}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
