```python
def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T_active(a, active_indices):
    """Build sparse-ish T matrix rows for active constraints only."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(active_indices), n))
    for m, k in enumerate(active_indices):
        # T[m, j] = a[|k-j|] if 0 <= |k-j| <= n-1
        for j in range(n):
            d = abs(k - j)
            if d < n:
                T[m, j] = a[d]
    return T

def lp_solve_constraint_gen(a, M, solver, max_inner=10, initial_frac=0.1):

    n = len(a)
    size = 2 * n - 1

    # Use a probe: b = ones (or current best)
    b_probe = np.ones(n)
    conv_probe = fast_conv_cross(a, b_probe)
    
    # Initial active set: top fraction of constraints
    n_initial = max(20, int(initial_frac * size))
    active = np.argsort(conv_probe)[-n_initial:]
    
    b = None

        # Solve LP with current active constraints
        T = build_T_active(a, active)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None

        if b.sum() < 1e-15:
            return None

        conv_check = fast_conv_cross(a, b)
        violated = np.where(conv_check > M * (1 + 1e-6))[0]
        if len(violated) == 0:
            return b

        active = np.union1d(active, violated)
        # Cap active set size

            # Keep the most violated

            active = active[order]

        top_k = np.argsort(conv_check)[-50:]

    # If we didn't converge, do one final solve with all detected constraints

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(30.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient computation via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # g1[i] = sum_k a[k] * w[k+i] (circular cross-correlation)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            # g2[i] = sum_k a[k] * w[i-k] = conv(a,w)[i]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            # Line search
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = 1.0 / (T_val * np.sqrt(n) * grad_norm / norm)
            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a * norm / s

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Efficient pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score_fast(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Find best j to transfer to
            # Use gradient info: peak lags indicate where conv is highest
            peak_threshold = cur_max * 0.9
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Build gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.01, ai * 0.99, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.08 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.exp(-5.0 * (x - 0.5) ** 2)
    else:
        a = 0.4 + 0.6 * np.cos(np.pi * x) ** 4
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

```python
        T[m, lo:hi] = a[k-hi+1:k-lo+1][::-1]
    return T
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations + parent loading
2. LSE smooth gradient descent for fast initial progress
3. LP best-response with constraint generation + Nesterov acceleration + adaptive restart
4. Pair coordinate descent with vectorized 1D minimization for final polishing
5. Multi-length exploration via interpolation

Key improvements over previous:
- Corrected constraint matrix construction for LP
- Constraint generation (start with subset, add violated) for faster LP solves
- Adaptive restart when stuck
- Running average tracking alongside Nesterov iterate
- Better time allocation and more robust edge handling

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print(f"  Using solver (fallback): ECOS")
    return cp.ECOS


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score_fast(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def build_T_active(a, active_indices):
    """Build T matrix rows for given constraint indices.
    T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0."""
    n = len(a)
    n_act = len(active_indices)
    T = np.zeros((n_act, n))
    for m, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[m, lo:hi] = a[k - hi + 1: k - lo + 1][::-1]
    return T


def lp_solve_constraint_gen(a, M, solver, max_inner=8, initial_frac=0.08):
    """LP with constraint generation: start with top constraints, add violated ones iteratively."""
    n = len(a)
    size = 2 * n - 1

    # Use a probe to find initial active set
    b_probe = np.ones(n)
    conv_probe = fast_conv_cross(a, b_probe)

    # Initial active set: top fraction + always include center region
    n_initial = max(30, int(initial_frac * size))
    top_idx = np.argsort(conv_probe)[-n_initial:]
    # Also include the center (most likely to be binding for self-conv)
    center = n - 1
    active = np.unique(np.concatenate([top_idx, [center, max(0, center-5), min(size-1, center+5)]]))

    b = None
    for inner in range(max_inner):
        T = build_T_active(a, active)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b.sum() < 1e-15:
            return None

        # Check all constraints
        conv_check = fast_conv_cross(a, b)
        violated = np.where(conv_check > M * (1 + 1e-5))[0]
        if len(violated) == 0:
            return b

        # Add violated constraints
        active = np.union1d(active, violated)

        # Cap active set: keep most violated + top original
        if len(active) > 400:
            viol_amt = conv_check[active] - M
            order = np.argsort(viol_amt)[-400:]
            active = active[order]

    # Final solve with all detected constraints
    if b is not None:
        conv_check = fast_conv_cross(a, b)
        all_active = np.where(conv_check > M * 0.3)[0]
        if 0 < len(all_active) <= size:
            T = build_T_active(a, all_active)
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                    if b_new.sum() > 1e-15:
                        b = b_new
            except Exception:
                pass

    return b


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with adaptive temperature schedule."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(40.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            # Gradient via FFT cross-correlations
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # g1[i] = sum_k a[k] * w[k+i] (circular)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            # g2[i] = conv(a, w)[i] = sum_k a[k] * w[i-k]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Backtracking line search
            step = norm / (T_val * np.sqrt(n) * grad_norm)
            for trial in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
            else:
                a = a * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, restart_patience=120):
    """LP best-response with Nesterov acceleration, constraint generation, and adaptive restart."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    avg_sum = np.zeros(n)
    n_avg = 0
    no_improve_count = 0
    rng = np.random.default_rng(777)

    max_iters = 80000
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # LP solve with constraint generation
        b = lp_solve_constraint_gen(y, M, solver, max_inner=6, initial_frac=0.06)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track running average (Catmull scheme)
        avg_sum += a
        n_avg += 1

        # Check improvement every few iterations
        if t % 10 == 0:
            candidates = []
            sc_y = fast_conv_self(y).max()
            candidates.append((sc_y, y))
            sc_a = fast_conv_self(a).max()
            candidates.append((sc_a, a))
            if n_avg >= 50:
                avg_a = avg_sum / n_avg
                s_avg = avg_a.sum()
                if s_avg > 1e-15:
                    avg_a = avg_a * norm / s_avg
                    sc_avg = fast_conv_self(avg_a).max()
                    candidates.append((sc_avg, avg_a))

            cur_best_sc, cur_best_arr = min(candidates, key=lambda x: x[0])
            if cur_best_sc < best_score - 1e-10:
                best_score = cur_best_sc
                best_a = cur_best_arr.copy()
                no_improve_count = 0
            else:
                no_improve_count += 10

        # Adaptive restart: perturb when stuck
        if no_improve_count >= restart_patience:
            n_perturb = max(10, n // 15)
            idx = rng.choice(n, size=n_perturb, replace=False)
            scale = 1.0 + 0.2 * rng.standard_normal(n_perturb)
            a[idx] *= scale
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            y = a.copy()
            no_improve_count = 0
            if verbose:
                print(f"    [Restart at t={t}] best={best_score:.8f}")

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Efficient pair coordinate descent with gradient-informed candidate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score_fast(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Gradient-informed j selection
            peak_threshold = cur_max * 0.88
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:5]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Build linear gain vector g where new_conv ≈ conv + delta * g
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta via sampling
                n_samples = 25
                deltas = np.linspace(0.01, ai * 0.98, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    new_conv = conv + d * g
                    # Quadratic corrections at diagonal
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2.0 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2.0 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve == 2:
                n_perturb = max(5, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x ** 0.6) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.exp(-5.0 * (x - 0.5) ** 2)
    elif choice == 9:
        a = 0.4 + 0.6 * np.cos(np.pi * x) ** 4
    elif choice == 10:
        a = np.exp(-4.0 * (x - 0.2) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.8) ** 2)
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Allocates time across optimization phases."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Quick LSE polish (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Main LP Nesterov (40%)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.42, 700.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: Pair CD polish (14%)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.14, 250.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_optimize(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, "pcd1")

    # Phase 4: Second LP round (18%)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t4, solver, verbose=True, restart_patience=80)
        try_update(a_out, "lp2")

    # Phase 5: Try different length (12%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    print(f"\n=== Phase 5: n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        init_long = interpolate_seq(best_seq, n_long)
        t5 = min(remaining * 0.35, 150.0)
        a_out, _ = nesterov_lp_optimize(init_long, t5, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.4, 80.0)
            a_out2, _ = pair_cd_optimize(a_out if a_out is not None else init_long, t5b, rng, verbose=False)
            try_update(a_out2, f"pcd_long")

    # Phase 6: Final polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t6 = min(remaining * 0.5, 60.0)
        a_out, _ = lse_gradient_descent(best_seq.copy(), t6, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd_optimize(best_seq.copy(), min(remaining * 0.7, 40.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
