            restart_score = evaluate_objective(restart_vector)
            if restart_score < best_score and restart_score < best_ever:
                best_seq = restart_vector
                best_score = restart_score
                print(f"Restart improved with score {best_score}")
            remaining -= t_restart
        else:
            break

    # Phase III: Length expansion exploration
    lengths_to_try = [n0 + 100, n0 + 200, n0 + 300, n0 + 400]  # sample some expanded lengths
    for new_n in lengths_to_try:
        # Check if new_n is too large (convolution cost is O(n^2))
        if new_n > 1000000:
            print("Skipping too large sequence lengths")
            break
        t_length = remaining / len(lengths_to_try) * (0.7 if new_n < best_ever_sequence]
        return best_ever_sequence

    # If we haven't found anything, return the given parent (but parent might be invalid)
    fallback = height_sequence_1 if parent is not None else None
    if fallback is None:
        fallback = parent if valid_parent else None

    # If fallback is still None, create a fallback from scratch
    if fallback is None or sum(fallback) < 0.001:
        fallback = structured_init(n0, rng)

    # Always normalize the fallback to our required sum if necessary
    fallback_sum = np.sqrt(2 * n0)
    fallback_score = evaluate_sequence(fallback)
    if fallback_score < best_ever:
        best_ever_sequence = fallback
        best_ever = fallback_score

    print(f"Final candidate sequence: n={len(best_ever_sequence)}, score={best_ever}")
    return best_ever_sequence```
