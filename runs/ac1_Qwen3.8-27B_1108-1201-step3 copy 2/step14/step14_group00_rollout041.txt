    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, eps):
    """Compute gradient of the log-sum-exp smoothed max autocorrelation."""
    n = len(a)
    c = fft_autocorr(a)
    # Numerically stable softmax
    c_shift = c - c.max()
    w = np.exp(c_shift / eps)
    W = w.sum()
    w /= W  # Now w is a probability distribution over lags
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * (w * a)[i] in the "cross-correlation" sense
    # Using FFT: compute conv(w_rev, a) and extract the right portion
    
    w_rev = w[::-1]  # length 2n-1
    full_size = len(w_rev) + n - 1  # 3n-2
    fft_size = 1 << (full_size - 1).bit_length()
    Wr = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(Wr * A, fft_size)[:full_size]

    # conv[2n-2-i] for i=0 is conv[2n-2], for i=n-1 is conv[n-1]
    idx = 2*n - 2 - np.arange(n)  # [2n-2, 2n-3, ..., n-1]
    grad = 2.0 * conv[idx]

    n = len(v)
    if n == 0:
        return v

    u = np.sort(v)[::-1]

import numpy as np

from scipy.fft import rfft, irfft

def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, eps):
    """Compute (1/2) * gradient of log-sum-exp smoothing of max autocorr."""
    n = len(a)
    c = fft_autocorr(a)
    c_max = c.max()
    w = np.exp((c - c_max) / eps)
    W = w.sum()
    w /= W
    
    # Compute grad_half[i] = sum_j w[i+j] * a[j]
    w_rev = w[::-1]
    full_len = len(w_rev) + n - 1
    fft_size = 1 << (full_len - 1).bit_length()
    Wr = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(Wr * A, fft_size)[:full_len]
    idx = 2 * n - 2 - np.arange(n)
    grad_half = conv[idx]
    return 2.0 * grad_half, c, w

def project_simplex(v, s):
    n = len(v)
    if n <= 1:
        return np.array([max(s, 0.0)]) if n == 1 else v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, s / n)
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    rs = result.sum()
    if rs > 0 and abs(rs - s) > 1e-10 * s:
        result *= s / rs
    return result
