def softmax_gradient_step(a, T, lr):
    """One step of soft-max gradient descent."""
    n = len(a)
    size = 2 * n - 1

    a_new = np.maximum(a_new, 0.0)
    
    return a_new, max_c, w

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    # Build convolution matrix T where (T@b)_i = sum_j a[|i-j|]... 
    # Actually (T@b)_i = sum_j a_j * b_{i-j} for valid indices
    # T is size x n
    T = np.zeros((size, n))
    for j in range(n):
        # a_j contributes to c_i for i = j, j+1, ..., j+n-1 (as first factor)
        # and for i = n-1-j, n-j, ..., 2n-2-j (as second factor, via symmetry)
        # Wait, I need to be more careful.
        # c_i = sum_k a_k * b_{i-k} (linear convolution)
        # For fixed j (index in b), b_j appears in c_i when i-k = j, i.e., i = k+j
        # for k = 0, ..., n-1 (where a_k is defined)
        # So b_j appears in c_{j}, c_{j+1}, ..., c_{j+n-1} with coefficients a_0, a_1, ..., a_{n-1}
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max gradient descent with geometric T-annealing (primary workhorse)
2. LP best-response with Nesterov acceleration (basin jumping)
3. Pair coordinate descent for local polish
4. Multi-n exploration and random restarts
Key difference from previous: much more aggressive smooth gradient phase with
adaptive step sizes, and a "top-k focused" gradient variant that concentrates
on the largest peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    if not HAS_CVXPY:
        return None
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def smooth_gradient_descent(a0, time_limit, T_start=5.0, T_end=0.05,
                            lr_init=0.01, anneal_rate=0.999,
                            top_k_frac=0.0, verbose=True):
    """
    Soft-max gradient descent with temperature annealing.
    top_k_frac: if > 0, only use top fraction of autocorrelation values in the weights.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    T = T_start
    lr = lr_init
    t0 = time.monotonic()
    n_steps = 0
    no_improve = 0

    size = 2 * n - 1
    # FFT size for cross-correlation: need >= (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    k_focus = max(1, int(top_k_frac * size)) if top_k_frac > 0 else size

    while time.monotonic() - t0 < time_limit:
        if T < T_end:
            break

        c = fast_autocorr(a)
        max_c = c.max()

        if max_c < best_score - 1e-12:
            best_score = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        # Compute soft-max weights
        exp_c = np.exp((c - max_c) / T)
        if top_k_frac > 0 and k_focus < size:
            # Zero out weights for non-top-k elements
            threshold = np.sort(exp_c)[-k_focus]
            exp_c[exp_c < threshold] = 0.0
        w = exp_c / (exp_c.sum() + 1e-30)

        # Gradient: 2 * (w cross-correlated with a)[:n]
        # result[k] = sum_i w[i] * a[i-k] for valid i, a
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(W * np.conj(A), fft_size)
        grad = 2.0 * cross[:n]

        # Adaptive step size via backtracking
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-30:
            alpha = min(lr, 1.0 / (np.linalg.norm(grad) + 1e-10))
        else:
            alpha = 0.0

        improved = False
        for _ in range(8):
            if alpha < 1e-12:
                break
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try, norm)
            c_try = fast_autocorr(a_try)
            if c_try.max() < max_c - 1e-12:
                a = a_try
                improved = True
                n_steps += 1
                break
            alpha *= 0.5

        if not improved:
            # Take a very small step to avoid complete stall
            a_try = a - lr * 0.001 * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try, norm)
            a = a_try
            n_steps += 1

        # Anneal
        T *= anneal_rate

        if verbose and n_steps % 2000 == 0 and n_steps > 0:
            elapsed = time.monotonic() - t0
            print(f"    SMOOTH-GD step={n_steps} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def concentrated_gradient_descent(a0, time_limit, verbose=True):
    """
    Gradient descent that focuses on the top few peaks using a hard threshold.
    Iteratively targets the single largest peak, then the next, etc.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    n_rounds = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_rounds += 1
        c = fast_autocorr(a)
        max_c = c.max()

        if max_c < best_score - 1e-12:
            best_score = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        # Focus on top 5% of peaks
        k = max(1, size // 20)
        top_indices = np.argsort(c)[-k:]
        c_focus = np.zeros(size)
        c_focus[top_indices] = c[top_indices]
        w = c_focus / (c_focus.sum() + 1e-30)

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(W * np.conj(A), fft_size)
        grad = 2.0 * cross[:n]

        # Line search
        alpha = 0.01 / (np.linalg.norm(grad) / n + 1e-10)
        for _ in range(6):
            if alpha < 1e-14:
                break
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try, norm)
            if fast_autocorr(a_try).max() < max_c - 1e-12:
                a = a_try
                break
            alpha *= 0.5
        else:
            a_try = a - alpha * 0.1 * grad
            a_try = np.maximum(a_try, 0.0)
            a = normalize(a_try, norm)

        if verbose and n_rounds % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CONCENTRATED step={n_rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)_i = sum_j a_j * b_{i-j}."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, window=30, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j = (i + offset) % n
                if j == i:
                    continue
                if a[i] < norm * 1e-10:
                    break

                # Per-unit change vector for moving mass from i to j
                v = np.zeros(size)
                # +2*a shifted to position j
                lo_j = j
                hi_j = min(size, j + n)
                v[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
                # -2*a shifted to position i
                lo_i = i
                hi_i = min(size, i + n)
                v[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]

                # Find best d via sampling
                best_d = 0.0
                best_val = conv.max()
                n_samples = 12
                max_d = a[i] * 0.99
                if max_d <= 0:
                    continue
                for d in np.linspace(0, max_d, n_samples)[1:]:
                    new_conv = conv + d * v
                    # Diagonal corrections
                    if 2 * i < size:
                        new_conv[2 * i] += d * d - 0  # extra d^2 from (a[i]-d)^2 vs a[i]^2
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if i + j < size and i != j:
                        new_conv[i + j] -= 2 * d * d
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = normalize(a, norm)
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = normalize(a, norm)
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

    return best_a, best_score


def random_restart(a0, rng, n_preserve=0.95):
    """Create a perturbed version of a0 to escape local minima."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return normalize(a, norm)
    a = a / s

    n_perturb = int(n * (1 - n_preserve))
    idx = rng.choice(n, size=n_perturb, replace=False)
    a[idx] = 0.0
    a[idx] += rng.random(n_perturb) * a.mean()
    a = np.maximum(a, 0.0)
    return normalize(a, norm)
