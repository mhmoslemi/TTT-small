def mirror_descent(a0, time_limit, verbose=True):
    """
    Mirror descent with negative entropy mirror map.
    Update: a_j <- a_j * exp(-eta * grad_j) / Z
    This naturally maintains positivity and can find different optima.
    """

    norm = np.sqrt(2.0 * n)

    t0 = time.monotonic()
    steps = 0
    eta = 1.0 / n
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        M_curr = b.max()
        
        # Compute subgradient of max autocorrelation
        p = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        if js <= je:
            grad[js:je + 1] = 2.0 * a[p - np.arange(js, je + 1)]
        
        # Also add contribution from second peak for robustness
        b_sorted = b.copy()
        b_sorted[p] = -np.inf
        p2 = int(np.argmax(b_sorted))
        if b_sorted[p2] > 0.5 * M_curr:
            w2 = b_sorted[p2] / M_curr
            js2 = max(0, p2 - n + 1)
            je2 = min(n - 1, p2)
            if js2 <= je2:
                grad[js2:je2 + 1] += w2 * 2.0 * a[p2 - np.arange(js2, je2 + 1)]
        
        # Project gradient onto sum-preserving subspace
        grad -= np.mean(grad)
        
        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        # Normalize: we want sum(exp(log_a_new)) = norm
        log_a_new -= np.logaddexp.reduce(log_a_new)  # normalize so sum exp = 1
        log_a_new += np.log(norm)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score_new = fast_autocorr(a_new).max()

            a = a_new
            best_M = score_new
            best_a = a.copy()
            eta = min(eta * 1.02, 5.0 / n)
        else:
            eta *= 0.85
        
        # Anneal eta
        eta = max(eta, 0.01 / n)
        
        steps += 1
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc step={steps} ({elapsed:.0f}s): M={best_M:.8f}, eta={eta:.4e}")

def lse_adam(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with Adam optimizer via FFT-based gradients."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    lr0 = 0.5 / n
    eps = 1e-8
    
    # Temperature schedule
    T_start = best_M * 0.1
    T_end = best_M * 0.001
    
    # FFT setup for gradient computation
    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, 1e-10 * best_M)
        
        b = fast_autocorr(a)
        M_curr = b.max()
        if M_curr < 1e-15:
            break
        
        # LSE weights
        w = np.exp((b - M_curr) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT: grad = 2 * (w * a_rev)[n-1:2n-1]
        a_rev = a[::-1]
        W_fft = rfft(w, fft_size)
        AR_fft = rfft(a_rev, fft_size)
        conv_grad = irfft(W_fft * AR_fft, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient to sum-preserving subspace
        grad -= np.mean(grad)
        
        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        lr = lr0 / (1 + progress * 5)  # decay learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score_new = fast_autocorr(a_new).max()
        
        if score_new < best_M:
            a = a_new
            best_M = score_new
            best_a = a.copy()
        elif score_new < best_M * 1.01:
            # Accept small worsening with probability
            if np.random.random() < 0.3 * (1 - progress):
                a = a_new
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): M={best_M:.8f}, T={T:.4e}, lr={lr:.4e}")
        
        # Periodic restart
        if steps % 2000 == 0:
            rng = np.random.default_rng(steps)
            n_pert = max(5, n // 50)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in idx:
                a_trial[i] *= rng.uniform(0.5, 1.5)
            a_trial = np.maximum(a_trial, 0)
            if a_trial.sum() > 1e-15:
                a_trial = a_trial * norm / a_trial.sum()
                if fast_autocorr(a_trial).max() < best_M * 1.05:
                    a = a_trial
                    m = np.zeros(n)
                    v = np.zeros(n)
    
    return best_a, best_M

            candidate_dst = [i for i in range(n) if i not in bad_positions and i != src]
            if not candidate_dst:
                candidate_dst = [i for i in range(n) if i != src]
            
            # Sort by value (prefer low-value destinations)
            candidate_dst.sort(key=lambda i: a[i])
            n_try = min(5, len(candidate_dst))
            
            for dst in candidate_dst[:n_try]:
                for frac in [0.1, 0.2, 0.3]:
                    amt = src_val * frac
                    if amt < 1e-20:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_move_score:
                        best_move_score = sc
                        best_move = (src, dst, frac)
        
        if best_move is not None:
            src, dst, frac = best_move
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            
            if sc < best_M:
                a = a_trial
                best_M = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            n_stall += 1
        
        # Random exploration when stalled
        if n_stall > 30:
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-15:
                    continue
                frac = rng.uniform(0.05, 0.3)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_M:
                    a = a_trial
                    best_M = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    break
                elif sc < best_M * 1.002 and rng.random() < 0.1:
                    a = a_trial
                    n_stall = 0
                    break
            else:
                n_stall += 1
        
        # Large perturbation when very stalled
        if n_stall > 100:
            n_pert = max(5, n // 40)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                j = rng.integers(n)
                amt = a_trial[idx] * rng.uniform(0.1, 0.5)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_M * 1.02:
                    a = a_trial
                n_stall = 0
        
        if verbose and n_improve > 0 and n_improve % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): M={best_M:.8f}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def make_initialization(n, method='sine', seed=42):
    """Create various initialization patterns."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    if method == 'sine':
        a = np.sin(np.pi * x) ** 1.5
    elif method == 'triangular':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif method == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif method == 'flat_top':
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.3)
    elif method == 'random_smooth':
        a = np.ones(n) + 0.3 * np.cumsum(rng.normal(0, 1, n))
        a = np.maximum(a, 0.1)
    elif method == 'bump':
        a = np.exp(-8.0 * (x - 0.3) ** 2) + np.exp(-8.0 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines multiple optimization strategies:
    - Multi-length LP Nesterov
    - Mirror descent
    - LSE Adam gradient
    - Peak shaving
    - Length fine-tuning
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize if no parent
    if best_seq is None:
        best_seq = make_initialization(n_base, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    # === Phase 1: Quick multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-length quick exploration (budget={remaining:.0f}s) ===")
    
    length_candidates = sorted(set([n_base, n_base - 100, n_base + 100, n_base - 200, n_base + 200]))
    length_candidates = [n for n in length_candidates if 500 <= n <= 5000]
    
    quick_time_per = min(60.0, remaining * 0.15 / max(1, len(length_candidates)))
    
    for n_try in length_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        if n_try == n_base:
            init = best_seq.copy()
        else:
            init = interpolate_sequence(best_seq, n_try)
        
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = nesterov_lp(a_norm, min(quick_time_per, remaining * 0.8), solver, verbose=False, max_active=150)
        try_update(a_out, f"explore_n{n_try}")
    
    # === Phase 2: Main LP Nesterov at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True, max_active=200)
        try_update(a_out, f"lp_main_n{n_cur}")
    
    # === Phase 3: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Mirror Descent n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent(a_norm, t3, verbose=True)
        try_update(a_out, f"mirror_n{n_cur}")
    
    # === Phase 4: LSE Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LSE-Adam n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam(a_norm, t4, verbose=True)
        try_update(a_out, f"lseadam_n{n_cur}")
    
    # === Phase 5: LP Nesterov round 2 (refined start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True, max_active=150)
        try_update(a_out, f"lp_r2_n{n_cur}")
    
    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t6, seed=7777, verbose=True)
        try_update(a_out, f"peak_n{n_cur}")
    
    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Length fine-tuning (budget={remaining:.0f}s) ===")
    
    for delta in [50, -50, 100, -100]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        n_try = n_cur + delta
        if n_try < 500 or n_try > 5000:
            continue
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        t_delta = min(60.0, remaining * 0.3)
        a_out, _ = nesterov_lp(a_norm, t_delta, solver, verbose=False, max_active=150)
        try_update(a_out, f"len_n{n_try}")
    
    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t8, solver, verbose=True, max_active=100)
        try_update(a_out, f"lp_final_n{n_cur}")
    
    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.20, 120.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t9, seed=31415, verbose=True)
        try_update(a_out, f"peak_final_n{n_cur}")
    
    # === Phase 10: Final LSE-Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final LSE-Adam n={n_cur} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam(a_norm, remaining * 0.5, verbose=True)
        try_update(a_out, f"lse_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
