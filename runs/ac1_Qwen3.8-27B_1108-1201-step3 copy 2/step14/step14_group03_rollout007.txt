def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        end = min(size, j + n)
        T[j:end, j] = a[:end - j]
    return T


def lp_best_response(a, M):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

        return float(sc)
    except Exception:

def normalize_sum(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return a
    return a * target_sum / s


def nesterov_lp_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:

            s_avg = avg_check.sum()
            if s_avg > 1e-15:

                if score_avg < best_score:

            elapsed = time.monotonic() - t0

def precompute_effects(a):
    n = len(a)
    size = 2 * n - 1
    effects = np.zeros((n, size))
    for i in range(n):
        fwd_start = i
        fwd_end = min(size, i + n)
        effects[i, fwd_start:fwd_end] += a[:fwd_end - fwd_start]
        bwd_start = max(0, i - n + 1)
        bwd_end = min(size, i + 1)
        idx = np.arange(bwd_start, bwd_end)
        src = i - idx
        valid = (src >= 0) & (src < n)
        effects[i, idx[valid]] += a[src[valid]]
    return effects


def pair_cd_enhanced(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    effects = precompute_effects(a)

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = 0

        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            n_tries = 0
            for offset in range(1, n):
                if n_tries >= 30:
                    break
                for sign in [1, -1]:
                    j = (i + sign * offset) % n
                    if j == i:
                        continue
                    n_tries += 1

                    delta = effects[j] - effects[i]

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 25
                    d_max = a[i] * 0.95
                    if d_max < 1e-15:
                        break
                    deltas = np.linspace(0.0, d_max, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2.0 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2.0 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-14:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved += 1
                            no_improve = 0
                            effects = precompute_effects(a)
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if pass_improved == 0:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_global(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    effects = precompute_effects(a)

    n_iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        n_improved = 0

        n_pairs = min(500, n * (n - 1) // 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < norm * 1e-12:
                continue

            delta = effects[j] - effects[i]
            best_d = 0.0
            best_val = conv.max()
            d_max = a[i] * 0.9
            if d_max < 1e-15:
                continue
            n_samples = 20
            deltas = np.linspace(0.0, d_max, n_samples)
            for d in deltas:
                if d <= 0:
                    continue
                new_conv = conv + d * delta
                if 2 * i < size:
                    new_conv[2 * i] = conv[2 * i] - 2.0 * a[i] * d + d * d
                if 2 * j < size:
                    new_conv[2 * j] = conv[2 * j] + 2.0 * a[j] * d + d * d
                new_max = new_conv.max()
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_d = d

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    n_improved += 1
                    no_improve = 0
                    effects = precompute_effects(a)
                else:
                    a[j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if n_improved == 0:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-global iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_iters = 0
    for step_size_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
        if time.monotonic() - t0 > time_limit:
            break
        base_step = norm * step_size_factor / np.sqrt(n)
        n_steps = max(50, int((time_limit - (time.monotonic() - t0)) * 10))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            k_star = conv.argmax()

            grad = np.zeros(n)
            for i in range(n):
                ki = k_star - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * a[ki]
            if k_star == 2 * i and 0 <= k_star // 2 < n:
                pass

            a_new = a - base_step / (1.0 + 0.01 * it) * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new * norm / s
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
            n_iters += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad factor={step_size_factor} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on parent (main workhorse, 50% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    if t1 > 30:
        print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Pair CD local (20% time)
    remaining = time_budget - (time.monotonic() - t_global)

    t2 = remaining * 0.20
    if t2 > 15:
        print(f"\n=== Phase 2: Pair CD local n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_enhanced(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd1_n{n_best}")

    # Phase 3: Pair CD global (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    if t3 > 15:
        print(f"\n=== Phase 3: Pair CD global n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_global(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcdg_n{n_best}")

    # Phase 4: Second LP Nesterov round (15% time)

    if t4 > 20:

        a_out, _ = nesterov_lp_optimize(a_norm, t4, verbose=True)

    n_best = len(best_seq)
    t5 = remaining * 0.15
    if t5 > 10:
        print(f"\n=== Phase 5: Final pair CD n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_enhanced(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 6: Subgradient polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
