```python
def lp_solve_adaptive(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b = np.ones(n) / n  # initial feasible point
    
    # Start with a subset of constraints
    conv_a = fast_conv_self(a)
    # Sort constraints by tightness
    tightness = 1.0 - conv_a / M  # 0 means tight, 1 means loose
    # Start with the tightest 100 constraints
    n_init = max(50, size // 10)
    active_idx = np.argsort(tightness)[:n_init]
    
    for iteration in range(20):
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T[active_idx, :] @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b = np.maximum(b_var.value, 0)
        
        # Check all constraints
        viol = T @ b - M
        worst = viol.max()
        if worst <= 1e-6 * M:
            return b  # Feasible
        
        # Add violated constraints
        violated = np.where(viol > 1e-4 * M)[0]
        active_idx = np.unique(np.concatenate([active_idx, violated]))
        if len(active_idx) >= size:
            # All constraints active, solve full problem
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(b_var.value, 0)
            return None
    
    return b
```

```python
def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve_adaptive(a, M, solver, max_rounds=30):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0, using constraint generation."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    # Start with the tightest constraints
    conv_a = fast_conv_self(a)
    tightness = 1.0 - conv_a / max(M, 1e-15)
    n_init = max(100, min(size, size // 8))
    active_idx = np.argsort(tightness)[:n_init]
    
    b_sol = None
    for round_i in range(max_rounds):
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T[active_idx, :] @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_sol = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        
        # Check all constraints
        viol = T @ b_sol - M
        worst = viol.max()
        if worst <= 1e-5 * max(M, 1.0):
            return b_sol
        
        # Add violated constraints
        violated = np.where(viol > 1e-6 * max(M, 1.0))[0]
        active_idx = np.unique(np.concatenate([active_idx, violated]))
        
        if len(active_idx) >= size:
            # Solve with all constraints
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                return b_sol
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            return b_sol
    
    return b_sol


def lse_gradient_descent(a0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    temperatures = [30.0, 10.0, 4.0, 1.5, 0.5]
    for T_val in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(10, min(100, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = np.minimum((conv - max_c) / T_val, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)
            
            # Backtracking
            step = norm / (T_val * np.sqrt(n)) * 0.5
            for trial in range(8):
                a_try = a - step * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step *= 0.5
                    continue
                a_try = a_try * norm / s_try
                if fast_conv_self(a_try).max() <= max_c + 0.05 * T_val:
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a * norm / s
            
            cur_max = fast_conv_self(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE T={T_val}: best_max={best_max:.10f}")
    
    return best_a, best_max


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    # Track best sum(b) for Polyak step
    best_gain = 0.0
    initial_max = best_max
    
    max_iter = 50000
    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        b = lp_solve_adaptive(y, M, solver, max_rounds=20)
        if b is None:
            b = lp_solve_adaptive(y, M, solver, max_rounds=40)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        sc_y = fast_conv_self(y).max()
        if sc_y < best_max:
            best_max = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv_self(a).max()
        if sc_a < best_max:
            best_max = sc_a
            best_a = a.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best_max={best_max:.10f}")
        
        # Early stopping if converged
        if t > 100 and best_max < initial_max * (1 - 0.999):
            break
    
    return best_a, best_max


def spectral_smooth(a0, time_limit, rng, verbose=False):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Target: make |FFT(a)| as flat as possible
    # Use iteratively reweighted approach
    lr = 0.1
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        
        # We want to reduce peaks in the spectrum
        # Gradient of max |A|^2 w.r.t. a: 
        # d/d a_i |A(f)|^2 = 2 * Re(exp(-2*pi*i*f*k/n) * A(f))
        # For the max, we focus on the frequency bin with largest |A|
        peak_freq = np.argmax(mag[1:] + 1e-10) + 1  # exclude DC
        if peak_freq >= fft_size // 2:
            peak_freq = np.argmin(mag[1:fft_size//2] + 1e-10) + 1
        
        # Compute gradient at peak frequency
        k = peak_freq
        phase = np.exp(-2j * np.pi * k * np.arange(n) / fft_size)
        A_peak = A[k]
        grad = 2.0 * np.real(A_peak * phase)
        
        # Also consider the negative frequency
        k_neg = fft_size - k
        if k_neg < fft_size:
            A_neg = A[k_neg]
            phase_neg = np.exp(-2j * np.pi * k_neg * np.arange(n) / fft_size)
            grad += 2.0 * np.real(A_neg * phase_neg)
        
        step = lr / (np.linalg.norm(grad) + 1e-10)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new
        
        cur_max = fast_conv_self(a_new).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_a, best_max


def pair_cd_optimize(a0, time_limit, rng, verbose=False):
    """Gradient-informed pair coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a * norm / s
    best_a = a.copy()
    best_max = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_imp = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        # Focus on peaks
        peak_threshold = cur_max * 0.9
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Candidate j's
            j_cands = []
            for _ in range(5):
                j_cands.append(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_cands.append((kl - i) % n)
                j_cands.append((kl + i) % n)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_cands:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal transfer
                n_samples = 20
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max - 1e-12:
                        best_max = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_imp += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve >= 4:
                n_pert = max(10, n // 40)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and total_imp % 200 == 0 and total_imp > 0:
            print(f"    CDs={total_imp}: best_max={best_max:.10f}")
    
    return best_a, best_max


def structured_init(n, rng):

    elif choice == 1:

    elif choice == 3:

    elif choice == 4:

        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8

    elif choice == 8:
        # Two narrow peaks
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        # Random smooth
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)

        a = np.ones(n) * 0.5
        for f, p, amp in zip(freqs, phases, amps):

    a = np.maximum(a, 0.0)
    s = a.sum()

    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Total budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try loading parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # Phase 1: Quick LSE (10% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_gradient_descent(best_seq, t1)
        try_update(a_out, "lse1")
    
    # Phase 2: LP Nesterov (main, 35% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")
    
    # Phase 3: Pair CD (20% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Pair CD (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng)
        try_update(a_out, "cd1")
    
    # Phase 4: Spectral smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 4: Spectral (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = spectral_smooth(best_seq, t4, rng)
        try_update(a_out, "spec1")
    
    # Phase 5: LP Nesterov round 2 (15% budget)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP Round 2 (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, "lp2")
    
    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.20, 150.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t6, 100.0), solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out2, _ = pair_cd_optimize(best_seq, min(remaining * 0.6, 60.0), rng, verbose=False)
            try_update(a_out2, "cd_long")
    
    # Phase 7: Final CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.50, 120.0)
    print(f"\n=== Phase 7: Final CD (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t7, rng)
        try_update(a_out, "cd_final")
    
    # Phase 8: Quick LSE + CD alternating
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.6, 80.0)
        print(f"\n=== Phase 8: LSE+CD (budget={t8:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, min(t8 * 0.5, 40.0))
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 50.0), rng)
            try_update(a_out, "cd_final2")
    
    # Final
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
```
