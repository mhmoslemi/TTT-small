1. Nesterov accelerated projected subgradient method (primary)
2. LP-based Frank-Wolfe with Nesterov acceleration (secondary)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full linear convolution a * a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v, target_sum=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    # Algorithm from Duchi et al. (2012), "Efficient Projections onto the l1-Ball"
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    w = np.arange(1, n + 1)
    # Find largest rho such that v_sorted[rho] - (cumsum[rho] - target_sum) / rho > 0
    cond = (v_sorted * w - (cumsum - target_sum)) > 0
    rho = cond.sum()
    if rho == 0:
        return np.full(n, target_sum / n)
    theta = (cumsum[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k):
    """Compute exact gradient of conv(a,a)[k] w.r.t. a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    k = int(k)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min > j_max:
        return grad
    valid_j = np.arange(j_min, j_max + 1)
    valid_idx = k - valid_j
    grad[valid_j] = 2.0 * a[valid_idx]
    return grad

def compute_soft_subgrad(a, top_k=15, beta=4.0):
    """Compute soft subgradient using top-k lags with power weighting."""
    n = len(a)
    ac = fast_conv(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = np.maximum(ac[top_idx], 0.0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_min = max(0, k_int - n + 1)
        j_max = min(n - 1, k_int)
        if j_min > j_max:
            continue
        valid_j = np.arange(j_min, j_max + 1)
        valid_idx = k_int - valid_j
        grad[valid_j] += wi * 2.0 * a[valid_idx]
    return grad, ac.max()

    n = len(a0)

    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov lookahead point
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall_count > max_stall:
            break
        
        # Compute soft subgradient at y
        g, m_y = compute_soft_subgrad(y, top_k=top_k, beta=beta)
        
        # Step size: use Lipschitz-like bound
        # f is approximately L-Lipschitz where L ~ 2*max(y)
        # For step: c / t (Nesterov) or c / sqrt(t) (vanilla)
        lr = 0.5 / t
        # Also scale by 1/||g|| to normalize the step direction
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            lr = lr / (g_norm / n)  # Scale so typical step is O(1/sqrt(n))
        
        # Projected step
        y_cand = y - lr * g
        a_new = simplex_project(y_cand, 1.0)
        
        # Check improvement
        m_new = fast_conv(a_new).max()
        
        # Nesterov update
        if t == 1:
            x_new = a_new
        else:

            x_new = a_new + beta_nest * (a_new - x_prev)
            x_new = simplex_project(x_new, 1.0)
            m_xnew = fast_conv(x_new).max()
            if m_xnew < m_new:
                a_new = x_new
                m_new = m_xnew
        
        x_prev = a_new
        y = a_new
        
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NPS t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = simplex_project(a0.astype(np.float64), 1.0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Solve LP: max sum(b) s.t. T@b <= M, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Nesterov Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass between pairs of positions."""
    n = len(a0)
    a = simplex_project(a0.astype(np.float64), 1.0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find the position in this lag's support that contributes most
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Try moving a small amount from src to several candidates
            # Score each candidate by how much it increases the max convolution
            candidates = np.random.choice(n, size=min(n, 50), replace=False)
            best_dst = -1
            best_score = float('inf')
            
            for dst in candidates:
                if dst == src:
                    continue
                # Estimate: adding to dst increases conv[k2] by 2*a[|k2-dst|] for nearby k2
                # Removing from src decreases conv[k2] by 2*a[|k2-src|]
                # Net effect on max: approximate by effect on top lags
                sens = 0.0
                for k2 in top_k:
                    k2i = int(k2)
                    add = 0.0
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        add += a[d1]
                    d2 = dst - k2i
                    if 0 <= d2 < n:
                        add += a[d2]
                    rem = 0.0
                    d1 = k2i - src
                    if 0 <= d1 < n:
                        rem += a[d1]
                    d2 = src - k2i
                    if 0 <= d2 < n:
                        rem += a[d2]
                    sens += (add - rem)
                
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1 and non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Compute full linear convolution a * a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v, target_sum=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    w = np.arange(1, n + 1)
    cond = (v_sorted * w - (cumsum - target_sum)) > 0
    rho = cond.sum()
    if rho == 0:
        return np.full(n, target_sum / n)
    theta = (cumsum[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)


def compute_soft_subgrad(a, top_k=15, beta=4.0):
    """Compute soft subgradient using top-k lags with power weighting.
    Returns (gradient, max_conv_value)."""
    n = len(a)
    ac = fast_conv(a)
    M_val = ac.max()
    k_actual = min(top_k, len(ac))
    top_idx = np.argpartition(ac, -k_actual)[-k_actual:]
    top_vals = ac[top_idx]
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_actual) / k_actual
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_min = max(0, k_int - n + 1)
        j_max = min(n - 1, k_int)
        if j_min > j_max:
            continue
        valid_j = np.arange(j_min, j_max + 1)
        valid_idx = k_int - valid_j
        grad[valid_j] += wi * 2.0 * a[valid_idx]
    return grad, M_val


def exact_subgrad_at_lags(a, lags):
    """Compute exact subgradient: average of gradients at specified lags."""
    n = len(a)
    grad = np.zeros(n)
    cnt = 0
    for k in lags:
        k_int = int(k)
        j_min = max(0, k_int - n + 1)
        j_max = min(n - 1, k_int)
        if j_min > j_max:
            continue
        valid_j = np.arange(j_min, j_max + 1)
        valid_idx = k_int - valid_j
        grad[valid_j] += 2.0 * a[valid_idx]
        cnt += 1
    if cnt > 0:
        grad /= cnt
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = simplex_project(a0.astype(np.float64), 1.0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m
