9. **New approach: Smooth minimization with FISTA**:
   
   Let `f(a) = max_k conv(a,a)[k]` and `g(a) = (sum(a))^2 / (2n)` (the normalization constraint).
   
   Actually, since the objective is homogeneous, let's fix `sum(a) = 1` and minimize `2n * max_k conv(a,a)[k]`.
   
   Or better: fix `sum(a) = sqrt(2n)` and minimize `max_k conv(a,a)[k]`.
   
   The function `f(a) = max_k 2*a[k*-j]` (as a subgradient) is convex but non-smooth. We can:
   
   a. Use FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with a smooth approximation.
   b. Use the log-sum-exp smoothing: `f_T(a) = T * log(sum_k exp(conv(a,a)[k]/T))` which is smooth and approximates `max_k conv(a,a)[k]` as `T -> 0`.
   
   The gradient of `f_T(a)` is: `f_T'(a)[j] = sum_k p_k * 2*a[k-j]` where `p_k = exp(conv(a,a)[k]/T) / sum_l exp(conv(a,a)[l]/T)`.
   
   This is a weighted average of the "active" subgradients, weighted by the softmax of the convolution values.
   
   This is what the previous code's `lse_polish` does, but perhaps we can make it much more aggressive as the PRIMARY method rather than just a polish.

def fast_autocorr(a):
    """Compute full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fast(p, a):
    """Compute sum_k p[k+j]*a[k] for j=0..n-1 via FFT.
    Returns grad[j] = sum_{l=0}^{n-1} p[j+l]*a[l] (with p zero-padded)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(np.asarray(p, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = irfft(P * np.conj(A), fft_size)
    return result[:n]


def soft_max_gradient(a, T):
    """Compute gradient of T*log(sum_k exp(conv(a,a)[k]/T)) w.r.t. a.
    
    grad[j] = 2 * sum_k p[k] * a[k-j] where p = softmax(conv/T)
    This equals 2 * cross_corr(p, a)[j] (with appropriate indexing).
    """
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    # Numerical stability: subtract max before exp
    logits = (conv - M) / T
    # Clip for numerical stability
    logits = np.clip(logits, -100, 100)
    exp_logits = np.exp(logits)
    p = exp_logits / exp_logits.sum()
    
    # grad[j] = 2 * sum_{k} p[k] * a[k-j] where 0<=k-j<n
    # = 2 * sum_l p[j+l] * a[l] where l = k-j
    # = 2 * cross_corr(p, a)[j]
    grad = 2.0 * cross_corr_fast(p, a)
    return grad, conv


def project_nonneg_sum(x, S):
    """Project x onto {z >= 0, sum(z) = S} via bisection on threshold."""
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    
    # If all elements are already >= 0 and sum is close to S, just scale
    if x.min() >= 0:
        s = x.sum()
        if abs(s - S) < 1e-12:
            return x.copy()
        if s > 0:
            return x * (S / s)
    
    # Bisection on tau: find tau such that sum(max(x-tau, 0)) = S
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    
    # Check if tau = min(x) gives sum >= S
    if (x - lo).clip(0).sum() <= S:
        return np.maximum(x - lo, 0) * S / max((x - lo).clip(0).sum(), 1e-30)
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = np.maximum(x - mid, 0.0).sum()
        if val > S:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2.0
    result = np.maximum(x - tau, 0.0)
    # Fix any numerical issues
    s = result.sum()
    if s > 0:
        result = result * (S / s)
    else:
        result = np.full(n, S / n)
    return result


def fista_soft(a0, time_limit, T_start=5.0, T_end=0.1, n_anneal=200, verbose=True):
    """FISTA with log-sum-exp smoothing and temperature annealing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s
    
    y = a.copy()
    t_mom = 1.0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    n_iters = 0
    n_anneal_done = 0
    
    # Adaptive step size
    step = 1.0
    
    for outer in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule
        if n_anneal_done < n_anneal:
            progress = min(1.0, (time.monotonic() - t0) / (time_limit * 0.5))
            T = T_start * (T_end / T_start) ** progress
        else:
            # Keep decreasing slowly
            T = max(T_end * 0.5, T * 0.999)
        
        n_anneal_done = min(n_anneal_done + 1, n_anneal)
        
        # Compute gradient at y
        grad, conv_y = soft_max_gradient(y, T)
        
        # FISTA step
        z = y - step * grad
        z = project_nonneg_sum(z, S)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = z + (t_mom - 1.0) / t_new * (z - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 0:
            y_new = y_new * S / s_y
        else:
            y_new = z.copy()
        
        y = y_new
        a = z
        t_mom = t_new
        n_iters += 1
        
        # Check improvement
        score_a = fast_autocorr(a).max()
        if score_a < best_score:

import numpy as np
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    try:
        import cvxpy as cp
        for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fast(p, a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(np.asarray(p, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = irfft(P * np.conj(A), fft_size)
    return result[:n]


def soft_max_gradient(a, T):
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    logits = (conv - M) / T
    logits = np.clip(logits, -700, 700)
    exp_logits = np.exp(logits)
    s = exp_logits.sum()
    if s < 1e-30:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        p = exp_logits / s
    grad = 2.0 * cross_corr_fast(p, a)
    return grad, conv, p


def project_nonneg_sum(x, S):
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    if x.min() >= -1e-15:
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s > 1e-15:
            return x * (S / s)
        return np.full(n, S / n)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val = np.maximum(x - mid, 0.0).sum()
        if val > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(x - tau, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * (S / s)
    else:
        result = np.full(n, S / n)
    return result


def fista_smooth(a0, time_limit, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s

    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    n_iters = 0
    T_curr = 2.0
    step = 0.5

    phase1_end = time_limit * 0.6
    phase2_end = time_limit

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        elapsed_frac = (time.monotonic() - t0) / max(time_limit, 1e-6)
        if elapsed_frac < 0.5:
            T_curr = max(0.05, 2.0 * (0.9997 ** n_iters))
        else:
            T_curr = max(0.02, 0.5 * (0.9999 ** (n_iters - int(0.5 * time_limit / max(step, 0.01))))

        n_iters += 1

        grad, conv_y, p = soft_max_gradient(y, T_curr)

        z = y - step * grad
        z = project_nonneg_sum(z, S)

        conv_z = fast_autocorr(z)
        if conv_z.max() > conv_y.max() * 1.01 + T_curr:
            step *= 0.8
            z = y - step * grad
            z = project_nonneg_sum(z, S)
        elif step < 1.0:
            step = min(step * 1.02, 2.0 / (a.max() + 1e-6))

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = z + (t_mom - 1.0) / t_new * (z - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * S / s_y
        else:
            y_new = z.copy()

        y = y_new
        a = z
        t_mom = t_new

        score_a = conv_z.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={n_iters} T={T_curr:.4f} step={step:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fista_with_restarts(a0, time_limit, n_restarts=3, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()

    t0 = time.monotonic()

    # First run on the given initialization
    time_per = time_limit / (n_restarts + 1)
    a_out, sc = fista_smooth(a0, time_per, verbose=verbose)
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()

    # Subsequent runs with perturbations
    rng = np.random.default_rng(12345)
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        time_per_run = remaining / (n_restarts - r)

        a_pert = best_a.copy()
        noise = rng.normal(0, 0.1 * a_pert.mean(), n)
        a_pert = a_pert + noise
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert * S / max(a_pert.sum(), 1e-15)

        a_out, sc = fista_smooth(a_pert, time_per_run, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    Restart {r+1} improved: {best_score:.8f}")

    return best_a, best_score


def pair_cd_fast(a0, time_limit, window=15, verbose=True):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(42)

    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < S * 1e-12:
                continue

            for offset in range(1, window + 1):
                j = (i + offset) % n
                if j == i or a[i] < S * 1e-12:
                    continue

                base_k_i = np.zeros(size)
                for k in range(max(0, i), min(size, i + n)):
                    base_k_i[k] = -2.0 * a[k - i]

                base_k_j = np.zeros(size)
                for k in range(max(0, j), min(size, j + n)):
                    base_k_j[k] = 2.0 * a[k - j]

                base = base_k_i + base_k_j

                quad_positions = set()
                if 2 * i < size:
                    quad_positions.add(2 * i)
                if i + j < size:
                    quad_positions.add(i + j)
                if 2 * j < size:
                    quad_positions.add(2 * j)

                top_k_idx = np.argsort(conv)[-30:][::-1]

                n_samples = 10
                d_candidates = np.linspace(0, a[i] * 0.95, n_samples)

                best_d = 0.0
                best_new_max = best_score

                for d in d_candidates[1:]:
                    delta = d * base
                    for qp in quad_positions:
                        if qp == 2 * i:
                            delta[qp] += d * d
                        elif qp == i + j:
                            delta[qp] -= 2.0 * d * d
                        elif qp == 2 * j:
                            delta[qp] += d * d

                    new_conv_check = conv[top_k_idx] + delta[top_k_idx]
                    for qp in quad_positions:
                        if qp in top_k_idx:
                            continue
                        new_val = conv[qp] + delta[qp]
                        if new_val > new_conv_check.max():
                            new_conv_check = np.append(new_conv_check, new_val)

                    new_max = new_conv_check.max()
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s_chk = a.sum()
                    if s_chk > 1e-15:
                        a = a * S / s_chk
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        continue

                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and (no_improve_passes >= 1 or int((time.monotonic() - t0)) % 30 < 2):
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a, M, solver):
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 15:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:

        else:

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_new.copy()

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def make_init_cosine(n, S):
    i = np.arange(n)
    a = np.cos(np.pi * i / (2 * n + 2))
    a = np.maximum(a, 0.0)
    a = a * S / max(a.sum(), 1e-15)
    return a


def make_init_triangle(n, S):
    i = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(i - center) / (center + 1)
    a = np.maximum(a, 0.0)
    a = a * S / max(a.sum(), 1e-15)
    return a


def make_init_bell(n, S, sigma_frac=0.4):
    i = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a * S / max(a.sum(), 1e-15)
    return a

            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: FISTA smooth (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: FISTA Smooth n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        S = np.sqrt(2.0 * n_base)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = fista_with_restarts(a_norm, t1, n_restarts=2, verbose=True)
        try_update(a_out, f"fista_n{n_base}")

    # === Phase 2: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        S = np.sqrt(2.0 * n_best)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t2, window=12, verbose=True)
        try_update(a_out, f"pcd_n{n_best}")

    # === Phase 3: FISTA again (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: FISTA round 2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        S = np.sqrt(2.0 * n_best)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = fista_smooth(a_norm, t3, verbose=True)
        try_update(a_out, f"fista2_n{n_best}")

    # === Phase 4: LP Nesterov (if available) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 200.0)
    print(f"\n=== Phase 4: LP Nesterov n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        solver = get_solver()
        if solver is not None:
            n_best = len(best_seq)
            S = np.sqrt(2.0 * n_best)
            a_norm = best_seq * S / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
            try_update(a_out, f"lp_n{n_best}")

    # === Phase 5: Final pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.50
    print(f"\n=== Phase 5: Final PCD n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        S = np.sqrt(2.0 * n_best)
        a_norm = best_seq * S / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t5, window=20, verbose=True)
        try_update(a_out, f"pcd_final_n{n_best}")

    # === Phase 6: Try different initialization + FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 6: Alternative init + FISTA (budget={remaining * 0.6:.0f}s) ===")
        t6 = remaining * 0.6
        S = np.sqrt(2.0 * n_base)

        # Try triangle init
        a_tri = make_init_triangle(n_base, S)
        a_out, _ = fista_smooth(a_tri, t6 * 0.4, verbose=False)
        try_update(a_out, f"fista_tri_n{n_base}")

        # Try bell init
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 20:
            a_bell = make_init_bell(n_base, S, sigma_frac=0.3)
            a_out, _ = fista_smooth(a_bell, min(remaining6 * 0.5, 60.0), verbose=False)
            try_update(a_out, f"fista_bell_n{n_base}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
