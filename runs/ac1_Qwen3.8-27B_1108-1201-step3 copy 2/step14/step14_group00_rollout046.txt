**Greedy best response (non-convex):**
The LP best response maximizes sum(b) subject to (a*b)_k <= M, b >= 0.
A simple greedy approach:
1. Start with b = 0.
2. At each step, find the j that maximizes the "benefit" of increasing b_j.
3. Increase b_j as much as possible without violating any constraint.
4. Repeat.

The "benefit" of increasing b_j by 1 is: it increases sum(b) by 1, but it increases (a*b)_k by a_{k-j}^+ for each k.
So the "cost" is the maximum over k of a_{k-j}^+ / (M - (a*b)_k).

import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_rows(a, indices):
    """Build rows of convolution matrix for given k indices."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(indices), n))
    for i, k in enumerate(indices):
        # (a*b)_k = sum_j a_{k-j} * b_j where 0 <= k-j < n and 0 <= j < n
        # So row k has: row[k, j] = a_{k-j} if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_end > j_start:
            a_indices = k - np.arange(j_start, j_end)
            valid = (a_indices >= 0) & (a_indices < n)
            rows[i, j_start + np.where(valid)[0]] = a[a_indices[valid]]
    return rows

def lp_best_response(a, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_conv_rows(a, active_set)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    else:

    best_score = fast_autocorr(a).max()

    x = a.copy()
    avg_x = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-14:
            break
        
        # Constraint screening
        tol = 0.05 + 0.02 * np.exp(-t / 300.0)
        active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
        if len(active_set) < 10:
            active_set = np.where(conv_y > M_curr * 0.7)[0]
        
        b = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            # Fall back to full constraint set
            active_set = np.arange(2*n-1)
            b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:

        x = x_new

                if sc < best_score:

```python
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def projected_gradient(a0, time_limit, beta=100):
    n = len(a0)
    a = simplex_project(a0.astype(np.float64))
    t0 = time.monotonic()
    t = 1
    eta = 0.01  # Step size
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        m = conv.max()
        softmax = np.exp(beta * (conv - m)) / np.sum(np.exp(beta * (conv - m)))
        
        # Gradient of (1/beta) * log(sum exp(beta * conv_k)) with respect to a_i
        # = sum_k softmax_k * d(conv_k)/da_i
        # d(conv_k)/da_i = a_{k-i}^+ + a_{i-k}^+ (counting both if they exist)
        
        grad = np.zeros(n)

        # add softmax_k * (a shifted by k + a shifted by -k) to grad
        for k in range(2*n-1):

                continue

            # For fixed k, a_{k-i} as a function of i is: a[k-i] for i in [max(0,k-n+1), min(n,k+1))

def build_conv_rows(a, indices):

    n = len(a)

            valid = (a_idx >= 0) & (a_idx < n)
