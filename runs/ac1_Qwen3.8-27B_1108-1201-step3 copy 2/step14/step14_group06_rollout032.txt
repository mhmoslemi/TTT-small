The Welch bound says that for n sequences of length m in C^m, the maximum aperiodic autocorrelation is at least `(n-1)/m`. But our problem is different.

Actually, the problem is related to "Golay sequences" and " Barker sequences". A Barker sequence of length n has the property that its aperiodic autocorrelation is at most 1 for all non-zero shifts. If such sequences existed for all n, we could achieve `max(conv(a,a)) = n` (for the zero lag) or less. But Barker sequences are only known for specific lengths.

For our problem, we're looking for non-negative sequences (not ±1 sequences), so the problem is different. The best known non-negative sequences with flat autocorrelation are related to "Sidon sets" and "Golay complementary sequences" adapted to the non-negative case.

For the specific objective `2n * max(conv(a,a)) / (sum(a))^2`, the theoretical lower bound is 1 (achieved by "perfect" sequences where all autocorrelation coefficients are equal). The best known constructions achieve about 1.4-1.5 for large n.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on LSE-smoothed objective
2. Multiplicative weight initialization for fast warm-start
3. Multi-scale optimization (short -> medium -> long)
4. Peak-aware coordinate descent for final polish
5. LP best-response with Nesterov for refinement
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_sum_nonneg(x, c):
    """Project x onto {a >= 0 : sum(a) = c}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    # Binary search for tau such that sum(max(0, x - tau)) = c
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) / 2
        s = np.maximum(x - tau, 0).sum()
        if s > c:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.maximum(x - tau, 0)

def lse_grad(a, n, T):
    """Compute gradient of LSE-smoothed max autocorrelation."""
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T * (b - max_b)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: sum_k w_k * 2 * a_{k-j} (with bounds)
    # This is 2 * (w * a) (cross-correlation)
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    return 2.0 * cross

def lse_value(a, n, T):
    """Compute LSE-smoothed max autocorrelation value."""
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 0.0
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    return max_b + np.log(exp_vals.sum()) / T

def fista_lse(a0, n, T_schedule, time_limit, verbose=True):
    """FISTA on LSE-smoothed objective with projection onto {a>=0: sum(a)=c}."""
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    x = a.copy()
    y = a.copy()
    best_x = a.copy()
    best_val = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    n_iter = 0
    
    for T_idx, T in enumerate(T_schedule):
        stage_start = time.monotonic()
        stage_limit = min(120.0, time_limit - (time.monotonic() - t0))
        if stage_limit < 5:
            break
        
        # Estimate Lipschitz constant
        # L ≈ 4 * T * max(a)^2 (conservative)
        L_est = 4.0 * T * np.max(a)**2 + 1e-10
        step = 1.0 / L_est
        
        t_fista = 1.0
        y = x.copy()
        prev_x = x.copy()
        
        while time.monotonic() - stage_start < stage_limit:
            # Compute gradient at y
            gy = lse_grad(y, n, T)
            
            # FISTA update
            x_new = y - step * gy
            # Project onto {a>=0: sum(a)=c}
            x_new = project_sum_nonneg(x_new, c)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista**2))
            y = x_new + ((t_fista - 1.0) / t_new) * (x_new - prev_x)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x_new.copy()
            else:
                y = y * c / sy
            
            prev_x = x
            x = x_new
            t_fista = t_new
            n_iter += 1
            
            # Check best
            val = fast_autocorr(x).max()
            if val < best_val:
                best_val = val
                best_x = x.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA T={T:.1f}: val={best_val:.8f} (iter={n_iter}, {elapsed:.0f}s)")
        
        # Update x for next stage
        if fast_autocorr(x).max() < fast_autocorr(best_x).max():
            best_x = x.copy()
        x = best_x.copy()
    
    return best_x, best_val

def mwu_refine(a0, n, time_limit, verbose=True):
    """Multiplicative weight refinement."""
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.1  # learning rate
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        M = b.max()
        if M < 1e-12:
            break
        
        # Find peak position
        k_star = int(np.argmax(b))
        
        # Update: elements that contribute to the peak are reduced
        # a_j contributes a_j * a_{k_star - j} to the peak
        # Reduce a_j proportionally to its contribution
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib[j] = a[j] * a[k2]
        
        # Multiplicative update
        norm_contrib = contrib / (contrib.max() + 1e-15)
        a_new = a * (1.0 - eta * norm_contrib)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new * c / a_new.sum()
        
        val = fast_autocorr(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            eta = eta * 0.9
            if eta < 1e-4:
                break
    
    return best_a, best_val

def peak_coord_descent(a0, n, time_limit, seed=42, verbose=True):

            if 0 <= k2 < n:
                contrib[j] = a[j] * a[k2]

        # Also consider other high peaks

                    val = fast_autocorr(a_trial).max()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.1 * np.cos(3.0 * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 5:
        # Beta(2,2) distribution
        x = np.linspace(0.0, 1.0, n)
        a = 6.0 * x * (1 - x)
    elif choice == 6:
        # Beta(1,1) = uniform with edges boosted
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    else:
        # Two Gaussian peaks
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + 0.8 * np.exp(-20 * (x - 0.75) ** 2)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated Frank-Wolfe with LP best-response."""
    import cvxpy as cp
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break
        
        # Try tight M
        eps = min(0.02, 0.002 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        
        # Build constraint matrix
        size = 2 * n - 1
        T = np.zeros((size, n))
        for i in range(n):
            start = i
            end = min(i + n, size)
            idx = np.arange(start, end)
            T[idx, i] = a[idx - i]
        
        # LP best response
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_used]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraints = [T @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            break
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g = b_val * c / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / sy
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * c / sa
        
        y = y_new
        a = a_new
        
        # Update best
        val_y = fast_autocorr(y).max()
        if val_y < best_val:
            best_val = val_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        val_a = fast_autocorr(a).max()
        if val_a < best_val:
            best_val = val_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        # Stall detection
        if time.monotonic() - last_improve > 45.0:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
        
        if verbose and n_iters % 50 == 0:
            print(f"    LP-Nest iter={n_iters}: best={best_val:.8f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_val

def get_solver():

        s = np.maximum(x - tau, 0.0).sum()
        if s > c:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.maximum(x - tau, 0.0)


def lse_obj_and_grad(a, n, T):
    """Compute LSE-smoothed max autocorrelation value and gradient.
    
    f(a) = (1/T) * log(sum_k exp(T * b_k - T * max_b)) + max_b
    grad_j = 2 * sum_k w_k * a_{k-j} (cross-correlation of weights with a)
    """
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return max_b, np.zeros(n)
    
    shifted = T * (b - max_b)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    obj_val = max_b + np.log(exp_vals.sum()) / T
    
    # Gradient = 2 * (w cross-correlated with a)
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    grad = 2.0 * cross
    return obj_val, grad


def fista_lse(a0, n, T_schedule, time_limit, verbose=True):
    """FISTA on LSE-smoothed objective. Optimizes over {a>=0: sum(a)=1}.
    
    Uses multi-stage temperature annealing. Returns best a (sum=1) and its true max autocorr.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_x = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    total_iters = 0
    
    for T_idx, T in enumerate(T_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break
        stage_limit = min(remaining / max(1, len(T_schedule) - T_idx), 150.0)
        if stage_limit < 3.0:
            break
        
        # FISTA initialization for this stage
        x = best_x.copy()
        y = x.copy()
        prev_x = x.copy()
        t_fista = 1.0
        
        # Step size: L <= 4*T*max(a)^2 (conservative for sum(a)=1, max(a)<=1)
        # Use L = 4*T*max(a)^2 * safety_factor
        L_est = 4.0 * T * np.max(x) ** 2 + 1e-8
        step = 1.0 / L_est
        # Cap step to avoid too-large initial step
        step = min(step, 0.5)
        
        stage_start = time.monotonic()
        stag_iter = 0
        last_stage_improve = stage_start
        
        while time.monotonic() - stage_start < stage_limit:
            # Gradient at y
            gy = np.zeros(n)
            b_y = fast_autocorr(y)
            max_by = b_y.max()
            if max_by > 1e-15:
                shifted = T * (b_y - max_by)
                shifted = np.clip(shifted, -700.0, 700.0)
                exp_v = np.exp(shifted)
                w = exp_v / exp_v.sum()
                
                fft_sz = 1 << (2 * n - 1).bit_length()
                w_f = np.zeros(fft_sz)
                w_f[:2 * n - 1] = w
                a_f = np.zeros(fft_sz)
                a_f[:n] = y
                cross = irfft(rfft(w_f, fft_sz) * np.conj(rfft(a_f, fft_sz)), fft_sz)[:n]
                gy = 2.0 * cross
            
            # FISTA update
            x_new = y - step * gy
            x_new = project_sum_nonneg(x_new, 1.0)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
            y = x_new + ((t_fista - 1.0) / t_new) * (x_new - prev_x)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x_new.copy()
            else:
                y = y / sy
            
            prev_x = x
            x = x_new
            t_fista = t_new
            stag_iter += 1
            total_iters += 1
            
            # Track best (check every 5 iterations to save time)
            if stag_iter % 5 == 0 or (time.monotonic() - stage_start) > stage_limit - 1.0:
                val = fast_autocorr(x).max()
                if val < best_val:
                    best_val = val
                    best_x = x.copy()
                    last_stage_improve = time.monotonic()
        
        if fast_autocorr(x).max() < best_val:
            best_val = fast_autocorr(x).max()
            best_x = x.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA T={T:.1f}: max_autocorr={best_val:.8f} (iters={total_iters}, {elapsed:.0f}s)")
        
        # Update step size for next stage
        x = best_x.copy()
    
    return best_x, best_val


def mwu_refine(a0, n, time_limit, verbose=True):
    """Multiplicative weight refinement: iteratively reduce mass at peak-contributing positions."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.05
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        M = b.max()
        if M < 1e-12:
            break
        
        # Find all significant peaks (within 5% of max)
        threshold = M * 0.95
        peak_indices = np.where(b >= threshold)[0]
        
        if len(peak_indices) == 0:
            break
        
        # Compute contribution of each position to the peaks
        contrib = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib[j] += a[j] * a[k - j]
        
        if contrib.max() < 1e-15:
            break
        
        # Multiplicative reduction of high-contributing positions
        norm_c = contrib / contrib.max()
        a_new = a * (1.0 - eta * norm_c)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        
        val = fast_autocorr(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.95
            if eta < 1e-5:
                break
    
    if verbose:
        print(f"    MWU: best={best_val:.8f} ({time.monotonic()-t0:.0f}s)")
    return best_a, best_val

        a = a / s

        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Top contributors to the peak

        je = min(n - 1, k_peak)
        contrib = a[js:je+1] * a[k_peak - (js:je+1)]
        top_local = np.argsort(contrib)[::-1][:15]
        top_global = js + top_local
        
        # Low-value positions (good targets for receiving mass)
        low_idx = np.argsort(a)[:25]
        
        improved = False
