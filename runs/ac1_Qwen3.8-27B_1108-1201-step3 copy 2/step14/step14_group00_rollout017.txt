For the coordinate descent, the key insight is:
- At each step, find the (i, j, delta) that most reduces the peak autocorrelation
- This is equivalent to: find the direction in the "mass transfer" space that most reduces max_k c_k

A good heuristic:
1. Find the top-K peak lags: k_1, k_2, ..., k_K
2. For each peak lag k_m, compute the "heat" vector h^{(m)}_i = a_{k_m - i}
3. The combined "pressure" on coordinate i is: P_i = sum_m lambda_m * h^{(m)}_i
   where lambda_m are the dual weights (or just use equal weights for simplicity)
4. Move mass from high-P coordinates to low-P coordinates
5. Use bisection to find the optimal amount

Actually, for a single coordinate move (delta from i to j), the change in the peak is:
Δmax ≈ max_k 2*delta*(a_{k-j} - a_{k-i})

To minimize this (most negative), we want:
- i such that a_{k*-i} is large for the peak k*
- j such that a_{k*-j} is small for the peak k*

So: i = argmax_i a_{k*-i} * [a_i > 0] and j = argmin_j a_{k*-j}

But we also need to consider that moving mass changes the autocorrelation at ALL lags, not just the peak. The bisection line search handles this.

    return a * norm_val / s

    Coordinate descent: move mass between pairs (i,j) to reduce peak autocorrelation.
    Uses bisection for exact line search.

    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_fn(a0, n)

    best_score = conv.max()
    best_a = a.copy()

    size = 2 * n - 1

            break

        k_star = int(np.argmax(conv))

                for k in range(i, min(size, i + n)):

                lo_d = 0.0
                hi_d = a[i]

                # At delta=0: max is conv.max() = best_score (approximately)
                # At delta=a[i]: max is (conv + 2*a[i]*d_vec).max()

                        new_conv[ki] += opt_d * opt_d
                        # Wait, I need to be more careful.
                        # c_{2i} = sum_j a_j * a_{2i-j}

                        # After changing a_i to a_i - d: (a_i-d)^2 = a_i^2 - 2*a_i*d + d^2
                        # So the change is -2*a_i*d + d^2
                        # The linear term gives: 2*d*(a_{2i-i} - a_{2i-i})... hmm, this is getting confusing.
                        # Let me just do the exact computation.
                        pass
                    kj = 2 * j
                    if kj < size:
                        pass  # Similar correction
                    
                    new_max = new_conv.max()
                    if new_max < best_new_max - 1e-12:
                        best_new_max = new_max
                        best_j = j
                        best_delta = opt_d
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                a = a * norm_val / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True

            # Try a different strategy: random perturb + re-optimize
            if p > 50:
                break
    
    return best_a, best_score

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_autocorr(a):
    """FFT-based autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_fn(a, n):
    """Normalize to sum = sqrt(2n), clip negatives."""
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm_val / n
    return a * norm_val / s

def lp_best_response_fast(a, M, active_indices):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_indices, b >= 0."""
    n = len(a)
    m = len(active_indices)
    # Build constraint matrix
    A_ub = np.zeros((m, n))
    b_y = a  # shorthand
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A_ub[idx, j] = b_y[k - j]
    
    c_obj = -np.ones(n)
    b_ub = M * np.ones(m)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'disp': False, 'presolve': True})
    
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None
