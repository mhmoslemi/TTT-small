def proj_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)

def softmax_weights(b, temperature):
    """Compute softmax weights over b with given temperature."""
    b_shifted = b - b.max()
    w = np.exp(b_shifted / temperature)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / s
    return w

    n = len(a)

    for j in range(n):

def lp_best_response(y, M, solver, max_active=None):
    """Solve LP: max sum(b) s.t. T_y @ b <= M, b >= 0."""
    n = len(y)
    size = 2 * n - 1
    T = build_conv_matrix(y)
    b_var = cp.Variable(n, nonneg=True)

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    for _ in range(4):
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())

        if max_active is not None and n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_k = min(max_active, n_active)
            top_idx = active_idx[np.argsort(active_vals)[::-1][:top_k]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True
            # Also add a few near-active constraints for robustness
            near_active = (circ >= M * 0.95) & (~active)
            n_na = int(near_active.sum())
            if n_na > 0:
                na_idx = np.where(near_active)[0]
                na_vals = circ[na_idx]
                extra_k = min(30, n_na)
                extra_idx = na_idx[np.argsort(na_vals)[::-1][:extra_k]]
                active[extra_idx] = True

        if int(active.sum()) < 10:
            break

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                          [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            break

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_active=200, gamma_init=1.8):
    """Fessler-Parks accelerated LP best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    M_hist = []
    gamma = gamma_init

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, max_active=max_active)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            osc = np.sum(np.abs(np.diff(recent))) / (30 * max(recent[0], 1e-15))
            if osc > 0.08:
                gamma = max(1.0, gamma - 0.15)
            elif osc < 0.005:
                gamma = min(2.0, gamma + 0.1)

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, g={gamma:.2f}, stall={stall}")

        if stall > 200:
            rng = np.random.default_rng(n_iters * 17 + 3)
            perturb_scale = 0.05 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = gamma_init
            M_hist = []

    return best_a, best_score


def accelerated_proximal_gradient(a0, time_limit, verbose=True):
    """Nesterov-accelerated proximal gradient for min max_k (Ta)_k on simplex."""
    n = len(a0)
    a = proj_simplex(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 1.0 / (n * 2.0)
    L_estimate = 2.0 * fast_autocorr(a).max()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        b_y = fast_autocorr(y)
        max_b = b_y.max()
        if max_b < 1e-14:
            break

        # Multi-peak softmax subgradient
        b_range = max_b - b_y.min() + 1e-12
        T_temp = max(0.02, 0.1 * b_range * (1.0 - 0.5 * progress))
        w = softmax_weights(b_y, T_temp)

        # Gradient: T_y^T @ w
        grad = compute_TT_v(w, y)

        # Adaptive step size
        L_curr = 2.0 * max_b * (1.0 + 0.1 * np.max(w) * (2 * n - 1))
        L_estimate = 0.9 * L_estimate + 0.1 * L_curr
        eta = 1.0 / (L_estimate * t)
        eta = min(eta, 2.0 / n)
        eta = max(eta, 1e-8 / n)

        # Proximal gradient step (project onto simplex)
        a_new = proj_simplex(y - eta * grad)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = a_new + (t - 1.0) / t_new * (a_new - a)
        y_new = proj_simplex(y_new)

        a, y, t = a_new, y_new, t_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 1000 == 0:
            print(f"    AccProx step={steps} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.2e}, L={L_estimate:.4f}")

    return best_a, best_score


def multiplicative_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent with entropy mirror map (multiplicative weight updates)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 1.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        eta = (1.0 / np.sqrt(n)) * (1.0 - 0.7 * progress)
        eta = max(eta, 0.01 / np.sqrt(n))

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-14:
            break

        # Multi-peak weights
        b_range = max_b - b.min() + 1e-12
        T_temp = max(0.02, 0.08 * b_range * (1.0 - 0.5 * progress))
        w = softmax_weights(b, T_temp)

        # Subgradient
        grad = compute_TT_v(w, a)

        # Multiplicative update: a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = fast_autocorr(a_new).max()
        a = a_new
        steps += 1

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            print(f"    MirrorDesc step={steps} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def simulated_annealing_polish(a0, time_limit, seed=42, verbose=True):
    """Simulated annealing with structured moves (peak shaving + spreading)."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T_init = 0.01 * best_score
    T_min = 0.0001 * best_score
    n_accept = 0
    n_reject = 0
    n_total = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_min / T_init) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing indices to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        move_type = rng.integers(3)

        if move_type == 0 and contrib.max() > 1e-18:
            # Peak shaving: move mass from peak contributor to non-contributor
            src = idx[np.argmax(contrib)]
            non_contrib = np.zeros(n, dtype=bool)
            if js > 0:
                non_contrib[:js] = True
            if je < n - 1:
                non_contrib[je + 1:] = True
            non_contrib[src] = True
            nc_idx = np.where(non_contrib)[0]
            if len(nc_idx) == 0:
                nc_idx = np.array([i for i in range(n) if i != src])
            dst = nc_idx[rng.integers(len(nc_idx))]
            amt = a[src] * rng.uniform(0.05, 0.25)
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st

        elif move_type == 1:
            # Random transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.02, 0.2)
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st

        else:
            # Multi-point perturbation
            n_pert = rng.integers(3, 10)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                amt = a_trial[i] * rng.uniform(0.02, 0.15)
                a_trial[i] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st

        sc = fast_autocorr(a_trial).max()
        n_total += 1

        if sc < best_score:
            a = a_trial
            best_score = sc
            best_a = a.copy()
            n_accept += 1
        else:
            # Annealing acceptance
            delta = sc - max_b
            if delta < 0:
                a = a_trial
                n_accept += 1
            else:
                accept_prob = np.exp(-delta / max(T, 1e-15))
                if rng.random() < accept_prob:
                    a = a_trial
                    n_accept += 1
                else:
                    n_reject += 1

        if verbose and n_total % 2000 == 0:
            print(f"    Anneal iter={n_total} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.6f}, acc={n_accept/n_total:.2f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def quick_proximal(a0, time_limit, verbose=False):
    """Fast proximal gradient without acceleration (for quick polishing)."""
    n = len(a0)
    a = proj_simplex(np.maximum(np.asarray(a0, dtype=np.float64), 0.0))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 2.0 / n

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-14:
            break

        b_range = max_b - b.min() + 1e-12
        T_temp = max(0.02, 0.05 * b_range)
        w = softmax_weights(b, T_temp)
        grad = compute_TT_v(w, a)

        a_new = proj_simplex(a - eta * grad)
        score = fast_autocorr(a_new).max()
        a = a_new
        steps += 1

        if score < best_score:
            best_score = score
            best_a = a.copy()

        eta *= 0.9995
        eta = max(eta, 1e-6 / n)

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QuickProx step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_active=150, gamma_init=1.7)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Accelerated Proximal Gradient (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: AccProxGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_out, _ = accelerated_proximal_gradient(best_seq, t2, verbose=True)
        do_try_update(a_out, f"accprox_n{n_cur}")

    # === Phase 3: Multiplicative Mirror Descent (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: MirrorDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_out, _ = multiplicative_mirror_descent(best_seq, t3, verbose=True)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # === Phase 4: LP Nesterov refined ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, max_active=100, gamma_init=1.5)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_candidates = [n_base - 50, n_base + 50, n_base + 100]
    n_candidates = [n for n in n_candidates if n >= 100]
    time_per_len = max(30.0, min(remaining * 0.05, 80.0))
    for n_try in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < time_per_len + 30:
            break
        print(f"\n=== Phase 5: LP Nesterov n={n_try} (budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.05, time_per_len)
        if t5 > 20:
            init_try = interpolate_sequence(best_seq, n_try)
            a_norm = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, max_active=120, gamma_init=1.6)
            do_try_update(a_out, f"lp_n{n_try}")

    # === Phase 6: Proximal gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: AccProxGrad r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        a_out, _ = accelerated_proximal_gradient(best_seq, t6, verbose=True)
        do_try_update(a_out, f"accprox2_n{n_cur}")

    # === Phase 7: Mirror descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: MirrorDesc r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 10:
        a_out, _ = multiplicative_mirror_descent(best_seq, t7, verbose=True)
        do_try_update(a_out, f"mirror2_n{n_cur}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, max_active=80, gamma_init=1.3)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 9: Simulated annealing polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Anneal n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = max(30.0, remaining * 0.15) if remaining > 100 else max(10.0, remaining * 0.3)
    if t9 > 10:
        a_out, _ = simulated_annealing_polish(best_seq, t9, seed=12345, verbose=True)
        do_try_update(a_out, f"anneal_n{n_cur}")

    # === Phase 10: Quick proximal final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t10 = max(10.0, remaining * 0.5) if remaining > 50 else remaining
    if t10 > 5:
        print(f"\n=== Phase 10: QuickProx n={n_cur} (budget={t10:.0f}s) ===")
        a_out, _ = quick_proximal(best_seq, t10, verbose=True)
        do_try_update(a_out, f"quickprox_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
