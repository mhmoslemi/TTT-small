            a_new = norm_sequence(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Geometric-mean ergodic average
        total_weight = total_weight * avg_rho + 1.0
        avg_a = avg_a * avg_rho + a
        
        # Check score periodically
        if n_iters % 10 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0
            
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            
            # Check geometric average
            avg_check = norm_sequence(avg_a / total_weight, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        # Stall detection
        if n_iters % 50 == 0:
            if stall_count >= stall_threshold:
                # Restart from best
                y = best_a.copy()
                a = best_a.copy()
                avg_a = best_a.copy()
                total_weight = 1.0
                stall_count = 0
                stall_threshold = max(100, stall_threshold // 2)
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    # Final check
    score_y = fast_autocorr(y).max()
    if score_y < best_score:
        best_score = score_y
        best_a = y.copy()
    avg_check = norm_sequence(avg_a / total_weight, norm)
    score_avg = fast_autocorr(avg_check).max()
    if score_avg < best_score:
        best_score = score_avg
        best_a = avg_check.copy()
    
    return best_a, best_score

def pair_reduction_polish(a0, time_limit, seed=42):
    """
    Polish by reducing pairs that contribute to peak convolution entries.
    For each peak lag k, find the pair (i, k-i) with maximum product a[i]*a[k-i],
    reduce the larger element, and redistribute to a low-contribution position.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        peak_lags = np.where(b >= M_curr * 0.95)[0]
        
        improved = False
        for k in peak_lags[:8]:
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src_i = idx[np.argmax(contrib)]
            
            # Find a good destination: position that doesn't contribute to any peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src_i)[0]
            
            # Choose destination with small current value
            candidates_sorted = candidates[np.argsort(a[candidates])]
            dst = candidates_sorted[rng.integers(min(len(candidates_sorted), max(1, len(candidates) // 4)))]
            
            # Try different reduction fractions
            for frac in [0.0003, 0.0007, 0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src_i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst] += amt
                if a_trial[src_i] < 0:
                    continue
                a_trial = norm_sequence(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = norm_sequence(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 1000:
                break
    
    return best_a, best_score

def lse_fista(a0, T_schedule, time_limit, verbose=False):
    """Log-sum-exp FISTA for fast warm-up."""
    n = len(a0)
    a = norm_sequence(a0, 1.0)
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for T in T_schedule:
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            # Compute LSE gradient
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                grad = np.zeros(n)
            else:
                shifted = (ac - ac_max) / T
                shifted = np.clip(shifted, -60.0, 60.0)
                exp_s = np.exp(shifted)
                w = exp_s / exp_s.sum()
                # Gradient of LSE w.r.t. y: grad_y[i] = 2 * sum_k w[k] * (dT/dy_i)[k]
                # (T @ w)[i] = sum_k T[i,k] * w[k] = conv(a, w)[i] ... 
                # Actually grad_i = 2 * (w * y_rev)[i] where y_rev is reversed
                N = 1 << (3 * n - 2).bit_length()
                W = rfft(w, N)
                Y_rev = rfft(y[::-1], N)
                cross = irfft(W * Y_rev, N)
                grad = 2.0 * cross[n - 1: n - 1 + n]
            
            step = 1.0 / L
            z = y - step * grad
            a_new = norm_sequence(z, 1.0)
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = norm_sequence(y_new, 1.0)
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    sc = fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return norm_sequence(best_a, 1.0), best_score

1. Multi-start LSE-FISTA warm-up with varied shapes
2. LP-Nesterov with geometric-mean averaging and adaptive M perturbation

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    best_a = a.copy()

    sc = fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return norm_sequence(best_a, 1.0), best_score


def lp_nesterov(a0, time_limit, solver, verbose=True, m_multiplier=1.0, avg_rho=0.95):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = a.copy()
    total_weight = 1.0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    stall_threshold = 300

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        M_use = M_curr * m_multiplier
        b = lp_best_response(y, M_use, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = norm_sequence(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = norm_sequence(a_new, norm)

        y = y_new
        a = a_new

        total_weight = total_weight * avg_rho + 1.0
        avg_a = avg_a * avg_rho + a

        if n_iters % 10 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            avg_check = norm_sequence(avg_a / total_weight, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if n_iters % 50 == 0:
            if stall_count >= stall_threshold:
                y = best_a.copy()
                a = best_a.copy()
                avg_a = best_a.copy()
                total_weight = 1.0
                stall_count = 0
                stall_threshold = max(150, stall_threshold * 2 // 3)

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    score_y = fast_autocorr(y).max()
    if score_y < best_score:
        best_score = score_y
        best_a = y.copy()
    avg_check = norm_sequence(avg_a / total_weight, norm)
    score_avg = fast_autocorr(avg_check).max()
    if score_avg < best_score:
        best_score = score_avg
        best_a = avg_check.copy()
    return best_a, best_score


def pair_reduction_polish(a0, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        peak_lags = np.where(b >= M_curr * 0.98)[0]

        improved = False
        for k in peak_lags[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src_i = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src_i)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(1, len(candidates_sorted) // 5)
            dst_candidates = candidates_sorted[:n_try]
            dst = dst_candidates[rng.integers(len(dst_candidates))]

            for frac in [0.0002, 0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src_i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst] += amt
                if a_trial[src_i] < 0:
                    continue
                a_trial = norm_sequence(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.001, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = norm_sequence(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 1200:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase:
    Phase 1: Multi-start LSE-FISTA warm-up (120s)
    Phase 2: LP-Nesterov main (700s) with geometric averaging
    Phase 3: Pair-reduction polish (120s)
    Phase 4: LP round 2 (280s)
    Phase 5: Multi-length exploration (150s)
    Phase 6: Final polish + LP (remaining time)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Multi-start warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.07, 120.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Multi-start warm-up (budget={t1:.0f}s) ===")
        x = np.linspace(0.0, 1.0, n_base)
        shapes = [
            np.sin(np.pi * x) ** 1.5,
            np.sin(np.pi * x) ** 2.0,
            0.5 * np.sin(np.pi * x) + 0.5 * np.sin(2.0 * np.pi * x * 0.5) ** 2,
        ]
        per_shape_time = t1 / len(shapes)
        for si, shape in enumerate(shapes):
            remaining = time_budget - (time.monotonic() - t_global)
            t_sub = min(per_shape_time, remaining)
            if t_sub < 5:
                break
            shape = np.maximum(shape, 0.0)
            s_val = shape.sum()
            if s_val < 1e-10:
                continue
            shape = shape / s_val
            print(f"  Shape {si+1}, budget={t_sub:.0f}s")
            a_out, _ = lse_fista(shape, [10.0, 3.0, 1.0, 0.3], t_sub, verbose=False)
            if a_out is not None:
                a_out = np.maximum(a_out, 0.0)
                s_val = a_out.sum()
                if s_val > 0.01:
                    a_out = a_out / s_val
                    sc = evaluate_safe(a_out)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"  *** NEW BEST [warm_s{si+1}]: {sc:.6f}")

    # === Phase 2: Main LP-Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.42, 700.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True, m_multiplier=1.0, avg_rho=0.95)
        if a_out is not None:
            a_out = np.maximum(a_out, 0.0)
            s_val = a_out.sum()
            if s_val > 0.01:
                a_out = a_out / s_val
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [lp_main]: {sc:.6f} (n={n_best})")

    # === Phase 3: Pair-reduction polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.08, 120.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Pair-reduction polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_reduction_polish(a_norm, t3, seed=42)
        if a_out is not None:
            a_out = np.maximum(a_out, 0.0)
            s_val = a_out.sum()
            if s_val > 0.01:
                a_out = a_out / s_val
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [polish1]: {sc:.6f}")

    # === Phase 4: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.22, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP round 2 n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True, m_multiplier=1.002, avg_rho=0.97)
        if a_out is not None:
            a_out = np.maximum(a_out, 0.0)
            s_val = a_out.sum()
            if s_val > 0.01:
                a_out = a_out / s_val
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [lp2]: {sc:.6f}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        longer_n = min(n_best + 400, 2400)
        t5 = min(remaining * 0.10, 130.0)
        if t5 > 30:
            print(f"\n=== Phase 5a: Longer n={longer_n} (budget={t5:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, longer_n)
            a_norm = init_long * np.sqrt(2.0 * longer_n) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=True, m_multiplier=1.0, avg_rho=0.95)
            if a_out is not None:
                a_out = np.maximum(a_out, 0.0)
                s_val = a_out.sum()
                if s_val > 0.01:
                    a_out = a_out / s_val
                    sc = evaluate_safe(a_out)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"  *** NEW BEST [long_n{longer_n}]: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            shorter_n = max(n_best - 400, 1000)
            if shorter_n != n_best:
                t5b = min(remaining * 0.06, 80.0)
                if t5b > 20:
                    print(f"\n=== Phase 5b: Shorter n={shorter_n} (budget={t5b:.0f}s) ===")
                    init_short = interpolate_sequence(best_seq, shorter_n)
                    a_norm = init_short * np.sqrt(2.0 * shorter_n) / init_short.sum()
                    a_out, _ = lp_nesterov(a_norm, t5b, solver, verbose=True, m_multiplier=1.0, avg_rho=0.95)
                    if a_out is not None:
                        a_out = np.maximum(a_out, 0.0)
                        s_val = a_out.sum()
                        if s_val > 0.01:
                            a_out = a_out / s_val
                            sc = evaluate_safe(a_out)
                            if np.isfinite(sc) and sc < best_score:
                                best_score = sc
                                best_seq = a_out.copy()
                                print(f"  *** NEW BEST [short_n{shorter_n}]: {sc:.6f}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t6 = min(remaining * 0.12, 100.0)
        if t6 > 20:
            print(f"\n=== Phase 6: Final polish n={n_best} (budget={t6:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = pair_reduction_polish(a_norm, t6, seed=999)
            if a_out is not None:
                a_out = np.maximum(a_out, 0.0)
                s_val = a_out.sum()
                if s_val > 0.01:
                    a_out = a_out / s_val
                    sc = evaluate_safe(a_out)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"  *** NEW BEST [polish_final]: {sc:.6f}")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.6, 150.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final LP n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=True, m_multiplier=1.0, avg_rho=0.98)
            if a_out is not None:
                a_out = np.maximum(a_out, 0.0)
                s_val = a_out.sum()
                if s_val > 0.01:
                    a_out = a_out / s_val
                    sc = evaluate_safe(a_out)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_out.copy()
                        print(f"  *** NEW BEST [lp_final]: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
