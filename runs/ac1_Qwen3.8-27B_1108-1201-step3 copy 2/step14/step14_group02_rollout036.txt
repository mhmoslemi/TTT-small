import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.signal import correlate
import cvxpy as cp

def fast_autocorr(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smoothmax_grad(a, tau=0.01, top_k=50):
    """
    Compute gradient of smooth max of convolution a*a.
    Uses softmax weights and FFT-based cross-correlation.
    
    grad[j] = 2 * sum_k w_k * a[k-j] (for valid k-j)
            = 2 * correlate(w, a, 'valid')[j]
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Get weights (top-k for efficiency and numerical stability)
    if top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        ac_top = ac[top_idx]
        ac_max = ac_top.max()
        w_top = np.exp((ac_top - ac_max) / tau)
        w_top /= w_top.sum()
        
        # Build full weight vector (sparse)
        w = np.zeros(2*n-1)
        w[top_idx] = w_top
    else:
        ac_max = ac.max()
        w = np.exp((ac - ac_max) / tau)
        w /= w.sum()
    
    # grad = 2 * correlate(w, a, mode='valid')
    grad = 2.0 * correlate(w, a, mode='valid')
    return grad

def mirror_descent_smooth(a0, time_limit, tau_init=0.05, tau_min=0.001, 
                          tau_decay=0.98, top_k=50, verbose=True):
    """
    Mirror descent for minimizing smooth max of convolution.
    Uses entropy mirror (exponential) for the simplex constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = tau_init
    step = 1.0 / np.sqrt(n)  # initial step size
    
    # Nesterov momentum
    y = a.copy()
    beta = 0.0
    
    # Barzilai-Borwein
    prev_a = None
    prev_g = None
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradient at Nesterov point
        g = smoothmax_grad(y, tau, top_k)
        
        # Mirror descent: a_new = exp(log(a) - step * g) / Z
        log_a = np.log(a + 1e-30)
        log_new = log_a - step * g
        log_new -= log_new.max()  # numerical stability
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta**2)) / 2.0
        y_new = a_new + (beta_new - 1.0) / (beta + 1.0) * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new /= sy
        
        # Track best
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        M_y = fast_autocorr(y_new).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y_new.copy()
        
        # BB step size
        if prev_a is not None and prev_g is not None:
            s_vec = a_new - prev_a
            y_vec = g - prev_g
            s_y = np.dot(s_vec, y_vec)
            s_s = np.dot(s_vec, s_vec)
            y_y = np.dot(y_vec, y_vec)
            if s_y > 1e-12 and s_s > 1e-12 and y_y > 1e-12:
                step_bb = s_s / s_y
                step = np.clip(step_bb, 1e-8, 10.0 / n)
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        y = y_new
        beta = beta_new
        
        # Decay tau
        tau = max(tau_min, tau * tau_decay)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, step={step:.2e}, tau={tau:.4f}")
    
    return best_a, best_m
