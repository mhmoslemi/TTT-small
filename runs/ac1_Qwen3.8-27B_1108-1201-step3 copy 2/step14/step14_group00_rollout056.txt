def build_rows(a, k_indices):
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            val = k - j
            if 0 <= val < n:
                rows[idx, j] = a[val]
    return rows


def lp_solve(a, M, active_set, solver):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active_set is not None and len(active_set) > 0:
        T = build_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_rows(a, np.arange(size))
        active_set = np.arange(size)
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set
    return b_val, active_set

def block_perturb(a, rng, block_sizes=None):
    n = len(a)
    if block_sizes is None:
        block_sizes = [5, 10, 20, 50]

    start = int(rng.integers(0, max(1, n - bs)))
    end = min(start + bs, n)

    new_block *= block_sum
    a_new = a.copy()
    a_new[start:end] = new_block
    return a_new

                            delta -= d * a[ki]
                        kj = k - j
                        if 0 <= kj < n:
                            delta += d * a[kj]
                        ki2 = i - k
                        if 0 <= ki2 < n:
                            delta -= d * a[k + ki2]
                        kj2 = j - k
                        if 0 <= kj2 < n:
                            delta += d * a[k + kj2]
                        new_val = conv[k] + delta
                        if new_val > new_max_est:
                            new_max_est = new_val
                    if new_max_est < best_val - 1e-12:
                        best_val = new_max_est
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = normalize_seq(a, n)
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-13:
                        best_score = conv.max()
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        a = normalize_seq(a, n)
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and (n - a.sum()) != 0:
            pass

    return best_a, best_score


def subgrad_fast(a0, time_limit, verbose=False, seed=0):
    """Fast subgradient method with O(n) per step."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()
        k_star = int(np.argmax(conv_y))

        # O(n) subgradient
        grad = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n, k_star + 1)
        if hi1 > lo1:
            indices = np.arange(lo1, hi1)
            grad[indices] += y[k_star - indices]
        lo2 = max(0, k_star)
        hi2 = min(n, k_star + n)
        if hi2 > lo2:
            indices = np.arange(lo2, hi2)
            grad[indices] += y[indices - k_star]
        grad *= 2.0

        # Multiplicative update
        eta = 0.5 / (1.0 + 0.05 * t)
        log_upd = -eta * grad / max(M, 1e-15)
        log_upd = np.clip(log_upd, -1.5, 1.5)
        v = y * np.exp(log_upd)
        v = np.maximum(v, 0.0)
        sv = v.sum()
        if sv < 1e-15:
            v = y.copy()
        else:
            v = v * norm / sv

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        if t == 1:
            a_new = v.copy()
        else:
            mom = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa

        y = y_new
        a = a_new

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            ac = avg_a / n_avg
            sac = ac.sum()
            if sac > 1e-15:
                ac = ac * norm / sac
                scc = fast_autocorr(ac).max()
                if scc < best_score:
                    best_score = scc
                    best_a = ac.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, variant=0):
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2) + 0.5 * np.exp(-8 * (x - 0.3) ** 2) + 0.5 * np.exp(-8 * (x - 0.7) ** 2)
    elif variant == 3:
        a = np.ones(n) * 0.8 + 0.2 * np.sin(np.pi * np.arange(n) / n)
    elif variant == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.15 * np.cos(x) + 0.1 * np.cos(3 * x) + 0.05 * np.cos(5 * x)
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = (1.0 - np.abs(2 * x - 1.0)) ** 0.5 + 0.3
    elif variant == 6:
        a = np.ones(n)
        # Add small structured noise
        a += 0.1 * np.random.default_rng(42).standard_normal(n)
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return fi / ss
