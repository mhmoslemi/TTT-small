    """LP best response with only top-K constraints."""
    n = len(y)
    ac = fast_autocorr(y)  # current autocorrelation
    # Sort constraints by "tightness"
    # A constraint conv(y,b)[k] <= M is tight if conv(y,b)[k] is close to M
    # We can approximate this by looking at which lag k has the largest y-contribution
    
    if K is None:
        K = min(2*n-1, 100)  # use at most 100 constraints
    
    # Select top K constraints
    # The most important constraints are those where the current autocorrelation is highest
    # (since those are the ones likely to be binding)
    top_k = np.argsort(ac)[::-1][:K]
    
    # Build reduced LP
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            T[idx, j] = y[k_int - j]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=cp.CLARABEL, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(b_var.value, 0)

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_best_response(y, M, solver, K=None):
    n = len(y)
    ac = fast_autocorr(y)
    if K is None:
        K = min(2 * n - 1, 200)
    top_k = np.argsort(ac)[::-1][:K]
    
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            T[idx, j] = y[k_int - j]
    
    constraints = [T @ b_var <= M, cp.sum(b_var) <= 2.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    g = b_val / b_sum
    return g

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, K=200):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        g = lp_best_response(y, M_curr, solver, K=K)
        if g is None:
            g = a.copy()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def compute_soft_gradient(a, beta):
    n = len(a)
    ac = fast_autocorr(a)
    g = ac
    max_g = g.max()
    logits = beta * (g - max_g)
    w = np.exp(logits)
    w = w / w.sum()
    
    grad_soft = np.zeros(n)
    for i in range(n):
        grad_soft[i] = np.dot(a, w[i:i+n])
    
    h = w * g
    cumsum_h = np.cumsum(np.concatenate([[0.0], h]))
    sliding_sum = cumsum_h[n:2*n] - cumsum_h[0:n]
    
    total_hw = np.dot(w, g)
    grad_F = 4.0 * n * a * (grad_soft - total_hw)
    
    soft_val = (1.0 / beta) * np.log(np.sum(np.exp(1e-10) * beta) / (beta * np.exp(1e-10 * max_g)))
    soft_val = max_g + (1.0 / beta) * np.log(np.sum(np.exp(beta * (g - max_g))))
    
    return grad_F, soft_val

def soft_max_objective_and_grad(x, beta, n):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    a = a / s
    
    grad_F, soft_val = compute_soft_gradient(a, beta)
    obj = 2.0 * n * soft_val
    return obj, grad_F

def lbfgs_log_optimize(a0, time_limit, beta=100.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x0 = np.log(a + 1e-15)
    
    def obj_and_grad(x):
        x_shifted = x - x.max()
        a_local = np.exp(x_shifted)
        s = a_local.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        a_local = a_local / s
        grad_F, soft_val = compute_soft_gradient(a_local, beta)
        obj = 2.0 * n * soft_val
        
        if obj < best_score:
            pass  # track externally
        
        return obj, grad_F
    
    # Use a loop with time checking
    result = None
    try:
        result = scipy_minimize(
            obj_and_grad, x0, method='L-BFGS-B',
            options={'maxiter': 10000, 'maxfun': 5000, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxls': 50}
        )
    except Exception as e:
        print(f"    L-BFGS error: {e}")
    
    if result is not None and result.x is not None:
        x_opt = result.x
        x_shifted = x_opt - x_opt.max()
        a_opt = np.exp(x_shifted)
        s = a_opt.sum()
        if s > 1e-15:
            a_opt = a_opt / s
            a_opt = np.maximum(a_opt, 0.0)
            score = 2.0 * n * fast_autocorr(a_opt).max()
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS-B ({elapsed:.1f}s): J={best_score:.6f}")

    return best_a, best_score

def coordinate_polish(a0, time_limit, verbose=False, rng=None):
    n = len(a0)
    if rng is None:
        rng = np.random.default_rng(777)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            k = top_k[rng.integers(len(top_k))]
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js >= je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            peak_indices = top_k[:8]
            best_dst = -1
            best_score = float('inf')
            
            n_cand = min(n, 300)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_indices:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = rng.integers(0, n, n_cand)
            else:
                candidates = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                sens = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                rem_sens = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - src
                    d2 = src - k2i
                    rem = 0.0
                    if 0 <= d1 < n:
                        rem += a[d1]
                    if 0 <= d2 < n:
                        rem += a[d2]
                    rem_sens += rem
                net_sens = sens - rem_sens
                if net_sens < best_score:
                    best_score = net_sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def soft_subgrad_mirror_descent(a0, time_limit, top_k=30, beta=5.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    gamma = 0.5
    avg_a = a.copy()
    n_avg = 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        step = gamma / (np.sqrt(t) + 1.0)
        ac = fast_autocorr(a)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score-sc:.6f})")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP Frank-Wolfe at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP n={n0} ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize(a_in, t1, solver, verbose=True, K=200)
        try_update(a_out, f"lp1_n{n0}")

    # === Phase 2: L-BFGS-B in log-space ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: L-BFGS-B n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        for beta_try in [80.0, 150.0, 250.0]:
            if time.monotonic() - t_global > time_budget - 20:
                break
            t2_sub = min(t2 / 3.0, time_budget - (time.monotonic() - t_global))
            if t2_sub < 10:
                break
            a_out, _ = lbfgs_log_optimize(a_in, t2_sub, beta=beta_try, verbose=True)
            try_update(a_out, f"lbfgs_n{len(best_seq)}_b{int(beta_try)}")

    # === Phase 3: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = soft_subgrad_mirror_descent(a_in, t3, top_k=30, beta=5.0, verbose=True)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord n={len(best_seq)} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng=rng)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 5: LP again (on improved sequence) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP round2 n={n_best} ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 30 and n_best <= 2500:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_lp_optimize(a_in, t5, solver, verbose=True, K=250)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: L-BFGS-B again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: L-BFGS-B round2 n={n_best} ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_log_optimize(a_in, t6, beta=200.0, verbose=True)
        try_update(a_out, f"lbfgs2_n{n_best}")

    # === Phase 7: Try scaling to larger length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_target = max(n_best + 800, 2400)

    t7 = min(remaining * 0.20, 200.0)

        if n_target <= 2500:

            try_update(a_out, f"lp_n{n_target}")
        else:
            a_out, _ = soft_subgrad_mirror_descent(a_in, t7 * 0.5, top_k=30, beta=4.0, verbose=False)
            try_update(a_out, f"md_n{n_target}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = lbfgs_log_optimize(norm_sum1(best_seq), min(remaining*0.5, 80.0), beta=150.0, verbose=False)
            try_update(a_out, f"lbfgs_n{n_target}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), min(remaining*0.6, 60.0), verbose=False, rng=rng)
            try_update(a_out, f"coord_n{n_target}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 15:
        t8 = min(remaining * 0.5, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng=rng)
        try_update(a_out, f"final_coord_n{n_best}")

    # === Phase 9: Random restart exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Random restarts n={n_best} ===")
        for restart in range(5):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_sub = min(remaining / 5.0, 60.0)
            if t_sub < 15:
                break
            
            # Different initializations
            if restart == 0:
                x = np.linspace(0, 1, n_best)
                init = np.cos(np.pi * (x - 0.5)) ** 2
            elif restart == 1:
                init = np.random.default_rng(restart*7).dirichlet(np.ones(n_best) * 2)
            elif restart == 2:
                x = np.linspace(0, 1, n_best)
                init = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
                init = np.maximum(init, 0.01)
            else:
                init = 0.7 * norm_sum1(best_seq) + 0.3 * np.random.default_rng(restart*13).dirichlet(np.ones(n_best))
            
            init = norm_sum1(init)
            
            if n_best <= 2000:

                a_out, _ = soft_subgrad_mirror_descent(init, t_sub * 0.5, top_k=20, beta=4.0, verbose=False)
            try_update(a_out, f"restart{restart}_n{n_best}")
            
            a_out, _ = lbfgs_log_optimize(norm_sum1(best_seq), min(t_sub*0.3, 30.0), beta=120.0, verbose=False)
            try_update(a_out, f"restart{restart}_lbfgs")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
