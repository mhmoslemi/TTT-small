    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    best_M = fast_conv(a).max()

        if b_sum < 1e-15:

        alpha = min(alpha, 0.5)

        y_new = np.maximum(y_new, 0.0)

            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new

def interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n

    result = np.maximum(result, 0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return result * np.sqrt(2.0 * new_n) / rs

                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = normalize(a_t, norm_val)
                sc = fast_conv(a_t).max()
                if sc < best_M:
                    a = a_t
                    best_M = sc
                    best_a = a.copy()
                    n_improve += 1

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            print(f"      CoordPolish improve={n_improve}: best_M={best_M:.8f}")

    return best_a, best_M

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, f_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    return result * np.sqrt(2.0 * new_n) / rs

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    norm_val = np.sqrt(2.0 * n_base)
    a_start = best_seq * norm_val / best_seq.sum()

    # Phase 1: LP Nesterov (main)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n_base} (t={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp(a_start, t1, solver, verbose=True)
        do_try(a_out, "lp1")

    # Phase 2: LSE gradient Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    a_in = best_seq * norm_val / best_seq.sum()
    t2 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 2: LSE-Grad n={n_cur} (t={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = lse_gradient_nesterov(a_in, t2, verbose=True)
        do_try(a_out, "lse1")

    # Phase 3: Deconvolution smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    a_in = best_seq * norm_val / best_seq.sum()
    t3 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 3: DeconvSmooth n={n_cur} (t={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = deconvolution_smooth(a_in, t3, verbose=True)
        do_try(a_out, "deconv")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    a_in = best_seq * norm_val / best_seq.sum()
    t4 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 4: LP-Nesterov r2 n={n_cur} (t={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp(a_in, t4, solver, verbose=True)
        do_try(a_out, "lp2")

    # Phase 5: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base + 80
    t5 = min(remaining * 0.10, 140.0)
    print(f"\n=== Phase 5: LP-Nesterov n={n2} (t={t5:.0f}s) ===")
    if t5 > 20:
        init_long = interpolate_seq(best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum(), n2)
        a_out, _ = nesterov_lp(init_long, t5, solver, verbose=True)
        do_try(a_out, f"lp_n{n2}")

    # Phase 6: Antisymmetric perturbation exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    t6 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 6: AntiSym explore n={n_cur} (t={t6:.0f}s) ===")
    if t6 > 10:
        base = best_seq * norm_val / best_seq.sum()
        for eps in [0.01, 0.02, 0.03, 0.05]:
            x = np.arange(n_cur)
            pert = 1.0 + eps * np.sin(2.0 * np.pi * (x + 0.5) / n_cur)
            a_pert = np.maximum(base * pert, 0.0)
            a_pert = normalize(a_pert, norm_val)
            a_out, _ = nesterov_lp(a_pert, min(t6 / 4, 25.0), solver, verbose=False)
            do_try(a_out, f"antisym_eps{eps}")

    # Phase 7: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    a_in = best_seq * norm_val / best_seq.sum()
    t7 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 7: LSE polish n={n_cur} (t={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = lse_gradient_nesterov(a_in, t7, verbose=True)
        do_try(a_out, "lse_final")

    # Phase 8: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    norm_val = np.sqrt(2.0 * n_cur)
    a_in = best_seq * norm_val / best_seq.sum()
    t8 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 8: CoordPolish n={n_cur} (t={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = coordinate_polish(a_in, t8, seed=777, verbose=True)
        do_try(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
