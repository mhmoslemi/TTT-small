    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, max_active=150):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    # Active-set reduction: iteratively identify binding constraints
    circ = T @ b_val
    for iteration in range(8):
        tol = 1e-7 * M + 1e-15
        active = circ >= M - tol
        n_active = int(active.sum())

        if n_active >= min(size, max_active + 100) or n_active < 5:
            break

        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            # Slightly violated: scale down to satisfy
            viol = circ_new.max() / M
            if viol > 1.0:
                b_val = b_val / viol
        circ = T @ b_val

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, max_active=120, gamma_init=1.9):
    """Fessler-Parks accelerated LP Nesterov with adaptive gamma and restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = gamma_init
    M_hist = []

    while time.monotonic() - t_start < time_limit:
        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, max_active=max_active)
        if b is None:
            # Try with relaxed M
            b = lp_best_response(y, M_curr * 1.001, solver, max_active=max_active)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            if momentum > 1.0:
                momentum = 1.0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            osc = np.sum(np.abs(np.diff(recent))) / (30 * max(recent[0], 1e-15))
            if osc > 0.04:
                gamma = max(1.0, gamma - 0.15)
            elif osc < 0.008:
                gamma = min(2.0, gamma + 0.05)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")

        # Adaptive restart
        if stall > 200:
            rng = np.random.default_rng(n_iters * 31 + 13)
            perturb_scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = max(1.5, gamma - 0.2)
            M_hist = []
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def subgradient_refine(a0, time_limit, verbose=True):
    """Multi-peak subgradient with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = 0.05 / n
    best_local = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top K peaks
        K = min(8, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:K]
        peak_vals = b[top_peaks]

        # Softmax weighting
        T_speed = max(0.005, 0.05 * (max_b - peak_vals.min() + 1e-10) * (1.0 - elapsed_frac))
        w = np.exp((peak_vals - max_b) / T_speed)
        w = w / w.sum()

        # Compute combined subgradient
        grad = np.zeros(n)
        for m, p in enumerate(top_peaks):
            if w[m] < 1e-6:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * w[m] * a[p - idx]

        # Project onto tangent space
        grad = grad - np.mean(grad)

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        # Polyak step with upper bound
        polyak_step = max(0.0, max_b - best_score * 0.999) / grad_norm_sq
        step = min(step_size, polyak_step) if polyak_step > 0 else step_size
        step = max(step, 1e-8 / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()

        if score_new < best_local:
            a = a_new
            best_local = score_new
            step_size = min(step_size * 1.08, 0.5 / n)
            stagnation = 0
        else:
            stagnation = getattr(subgradient_refine, '_stagnation', 0) + 1
            if score_new < max_b * 1.002 and np.random.random() < (1.0 - elapsed_frac) * 0.3:
                a = a_new
            step_size *= 0.85

        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()

        steps += 1

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, M={max_b:.6f}")

        # Periodic perturbation
        if steps % 1500 == 0 and step_size < 1e-10 / n:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
            a = a * norm / a.sum()
            step_size = 0.05 / n
            best_local = fast_autocorr(a).max()

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with multi-destination moves and annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top peaks
        top_K = min(5, 2 * n - 1)
        peak_order = np.argsort(b)[::-1][:top_K]

        moved = False
        for p_star in peak_order:
            # Identify contributing pairs
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[p_star - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find non-contributing positions
            non_contrib = np.zeros(n, dtype=bool)
            if js > 0:
                non_contrib[:js] = True
            if je < n - 1:
                non_contrib[je + 1:] = True
            non_contrib[src] = True

            # Check if src contributes to any other top peak
            # (to avoid making other peaks worse)
            non_contrib_idx = np.where(non_contrib)[0]
            if len(non_contrib_idx) == 0:
                non_contrib_idx = np.array([i for i in range(n) if i != src])

            n_select = min(15, len(non_contrib_idx))
            sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]

            for dst in sorted_dst[:5]:
                # Try multiple transfer fractions
                best_trial_score = max_b
                best_frac = 0
                for frac in [0.03, 0.06, 0.1, 0.15, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_trial_score:
                        best_trial_score = sc
                        best_frac = frac

                if best_frac > 0 and best_trial_score < best_score:
                    a = a.copy()
                    a[src] -= src_val * best_frac
                    a[dst] += src_val * best_frac
                    a = np.maximum(a, 0)
                    a = a * norm / a.sum()
                    best_score = fast_autocorr(a).max()
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    moved = True
                break

            if moved:
                break
        else:
            n_stall += 1

        if not moved:
            # Random exploration
            if n_stall > 20:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.03, 0.15)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                        else:
                            # Annealing
                            T = best_score * 0.002 * (1.0 - elapsed_frac)
                            if rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
                                a = a_trial
                            n_stall += 1

        if n_stall > 200:
            # Larger perturbation
            n_pert = max(10, n // 40)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.03:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def fourier_preprocess(a0, n_iters=10):
    """Fourier-domain flattening: adjust phases to reduce peak autocorrelation."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    norm = np.sqrt(2.0 * n)
    a = a * norm / a.sum()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    for _ in range(n_iters):
        A = rfft(a, 2 * n)
        mag = np.abs(A)
        # Try to flatten magnitude spectrum (reduce peaks)
        # Where |A| is large, reduce it; where small, increase
        mean_mag = mag.mean()
        mag_flat = mean_mag + 0.3 * (mag - mean_mag)
        mag_flat = np.maximum(mag_flat, 0.1 * mean_mag)
        phase = np.angle(A)
        A_new = mag_flat * np.exp(1j * phase)
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new * norm / s
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-phase optimization combining LP Nesterov,
    subgradient, Fourier preprocessing, and peak shaving with adaptive
    time allocation and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        print("No parent available, using default init")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.ones(n_base)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Fourier preprocessing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t0 = time_budget * 0.05
    if remaining > t0 + 100:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out = fourier_preprocess(a_norm, n_iters=15)
        do_try_update(a_out, "fourier_pre")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 420.0)
    if t1 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, max_active=130, gamma_init=1.9)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Subgradient on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 100.0)
    if t2 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_refine(a_norm, t2, verbose=True)
        do_try_update(a_out, f"subgrad1_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, max_active=100, gamma_init=1.7)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t4, seed=717, verbose=True)
        do_try_update(a_out, f"peak1_n{n_cur}")

    # === Phase 5: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_candidates = [len(best_seq) + 50, len(best_seq) - 50, len(best_seq) + 100]
    n_candidates = [nc for nc in n_candidates if nc >= 200 and nc <= 5000]

    for n_try in n_candidates[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        t_try = min(remaining * 0.12, 180.0)
        if t_try < 30:
            break
        print(f"\n=== Trying n={n_try} (budget={t_try:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t_try, solver, verbose=True, max_active=100, gamma_init=1.8)
        do_try_update(a_out, f"lp_n{n_try}")

    # === Phase 6: LP Nesterov final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.18, 250.0)
    if t6 > 40:
        print(f"\n=== Phase 6: Final LP Polish n={n_cur} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, max_active=80, gamma_init=1.5)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 7: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 15:
        print(f"\n=== Phase 7: SubGrad Polish n={n_cur} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_refine(a_norm, t7, verbose=True)
        do_try_update(a_out, f"subgrad_final_n{n_cur}")

    # === Phase 8: Peak shaving final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 15:
        print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=271828, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # === Phase 9: One more LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Last LP n={n_cur} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, remaining * 0.7, solver, verbose=True, max_active=60, gamma_init=1.3)
        do_try_update(a_out, f"lp_very_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
